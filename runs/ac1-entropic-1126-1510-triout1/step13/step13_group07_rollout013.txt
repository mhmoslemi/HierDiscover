# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Returns a vector of length 2·n‑1.
    """
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the index grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:

    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Exponentiated‑gradient descent with soft‑max smoothing ---------------
def _softmax_grad(a: np.ndarray, temp: float) -> np.ndarray:
    """
    Returns the gradient of the soft‑max approximation
    (1/τ)·log Σ_k exp(τ·(a*a)[k]) with respect to a.
    Computed via two FFT‑based convolutions:
        * conv = a*a   (length 2n‑1)
        * w   = soft‑max weights over conv      (length 2n‑1)
        * grad = 2·(a * w)  (full convolution, centre slice of length n)
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _fft_self_convolve(a)                # length 2n‑1

    # 2) soft‑max weights (stable version)
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / temp)   # (conv‑max)/τ
    w = exp_vals / exp_vals.sum()              # probabilities, sum to 1

    # 3) gradient = 2 * (a ⋆ w)  where ⋆ denotes cross‑correlation.
    #    It can be obtained from a linear convolution of a and w
    #    and then extracting the central n entries.
    need = n + (2 * n - 1) - 1                  # = 3n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    c = np.fft.irfft(A * W, n=size)[:need]     # full convolution, length 3n‑2
    grad = 2.0 * c[n - 1 : n - 1 + n]          # central slice, length n
    return grad


def _exp_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    temp0: float = 0.2,
    temp_decay: float = 0.9995,
    min_temp: float = 1e-5,
    jitter_interval: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the convex objective
    max_k (a*a)[k] (sum of a is kept equal to 1).
    The method uses a smooth soft‑max surrogate that is gradually
    sharpened.  Multiplicative updates keep the iterate strictly
    positive; after each step the vector is renormalised to lie on the
    probability simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    cur_score = _objective(a)

    temp = temp0
    while time.time() < deadline and t < 200_000:
        t += 1

        # evaluate true objective
        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] exp‑grad it={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # gradient of the smooth surrogate
        grad = _softmax_grad(a, temp)

        # multiplicative (exponentiated) update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (simple L1 renorm works because all entries stay ≥0)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = _distribution_uniform(n)
        else:
            a = a / s

        # temperature annealing
        temp = max(temp * temp_decay, min_temp)

        # jitter if stagnating
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full a*a convolution using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]

    return lam * uniform + (1.0 - lam) * base

# Soft‑max gradient (used inside the exponentiated‑gradient loop) -----
def _softmax_grad(a: np.ndarray, temp: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate
        (1/τ)·log Σ_k exp(τ·(a*a)[k])
    with respect to *a*.
    The gradient equals 2·(a ⋆ w) where w are the soft‑max weights.
    Both the self‑convolution and the cross‑convolution are obtained
    via FFTs.
    """
    n = a.shape[0]

    # 1) self‑convolution a*a
    conv = _fft_self_convolve(a)          # length 2n‑1

    # 2) soft‑max weights (stable)
    shift = conv.max()
    exp_vals = np.exp((conv - shift) / temp)
    w = exp_vals / exp_vals.sum()          # Σ w = 1, length 2n‑1

    # 3) gradient = 2·(a ⋆ w) = 2·central slice of a * w (linear convolution)
    need = n + (2 * n - 1) - 1               # = 3n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    c = np.fft.irfft(A * W, n=size)[:need]  # full convolution
    grad = 2.0 * c[n - 1 : n - 1 + n]        # central n entries
    return grad


# ----------------------------------------------------------------------
def _exp_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.5,
    temp0: float = 0.2,
    temp_decay: float = 0.9995,
    min_temp: float = 1e-5,
    jitter_interval: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the convex objective
    ``max_k (a*a)[k]`` while staying on the probability simplex.
    The surrogate temperature is annealed from *temp0* to *min_temp*.
    Multiplicative updates keep the iterate strictly positive.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    cur_score = _objective(a)

    temp = temp0
    while time.time() < deadline and t < 200_000:
        t += 1

        # true objective (used for bookkeeping)
        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] exp‑grad it={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # gradient of the smooth surrogate
        grad = _softmax_grad(a, temp)

        # multiplicative (exponentiated) update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex (simple L1 normalisation)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = _distribution_uniform(n)
        else:
            a = a / s

        # anneal temperature
        temp = max(temp * temp_decay, min_temp)

        # occasional jitter to escape plateaus
        if no_improve >= jitter_interval:
            jitter = rng.exponential(scale=0.002, size=n)
            a = a + jitter
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic local mass‑transfer moves; keep improvements."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing walk."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """Main entry point – returns a list of heights for ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 3.0            # tiny safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    all_candidates: list[tuple[float, np.ndarray]] = []

    # --------------------------------------------------------------
    # 1️⃣  Seed generation
    lengths = [1500, 2500, 3500, 4500, 6000, 8000]
    beta_ps = [0.05, 0.10, 0.20, 0.40, 0.70, 1.0, 1.5, 2.0]
    edge_ps = [0.05, 0.15, 0.30, 0.45, 0.70, 1.0]
    sine_ps = [0.5, 1, 2, 3, 4, 5]
    mix_lams = [0.0, 0.25, 0.5, 0.75, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        all_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"  new global best (uniform) len={n} score={sc:.6f}")

        # arcsine
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        all_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"  new global best (arcsine) len={n} score={sc:.6f}")

        # symmetric beta
        for p in beta_ps:
            if time.time() >= deadline:
                break
            base = _distribution_beta_sym(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"  new global best βsym p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # edge‑weighted
        for p in edge_ps:
            if time.time() >= deadline:
                break
            base = _distribution_edge_weighted(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"  new global best edge p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # sine‑weighted
        for p in sine_ps:
            if time.time() >= deadline:
                break
            base = _distribution_sine_weighted(n, p)
            for lam in mix_lams:
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                all_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"  new global best sine p={p} lam={lam:.2f} len={n} score={sc:.6f}"
                    )

        # asymmetric beta (random parameters)
        for _ in range(3):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0)      # 0.01 … 1
            b_param = 10 ** rng.uniform(-2, 0)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"  new global best asym‑beta a={a_param:.3g} b={b_param:.3g} len={n} score={sc:.6f}"
                )

        # random Dirichlet vectors
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.005, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"  new global best Dirichlet α={alpha:.3g} len={n} score={sc:.6f}"
                )

        # sparse random vectors
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            all_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"  new global best sparse len={n} score={sc:.6f}")

    # keep the ten best seeds
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in all_candidates[:10]]

    # --------------------------------------------------------------
    # 2️⃣  Optional: import a previously known construction
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"  imported previous best score={sc:.6f}")
            all_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / len(top_seeds)
        print(f"[{time.time():.1f}] start exp‑grad descent on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_grad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.6,
                temp0=0.2,
                temp_decay=0.9993,
                jitter_interval=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exp‑grad seed {idx} new global best {overall_best_score:.6f}"
                )
            if overall_best_score <= 1.45:      # early stop if we are already very good
                break

    # --------------------------------------------------------------
    # 4️⃣  Deterministic pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # --------------------------------------------------------------
    # 5️⃣  Final simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # --------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
