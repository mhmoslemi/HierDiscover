def _adam_refine(a, deadline, best_seq=None, best_score=float('inf'), max_iters=200000,
                lr0=0.5, active_frac=0.03, max_active=25, beta1=0.9, beta2=0.999,
                eps=1e-8, eval_interval=200):
    n = a.shape[0]
    # initialise x = log(a) - could use np.log(a+1e-12)
    x = np.log(a + 1e-12)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t = 0
    if best_seq is None:
        best_seq = a.copy()
        best_score = _objective(a)
    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # active set
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep top max_active by conv
            idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[idx]
        # subgradient w.r.t a
        grad_a = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]
        # gradient w.r.t x via softmax Jacobian
        # grad_x = a * (grad_a - np.dot(a, grad_a))
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = lr0 / np.sqrt(t)  # incorporate decaying step size
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                # print
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")
    # final projection back to simplex: a already sum to 1 due to softmax, but ensure
    best_seq = _project_onto_simplex(best_seq, 1.0)
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.
The search consists of three phases:

1️⃣  **Broad seeding** –  a collection of analytically‑motivated families
    (uniform, sine‑weighted, centre‑exponential, left‑skewed exponential,
    random Dirichlet with low/high concentration, sparse random) is
    sampled for several lengths.  The 12 best seeds (lowest exact objective)
    are kept.

2️⃣  **Adam‑based mirror descent** –  each retained seed is short‑run
    optimised using an Adam optimiser on the soft‑max parametrisation
    (log‑weights → probabilities).  The sub‑gradient is assembled from the
    active set of convolution entries that are within a small fraction of
    the current maximum.  A long final run from the overall best seed uses
    the remaining time budget.

3️⃣  **Local polishing** –  a greedy pairwise mass‑transfer step followed by a
    cheap simulated‑annealing style search cleans up any remaining irregularities.

All steps respect the 1000 s global budget.  The returned list can be fed
directly to ``evaluate_sequence`` – the evaluator will normalise it anyway.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a (real) FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective  2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector – sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_exponential_center(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = np.exp(-rate * np.abs(i - half))
    w /= w.sum()
    return w


def _distribution_exponential_left(n: int, rate: float) -> np.ndarray:
    """Monotonically decreasing exponential from the left side."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_exponential_right(n: int, rate: float) -> np.ndarray:
    """Mirror of the left‑skewed exponential."""
    return _distribution_exponential_left(n, rate)[::-1]


def _distribution_dirichlet(n: int, rng: np.random.Generator, conc: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    return a.astype(np.float64)


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Adam‑based mirror descent on the exact objective
def _adam_refine(
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    lr0: float = 0.8,
    active_frac: float = 0.025,
    max_active: int = 30,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    eval_interval: int = 250,
    best_seq: np.ndarray | None = None,
    best_score: float = float("inf"),
) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector using Adam on the soft‑max parameters.
    The objective is the exact ``_objective`` (so no stochastic estimate).
    """
    n = init_a.shape[0]
    # initialise log‑weights (avoid log(0))
    x = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    ind_grid = np.arange(n, dtype=np.int32)
    t = 0
    if best_seq is None:
        best_seq = init_a.copy()
        best_score = _objective(init_a)

    while time.time() < deadline and t < max_iters:
        t += 1
        a = _softmax(x)          # always lies on the simplex
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep the largest max_active entries only
            top = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top]

        # sub‑gradient of the max (using the active set)
        grad_a = np.zeros_like(a)
        for idx_k in act:
            offset = idx_k - ind_grid
            mask = (offset >= 0) & (offset < n)
            grad_a[mask] += 2.0 * a[offset[mask]]

        # convert gradient w.r.t. a into gradient w.r.t. log‑weights x
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update with a decaying step‑size
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = lr0 / math.sqrt(t)
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam it={t:,} new best {best_score:.6f}")

    # final projection (soft‑max already sums to 1, but keep a safe guard)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq, best_score

# Greedy pairwise refinement (same as in the reference code)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

# Very cheap simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
