def _adam_descent(a, deadline, best_seq, best_score):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5
    t = 0
    while time.time() < deadline:
        t += 1
        # compute active constraints as in original
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        act = np.where(conv >= max_c * (1 - 0.05))[0]  # maybe keep 5%?
        if act.size > 30:
            # keep top 30 largest conv values
            act = act[np.argpartition(conv[act], -30)[-30:]]
        # gradient
        grad = np.zeros_like(a)
        ind = np.arange(a.size)
        for idx in act:
            offset = idx - ind
            mask = (offset >= 0) & (offset < a.size)
            grad[mask] += 2.0 * a[offset[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 / (np.sqrt(v_hat) + eps) * m_hat
        # exponentiated gradient update: a = a * np.exp(-step)
        a = a * np.exp(-step)  # ensures nonnegativity
        a = _project_onto_simplex(a, s=1.0)
        # evaluate occasionally
        if t % 200 == 0:
            cur = _objective(a)
            if cur < best_score:
                best_score, best_seq = cur, a.copy()
    return best_seq, best_score

def _smooth_gradient(a, T):
    conv = _fft_self_convolve(a)  # shape 2n-1
    max_c = conv.max()
    # compute softmax weights scaled by temperature T
    # w_k = exp((conv_k - max_c) / T)
    # to avoid overflow, shift by max.
    w = np.exp((conv - max_c) / T)
    w_sum = w.sum()
    p = w / w_sum
    # compute gradient: grad_i = 2 * sum_k p_k * a_{k-i}
    # implement via FFT
    # first compute weighted sum of a shifts: do conv of a with p_rev
    p_rev = p[::-1]

def _gradient_from_weights(a, w):
    # a length n, w length 2n-1
    n = a.shape[0]
    # compute conv of a with reversed w
    w_rev = w[::-1]
    # size of FFT
    need = n + w_rev.shape[0] - 1  # = 3n-2
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    W = rfft(w_rev, n=size)
    conv = irfft(A * W, n=size)
    conv = conv[:need]  # length need
    # gradient_i = conv[2n-2 - i]
    grad = conv[2 * n - 2 - np.arange(n)]
    return grad
```

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute gradient g_i = sum_k w_k * a_{k-i} efficiently via FFT.
    a shape (n,)
    w shape (2n-1,)
    Returns array of shape (n,) where missing indices are treated as 0.
    """
    n = a.shape[0]
    # reverse w
    w_rev = w[::-1]
    # compute convolution length n + (2n-1) - 1 = 3n-2
    need = n + w_rev.shape[0] - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    W = rfft(w_rev, n=size)
    conv = irfft(A * W, n=size)
    conv = conv[:need]  # length need
    # gradient_i = conv[2*n-2 - i]
    indices = 2 * n - 2 - np.arange(n)
    grad = conv[indices]
    return grad

"""
search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

The algorithm consists of three phases:

1️⃣ **Diverse seed generation** – analytic families (uniform,
    edge‑weighted, edge‑inverse‑power, sine‑weighted, symmetric beta),
    random Dirichlet samples and sparse vectors are created for several
    lengths.  The ~10 best seeds (lowest exact objective) are kept.

2️⃣ **Two‑stage continuous optimisation** – each of the retained seeds is
    refined by (i) a smoothed‑gradient descent that minimises a
    temperature‑softened version of the max‑convolution (full softmax
    weights, gradient computed by FFT) and (ii) a mirror‑descent with
    momentum on the true sub‑gradient (active‐set).  The best result of
    all runs is kept.

3️⃣ **Local polishing** – a greedy pair‑wise mass‑transfer step followed
    by a cheap simulated‑annealing style local search.

All updates stay on the probability simplex (non‑negative, sum = 1) by
projecting with Euclidean projection.  The routine respects a 1000 s
budget and returns a plain Python list of heights ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2n‑1) using a real FFT."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.006
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Helper for the soft‑max gradient --------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k * a_{k-i} efficiently (zero outside range).
    a – (n,)
    w – (2n‑1,)
    Returns (n,) gradient (without the factor 2 that appears in the exact
    sub‑gradient of the max‑convolution).
    """
    n = a.shape[0]
    # reverse w and convolve with a
    w_rev = w[::-1]
    need = n + w_rev.shape[0] - 1          # = 3n‑2
    size = 1 << ((need - 1).bit_length())
    A = rfft(a, n=size)
    W = rfft(w_rev, n=size)
    conv = irfft(A * W, n=size)
    conv = conv[:need]                     # length 3n‑2
    # gradient_i = conv[2n‑2‑i]
    idx = 2 * n - 2 - np.arange(n)
    grad = conv[idx]
    return grad

# Mirror‑descent with momentum on the exact (active‑set) objective ---
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.08,
    max_active: int = 40,
    beta: float = 0.9,
    eval_interval: int = 250,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent with momentum using an active set."""
    n = a.shape[0]
    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 400_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for idx in act:
            offset = idx - ind
            mask = (offset >= 0) & (offset < n)
            grad[mask] += 2.0 * a[offset[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] momentum it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Smoothed‑gradient descent (soft‑max of the convolution) ----------
def _smoothed_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    temp0: float = 0.12,
    temp_decay: float = 0.9994,
    eval_interval: int = 200,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a temperature‑softened version of max(conv).
    The soft‑max weights are updated each iteration; the gradient is
    computed via FFT (linear time).  An exponential (mirror) update
    keeps the iterates on the simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 300_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # temperature schedule
        T = max(1e-4, temp0 * (temp_decay ** t))
        # soft‑max weights (stable)
        w = np.exp((conv - max_c) / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            p = np.full_like(w, 1.0 / w.size)
        else:
            p = w / w_sum

        # gradient = 2 * Σ_k p_k * a shifted by k
        grad = 2.0 * _gradient_from_weights(a, p)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smoothed it={t:,} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score

# Greedy pairwise refinement -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Transfer a random fraction of mass from i to j if it improves the objective."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Simulated‑annealing style local search -------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

    p_grid = np.logspace(-3, 1.0, num=12)           # 0.001 … 10
    lam_grid = np.linspace(0.0, 0.4, 5)            # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families (no mixing)
        for cand in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_edge_weighted(n, 0.5),
            _distribution_edge_inversepow(n, 0.5),
        ):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # parametrised families + uniform mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            families = (
                _distribution_edge_weighted(n, p),
                _distribution_edge_inversepow(n, p),
                _distribution_sine_weighted(n, p),
                _distribution_beta_sym(n, max(p, 0.1)),   # avoid p<0.1 for beta (log‐singular)
            )
            for base in families:
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best (mix) len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the strongest ~12 seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard against missing seed
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Continuous optimisation on the best seeds
    remaining = deadline - time.time()
    # Reserve about 30 s for final polishing
    optimise_deadline = time.time() + max(10.0, remaining - 30.0)
    print(f"[{time.time():.1f}] start optimisation phase (≈{optimise_deadline - time.time():.1f}s)")

    # Use the top few seeds for a short run each
    seeds_to_use = [seq for _, seq in top_candidates[:6]]
    for idx, seed_vec in enumerate(seeds_to_use, 1):
        if time.time() >= optimise_deadline:
            break
        sub_deadline = time.time() + max(5.0, (optimise_deadline - time.time()) / (len(seeds_to_use) - idx + 1))
        print(f"[{time.time():.1f}] seed {idx}/{len(seeds_to_use)} – length {len(seed_vec)} – time budget {sub_deadline - time.time():.1f}s")

        # smoothed gradient descent (first half of the allocated time)
        mid = time.time() + 0.5 * (sub_deadline - time.time())
        best_seq, best_score = _smoothed_gradient_descent(
            seed_vec.copy(),
            mid,
            best_seq=overall_best_seq.copy(),
            best_score=overall_best_score,
            lr0=0.8,
            temp0=0.14,
            temp_decay=0.9992,
            eval_interval=150,
            patience=2000,
        )
        if best_score < overall_best_score:
            overall_best_score, overall_best_seq = best_score, best_seq.copy()
            print(f"[{time.time():.1f}] after smoothed descent new global best {overall_best_score:.6f}")

        # mirror‑descent with momentum (remaining half)
        best_seq, best_score = _subgradient_descent_momentum(
            best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq.copy(),
            best_score=overall_best_score,
            lr0=0.9,
            active_frac=0.08,
            max_active=40,
            beta=0.9,
            eval_interval=200,
            patience=3000,
        )
        if best_score < overall_best_score:
            overall_best_score, overall_best_seq = best_score, best_seq.copy()
            print(f"[{time.time():.1f}] after momentum descent new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=400_000,
        )
        print(f"[{time.time():.1f}] after pairwise best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final simulated‑annealing search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.012,
            decay=0.9994,
        )
        print(f"[{time.time():.1f}] after SA best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score (exact) ≈ {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
