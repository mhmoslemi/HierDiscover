"""
Propose a non‑negative step function (list of heights) that (almost) minimizes
``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
--------------------------------
1. **Seed generation** – Build a diverse pool of probability vectors from many
   analytic families (uniform, arcsine, beta, edge‑weighted, Hann, generalized‑normal,
   power‑law, random Dirichlet, sparse, triangular, Gaussian bumps).  Keep the best
   30 candidates of a *single* length (the length that yields the smallest objective).

2. **Exact mirror descent** – Run exponentiated‑gradient descent on each of the
   top seeds (sub‑gradient of the active convolution entry).  This quickly drives
   the max‑convolution down.

3. **Genetic refinement** – Maintain a population of the best few vectors of the
   chosen length.  Repeatedly generate children by convex combination or by
   small random perturbations, keep the best individuals and update the global
   best.  This explores the neighbourhood of good solutions more globally than
   pure gradient steps.

4. **Progressive up‑sampling** – While time permits, double the length of the
   current best solution, smooth it, and repeat steps 2–3 on the larger vector.
   Larger n allows a smaller coefficient *c* = n·max(conv) and therefore a lower
   final objective.

5. **Final annealing** – Pairwise mass‑transfer simulated annealing (as in the
   previous baseline) for polishing.

All operations respect the simplex (non‑negative, sum = 1), which makes the
objective scale‑invariant.  The code prints intermediate scores so you can see
the progress when it runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast linear self‑convolution using real FFT
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective used by ``evaluate_sequence`` (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – always return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_generalized_normal(n: int, q: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n - 0.5
    w = np.exp(- (np.abs(x) / 0.5) ** q)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    """Linear rise then fall (peak at centre)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = 1.0 - np.abs(i - center) / (center + 1)
    w /= w.sum()
    return w


def _distribution_gaussian_bump(n: int, rng: np.random.Generator) -> np.ndarray:
    """A single Gaussian bump placed at a random location."""
    mu = rng.uniform(0.2, 0.8) * (n - 1)
    sigma = rng.uniform(0.05, 0.15) * n
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the *active* convolution entry (sub‑gradient of max‑conv)
# ----------------------------------------------------------------------
def _grad_of_max_conv(a: np.ndarray) -> tuple[np.ndarray, int]:
    """
    Compute the sub‑gradient of the currently maximal convolution entry.
    Returns (grad, active_index).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    k = int(np.argmax(conv))

    grad = np.zeros_like(a)
    if k < n:
        grad[: k + 1] = 2.0 * a[k::-1]
    else:
        start = k - (n - 1)
        grad[start:] = 2.0 * a[start:][::-1]
    return grad, k


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (mirror descent on the simplex)
# ----------------------------------------------------------------------
def _exp_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """Perform exponentiated‑gradient descent on the exact objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _ = _grad_of_max_conv(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] EG it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise mass‑transfer simulated annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; accept according to annealing."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling while preserving the simplex
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Simple smoothing (3‑point moving average)
# ----------------------------------------------------------------------
def _smooth(a: np.ndarray) -> np.ndarray:
    kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
    a_s = np.convolve(a, kernel, mode='same')
    a_s = np.maximum(a_s, 0.0)
    if a_s.sum() == 0.0:
        a_s[:] = 1.0 / a_s.shape[0]
    else:
        a_s /= a_s.sum()
    return a_s


# ----------------------------------------------------------------------
# Genetic‑style population improvement (crossover + mutation)
# ----------------------------------------------------------------------
def _genetic_improvement(
    population: list[tuple[np.ndarray, float]],
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float, list[tuple[np.ndarray, float]]]:
    """
    Maintain a small population (≤30) of good candidates.
    Repeatedly create children by convex combination or by random perturbation,
    keep the best individuals and update the global optimum.
    """
    n = best_seq.shape[0]
    pop_scores = [sc for _, sc in population]

    while time.time() < deadline:
        # decide crossover or mutation
        if rng.random() < 0.7:
            # crossover
            i, j = rng.choice(len(population), size=2, replace=False)
            w = rng.random()
            child = w * population[i][0] + (1.0 - w) * population[j][0]
        else:
            # mutation of a random parent
            i = rng.choice(len(population))
            child = population[i][0].copy()
            # pick a handful of coordinates to nudge
            k = max(1, int(n * 0.002))  # ~0.2 % of entries
            idx = rng.choice(n, size=k, replace=False)
            delta = (rng.random(k) - 0.5) * 0.04  # up to ±2 % locally
            child[idx] = np.clip(child[idx] + delta, 0.0, None)

        # re‑normalize to simplex
        s = child.sum()
        if s == 0.0:
            child = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            child /= s

        child_score = _objective(child)

        # update global best
        if child_score < best_score - 1e-12:
            best_score = child_score
            best_seq = child.copy()
            print(f"[{time.time():.1f}] genetic new global best {best_score:.6f}")

        # replace worst individual if child is better
        worst_idx = int(np.argmax(pop_scores))
        if child_score < pop_scores[worst_idx] - 1e-14:
            population[worst_idx] = (child, child_score)
            pop_scores[worst_idx] = child_score

    # sort population for convenience
    population.sort(key=lambda x: x[1])
    return best_seq, best_score, population

        # Gaussian bump seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_gaussian_bump(n, rng)
            sc = _objective(cand)
            length_candidates.append((sc, cand.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                best_len = n
                print(f"[{time.time():.1f}] new best bump len={n} score={sc:.6f}")

        # keep only the best few of *this* length
        length_candidates.sort(key=lambda x: x[0])
        length_candidates = length_candidates[:top_k]

        # if this length is the current best, store its top candidates
        if best_len == n:
            candidates_of_best_len = length_candidates

    # ------------------------------------------------------------------
    # 2.  Include a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_len = prev.shape[0]
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            # add to candidate pool for its length
            if best_len == prev.shape[0]:
                candidates_of_best_len.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Guarantee a non‑empty result.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_len = 2048
        best_seq = _distribution_uniform(best_len)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the top seeds (exponentiated gradient descent)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and candidates_of_best_len:
        # keep the 5 best seeds of the chosen length
        top_seeds = [seq for _, seq in sorted(candidates_of_best_len, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.30 / max(1, len(top_seeds))  # ~30 % of remaining time
        print(f"[{time.time():.1f}] EG descent on {len(top_seeds)} seeds (len={best_len})")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _exp_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=best_seq,
                best_score=best_score,
                eta0=0.9,
                eval_every=80,
            )
            if sc_opt < best_score - 1e-12:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after EG seed {idx} new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Genetic refinement on the current best length
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # construct a small population from the best candidates we already have
        pop = sorted(candidates_of_best_len, key=lambda x: x[0])[:min(12, len(candidates_of_best_len))]
        # ensure the current best is in the population
        if not any(np.allclose(seq, best_seq) for seq, _ in pop):
            pop.append((best_seq.copy(), best_score))
        pop.sort(key=lambda x: x[1])
        print(f"[{time.time():.1f}] genetic refinement (pop size={len(pop)})")
        sub_deadline = time.time() + (deadline - time.time()) * 0.20
        best_seq, best_score, _ = _genetic_improvement(
            pop, sub_deadline, best_seq, best_score, rng
        )
        print(f"[{time.time():.1f}] after genetic best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Progressive up‑sampling loop
    # ------------------------------------------------------------------
    while time.time() < deadline and best_seq.shape[0] < max_len_allowed:
        cur_n = best_seq.shape[0]
        new_n = cur_n * 2
        if new_n > max_len_allowed:
            break
        print(f"[{time.time():.1f}] upsampling {cur_n} → {new_n}")

        # upsample and smooth
        best_seq = _upsample(best_seq, new_n)
        best_seq = _smooth(best_seq)

        # evaluate immediately
        cur_score = _objective(best_seq)
        if cur_score < best_score:
            best_score = cur_score
            print(f"[{time.time():.1f}] upsampled new best {best_score:.6f}")

        # short EG burst
        sub_deadline = time.time() + (deadline - time.time()) * 0.15
        best_seq, best_score = _exp_gradient_descent(
            best_seq.copy(),
            sub_deadline,
            best_seq,
            best_score,
            eta0=0.7,
            eval_every=60,
        )
        print(f"[{time.time():.1f}] EG after upsample score {best_score:.6f}")

        # brief genetic polish on the larger vector
        pop = [(best_seq.copy(), best_score)]
        # add a few random Dirichlet perturbations as extra individuals
        for _ in range(4):
            pert = rng.dirichlet(np.full(new_n, 0.5))
            child = 0.8 * best_seq + 0.2 * pert
            child = _project_onto_simplex(child)
            pop.append((child, _objective(child)))
        sub_deadline = time.time() + (deadline - time.time()) * 0.10
        best_seq, best_score, _ = _genetic_improvement(
            pop, sub_deadline, best_seq, best_score, rng
        )
        print(f"[{time.time():.1f}] genetic after upsample score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7.  Final annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_score = _pairwise_annealing(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
            rng,
            max_trials=None,
        )
        print(f"[{time.time():.1f}] after annealing new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)/sum²): {best_score:.6f} "
        f"(length={len(best_seq)})"
    )
    return best_seq.tolist()
