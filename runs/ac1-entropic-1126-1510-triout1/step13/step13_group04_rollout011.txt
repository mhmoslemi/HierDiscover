"""
Propose a non‑negative step function (probability vector) that (almost)
minimises ``evaluate_sequence``.

Algorithm Overview
------------------
1. **Diverse seeding** – Build a pool of candidate probability vectors from a
   variety of analytic families (uniform, Hann, exponential, logistic,
   power‑law, Dirichlet, sparse).  Keep the best few (by the true objective).

2. **Smoothed‑gradient optimisation** – Replace the non‑smooth max‑convolution
   with a soft‑max (log‑sum‑exp) surrogate.
   For a current vector `a` we compute
        w_k ∝ exp(α·conv_k) ,   Σ w = 1
   where `conv = a*a`.  The gradient of the surrogate is
        ∇ = 2·(w ⊛ a_rev)      (cross‑correlation),
   obtained efficiently with one FFT per iteration.
   We run an Adam optimiser on this smooth objective while keeping `a`
   on the probability simplex (Euclidean projection).  The temperature
   parameter `α` is increased gradually (coarse → fine optimisation).

3. **Upsampling & refinement** – After the main phase we linearly up‑sample
   the current best vector to a larger length and repeat a short
   optimisation burst at the higher resolution.

4. **Final polishing** – A simple pairwise “mass‑transfer” annealing
   (random moves that are accepted if they improve the true objective) is
   used as a last polishing step.

All steps respect a 1000 s total budget (2 s safety margin).  The routine
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

# Basic utilities

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (unnormalised)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Analytic families – each returns a *probability* vector (sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float, decreasing: bool = True) -> np.ndarray:
    """a_i ∝ exp(‑lam·i) (decreasing) or exp(lam·i) (increasing)."""
    i = np.arange(n, dtype=np.float64)
    if decreasing:
        w = np.exp(-lam * i)
    else:
        w = np.exp(lam * i)
    w /= w.sum()
    return w


def _distribution_logistic(n: int, steep: float, decreasing: bool = True) -> np.ndarray:
    """a_i ∝ 1/(1+exp(steep·(i‑mid))) – monotone sigmoid."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    if decreasing:
        w = 1.0 / (1.0 + np.exp(steep * (i - mid)))
    else:
        w = 1.0 / (1.0 + np.exp(-steep * (i - mid)))
    w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law (p>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.02) -> np.ndarray:
    """Sparse random vector with given density."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Smoothed gradient of the max‑convolution objective
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Return gradient of the soft‑max surrogate of max(conv) for vector a.
    Also returns the raw convolution (needed for evaluation).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)                     # length 2n‑1
    # soft‑max weights (log‑sum‑exp trick)
    maxc = conv.max()
    shifted = conv - maxc
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / exp_vals.sum()                    # length 2n‑1, Σw = 1

    # gradient = 2·(w ⊛ a_rev)   (cross‑correlation)
    a_rev = a[::-1]
    size = 1 << ((len(w) + n - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    grad_full = np.fft.irfft(W * A, n=size)[: len(w) + n - 1]
    grad = 2.0 * grad_full[len(w) - 1 : len(w) - 1 + n]
    return grad, conv

    eval_every: int = 40,
) -> tuple[np.ndarray, float]:
    """Run Adam on the soft‑max surrogate, projecting onto the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    n = a.shape[0]

    for alpha in alpha_schedule:
        # a few dozen inner steps per α; stop early if out of time
        while time.time() < deadline:
            t += 1
            # compute gradient of smoothed objective
            grad, conv = _smooth_gradient(a, alpha)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            step = lr0 / (math.sqrt(t) + 1e-8) * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = a - step
            # keep on the probability simplex
            a = _project_onto_simplex(a, s=1.0)

            if t % eval_every == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam α={alpha:.1f} it={t} new best {best_score:.6f}")

            # advance to next α once a small number of inner iterations have been done
            if t % 2000 == 0:      # switch α after ~2000 steps (adjustable)
                break
        if time.time() >= deadline:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple pairwise annealing (final polishing)
# ----------------------------------------------------------------------
def _pairwise_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two bins; occasional uphill moves."""
    n = a.shape[0]
    start_time = time.time()
    total_time = deadline - start_time
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4
        if delta < 1e-16:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        elapsed = time.time() - start_time
        temperature = max(0.0, 1.0 - elapsed / total_time) * best_score * 0.01

        if cur < best_score - 1e-14:
            accept = True
        else:
            accept = temperature > 0.0 and rng.random() < math.exp(-(cur - best_score) / temperature)

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal i={i}->j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear up‑sampling (keeps total mass = 1)
# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a probability vector to a larger length."""
    old_n = a.shape[0]
    if new_n == old_n:
        return a.copy()
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a, left=0.0, right=0.0)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() == 0.0:
        a_new[:] = 1.0 / new_n
    else:
        a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the total 1000 s budget (2 s safety margin) and
    returns a plain ``list[float]`` ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seeds (different lengths & families)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]          # moderate sizes, later we may upsample
    exp_lams = np.logspace(-4, -0.5, 8)        # for exponential families
    log_steep = np.logspace(0.0, 2.0, 8)       # for logistic families
    power_ps = np.linspace(0.2, 2.5, 9)        # power‑law exponents

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating seeds for length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_hann(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} (det) score={sc:.6f}")

        # exponential decreasing / increasing
        for lam in exp_lams:
            if time.time() >= deadline:
                break
            cand_dec = _distribution_exponential(n, lam, decreasing=True)
            cand_inc = _distribution_exponential(n, lam, decreasing=False)
            for cand in (cand_dec, cand_inc):
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    typ = "dec" if cand is cand_dec else "inc"
                    print(f"[{time.time():.1f}] new best exp_{typ} lam={lam:.3g} len={n} score={sc:.6f}")

        # logistic families (monotone S‑shapes)
        for steep in log_steep:
            if time.time() >= deadline:
                break
            cand_dec = _distribution_logistic(n, steep, decreasing=True)
            cand_inc = _distribution_logistic(n, steep, decreasing=False)
            for cand in (cand_dec, cand_inc):
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    typ = "dec" if cand is cand_dec else "inc"
                    print(f"[{time.time():.1f}] new best logistic_{typ} s={steep:.3g} len={n} score={sc:.6f}")

        # power‑law (decreasing)
        for pwr in power_ps:
            if time.time() >= deadline:
                break
            cand = _distribution_power_law(n, pwr)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best powerlaw p={pwr:.3g} len={n} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.02, 2.0)
            cand = _distribution_random_dirichlet(n, rng, alpha=alpha)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet α={alpha:.3g} len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(3):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.02)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # keep only the most promising (30 total)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Include a previously discovered construction if available.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback if nothing was found.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam optimisation on the top seeds (smoothed surrogate)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        # select a few best seeds for deep optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining * 0.45 / max(1, len(seeds))   # ~45 % of remaining time
        print(f"[{time.time():.1f}] Adam optimisation on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            # α schedule from smooth to sharp
            alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0]
            best_seq, best_sc = _adam_optimize(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha_schedule=alpha_sched,
                lr0=0.07,
                eval_every=80,
            )
            if best_sc < overall_best_score - 1e-12:
                overall_best_score = best_sc
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional up‑sampling and short re‑optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_n = overall_best_seq.shape[0]
        target_n = cur_n * 2
        if target_n <= 64000:
            print(f"[{time.time():.1f}] upsampling {cur_n} → {target_n}")
            up_seq = _upsample(overall_best_seq, target_n)
            up_score = _objective(up_seq)
            if up_score < overall_best_score:
                overall_best_score = up_score
                overall_best_seq = up_seq
                print(f"[{time.time():.1f}] upsampled new best {overall_best_score:.6f}")

            # short Adam burst at larger size
            sub_deadline = time.time() + (deadline - time.time()) * 0.15
            alpha_sched = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
            best_seq, best_sc = _adam_optimize(
                overall_best_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha_schedule=alpha_sched,
                lr0=0.05,
                eval_every=60,
            )
            if best_sc < overall_best_score - 1e-12:
                overall_best_score = best_sc
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after up‑sample Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final polishing with pairwise annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_sc = _pairwise_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
            max_trials=None,
        )
        if best_sc < overall_best_score - 1e-12:
            overall_best_score = best_sc
            overall_best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] after final anneal new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)/sum²) = {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
