"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The search combines:

1.  A rich family scan (uniform, arcsine, sine, cosine, quadratic,
    Beta(p,q), edge‑weighted, power‑sum, random Dirichlet and sparse
    seeds).  The ~30 best candidates (by the exact objective
    ``2·n·max(conv)`` with the sum normalised to 1) are retained.

2.  Exact exponentiated‑gradient descent on the true sub‑gradient of the
    max‑convolution objective (mirror descent on the simplex).  A few
    top seeds are refined for a significant share of the time budget.

3.  Simulated‑annealing polishing: a random “move” transfers a fraction
    of mass from a source index to a destination index, updating the
    convolution in O(n) using the analytic formula for the change of a
    self‑convolution under a mass shift.  Worse moves are occasionally
    accepted according to a linear cooling schedule, allowing the
    algorithm to escape shallow local minima.

All helpers are top‑level functions; no closures or file IO are used.
The returned list sums to 1 (scale‑invariant for ``evaluate_sequence``)
and contains only non‑negative numbers.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution ``a*a`` using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()     # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective ``2·n·max(conv(a,a))`` for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Symmetric (or asymmetric) Beta(p,q) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w = np.clip(w, 0.0, None)
    total = w.sum()
    if total > 0:
        w /= total
    else:
        w = np.full(n, 1.0 / n)
    return w


def _distribution_power_sum(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{‑p} + (n‑i)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n + 1 - i) ** (-p)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Exact sub‑gradient of the max‑convolution -----------------------------
def _max_convolution_subgradient(a: np.ndarray) -> np.ndarray:
    """Sub‑gradient of  max_k Σ_i a_i a_{k-i}."""
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * max_val
    active = np.where(conv >= max_val - tol)[0]
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[k - idx]
    if active.size:
        grad /= active.size
    return grad

def _exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    report_every: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror descent on the exact max‑convolution
    objective, projected onto the simplex after each update.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        grad = _max_convolution_subgradient(a)
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)                     # normalise to sum 1
        if t % report_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑gd it={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    init_best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the exact objective.
    A random move transfers a fraction of mass from a source index to a
    destination index.  The full convolution is updated in O(n) using the
    analytic formula for its change under a mass shift; uphill moves are
    accepted with probability exp(‑Δ/T) where T linearly cools to 0.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_seq = a.copy()

    start = time.time()
    total_time = deadline - start
    max_move_frac = 0.4               # max fraction of source mass that may be moved

    # initial temperature – about a few percent of the current objective
    T0 = max(1e-6, init_best_score * 0.02)

    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling
        T = T0 * max(0.0, 1.0 - elapsed / total_time)

        # pick source (prefer larger mass) and destination (different)
        if a.sum() == 0.0:
            break
        src = rng.choice(n, p=a)          # weighted choice
        if a[src] < 1e-14:
            continue
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # propose a move
        delta = rng.random() * a[src] * max_move_frac
        if delta < 1e-15:
            continue

        # compute the new convolution (copy → O(n))
        new_conv = conv.copy()
        new_conv[src:src + n] -= 2.0 * delta * a
        new_conv[dst:dst + n] += 2.0 * delta * a
        new_conv[2 * src] += delta * delta
        new_conv[2 * dst] += delta * delta
        new_conv[src + dst] -= 2.0 * delta * delta

        new_max = new_conv.max()
        new_score = 2.0 * n * new_max
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
                accept = True

        if accept:
            # commit the move
            a[src] -= delta
            a[dst] += delta
            conv = new_conv
            cur_score = new_score
            if cur_score < best_score - 1e-15:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1. Rich family scan
    # ------------------------------------------------------------------
    lengths = [2048, 4096, 8192]
    # grids for Beta(p,q) and edge‑weighted families
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 10),      # 0.01 … 0.5
            np.linspace(0.5, 2.0, 12),
            np.linspace(2.0, 5.0, 8),
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 5)   # convex mixture with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric families -------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            # asymmetric beta
            for q in p_grid:
                beta_seq = _distribution_beta(n, p, q)
                for lam in lam_grid:
                    cand = _mix_with_uniform(beta_seq, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new best len={n} p={p:.3g} q={q:.3g} lam={lam:.2f} score={sc:.6f}"
                        )
                # limit the double loop a bit to respect time
                if len(p_grid) * len(p_grid) > 200:
                    break

            # edge‑weighted and power‑sum families
            edge_seq = _distribution_edge_weighted(n, p)
            pow_seq = _distribution_power_sum(n, p)
            for base in (edge_seq, pow_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds ---------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds -------------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the most promising ~30 candidates overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2. Include a previously discovered construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4. Refine top seeds with exact exponentiated gradient descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start exp‑gd on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            a_opt, sc_opt = _exp_gradient_descent(
                seed_seq,
                time.time() + per_seed,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.7,
                report_every=300,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exp‑gd seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Final simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing polishing")
        a_opt, sc_opt = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            init_best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Normalise and return
    # ------------------------------------------------------------------
    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    total = overall_best_seq.sum()
    if total == 0.0:
        overall_best_seq = _distribution_uniform(overall_best_seq.size)
    else:
        overall_best_seq = overall_best_seq / total

    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
