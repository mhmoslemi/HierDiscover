"""
Hybrid global+local optimisation for the “max‑convolution” step‑function problem.

* Global seed pool – diverse analytic families (uniform, arcsine,
  symmetric/asymmetric beta, edge‑weighted, edge‑inverse‑power,
  triangular, sine‑weighted) together with random Dirichlet/sparse vectors.
  A very wide range of shape parameters (including extreme edge‑heavy
  exponents) is explored for several moderate lengths (2‑20 k).

* Heavy refinement – exponentiated‑gradient (entropy mirror) descent on the
  exact non‑smooth objective 2·n·max(conv(a,a)) (with the sum of a fixed to
  1).  The active set consists of all convolution entries whose value is
  within a configurable fraction of the current maximum; a momentum term
  (β≈0.9) and a 1/√t step‑size are used.  Each seed receives a time‑slice of
  the remaining budget.

* Upsampling – the best distribution is linearly interpolated to a larger
  length (≈ 2×) and refined again, because the theoretical lower bound
  strictly decreases with n.

* Deterministic & stochastic clean‑up – a greedy pairwise mass‑transfer
  sweep followed by a cheap Metropolis‑style simulated‑annealing search
  removes any remaining spikes.

The whole procedure respects the global time budget (default 1000 s) and
returns a plain Python list of heights.  All vectors are projected onto
the probability simplex after each update, guaranteeing non‑negativity
and unit sum, which makes the objective scale‑invariant.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    # shift for numerical stability
    logw -= np.max(logw)
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^(‑p) where i,j are distances from the edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_inversepow(n: int, p: float) -> np.ndarray:
    """Weight ∝ i^(‑p) + j^(‑p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = i ** (-p) + j ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape (peak in the centre)."""
    i = np.arange(n, dtype=np.float64)
    w = np.minimum(i + 1, n - i)
    w = w.astype(np.float64)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_inverted_triangle(n: int) -> np.ndarray:
    """U‑shaped distribution (high at the edges, low in centre)."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / (np.minimum(i + 1, n - i) + 1.0)  # +1 avoids div‑by‑zero
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.008
) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """lam·U + (1‑lam)·base, U = uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


def _power_transform(seq: np.ndarray, q: float) -> np.ndarray:
    """Raise all entries to power q (q>0) and renormalise."""
    if q == 1.0:
        return seq.copy()
    w = np.power(seq, q)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        return seq.copy()
    return w / w_sum


# ----------------------------------------------------------------------
def _subgradient_descent_momentum(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    active_frac: float = 0.05,
    max_active: int = 40,
    beta: float = 0.9,
    eval_interval: int = 80,
    patience: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (entropy mirror) descent on 2·n·max(conv).

    The subgradient is built from all convolution entries whose value is
    at least ``active_frac`` below the current maximum.  A momentum term
    (β) smooths the direction; the step‑size decays as η₀/√t.
    """

    t = 0
    no_improve = 0
    momentum = np.zeros_like(a)
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 400_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)

        act = np.where(conv >= thresh)[0]
        if act.size > max_active:
            # keep only the largest contributors
            top_idx = np.argpartition(conv[act], -max_active)[-max_active:]
            act = act[top_idx]

        grad = np.zeros_like(a)
        for k in act:
            shift = k - ind_grid
            mask = (shift >= 0) & (shift < n)
            grad[mask] += 2.0 * a[shift[mask]]

        momentum = beta * momentum + (1.0 - beta) * grad
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * momentum)

        # projection onto simplex (ensures sum=1)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur + 1e-14 < best_score:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] it={t:,}   new global best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            # shake‑up to escape plateaus
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Greedy random mass transfer between two coordinates.
    Accept only improving moves.
    """
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e}  best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑style annealing: propose random mass transfers,
    accept if they improve or with probability exp(-(Δ)/T).
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur + 1e-14 < best_score:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e}  best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger size."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and
    returns a plain list of heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # containers for the global best
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Seed generation – very diverse families, broad parameter ranges
    lengths = [2000, 4000, 8000, 12000, 16000]   # moderate sizes for cheap evaluation

    # parameter grids
    p_sym_grid = np.concatenate(
        [
            np.logspace(-5, -1, 7),          # 1e‑5 … 0.1
            np.logspace(-1, 0.5, 7),         # 0.1 … 3.16
        ]
    )
    p_edge_grid = np.concatenate(
        [
            np.logspace(-4, -1, 6),          # 1e‑4 … 0.1
            np.logspace(-1, 1, 6),           # 0.1 … 10
        ]
    )
    lam_grid = np.linspace(0.0, 0.8, 9)  # mixing weight with uniform
    power_grid = [0.4, 0.6, 0.8, 1.2, 1.5, 2.0]  # for power transforms

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- sampling length {n}")

        # 1a) deterministic “simple” families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n),
                     _distribution_triangle(n), _distribution_inverted_triangle(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best (simple) {sc:.6f}")

        # 1b) symmetric beta, edge‑weighted, edge‑inverse‑power, sine‑weighted
        for p in p_sym_grid:
            if time.time() >= deadline:
                break
            beta_sym = _distribution_beta_sym(n, p)
            edge_w = _distribution_edge_weighted(n, p)
            edge_i = _distribution_edge_inversepow(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta_sym, edge_w, edge_i, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    mix = _mix_with_uniform(base, lam)
                    sc = _objective(mix)
                    top_candidates.append((sc, mix.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, mix.copy()
                        print(f"[{time.time():.1f}] new global best p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # 1c) edge‑weighted with larger exponents (p up to 10)
        for p in p_edge_grid:
            if time.time() >= deadline:
                break
            edge_w = _distribution_edge_weighted(n, p)
            edge_i = _distribution_edge_inversepow(n, p)
            for base in (edge_w, edge_i):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    mix = _mix_with_uniform(base, lam)
                    sc = _objective(mix)
                    top_candidates.append((sc, mix.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, mix.copy()
                        print(f"[{time.time():.1f}] new global best big‑p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # 1d) asymmetric Beta (two‑parameter random draws)
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = rng.uniform(0.001, 3.0)
            b = rng.uniform(0.001, 3.0)
            cand = _distribution_beta_asym(n, a, b)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best asym a={a:.3g} b={b:.3g} score={sc:.6f}")

        # 1e) random Dirichlet
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_dirichlet(n, rng)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet score={sc:.6f}")

        # 1f) random sparse seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.012)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best sparse score={sc:.6f}")

        # 1g) power transforms of a few base families
        base_families = [_distribution_arcsine(n), _distribution_beta_sym(n, 0.5),
                        _distribution_edge_weighted(n, 0.5), _distribution_inverted_triangle(n)]
        for base in base_families:
            for q in power_grid:
                cand = _power_transform(base, q)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best power q={q:.2f} score={sc:.6f}")

        # keep only the very best seeds (compact set)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0 or not np.isfinite(prev.sum()):
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            top_candidates.append((sc, prev.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Safe fallback – should never trigger
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Heavy mirror‑descent on a handful of elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        per_seed_time = remaining * 0.55 / max(1, len(seeds))
        print(f"[{time.time():.1f}] heavy descent on {len(seeds)} seeds, ~{per_seed_time:.1f}s each")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            a_opt, sc_opt = _subgradient_descent_momentum(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                active_frac=0.06,
                max_active=45,
                beta=0.88,
                eval_interval=100,
                patience=1300,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Upsample to a larger length and refine again (if time permits)
    remaining = deadline - time.time()
    if remaining > 120.0:
        new_len = min(35000, len(overall_best_seq) * 2)
        print(f"[{time.time():.1f}] up‑sampling to {new_len} points for final refinement")
        up_seq = _upsample(overall_best_seq, new_len)

        sub_deadline = time.time() + remaining * 0.65
        up_seq_opt, up_score = _subgradient_descent_momentum(
            up_seq,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            active_frac=0.045,
            max_active=30,
            beta=0.92,
            eval_interval=80,
            patience=1800,
        )
        if up_score < overall_best_score:
            overall_best_score, overall_best_seq = up_score, up_seq_opt.copy()
            print(f"[{time.time():.1f}] up‑sampled descent yielded new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Deterministic pairwise refinement (greedy)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] deterministic pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=500_000,
        )

    # ------------------------------------------------------------------
    # 7.  Final cheap simulated‑annealing clean‑up
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing clean‑up")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.009,
            decay=0.9994,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list (evaluate_sequence performs its own clipping)
    return overall_best_seq.tolist()
