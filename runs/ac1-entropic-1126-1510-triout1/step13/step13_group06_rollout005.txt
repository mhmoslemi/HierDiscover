"""
Propose a step‑function that minimizes the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is a full linear convolution.  The algorithm builds a large
collection of analytic and random seeds, keeps the most promising ones,
and then refines them with a fast *max‑subgradient* descent (Adam on the
soft‑max of the convolution).  The descent works directly on the logits
of a soft‑max representation, guaranteeing non‑negativity and unit sum.
After the gradient phase a cheap pairwise‑swap local search and a short
simulated‑annealing walk are applied.  The routine respects a total time
budget (default 1000 s) and returns a plain Python list.
"""

from __future__ import annotations

import math
import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of L = Σ_i w_i·(a*a)[i] w.r.t. a (as a vector of length n).
    Closed form: ∇L = 2·(rev(a) ⊛ w) restricted to the central n entries.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # length n
    need = len(w) + n - 1                # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)
    conv = conv[:need]                  # length need
    grad = 2.0 * conv[n - 1 : n - 1 + n]   # slice of length n
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    """Arcsine (Beta(½,½)) on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_law(n: int, s: float) -> np.ndarray:
    """Power‑law (Zipf) shape w_i ∝ (i+1)^{‑s}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / (i ** s)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(
    n: int, rng: np.random.Generator, alpha: float
) -> np.ndarray:
    """Dirichlet draw with symmetric concentration α."""
    return rng.dirichlet(np.full(n, alpha))


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Max‑subgradient (Adam) descent ---------------------------------------
def _max_subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.05,
    tol_factor: float = 1e-3,
    eval_interval: int = 100,
    patience: int = 2500,
) -> Tuple[np.ndarray, float]:
    """
    Adam on logits of a soft‑max parametrisation, using a sub‑gradient of the
    max‑convolution (uniform weight over the near‑max entries).
    """
    n = a.shape[0]
    eps = 1e-12

    # initialise logits from the current probabilities
    z = np.log(np.maximum(a, eps))

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    beta1_pow = 1.0
    beta2_pow = 1.0

    it = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < 300_000:
        it += 1

        # soft‑max (stable)
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # full convolution
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max

        # book‑keeping
        if cur_obj < best_score - 1e-14:
            best_score = cur_obj
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] descent it={it} new best {best_score:.6f}")
        else:
            no_improve += 1

        # inject jitter if stuck
        if no_improve >= patience:
            jitter = rng.normal(scale=0.02, size=n)
            z += jitter
            no_improve = 0
            continue

        # construct mask for the (approximately) maximal entries
        tol = cur_max * tol_factor
        mask = conv >= (cur_max - tol)
        w = np.zeros_like(conv)
        if mask.any():
            w[mask] = 1.0 / mask.sum()
        else:
            w[np.argmax(conv)] = 1.0

        # gradient w.r.t. a (weighted sum of convolution entries)
        grad_a = _weighted_gradient(a, w)

        # gradient w.r.t. logits (soft‑max chain rule)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        z -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # occasional exact evaluation (already done above)
        if it % eval_interval == 0:
            pass

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap local refinement -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> Tuple[np.ndarray, float]:
    """Transfer a random amount of mass between two coordinates if it helps."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9996,
) -> Tuple[np.ndarray, float]:
    """Cheap stochastic walk that accepts uphill moves with a decaying temperature."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}"
            )
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimizes `evaluate_sequence`.
    Returns a plain Python list; `evaluate_sequence` will normalise it.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 4.0  # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Generate a diverse pool of analytic / random seeds
    lengths = [2500, 3500, 5000, 7500, 11000]
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 10, base=10),   # 0.01 … 0.1
            np.logspace(-1, 0, 12, base=10),    # 0.1 … 1
            np.logspace(0, 0.7, 12, base=10),   # 1 … 5
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 9)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # deterministic families (uniform & arcsine)
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + optional mixing with uniform
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            power = _distribution_power_law(n, p)  # treat p as exponent

            for base in (beta, edge, sine, power):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # asymmetric beta (random parameters)
        for _ in range(5):
            if time.time() >= deadline:
                break
            a_param = 10 ** rng.uniform(-2, 0.7)   # 0.01 … ~5
            b_param = 10 ** rng.uniform(-2, 0.7)
            cand = _distribution_beta_asym(n, a_param, b_param)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] asym‑beta len={n} a={a_param:.3g} b={b_param:.3g} score={sc:.6f}"
                )

        # random Dirichlet seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = _distribution_dirichlet(n, rng, alpha)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] Dirichlet len={n} α={alpha:.3f} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] sparse len={n} score={sc:.6f}"
                )

        # keep only the most promising seeds (global ranking)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2.  Include any previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never trigger)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Max‑subgradient (Adam) descent on a handful of top seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # take the best few seeds for intensive optimisation
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting Adam descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _max_subgradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.05,
                tol_factor=1e-3,
                eval_interval=120,
                patience=2000,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 6.  Final simulated‑annealing walk
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final stochastic local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – `evaluate_sequence` will perform final checks.
    return overall_best_seq.tolist()
