"""
Propose a near‑optimal non‑negative step function for ``evaluate_sequence``.

The algorithm works in several stages within the allotted time budget:

1. **Broad analytic seeding** – generate many candidate vectors from
   symmetric and *asymmetric* beta families, sine‑power windows,
   cosine windows, power‑law decays, exponential tails, sparse Dirichlet
   draws, etc.  All vectors are normalised to sum = 1 and their
   ``2·n·max(a*a)`` score (identical to ``evaluate_sequence`` for a
   normalised vector) is evaluated with an FFT‑based convolution.
   The best candidate seeds the optimisation.

2. **Soft‑max Adam descent** – minimise a smooth approximation of the
   ``max`` (log‑sum‑exp) using Adam with a schedule of increasing
   soft‑max steepness ``α``.  After each update the vector is projected
   onto the probability simplex.

3. **Exact sub‑gradient descent** – using the true active set of
   maximal convolution entries, perform a momentum sub‑gradient method
   with a diminishing step size.

4. **Diffusion smoothing** – a few passes of a small 3‑point kernel to
   reduce isolated spikes while keeping the sum fixed.

5. **Precise simulated‑annealing** – a hill‑climber that moves a small
   mass ``δ`` from a source index to a target index.  The self‑convolution
   is updated *exactly* in O(n) time (including the quadratic correction
   term).  The temperature is slowly cooled; occasional jitter helps
   escape plateaus.

All stages respect the global time budget (default 1000 s).  The best
vector found is returned as a plain list of floats.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a):
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors via real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Return 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_unsym(n, p, q):
    """Beta density on [0,1] with shape parameters p,q (both>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) – clamp
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer mass `delta` from index i to index j.
    Updates vector `a` and its self‑convolution `conv` *exactly* in O(n).
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()

    # linear part: shift the old vector
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old

    # quadratic correction
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta

    a[i] -= delta
    a[j] += delta

    it_no_improve = 0
    T = cur_score * 0.08               # initial temperature (~8 % of score)
    decay = 0.9998                     # slower cooling for thorough search
    max_delta_factor = 0.025           # move up to 2.5 % of source mass per step

    while time.time() < deadline:
        it += 1
        # source sampled proportionally to its mass
        source = np.random.choice(n, p=a)
        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * max_delta_factor, 0.03)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snapshot = a.copy()
        conv_snapshot = conv.copy()

        _apply_transfer(a, conv, source, target, delta)

        new_score = 2.0 * n * conv.max()

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snapshot
            conv = conv_snapshot
            it_no_improve += 1

        T *= decay

        # occasional jitter to shake out stagnation
        if it % 4000 == 0 and it_no_improve > 1500:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
```
