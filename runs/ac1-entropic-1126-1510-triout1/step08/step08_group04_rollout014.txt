def _lp_optimize_g(f, M):
    n = f.shape[0]
    # Build matrix A of shape (2*n-1, n)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for k in range(2*n-1):
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # j from lo..hi, i = k - j
        A[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]
    # constraints: A @ g <= M
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g) => minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None
    g = res.x
    # make sure non-negative
    g = np.maximum(g, 0.0)
    # normalize to sum 1
    s = g.sum()
    if s <= 0:
        return None
    g /= s
    return g
```

            g = _lp_optimize_g(a, M)
            if g is None:
                print(f"[{time.time():.1f}] LP failed at iter {lp_it}")

            else:

                blend_t *= 0.5
                if blend_t < 1e-4:
                    break

    # final hill climbing
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # locate a maximal convolution entry
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find pair i,j with max product within this k
        best_i = i_start
        best_j = k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        # choose source = index with larger a among the pair
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)

            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill iter={it} new best {best_score:.6f}")
        else:

            conv = conv_snap

            a = a + jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter at hill iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative vector) with a small value of
evaluate_sequence().  The algorithm proceeds in four phases:

1. **Broad initial search** – analytic families (uniform, arcsine,
   triangular, exponential, beta, sine‑power, Gaussian) and many random
   Dirichlet draws are generated for a handful of lengths.  The best
   candidate (lowest 2·n·max self‑convolution) is kept.

2. **Soft‑max mirror descent** – the current best vector is refined for a
   large number of iterations.  The max‑convolution is approximated by a
   smooth soft‑max with temperature α that is exponentially increased,
   while an exponentially decreasing learning‑rate drives an
   exponentiated‑gradient (mirror) step.  The simplex constraint
   Σa_i = 1, a_i ≥ 0 is enforced by a Euclidean projection.

3. **Linear‑programming refinement** (optional, requires SciPy) – for the
   current vector f, solve the LP
        maximise   Σ g_j      subject to   (f∗g)_k ≤ M   ∀k,   g_j ≥ 0,
   where M = max(f∗f).  The resulting g is blended with f
   (f←(1–t)f+t g) and renormalised.  Repeating a few times typically
   reduces the peak further.

4. **Hill‑climbing** – a cheap O(n) update transfers a small amount of
   mass from an index that participates in the maximal convolution entry
   to a random low‑mass index.  Moves that decrease the peak are kept;
   occasional jitter helps escape plateaus.

The whole routine respects a 1000 s wall‑clock budget and prints
intermediate scores.  The returned list is directly usable by
evaluate_sequence().
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """
    Objective 2·n·max(a∗a) for a vector already normalised to Σa=1.
    Smaller is better.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families – all return a *vector* with Σ=1
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the grid {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    # keep a copy of the old a for the O(n) update
    a_old = a.copy()
    # convolution update (linear in delta)
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # the diagonal terms (i+i) and (j+j) receive the delta^2 contribution
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    # finally move the mass
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _lp_optimize_g(f, M):
    """
    Solve the LP: maximise Σg_j  subject to f∗g ≤ M  (component‑wise)  and g≥0.
    Returns a normalised vector g (Σg=1) or None on failure.
    """
    try:
        from scipy.optimize import linprog
    except Exception:  # SciPy not available
        return None

    n = f.shape[0]
    # Build the dense constraint matrix A such that A @ g ≤ M
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # rows k, columns lo..hi get f[k - j]
        idx = np.arange(lo, hi + 1)
        A[k, lo:hi + 1] = f[k - idx]
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximise Σg  ↔ minimise -Σg
    bounds = [(0, None)] * n

    # Highs is fast and works with dense matrices
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s <= 0:
        return None
    g /= s
    return g

        for alpha in (0.1, 0.5, 1.0, 2.0):

                candidates.append(vec)

        pass

    # ------------------------------------------------------------------

        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Soft‑max mirror descent with a long annealing schedule
    a = best_seq.copy()
    n = a.shape[0]

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
