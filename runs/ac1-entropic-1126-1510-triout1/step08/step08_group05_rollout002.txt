def lp_refine(a, iterations=10, tau=0.2):
    n = len(a)
    a = a.copy()
    for it in range(iterations):
        b = _fft_self_convolve(a)
        M = b.max()
        T = np.empty((2*n-1, n), dtype=np.float64)
        for i in range(n):
            T[i:i+n, i] = a
        # Solve LP
        res = linprog(c=np.full(n, -1.0), A_ub=T, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 / sum_g0  # now sum =1
        # Interpolation
        a = (1 - tau) * a + tau * g
        # Normalize
        a = np.maximum(a, 1e-12)
        a /= a.sum()
    return a

"""
Propose a low‑peak non‑negative step function.

The optimiser works in several stages:

1. **Broad initialisation** – evaluate many analytic families (uniform,
   arcsine, beta, power‑law, exponential, sine‑power, cosine windows)
   together with a few sparse and Dirichlet random draws for several
   lengths.  The best of these candidates seeds the subsequent optimisation.

2. **Soft‑max Adam descent** – minimise a smooth approximation of the
   ``max`` of the self‑convolution using the log‑sum‑exp trick.
   The “softness’’ parameter `alpha` is increased gradually; after each
   step the vector is projected onto the probability simplex.

3. **Linear‑program refinement** – for a fixed vector `a` we solve

        maximise   sum(g)
        subject to  a * g ≤ M   (full linear convolution)
                    g ≥ 0

   where `M = max(a*a)`.  This is a dense LP whose optimal solution is an
   extreme point; mixing the current `a` with the LP solution reduces the
   peak of the convolution.  The LP is solved with SciPy’s HiGHS
   optimiser.  A few (≤ 15) alternating LP / convex‑combination steps are
   performed.

4. **Mild diffusion smoothing** – a few passes of a 3‑point kernel to
   damp high‑frequency artefacts.

5. **Simulated‑annealing hill‑climber** – an O(n) incremental update
   (`_apply_transfer`) moves a small amount of mass from a source index
   to a target index.  Moves are accepted if they improve the objective
   or with a temperature‑controlled probability.  Temperature decays
   exponentially; occasional jitter re‑injects diversity.

All steps respect the global time budget (default 1000 s).  The best
sequence seen during the run is returned as a plain list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is proportional to ``evaluate_sequence`` (the denominator is 1²).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families – each returns a vector with sum = 1
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals

def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Transfer a mass `delta` from index i to index j,
    updating the vector `a` and its full self‑convolution `conv` in O(n)
    time (the convolution is the linear convolution of length 2·n‑1).
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    # The contribution of a[i] (or a[j]) to the full self‑convolution
    # appears exactly n times, shifted.
    a_i_old = a[i]
    a_j_old = a[j]

    # Update convolution: subtract old contribution of a[i], add new contribution of a[j]
    conv[i:i + n] -= 2.0 * delta * a_i_old
    conv[j:j + n] += 2.0 * delta * a_i_old   # note: use the *old* a_i for cross terms
    conv[i:i + n] -= 2.0 * delta * a_j_old
    conv[j:j + n] += 2.0 * delta * a_j_old

    # Quadratic self‑terms (i,i) and (j,j)
    conv[2 * i] += delta * delta - 2 * delta * a_i_old
    conv[2 * j] += delta * delta - 2 * delta * a_j_old

    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix T such that T @ g = a * g  (full linear convolution).
    Shape of T: (2·n‑1) × n.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    T = np.empty((rows, n), dtype=np.float64)
    # Each column j contains a shifted copy of a
    for j in range(n):
        T[j:j + n, j] = a
    return T


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, iterations: int = 12, tau: float = 0.2,
               min_improve: float = 1e-8) -> np.ndarray:
    """
    Repeated linear‑program refinement.
    Returns a (potentially) improved vector with sum = 1.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(best)
    for it in range(iterations):
        # compute current bound
        b = _fft_self_convolve(a)
        M = b.max()

        # Build convolution matrix for constraints a * g ≤ M
        T = _build_convolution_matrix(a)

        c = -np.ones(n, dtype=np.float64)               # maximise sum(g)  -> minimise -sum(g)
        ub = np.full(2 * n - 1, M, dtype=np.float64)   # right‑hand side
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=T, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            # LP failed – stop refining
            break

        g0 = res.x
        s = g0.sum()
        if s <= 0:
            break
        g = g0 / s  # normalise to sum = 1

        # convex combination
        a = (1.0 - tau) * a + tau * g
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        cur_score = _max_convolution_score(a)
        if cur_score + min_improve < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] LP iter={it+1} new best {cur_score:.6f}")

        # early stop if improvement stalls
        if cur_score >= best_score - min_improve:
            break
    return best
