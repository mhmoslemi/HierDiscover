def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Determine n
    try:
        n = len(height_sequence_1)  # if exists
    except NameError:
        n = 3000
    if n < 500:
        n = 500
    # Starting sequence
    try:
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() == 0:
            a = _uniform_distribution(n)
        else:
            a = a / a.sum()
            if a.shape[0] != n:
                # if length mismatch, resize using interpolation
                a = _interpolate_seq(a, n)
    except Exception:
        a = _arcsine_distribution(n)  # or uniform
    # Compute best as we go
    best_a = a.copy()
    best_score = _max_convolution_score(best_a)
    print(f"[{time.time():.1f}] initial n={n} cheap score={best_score:.6f}")

    # LP iteration hyperparams
    max_lp_iters = 12
    iteration = 0
    while time.time() < deadline - 5 and iteration < max_lp_iters:
        iteration += 1
        # compute max conv
        conv = _fft_self_convolve(a)
        M = conv.max()
        # build A matrix
        n = a.shape[0]
        # Compute rows and cols as described
        i = np.arange(n)[:, None]
        j = np.arange(n)[None, :]
        rows = (i + j).ravel()
        cols = j.ravel()
        data = np.tile(a, n)
        A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        # Solve LP
        # maximize sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds,
                                    method='highs',
                                    options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})
        if not res.success:
            print(f"[{time.time():.1f}] LP iteration {iteration} failed: {res.message}")
            break
        g0 = res.x
        S = g0.sum()
        if S <= 1e-12:
            print(f"[{time.time():.1f}] LP iteration {iteration} got zero sum")
            break
        g = g0 / S
        # combine
        t = 0.2  # maybe adapt
        a = (1 - t) * a + t * g
        # compute cheap score
        cur_score = _max_convolution_score(a)
        print(f"[{time.time():.1f}] LP iteration {iteration} cheap score={cur_score:.6f}")
        if cur_score < best_score - 1e-12:
            best_score, best_a = cur_score, a.copy()
    # After LP loop, run subgradient descent as fallback

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Initial seeding** – build a handful of analytically‑motivated
   distributions (arcsine, triangular, Gaussian, Cauchy, uniform) and a few
   random Dirichlet draws, all normalised to sum = 1.  Keep the best cheap
   score ( ``2·n·max(conv)`` ) as the starting vector.

2. **Iterative LP refinement** – for a fixed length *n* the problem
   “maximise Σ g_j subject to g≥0 and (f∗g)_k ≤ M” is a linear programme.
   Solving it yields a new feasible vector *g* that can be mixed with the
   current *f* without increasing the max‑convolution bound *M*.
   Repeatedly solve the LP, renormalise *g* and replace
   ``f ← (1‑t)·f + t·g`` (with a moderate mixing weight *t*).  Each LP
   reduces the objective or leaves it unchanged.

3. **Sub‑gradient descent** – after the LP phase, run a plain projected
   sub‑gradient method for a few hundred‑thousand iterations.  This gives
   a finer local optimisation.

4. **Hill‑climbing fine‑tuning** – finally perform cheap random mass‑transfer
   moves guided by the sub‑gradient.  Small transfers are accepted only
   if they improve the cheap score.

All steps respect the global time budget (default 1000 s).  The routine
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize

# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# --------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# --------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# --------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 0.6):
    """Plain projected sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families ------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.33) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw from Dirichlet(α,…,α)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# --------------------------------------------------------------
def _pick_initial_vector(n: int, seed: int = 42) -> np.ndarray:
    """Return the best among a small portfolio of analytic/random candidates."""
    np.random.seed(seed)
    candidates = [
        _arcsine_distribution(n),
        _triangular_distribution(n),
        _gaussian_distribution(n, sigma_frac=0.33),
        _cauchy_distribution(n, scale_frac=0.30),
        _uniform_distribution(n),
    ]
    # a few Dirichlet draws
    for a in (0.3, 0.6, 1.0):
        for _ in range(2):
            candidates.append(_random_dirichlet(n, alpha=a))
    # pick the one with smallest cheap score
    best = min(candidates, key=_max_convolution_score)
    return best.copy()
