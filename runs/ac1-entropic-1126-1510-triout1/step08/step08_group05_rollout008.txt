       full self‑convolution using O(n) incremental updates.
All phases respect a hard time budget (default 1000 s).  The routine
returns the best vector (as a list of floats) found within the budget.
"""

# ----------------------------------------------------------------------
# FFT based convolution utilities
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Simple softmax (stable) – used for the log‑parameterisation
def _softmax(z: np.ndarray) -> np.ndarray:
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact objective (2·n·max(b)) for a probability vector (sum=1)
def _max_convolution_score(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families – all return a vector whose entries sum to 1
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Incremental self‑convolution update for a mass transfer move
def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Transfer a mass `delta` from index i to index j.
    Updates the probability vector a and its full self‑convolution conv
    in O(n) time (conv has length 2·n‑1).
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part: subtract 2*delta*a_old[i] from conv[i : i+n]
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    # finally move the mass
    a[i] -= delta
    a[j] += delta

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function with minimal evaluation score."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5        # tiny safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Generate a large pool of analytic and random candidates.
    length_options = [2000, 3000, 4000, 5000, 8000]
    pool = []

    for n in length_options:
        if time.time() >= deadline:
            break

        # deterministic families
        families = [
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_triangular,
            lambda n=n: _distribution_exponential(n, lam=0.003),
            lambda n=n: _distribution_exponential(n, lam=0.02),
        ]

        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            families.append(lambda n=n, p=p: _distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            families.append(lambda n=n, pw=pw: _distribution_sine_power(n, pw))

        for c in (0.5, 0.46, 0.42):
            families.append(lambda n=n, c=c: _distribution_cosine_window(n, c))

        for p in (0.5, 1.0, 1.5, 2.0):
            families.append(lambda n=n, p=p: _distribution_power_law(n, p))

        # random Dirichlet / Gamma draws
        for _ in range(3):
            families.append(lambda n=n: np.random.gamma(0.5, 1.0, size=n))

        # a sparse random seed (few non‑zeros)
        def sparse_seed(n=n):
            vec = np.zeros(n, dtype=np.float64)
            k = max(2, n // 200)
            idx = np.random.choice(n, k, replace=False)
            vec[idx] = np.random.rand(k)
            vec /= vec.sum()
            return vec

        families.append(sparse_seed)

        for make_vec in families:
            if time.time() >= deadline:
                break
            vec = make_vec(n)
            vec = np.asarray(vec, dtype=np.float64)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            pool.append(vec)

    # optionally add the previous best solution if it exists

    except Exception:
        pass

    if not pool:  # safety fallback
        n = 2000
        pool.append(_distribution_uniform(n))

    # ------------------------------------------------------------------
    # 2.  Keep the top‑k seeds according to the exact objective.
    scores = [_max_convolution_score(v) for v in pool]
    idx_sorted = np.argsort(scores)
    top_k = min(5, len(pool))
    best_indices = idx_sorted[:top_k]
    seeds = [pool[i] for i in best_indices]

    # initialise global best from these seeds
    for vec in seeds:
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
    print(f"[{time.time():.1f}] initial best score {best_score:.6f} (len={best_seq.shape[0]})")

    # ------------------------------------------------------------------
    # 3.  Smooth‑max optimisation with L‑BFGS‑B (continuation in α).
    if minimize is not None:
        for seed_vec in seeds:
            if time.time() >= deadline:
                break
            n = seed_vec.shape[0]
            # logits initialisation
            x = np.log(seed_vec + eps)

            # continuation schedule (α, max iterations for this stage)
            schedule = [(1.0, 50), (5.0, 50), (20.0, 50),
                        (100.0, 50), (500.0, 50), (2000.0, 30)]

            for alpha, maxiter in schedule:
                if time.time() >= deadline:
                    break

                # objective+gradient for the current α
                def fun(x, alpha=alpha, n=n):
                    a = _softmax(x)                         # a_i ≥0, sum=1
                    b = _fft_self_convolve(a)                # length 2n‑1
                    maxb = b.max()
                    # soft‑max weights over the convolution entries
                    c = alpha * (b - maxb)
                    w = np.exp(c)
                    w_sum = w.sum()
                    w /= w_sum
                    # gradient w.r.t. a (formula from the reference)
                    conv_rev = _fft_convolve(a[::-1], w)
                    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]
                    # chain rule: ∂obj/∂x = a * (grad_a - a·grad_a)
                    dot = np.dot(a, grad_a)
                    grad_x = a * (grad_a - dot)
                    # smooth approximation of max
                    obj = maxb + (1.0 / alpha) * np.log(w_sum)
                    return obj, grad_x

                # L‑BFGS‑B step
                res = minimize(fun, x, method="L-BFGS-B", jac=True,
                               options={"maxiter": maxiter, "disp": False})
                x = res.x
                a = _softmax(x)
                a = np.maximum(a, eps)
                a /= a.sum()
                sc = _max_convolution_score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] improved score {sc:.6f} (α={alpha})")
    else:
        # Fallback: a very small number of Adam steps (if SciPy missing)
        print("[WARN] SciPy not available – skipping L‑BFGS‑B stage.")

    # ------------------------------------------------------------------
    # 4.  Final simulated‑annealing refinement (mass‑transfer moves).
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        best_score_sa = cur_score
        best_seq_sa = a.copy()

        T = cur_score * 0.1           # start temperature (≈10 % of current score)
        decay = 0.9999                # exponential cooling
        max_delta_factor = 0.02       # at most 2 % of a source mass per move
        it = 0

            it += 1
            # pick source proportional to its mass – high‑mass indices move more often
            source = np.random.choice(n, p=a)
            target = np.random.randint(n)
            if target == source:
                continue

            max_delta = min(a[source] * max_delta_factor, 0.05)
            if max_delta <= eps:
                continue
            delta = max_delta * np.random.rand()
            if delta <= eps:
                continue

            a_snapshot = a.copy()
            conv_snapshot = conv.copy()

            _apply_transfer(a, conv, source, target, delta)

            new_score = 2.0 * n * conv.max()
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                if T > 0:
                    prob = math.exp(-(new_score - cur_score) / T)
                    if np.random.rand() < prob:
                        accept = True

            if accept:
                cur_score = new_score
                if new_score < best_score_sa - 1e-12:
                    best_score_sa = new_score
                    best_seq_sa = a.copy()
                    print(f"[{time.time():.1f}] SA improved to {new_score:.6f}")
                T *= decay
            else:
                a = a_snapshot
                conv = conv_snapshot
                T *= decay

        # update global best if SA found better
        if best_score_sa < best_score:
            best_score = best_score_sa
            best_seq = best_seq_sa.copy()

        print("Could not evaluate final sequence:", exc)
