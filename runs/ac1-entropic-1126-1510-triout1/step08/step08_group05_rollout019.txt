# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution a*b (real FFT)."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """2·n·max(a*a) for a already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_triangular(n):

    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """a_i ∝ 1/(i+1)^p (decreasing)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


def _distribution_decreasing_power(n, p):
    """a_i ∝ (n‑i)^p   (monotone decreasing, p≥0)."""
    i = np.arange(n, dtype=np.float64)
    vals = (n - i) ** p
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j,
    updating both `a` and the full self‑convolution `conv` in O(n) time.
    """
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    conv[i:i + a.shape[0]] -= 2.0 * delta * a_old
    conv[j:j + a.shape[0]] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def propose_candidate(budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (approximately) minimises
    ``evaluate_sequence``.  A mixture of analytic seeding, Adam optimisation,
    exact sub‑gradient descent and simulated annealing is used.
    The routine respects the global time budget (default 1000 s) and
    returns the best sequence found as a plain list of floats.
    """
    np.random.seed(0)
    random.seed(0)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    # ------------------------------------------------------------------
    # 1.  Broad sampling of analytic families + random Dirichlet draws.
    lengths = [2000, 4000, 6000, 8000, 10000]
    candidate_pool = []                       # store (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        # analytic families -------------------------------------------------
        families = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.003),
            _distribution_exponential(n, lam=0.02),
        ]

        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            families.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            families.append(_distribution_sine_power(n, pw))

        for c in (0.5, 0.46, 0.42):
            families.append(_distribution_cosine_window(n, c))

        for p in (0.5, 1.0, 1.5, 2.0):
            families.append(_distribution_power_law(n, p))

        for p in (0.5, 1.0, 2.0, 3.0):
            families.append(_distribution_decreasing_power(n, p))

        # a sparse deterministic seed
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        families.append(sparse)

        # a few Dirichlet draws (shape=0.5 gives high variance)
        for _ in range(8):
            x = np.random.gamma(0.5, 1.0, size=n)
            x /= x.sum()
            families.append(x)

        for vec in families:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            candidate_pool.append((sc, vec))
            if sc < best_score:
                best_score, best_seq = sc, vec.copy()
                print(f"[{time.time():.1f}] init length={n} score={sc:.6f}")

    # keep only the ten best seeds -----------------------------------------
    candidate_pool.sort(key=lambda x: x[0])
    seeds = [vec.copy() for _, vec in candidate_pool[:10]]

    # ------------------------------------------------------------------
    # 2.  Local optimisation (Adam + exact sub‑gradient) for each seed.
    def _adam_optimize(a):
        """Adam on the soft‑max approximation of max(b)."""
        n = a.shape[0]
        schedule = [
            (1.0, 4000),
            (5.0, 4000),
            (20.0, 4000),
            (100.0, 4000),
            (500.0, 4000),
            (2000.0, 3000),
        ]
        beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
        lr = 0.12
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        best_local = a.copy()
        best_local_score = _max_convolution_score(a)

        for alpha, steps in schedule:
            for _ in range(steps):
                if time.time() >= deadline:
                    return best_local, best_local_score
                t += 1
                b = _fft_self_convolve(a)
                maxb = b.max()
                # softmax weights (numerically stable)
                w = np.exp(alpha * (b - maxb))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(b, 1.0 / b.size)
                else:
                    w /= w_sum
                grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad * grad)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

                a = _project_onto_simplex(a, 1.0)
                a = np.maximum(a, eps)
                a = a / a.sum()

                if t % 500 == 0:
                    cur = _max_convolution_score(a)
                    if cur < best_local_score - 1e-12:
                        best_local_score = cur
                        best_local = a.copy()
        return best_local, best_local_score

    def _subgradient_optimize(a):
        """Exact sub‑gradient descent (no momentum)."""
        n = a.shape[0]
        best_local = a.copy()
        best_local_score = _max_convolution_score(a)
        it = 0
        beta = 0.9
        velocity = np.zeros_like(a)

        while time.time() < deadline:
            it += 1
            b = _fft_self_convolve(a)
            cur_max = b.max()
            eps_rel = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(b >= cur_max - eps_rel)

            w = np.zeros_like(b)
            w[active] = 1.0
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

            velocity = beta * velocity + (1.0 - beta) * grad
            eta = 0.2 / np.sqrt(it)
            a = a - eta * velocity
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            cur = _max_convolution_score(a)
            if cur < best_local_score - 1e-12:
                best_local_score = cur
                best_local = a.copy()
                print(f"[{time.time():.1f}] subgrad it={it} improved {cur:.6f}")

        return best_local, best_local_score

    # run optimisation for each seed (time is shared)
    for seed_id, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        a = seed_vec.copy()
        a, sc = _adam_optimize(a)
        if sc < best_score - 1e-12:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] after Adam seed {seed_id} best {sc:.6f}")

        a, sc = _subgradient_optimize(a)
        if sc < best_score - 1e-12:
            best_score, best_seq = sc, a.copy()
            print(f"[{time.time():.1f}] after subgrad seed {seed_id} best {sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing hill‑climber on the exact convolution.
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_M = conv.max()
    cur_score = 2.0 * n * cur_M
    best_score = cur_score
    best_seq = a.copy()

    it = 0
    T = cur_score * 0.05                # start temperature (≈5 % of current score)
    decay = 0.9995                      # exponential cooling
    max_delta_factor = 0.02             # up to 2 % of source mass per move

    # pre‑compute a gradient for smart source/target choice
    while time.time() < deadline:
        it += 1
        # compute exact sub‑gradient (active set) only every few iterations
        if it % 200 == 0:
            b = conv
            cur_max = b.max()
            eps_rel = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(b >= cur_max - eps_rel)
            w = np.zeros_like(b)
            w[active] = 1.0
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        else:
            grad = None

        # choose source (large positive gradient) and target (small/negative)
        if grad is not None:
            src = int(np.argmax(grad))
            trg = int(np.argmin(grad))
        else:
            src = np.random.choice(n, p=a)
            trg = np.random.randint(n)

        if src == trg:
            continue

        max_delta = min(a[src] * max_delta_factor, 0.02)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()

        _apply_transfer(a, conv, src, trg, delta)

        new_M = conv.max()
        new_score = 2.0 * n * new_M

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            cur_M = new_M
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap

        T *= decay

        # occasional gentle perturbation to escape stagnation
        if it % 5000 == 0 and T < 1e-8:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            cur_M = conv.max()
            cur_score = 2.0 * n * cur_M
            T = cur_score * 0.05
            print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

    print(f"Final best score ≈ {best_score:.6f} (length={n})")
    return best_seq.tolist()
