"""
Propose a step function (non‑negative vector) that approximately minimises
``evaluate_sequence``.  

The algorithm consists of several stages:

1. **Broad initial scan** – generate many candidate distributions
   (uniform, arcsine, triangular, exponential, beta, sine‑power,
   cosine window, power‑law, Gaussian and a few random Dirichlet draws) for
   several lengths (2000–8000).  Keep the one with the smallest
   ``2·n·max(conv)`` score.

2. **Adam descent with a soft‑max approximation** – use a gradually
   sharpening exponential weight (parameter ``alpha``) to smooth the
   max‑operator, perform Adam updates on the simplex and keep the best
   iterate.

3. **Exact sub‑gradient refinement** – repeatedly move in the sub‑gradient
   direction of the true max‑convolution (active set) while projecting onto
   the simplex.

4. **Optional diffusion smoothing** (a few passes of a 3‑point kernel)
   to slightly spread mass without increasing the peak.

5. **Simulated‑annealing hill‑climber** – work directly on the full
   self‑convolution using O(n) incremental updates (mass transfer from a
   source index to a target index).  The annealing schedule is
   conservative (slow cooling) and the global best score is never
   overwritten by the current state (bug fix compared to the reference
   implementation).  Occasional jitter is applied if progress stalls.

All steps respect the global time budget (default 1000 s).  The final
vector is normalised to sum = 1 and all entries are non‑negative.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma_factor=0.25):
    """Discrete Gaussian centred in the interval."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part (each off‑diagonal pair contributes twice)
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (approximately) minimises
    ``evaluate_sequence``.  The routine follows the stages described in the
    module docstring and respects the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin
    eps = 1e-12

    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1.  Scan analytic families + random draws for several lengths.
    lengths = [2000, 3000, 4000, 5000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # core families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.003))
        candidates.append(_distribution_exponential(n, lam=0.02))

        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))

        for c in (0.5, 0.46, 0.42):
            candidates.append(_distribution_cosine_window(n, c))

        for p in (0.5, 1.0, 1.5, 2.0):
            candidates.append(_distribution_power_law(n, p))

        # Gaussian
        candidates.append(_distribution_gaussian(n, sigma_factor=0.22))

        # sparse random seed
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # Dirichlet / Gamma draws with different concentration parameters
        for conc in (0.1, 0.3, 0.5, 1.0, 2.0):
            x = np.random.gamma(conc, 1.0, size=n)
            if x.sum() == 0:
                continue
            x /= x.sum()
            candidates.append(x)

        # evaluate candidates
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously‑found global solution if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback if nothing was generated.
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Adam descent with a gradually sharpening soft‑max.
    a = best_seq.copy()
    n = a.shape[0]

    schedule = [
        (1.0,   2000),
        (5.0,   2000),
        (20.0,  4000),
        (100.0, 4000),
        (500.0, 4000),
        (2000.0,3000),
        (5000.0,2000)
    ]

    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_seq_adam = a.copy()
    best_score_adam = _max_convolution_score(a)

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if t % 250 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score_adam - 1e-12:
                    best_score_adam = cur
                    best_seq_adam = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

    # Update global best if Adam found improvement
    if best_score_adam < best_score:
        best_score = best_score_adam
        best_seq = best_seq_adam.copy()
        print(f"[{time.time():.1f}] Updated global best after Adam {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Exact max‑sub‑gradient refinement.
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cur:.6f}")

        if time.time() + 0.5 >= deadline:
            break

    # ------------------------------------------------------------------
    # 6.  Diffusion smoothing (optional – 8 short passes).
    if time.time() < deadline:
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        a = best_seq.copy()
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, eps)
            a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Diffusion smoothing new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 7.  Simulated‑annealing hill‑climber (full O(n) updates).
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    cur_M = conv.max()
    cur_score = 2.0 * n * cur_M

    # **Do NOT** reset best_score here – keep the best found so far.
    T = cur_score * 0.05
    decay = 0.9998
    max_delta_factor = 0.04
    it = 0
    it_no_improve = 0

    while time.time() < deadline:
        it += 1

        # pick source index proportionally to its mass
        source = np.random.choice(n, p=a)
        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * max_delta_factor, 0.1)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()

        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        new_score = 2.0 * n * new_M
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            cur_M = new_M
            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        T *= decay

        # occasional jitter to escape flat regions
        if it % 5000 == 0 and it_no_improve > 2000:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            cur_M = conv.max()
            cur_score = 2.0 * n * cur_M
            T = cur_score * 0.05
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

    # ------------------------------------------------------------------
    # Final normalisation and logging.
    best_seq = np.maximum(best_seq, 0.0)
    if best_seq.sum() == 0:
        # should not happen, but guard against division by zero
        best_seq = _distribution_uniform(len(best_seq))
    best_seq = best_seq / best_seq.sum()

    print(f"Final best score ≈ {best_score:.6f} (length={len(best_seq)})")
    # optional sanity check with the official evaluator (if available)
    try:
        real = evaluate_sequence(best_seq.tolist())
        print(f"evaluate_sequence reports: {real}")
    except Exception:
        pass

    return best_seq.tolist()
