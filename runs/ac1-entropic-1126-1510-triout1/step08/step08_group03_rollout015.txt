        # Implementation: For each i from 0..n-1, row index offset: for each j where f_i>0, we add coefficient f_i to column j at row i+j.

        data = []

            if fi == 0:
                continue
            # columns j: 0..n-1, rows k = i + j

            # but we need to restrict rows to length 2n-1
            rows_i = i + np.arange(n)
            cols_i = np.arange(n)

            rows.append(rows_i)
            cols.append(cols_i)
            data.append(data_i)

        cols = np.concatenate(cols)
        data = np.concatenate(data)
        A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

        # using HiGHS

            break
        g = res.x

        if sum_g <= 0 or sum_g <= 1.0 + 1e-12: # no improvement
            # maybe break
            break
        # scale to sum=1

        # recompute score
        cur_score = _max_convolution_score(a)

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The algorithm consists of four stages:

1. **Broad parametric sampling** – generate many candidate probability
   vectors from analytic families (arcsine, triangular, exponential,
   sine, cosine, uniform, symmetric/asymmetric Beta) and from random
   Dirichlet draws.  All candidates are normalised to sum = 1 and scored
   with the cheap surrogate 2·n·max(a*a).

2. **Heavy local optimisation** – the top‑k candidates are refined with
   several projected first‑order methods that respect the simplex:
   exponentiated‑gradient, Adam, a continuation method that smooths the
   max‑operator, and plain sub‑gradient descent.

3. **Linear‑program refinement** – for the current best sequence (if its
   length is moderate) we solve the LP “maximise Σ g  subject to
   (f * g) ≤ M” where ``f`` is the current sequence and ``M`` is its
   self‑convolution maximum.  The LP solution is projected back to the
   simplex and blended with ``f``; this operation provably cannot increase
   the objective and often yields a noticeable drop.

4. **Simulated‑annealing hill‑climbing** – finally we perform a
   temperature‑controlled random walk on the simplex, accepting
   worsening moves with a Metropolis probability.  Tiny jitter is added
   regularly to escape plateaus.

All steps respect a global 1000 s wall‑clock budget; the function returns
the best sequence found.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (once for k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 0.8,
                             gamma_factor: float = 1.005):
    """Continuation method: start with a smooth max (γ≈0.8) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ‑Dirichlet with α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()

def _build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Build the matrix A for the constraints (f * g)_k ≤ M.
    Row k, column j carries coefficient f_{k‑j} (zero when out of range).
    """
    n = f.size
    rows = []
    cols = []
    data = []
    for i, fi in enumerate(f):
        if fi == 0.0:
            continue
        rows.append(np.arange(i, i + n, dtype=np.int64))
        cols.append(np.arange(n, dtype=np.int64))
        data.append(np.full(n, fi, dtype=np.float64))
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine(a0: np.ndarray, deadline: float):
    """
    Linear‑program refinement:
    maximise Σ g subject to (a0 * g) ≤ M and g ≥ 0,
    then blend the solution back into the current point.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    while time.time() < deadline:
        M = _fft_self_convolve(a).max()
        # Build constraint matrix for (a * g) ≤ M
        A = _build_conv_matrix(a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear program: maximise 1ᵀg  ⇔ minimise -1ᵀg
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:  # no improvement possible
            break

        # normalise to the simplex and blend
        g = g / sum_g
        t = 0.5
        a = (1 - t) * a + t * g

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP refinement → {cur:.6f}")

        # stop if the improvement becomes negligible
        if best_score - cur < 1e-9:
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine follows the four‑stage pipeline described in the module
    doc‑string and respects the global time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Choose a baseline length; fall back to 3500 if no historic data.
    try:
        base_n = len(height_sequence_1)             # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    lengths = set()
    for f in (0.6, 0.8, 1.0, 1.2, 1.4, 1.6, 2.0, 2.5):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(15000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1. Broad sampling of analytic families and random draws
    candidates = []                 # (array, cheap_score)

    # a geometric grid for symmetric beta shape parameters
    beta_ps = np.geomspace(0.04, 0.9, num=30)

    for n in lengths:
        if time.time() >= deadline:
            break

        # Analytic families (no lambdas‑style calls)
        families = [
            _arcsine_distribution,
            _triangular_distribution,
            lambda n: _exponential_distribution(n, lam=0.003),
            _sine_distribution,
            _cosine_distribution,
            lambda n: np.full(n, 1.0 / n)
        ]
        for fun in families:
            a = fun(n)
            candidates.append((a, _max_convolution_score(a)))

        # Symmetric beta sweep
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # Random asymmetric beta
        for _ in range(3):
            p = np.random.choice(beta_ps)
            q = np.random.choice(beta_ps)
            a = _beta_asymmetric_distribution(n, p, q)
            candidates.append((a, _max_convolution_score(a)))

        # Random Dirichlet draws
        for _ in range(5):
            a = _random_dirichlet(n)
            candidates.append((a, _max_convolution_score(a)))

        # Random mixtures of two families
        for _ in range(3):
            # pick two base families at random
            fam1 = np.random.choice(['arcsine', 'triangular', 'uniform', 'sine', 'cosine'])
            fam2 = np.random.choice(['beta_sym', 'beta_asym', 'exponential'])

            if fam1 == 'arcsine':
                a1 = _arcsine_distribution(n)
            elif fam1 == 'triangular':
                a1 = _triangular_distribution(n)
            elif fam1 == 'uniform':
                a1 = np.full(n, 1.0 / n)
            elif fam1 == 'sine':
                a1 = _sine_distribution(n)
            else:  # cosine
                a1 = _cosine_distribution(n)

            if fam2 == 'beta_sym':
                p = np.random.choice(beta_ps)
                a2 = _beta_symmetric_distribution(n, p)
            elif fam2 == 'beta_asym':
                p = np.random.choice(beta_ps)
                q = np.random.choice(beta_ps)
                a2 = _beta_asymmetric_distribution(n, p, q)
            else:  # exponential
                lam = 10 ** np.random.uniform(-4, -2)
                a2 = _exponential_distribution(n, lam=lam)

            lam_mix = np.random.rand()
            a = lam_mix * a1 + (1.0 - lam_mix) * a2
            a = a / a.sum()
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 2. Include the previous best sequence, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top_seeds = [seq for seq, _ in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy optimisation on each seed (time‑sharing)
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        return best_seq.tolist()
    per_seed_budget = remaining_time * 0.7 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        seed_deadline = time.time() + per_seed_budget

        # Exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, seed_deadline,
                                        max_iters=80_000, eta0=0.9)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} exp‑grad → {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, seed_deadline,
                                max_iters=80_000, step0=0.3)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} Adam → {sc:.6f}")

        # Smooth‑max continuation
        seq, sc = _smooth_gradient_descent(seed_seq, seed_deadline,
                                            max_iters=150_000,
                                            eta0=0.6,
                                            gamma0=0.8,
                                            gamma_factor=1.005)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} smoothed → {sc:.6f}")

        # Sub‑gradient descent
        seq, sc = _subgradient_descent(seed_seq, seed_deadline,
                                       max_iters=300_000, eta0=0.7)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} subgrad → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Optional linear‑program refinement (only for moderate sizes)
    if len(best_seq) <= 5000 and time.time() < deadline - 10:
        try:
            best_seq, best_score = _lp_refine(best_seq, deadline)
        except Exception as e:
            print("LP refinement failed:", e)

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing / fine‑grained hill climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.05  # start with up to 5 % of total mass per move

    while time.time() < deadline:
        it += 1
        # geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (it / 8000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] SA iter={it} new best {best_score:.6f}")
        else:
            # Metropolis acceptance with a slowly cooling temperature
            T = max(1e-4, 0.1 * (0.99 ** (it // 1000)))
            if np.random.rand() < np.exp(-(new_score - cur_score) / T):
                a = a_new
                cur_score = new_score

        # occasional jitter to break flat plateaus
        if it % 4000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 1e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
