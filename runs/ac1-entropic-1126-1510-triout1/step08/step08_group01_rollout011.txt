"""
Optimiser for the step‑function upper‑bound problem.

The function builds many analytic seed vectors (arcsine, exponential,
beta, Dirichlet draws, the previous best etc.), keeps the few best
candidates and then refines them with two complementary optimisation
schemes:

1.  Classical sub‑gradient descent on the exact ∞‑norm objective.
2.  A smoothed “soft‑max” descent that gradually concentrates the
    weight on the current worst convolution index (p‑norm annealing).
Both schemes use Euclidean projection onto the probability simplex
(∑a_i=1, a_i≥0) after each update.  The best vector found before the
time budget expires is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, ∑x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    rho = rho[-1] if rho.size else -1
    theta = 0.0 if rho == -1 else cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate used in the original code: 2·n·max(a*a) for ∑a=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# exact sub‑gradient of the ∞‑norm (used for the classic scheme) -----
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(a*a) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # w is 1 on every active index, else 0
    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct the double‑counting on the diagonal (i=j) for even indices
    even_mask = (active % 2 == 0)
    diag_idx = active[even_mask] // 2
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# smoothed “soft‑max” descent (p‑norm annealing) -------------------------
def _softmax_descent(a0: np.ndarray, deadline: float,
                     p_schedule=None,
                     max_iters_per_p: int = 80_000,
                     step0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Minimise the ∞‑norm of a*a by a sequence of soft‑max approximations.
    p_schedule – list of integer powers, low to high (e.g. [2,4,8,16,...]).
    Returns the best vector seen and its surrogate score.
    """
    if p_schedule is None:
        p_schedule = [2, 4, 8, 16, 32, 64]

    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]

    for p in p_schedule:
        it = 0
        while time.time() < deadline and it < max_iters_per_p:
            it += 1
            conv = _fft_self_convolve(a)                 # length 2n‑1
            max_c = conv.max()
            # soft‑max weights – always ≤ 1, 1 on the current maximal entries
            w = np.exp(p * (conv - max_c))

            # weighted convolution for the gradient
            conv_rev = _fft_convolve(a[::-1], w)        # length 3n‑2
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # diagonal correction (the a_i·a_i term appears only once)
            diag_corr = a * w[2 * np.arange(n)]
            grad -= diag_corr

            # diminishing additive step
            eta = step0 / np.sqrt(it)
            a = a - eta * grad
            a = _project_onto_simplex(a, s=1.0)

            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
                # print progress for this p
                print(f"[{time.time():.1f}] p={p:2d} iter={it:6d} score={best_score:.6f}")

        # restart from the current best before increasing p
        a = best_a.copy()

        # early abort if time ran out
        if time.time() >= deadline:
            break

    return best_a, best_score

def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 200_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Classic sub‑gradient descent on the exact max‑norm."""
    a = a0.copy()

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()

    candidates.sort(key=lambda x: x[1])
    top_k = min(7, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Current best seed score = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # --------------------------------------------------------------
    # 4. intensive refinement of each top seed
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]
        print(f"\n=== refining seed {idx+1}/{len(top_candidates)} (n={n}) ===")

        # (a) classic sub‑gradient SGD
        seq, sc = _sgd_subgrad(seed_seq, deadline,
                               max_iters=200_000,
                               step0=0.5)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] SGD improved to {best_score:.6f}")

        # (b) soft‑max annealing descent
        seq, sc = _softmax_descent(seed_seq, deadline,
                                   p_schedule=[2, 4, 8, 16, 32, 64],
                                   max_iters_per_p=40_000,
                                   step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] Soft‑max improved to {best_score:.6f}")

        # (c) a quick extra SGD pass from the current best of this seed
        seq, sc = _sgd_subgrad(best_seq, deadline,
                               max_iters=80_000,
                               step0=0.4)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] final SGD tweak gave {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. global Frank‑Wolfe‑style pairwise mass transfer (lightweight)
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 60_000:
        it_fw += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # where we have too much mass
        i_dst = int(np.argmin(grad))   # where we need more mass
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta < 1e-14:
            break
        # full transfer
        a2 = a.copy()
        a2[i_src] -= delta
        a2[i_dst] += delta
        sc2 = _max_convolution_score(a2)
        if sc2 < cur_score - 1e-13:
            a = a2
            cur_score = sc2
            if sc2 < best_score - 1e-13:
                best_score, best_seq = sc2, a.copy()
                print(f"[{time.time():.1f}] FW iter {it_fw} new best {best_score:.6f}")
        else:
            # try half‑transfer as a fallback
            half = 0.5 * delta
            a2[i_src] = a[i_src] - half
            a2[i_dst] = a[i_dst] + half
            sc2 = _max_convolution_score(a2)
            if sc2 < cur_score - 1e-13:
                a = a2
                cur_score = sc2
                if sc2 < best_score - 1e-13:
                    best_score, best_seq = sc2, a.copy()
                    print(f"[{time.time():.1f}] FW half‑transfer new best {best_score:.6f}")

        # occasional tiny jitter to break stagnation
        if it_fw % 5000 == 0 and np.random.rand() < 0.1:
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    print(f"\nFinal best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # The optimiser works with l1‑normalised vectors; ``evaluate_sequence``
    # is homogeneous, so returning the normalised vector is fine.
    return best_seq.tolist()
