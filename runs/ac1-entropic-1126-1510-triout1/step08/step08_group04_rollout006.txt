def upsample(a, new_n):
    old_n = len(a)
    x_old = np.linspace(0,1,old_n)
    x_new = np.linspace(0,1,new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new
```

def hill_climb(a, conv, M, deadline):
    n = len(a)
    while time.time() < deadline:
        # Find current max index and active contributions
        k = int(conv.argmax())
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        # compute contributions
        con = a[i_start:i_end+1] * a[k - np.arange(i_start, i_end+1)]
        # source is index with max contribution
        idx_relative = np.argmax(con)
        source = i_start + idx_relative
        # choose target distinct randomly
        target = random.randint(0, n-1)
        if target == source:
            continue
        # delta magnitude: sample from uniform up to min(source mass*0.5, 0.05)
        max_delta = min(a[source]*0.5, 0.05)
        if max_delta <= 1e-12:
            continue
        delta = max_delta * random.random()
        # perform transfer
        a_old = a.copy()
        conv_old = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        # accept if new_M <= M or with probability
        if new_M < M - 1e-14:
            M = new_M
            # maybe update best
            # ...
        else:
            # revert
            a = a_old
            conv = conv_old
        # occasional jitter
        if random.random() < 0.001:
            a += np.random.randn(n)*1e-5
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a, conv, M
```

def _distribution_beta(p, q, n):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute pdf components, avoid underflow/overflow via log
    log_w = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(log_w)
    if np.all(w == 0) or np.isnan(w).any():
        w = np.full_like(x, 1.0 / n)
    else:
        w_sum = w.sum()
        if w_sum == 0: w = np.full_like(x, 1.0 / n)
        else: w = w / w_sum
    return w
```

"""
A multi‑start, multi‑scale local optimiser for the max‑self‑convolution problem.

- Generates many analytic families (uniform, arcsine, triangular, exponential,
  Beta(p,q), Gaussian, cosine‑window) and Dirichlet draws for several lengths.
- Keeps the top‑K candidates (by cheap 2·n·max (a*a) score) and refines each
  with a fast O(n) hill‑climber that moves a small amount of mass from the
  index contributing most to the current maximal convolution entry to a random
  target.  The hill‑climber accepts any improving move and occasionally
  accepts a worsening move (simulated‑annealing style) to escape plateaus.
- After a sweep without improvement a jittered copy of the best sequence is
  injected to maintain diversity.
- Finally the current best sequence is repeatedly up‑sampled (linear
  interpolation) to a larger length and hill‑climbed again – larger vectors
  allow a finer approximation of the (conjectured) optimal continuous shape.
- The whole search respects a 1000 s wall‑clock budget and returns the best
  vector (as a plain Python list of floats) found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """
    Cheap objective for a vector already normalised to sum = 1:
    2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – all return a vector whose sum is 1

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta(p, q, n):
    """Discrete Beta(p,q) on the grid { (i+0.5)/n }."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use logs to avoid under‑/overflow when p,q<1
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if np.isnan(w).any() or w.sum() == 0:
        w = np.full_like(x, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2, width = sigma·n."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # convolution update according to (a-δe_i+δe_j)*(a-δe_i+δe_j)
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _hill_climb(a, conv, cur_max, deadline):
    """
    Fast O(n) hill‑climber.
    Repeatedly moves a little mass from the index contributing most to the
    current maximal convolution entry to a random target.  Accepts any
    improvement; otherwise accepts with a tiny probability to permit escapes.
    Returns the (possibly) improved vector, its convolution and new max.
    """
    n = a.shape[0]
    eps = 1e-12
    while time.time() < deadline:
        # 1) locate the current maximal convolution index
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idxs = np.arange(i_start, i_end + 1)
        # contributions a[i]*a[k-i]
        contrib = a[idxs] * a[k - idxs]
        src = idxs[np.argmax(contrib)]

        # 2) pick a random destination (different from src)
        tgt = random.randint(0, n - 1)
        if tgt == src:
            continue

        # 3) choose a small transfer amount
        max_delta = min(a[src] * 0.5, 0.02)
        if max_delta <= eps:
            continue
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        # 4) trial move
        a_backup = a.copy()
        conv_backup = conv.copy()
        _apply_transfer(a, conv, src, tgt, delta)
        new_max = conv.max()

        # 5) acceptance test
        if new_max < cur_max - 1e-14:
            cur_max = new_max
        else:
            # occasional uphill moves (≈0.1 %)
            if random.random() < 0.001:
                cur_max = new_max
            else:
                a = a_backup
                conv = conv_backup

        # occasional global jitter to shake out stagnation
        if random.random() < 0.0005:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            cur_max = conv.max()

    return a, conv, cur_max


# ----------------------------------------------------------------------
def _upsample(a, new_n):
    """Linear interpolation of a probability vector to a longer length."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector (as a Python list of floats) found within the given
    wall‑clock budget (seconds).
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # a little safety margin
    eps = 1e-12

    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1) generate a broad set of analytic / random candidates for several lengths
    lengths = [1024, 2048, 4096, 8192, 16384]
    candidate_heap = []            # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        cand = []

        # Analytic families
        cand.append(_distribution_uniform(n))
        cand.append(_distribution_arcsine(n))
        cand.append(_distribution_triangular(n))
        cand.append(_distribution_exponential(n, lam=0.01))
        cand.append(_distribution_exponential(n, lam=0.05))

        # Beta families – a fairly dense grid
        for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0):
            for q in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
                cand.append(_distribution_beta(p, q, n))

        # Gaussian kernels
        for sigma in (0.07, 0.10, 0.15, 0.20, 0.30):
            cand.append(_distribution_gaussian(n, sigma))

        # Cosine windows
        for coeff in (0.5, 0.75, 0.9):
            cand.append(_distribution_cosine_window(n, coeff))

        # Random Dirichlet draws with various concentrations
        for alpha in (0.05, 0.1, 0.3, 0.5, 1.0, 2.0):
            cand.append(np.random.dirichlet(np.full(n, alpha)))

        # Sparse random start (few spikes)
        k = max(2, n // 200)
        sparse = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        cand.append(sparse)

        # Evaluate cheap score for each candidate
        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            candidate_heap.append((sc, vec.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] init len={n} new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2) incorporate a previously found solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            candidate_heap.append((sc, prev.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) keep only the top‑K promising candidates
    candidate_heap.sort(key=lambda x: x[0])
    K = 10
    candidates = candidate_heap[:K]   # list of (score, vector)

    # ------------------------------------------------------------------
    # 4) local hill‑climbing on each candidate (time‑sliced)
    while time.time() < deadline:
        any_improved = False
        for idx, (score, a) in enumerate(candidates):
            if time.time() >= deadline:
                break
            n = a.shape[0]
            # fresh copies for the local optimiser
            a_local = a.copy()
            conv = _fft_self_convolve(a_local)
            cur_max = conv.max()
            # allocate a short budget for this candidate (≈2 s)
            local_dead = min(deadline, time.time() + 2.0)
            a_local, conv, cur_max = _hill_climb(a_local, conv, cur_max, local_dead)
            new_score = 2.0 * n * cur_max
            # store the possibly improved candidate
            candidates[idx] = (new_score, a_local)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_local.copy()
                best_n = n
                any_improved = True
                print(f"[{time.time():.1f}] hill‑climb len={n} new best {new_score:.6f}")

            # early exit if we already beat the target
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

        # If a full sweep produced no improvement, inject a jittered copy of the current best
        if not any_improved:
            if time.time() >= deadline:
                break
            n = best_n
            jitter = np.exp(0.02 * np.random.randn(n))   # multiplicative jitter
            a_new = best_seq * jitter
            a_new = np.maximum(a_new, eps)
            a_new /= a_new.sum()
            sc = _max_convolution_score(a_new)
            candidates.append((sc, a_new))
            candidates.sort(key=lambda x: x[0])
            candidates = candidates[:K]   # keep the elite set
            print(f"[{time.time():.1f}] jitter injection, score {sc:.6f}")

    # ------------------------------------------------------------------
    # 5) Upsample the best solution to larger lengths (if time permits)
    while time.time() < deadline and best_n < 20000:
        new_n = min(best_n * 2, 20000)
        if new_n == best_n:
            break
        up = _upsample(best_seq, new_n)
        conv = _fft_self_convolve(up)
        cur_max = conv.max()
        # give the up‑sampled vector a modest amount of hill‑climbing time
        local_dead = min(deadline, time.time() + 5.0)
        up, conv, cur_max = _hill_climb(up, conv, cur_max, local_dead)
        new_score = 2.0 * new_n * cur_max
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_seq = up.copy()
            best_n = new_n
            print(f"[{time.time():.1f}] up‑sampled len={new_n} new best {new_score:.6f}")
        else:
            # even if not better, keep the up‑sampled version as a new candidate for future sweeps
            candidate_heap.append((new_score, up.copy()))
        # continue trying larger sizes while we have budget
        best_n = new_n
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 6) Final report
    print(f"[{time.time():.1f}] final best score {best_score:.6f} (length={best_n})")
    return best_seq.tolist()
