"""
A time‑bounded optimiser for ``evaluate_sequence`` that expands on the
previous version with a few extra ingredients:

1. **Richer analytic families** – a power‑law family
   a_i ∝ ( (i+½)(n‑i‑½) )^{-γ}  (γ ∈ [0.1,1.5]) is added.
2. **Smooth max surrogate** – minimise
   F_λ(a)= (1/λ)·log∑_k exp(λ·(a∗a)_k) with an Adam optimiser.
   λ is increased geometrically (5 → 40) to approach the true
   ``max(a∗a)`` while keeping the landscape smooth.
3. **More exhaustive hyper‑parameter sweeps** – finer grids for the
   Gaussian σ‑factor, the exponential rate and the β‑parameters.
4. **Mixture of top seeds** – cheap convex combinations of the best
   few candidates are inspected.
5. **More aggressive local optimisation** – longer Adam/Exp‑Grad/SGD
   runs and the smooth‑Adam phase.
6. **Frank–Wolfe style mass transfer** with a tiny line‑search (full
   or half transfer) and occasional random jitter.

The routine respects the supplied budget (≈ 1000 s) and returns the
best sequence found (normalised to sum = 1, which is the natural scale
for the surrogate 2·n·max(conv)).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for ``evaluate_sequence`` when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (sum(a)=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
# smooth surrogate (log‑sum‑exp) -------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float):
    """
    Gradient of  (1/lam)·log∑_k exp(lam·(a∗a)_k)  with respect to a.
    Returns (gradient, max_entry_of_convolution).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    mx = b.max()
    # stable softmax weights
    w = np.exp(lam * (b - mx))
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for the terms where i=j (even k)
    even_idx = np.arange(0, 2 * n - 1, 2)
    w_even = w[even_idx]                 # length n
    grad -= a * w_even
    return grad, mx

# analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _powerlaw_distribution(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑i‑½))^{‑γ} ."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / ((i + 0.5) * (n - i - 0.5)) ** gamma
    vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Dirichlet(α,…,α) on the simplex."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# mirror‑descent variants -----------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8,
                          decay: float = 0.9999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 200_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _adam_smooth_descent(a0: np.ndarray, deadline: float,
                         lam_start: float = 5.0,
                         lam_factor: float = 2.0,
                         outer_steps: int = 4,
                         inner_iters: int = 40_000,
                         step0: float = 0.3) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed Log‑Sum‑Exp surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    lam = lam_start
    for outer in range(outer_steps):
        # Adam state reset for each λ
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        for it in range(1, inner_iters + 1):
            if time.time() >= deadline:
                break
            grad, _ = _smooth_gradient(a, lam)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            eta = step0 / (np.sqrt(v_hat) + 1e-8)
            a = a - eta * m_hat
            a = _project_onto_simplex(a)
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-15:
                best_score = cur_score
                best_a = a.copy()
        lam *= lam_factor
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
def _mix_candidates(base: np.ndarray, others: list[np.ndarray],
                    deadline: float) -> tuple[np.ndarray, float]:
    """Try convex combinations of *base* with each vector in *others*."""
    best_a = base.copy()
    best_score = _max_convolution_score(base)
    alphas = np.linspace(0.0, 0.5, 11)   # 0,0.05,…,0.5
    for other in others:
        for a in alphas:
            if time.time() >= deadline:
                return best_a, best_score
            cand = (1 - a) * base + a * other
            cand = _project_onto_simplex(cand)
            sc = _max_convolution_score(cand)
            if sc < best_score - 1e-14:
                best_score, best_a = sc, cand.copy()
    return best_a, best_score
