def _distribution_root_uniform(n):
    # Build target convolution vector of length N = 2*n-1: uniform values = 1/(2*n-1) (to sum to 1?). Actually we need conv target ~ uniform (sum=1). So each entry should be 1/(2*n-1). We'll use raw ones to compute FFT.

    # Zero pad to length L (next power of two)
    L = 1 << ((2 * n - 1).bit_length())
    # Compute FFT
    B = np.fft.rfft(target, n=L)
    # sqrt magnitude
    A = np.sqrt(np.abs(B))
    # Inverse FFT to get candidate a (complex)
    a_complex = np.fft.irfft(A, n=L)
    # keep first n entries
    a = a_complex[:n]
    # Clip negatives
    a[a < 0] = 0.0
    # Normalize sum
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # schedule: list of (steps, lr)
    schedule = [
        (2000, 0.3),
        (2000, 0.2),
        (3000, 0.15),
        (3000, 0.1),
        (4000, 0.07),
        (4000, 0.05),
    ]
    for steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1
            # compute self-convolution

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.  

The search consists of three stages

1. **Rich seeding** – analytic families (uniform, arcsine, beta, …),
   a Fourier‑root of the uniform convolution and a few random Dirichlet
   vectors are generated for several lengths.  Their exact objective
   (2·n·max(conv(a,a))) is evaluated and the best few are kept.

2. **Projected Adam on the exact max‑convolution** – an Adam optimiser
   works on the true non‑smooth objective.  At every iteration the
   sub‑gradient of ``max(conv(a,a))`` is used (the weight vector is an
   indicator of the maximal convolution entries).  After the Adam step
   the vector is projected onto the probability simplex and symmetrised.
   A short schedule with decreasing learning‑rates is employed; the best
   vector seen is retained.

3. **Greedy pairwise mass transfers** – a cheap local walk moves a
   random amount of mass from a source index to a destination index and
   keeps the change only if the true objective improves.  This final
   polishing typically yields the lowest bound.

All stages respect the global time budget (default 1000 s).  The returned
list is a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    """Half‑sine shape, zero at the edges, positive inside."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A fairly broad pool of start vectors for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # beta family – many shape parameters
    for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials
    for lam in (0.001, 0.002, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # power‑law at the edges
    for gamma in (0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4):
        cand.append(_distribution_powerlaw_edges(n, gamma))

    # sine shape
    cand.append(_distribution_sine(n))

    # alternating patterns
    for stride in (2, 3, 4):
        cand.append(_distribution_alternating(n, stride))

    # two‑block spikes
    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random sparse start (~1 % non‑zeros)
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
# New analytic seed – square‑root of uniform convolution
def _distribution_root_uniform(n: int) -> np.ndarray:
    """
    Heuristic: take the (real) square‑root of the DFT of the uniform
    convolution (length 2n‑1) and invert the transform.  The result is
    clipped to non‑negative values and renormalised.
    """
    # target convolution: constant vector of length 2n‑1
    target = np.ones(2 * n - 1, dtype=np.float64) / (2 * n - 1)
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    B = np.fft.rfft(target, n=L)                 # FFT of target
    A = np.sqrt(np.abs(B))                       # keep only the magnitude
    a_complex = np.fft.irfft(A, n=L)             # inverse transform
    a = a_complex[:n]                            # keep first n coefficients
    a = np.where(a > 0, a, 0.0)                  # clip negatives
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Make the vector symmetric (v[i] = v[n‑1‑i]) while preserving sum."""
    return (v + v[::-1]) / 2.0


# ----------------------------------------------------------------------
def _adam_subgradient(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser that uses the exact sub‑gradient of
    max(conv(a,a)).  The weight vector is an indicator of the maximal
    convolution entries (normalised to sum‑1).  After each Adam update
    the iterate is projected onto the probability simplex and symmetrised.
    """
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule – (steps, learning‑rate)
    schedule = [
        (2000, 0.30),
        (2000, 0.20),
        (3000, 0.15),
        (3000, 0.10),
        (4000, 0.07),
        (4000, 0.05),
    ]

    for steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1

            # current self‑convolution and its maximum
            conv = _fft_self_convolve(a)
            maxc = conv.max()

            # indicator of the maximal entries (with a tiny tolerance)
            mask = conv >= maxc - 1e-12
            w = mask.astype(float)
            if w.sum() == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w.sum()

            # sub‑gradient = 2 * (w * a) convolution (as in the smooth case)
            grad_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * grad_rev[n - 1 : n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection + symmetrisation
            a = _project_onto_simplex(a, 1.0)
            a = _symmetrize(a)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] Adam step {t} new best {best_score:.6f}"
                    )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Tiny simulated‑annealing walk – move random mass from a source index
    to a destination index and keep the change only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # ensure j != i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random fraction of the source mass (up to 30 %)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths (longer vectors give a finer discretisation)
    lengths = [8000, 12000, 20000, 30000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families + the Fourier‑root seed for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating candidates for length {n}")

        cand_list = _initial_candidates_extended(n, rng)
        cand_list.append(_distribution_root_uniform(n))

        # a few random Dirichlet vectors (different concentration parameters)
        for alpha in (0.3, 0.7, 1.5):
            rand_vec = rng.gamma(alpha, size=n)
            if rand_vec.sum() == 0:
                continue
            rand_vec /= rand_vec.sum()
            cand_list.append(rand_vec)

        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            if cand.sum() == 0:
                continue
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] new global best length={n} score={sc:.6f}"
                )
            top_candidates.append((sc, cand.copy()))

        # keep a small set of best seeds (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:8]

    # ------------------------------------------------------------------
    # 3. If a previous construction exists, add it (optional)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0:
                prev /= prev.sum()
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous best score={sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – never needed but kept for safety
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(2000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline
    a = overall_best_seq.copy()

    # 5a – Adam on the exact max‑convolution (few top seeds)
    remaining_time = deadline - time.time()
    if remaining_time > 30:
        # select up to three most promising seeds (including the current best)
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]]
        # split the remaining budget evenly
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            best_seq, best_score = _adam_subgradient(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(
                    f"[{time.time():.1f}] after Adam seed best {overall_best_score:.6f}"
                )

    # 5b – final greedy pairwise swaps / tiny simulated‑annealing fine‑tuning
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        overall_best_seq = best_seq.copy()
        overall_best_score = best_score

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
