def _objective(a):
    n = a.shape[0]
    maxb = _fft_self_convolve(a).max()
    return 2.0 * n * maxb

def _distribution_beta(p,q,n):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute log w = (p-1)*log(x)+(q-1)*log(1-x)
    # clip etc.
    logw = (p-1)*np.log(x) + (q-1)*np.log(1 - x)
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0/n)
    w /= w_sum
    return w

def _distribution_powerlaw_sum(gamma,n):
    i = np.arange(n, dtype=np.float64) + 0.5
    left = i**(-gamma)
    right = (n - i)**(-gamma)
    w = left + right
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full(n, 1.0/n)
    else:
        w /= w_sum
    return w

def _distribution_exponential(lam,n):
    i = np.arange(n, dtype=np.float64)
    centre = (n-1)/2.0
    w = np.exp(-lam * np.abs(i - centre))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full(n, 1.0/n)
    else:
        w /= w_sum
    return w

def _subgradient_optimize(a, deadline, best_seq, best_score):
    n = a.shape[0]
    t = 0
    lr = 0.1
    while time.time() < deadline:
        # compute conv and max index
        conv = _fft_self_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_score:
            best_score = cur_obj
            best_seq = a.copy()
        # gradient steps

                best_score = cur_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad iter={t} new best {best_score:.6f}")
    return best_seq, best_score

def _pairwise_swap_local(a, deadline, best_seq, best_score):
    # as earlier but earlier uses _objective (already correct)
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.3
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Heuristic optimizer for the step‑function problem.
    Generates many analytic candidates, refines the best few with a
    simple sub‑gradient descent on the exact max‑convolution and finishes
    with a tiny stochastic local search.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    lengths = [4000, 8000, 12000, 20000]  # a few moderate sizes
    overall_best_seq = None
    overall_best_score = float('inf')
    top_candidates = []   # list of (score, vector)

    # 1. generate analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")
        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))
        # Keep only the most promising few
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # 2. local sub‑gradient refinement of the top seeds
    # allocate about half the remaining budget to refinement
    remaining = deadline - time.time()
    if remaining > 30 and len(top_candidates) > 0:
        # give each seed equal slice of the remaining time
        per_seed = remaining / len(top_candidates)
        new_top = []
        for init_score, init_vec in top_candidates:
            sub_deadline = time.time() + per_seed
            best_vec, best_sc = _subgradient_optimize(init_vec.copy(),
                                                      sub_deadline,
                                                      best_seq=overall_best_seq,
                                                      best_score=overall_best_score)
            overall_best_seq = best_vec.copy()
            overall_best_score = best_sc
            new_top.append((best_sc, best_vec.copy()))
        top_candidates = new_top

    # 3. final tiny stochastic refinement
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(overall_best_seq.copy(),
                                                                   deadline,
                                                                   best_seq=overall_best_seq,
                                                                   best_score=overall_best_score)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()

"""
A fast heuristic optimizer for the “minimise max self‑convolution” problem.

The code generates a large collection of analytic candidates (uniform,
Beta‑laws, power‑law edges, exponentials, random sparse vectors) for a few
different lengths, keeps the best few, and then refines each of them with a
simple sub‑gradient descent that directly minimizes the exact objective
2·n·max(conv(a,a)) (the sum of the vector is kept equal to 1 by projecting
onto the simplex).  Finally a tiny stochastic “pairwise‑swap’’ walk is run.
The whole search obeys the global time budget (default 1000 s) and returns
the best vector found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    maxb = _fft_self_convolve(a).max()
    return 2.0 * n * maxb


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta(p: float, q: float, n: int) -> np.ndarray:
    """Beta(p,q) on the discrete grid (p,q>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density; safe because x∈(0,1)
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0.0 or np.isnan(w_sum):
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= w_sum
    return w


def _distribution_powerlaw_sum(gamma: float, n: int) -> np.ndarray:
    """a_i ∝ (i+½)^−γ + (n‑i‑½)^−γ .  (γ∈(0,½) gives a U‑shape)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    left = i ** (-gamma)
    right = (n - i) ** (-gamma)
    w = left + right
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= w_sum
    return w


def _distribution_exponential(lam: float, n: int) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-lam * np.abs(idx - centre))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= w_sum
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.05) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    w = np.zeros(n, dtype=np.float64)
    w[idx] = rng.random(k)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[0] = 1.0
        w_sum = 1.0
    w /= w_sum
    return w


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad set of start vectors for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta(0.5, 0.5, n))          # arcsine

    # symmetric beta – many shape parameters (explores the U‑shape)
    for p in np.geomspace(0.01, 2.0, num=20):
        cand.append(_distribution_beta(p, p, n))

    # asymmetric beta – pushes mass to one side
    for p in (0.01, 0.02, 0.05, 0.1):
        cand.append(_distribution_beta(p, 1.0, n))
        cand.append(_distribution_beta(1.0, p, n))

    # power‑law at the edges (sum of two one‑sided power laws)
    for gamma in np.linspace(0.01, 0.49, num=20):
        cand.append(_distribution_powerlaw_sum(gamma, n))

    # exponentials – centre‑peak vs edge‑peak
    for lam in np.geomspace(0.0005, 0.02, num=12):
        cand.append(_distribution_exponential(lam, n))

    # a few random sparse starts
    cand.append(_distribution_random_sparse(n, rng, density=0.10))
    cand.append(_distribution_random_sparse(n, rng, density=0.05))

    return cand


# ----------------------------------------------------------------------
# Sub‑gradient optimisation on the exact max‑convolution ----------------
def _subgradient_optimize(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent that minimises 2·n·max(conv(a,a)).
    Each iteration uses the actual worst shift, takes a step opposite to its
    gradient and projects back onto the probability simplex.
    A tiny Gaussian perturbation is added every 1000 steps to help escape
    shallow local minima.
    """
    n = a.shape[0]
    lr = 0.5                     # initial learning rate
    it = 0

    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        cur = 2.0 * n * conv.max()
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        # index of the maximal convolution coefficient
        k = int(conv.argmax())

        # gradient of conv[k] w.r.t. a (exact, not smoothed)
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)

        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # diagonal terms (i==j) appear only once
        diag = mask & (i == j)
        grad[diag] = a[i[diag]]

        # simple step and projection
        a = a - lr * grad
        a = _project_onto_simplex(a, 1.0)

        # learning‑rate schedule
        it += 1
        if it % 500 == 0:
            lr *= 0.7

        # occasional random kick to explore new basins
        if it % 1000 == 0:
            a += rng.normal(scale=0.001, size=n)
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, 1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Tiny stochastic refinement (pairwise mass swaps) --------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep the move only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30       # up to 30 % of source mass
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} Δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Heuristic optimiser for the step‑function problem.
    Returns a list of non‑negative heights that (empirically) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5        # safety margin

    # a few moderate lengths – longer vectors give a finer discretisation
    lengths = [4000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. scan analytic families
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()               # enforce sum‑1
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the very best few seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2. sub‑gradient refinement of the best few seeds
    if top_candidates and time.time() < deadline:
        # pick up to three most promising seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]]
        per_seed = (deadline - time.time()) / max(1, len(seeds))
        for seed_vec in seeds:
            sub_deadline = time.time() + per_seed
            overall_best_seq, overall_best_score = _subgradient_optimize(
                seed_vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                rng=rng,
            )

    # ------------------------------------------------------------------
    # 3. final tiny stochastic walk (pairwise swaps)
    if time.time() < deadline and overall_best_seq is not None:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
