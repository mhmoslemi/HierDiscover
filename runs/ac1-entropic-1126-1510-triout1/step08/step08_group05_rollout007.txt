"""
Propose a candidate step‐function that (approximately) minimises
``evaluate_sequence``.  The strategy is:

1.  Broad sweep of analytic families (uniform, arcsine, beta, sine‑power,
    cosine‑window, power‑law, exponential, Gaussian, triangular) together
    with random Dirichlet draws for several lengths.  The best few are kept.

2.  For each keep‑alive candidate we run a **mirror (exponential) descent**
    on the simplex.  The smooth max of the self‑convolution is used as a
    surrogate for the hard ``max``; the “softness’’ parameter ``α`` is
    increased gradually.  The update is
        a ← a·exp(−η·∇)   (component‑wise)  and then renormalised.
    A tiny ``eps`` guarantees positivity.

3.  A short diffusion (3‑point averaging) pass is applied – this often
    reduces the peak of the self‑convolution without destroying the overall
    shape.

4.  A **simulated‑annealing** hill‑climber works directly on the exact
    self‑convolution.  A mass transfer ``δ`` from a source index to a target
    index is performed with an O(n) incremental convolution update.  The
    temperature schedule is relatively aggressive at the start and cools
    exponentially.

5.  The best sequence seen (as plain ``list[float]``) is returned.
   The whole search respects the supplied time budget (default 1000 s).

All helper functions are defined at module level – no closures or lambdas.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection onto the simplex {x≥0, sum x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families – each returns a **probability vector** (sum = 1).
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law a_i ∝ 1/(i+1)^p, normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2, wrapped onto the finite support."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n, width_frac=1.0):
    """Triangular shape – width_frac∈(0,1] controls how sharp the tip is."""
    half = int(np.round(width_frac * n / 2.0))
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 else left
    vals = np.concatenate([left, right])
    if vals.size < n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j,
    updating both `a` and its full self‑convolution array `conv` in O(n)
    time.
    """
    if delta <= 0 or i == j:
        return
    # store a copy of the old a for the linear term
    a_old = a.copy()
    n = a.shape[0]
    # linear contributions (subtract from source, add to target)
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

def _smooth_max_weights(b, alpha):
    """Soft‑max weights for the smooth max with parameter α."""
    m = b.max()
    w = np.exp(alpha * (b - m))
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _softmax_gradient(a, alpha):
    """Gradient of the smooth max (softmax) of a*a."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    w = _smooth_max_weights(b, alpha)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (finite vector) that (approximately)
    minimises the black‑box ``evaluate_sequence`` function.
    The routine respects the time budget and returns the best sequence (as a
    plain list of floats) found.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Broad scan of families (several lengths, many parameter choices)
    lengths = [2000, 3000, 4000, 5000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---------- deterministic families ----------
        candidates = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
        ]

        # ---------- parametrised families ----------
        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))

        for c in (0.3, 0.4, 0.5, 0.6, 0.8):
            candidates.append(_distribution_cosine_window(n, c))

        for p in (0.3, 0.5, 1.0, 1.5, 2.0):
            candidates.append(_distribution_power_law(n, p))

        for lam in (0.001, 0.003, 0.01, 0.03):
            candidates.append(_distribution_exponential(n, lam))

        # Gaussian family – sigma proportional to n
        for sig_frac in (0.15, 0.20, 0.25, 0.30):
            sigma = sig_frac * n
            candidates.append(_distribution_gaussian(n, sigma))

        # Triangular with different widths
        for wf in (0.3, 0.5, 0.7, 1.0):
            candidates.append(_distribution_triangular(n, width_frac=wf))

        # ---------- random Dirichlet / Gamma draws ----------
        for _ in range(8):
            # random concentration between 0.1 and 5.0
            conc = 10 ** np.random.uniform(-1, 0.7)
            x = np.random.gamma(conc, 1.0, size=n)
            x /= x.sum()
            candidates.append(x)

        # ---------- sparse seeds ----------
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # ---------- evaluate ----------
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()                     # enforce sum = 1
            scr = _max_convolution_score(vec)
            if scr < best_score:
                best_score = scr
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} new init best {scr:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Seed from the previous global solution (if it exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        n = 3000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Mirror (exponential) descent on the simplex.
    #    We use a sequence of soft‑max sharpness values.
    a = best_seq.copy()
    n = a.shape[0]

    # Soft‑max schedule – start very smooth, finish extremely sharp.
    alpha_schedule = [
        (0.5, 2000),
        (2.0, 2000),
        (10.0, 3000),
        (50.0, 3000),
        (200.0, 3000),
        (800.0, 2000),
        (3000.0, 1500)
    ]

    # Base learning rate for exponentiated gradient.
    eta = 0.4
    best_seq_adam = a.copy()
    best_score_adam = _max_convolution_score(a)

    for alpha, steps in alpha_schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            # compute smooth‑max gradient
            grad = _softmax_gradient(a, alpha)

            # exponential (mirror) update
            a = a * np.exp(-eta * grad)
            a = np.maximum(a, eps)          # avoid zeros
            a /= a.sum()

            # periodically check improvement
            if random.random() < 0.004:   # roughly every 250 steps on average
                cur = _max_convolution_score(a)
                if cur < best_score_adam - 1e-12:
                    best_score_adam = cur
                    best_seq_adam = a.copy()
                    print(f"[{time.time():.1f}] Mirror α={alpha:.1f} new best {cur:.6f}")

    a = best_seq_adam.copy()
    best_seq = a.copy()
    best_score = best_score_adam

    # ------------------------------------------------------------------
    # 4️⃣  Light diffusion – often reduces the peak further.
    if time.time() < deadline:
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, eps)
            a /= a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Diffusion new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing hill‑climber on the exact self‑convolution.
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_seq = a.copy()
    it = 0
    it_no_improve = 0
    # start temperature at ~10 % of the current score (more aggressive than before)
    T = cur_score * 0.10
    decay = 0.9992
    max_delta_factor = 0.04           # up to 4 % of source mass in early phase

    while time.time() < deadline:
        it += 1
        # pick source index proportional to its mass (more influential)
        src = np.random.choice(n, p=a)
        tgt = np.random.randint(n)
        if tgt == src:
            continue

        # allowed mass transfer – larger initially, shrinks with temperature
        max_delta = min(a[src] * max_delta_factor, 0.07)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        # keep snapshots for possible rejection
        a_snap = a.copy()
        conv_snap = conv.copy()

        _apply_transfer(a, conv, src, tgt, delta)

        new_score = 2.0 * n * conv.max()
        improve = new_score < cur_score

        if improve:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            accept = np.random.rand() < prob

        if accept:
            cur_score = new_score
            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        # cooling
        T *= decay

        # occasional large‑scale jiggle if we seem stuck
        if it % 4000 == 0 and it_no_improve > 1500:
            jitter = np.random.randn(n) * 2e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

    # ------------------------------------------------------------------
    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    # Return the best sequence found (already normalised to sum = 1)
    return best_seq.tolist()
