def build_constraint_matrix(f, active_idxs):
    n = len(f)
    A = np.zeros((len(active_idxs), n), dtype=np.float64)
    for row, k in enumerate(active_idxs):
        # shift f by k: indices i where 0 <= k-i < n => i in [max(0,k-n+1), min(k,n-1)]
        start_i = max(0, k - n + 1)
        end_i = min(k, n - 1)
        # for i in this range, coeff = f[k-i]
        i_range = np.arange(start_i, end_i+1)
        coeffs = f[k - i_range]
        A[row, i_range] = coeffs
    return A

def propose_candidate(seed=42, budget_s=1000):
    import time, random, math, numpy as np, cvxpy as cp
    start = time.time()
    deadline = start + budget_s - 0.5

    np.random.seed(seed)
    random.seed(seed)

    best_seq = None
    best_val = float('inf')

    def evaluate_local(vec):
        # compute our metric approximation: sum normalized to 1.
        n = len(vec)
        conv = _fft_self_convolve(vec)
        return 2.0 * n * conv.max()

    # Step A: candidate generation
    lengths = [2000, 3000, 4000, 5000, 8000]
    # param grids
    for n in lengths:
        if time.time() >= deadline: break

        # uniform
        cand = [_distribution_uniform(n)]
        # arcsine
        cand.append(_distribution_arcsine(n))
        # triangular
        cand.append(_distribution_triangular(n))
        # Exponential lam
        for lam in np.logspace(-4, -1, 6): # from 1e-4 to 1e-1
            cand.append(_distribution_exponential(n, lam))
        # Beta symmetric p
        for p in np.logspace(-1, 1, 10):  # 0.1 to 10
            cand.append(_distribution_beta_sym(n, p))
        # Power law
        for p in np.logspace(-1, 0.5, 8): # 0.1 to 3.16
            cand.append(_distribution_power_law(n, p))
        # sine power
        for pw in np.linspace(0.2, 8.0, 12):
            cand.append(_distribution_sine_power(n, pw))
        # cosine window
        for coeff in np.linspace(0.2, 0.8, 7):
            cand.append(_distribution_cosine_window(n, coeff))

        # random Dirichlet draws
        for _ in range(5):
            vec = np.random.gamma(0.5, 1.0, size=n)
            vec /= vec.sum()
            cand.append(vec)

        # iterate candidates
        for vec in cand:
            if time.time() >= deadline: break
            # ensure positivity, sum=1
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()
            val = evaluate_local(vec)
            if val < best_val:
                best_val = val
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] init len={n} new best {val:.6f}")

    # incorporate previous height_sequence_1
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64) # global variable
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev = prev / prev.sum()
            val = evaluate_local(prev)
            if val < best_val:
                best_val = val
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {val:.6f}")
    except Exception:
        pass

    # If none, fallback
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_val = evaluate_local(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_val:.6f}")

    # Phase C: Adam descent
    a = best_seq.copy()
    n = a.shape[0]
    # schedule of alphas and step counts (adapt)
    schedule = [(0.5, 2000), (2.0, 2000), (10.0, 3000), (50.0, 4000), (200.0, 4000), (1000.0, 3000)]
    eps = 1e-12
    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.1
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_adam_seq = a.copy()
    best_adam_val = evaluate_local(a)

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1

            # compute softmax weights
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w = w / w_sum

            # gradient of smooth max: conv_rev = a (reversed) convolve w
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # projection to simplex
            a = _project_onto_simplex(np.maximum(a, eps), 1.0)

            if t % 500 == 0:
                cur = evaluate_local(a)
                if cur < best_adam_val:
                    best_adam_val = cur
                    best_adam_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

    # after Adam, use best found
    a = best_adam_seq.copy()
    best_seq = a.copy()
    best_val = best_adam_val

    # We'll perform at most 4 refinement cycles or until time runs out.
    def active_lp_refine(f, t_factor=0.3):
        n = len(f)
        conv_b = _fft_self_convolve(f)
        M = conv_b.max()
        # active indices (within tolerance)
        tol = 1e-9 * (abs(M) + 1.0)
        act = np.where(conv_b >= M - tol)[0]
        if act.size == 0:
            return f, False
        # Build constraint matrix A shape (len(act), n)
        A = np.zeros((act.size, n), dtype=np.float64)
        for idx, k in enumerate(act):
            # i from max(0, k-n+1) to min(k, n-1)
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1)
            A[idx, i_range] = f[k - i_range]
        # Solve LP
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
        except Exception as e:
            # fallback
            return f, False
        if g.value is None:
            return f, False
        g_opt = np.maximum(g.value, 0.0)
        sum_g = g_opt.sum()
        if sum_g <= 0:
            return f, False
        g_norm = g_opt / sum_g
        # convex combination
        new_f = (1 - t_factor) * f + t_factor * g_norm
        new_f = _project_onto_simplex(new_f, 1.0)
        return new_f, True

    lp_iter = 0
    while time.time() < deadline - 2.0 and lp_iter < 5:
        new_a, changed = active_lp_refine(a, t_factor=0.4)
        if not changed:
            break
        a = new_a
        cur_val = evaluate_local(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LP refine iter={lp_iter} new best {cur_val:.6f}")
        lp_iter += 1

    # Phase E: Subgradient descent with momentum (same as earlier but more steps)
    velocity = np.zeros_like(a)
    beta = 0.85
    step0 = 0.2
    it = 0
    while time.time() < deadline - 2.0 and it < 500000:
        it += 1
        convb = _fft_self_convolve(a)
        cur_max = convb.max()
        # identify active indices within tolerance
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(convb >= cur_max - eps_rel)

        w = np.zeros_like(convb)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        velocity = beta * velocity + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(np.maximum(a, eps), 1.0)

        if it % 2000 == 0:
            cur_val = evaluate_local(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = a.copy()

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)

    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer `delta` mass from index i to index j.
    `a` and its full self‑convolution `conv` are updated in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return

    a_old = a.copy()

    # linear part: -2*delta * (a_old * e_i)  + 2*delta * (a_old * e_j)
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old

    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta

    # cross term: -2*delta^2 at index i + j
    cross = i + j
    if 0 <= cross < conv.size:
        conv[cross] -= 2.0 * delta * delta

    a[i] -= delta
    a[j] += delta

        g_norm = g_opt / total

        # mix the old vector with the new one
        new_f = (1 - mix_factor) * f + mix_factor * g_norm
        new_f = _project_onto_simplex(new_f, 1.0)
        return new_f, True

    lp_iter = 0
    while time.time() < deadline - 2.0 and lp_iter < 5:
        a, changed = _lp_refine(a, mix_factor=0.4)
        if not changed:
            break
        cur = _local_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LP refine #{lp_iter} new best {cur:.6f}")
        lp_iter += 1

    # ------------------------------------------------------------------
    # 5. Exact max‑sub‑gradient descent with momentum
    velocity = np.zeros_like(a)
    beta = 0.85
    step0 = 0.2
    it = 0
    while time.time() < deadline - 2.0 and it < 500_000:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        velocity = beta * velocity + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(np.maximum(a, eps), 1.0)

        if it % 2000 == 0:
            cur = _local_score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Subgrad it={it} new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 6. Diffusion smoothing (light blur)
    if time.time() < deadline - 1.0:
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        a = best_seq.copy()
        for _ in range(12):
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, eps)
            a /= a.sum()
        cur = _local_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Diffusion new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 7. Simulated‑annealing hill‑climber with gradient‑guided source
    if time.time() < deadline - 1.0:
        a = best_seq.copy()
        conv = _fft_self_convolve(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        best_sa_score = cur_score
        best_sa_seq = a.copy()

        T = cur_score * 0.05   # initial temperature ≈5 % of current score
        decay = 0.9997
        max_delta_factor = 0.02
        it_sa = 0

        while time.time() < deadline:
            it_sa += 1

            # sub‑gradient contribution (used as source‑selection weight)
            active = np.flatnonzero(conv >= M - 1e-9 * (abs(M) + 1.0))
            w = np.zeros_like(conv)
            w[active] = 1.0
            contrib = _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
            prob_src = contrib / contrib.sum()

            source = np.random.choice(n, p=prob_src)
            target = np.random.randint(n)
            if target == source:
                continue

            max_delta = min(a[source] * max_delta_factor, 0.05)
            if max_delta <= eps:
                continue
            delta = max_delta * np.random.rand()
            if delta <= eps:
                continue

            a_snapshot = a.copy()
            conv_snapshot = conv.copy()

            _apply_transfer(a, conv, source, target, delta)

            new_M = conv.max()
            new_score = 2.0 * n * new_M

            accept = False
            if new_score < cur_score:
                accept = True
            else:
                prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
                if np.random.rand() < prob:
                    accept = True

            if accept:
                cur_score = new_score
                M = new_M
                it_no_improve = 0
                if new_score < best_sa_score - 1e-12:
                    best_sa_score = new_score
                    best_sa_seq = a.copy()
                    print(f"[{time.time():.1f}] SA iter={it_sa} new best {new_score:.6f}")
            else:
                a = a_snapshot
                conv = conv_snapshot
                it_no_improve = it_no_improve + 1 if 'it_no_improve' in locals() else 1

            T *= decay

            # occasional restart if the process stalls
            if it_sa % 5000 == 0 and it_no_improve > 3000:
                T = cur_score * 0.05
                a = best_sa_seq.copy()
                conv = _fft_self_convolve(a)
                M = conv.max()
                cur_score = 2.0 * n * M
                it_no_improve = 0
                print(f"[{time.time():.1f}] SA restart at iter {it_sa}")

    # ------------------------------------------------------------------
    # 8. Choose the best of the two final candidates (if SA was run)
    final_seq = best_seq
    if 'best_sa_seq' in locals():
        try:
            val_seq = evaluate_sequence(final_seq.tolist())
            val_sa = evaluate_sequence(best_sa_seq.tolist())
            if val_sa < val_seq:
                final_seq = best_sa_seq
                print(f"[{time.time():.1f}] Final choice: SA sequence with score {val_sa:.6f}")
            else:
                print(f"[{time.time():.1f}] Final choice: Adam/LP sequence with score {val_seq:.6f}")
        except Exception:
            # fallback to Adam/LP sequence if evaluation fails
            pass

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = final_seq / final_seq.sum()
    return final_seq.tolist()
