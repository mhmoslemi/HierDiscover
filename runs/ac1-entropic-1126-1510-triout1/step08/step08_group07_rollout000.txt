"""
Multi‑start Entropic‑Gradient optimiser for the min‑max self‑convolution
problem.

The search proceeds in four stages:

1. **Rich initial pool** – a collection of analytic families (uniform,
   arcsine, beta, exponential, harmonic, sparse random, etc.) together
   with a handful of Dirichlet draws (different concentration parameters)
   is created.  The best among them (according to the exact objective
   2·n·max(conv(a,a))) is selected as a seed.

2. **Entropic (exponentiated) gradient descent** – the non‑convex “max‑convolution”
   objective is smoothed by a soft‑max with temperature τ.
   The gradient of the smooth objective w.r.t. the probability vector is
   computed analytically via a single FFT‑based convolution.
   An exponentiated‑gradient update (mirror descent on the simplex) is
   applied and the vector is re‑normalised.  τ and the learning rate
   η are annealed during the run.

3. **Greedy active‑set refinement** – the vector obtained from stage 2
   is handed to the proven greedy local‑search routine (pairwise mass
   transfers driven by the exact active‑set sub‑gradient).  This sharply
   reduces the true max‑convolution value.

4. **Simulated‑annealing fine‑tuning** – a short Metropolis walk with
   exponential cooling polishes the solution.

All stages respect the overall time budget (default 1000 s).  The final
vector is projected onto the probability simplex and returned as a plain
Python list of non‑negative floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta distribution with shape parameter p (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_harmonic(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1) – a simple heavy‑tail shape."""
    w = 1.0 / (np.arange(1, n + 1, dtype=np.float64))
    w /= w.sum()
    return w

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k (a*a)[k] = 2 * (w * a) (convolution).
    Both a (length n) and w (length 2n‑1) are real.
    Returns a gradient vector of length n.
    """
    n = a.shape[0]
    size = 1 << ((len(w) + n - 1) - 1).bit_length()
    Wa = np.fft.rfft(w, n=size)
    Aa = np.fft.rfft(a, n=size)
    conv_wa = np.fft.irfft(Wa * Aa, n=size)
    # slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_wa[n - 1:n - 1 + n]
    return grad

def _exponential_gradient_descent(start_a: np.ndarray,
                                 deadline: float,
                                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Entropic (exponentiated) gradient descent on the soft‑max smoothed
    max‑convolution objective.
    Returns the best vector seen and its exact objective.
    """
    a = start_a.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    tau = 5.0               # smoothing temperature
    eta = 0.5               # base step size for the exponentiated update
    it = 0
    last_report = time.time()

    while time.time() < deadline:
        it += 1

        # ----- smooth max -----
        conv = _fft_self_convolve(a)                # length 2n‑1
        max_c = conv.max()
        # numerical stability: shift by max before exponentiation
        shifted = conv - max_c
        w = np.exp(tau * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient -----
        grad = _gradient_from_weights(a, w)

        # ----- exponentiated (mirror) update -----
        # avoid overflow/underflow by scaling the exponent
        scaled = -eta * grad
        max_abs = np.max(np.abs(scaled))
        if max_abs > 50.0:                     # empirical guard
            scaled = scaled * (50.0 / max_abs)

        a = a * np.exp(scaled)
        a = _project_onto_simplex(a, 1.0)

        # ----- evaluate exact objective -----
        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best = a.copy()
            best_score = cur_score
            print(f"[{time.time():.1f}] EG it={it} improved to {best_score:.6f}")

        # ----- schedule adjustments -----
        if it % 200 == 0:
            tau = min(200.0, tau * 1.2)          # sharpen the approximation
            eta = max(0.02, eta * 0.9)           # slowly decay step size

            # occasional random jitter to escape plateaus
            if rng.random() < 0.3:
                i = rng.integers(n)
                if a[i] > 1e-12:
                    j = rng.integers(n - 1)
                    if j >= i:
                        j += 1
                    eps = min(a[i] * 0.05, 0.001)
                    a[i] -= eps
                    a[j] += eps
                    a = _project_onto_simplex(a, 1.0)

        # thin reporting (prevent spamming)
        if time.time() - last_report > 30.0:
            print(f"[{time.time():.1f}] EG it={it} best={best_score:.6f} tau={tau:.2f} eta={eta:.3f}")
            last_report = time.time()

    return best, best_score

def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Aggressive pairwise mass transfer based on the exact active‑set sub‑gradient."""
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient: g_i = 2 * Σ_{k∈active} a_{k-i}
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i_range = np.arange(i_low, i_high + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # try moving most of the source mass (but limit to avoid instability)
        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # reduced transfer if full move failed
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # occasional jitter
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps_jit = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps_jit
                a_tmp[j] += eps_jit
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} improved to {best_score:.6f}")

    return best_vec, best_score

def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    T = 0.03

    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9997

    return best_a, best_score
