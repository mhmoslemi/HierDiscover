"""
Smooth‑max optimisation for the step‑function score.

The routine works in three stages:

1. **Broad sampling** – generate a modest set of analytic families
   (arcsine, triangular, exponential, Gaussian, Cauchy, symmetric / 
   asymmetric Beta, Dirichlet draws).  All vectors are ℓ₁‑normalised and
   scored with the cheap surrogate 2·n·max(conv(a,a)).

2. **Smooth‑max Adam descent** – for the top few seeds we minimise a
   smooth approximation of the max‑convolution,
        f(a)= (1/β)·log Σ_k exp(β·(a*a)_k) .
   Its gradient is 2·conv_rev(a,softmax(β·b)) and is used inside an
   Adam optimiser.  The temperature β is annealed from a modest value
   (≈8) up to a few hundred, gradually turning the smooth objective
   into the true max.

3. **Frank‑Wolfe mass‑transfer polishing** – a cheap pairwise transfer
   loop (identical to the previous code) cleans up any remaining
   spikes.

The algorithm respects the supplied time budget (≈ 1000 s) and returns
the best normalised sequence found.  Scaling does not affect the
evaluation, so the returned vector is left with sum = 1.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for evaluate_sequence when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (Σa = 1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
# analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# smooth‑max Adam optimiser --------------------------------------------
def _smooth_optimize(a0: np.ndarray, deadline: float,
                    max_iters: int = 200_000,
                    beta_start: float = 8.0,
                    beta_max: float = 200.0,
                    beta_inc: float = 1.07,
                    beta_update: int = 2000) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of the max‑convolution using Adam.
    The temperature β is slowly increased so the objective converges to
    the true max.  Returns the best vector seen and its surrogate score.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # best ever seen
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    beta = beta_start
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        # anneal temperature
        if t % beta_update == 0 and beta < beta_max:
            beta = min(beta_max, beta * beta_inc)

        # self‑convolution
        b = _fft_self_convolve(a)

        # soft‑max weights w_k = exp(β b_k) / Σ exp(β b)
        xb = beta * b
        max_xb = xb.max()
        exp_xb = np.exp(xb - max_xb)
        w = exp_xb / exp_xb.sum()

        # gradient = 2 * conv_rev(a, w)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad ** 2)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        step = 0.3 / (np.sqrt(v_hat) + 1e-8)

        a = a - step * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # occasional true‑score check
        if t % 500 == 0:
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-14:
                best_score, best_a = cur_score, a.copy()
                print(f"[{time.time():.1f}] smooth‑Adam it={t} β={beta:.2f} score={best_score:.6f}")

    return best_a, best_score

# original Adam optimiser (used as a cheap fallback)
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

            candidates.append((a, _max_convolution_score(a)))

        # power‑transformed arcsine (adds extra curvature)
        base = _arcsine_distribution(n)
        for pwr in (0.4, 0.7, 1.3, 1.8):
            a = np.power(base, pwr)
            a /= a.sum()
            candidates.append((a, _max_convolution_score(a)))

    # inject the previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # keep the most promising seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(8, len(candidates))
    top_seeds = candidates[:top_k]

    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]

    print(f"Generated {len(candidates)} candidates, top {top_k} kept.")
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # intensive refinement of the top seeds
    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline - 5.0:
            break
        print(f"--- Refinement of seed {idx + 1}/{len(top_seeds)} (n={seed_seq.size}) ---")

        # smooth‑max Adam descent (the main workhorse)
        seq, sc = _smooth_optimize(seed_seq, deadline, max_iters=250_000)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"   smooth‑Adam improved to {best_score:.6f}")

        # cheap fallback Adam (sub‑grad) – sometimes finds different valleys
        seq, sc = _adam_descent(seed_seq, deadline, max_iters=80_000, step0=0.2)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"   sub‑grad Adam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # final Frank‑Wolfe style mass‑transfer polishing
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 60_000:
        it_fw += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-12:
            break

        # try a full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        if score_full < cur_score - 1e-12:
            a = a_full
            cur_score = score_full
        else:
            # try half‑transfer as a fallback
            half = 0.5 * max_transfer
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-12:
                a = a_half
                cur_score = score_half

        if cur_score < best_score - 1e-14:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] FW iter={it_fw} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if it_fw % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # the vector is already ℓ₁‑normalised; scaling does not affect the real score
    return best_seq.tolist()
