# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using log‑sum‑exp:
        f_eps(a) = eps * log Σ_i exp(conv_i/eps)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad

def _smooth_adam(a0: np.ndarray, deadline: float,
                 eps_start: float = 1e-2,
                 eps_end: float = 1e-7,
                 max_iters: int = 800_000,
                 step0: float = 0.25,
                 beta1: float = 0.9,
                 beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smoothed max.
    The smoothing parameter ε is decreased geometrically from eps_start
    to eps_end during the optimisation.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    # geometric schedule for ε
    if max_iters > 0:
        log_eps_factor = (np.log(eps_end) - np.log(eps_start)) / max_iters
    else:
        log_eps_factor = 0.0

    while time.time() < deadline and t < max_iters:
        t += 1
        eps = np.exp(np.log(eps_start) + log_eps_factor * t)

        grad = _smooth_grad(a, eps)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _adam(a0: np.ndarray, deadline: float,
          max_iters: int = 600_000,
          step0: float = 0.25,
          beta1: float = 0.9,
          beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Standard Adam on the cheap max‑convolution objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with diminishing step size."""

    best_a = a.copy()
    best_score = _max_convolution_score(a)

        a = a - eta * grad

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _power_law_distribution(n: int, p: float = 0.5) -> np.ndarray:
    """Weight ∝ ((i+½)(n-i‑½))⁻ᵖ, p≈0.5 recovers arcsine."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / ((i + 0.5) * (n - i - 0.5)) ** p
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0            # map to [-1,1]
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a probability vector (sum = 1).  The routine respects the
    supplied time budget and prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # Choose a length.  A few thousand entries gives a good trade‑off.
    # If a historic sequence exists, use its length as a reference.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # try a larger size if we still have enough time left after the first pass
    length_candidates = [int(base_n * f) for f in (0.9, 1.0, 1.1, 1.2, 1.4, 1.6)]
    length_candidates = [n for n in length_candidates if 800 <= n <= 12000]
    n = max(length_candidates)   # take the largest admissible size

    print(f"[{time.time():.1f}] Selected length n = {n}")

    # --------------------------------------------------------------
    # 1. Broad sampling – generate a pool of candidate distributions
    candidates = []   # (array, cheap_score)

    # parameter grids
    beta_ps = [0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.75, 1.0, 1.5, 2.0]
    power_ps = [0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]
    exp_lams = [0.001, 0.002, 0.003, 0.005]
    sigma_fracs = [0.20, 0.25, 0.30, 0.35]
    scale_fracs = [0.15, 0.20, 0.25, 0.30]
    dirichlet_alphas = [0.3, 0.6, 1.0]

    # deterministic families
    fams = [
        (_arcsine_distribution, "arcsine", {}),
        (_triangular_distribution, "triangular", {}),
        (_uniform_distribution, "uniform", {}),
    ]

    # generate from deterministic families
    for gen, name, kwargs in fams:
        a = gen(n, **kwargs)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] {name:10s} cheap score={score:.6f}")

    # power‑law (generalised arcsine)
    for p in power_ps:
        a = _power_law_distribution(n, p=p)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] power(p={p:.2f}) cheap score={score:.6f}")

    # symmetric beta
    for p in beta_ps:
        a = _beta_symmetric_distribution(n, p)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] beta(p={p:.2f}) cheap score={score:.6f}")

    # exponential
    for lam in exp_lams:
        a = _exponential_distribution(n, lam=lam)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] exponential(lam={lam:.3f}) cheap score={score:.6f}")

    # Gaussian
    for sf in sigma_fracs:
        a = _gaussian_distribution(n, sigma_frac=sf)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] gaussian(sig={sf:.2f}) cheap score={score:.6f}")

    # Cauchy
    for sf in scale_fracs:
        a = _cauchy_distribution(n, scale_frac=sf)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] cauchy(scale={sf:.2f}) cheap score={score:.6f}")

    # semicircle
    a = _semicircle_distribution(n)
    score = _max_convolution_score(a)
    candidates.append((a, score))
    print(f"[{time.time():.1f}] semicircle cheap score={score:.6f}")

    # Dirichlet draws
    for alpha in dirichlet_alphas:
        for _ in range(3):
            a = _random_dirichlet(n, alpha=alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] Dirichlet(α={alpha:.2f}) cheap score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Include the previous best sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size == n:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = _uniform_distribution(n)
            else:
                prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best cheap score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 30
    seeds = candidates[:top_k]
    print(f"Selected {len(seeds)} seeds for refinement; best cheap score={seeds[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy refinement – smooth Adam (most accurate) + Adam fallback
    best_seq = seeds[0][0].copy()
    best_score = seeds[0][1]

    for idx, (seed_seq, _) in enumerate(seeds):
        if time.time() >= deadline:
            break

        # smooth Adam – long run
        seq, sc = _smooth_adam(seed_seq, deadline,
                               eps_start=1e-2,
                               eps_end=1e-7,
                               max_iters=300_000,
                               step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} smooth‑Adam → {sc:.6f}")

        # standard Adam as a quick alternative
        seq, sc = _adam(seed_seq, deadline,
                        max_iters=200_000,
                        step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient descent on the current best
    subgrad_deadline = deadline - 30.0   # reserve ~30 s for final hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=2_000_000,
                                       eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained hill‑climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02               # initial max transfer (2 % of total mass)
    last_improve_iter = 0

    while time.time() < deadline:
        iteration += 1
        # geometric decay of transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        # pick a pair (i,j) – gradient‑guided every few steps
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                last_improve_iter = iteration
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if iteration % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 1e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

        # restart from best if stuck for a long time
        if iteration - last_improve_iter > 120_000:
            a = best_seq.copy()
            cur_score = best_score
            last_improve_iter = iteration
            print(f"[{time.time():.1f}] restart after stagnation at iter={iteration}")

    # ------------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final cheap score: {final_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
