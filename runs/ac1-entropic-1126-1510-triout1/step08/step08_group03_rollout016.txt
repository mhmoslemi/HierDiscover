Robust search for a non‑negative step‑function that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Broad sampling** – for a set of lengths (≈ 0.7–2× a base length) we
   generate many analytic families (arcsine, symmetric/asymmetric Beta,
   power‑law, Gaussian, sine, cosine, triangular, uniform) and a few
   random Dirichlet draws (with a range of concentration parameters).  All
   vectors are normalised to sum 1 and rated by the cheap surrogate

        cheap(a) = 2·n·max(a*a)        (a*a = full self‑convolution)

   The cheapest ``top_k`` vectors become seeds.

2. **Iterative refinement** – each seed is improved **sequentially** (the
   output of one optimiser becomes the start point of the next) by:

   * exponentiated‑gradient descent,
   * Adam‑style adaptive descent,
   * a continuation method that replaces ``max`` by a soft‑max (γ
     increases during the run),
   * plain sub‑gradient descent,
   * (for modest n ≤ 1200) a linear‑programming “dual‑update’’ that follows
     the scheme described in the statement:
       maximise   Σ g_j   s.t.   (f * g)_k ≤ M   and   g_j ≥ 0,
     where ``M = max(f*f)``.  The solution is renormalised to sum 1 and
     blended into the current iterate with a small step size ``t``.

   All steps observe the global deadline.

3. **Fine‑grained hill‑climbing** – a stochastic mass‑transfer scheme
   (tiny random Δ moved from one index to another) with a geometric decay
   of the transferred amount explores the local neighbourhood.  Occasional
   jitter helps to escape plateaus.

The routine respects the 1000 s budget, never returns an empty list, and
produces a vector that can be fed directly to ``evaluate_sequence``.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of ``max(a*a)`` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms where i=j (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.9):
    """Exponentiated‑gradient (mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        a = a / a_sum if a_sum != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.3,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000,
                         eta0: float = 0.8):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _smooth_gradient(a: np.ndarray, gamma: float):
    """Gradient of the soft‑max approximation of max(a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 1.0,
                             gamma_end: float = 80.0):
    """Continuation: start with a smooth max (small γ) and raise γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    # linear schedule for γ to avoid overflow
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        gamma = gamma0 + (gamma_end - gamma0) * (it - 1) / (max_iters - 1)
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet point with concentration `alpha`."""
    x = np.random.gamma(alpha, 1.0, size=n)
    total = x.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return x / total

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))

def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return w / w.sum()


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0.0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


def _powerlaw_distribution(n: int, p: float) -> np.ndarray:

    vals = 1.0 / ((i + 0.5) ** p * (n - i - 0.5) ** p)
    if vals.sum() == 0.0:

def _gaussian_distribution(n: int, sigma_frac: float) -> np.ndarray:

def _lp_dual_update(f: np.ndarray, M: float, t: float,
                   remaining_deadline: float) -> np.ndarray:
    """
    Perform one LP dual‑update step described in the statement:
    maximise Σ g_j  subject to (f * g)_k ≤ M, g ≥ 0.
    The solution is renormalised to sum 1 and blended with ``f`` using step
    size ``t`` (0 < t ≤ 1).  Returns the blended vector or the original ``f``
    if the LP fails.
    """
    n = f.size
    # Build sparse Toeplitz matrix for the convolution constraints
    i_idx = np.repeat(np.arange(n), n)               # rows: i
    j_idx = np.tile(np.arange(n), n)                 # columns: j
    row_idx = i_idx + j_idx                           # shift k = i+j
    data = f[i_idx]                                   # coefficient f_i

    A = coo_matrix((data, (row_idx, j_idx)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M)

    # maximise Σ g_j  →  minimise  -Σ g_j
    c = -np.ones(n)

    # Solve the LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if not res.success:
        return f

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return f
    g_norm = g / s                     # back to the simplex
    # simple convex blend
    f_new = (1.0 - t) * f + t * g_norm
    f_new = _project_onto_simplex(f_new, s=1.0)
    return f_new


# ----------------------------------------------------------------------
def _heavy_optimize(seed_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Apply a cascade of first‑order methods to a single seed.
    The output of each method becomes the start point for the next one.
    For modest dimensions (≤1200) a single LP dual‑update is inserted
    after the gradient passes.
    """
    a = seed_seq.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    # 1. Exponentiated‑gradient
    a, sc = _exp_gradient_descent(a, deadline,
                                  max_iters=120_000, eta0=0.9)
    if sc < best_score:
        best_a, best_score = a.copy(), sc

    # 2. Adam
    a, sc = _adam_descent(a, deadline,
                          max_iters=120_000, step0=0.3)
    if sc < best_score:
        best_a, best_score = a.copy(), sc

    # 3. Smooth‑max continuation
    a, sc = _smooth_gradient_descent(a, deadline,
                                      max_iters=250_000,
                                      eta0=0.7,
                                      gamma0=1.0,
                                      gamma_end=80.0)
    if sc < best_score:
        best_a, best_score = a.copy(), sc

    # 4. Sub‑gradient tidy‑up
    a, sc = _subgradient_descent(a, deadline,
                                 max_iters=400_000,
                                 eta0=0.8)
    if sc < best_score:
        best_a, best_score = a.copy(), sc

    # 5. Optional LP dual‑update (only for small n to keep it cheap)
    if a.shape[0] <= 1200 and time.time() < deadline:
        M = _subgrad(a)[1]            # current max convolution value
        a_lp = _lp_dual_update(a, M, t=0.2, remaining_deadline=deadline)
        sc_lp = _max_convolution_score(a_lp)
        if sc_lp < best_score:
            best_a, best_score = a_lp.copy(), sc_lp
        a = a_lp                      # continue from the LP result

    # final projection (guard against numerical drift)
    a = _project_onto_simplex(a, s=1.0)
    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (around a previously known length)
    try:
        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        base_n = 3400

    lengths = set()
    for f in (0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(6):
        n = np.random.randint(500, min(15000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Broad sampling of analytic families and random Dirichlet draws
    candidates = []                     # (array, cheap_score)

    beta_params = [0.2, 0.5, 0.8, 1.2, 2.0, 3.0, 5.0]
    power_params = [0.1, 0.25, 0.4, 0.6, 0.8, 1.0]
    sigma_fracs = [0.04, 0.07, 0.10, 0.15, 0.25]   # for Gaussian‑like families
    dirichlet_alphas = [0.1, 0.3, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families -----
        fams = [
            _arcsine_distribution(n),
            _triangular_distribution(n),
            _exponential_distribution(n, lam=0.003),
            _sine_distribution(n),
            _cosine_distribution(n),
            np.full(n, 1.0 / n)
        ]
        for a in fams:
            candidates.append((a, _max_convolution_score(a)))

        # symmetric beta
        for p in beta_params:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # asymmetric beta
        for p in beta_params:
            for q in beta_params:
                a = _beta_asymmetric_distribution(n, p, q)
                candidates.append((a, _max_convolution_score(a)))

        # power‑law (generalised arcsine)
        for p in power_params:
            a = _powerlaw_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # Gaussian‑like (tuned σ)
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sf)
            candidates.append((a, _max_convolution_score(a)))

        # ----- random Dirichlet draws with various concentration -----
        for alpha in dirichlet_alphas:
            for _ in range(2):
                a = _random_dirichlet(n, alpha=alpha)
                candidates.append((a, _max_convolution_score(a)))

        # ----- random mixtures of two families (to increase diversity) -----
        for _ in range(3):
            # pick two families at random
            fam1 = np.random.choice(['arcsine', 'triangular', 'uniform',
                                    'sine', 'cosine'])
            fam2 = np.random.choice(['beta_sym', 'beta_asym',
                                    'exponential', 'powerlaw', 'gaussian'])
            if fam1 == 'arcsine':
                a1 = _arcsine_distribution(n)
            elif fam1 == 'triangular':
                a1 = _triangular_distribution(n)
            elif fam1 == 'uniform':
                a1 = np.full(n, 1.0 / n)
            elif fam1 == 'sine':
                a1 = _sine_distribution(n)
            else:  # cosine
                a1 = _cosine_distribution(n)

            if fam2 == 'beta_sym':
                p = np.random.choice(beta_params)
                a2 = _beta_symmetric_distribution(n, p)
            elif fam2 == 'beta_asym':
                p = np.random.choice(beta_params)
                q = np.random.choice(beta_params)
                a2 = _beta_asymmetric_distribution(n, p, q)
            elif fam2 == 'exponential':
                lam = 10 ** np.random.uniform(-4, -2)
                a2 = _exponential_distribution(n, lam=lam)
            elif fam2 == 'powerlaw':
                p = np.random.choice(power_params)
                a2 = _powerlaw_distribution(n, p)
            else:  # gaussian
                sf = np.random.choice(sigma_fracs)
                a2 = _gaussian_distribution(n, sf)

            mix = np.random.rand()
            a = mix * a1 + (1.0 - mix) * a2
            a = a / a.sum()
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # Include the previously best sequence, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [seq for seq, _ in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Heavy optimisation on each seed (time‑sharing)
    remaining_time = deadline - time.time()
    per_seed_budget = remaining_time * 0.78 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_seed_budget
        seq, sc = _heavy_optimize(seed_seq, sub_deadline)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} heavy‑opt → {sc:.6f}")

    # ------------------------------------------------------------------
    # Fine‑grained hill climbing (tiny random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.04                     # start with up to 4 % of total mass

    while time.time() < deadline:
        it += 1
        # geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (it / 14000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to help escape flat regions
        if it % 4000 == 0 and np.random.rand() < 0.18:
            jitter = np.random.randn(n) * 6e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # The list is already normalised to sum 1, which is a valid input for
    # ``evaluate_sequence``.
    return best_seq.tolist()
