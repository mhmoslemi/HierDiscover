def _lp_step(f, M):
    n = f.shape[0]
    # Build constraint matrix A shape (2*n-1, n)
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # then j = k - i
        # coefficient for g_j = f_i
        # j in [0, n-1]
        # for i in range(i_min,i_max+1):
        #   j = k - i
        #   rows.append(k)
        #   cols.append(j)
        #   data.append(f[i])
        # Could vectorize: fix k, compute i array
        i = np.arange(i_min, i_max+1)
        j = k - i
        rows.extend([k]*len(i))
        cols.extend(j.tolist())
        data.extend(f[i].tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1,n))
    # bounds: g_j >= 0
    bounds = [(0, None)] * n
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds,
                                 method='highs', options={'presolve': True})
    if not res.success:
        return None
    g = res.x
    return g
```

def _beta_symmetric_distribution(n, p):
    # using log to avoid overflow
    x = (np.arange(n) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1-x)
    w = np.exp(logw - np.max(logw))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0/n)
    else:
        w /= w_sum
    return w
```

def _lp_step(a, deadline):
    n = a.shape[0]
    conv_aa = _fft_self_convolve(a)
    M = conv_aa.max()
    # Build constraints only for indices where conv_aa is close to M? Could build all for safety
    # Use sparse matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        i_vals = np.arange(i_min, i_max+1)
        j_vals = k - i_vals
        rows.extend([k]*len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(a[i_vals].tolist())
    A = scipy.sparse.csr_matrix((data,(rows,cols)), shape=(2*n-1, n))
    # objective maximize sum(g)
    c = -np.ones(n)
    # bounds
    bounds = [(0, None)] * n
    # LP solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds,
                                method='highs', options={'presolve': True, 'time_limit': max(1, int(deadline - time.time()))})
    if not res.success:
        return None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    g_scaled = g / sum_g
    return g_scaled
```

"""
Propose a low‑score non‑negative step function.

The search proceeds in stages:

1️⃣ Choose a modest set of sequence lengths (≈3000‑6000) and generate a
   rich pool of candidates from many analytic families (Beta, arcsine,
   Hann / Hamming / Black‑man windows, exponential, power‑law, random
   Dirichlet draws).  For the symmetric Beta family the shape parameter
   is optimised with a cheap 1‑D scalar optimisation.

2️⃣ Keep the best few (≤5) seeds and refine each of them with
   – Adam‑style adaptive moment descent on the simplex,
   – exponentiated‑gradient (KL‑mirror) descent,
   – a Frank‑Wolfe “mass‑transfer” move that greedily shifts probability
     from the component with the largest sub‑gradient to the one with the
     smallest,
   – a single LP‑step (maximise Σg under the current max‑convolution
     bound) followed by a line‑search update,
   – a light simulated‑annealing jitter phase.

All steps respect the global 1000 s budget and the sequence is always
projected onto the probability simplex (∑a=1, a≥0).  The surrogate
objective 2·n·max(conv(a,a)) equals the true `evaluate_sequence`
objective for such normalised sequences, so we minimise the surrogate
and finally return the best list found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate score 2·n·max(conv(a,a)) (exact when sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.20,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 1_200_000):
    """Adam‑style descent on the simplex (surrogate objective)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 800_000):
    """Exponentiated‑gradient (mirror descent w.r.t. KL)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families (all return a probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space for stability
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha          # in [-1,1]
    w = np.i0(beta * np.sqrt(1.0 - t * t)) / np.i0(beta)
    w /= w.sum()
    return w

def _optimize_beta_sym(n: int) -> float:
    """Find a p that (approximately) minimises the surrogate for a symmetric Beta."""
    def obj(p):
        a = _beta_symmetric_distribution(n, p)
        return _max_convolution_score(a)

    # search in a reasonable range; arcsine corresponds to p=0.5
    res = scipy.optimize.minimize_scalar(
        obj, bounds=(0.05, 2.0), method='bounded',
        options={'maxiter': 30, 'xatol': 1e-4})
    return res.x if res.success else 0.5

def _lp_step(f: np.ndarray, deadline: float) -> np.ndarray | None:
    """
    Solve the linear programme:
        maximise Σ g_j
        s.t.   (f * g)[k] ≤ M   for all k
               g_j ≥ 0
    where M = max_k (f*f)[k].
    Return the scaled solution (∑g=1) if it strictly improves the
    max‑convolution, otherwise None.
    """
    n = f.shape[0]
    # current bound
    conv_ff = _fft_self_convolve(f)
    M = conv_ff.max()

    # pick only constraints that are active (within 1e‑8 relative tolerance)
    eps = 1e-8 * M
    active = np.flatnonzero(conv_ff >= M - eps)
    if active.size == 0:
        active = np.arange(conv_ff.size)   # fall back to all constraints

    rows = []
    cols = []
    data = []
    for ridx, k in enumerate(active):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1, dtype=np.int64)
        j_vals = k - i_vals
        rows.extend([ridx] * len(i_vals))
        cols.extend(j_vals.tolist())
        data.extend(f[i_vals].tolist())

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active), n))
    b = np.full(len(active), M, dtype=np.float64)

    # maximise Σ g  <=> minimise -1ᵀ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Solve with HiGHS (fast interior‑point)
    try:
        lp_res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b, bounds=bounds,
            method='highs', options={'presolve': True})
    except Exception:
        return None

    if not lp_res.success:
        return None

    g = lp_res.x
    s = g.sum()
    if s <= 1e-12:
        return None
    g_scaled = g / s

    # check improvement
    new_max = _fft_self_convolve(g_scaled).max()
    if new_max < M - 1e-12:
        return g_scaled
    return None

            if a_lp is not None:
                lp_score = _max_convolution_score(a_lp)
                if lp_score < best_score - 1e-14:
                    best_seq, best_score = a_lp.copy(), lp_score
                    print(f"[{time.time():.1f}]   LP step improved to {best_score:.6f}")

        # ------------------------------------------------------------------
        # 7️⃣ Light simulated‑annealing / jitter phase
        a = best_seq.copy()
        cur_score = best_score
        anneal_iter = 0
        while time.time() < deadline and anneal_iter < 150_000:
            anneal_iter += 1
            # linear temperature schedule
            t_frac = (deadline - time.time()) / (deadline - start_time)
            temp = 0.001 + 0.02 * t_frac

            i = np.random.randint(a.size)
            j = np.random.randint(a.size - 1)
            if j >= i:
                j += 1
            max_delta = min(a[i], 1.0 - a[j])
            if max_delta <= 0:
                continue
            delta = max_delta * np.random.rand() * temp

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            new_score = _max_convolution_score(a_new)

            if new_score < cur_score - 1e-15:
                a = a_new
                cur_score = new_score
                if cur_score < best_score - 1e-14:
                    best_seq, best_score = a.copy(), cur_score
                    print(f"[{time.time():.1f}]   Anneal iter={anneal_iter} new best {best_score:.6f}")
            else:
                # Metropolis acceptance
                prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
                if np.random.rand() < prob:
                    a = a_new
                    cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list (already normalised to sum‑1)
    return best_seq.tolist()
