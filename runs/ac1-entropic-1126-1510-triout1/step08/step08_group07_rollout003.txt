assistantfinal**Solution Overview**

The search consists of four stages  

1. **Rich initialisation** – several analytic families (uniform, arcsine, beta,
   exponential, alternating, sparse random) are sampled and the best one is
   kept.  
2. **Continuation gradient descent** – we minimise a smooth surrogate of the
   max‑convolution

   \[
     M_\alpha(a)=\frac{1}{\alpha}\log\Bigl(\sum_k e^{\alpha\,b_k}\Bigr),
     \qquad b = a*a .
   \]

   For a fixed smoothing parameter `α` we can write the gradient in closed
   form

   \[
     \nabla M_\alpha(a)=2\;(a\star w),\qquad
     w_k=\frac{e^{\alpha(b_k-\max b)}}{\sum_j e^{\alpha(b_j-\max b)}} .
   \]

   The cross‑correlation `a⋆w` is obtained with two FFTs, so each iteration
   is cheap.  Starting with a small `α` we slowly increase it (continuation)
   which drives the solution toward the true `max` while keeping the landscape
   smooth enough for gradient steps.  After each gradient step we project back
   onto the probability simplex.

3. **Aggressive greedy active‑set optimisation** – the same routine that was
   used in the previous state‑of‑the‑art is applied to the result of stage 2,
   performing exact sub‑gradient moves on the active set of maximal
   convolution shifts.

4. **Simulated‑annealing fine‑tuning** – a cheap Metropolis walk with exponential
   cooling cleans up remaining tiny irregularities.

All stages respect the global deadline (`budget_s`).  The final vector is
projected onto the simplex and returned as a plain list of floats; it can be
directly fed to `evaluate_sequence`.

"""
Step‑function minimiser for `evaluate_sequence`.

The algorithm proceeds in four phases:
1.  Generate a diverse catalogue of analytic probability vectors and keep the
    best (by the true objective 2·n·max(conv)).
2.  Continuation gradient descent on a smoothed max‑convolution objective.
    The gradient is obtained analytically and evaluated with two FFTs per
    iteration; after each step the vector is projected onto the simplex.
3.  Aggressive greedy active‑set optimisation (identical to the previous SOTA).
4.  Short simulated‑annealing walk for final polishing.

All work stops before the supplied time budget.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a

# 2) Continuation gradient descent on smoothed max ----------------------
def _smooth_obj_and_grad(a: np.ndarray, alpha: float):
    """
    Returns
        max_b    – maximal entry of the self‑convolution,
        b        – full self‑convolution,
        grad     – gradient of (1/α)·log‑sum‑exp(b) w.r.t. a.
    """
    n = a.shape[0]

    # ----- self‑convolution -----------------------------------------
    size1 = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size1)
    b = np.fft.irfft(A * A, n=size1)[: 2 * n - 1]

    max_b = b.max()
    # ----- smoothed max ---------------------------------------------
    # w_k ∝ exp(α·(b_k‑max_b)), safe because max_b is subtracted.
    exp_term = np.exp(alpha * (b - max_b))
    w = exp_term / exp_term.sum()

    # ----- gradient (cross‑correlation a⋆w) -------------------------
    w_rev = w[::-1]                       # reversed weight vector
    size2 = _next_pow2(n + w_rev.size - 1)
    A = np.fft.rfft(a, n=size2)
    W = np.fft.rfft(w_rev, n=size2)
    conv_full = np.fft.irfft(A * W, n=size2)[: n + w_rev.size - 1]

    centre = w_rev.size - 1                # index that yields the cross‑correlation
    grad = 2.0 * conv_full[centre - np.arange(n)]

    return max_b, b, grad

    Returns the best vector and its exact objective.
    """

        it = 0

            for _ in range(8):
                a_new = a - step * grad
                a_new = _project_onto_simplex(a_new, 1.0)
                new_score = _objective(a_new)
                if new_score < cur_score - 1e-14:
                    a = a_new
                    cur_score = new_score
                    improved = True
                    break
                step *= 0.5
                if step < 1e-12:
                    break

            if not improved:
                # occasional random jitter to escape tiny plateaus
                if rng.random() < 0.05:
                    i = rng.integers(len(a))
                    if a[i] > 1e-12:
                        j = rng.integers(len(a) - 1)
                        if j >= i:
                            j += 1
                        eps = min(a[i] * 0.05, 0.001)
                        a_tmp = a.copy()
                        a_tmp[i] -= eps
                        a_tmp[j] += eps
                        a_tmp = _project_onto_simplex(a_tmp, 1.0)
                        scr = _objective(a_tmp)
                        if scr < cur_score:
                            a = a_tmp
                            cur_score = scr
            # early exit if the improvement becomes extremely small
            if it % 500 == 0:
                print(f"[{time.time():.2f}] grad‑desc α={alpha:.1f} it={it} score={cur_score:.6f}")

        print(f"[{time.time():.2f}] finished α={alpha:.1f} best={cur_score:.6f}")

    return a, cur_score


# ----------------------------------------------------------------------
# 3) Greedy active‑set optimiser (same as the previous SOTA) ---------
def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient w.r.t. the true max (active set)
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # try moving a decent chunk of mass
        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.2f}] greedy it={it} improved to {best_score:.6f}")
            continue

        # smaller fraction if the full move fails
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.2f}] greedy (small) it={it} improved to {best_score:.6f}")
            continue

        # occasional jitter
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.2f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score


# ----------------------------------------------------------------------
# 4) Simulated‑annealing fine‑tuning ------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03  # initial temperature (objective units)
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.2f}] SA new best {best_score:.6f}")

        T *= 0.9997

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    probability vector found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # length – a few thousand gives a good trade‑off between quality and speed
    n = 3500
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 1) rich initialisation
    candidates = _initial_candidates(n, rng)

    # include a previous construction if it is present in the environment
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.2f}] init cand {idx} score={sc:.6f}")

    if best_vec is None:
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) continuation gradient descent (smooth max)
    best_vec, best_score = _smooth_gradient_descent(best_vec, deadline, rng)

    # ------------------------------------------------------------------
    # 3) aggressive greedy optimisation
    best_vec, best_score = _greedy_local_search(best_vec, deadline, rng)

    # ------------------------------------------------------------------
    # 4) simulated‑annealing fine‑tuning (if time remains)
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)

    # ------------------------------------------------------------------
    # final sanity – enforce simplex and report final objective
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
