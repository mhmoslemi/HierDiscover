Main ideas

   (0.5 → 64) inside a projected Adam optimiser.
3. **Correct sub‑gradient for Frank‑Wolfe** – the previous implementation
   erroneously sub‑tracted a diagonal term.  Here we use the mathematically
   correct gradient  ∇(a*a)_i = 2·a shifted, which yields a more effective
   pairwise mass‑transfer loop.
4. **Longer refinement & more seeds** – top‑k increased to 10 and each
   seed receives up to 150 k Adam‑smooth steps plus a cheap SGD pass.
5. **Optional LP boost** – preserved for moderate lengths (≤2000).

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate of evaluate_sequence when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _raised_cosine_distribution(n: int) -> np.ndarray:
    """Hann window on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * (i + 0.5) / n)
    w /= w.sum()
    return w


def _sine_power_distribution(n: int, power: float = 1.0) -> np.ndarray:
    """|sin(π (i+0.5)/n)|^power, normalised."""
    i = np.arange(n, dtype=np.float64)
    w = np.abs(np.sin(np.pi * (i + 0.5) / n)) ** power
    w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float = 1.0) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^p  (symmetric, zero at edges for p>0)."""
    i = np.arange(n, dtype=np.float64)
    w = ((i + 1) * (n - i)) ** p
    w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Sample from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# ----------  smooth‑max gradient (log‑sum‑exp)  ------------------------
def _smooth_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of f_α(a) = (1/α)·log Σ_i exp(α·(a*a)_i).
    Returns (grad, f_val).  a is assumed to sum to 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                # length 2n‑1
    max_b = b.max()
    # weights for the log‑sum‑exp
    shifted = b - max_b                      # ≤ 0
    exp_vals = np.exp(alpha * shifted)
    sum_exp = exp_vals.sum()
    # avoid division by zero – when alpha is huge and only one entry is max,
    # exp_vals will be 1 at the max and ~0 elsewhere, still fine.
    w = exp_vals / sum_exp
    # convolution of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    # smooth objective value (equal to max_b + log(sum_exp)/α)
    f_val = float(max_b + np.log(sum_exp) / alpha) if alpha != 0 else float(max_b)
    return grad, f_val

# ----------  correct sub‑gradient of max(b) (average over active) -----
def _subgrad_max(a: np.ndarray):
    """
    Returns (gradient, cur_max, active_set) for the true max(b)
    where gradient = Σ_{i∈active} ∇b_i / |active|.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)
    # uniform weight over the active set
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    # gradient = 2 * conv_rev (no diagonal correction!)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, cur_max, active

# -------------------  Adam on smooth‑max  ---------------------------
def _adam_smooth_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 150_000,
                         step0: float = 0.2,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         alpha_start: float = 0.5,
                         alpha_end: float = 64.0,
                         alpha_update_iters: int = 20_000) -> tuple[np.ndarray, float]:
    """
    Projected Adam minimising the smoothed max objective.
    The smoothing parameter α is doubled every `alpha_update_iters`.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    alpha = alpha_start

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, alpha)
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
        # update smoothing parameter
        if it % alpha_update_iters == 0 and alpha < alpha_end:
            alpha = min(alpha_end, alpha * 2.0)

    return best_a, best_score

def _sgd_max_descent(a0: np.ndarray, deadline: float,
                     max_iters: int = 120_000,
                     step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Projected SGD using the exact sub‑gradient of max(b).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad_max(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

# -------------------  linear‑programming boost ----------------------
def _lp_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """One‑shot LP improvement for modest lengths (≤2000)."""
    n = a.shape[0]
    if n > 2000:
        return a, _max_convolution_score(a)

    cur_max = _subgrad_max(a)[1]
    # Build sparse A such that (a * g)_k = Σ_j a_{k-j} g_j  ≤ cur_max
    from scipy.sparse import csr_matrix
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, cur_max, dtype=np.float64)

    # maximise Σ g_j → minimise −Σ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    from scipy.optimize import linprog
    remaining = max(0.0, deadline - time.time())
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': remaining, 'presolve': True})
    if not res.success:
        return a, _max_convolution_score(a)

    g = res.x
    if g.sum() > 0:
        g = g / g.sum()
    else:
        g = a.copy()

    # blend a little bit of g and polish
    t = 0.2
    a_new = (1 - t) * a + t * g
    a_new = _project_onto_simplex(a_new, s=1.0)
    a_ref, sc_ref = _sgd_max_descent(a_new, deadline, max_iters=60_000, step0=0.4)
    return a_ref, sc_ref

        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # keep the top seeds (increase to 10)
    candidates.sort(key=lambda x: x[1])
    top_k = min(10, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Current best seed score = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # --------------------------------------------------------------
    # 2. intensive refinement of each top seed
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Adam on smoothed max
        seq, sc = _adam_smooth_descent(seed_seq, deadline,
                                        max_iters=200_000,
                                        step0=0.25,
                                        alpha_start=0.5,
                                        alpha_end=64.0,
                                        alpha_update_iters=15_000)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam‑smooth -> {best_score:.6f}")

        # fallback: plain SGD on the true max (helps escape plateaus)
        seq, sc = _sgd_max_descent(seed_seq, deadline, max_iters=120_000, step0=0.5)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} SGD‑max -> {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Frank‑Wolfe style pairwise mass transfer on the global best
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 120_000:
        it_fw += 1
        grad, cur_max, active = _subgrad_max(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-12:
            a = a_full
            cur_score = score_full
            improved = True
        else:
            # try half‑transfer
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-12:
                a = a_half
                cur_score = score_half
                improved = True

        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={it_fw} new best {best_score:.6f}")

        # occasional jitter to shake off stagnation
        if it_fw % 4000 == 0 and np.random.rand() < 0.13:
            jitter = np.random.randn(a.size) * 5e-6
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    # 4. linear‑programming boost (if time permits)
    if time.time() < deadline - 2.0:
        try:
            best_seq, best_score = _lp_refine(best_seq, deadline)
            print(f"[{time.time():.1f}] LP refinement gave new best {best_score:.6f}")
        except Exception as e:
            print("LP refinement failed:", e)

    # --------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
