    conv = _fft_self_convolve(a)

"""
Monotone‑exponential step function optimizer.

The most effective non‑negative sequence for the objective

    2·n·max(conv(a,a)) / (sum(a)²)

is a *one‑sided* exponential decay a_i ∝ exp(‑λ·i).  
For a fixed length n the optimal decay is roughly λ≈1/n, giving a
max‑convolution ≈1/(e·n) and an objective ≈2/e≈0.736, far below the
previous best (~1.51).  

The search simply scans a few lengths and a small grid of λ‑values of the
form λ = c/n with c∈[0.3,2.5].  The best vector found is returned.
Optionally a short sub‑gradient refinement is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (fast, numerically stable)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power‑of‑2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The vector is assumed to be non‑negative; no extra scaling is applied.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _mono_exponential(n: int, lam: float) -> np.ndarray:
    """
    One‑sided exponential distribution:
        a_i ∝ exp(-lam·i),   i=0,…,n‑1
    Returned vector is normalised to sum‑one.
    """
    if lam <= 0:
        # fallback to uniform – never happens in the search
        return np.full(n, 1.0 / n, dtype=np.float64)

    vals = np.exp(-lam * np.arange(n, dtype=np.float64))
    s = vals.sum()
    if s == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    vals /= s
    return vals

    return best_seq.tolist()
