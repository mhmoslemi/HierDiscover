def _upsample(a, new_n):
    old_n = a.shape[0]
    x_old = np.linspace(0, 1, old_n, endpoint=False) + 0.5/old_n
    x_new = np.linspace(0, 1, new_n, endpoint=False) + 0.5/new_n
    return np.interp(x_new, x_old, a)

def _adam_logsumexp(a0, deadline, best_seq, best_score):
    a = a0.copy()
    # convert to logits
    eps = 1e-12
    logits = np.log(a + eps)  # ensure >0
    n = a.shape[0]
    # Adam parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; epsilon=1e-8
    t = 0
    # schedule
    schedule = [(0.5, 500, 0.2), (1, 1000,0.15), ...] maybe truncated.

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                # compute best seq from current a
                cur_seq = np.exp(logits - np.max(logits))
                cur_seq /= cur_seq.sum()
                cur_score = _objective(cur_seq)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                return best_seq, best_score
            t += 1
            a = np.exp(logits - np.max(logits))
            a /= a.sum()
            conv = _fft_self_convolve(a)
            maxb = conv.max()
            w = np.exp(alpha * (conv - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad_a = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]

def _local_search_simulated_annealing(a, deadline, best_seq, best_score):
    n = a.shape[0]
    # Melt copy of a
    cur_a = a.copy()
    # record baseline
    best_seq_local = best_seq.copy()
    best_score_local = best_score
    # Temperature schedule: start high, exponential decay each iteration
    # choose max_iters set as number of possible loops (based on time). We'll keep loops while time<deadline.
    iter = 0
    # Determine approximate remaining loops = int((deadline - time.time()) * 200) maybe.
    while time.time() < deadline:
        # compute temperature based on iter
        T = max(1e-6, 1e-2 * (0.9999 ** iter))  # decays slowly
        # randomly pick source with some mass
        src_idx = np.random.choice(np.where(cur_a > 1e-12)[0])
        dest_idx = np.random.randint(n - 1)
        if dest_idx >= src_idx:
            dest_idx += 1
        # random fraction of source mass up to 0.5
        fraction = np.random.rand() * 0.5
        delta = fraction * cur_a[src_idx]
        if delta < 1e-15:
            continue
        new_a = cur_a.copy()
        new_a[src_idx] -= delta
        new_a[dest_idx] += delta
        # evaluate
        new_score = _objective(new_a)
        # acceptance
        if new_score < best_score_local or np.random.rand() < np.exp(-(new_score - best_score_local) / T):
            cur_a = new_a
            if new_score < best_score_local:
                best_score_local = new_score
                best_seq_local = new_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter} new best {best_score_local:.6f}")
        iter += 1
    return best_seq_local, best_score_local

"""
Improved step‑function optimiser.

The algorithm consists of three stages:

1. **Rich seeding** – generates a large variety of analytic and random
   probability vectors (uniform, arcsine, Beta(p,q), power‑law edges,
   exponential, block spikes, alternating, sparse Dirichlet, …) for a few
   discretisation sizes.  The best few seeds are kept.

2. **Smooth Adam descent** – optimises a log‑sum‑exp smooth approximation of
   the true objective `2·n·max(conv(a,a))`.  Adam is applied to the *logits*
   of the distribution (softmax parametrisation) which automatically enforces
   non‑negativity and the unit‑sum constraint.  A schedule of increasing
   smoothing parameters `α` gradually turns the smooth objective into the
   true max.

3. **Simulated‑annealing fine‑tuning** – a cheap Monte‑Carlo walk that moves
   random amounts of mass between indices.  Moves are accepted if they
   improve the exact objective or, with a temperature‑dependent probability,
   if they are slightly worse, allowing the search to escape shallow local
   minima.

All stages respect the global 1000 s time budget.  The best vector found is
returned as a plain Python list of `float`s, ready to be fed into
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (full convolution)

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Exact objective (sum of coefficients is assumed to be 1)

def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a unit‑sum vector a."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Probability‑distribution families (all return a unit‑sum vector)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta(p,q) on the grid { (i+0.5)/n }."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    return _distribution_beta_asym(n, p, p)


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           concentration: float = 1.0) -> np.ndarray:
    """Sample from a Dirichlet(α) with scalar concentration."""
    alpha = np.full(n, concentration, dtype=np.float64)
    sample = rng.gamma(alpha)
    sample /= sample.sum()
    return sample


# ----------------------------------------------------------------------
# Helper: softmax (logits → probability vector)

def _softmax(z: np.ndarray) -> np.ndarray:
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Upsampling (linear interpolation) – used when we increase the discretisation size

def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(a) to new_n, preserving mass."""
    old_n = a.shape[0]
    # centre positions on [0,1)
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
# Seed generation – a fairly broad pool for a given length

def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # classic closed‑form families
    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))
    seeds.append(_distribution_triangular(n))
    seeds.append(_distribution_sine(n))

    # symmetric Beta
    for p in (0.1, 0.2, 0.5, 1.0, 2.0, 3.0):
        seeds.append(_distribution_beta_sym(n, p))

    # asymmetric Beta
    for p in (0.2, 0.5, 1.0, 2.0):
        for q in (0.5, 1.0, 2.0, 5.0):
            seeds.append(_distribution_beta_asym(n, p, q))

    # power‑law at edges
    for gamma in (0.07, 0.15, 0.25, 0.35):
        seeds.append(_distribution_powerlaw_edges(n, gamma))

    # exponentials
    for lam in (0.001, 0.003, 0.006, 0.015):
        seeds.append(_distribution_exponential(n, lam))

    # alternating spikes
    for stride in (2, 3, 4):
        seeds.append(_distribution_alternating(n, stride))

    # two‑block spikes
    for frac in (0.03, 0.07, 0.12):
        seeds.append(_distribution_two_blocks(n, frac))

    # random Dirichlet draws
    for _ in range(4):
        conc = rng.uniform(0.05, 5.0)
        seeds.append(_distribution_dirichlet(n, rng, concentration=conc))

    # sparse random vectors
    for _ in range(4):
        density = rng.uniform(0.005, 0.03)
        seeds.append(_distribution_random_sparse(n, rng, density=density))

    return seeds


# ----------------------------------------------------------------------
# Adam optimisation on the smooth max‑convolution objective

def _adam_smooth(a0: np.ndarray, deadline: float,
                 global_best_seq: np.ndarray, global_best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam on logits with a log‑sum‑exp smoothing of the max.
    Returns the best sequence found during the run and its score.
    """
    n = a0.shape[0]
    # initialise logits from a0 (add a tiny epsilon for safety)
    eps = 1e-14
    logits = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, epsilon = 0.9, 0.999, 1e-8
    t = 0

    # smoothing schedule (α, number of steps, learning rate)
    schedule = [
        (0.5,   800, 0.20),
        (1.0,  1200, 0.16),
        (5.0,  1800, 0.13),
        (20.0, 2000, 0.10),
        (100.0,1500, 0.08),
        (500.0,1200, 0.06),
        (2000.0,800, 0.04),
        (5000.0,600, 0.03),
    ]

    best_seq = global_best_seq.copy()
    best_score = global_best_score

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                # final evaluate before early exit
                a = _softmax(logits)
                cur = _objective(a)
                if cur < best_score:
                    best_score = cur
                    best_seq = a.copy()
                return best_seq, best_score

            t += 1

            # current probability vector
            a = _softmax(logits)

            # smoothed max via log‑sum‑exp
            conv = _fft_self_convolve(a)
            max_conv = conv.max()
            w = np.exp(alpha * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smooth max w.r.t. a
            grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

            # back‑prop through softmax
            dot = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot)

            # Adam update on logits
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            logits = logits - lr * m_hat / (np.sqrt(v_hat) + epsilon)

            # occasional exact evaluation
            if t % 250 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    # final check
    a = _softmax(logits)
    cur = _objective(a)
    if cur < best_score:
        best_score = cur
        best_seq = a.copy()
        print(f"[{time.time():.1f}] final Adam best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local refinement (mass transfers)

def _simulated_annealing(a_start: np.ndarray, deadline: float,
                         global_best_seq: np.ndarray, global_best_score: float) -> tuple[np.ndarray, float]:
    """
    Random mass transfers between coordinates.
    Accepts better moves always; worse moves with a temperature‑dependent probability.
    """
    n = a_start.shape[0]
    cur_a = a_start.copy()
    cur_score = _objective(cur_a)

    best_seq = global_best_seq.copy()
    best_score = global_best_score

    iter_idx = 0
    T0 = 1e-2                # initial temperature
    T_min = 1e-8             # final temperature

    # estimate rough number of iterations we can afford
    # (convolution dominates runtime, approx 0.001–0.003 s per eval)
    # we pick a generous upper bound and let the time check stop us.
    while time.time() < deadline:
        # exponential cooling
        T = max(T_min, T0 * (0.9995 ** iter_idx))

        # choose a source index that actually has mass
        src_candidates = np.where(cur_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 50 % of the source mass
        delta = cur_a[i] * np.random.rand() * 0.5
        if delta < 1e-15:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _objective(new_a)

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter_idx} new best {best_score:.6f}")

        iter_idx += 1

    return best_seq, best_score
