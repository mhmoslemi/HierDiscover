def _mass_transfer_opt(a, grad):
    # compute removal capacity: indices where grad>0, capacity = a_i
    pos = np.where(grad > 0)[0]
    # compute addition capacity: indices where grad < 0, capacity = 1 - a_i
    neg = np.where(grad < 0)[0]
    if pos.size == 0 or neg.size == 0:
        return a, False
    total_rem = a[pos].sum()
    total_add = (1 - a[neg]).sum()
    total_transfer = min(total_rem, total_add)
    if total_transfer <= 1e-12:
        return a, False

    # sort pos by grad descending (largest gradient) and neg by grad ascending (most negative)
    pos_order = pos[np.argsort(-grad[pos])] # descending
    neg_order = neg[np.argsort(grad[neg])] # ascending

    a_new = a.copy()
    transfer_left = total_transfer

    # removal
    for i in pos_order:
        if transfer_left <= 0:
            break
        rem = a_new[i]
        delta = min(rem, transfer_left)
        a_new[i] -= delta
        transfer_left -= delta

    # At this point transfer_left = 0 (should be). But need to add same amount to neg.
    add_left = total_transfer
    for j in neg_order:
        if add_left <= 0:
            break
        cap = 1 - a_new[j]
        delta = min(cap, add_left)
        a_new[j] += delta
        add_left -= delta

    # ensure sum remains 1 (small floating error) then project onto simplex for safety
    a_new = _project_onto_simplex(a_new, s=1.0)
    # compute new score
    new_score = _max_convolution_score(a_new)
    return a_new, new_score

"""
propose_candidate
=================

Search for a non‑negative step function that minimizes the provided
``evaluate_sequence``.  The algorithm works in three stages:

1. **Broad sampling** – generate a large collection of analytic families
   (arcsine, smoothed arcsine, semicircle, parabolic, symmetric/asymmetric
   beta, power‑law, uniform, random Dirichlet).  All candidates are
   ℓ₁‑normalised (∑a=1) and scored with the cheap surrogate
   ``2·n·max(conv(a,a))`` which equals the true score when the sum is 1.

2. **Intensive refinement** – keep the best few seeds and improve them with
   (i) Adam‑style adaptive‑moment descent on the simplex,
   (ii) exponentiated‑gradient descent, and
   (iii) a *mass‑transfer* step that solves the linearised sub‑problem:
        move as much probability mass as possible from coordinates
        with large positive sub‑gradient to those with large negative
        sub‑gradient (the exact optimal move for the current active set).

3. **Polishing** – a short simulated‑annealing / jitter phase to escape
   tiny plateaus.

The routine respects the 1000 s budget, prints progress information and
always returns a Python list ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
# Simplex utilities ------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolutions -------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Surrogate objective -----------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Surrogate 2·n·max(conv(a,a)) (exact when ∑a = 1).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the L∞‑norm of the self‑convolution --------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current max value, active indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                # full self‑convolution
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
# Optimisers --------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 500_000):
    """
    Adam‑style descent on the simplex.  Returns the best point seen.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 400_000):
    """
    Exponentiated‑gradient (KL‑mirror) descent on the simplex.
    Returns the best point seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _mass_transfer_step(a: np.ndarray, grad: np.ndarray):
    """
    Perform the optimal linear mass transfer based on the current
    sub‑gradient.  Returns a new feasible vector and a flag whether the
    transfer was non‑trivial.
    """
    # Positive gradient → we would like to *decrease* this coordinate.
    pos = np.where(grad > 0)[0]
    # Negative gradient → we would like to *increase* this coordinate.
    neg = np.where(grad < 0)[0]
    if pos.size == 0 or neg.size == 0:
        return a, False

    # total removable mass and total capacity to add
    total_rem = a[pos].sum()
    total_add = (1.0 - a[neg]).sum()
    tot = min(total_rem, total_add)
    if tot <= 1e-15:
        return a, False

    # order of removal / addition
    pos_order = pos[np.argsort(-grad[pos])]   # largest positive first
    neg_order = neg[np.argsort(grad[neg])]   # most negative first

    a_new = a.copy()
    left = tot

    # remove mass
    for i in pos_order:
        if left <= 0:
            break
        removable = a_new[i]                # can go down to 0
        delta = min(removable, left)
        a_new[i] -= delta
        left -= delta

    # add the same amount
    left = tot
    for j in neg_order:
        if left <= 0:
            break
        capacity = 1.0 - a_new[j]           # can go up to 1
        delta = min(capacity, left)
        a_new[j] += delta
        left -= delta

    # final projection to guard against tiny numerical drift
    a_new = _project_onto_simplex(a_new, s=1.0)
    return a_new, True

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Sample a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _arcsine_smooth_distribution(n: int, eps_factor: float = 0.0) -> np.ndarray:
    """
    Same as the arcsine law but with a global additive epsilon
    (scaled with n²) to flatten the huge edge spikes.
    """
    i = np.arange(n, dtype=np.float64)
    eps = eps_factor * (n ** 2)
    denom = (i + 0.5) * (n - i - 0.5) + eps
    vals = 1.0 / np.sqrt(denom)
    vals /= vals.sum()
    return vals


def _semicircle_distribution(n: int) -> np.ndarray:
    """Discrete Wigner semicircle: √(1‑(2x‑1)²) on the grid."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    vals = np.sqrt(1.0 - (2 * x - 1) ** 2)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= s
    return vals


def _parabolic_distribution(n: int) -> np.ndarray:
    """Beta(2,2) – proportional to (i+½)(n‑i‑½)."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) * (n - i - 0.5)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)
