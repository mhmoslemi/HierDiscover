assistantfinal**Solution Overview**

The task is to find a non‑negative vector `a` (a step‑function) that minimises  

```
E(a) = 2·n·max( a * a ) / ( Σ a )²          ( * is the linear convolution * )
```

`E(a)` is invariant to scaling, therefore we can restrict ourselves to vectors with
`Σ a = 1`.  
The problem is non‑convex, but a good practical strategy is:

1. **Generate many diverse initial candidates** – analytic families (uniform,
   arcsine, beta, exponential, etc.), random Dirichlet draws and a few
   handcrafted asymmetric shapes.
2. **Pick the best of the candidates** using a cheap FFT based evaluation.
3. **Improve the chosen vector with a linear‑programming “mixing step”.**  
   For a fixed `a` and its current worst convolution value `M = max(a*a)` we
   solve  

```
max   Σ g_i
s.t.  (a * g)_k ≤ M          for all k
      g_i ≥ 0
```

   The constraints are linear in `g`.  The solution `g` tells us a direction
   that cannot increase the current worst entry.  We move a fraction `t` towards
   `g` and renormalise.
4. **Polish the result with a soft‑max mirror‑descent optimiser** (exponentiated
   gradient).  An increasing “temperature’’ `α` turns the max into a smooth
   maximum, while an Adam‑style adaptive step‑size accelerates convergence.
5. **Final O(n) hill‑climber** – mass is transferred between two indices;
   the convolution is updated in `O(n)` using the analytic formula from
   `_apply_transfer`.  Only moves that lower the maximal entry are kept.

All steps respect the global 1000 s wall‑clock limit; the most expensive part
( the LP ) is executed only on moderate sizes (`n ≤ 3000`) to keep memory
consumption reasonable.

Below is a self‑contained implementation that follows the plan described
above.  The function `propose_candidate` returns the best sequence found
within the time budget.

The algorithm works in three phases:

1.  Broad scan of analytic families (uniform, arcsine, beta, exponential,
    linear ramps, etc.) and random Dirichlet draws for several lengths.
    The cheapest FFT‑based score 2·n·max(conv) is used to keep the best
    candidate.

2.  An LP‑based “mixing’’ step: with the current vector `a` and its worst
    convolution value `M`, maximise Σg under the linear constraints
    (a * g)_k ≤ M.  The optimal `g` never raises the current maximum.  A
    small convex combination `a ← (1‑t)·a + t·g` (t≈0.15) is performed and the
    vector is projected back onto the probability simplex.

3.  A refined Adam‑accelerated mirror‑descent that minimises a smooth
    approximation of the max (soft‑max with growing temperature) followed by
    a cheap hill‑climber that performs O(n) mass‑transfer updates.

The routine respects a 1000 s wall‑clock limit and returns the best sequence
as a plain list of floats.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)[: a.size + b.size - 1]
    return conv


def _max_convolution_score(a: np.ndarray) -> float:
    """2·n·max(a*a) for a normalised to Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with Σ = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential_centered(n, lam=0.005):
    """Symmetric exponential around the centre."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_exponential_decay(n, lam=0.005):
    """Monotone decreasing exponential (no symmetry)."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


def _distribution_beta(n, p, q):
    """Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Guard against p,q < 1 producing infinities at the borders
    w = np.exp((p - 1) * np.log(np.clip(x, 1e-12, 1 - 1e-12))
               + (q - 1) * np.log(np.clip(1 - x, 1e-12, 1 - 1e-12)))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_linear_ramp(n, up=True):
    """Linear ramp: up=True → increasing, up=False → decreasing."""
    if up:
        vals = np.arange(1, n + 1, dtype=np.float64)
    else:
        vals = np.arange(n, 0, -1, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Move a mass `delta` from index i to index j.
    Updates both the vector `a` and its full self‑convolution `conv` in O(n) time.
    """
    if delta <= 0 or i == j:
        return
    n = a.shape[0]
    # old value of a[i] before the move
    ai = a[i]
    aj = a[j]

    # contribution of the moved mass to the convolution:
    #   for each k, (a[i] a[k-i] + a[k-i] a[i]) changes linearly.
    # We use the formula from the original implementation.
    conv[i : i + n] -= 2.0 * delta * a
    conv[j : j + n] += 2.0 * delta * a
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta

    a[i] = ai - delta
    a[j] = aj + delta

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine combines a broad initial sweep, a linear‑programming mixing
    step, Adam‑accelerated mirror descent on a soft‑max approximation of the
    max, and a final O(n) hill‑climber.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0     # tiny safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # --------------------------------------------------------------
    # 1. Scan a collection of lengths & families
    lengths = [800, 1500, 2500, 3500, 5000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # Analytic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential_centered(n, lam=0.004))
        candidates.append(_distribution_exponential_centered(n, lam=0.02))
        candidates.append(_distribution_exponential_decay(n, lam=0.001))
        candidates.append(_distribution_exponential_decay(n, lam=0.01))
        candidates.append(_distribution_linear_ramp(n, up=True))
        candidates.append(_distribution_linear_ramp(n, up=False))

        # Asymmetric Beta families
        for p, q in ((0.3, 0.8), (0.8, 0.3), (0.5, 2.0), (2.0, 0.5), (0.2, 0.2)):
            candidates.append(_distribution_beta(n, p, q))

        # Random Dirichlet draws (various concentrations)
        for alpha in (0.2, 0.4, 0.8, 1.5, 3.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            candidates.append(vec)

        # Sparse random start (few spikes)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # cheap evaluation of the pool
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} new init best {sc:.6f}")

    # --------------------------------------------------------------
    # 2. If an old solution exists, try it as a warm start
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. LP‑based mixing step (only on moderate size to keep memory OK)
    if best_seq is not None and best_n <= 3000:
        a = best_seq.copy()
        n = a.shape[0]
        # Ensure every entry is strictly positive (otherwise the LP is unbounded)
        a = np.maximum(a, eps)
        a /= a.sum()

        # We will perform at most 5 mixing rounds
        mix_rounds = 5
        for rnd in range(mix_rounds):
            if time.time() >= deadline:
                break
            conv = _fft_self_convolve(a)
            M = conv.max()
            # Build convolution matrix A where A @ g = a * g
            col = np.concatenate([a, np.zeros(n - 1)])
            row = np.concatenate([a[:1], np.zeros(n - 1)])
            A = toeplitz(col, row)                     # shape (2n‑1, n)

            # Linear program: maximise Σ g_i  <=> minimise -Σ g_i
            c = -np.ones(n)
            b_ub = M * np.ones(2 * n - 1)
            bounds = [(0.0, None)] * n

            # Solve – use the high‑performance HiGHS interior‑point method
            try:
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True})
                if res.success and res.x is not None:
                    g = res.x
                    # Normalise g to have unit sum
                    if g.sum() > 0:
                        g = g / g.sum()
                    else:
                        continue
                    # Convex combination
                    t = 0.15
                    a = (1 - t) * a + t * g
                    a = _project_onto_simplex(a, 1.0)
                    # re‑evaluate
                    new_sc = _max_convolution_score(a)
                    if new_sc < best_score:
                        best_score = new_sc
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] after LP round {rnd+1} new best {new_sc:.6f}")
                else:
                    # LP failed – fall back to the previous vector
                    pass
            except Exception:
                # In case SciPy is not available or numerical trouble occurs
                pass

    # --------------------------------------------------------------
    # 4. Soft‑max mirror‑descent with Adam (focuses on the max entry)
    a = best_seq.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    total_iters = 12000
    for it in range(1, total_iters + 1):
        if time.time() >= deadline:
            break
        # geometric schedule for the soft‑max temperature
        alpha = 1.0 * (50000.0 / 1.0) ** (it / total_iters)   # grows from 1 to 5e4
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth maximum
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update on the natural parameter (exponentiated‑gradient style)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # learning‑rate schedule (decreases slowly)
        lr = 0.25 / (1 + it / 2000.0)

        a = a * np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps_adam))
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if it % 1000 == 0 or it == total_iters:
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score = cur_sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam iter={it} new best {cur_sc:.6f}")

    # --------------------------------------------------------------
    # 5. Final O(n) hill‑climber (mass transfer moves)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # locate a maximal entry of the convolution
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair (i,j) that contributes most to this entry
        best_i, best_j = i_start, k - i_start
        best_pair = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_pair:
                best_pair = prod
                best_i, best_j = i, j

        # move a small chunk from the larger of the two to a random other index
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.4, 0.02)          # keep a non‑negative
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a_snapshot = a.copy()
        conv_snapshot = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[{time.time():.1f}] hill iter={it} new best {best_score:.6f}")
        else:
            a = a_snapshot
            conv = conv_snapshot

        # occasional jitter to shake out plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {it}")

    final_sc = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_sc:.6f} (length={n})")
    return best_seq.tolist()
