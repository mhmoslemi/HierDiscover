"""
Propose a step‑function sequence that (approximately) minimises
`evaluate_sequence`.  The algorithm works in three stages:

1. **Broad initialisation** – a cheap sweep over many analytic families
   (uniform, beta, sine‑power, cosine‑window, power‑law, exponential,
   sparse Dirichlet draws) for a handful of lengths.  The best candidate
   (normalised to sum = 1) seeds the optimisation.

2. **Linear‑programming refinement** – given a current vector `a`,
   we compute `M = max(a*a)`.  The linear programme

       maximise   Σ g_i
       subject to  (a * g)_k ≤ M   for all k,
                   g_i ≥ 0

   is an exact implementation of the “LP‑step” described in the reference
   paper.  The solution `g` is mixed with `a` (`a ← (1‑t)a + t g`,
   `t≈0.3`) and projected back onto the probability simplex.  A few
   iterations (while time permits) dramatically lower the peak of the
   self‑convolution.

3. **Fine‑tuning** – a short sub‑gradient descent, a diffusion smoothing
   pass and a simulated‑annealing hill‑climber (mass‑transfer moves with
   O(n) convolution updates) finish the search.

All steps respect the global time budget (default 1000 s).  The returned
list has non‑negative entries and is scaled to sum = 1 (the scaling does
not affect the final score).
"""
import time

import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Score = 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families – each returns a vector of length n with sum = 1.
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _lp_optimize(a, M):
    """
    Solve
        maximise Σ g_i
        subject to (a * g)_k ≤ M  (k = 0 … 2n‑2)
                    g_i ≥ 0
    Returns the optimal g (as a NumPy array) or None on failure.
    """
    n = a.shape[0]
    # Build sparse Toeplitz matrix of shape (2n‑1, n) with constant diagonals a[i]
    diags = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    A = sparse.diags(diags, offsets=range(n), shape=(2 * n - 1, n), format="csc")
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i  <=> minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # tiny safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Wide scan of analytic families (few lengths, keeps time modest).
    lengths = [1500, 2000, 2500, 3000, 3500]
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.003),
            _distribution_exponential(n, lam=0.02),
        ]

        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))

        for c in (0.5, 0.46, 0.42):
            candidates.append(_distribution_cosine_window(n, c))

        for p in (0.5, 1.0, 1.5, 2.0):
            candidates.append(_distribution_power_law(n, p))

        # sparse random seed
        sparse_vec = np.zeros(n, dtype=np.float64)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse_vec[idx] = np.random.rand(k)
        sparse_vec /= sparse_vec.sum()
        candidates.append(sparse_vec)

        # a few Dirichlet draws (shape 0.5 → fairly sparse)
        for _ in range(3):
            x = np.random.gamma(0.5, 1.0, size=n)
            x /= x.sum()
            candidates.append(x)

        # evaluate all candidates
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Bring in the previous global solution if it exists.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # Fallback (should never happen)
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Linear‑programming refinement loop.
    a = best_seq.copy()
    n = a.shape[0]
    lp_iters = 0
    max_lp_iters = 25                     # enough for a noticeable drop
    while time.time() < deadline and lp_iters < max_lp_iters:
        lp_iters += 1
        conv = _fft_self_convolve(a)
        M = conv.max()
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LP iter={lp_iters} new best {cur_score:.6f}")

        # solve LP: maximise Σ g_i  subject to (a * g) ≤ M
        g = _lp_optimize(a, M)
        if g is None:
            print(f"[{time.time():.1f}] LP failed at iter {lp_iters}")
            break

        # normalise g to a probability vector
        g = np.maximum(g, eps)
        g /= g.sum()

        # convex combination (t controls step size – a modest 0.3 works well)
        t = 0.3
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, eps)
        a /= a.sum()
        # loop continues; the score will be re‑computed at the top

    # ------------------------------------------------------------------
    # 4. Exact max‑sub‑gradient descent (few quick iterations).
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur_score = 2.0 * n * _fft_self_convolve(a).max()

            a = np.convolve(a, kernel, mode="same")
            a = np.maximum(a, eps)
            a /= a.sum()
        cur_score = 2.0 * n * _fft_self_convolve(a).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Diffusion new best {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing hill‑climber on the full convolution.
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_seq = a.copy()

    it = 0
    T = cur_score * 0.05                    # start temperature ≈5 % of current score
    decay = 0.9997                           # slower cooling for a finer search
    max_delta_factor = 0.02                 # at most 2 % of source mass per move

    while time.time() < deadline:
        it += 1
        # pick source proportionally to its mass (higher mass ⇒ more impact)
        source = np.random.choice(n, p=a)
        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * max_delta_factor, 0.05)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()

        # O(n) incremental update of the full self‑convolution
        n2 = 2 * n - 1
        a_old = a_snap
        conv[source : source + n] -= 2.0 * delta * a_old
        conv[target : target + n] += 2.0 * delta * a_old
        conv[2 * source] += delta * delta
        conv[2 * target] += delta * delta
        a[source] -= delta
        a[target] += delta

        new_score = 2.0 * n * conv.max()

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap

        T *= decay

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
