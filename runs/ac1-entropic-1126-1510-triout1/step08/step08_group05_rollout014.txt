"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The algorithm consists of three stages:

1. **Initial guess** – a heavy‑tailed beta distribution (p≈0.2) or the
   previously best known vector (if the global variable ``height_sequence_1``
   exists).  The vector is normalised to L¹‑norm 1.

2. **Linear‑programming refinement** – for the current vector *a* we solve
   the packing LP

       max   1ᵀ g
       s.t.  A(a) g ≤ M,
             g ≥ 0 ,

   where ``M = max(a*a)`` and ``A(a)`` is the convolution matrix
   (row k contains the coefficients of ``a`` that multiply ``g`` to form
   ``(a*g)[k]``).  ``g`` is then rescaled to sum 1 and mixed with *a*.
   This step reduces the peak of the self‑convolution dramatically and is
   repeated a few times (the LP is solved with SciPy’s HiGHS interior‑point
   solver).

3. **Short simulated‑annealing (SA) hill‑climber** – starting from the best
   vector obtained above we perform a fast O(n) mass‑transfer SA that works
   directly on the self‑convolution.  The move set consists of moving a
   small amount of mass from a source index to a target index while updating
   the convolution incrementally.  The temperature decays exponentially.

The routine respects the given time budget (default 1000 s) and returns the
best sequence found as a plain list of floats.
"""

import time
import math
import random
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta‑symmetric density proportional to (x·(1−x))^{p−1} on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Transfer a mass `delta` from index i to index j, updating the
    vector `a` and its full self‑convolution `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_i_old = a[i]
    a_j_old = a[j]

    # quadratic self‑terms
    conv[2 * i] -= delta * delta
    conv[2 * j] += delta * delta

    # linear part (the cross terms)
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a

    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n convolution matrix A such that (A @ g)[k] = (a*g)[k].
    Each row k contains a shifted copy of the vector a.
    """
    n = a.shape[0]
    # row indices are i + j, column indices are j, data = a_i
    idx = np.arange(n, dtype=np.int64)
    rows = (idx[:, None] + idx[None, :]).ravel()
    cols = np.tile(idx, n)
    data = np.repeat(a, n)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (approximately) minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence (as a plain list of floats) found within that budget.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 4.0   # safety margin

    eps = 1e-12

    # --------------------------------------------------------------
    # 1. Initial candidate – heavy‑tailed beta distribution (p≈0.2)
    n = 3000                # a size that balances quality and LP tractability
    a = _distribution_beta_sym(n, p=0.2)
    a = np.maximum(a, eps)
    a /= a.sum()

    # incorporate a previously known solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            # a simple averaging of the two candidates
            a = 0.5 * a + 0.5 * prev
            a /= a.sum()
    except Exception:
        pass

    best_seq = a.copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] init length={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Linear‑programming refinement (a few iterations)

                if np.random.rand() < prob:

                it_no_improve = 0

            best_seq = best_seq_sa.copy()
