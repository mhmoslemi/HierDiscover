"""
Propose a low‑score non‑negative step function.

The search consists of four phases:

1️⃣  Broad generation: many analytic families (arcsine, beta,
    exponential, power‑law, etc.) and random Dirichlet draws are
    created for a *base* length.  All candidates are ℓ₁‑normalised
    (∑a=1) and scored with the cheap surrogate
    ``2·n·max(conv(a,a))`` which equals ``evaluate_sequence`` for a simplex
    point.

2️⃣  Intensive refinement: every top‑k seed is polished by two
    first‑order methods – Adam with Euclidean projection and
    exponentiated‑gradient (KL‑mirror) descent – both run until the
    global deadline.  The best point seen is kept.

3️⃣  Mass‑transfer (Frank‑Wolfe) polishing: repeatedly move mass from
    the component with the largest sub‑gradient to the one with the
    smallest, using a full‑transfer line‑search.  This aggressively
    reduces the max‑convolution value.

4️⃣  Upscaling & final annealing: the current best distribution is
    linearly interpolated to several larger lengths (up to ~25 k) and
    each scaled version is refined again with a short Adam run.
    Finally a light simulated‑annealing loop performs random pairwise
    transfers, accepting improvements or Metropolis steps.

The routine respects the 1000 s budget and returns the best list of
non‑negative floats found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Cheap surrogate for ``evaluate_sequence`` when sum(a)=1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 300_000):
    """Adam‑style descent with Euclidean projection onto the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a -= eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 250_000):
    """Exponentiated‑gradient descent (KL‑mirror)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _mass_transfer_step(a: np.ndarray, deadline: float,
                       max_iters: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe‐style mass transfer with an exact line‑search (full or half
    transfer).  Returns the best vector found and its surrogate score.
    """
    cur_score = _max_convolution_score(a)
    best_a, best_score = a.copy(), cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        s_full = _max_convolution_score(a_full)

        if s_full < cur_score - 1e-13:
            a, cur_score = a_full, s_full
        else:
            # half transfer as a cheap back‑off
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            s_half = _max_convolution_score(a_half)
            if s_half < cur_score - 1e-13:
                a, cur_score = a_half, s_half
            else:
                # occasional jitter to avoid stagnation
                if it % 5000 == 0 and np.random.rand() < 0.12:
                    jitter = np.random.randn(a.size) * 2e-5
                    a = _project_onto_simplex(a + jitter, s=1.0)
                    cur_score = _max_convolution_score(a)

        if cur_score < best_score - 1e-14:
            best_a, best_score = a.copy(), cur_score
    return best_a, best_score


def _annealing_phase(a: np.ndarray, deadline: float,
                     max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing: random pairwise transfers with a
    decreasing temperature schedule.
    """
    cur_score = _max_convolution_score(a)
    best_a, best_score = a.copy(), cur_score
    t0, t1 = 0.025, 0.001
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # linear schedule based on remaining time
        remaining = deadline - time.time()
        total = max(1e-9, deadline - (deadline - max_iters * 0.0))
        frac = remaining / (total if total > 0 else 1)
        temp = t1 + (t0 - t1) * frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1

        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        s_new = _max_convolution_score(a_new)

        if s_new < cur_score - 1e-15:
            a, cur_score = a_new, s_new
            if cur_score < best_score - 1e-14:
                best_a, best_score = a.copy(), cur_score
        else:
            # Metropolis acceptance
            prob = np.exp(-(s_new - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a, cur_score = a_new, s_new
    return best_a, best_score

def propose_candidate(seed: int = 43, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s budget and
    returns the best sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 0️⃣ Determine a sensible base length (use previous best if present)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # ------------------------------------------------------------------
    # 1️⃣ Choose a set of lengths to explore
    lengths = set()
    for factor in (0.8, 0.95, 1.0, 1.1, 1.3):
        n = int(base_n * factor)
        if 2000 <= n <= 25000:
            lengths.add(n)
    # add a few random lengths
    for _ in range(8):
        n = np.random.randint(2000, min(30000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring {len(lengths)} lengths (base≈{base_n})")

    # ------------------------------------------------------------------
    # 2️⃣ Broad generation of candidates (analytic families + random draws)
    candidates = []   # (array, surrogate_score)

    # parameter grids
    beta_ps = np.concatenate([
        np.arange(0.02, 0.21, 0.02),
        np.arange(0.25, 0.71, 0.05),
        np.arange(0.80, 1.51, 0.10),
        np.array([2.0])
    ])
    beta_asym_grid = [(0.2, 1.0), (0.5, 0.5), (0.8, 0.3), (0.3, 0.7), (1.5, 0.5)]
    power_ps = [0.4, 0.6, 0.8, 1.0, 1.3, 1.6, 2.0]
    exponential_lams = [0.0005, 0.001, 0.005, 0.01, 0.02]
    dirichlet_alphas = [0.05, 0.2, 0.5, 1.0, 2.0, 5.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families -------------------------------------------------
        for gen, name in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exp0.005"),
        ):
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} {name:12s}")

        # exponentials with a few lambdas
        for lam in exponential_lams:
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} exp(lam={lam})")

        # symmetric beta
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # asymmetric beta
        for a_alpha, a_beta in beta_asym_grid:
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _max_convolution_score(a)))

        # power‑law
        for p in power_ps:
            a = _power_law_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws
        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 3️⃣ Add the previous best sequence (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4️⃣ Keep the top‑k seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")
    best_seq, best_score = top_seeds[0][0].copy(), top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Intensive refinement of each seed
    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Adam descent
        seq_adam, sc_adam = _adam_descent(seed_seq, deadline, max_iters=400_000)
        if sc_adam < best_score - 1e-14:
            best_seq, best_score = seq_adam.copy(), sc_adam
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam improved to {best_score:.6f}")

        # Exponentiated‑gradient descent
        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, deadline, max_iters=300_000)
        if sc_eg < best_score - 1e-14:
            best_seq, best_score = seq_eg.copy(), sc_eg
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Frank‑Wolfe style mass‑transfer polishing
    a = best_seq.copy()
    a, best_score = _mass_transfer_step(a, deadline, max_iters=70_000)
    best_seq = a.copy()
    print(f"[{time.time():.1f}] After mass‑transfer best surrogate = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣ Upscaling to larger lengths and quick refinement
    upscale_factors = [1.5, 2.0, 2.5]
    for factor in upscale_factors:
        if time.time() >= deadline:
            break
        new_n = int(best_seq.size * factor)
        if new_n > 30000:
            continue
        # linear interpolation to new size
        x_old = np.linspace(0, 1, best_seq.size)
        x_new = np.linspace(0, 1, new_n)
        a_up = np.interp(x_new, x_old, best_seq)
        a_up = _project_onto_simplex(a_up, s=1.0)

        # short Adam run on the upscaled vector
        up_seq, up_score = _adam_descent(a_up, deadline, max_iters=150_000)
        if up_score < best_score - 1e-14:
            best_seq, best_score = up_seq.copy(), up_score
            print(f"[{time.time():.1f}] Upscaled (×{factor:.1f}) improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8️⃣ Final simulated‑annealing polishing
    a = best_seq.copy()
    a, final_score = _annealing_phase(a, deadline, max_iters=120_000)
    if final_score < best_score - 1e-14:
        best_seq, best_score = a.copy(), final_score
        print(f"[{time.time():.1f}] Final annealing improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Finished. Best surrogate score = {best_score:.6f} (length={len(best_seq)})")
    # Return plain Python list – evaluate_sequence will clip/normalise if needed
    return best_seq.tolist()
