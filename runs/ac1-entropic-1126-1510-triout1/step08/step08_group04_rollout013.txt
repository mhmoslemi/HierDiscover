def propose_candidate(seed=0, budget_s=1000):
    import time, random, numpy as np, math
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    def _norm(v):
        v = np.maximum(v, 1e-12)
        s = v.sum()
        if s < 1e-12:
            return v
        v = v / s
        return v

    best_seq, best_score = None, float('inf')
    best_n = None

    # candidate lengths
    lengths = [2000, 4000, 8000, 12000]  # maybe also 16000 ...
    for n in lengths:
        if time.time() >= deadline:
            break

        # generate candidates
        cand = []

        cand.append(_distribution_uniform(n))
        cand.append(_distribution_arcsine(n))
        cand.append(_distribution_triangular(n))
        cand.append(_distribution_exponential(n, lam=0.005))
        cand.append(_distribution_gaussian(n, sigma=0.3))

        # reversed of some families
        cand.append(_distribution_uniform(n)[::-1])
        cand.append(_distribution_exponential(n, lam=0.005)[::-1])

        for p in (0.4,0.8,1.5,3.0):
            cand.append(_distribution_beta_sym(n, p))
            cand.append(_distribution_beta_sym(n, p)[::-1])

        for power in (0.5, 1.0, 2.0, 4.0):
            cand.append(_distribution_sine_power(n, power))
            cand.append(_distribution_sine_power(n, power)[::-1])

        # random Dirichlet
        for alpha in (0.5, 1.0, 2.0):
            cand.append(np.random.dirichlet(np.full(n, alpha)))
        # random sparse
        sparse = np.zeros(n)
        k = max(2, n // 300)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        cand.append(sparse)

        for vec in cand:
            if time.time() >= deadline:
                break
            vec = _norm(vec)
            sc = 2.0 * n * _fft_self_convolve(vec).max()
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} init best {sc:.6f}")
    # also try previous sequence if available
    # ... (similar as before)

    # define inner functions for SA
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * best_M

    # SA parameters
    T0 = 0.5 * M  # initial temperature relative to magnitude
    T_end = 1e-6
    max_iters = 2000000  # up to something
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # compute active indices (where conv close to M)
        eps_rel = 1e-12 * (abs(M)+1.0)

        for k in active:

            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            # for each i, j = k - i

            g[idx_i] += a[j]  # broadcast
        # Now g contains gradient (subgradient) of each variable for active constraints.

        # source selection: choose i with probability proportional to g
        g_sum = g.sum()
        if g_sum <= 0:

        else:
            # random weighted selection
            r = random.random() * g_sum
            cum = 0.0
            for i_idx, val in enumerate(g):
                cum += val
                if r <= cum:
                    source = i_idx
                    break
        # target random
        target = random.randint(0, n - 1)
        if target == source:

        # delta magnitude
        # Choose delta relative to source mass
        max_delta = min(a[source] * 0.5, 0.05)  # 5% absolute limit
        if max_delta <= 0:

        delta = max_delta * random.random()
        if delta <= 1e-15:

        # apply transfer

            # M stays same

        if time.time() + 0.5 > deadline:

    a = best_a.copy()

        # choose source from indices that contribute to this max (i where a_i > 0)

        i_max = min(n - 1, k)
        # pick source with largest a_i * a_{k-i}? Actually choose whichever has larger a_i.
        # Let's compute products
        idx_i = np.arange(i_min, i_max + 1)
        j = k - idx_i
        products = a[idx_i] * a[j]
        # find pair with max product
        max_idx = np.argmax(products)
        source = idx_i[max_idx] if a[idx_i[max_idx]] >= a[j[max_idx]] else j[max_idx]
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.03)
        delta = max_delta * random.random()
        if delta <= 1e-15:
            continue
        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_a = a.copy()
            if it2 % 500 == 0:
                print(f"[{time.time():.1f}] final hill iter={it2} best {best_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap

        if it2 % 5000 == 0:
            # small jitter
            jitter = np.random.randn(n) * 1e-6
            a += jitter
            a = _norm(a)
            conv = _fft_self_convolve(a)
            M = conv.max()
    # final return
    final_score = 2.0 * n * M
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_a.tolist()

def simulated_annealing(a, max_iter, time_deadline):
    n = len(a)
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M
    best_score = 2.0 * n * best_M
    T0 = M * 0.5
    T_end = M * 1e-6
    it = 0
    while time.time() < time_deadline and it < max_iter:
        it += 1
        # compute source based on max product pair
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # compute product of pairs
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = k - idx_i
        # vectorized product

"""
Propose a step‑function (non‑negative vector) that attains a low value of

    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)^2)

The algorithm works in three stages:

1. **Broad initialization** – generate many analytic families (uniform,
   arcsine, triangular, exponential, Beta, sine‑power, Gaussian) for a few
   candidate lengths, also add random Dirichlet draws and a few sparse starts.
   The best among them (with the sum normalised to 1) is kept.

2. **Simulated‑annealing mass‑transfer** – starting from the best vector,
   repeatedly move a small amount of mass from an index that contributes most
   to the current maximal convolution entry to a (mostly) unrelated index.
   Moves that increase the maximal entry are accepted with a temperature‑
   dependent probability, allowing the search to escape local minima.
   The temperature is cooled geometrically.

3. **Deterministic hill‑climbing polish** – after the annealing phase a cheap
   O(n) hill‑climber (identical to the one used previously) is run for a few
   thousand iterations to polish the solution.

All vectors are kept on the simplex (sum = 1) and clipped to stay within the
security limits of ``evaluate_sequence``.  The routine respects a 1000 s
wall‑clock budget and prints intermediate scores.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # ordered pairs (i,·) and (·,i) are removed
    conv[i:i + n] -= 2.0 * delta * a_old
    # ordered pairs (j,·) and (·,j) are added
    conv[j:j + n] += 2.0 * delta * a_old
    # diagonal entries (i,i) and (j,j) need the quadratic term
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _normalise(v):
    """Clip tiny negatives, enforce positivity and normalise to sum = 1."""
    eps = 1e-12
    v = np.maximum(v, eps)
    s = v.sum()
    if s < eps:
        return v  # should never happen
    return v / s

    The wall‑clock limit is respected.
    """

    candidate_lengths = [2000, 4000, 8000, 12000]

        # Analytic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.004))
        candidates.append(_distribution_exponential(n, lam=0.02))
        candidates.append(_distribution_gaussian(n, sigma=0.3))
        candidates.append(_distribution_cosine_window(n, coeff=0.4))
