def _adam_optimize_softmax(a0, deadline, betas=(0.9,0.999), step0=0.15, schedule=None, iters_per=50000):
    if schedule is None:
        schedule = [2.0, 5.0, 10.0, 20.0, 40.0]
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    for beta in schedule:
        for _ in range(iters_per):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            # compute self-conv
            b = _fft_self_convolve(a)
            # compute softmax weights
            maxb = b.max()
            exp_terms = np.exp(beta * (b - maxb))
            w = exp_terms / exp_terms.sum()
            # gradient
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n-1:n-1+n]
            # Adam update
            m = betas[0] * m + (1 - betas[0]) * grad
            v = betas[1] * v + (1 - betas[1]) * (grad * grad)
            m_hat = m / (1 - betas[0]**t)
            v_hat = v / (1 - betas[1]**t)
            eta = step0 / (np.sqrt(v_hat) + 1e-8)
            a = a - eta * m_hat
            a = _project_onto_simplex(a, s=1.0)
            # evaluate per few iterations
            if t % 5000 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
    # final check
    cur_score = _max_convolution_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score
```

"""
Robust step‑function optimiser (≈1000 s budget).

The optimiser builds a diverse set of initial probability vectors on several
lengths, then refines the most promising ones with a specialised Adam
optimiser that minimises a smooth approximation of the true objective
    f(a) = 2·n·max(a∗a) .
The smooth surrogate is the soft‑max of the convolution entries:
    f_β(a) = 2·n·log‑sum‑exp(β·(a∗a))/β  (β → ∞ ⇒ max).

A schedule (β = 2, 5, 10, 20, 40) is used, each stage performing many
projected Adam steps on the simplex.  The best vector found is
polished with a classic Frank‑Wolfe‑style pairwise mass transfer loop and,
if time permits, a short linear‑programming boost.

All operations are carried out with FFT‑based convolutions for speed.
The final sequence is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (sum(a)=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for the even active indices (where the term a_i*a_i appears only once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
# analytic families ----------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# optimisation primitives ----------------------------------------------
def _adam_optimize_softmax(a0: np.ndarray,
                           deadline: float,
                           betas: tuple[float, float] = (0.9, 0.999),
                           step0: float = 0.15,
                           schedule: list[float] | None = None,
                           iters_per_beta: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate
        f_β(a) = 2·n·log‑sum‑exp(β·(a∗a))/β.
    The schedule gradually increases β, turning the surrogate into the true
    max objective.
    """
    if schedule is None:
        schedule = [2.0, 5.0, 10.0, 20.0, 40.0]

    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    for beta in schedule:
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            # self‑convolution
            b = _fft_self_convolve(a)

            # soft‑max weights
            maxb = b.max()
            exp_terms = np.exp(beta * (b - maxb))
            w = exp_terms / exp_terms.sum()

            # gradient of the surrogate
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # Adam update
            m = betas[0] * m + (1.0 - betas[0]) * grad
            v = betas[1] * v + (1.0 - betas[1]) * (grad * grad)
            m_hat = m / (1.0 - betas[0] ** t)
            v_hat = v / (1.0 - betas[1] ** t)
            eta = step0 / (np.sqrt(v_hat) + 1e-8)
            a = a - eta * m_hat
            a = _project_onto_simplex(a, s=1.0)

            # occasional exact evaluation
            if t % 5_000 == 0:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
    # final check
    cur_score = _max_convolution_score(a)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent on the true max objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _lp_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """One‑shot linear‑programming improvement (used only for modest n)."""
    n = a.shape[0]
    if n > 2000:                     # keep the LP cheap
        return a, _max_convolution_score(a)

    # current bound
    cur_max = _subgrad(a)[1]

    # build constraint matrix for (a * g) ≤ cur_max
    from scipy.sparse import csr_matrix
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, cur_max, dtype=np.float64)

    # maximise ∑ g_j  → minimise −∑ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    from scipy.optimize import linprog
    remaining = max(0.0, deadline - time.time())
    res = linprog(c,
                  A_ub=A, b_ub=b,
                  bounds=bounds,
                  method='highs',
                  options={'time_limit': remaining, 'presolve': True})
    if not res.success:
        return a, _max_convolution_score(a)

    g = res.x
    if g.sum() > 0:
        g = g / g.sum()
    else:
        g = a.copy()

    # blend a little bit of g and polish
    t = 0.2
    a_new = (1 - t) * a + t * g
    a_new = _project_onto_simplex(a_new, s=1.0)
    a_ref, sc_ref = _sgd_subgrad(a_new, deadline, max_iters=60_000, step0=0.4)
    return a_ref, sc_ref
