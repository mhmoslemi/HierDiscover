"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Seed generation** – Build a diverse pool of candidate probability vectors
   (arcsine, very‑U‑shaped beta, asymmetric beta, uniform and a handful of random
   Dirichlet draws) for several lengths (≈ 3 k … 12 k).  All candidates are
   normalised to the simplex and scored with the cheap surrogate
   ``2·n·max(a*a)`` (identical to the true objective for a sum of 1).

2. **Top‑k selection** – Keep the cheapest few seeds.

3. **Continuation L‑BFGS** – Optimise each seed on a *smoothed* version of the
   max‑convolution using the soft‑max (log‑sum‑exp) with a growing smoothing
   parameter γ.  The optimisation is performed in the unconstrained **log‑softmax**
   space  ``y`` where ``a = softmax(y)`` guarantees non‑negativity and the
   sum‑to‑one constraint.  For each γ we run a limited‑budget L‑BFGS
   (max ≈ 2000 function evaluations).  The schedule used is
   ``γ = [5, 20, 60, 200]`` which gradually turns the surrogate into the true
   max–operator.

4. **Fine‑grained hill‑climbing** – A short stochastic “mass‑transfer”
   refinement that accepts only improving moves; it can escape the final
   L‑BFGS basin and fine‑tune the solution.

All steps obey a global 1000 s budget; the best sequence found is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, n: int, gamma: float):
    """Objective (soft‑max of a*a) and gradient w.r.t. y (log‑softmax coordinates)."""
    a = np.exp(y)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= a_sum

    b = _fft_self_convolve(a)                 # length 2n‑1
    b_max = b.max()
    # soft‑max weights, shifted for numerical stability
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # weighted sum = ∑ w_i b_i  (smooth approximation of max)
    weighted_sum = np.dot(w, b)
    obj = 2.0 * n * weighted_sum               # surrogate objective

    # gradient w.r.t a  (see paper for derivation)
    conv_rev = _fft_convolve(a[::-1], w)      # length 3n‑2, we need middle part
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]
    diag_w = w[2 * np.arange(n)]               # contributions where i=j
    grad_a -= a * diag_w

    # chain rule for a = softmax(y)
    dot = np.dot(grad_a, a)
    grad_y = a * (grad_a - dot)

    return obj, grad_y


# ----------------------------------------------------------------------
def _lbfgs_optimize(a0: np.ndarray, gamma: float, deadline: float,
                    maxfun: int = 2000):
    """
    L‑BFGS optimisation of the smooth surrogate for a fixed γ.
    Variables are the log‑softmax parameters y (unconstrained).
    """
    n = a0.shape[0]
    y0 = np.log(a0 + 1e-30)                     # safe log

    def fun(y):
        if time.time() > deadline:
            # abort the optimisation – signal via exception
            raise StopIteration
        obj, grad = _smooth_obj_and_grad(y, n, gamma)
        return obj, grad

    try:
        res = minimize(fun, y0, method='L-BFGS-B', jac=True,
                       options={'maxfun': maxfun, 'maxiter': maxfun,
                                'ftol': 1e-12, 'disp': False})
    except StopIteration:
        # Time ran out – fall back to the current a0
        return a0, _max_convolution_score(a0)

    if not res.success:
        # optimisation failed but we still have a point to return
        pass

    y_opt = res.x
    a_opt = np.exp(y_opt)
    a_opt /= a_opt.sum()
    return a_opt, _max_convolution_score(a_opt)


# ----------------------------------------------------------------------
def _continuation_optimize(a0: np.ndarray, deadline: float):
    """
    Run a sequence of L‑BFGS optimisations with increasing γ.
    Returns the final (a, cheap_score).
    """
    a = a0.copy()
    best_score = _max_convolution_score(a)

    # smoothing schedule – start fairly smooth and end with a very sharp max
    gammas = [5.0, 20.0, 60.0, 200.0]
    # give each stage a modest budget; the overall deadline controls the loop
    for gamma in gammas:
        if time.time() >= deadline:
            break
        a, score = _lbfgs_optimize(a, gamma, deadline, maxfun=2000)
        if score < best_score:
            best_score = score
    return a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose lengths to explore – keep them moderate so FFTs stay cheap
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    lengths = set()
    # deterministic multipliers around the current best length
    for f in (0.7, 0.9, 1.0, 1.2, 1.5, 2.0):
        n = int(base_n * f)
        if 500 <= n <= 18000:
            lengths.add(n)
    # a few random lengths for extra diversity
    for _ in range(5):
        n = np.random.randint(500, min(18000, base_n * 3) + 1)
        lengths.add(n)

    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1️⃣  Broad sampling across analytic families & random draws
    candidates = []   # (array, cheap_score)

    # extremely U‑shaped symmetric beta parameters (including very small p)
    beta_params = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine (often a strong baseline)
        a = _arcsine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))

        # uniform
        a = np.full(n, 1.0 / n)
        candidates.append((a, _max_convolution_score(a)))

        # symmetric beta with a wide range of p
        for p in beta_params:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # asymmetric beta – a few useful combos
        for p in [0.02, 0.05, 0.1]:
            for q in [0.5, 1.0, 2.0, 5.0]:
                a = _beta_asymmetric_distribution(n, p, q)
                candidates.append((a, _max_convolution_score(a)))

        # a few random Dirichlet draws
        for _ in range(2):
            a = _random_dirichlet(n)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 2️⃣  Include the previously best construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(10, len(candidates))
    top_seeds = [seq for seq, _ in candidates[:top_k]]

    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Heavy refinement via continuation L‑BFGS (time‑shared)
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        return best_seq.tolist()
    per_seed_budget = remaining_time * 0.8 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        seed_deadline = time.time() + per_seed_budget
        a_opt, sc = _continuation_optimize(seed_seq, seed_deadline)
        if sc < best_score - 1e-14:
            best_seq, best_score = a_opt.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1:02d} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Fine‑grained hill‑climbing (purely improving random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.03  # up‑to 3 % of total mass at the start

    while time.time() < deadline:
        it += 1
        # Exponential decay of the transferred amount
        transfer = max_transfer * (0.5 ** (it / 12000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it:,d} new best {best_score:.6f}")

        # occasional jitter to shake out plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
