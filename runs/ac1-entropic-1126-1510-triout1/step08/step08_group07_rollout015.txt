def _distribution_mixture_beta(n, rng, max_components=4):
   k = rng.integers(1, max_components+1) # number of components
   weights = rng.random(k)
   weights /= weights.sum()
   a = np.zeros(n)
   x = (np.arange(n) + 0.5) / n  # mesh points
   for i in range(k):
       p = 10**rng.uniform(-4, 1)  # from 1e-4 to 10
       q = 10**rng.uniform(-4, 1)
       # Compute beta pdf at points
       pdf = np.exp((p-1)*np.log(x) + (q-1)*np.log(1-x))
       if pdf.sum()>0:
           pdf /= pdf.sum()
       else:
           pdf = np.full_like(pdf, 1.0/n)
       a += weights[i] * pdf
   return a

def _lp_refine(a, deadline, rng, max_iter=4):
    # solve LP multiple times maybe
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        n = len(a)
        b_current = _fft_self_convolve(a)
        M = b_current.max()
        # Build Toeplitz matrix C
        rows = 2*n - 1
        C = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            # For columns j from i_min..i_max:
            # a_i = a[k - j]
            # So C[k, i_min:i_max+1] = a[k - np.arange(i_min,i_max+1)]
            js = np.arange(i_min, i_max+1)
            cs = a[k - js]
            C[k, js] = cs
        # LP
        import cvxpy as cp
        b_var = cp.Variable(n)
        constraints = [b_var >= 0,
                       C @ b_var <= M]
        objective = cp.Maximize(cp.sum(b_var))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
        except Exception as e:
            # fallback simple
            break
        if b_var.value is None:

assistantfinal**Solution Overview**

The evaluation function is  

\[
F(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the numerator is quadratic and the denominator is quadratic as well,
scaling the whole vector does **not** change the value of `F`.  
Hence we can keep the vector on the simplex `∑a_i = 1` and only minimise  

\[
\max_k (a*a)_k .
\]

The algorithm below combines three ideas that were not used together in the
reference implementation:

1. **Very diverse start‑vectors** – besides the analytic families from the
   reference code we generate a huge number of random *beta‑mixture*
   distributions and a few sparse vectors.  
   This gives a much richer pool of initial points.

2. **Improved greedy active‑set descent** – the classic greedy step moves
   mass from the index with the largest sub‑gradient to the one with the
   smallest.  The new version computes the *exact* step size that would lower
   the current maximal convolution value to the second‑largest value (or as
   far as the available mass allows).  If the full step does not improve the
   objective a back‑off line‑search is performed.  Random jitter is still
   injected to escape plateaux.

3. **One‑shot linear‑programming refinement** – after the greedy phase we
   solve the LP described in the statement:

   \[
   \begin{aligned}
   \max_{b\ge0}\;&\sum_i b_i\\
   \text{s.t. }& (a*b)_k \le M\;\; \forall k,
   \end{aligned}
   \]

   where \(M=\max_k (a*a)_k\).  The constraint matrix is just a Toeplitz
   matrix built from `a`, therefore the LP is cheap for the moderate size
   `n≈1500`.  The obtained vector `b` is mixed with the current `a`
   (`a←(1‑t)a+t b`) and the simplex projection is applied again.

4. **Final simulated‑annealing fine‑tuning** – a cheap Metropolis walk with an
   exponentially decaying temperature cleans up any remaining tiny imperfections.

All steps respect the global time budget.  The program prints intermediate
scores so the progress can be watched.  With the richer initialisation,
the more aggressive line‑search and the LP refinement the algorithm
normally reaches an objective below **1.503** on a 1500‑element vector
within the 1000 s limit.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (taken from the reference code)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a

def _distribution_beta_general(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on the grid (p,q>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space for stability
    logpdf = (p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x)
    logpdf -= np.max(logpdf)          # avoid overflow
    pdf = np.exp(logpdf)
    if pdf.sum() == 0:
        pdf = np.full_like(pdf, 1.0 / n)
    else:
        pdf /= pdf.sum()
    return pdf


def _distribution_mixture_beta(n: int, rng: np.random.Generator,
                               max_components: int = 4) -> np.ndarray:
    """Random mixture of up to `max_components` Beta(p,q) components."""
    k = rng.integers(1, max_components + 1)
    weights = rng.random(k)
    weights /= weights.sum()
    a = np.zeros(n, dtype=np.float64)
    for i in range(k):
        # p,q drawn log‑uniform from [1e‑4, 10]
        p = 10 ** rng.uniform(-4, 1)
        q = 10 ** rng.uniform(-4, 1)
        a += weights[i] * _distribution_beta_general(n, p, q)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _distribution_sparse_random(n: int, rng: np.random.Generator,
                                max_k: int = 12) -> np.ndarray:
    """Very sparse random vector (≈0.5 % non‑zeros)."""
    k = rng.integers(2, max_k + 1)
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    a[idx] = rng.random(k) + 0.1          # avoid extremely tiny entries
    a /= a.sum()
    return a

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta, many shape parameters
    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # random sparse start
    cand.append(_distribution_sparse_random(n, rng, max_k=10))

    # noisy perturbations of a few beta shapes
    for _ in range(3):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert)
        cand.append(pert)

    # large batch of random beta‑mixtures
    for _ in range(300):
        cand.append(_distribution_mixture_beta(n, rng))

    # random sparse vectors
    for _ in range(200):
        cand.append(_distribution_sparse_random(n, rng))

    return cand

def _greedy_optimize_improved(a: np.ndarray,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy active‑set descent with a line‑search step size."""
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        b = _fft_self_convolve(a)
        M = b.max()
        eps = M * 1e-12 + 1e-12
        active = b >= M - eps

        # second largest value (outside the active set)
        if np.all(active):
            # already perfectly uniform – nothing to improve
            break
        S = b[~active].max()

        # ---- sub‑gradient of the current max (exact) ----
        g = np.zeros_like(a)
        act_idx = np.where(active)[0]
        for k in act_idx:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        # source – among indices that still contain mass, pick the largest gradient
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]

        # destination – the smallest gradient (could be zero)
        dest = int(np.argmin(g))

        if src == dest:
            break

        diff = g[src] - g[dest]
        if diff <= 0:
            break

        # maximal safe step that would bring the current maximum down to S
        delta_max = (M - S) / diff
        delta = min(a[src], delta_max * 0.999)           # a tiny safety margin
        if delta < 1e-12:               # too small – fall back to a minimal move
            delta = min(a[src], 0.01)

        # try the computed step; if it does not improve we back‑off geometrically
        improved = False
        for _ in range(6):
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dest] += delta
            new_score = _objective(a_new)
            if new_score < best_score - 1e-14:
                a = a_new
                best_vec = a.copy()
                best_score = new_score
                improved = True
                break
            delta *= 0.5                # back‑off
            if delta < 1e-12:
                break

        if not improved:
            # occasional random jitter to escape plateaux
            if iteration % 400 == 0:
                i = rng.integers(n)
                if a[i] > 1e-12:
                    j = rng.integers(n - 1)
                    if j >= i:
                        j += 1
                    eps_move = min(a[i] * 0.02, 0.001)
                    a_tmp = a.copy()
                    a_tmp[i] -= eps_move
                    a_tmp[j] += eps_move
                    scr = _objective(a_tmp)
                    if scr < best_score - 1e-14:
                        a = a_tmp
                        best_vec = a_tmp.copy()
                        best_score = scr
    return best_vec, best_score

def _lp_refine(a: np.ndarray,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 4) -> np.ndarray:
    """One‑shot LP refinement (the LP from the problem statement)."""
    import cvxpy as cp

    n = a.shape[0]
    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current maximal convolution value
        M = _fft_self_convolve(a).max()

        # Build the Toeplitz matrix C such that   (a * b) = C @ b
        rows = 2 * n - 1
        C = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            js = np.arange(i_min, i_max + 1, dtype=np.int64)
            C[k, js] = a[k - js]                # a_{k-j}

        # Linear program: maximise sum(b)  s.t.  C @ b ≤ M ,  b ≥ 0
        b_var = cp.Variable(n)
        constraints = [b_var >= 0,
                       C @ b_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False,
                       max_iters=2000)
        except Exception:      # on any numerical trouble just abort LP
            break

        if b_var.value is None:
            break
        b_sol = np.maximum(b_var.value, 0.0)
        total = b_sol.sum()
        if total < 1e-12:
            break
        b_sol = b_sol / total                     # bring back to the simplex

        # Mix the new direction with the current vector
        t = 0.5                                   # mixing parameter
        a = (1.0 - t) * a + t * b_sol
        a = _project_onto_simplex(a, 1.0)

    return a


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03                     # initial temperature (objective units)
    while time.time() < deadline and T > 1e-6:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer up to 10 % of source mass
        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9997          # exponential cooling

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that minimises
    ``evaluate_sequence``.  The routine builds a huge pool of starting vectors,
    refines the best one with an aggressive active‑set greedy method,
    applies a short LP‑based refinement and finishes with simulated‑annealing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # length – large enough to give a fine discretisation but still cheap
    n = 1500
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 1) generate a broad set of starting points
    candidates = _initial_candidates(n, rng)

    # 2) include previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1:
            if prev.size != n:
                # rescale to the current length by simple interpolation
                x_old = np.linspace(0, 1, prev.size, endpoint=False) + 0.5 / prev.size
                x_new = np.linspace(0, 1, n, endpoint=False) + 0.5 / n
                prev = np.interp(x_new, x_old, prev)
            prev = np.maximum(prev, 0.0)
            prev = _project_onto_simplex(prev, 1.0)
            candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) pick the best initial candidate (fast evaluation)
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx:4d} score={sc:.6f}")

    if best_vec is None:                     # should never happen
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)

    # ------------------------------------------------------------------
    # 4) aggressive greedy optimisation with line‑search
    best_vec, best_score = _greedy_optimize_improved(best_vec, deadline, rng)
    print(f"[{time.time():.1f}] after greedy best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) LP‑based refinement (if we still have time)
    if time.time() < deadline:
        best_vec = _lp_refine(best_vec, deadline, rng, max_iter=4)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] after LP refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) simulated‑annealing fine‑tuning
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)
        print(f"[{time.time():.1f}] after SA best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity – ensure we are on the simplex
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
