def _distribution_beta_asym(n, p, q):
    x = (np.arange(n)+0.5) / n
    w = np.exp((p-1)*np.log(x) + (q-1)*np.log(1-x))
    if w.sum()==0:
        w = np.full(n, 1.0/n)
    else:
        w /= w.sum()
    return w

"""
Optimiser for the non‑negative step‑function problem.

The algorithm works in three phases:

1. **Rich seeding** – a wide family of analytic distributions (uniform,
   arcsine, beta (symmetric & asymmetric), exponential, power‑law, sine,
   alternating, block‑spikes, sparse random, and random mixtures) is
   generated for a modest size (≈1024).  The best few seeds are kept.

2. **Progressive up‑sampling + smoothed Adam** – the current best seeds are
   linearly interpolated to a larger grid (doubling the size each stage).
   For every new length a short Adam run on a log‑sum‑exp smooth approximation
   of the max‑convolution is performed.  The smoothing parameter α grows from
   0.5 to 5 000 000, and the schedule is repeatedly applied while time permits.
   The Adam updates are projected onto the probability simplex
   (∑a_i = 1, a_i ≥ 0).

3. **Intensive greedy refinement** – a Metropolis‑style walk that repeatedly
   transfers a random fraction of mass from a source index to a destination
   index, accepting only improvements of the *exact* objective.  The walk
   continues until the global time budget expires.

All phases respect a hard 1000 s (or the supplied ``budget_s``) deadline.
The returned list is a feasible step‑function; the provided ``evaluate_sequence``
will compute the true score.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a vector with sum == 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_mixture(
    n: int, rng: np.random.Generator, components: int = 2
) -> np.ndarray:
    """Convex combination of a few random analytic families."""
    mix = np.zeros(n, dtype=np.float64)
    for _ in range(components):
        # pick a family uniformly at random
        family = rng.integers(0, 9)
        if family == 0:
            part = _distribution_uniform(n)
        elif family == 1:
            part = _distribution_arcsine(n)
        elif family == 2:
            p = rng.uniform(0.05, 3.0)
            part = _distribution_beta_sym(n, p)
        elif family == 3:
            p = rng.uniform(0.05, 3.0)
            q = rng.uniform(0.05, 3.0)
            part = _distribution_beta_asym(n, p, q)
        elif family == 4:
            lam = rng.choice([0.001, 0.002, 0.003, 0.005, 0.01, 0.02])
            part = _distribution_exponential(n, lam)
        elif family == 5:
            gamma = rng.choice([0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4])
            part = _distribution_powerlaw_edges(n, gamma)
        elif family == 6:
            part = _distribution_sine(n)
        elif family == 7:
            stride = rng.choice([2, 3, 4])
            part = _distribution_alternating(n, stride)
        elif family == 8:
            block_frac = rng.choice([0.05, 0.10])
            part = _distribution_two_blocks(n, block_frac)
        else:
            part = _distribution_random_sparse(n, rng, density=0.02)
        weight = rng.random()
        mix += weight * part
    mix = np.maximum(mix, 0.0)
    if mix.sum() == 0:
        mix = _distribution_uniform(n)
    else:
        mix /= mix.sum()
    return mix


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of start vectors for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_sine(n))

    # symmetric beta family
    for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # asymmetric beta family
    for _ in range(6):
        p = rng.uniform(0.05, 3.0)
        q = rng.uniform(0.05, 3.0)
        cand.append(_distribution_beta_asym(n, p, q))

    # exponentials
    for lam in (0.001, 0.002, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # power‑law at the edges
    for gamma in (0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4):
        cand.append(_distribution_powerlaw_edges(n, gamma))

    # alternating patterns
    for stride in (2, 3, 4):
        cand.append(_distribution_alternating(n, stride))

    # two‑block spikes
    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random sparse start
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # few random mixtures
    for _ in range(3):
        cand.append(_distribution_mixture(n, rng, components=rng.integers(2, 4)))

    return cand

def _smooth_adam(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max‑convolution loss.
    ``a`` is assumed to sum to 1.
    """
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule of (smoothness α, number of steps, learning rate)
    schedule = [
        (0.5, 2000, 0.30),
        (2.0, 2000, 0.25),
        (10.0, 2000, 0.20),
        (50.0, 2000, 0.15),
        (200.0, 1500, 0.12),
        (1000.0, 1200, 0.09),
        (5000.0, 800, 0.07),
        (20000.0, 600, 0.05),
        (50000.0, 400, 0.04),
        (200000.0, 300, 0.03),
        (500000.0, 200, 0.025),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()

            # smooth‑max (log‑sum‑exp) weights
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0 or np.isinf(w_sum):
                # fallback – uniform weights (rare)
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of smooth‑max w.r.t. a
            conv_rev = _fft_convolve(a[::-1], w)        # length 3n‑2
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation to record improvements
            if t % 400 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.2f}] α={alpha:.0f} step={t} new best {best_score:.6f}"
                    )

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Random mass transfers accepted only if the *true* objective improves.
    The move size is up to 30 % of the donor mass.
    """
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # pick a distinct destination uniformly at random
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def _upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from ``len(vec)`` to ``new_n`` points,
    followed by renormalisation to sum 1.
    """
    old_n = vec.shape[0]
    if old_n == new_n:
        return vec.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up = _distribution_uniform(new_n)
    else:
        up /= s
    return up

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  The function returns the best vector found
    (as a plain Python list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # short safety margin

    # Lengths are doubled each stage – keep the last one as the final size.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    overall_best_seq = None
    overall_best_score = float("inf")
    # ------------------------------------------------------------------
    # 1.  Seed generation on the smallest grid
    n0 = lengths[0]
    seed_pool = _initial_candidates(n0, rng)

    # evaluate seeds (sum already = 1)
    scored = []
    for cand in seed_pool:
        sc = _objective(cand)
        scored.append((sc, cand))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(
                f"[{time.time():.2f}] initial global best n={n0} score={sc:.6f}"
            )
    # retain a few promising seeds
    scored.sort(key=lambda x: x[0])
    seeds = [seq for _, seq in scored[:5]]

    # ------------------------------------------------------------------
    # 2.  Progressive up‑sampling + Adam optimisation
    for n in lengths[1:]:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.2f}] ---- up‑sampling to length {n} ----")

        # up‑sample the current promising seeds
        new_seeds = [_upsample(s, n) for s in seeds]

        # add a few fresh random sparse starts at the new resolution
        for _ in range(3):
            new_seeds.append(_distribution_random_sparse(n, rng, density=0.02))

        # optimise each seed for a modest budget (share remaining time)
        per_seed_deadline = time.time() + (deadline - time.time()) / max(1, len(new_seeds))
        scored = []
        for s in new_seeds:
            if time.time() >= deadline:
                break
            a_opt, sc = _smooth_adam(
                s.copy(),
                per_seed_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            scored.append((sc, a_opt.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.2f}] new global best after Adam n={n} score={sc:.6f}"
                )

        # keep the best few seeds for the next level
        scored.sort(key=lambda x: x[0])
        seeds = [seq for _, seq in scored[:5]]

        # a quick greedy refinement on the current global best
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 3.  Final intensive local search (if time permits)
    if time.time() < deadline:
        print(f"[{time.time():.2f}] final greedy refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # Make sure the vector sums to 1 (numerical drift)
    overall_best_seq = np.maximum(overall_best_seq, 0.0)
    s = overall_best_seq.sum()
    if s == 0.0:
        overall_best_seq = _distribution_uniform(overall_best_seq.shape[0])
    else:
        overall_best_seq /= s

    print(
        f"[{time.time():.2f}] final best score (smooth objective) = {overall_best_score:.6f}, length={len(overall_best_seq)}"
    )
    return overall_best_seq.tolist()
