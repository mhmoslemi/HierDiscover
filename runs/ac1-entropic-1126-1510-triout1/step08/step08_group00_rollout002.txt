"""
Propose a low‑score non‑negative step function.

Algorithm
---------

The function builds a large pool of candidate distributions using many
analytic families (arcsine, triangular, exponential, several β‑distributions,
window functions such as Hann / Hamming / Blackman / Tukey, power‑law,
and random Dirichlet draws).  All candidates are projected onto the
probability simplex (∑a = 1) and scored by the cheap surrogate

    2·n·max(conv(a,a))

which is exactly the value returned by ``evaluate_sequence`` when the sum is 1.

The top‑k candidates are refined with two first‑order methods that operate
directly on the simplex:

* Adam‑style adaptive‑moment descent.
* Exponentiated‑gradient (KL‑mirror) descent.

Both methods use the analytically obtained sub‑gradient of
max(conv(a,a)) and run until the global deadline.  The best point seen
by either method is kept.

After the gradient phase we perform a few deterministic improvement steps:

* Pairwise “mass‑transfer’’ (Frank‑Wolfe style) moves that push probability
  mass from the component with the largest gradient to the one with the
  smallest.
* A small simulated‑annealing jitter phase.
* Convex‑combination line‑searches between the current best and the other
  top seeds.
* A pruning step that removes the smallest non‑zero entry if doing so
  reduces the surrogate.

Finally the best sequence (still satisfying ∑a = 1 and a_i ≥ 0) is returned.
The whole routine respects the 1000 s budget and always produces a list
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate score 2·n·max(conv(a,a)) (exact when sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 1_000_000):
    """
    Adam‑style adaptive moment descent on the simplex.
    Returns the best point and its surrogate score seen during the run.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 800_000):
    """
    Exponentiated‑gradient (KL‑mirror) descent on the simplex.
    Returns the best point and its surrogate score seen during the run.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families (all return a *normalized* probability vector)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * math.pi * i / (n - 1))
    w /= w.sum()
    return w


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * math.pi * i / (n - 1))
    w /= w.sum()
    return w


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * math.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * math.pi * i / (n - 1))
    )
    w /= w.sum()
    return w


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey (tapered cosine) window, alpha∈(0,1)."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    mask1 = i < alpha / 2
    mask2 = i > 1 - alpha / 2
    w[mask1] = 0.5 * (1 + np.cos(math.pi * (2 * i[mask1] / alpha - 1)))
    w[mask2] = 0.5 * (1 + np.cos(math.pi * (2 * (i[mask2] - 1) / alpha + 1)))
    w /= w.sum()
    return w


def _cosine_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin^p(π i/(n-1)), zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(math.pi * i / (n - 1)) ** p
    w[0] = 0.0
    w[-1] = 0.0
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w
