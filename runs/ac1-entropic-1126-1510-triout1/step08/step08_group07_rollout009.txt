"""
Propose a step‑function (non‑negative vector summing to 1) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

where conv is the ordinary discrete convolution.  The code runs within a
1000 s time budget and combines several complementary search ideas:

1.  **Rich initialisation** – many analytic families (uniform, arcsine,
    symmetric beta, exponential, alternating, block‑uniform, random
    Dirichlet, mixtures) are sampled for a fixed length `n`.  The best
    candidate (according to the fast‑computed objective) is kept.

2.  **Aggressive greedy active‑set optimisation** – at each iteration the
    maximal convolution value(s) are identified, the exact sub‑gradient
    of the max is built, and mass is transferred from the most “dangerous”
    index to the least one.  The whole transferable mass is tried first,
    then a geometric back‑off (`δ←δ/2`) until an improvement is found.
    Random jitter is added periodically to escape plateaus.

3.  **Smoothed projected gradient descent** – the non‑smooth max is
    approximated by a log‑sum‑exp with a temperature `β`.  Its gradient
    is obtained analytically (a convolution of the current vector with a
    weight derived from the soft‑max).  After each step the vector is
    projected back onto the probability simplex.  The temperature is
    increased gradually while the step size is reduced on non‑improving
    moves.

4.  **Short simulated‑annealing fine‑tuning** – a cheap Metropolis walk
    with exponential cooling performs small random mass transfers and
    keeps any improvement.

All stages respect the global deadline.  The final vector is projected
onto the simplex one last time and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT (linear, not circular)."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)^2) – works for any a."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max() / (sum_a * sum_a))

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta distribution with shape parameter p (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_len: int, start: int) -> np.ndarray:
    """Uniform mass on a contiguous block of length `block_len`."""
    a = np.zeros(n, dtype=np.float64)
    a[start:start + block_len] = 1.0
    a /= a.sum()
    return a

def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse collection of starting vectors."""
    cand = []

    # deterministic analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # random sparse start (~0.5 % non‑zeros)
    k = max(2, n // 200)
    sparse = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    sparse[idx] = rng.random(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # random Dirichlet draws
    for _ in range(8):
        d = rng.random(n) + 1e-6
        d /= d.sum()
        cand.append(d)

    # two‑block uniform families
    for _ in range(5):
        blk = rng.integers(int(0.05 * n), int(0.3 * n) + 1)
        st = rng.integers(0, n - blk + 1)
        cand.append(_distribution_two_blocks(n, blk, st))

    # mixtures of arcsine and uniform
    for _ in range(5):
        mix = rng.random()
        cand.append(mix * _distribution_arcsine(n) + (1 - mix) * _distribution_uniform(n))

    # noisy perturbations of a few beta shapes
    for _ in range(4):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, 1.0)
        cand.append(pert)

    return cand

def _greedy_optimum(a: np.ndarray, deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer based on the exact sub‑gradient of the
    active convolution shift(s).  The whole source mass is attempted; if it
    does not improve the objective a geometric back‑off (δ←δ/2) is used.
    Random jitter is injected every few thousand iterations.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # self‑convolution and active shifts
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = np.where(b >= maxb - eps)[0]

        # sub‑gradient of the max
        g = np.zeros_like(a)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        # select source (largest gradient among non‑zero entries)
        src_cand = np.where(a > 1e-12)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(g[src_cand])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        delta = a[src]          # try to move as much as possible
        improved = False

        while delta > 1e-12:
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dest] += delta
            new_score = _objective(a_new)
            if new_score < best_score - 1e-14:
                a = a_new
                best_vec = a.copy()
                best_score = new_score
                improved = True
                break
            delta *= 0.5

        if improved:
            continue

        # occasional random jitter to escape plateaus
        if it % 2000 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.02, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a_tmp.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score

def _smooth_gradient_descent(a: np.ndarray, deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of the max:
        max(b) ≈ maxb + (1/β)·log Σ exp(β·(b-maxb)).
    The gradient is obtained analytically via a convolution.
    β is slowly increased, η (step size) is reduced when the move is not
    improving.
    """
    n = a.shape[0]
    cur_score = _objective(a)
    beta = 5.0          # start with a gentle smoothing
    eta = 0.3           # initial step size
    iter_no = 0

    while time.time() < deadline:
        iter_no += 1

        b = _fft_self_convolve(a)
        maxb = b.max()
        shift = b - maxb                      # ≤ 0
        exp_vals = np.exp(beta * shift)       # safe because shift ≤ 0
        sum_exp = exp_vals.sum()
        w = exp_vals / sum_exp                 # soft‑max weights, length 2n‑1

        # gradient = 2 * (w_rev ∗ a) evaluated at indices len(w)-1‑i
        w_rev = w[::-1]
        conv_rev = _fft_convolve(w_rev, a)    # length 3n‑2
        idx_start = len(w_rev) - 1            # = 2n‑2
        grad = 2.0 * conv_rev[idx_start - np.arange(n)]

        # projected gradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, 1.0)

        new_score = _objective(a_new)

        if new_score < cur_score - 1e-14:
            a = a_new
            cur_score = new_score
        else:
            eta *= 0.6   # reduce step when not improving

        # schedule for β and η
        if iter_no % 250 == 0:
            beta = min(beta * 1.05, 200.0)   # ramp up smoothing
            eta = max(eta, 1e-6)

        if eta < 1e-7:
            break

    return a, cur_score

def _simulated_annealing(a: np.ndarray, deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑type walk that performs small random mass transfers.
    The temperature decays exponentially; any improvement is kept.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9997

    return best_a, best_score
