"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm (≈ 2 min):
   1. Generate many analytic and random seed vectors, normalised to sum = 1.
   2. Keep the N best seeds (according to the cheap proxy
      2·n·max(conv(a,a))) .
   3. Refine each seed with a cascade of optimisers:
        – Adam on the L²‑distance to the ideal *flat* convolution
        – exponentiated‑gradient on the true max‑objective
        – Adam on the max‑objective
        – continuation (soft‑max) gradient descent
        – plain sub‑gradient descent
   4. Final simulated‑annealing hill‑climb (tiny random mass transfers
      with a temperature schedule) to escape tiny plateaus.
All optimisation steps use the fast FFT‑based convolutions and respect the
simplex constraint via Euclidean projection.  The routine respects the 1000 s
budget and returns the best vector it found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap proxy = 2·n·max(a*a) (assumes ∑a = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 120_000, eta0: float = 0.9):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 120_000, step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style descent on the max‑objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 250_000, eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of conv(a,a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 180_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation: start with smooth max (γ≈5) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
        gamma *= gamma_factor
    return best_a, best_score

def _adam_l2_descent(a0: np.ndarray, deadline: float,
                     max_iters: int = 200_000,
                     step0: float = 0.5,
                     beta1: float = 0.9, beta2: float = 0.999):
    """
    Adam‑style optimisation of the L² distance between conv(a,a) and the
    flat target 1/(2·n‑1).  The loss is ``∑ (b - target)²``.
    """
    a = a0.copy()
    n = a.shape[0]
    target = 1.0 / (2 * n - 1)
    # initial loss
    b = _fft_self_convolve(a)
    loss = np.sum((b - target) ** 2)
    best_a = a.copy()
    best_loss = loss
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        b = _fft_self_convolve(a)
        diff = b - target
        grad = 2.0 * _fft_convolve(a[::-1], diff)[n - 1 : n - 1 + n]
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        loss = np.sum(( _fft_self_convolve(a) - target) ** 2)
        if loss < best_loss - 1e-18:
            best_loss, best_a = loss, a.copy()
    return best_a, best_loss

def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration parameter α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    total = x.sum()
    if total == 0:
        return np.full(n, 1.0 / n)
    return x / total


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # pick a base length – if a previous construction exists use its size,
    # otherwise fall back to a medium‑sized problem.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # explore lengths around the base size and a few random larger ones
    lengths = set()
    for f in (0.8, 1.0, 1.2, 1.5, 1.8, 2.0):
        n = int(base_n * f)
        if 500 <= n <= 12000:
            lengths.add(n)
    for _ in range(6):
        n = np.random.randint(500, min(12000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # generate a diverse pool of seed vectors
    candidates = []                     # list of (array, cheap_score)
    beta_ps = [0.1, 0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0, 12.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # analytic families (all normalised to sum=1)
        a = _arcsine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = _triangular_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = _exponential_distribution(n, lam=0.0025)
        candidates.append((a, _max_convolution_score(a)))
        a = _sine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = _cosine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))
        a = np.full(n, 1.0 / n)
        candidates.append((a, _max_convolution_score(a)))

        # symmetric Beta with many exponents
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # asymmetric Beta
        for p in beta_ps:
            for q in beta_ps:
                a = _beta_asymmetric_distribution(n, p, q)
                candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws – varied concentration
        for alpha in (1.0, 0.3, 0.1):
            a = _random_dirichlet(n, alpha=alpha)
            candidates.append((a, _max_convolution_score(a)))

        # random mixtures of two families
        for _ in range(2):
            fam1 = np.random.choice(['arcsine', 'triangular', 'uniform', 'sine'])
            fam2 = np.random.choice(['beta_sym', 'beta_asym', 'exponential'])
            if fam1 == 'arcsine':
                a1 = _arcsine_distribution(n)
            elif fam1 == 'triangular':
                a1 = _triangular_distribution(n)
            elif fam1 == 'uniform':
                a1 = np.full(n, 1.0 / n)
            else:  # sine
                a1 = _sine_distribution(n)

            if fam2 == 'beta_sym':
                p = np.random.choice(beta_ps)
                a2 = _beta_symmetric_distribution(n, p)
            elif fam2 == 'beta_asym':
                p = np.random.choice(beta_ps)
                q = np.random.choice(beta_ps)
                a2 = _beta_asymmetric_distribution(n, p, q)
            else:  # exponential
                lam = 10 ** np.random.uniform(-4, -2)
                a2 = _exponential_distribution(n, lam=lam)

            lam_mix = np.random.rand()
            a = lam_mix * a1 + (1.0 - lam_mix) * a2
            a = a / a.sum()
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # keep previous best sequence as an additional seed (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # select the most promising seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [seq for seq, _ in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # ------------------------------------------------------------------
    # heavy optimisation phase – each seed gets a fair share of the remaining time
    remaining_time = deadline - time.time()
    per_seed_budget = remaining_time * 0.8 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        seed_deadline = time.time() + per_seed_budget

        # 1) L²‑distance to a flat convolution (Adam)
        seq, _ = _adam_l2_descent(seed_seq, seed_deadline,
                                  max_iters=150_000, step0=0.6)
        cur = _max_convolution_score(seq)
        if cur < best_score - 1e-14:
            best_seq, best_score = seq.copy(), cur
            print(f"[{time.time():.1f}] seed {idx+1} L2‑Adam → {cur:.6f}")

        # 2) Exponentiated‑gradient on max‑objective
        seq, cur = _exp_gradient_descent(seed_seq, seed_deadline,
                                         max_iters=80_000, eta0=0.9)
        if cur < best_score - 1e-14:
            best_seq, best_score = seq.copy(), cur
            print(f"[{time.time():.1f}] seed {idx+1} exp‑grad → {cur:.6f}")

        # 3) Adam on max‑objective
        seq, cur = _adam_descent(seed_seq, seed_deadline,
                                 max_iters=80_000, step0=0.25)
        if cur < best_score - 1e-14:
            best_seq, best_score = seq.copy(), cur
            print(f"[{time.time():.1f}] seed {idx+1} Adam → {cur:.6f}")

        # 4) Smoothed‑max continuation
        seq, cur = _smooth_gradient_descent(seed_seq, seed_deadline,
                                            max_iters=120_000,
                                            eta0=0.7, gamma0=5.0,
                                            gamma_factor=1.001)
        if cur < best_score - 1e-14:
            best_seq, best_score = seq.copy(), cur
            print(f"[{time.time():.1f}] seed {idx+1} smoothed → {cur:.6f}")

        # 5) Plain sub‑gradient tidy‑up
        seq, cur = _subgradient_descent(seed_seq, seed_deadline,
                                        max_iters=250_000, eta0=0.5)
        if cur < best_score - 1e-14:
            best_seq, best_score = seq.copy(), cur
            print(f"[{time.time():.1f}] seed {idx+1} subgrad → {cur:.6f}")

    # ------------------------------------------------------------------
    # final simulated‑annealing hill‑climb
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    start_temp = 0.015
    while time.time() < deadline:
        it += 1
        # linear cooling
        remaining = deadline - time.time()
        T = start_temp * (remaining / (deadline - (deadline - budget_s + 5.0)))
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # propose a transfer of a small fraction of a[i]
        max_transfer = 0.02
        delta = np.random.rand() * min(max_transfer, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        diff = new_score - cur_score
        if diff < 0 or math.exp(-diff / max(T, 1e-12)) > np.random.rand():
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[{time.time():.1f}] SA iter={it} new best {best_score:.6f}")

        # occasional jitter to shake out plateaus
        if it % 7000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 8e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
