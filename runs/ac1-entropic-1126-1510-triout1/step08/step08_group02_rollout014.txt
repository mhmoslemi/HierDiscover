"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)

1. **Initial seed** – use the previous best sequence if available,
   otherwise an analytic arcsine distribution.  Normalise to sum = 1.

2. **LP‑based improvement** – for moderate sizes (n ≤ 3000) iterate:
      • compute M = max(conv(f,f))
      • solve   max Σ g_j   s.t.  f∗g ≤ M , g ≥ 0        (LP)
      • normalise g to sum = 1 and replace   f ← (1‑t) f + t g
   The linear program is built with a Toeplitz matrix; each iteration
   provably does not increase the objective and often yields a sizable
   drop.

3. **First‑order refinements** – run three mirror‑gradient style
   optimisers (exponentiated‑gradient, Adam, smoothed‑Adam) on the
   probability simplex.  All optimisers respect the time budget.

4. **Long sub‑gradient descent** – plain projected sub‑gradient descent
   with diminishing step size.

5. **Fine‑grained hill‑climbing** – repeatedly move a tiny amount of
   mass from the index with the largest sub‑gradient to the one with the
   smallest, accepting only improving moves; occasional random jitter
   helps escape plateaus.

All steps continuously record the best cheap score
``2·n·max(conv)`` (identical to ``evaluate_sequence`` when the vector is
normalised to sum = 1).  The final vector is returned as a plain Python
list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear algebra helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a∗a)  (assumes Σa = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (exact)
# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a∗a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (i = 2·k terms counted twice)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Smoothed gradient (log‑sum‑exp approximation)
# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of   eps·log Σ_i exp(conv_i/eps)   w.r.t. a."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w /= w.sum()

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad


# ----------------------------------------------------------------------
# First‑order optimiser implementations
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9):
    """Exponentiated‑gradient (mirror descent w.r.t. KL) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam optimiser on the simplex (projected after each update)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eps0: float = 1e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999):
    """Adam on the log‑sum‑exp smoothed max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # tighten smoothing occasionally
        if t % 50_000 == 0:
            eps = max(eps * 0.5, 1e-6)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.6):
    """Plain projected sub‑gradient descent with a 1/√t step rule."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families (used for an initial seed if no prior sequence)

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

    heavy_deadline = deadline - 30.0
    if time.time() < heavy_deadline:
        seq, sc = _exp_gradient_descent(best_seq, heavy_deadline,
                                        max_iters=150_000, eta0=0.9)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] exp‑grad → {sc:.6f}")

    if time.time() < heavy_deadline:
        seq, sc = _adam_descent(best_seq, heavy_deadline,
                                max_iters=150_000, step0=0.25)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] Adam → {sc:.6f}")

    if time.time() < heavy_deadline:
        seq, sc = _smooth_adam_descent(best_seq, heavy_deadline,
                                       max_iters=120_000, eps0=1e-3,
                                       step0=0.2)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] smooth‑Adam → {sc:.6f}")

    # --------------------------------------------------------------
    # 4.  Long sub‑gradient descent
    # --------------------------------------------------------------
    subgrad_deadline = deadline - 20.0
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=1_200_000, eta0=0.6)
        if sc < best_score - 1e-12:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] sub‑gradient → {sc:.6f}")

    # --------------------------------------------------------------
    # 5.  Fine‑grained hill‑climbing
    # --------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02   # initial transfer size as fraction of total mass

    while time.time() < deadline:
        iteration += 1
        # geometric decay of transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 10_000.0))

        # Guided move every few steps
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to help escape flat plateaus
        if iteration % 5_000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    # 6.  Return result
    # --------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # evaluate_sequence expects a plain Python list
    return best_seq.tolist()
