"""
Propose a non‑negative step‑function (probability vector) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Broad sampling** – generate many candidate distributions
   (analytic families, Dirichlet draws, random mixtures) over a handful
   of lengths.  All candidates are normalised to sum = 1 and scored by
   the cheap objective ``2·n·max(a*a)``.

2. **Select a seed of modest size** (≈ 800 – 1200 entries) and run a
   *linear‑programming refinement*: for the current vector ``f`` solve

        maximise   Σ b_j
        subject to 0 ≤ b,   (f * b)_k ≤ M  ∀k,
        where M = max(f*f).

   The LP solution is mixed into ``f`` (mirrored‑descent step) and the
   process repeats a few times.  This implements the “LP‑step” described
   in the problem statement and typically reduces the max‑convolution.

3. **Upsample** the refined small‑size vector to the target length
   (the size of the previously best construction) by linear interpolation.

4. **Heavy gradient‑based refinement** – run several first‑order
   optimisation schemes (exponentiated‑gradient, Adam, smooth‑Adam) on
   the upsampled vector.  The optimiser runs until the global deadline
   and always respects the simplex constraint.

5. **Long sub‑gradient descent** followed by a guided hill‑climbing
   phase transfers tiny amounts of mass from indices with large
   sub‑gradient to those with small sub‑gradient, with occasional random
   jitter.

All stages respect the 1000 s budget (a small safety margin is kept).
The returned list is a plain Python list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import sparse, optimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of the smoothed max (log‑sum‑exp)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimiser on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eps0: float = 1e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999):
    """Adam optimiser on the log‑sum‑exp smoothing of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        if t % 50_000 == 0:
            eps = eps * 0.5 if eps > 1e-7 else eps

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Analytic families -------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α (α=1 → uniform on simplex)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _upsample(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from length len(a) to new_n."""
    if a.size == new_n:
        return a.copy()
    x_old = np.arange(a.size, dtype=np.float64)
    x_new = np.linspace(0, a.size - 1, new_n, dtype=np.float64)
    return np.interp(x_new, x_old, a)


# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, deadline: float,
               max_iter: int = 12, t0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement of a probability vector.

    Repeatedly solves:
        maximise   Σ b_j
        subject to 0 ≤ b,   (f * b)_k ≤ M  ∀k,
    where M = max(f*f).  The optimizer mixes the solution back
    into ``f`` with a decreasing step size.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    a = f.copy()
    best_score = _max_convolution_score(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = _fft_self_convolve(a).max()
        # Build dense constraint matrix A where column j contains a shifted by j
        A = np.zeros((rows, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a
        c = -np.ones(n, dtype=np.float64)                # maximize sum(b)
        b_ub = np.full(rows, M, dtype=np.float64)
        bounds = [(0, None)] * n

        res = optimize.linprog(c, A_ub=A, b_ub=b_ub,
                               bounds=bounds, method='highs')
        if not res.success:
            # If LP fails we abort the refinement.
            print(f"[{time.time():.1f}] LP failed at iteration {it+1}")
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = b_opt / sum_b
        t = t0 * (0.8 ** it)            # geometric decay of mixing weight
        a_new = (1 - t) * a + t * g
        new_score = _max_convolution_score(a_new)
        if new_score < best_score - 1e-12:
            a = a_new
            best_score = new_score
            print(f"[{time.time():.1f}] LP iter {it+1}/{max_iter} score {best_score:.6f}")
        else:
            # No improvement – stop early.
            break
    return a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a probability vector (sum = 1).  The routine respects the supplied
    time budget and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Determine a base size (fallback to 3500 if no historic data).
    try:
        base_n = len(height_sequence_1)           # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # Lengths to explore – a mixture around the base size and a few random ones.
    lengths = set()
    for fct in (0.6, 0.8, 1.0, 1.2, 1.4):
        n = int(base_n * fct)
        if 500 <= n <= 12000:
            lengths.add(n)
    for _ in range(8):
        n = np.random.randint(500, min(12000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Helper to evaluate cheap score quickly
    candidates = []   # each entry: (array, cheap_score)

    beta_ps = [0.01, 0.04, 0.08, 0.15, 0.30, 0.60, 1.0, 1.5, 2.5, 4.0]
    exp_lams = [0.001, 0.004, 0.007, 0.012]
    sigma_fracs = [0.15, 0.28, 0.40]
    scale_fracs = [0.10, 0.20, 0.35]
    dirichlet_alphas = [0.05, 0.2, 0.7, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Analytic families -------------------------------------------------
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_uniform_distribution, "uniform"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
        ]
        for gen, name in fams:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Exponential families -------------------------------------------------
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Gaussian families --------------------------------------------------
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Cauchy families ----------------------------------------------------
        for sf in scale_fracs:
            a = _cauchy_distribution(n, scale_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Semicircle ---------------------------------------------------------
        a = _semicircle_distribution(n)
        candidates.append((a, _max_convolution_score(a)))

        # Symmetric beta ------------------------------------------------------
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Random Dirichlet draws ---------------------------------------------
        for alpha in dirichlet_alphas:
            for _ in range(2):
                if time.time() >= deadline:
                    break
                a = _random_dirichlet(n, alpha=alpha)
                candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # Add previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep top‑k seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 25
    top_candidates = candidates[:top_k]
    print(f"[{time.time():.1f}] Selected {len(top_candidates)} seeds; best cheap score={top_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # Choose a modest‑size seed (≤1200) for LP refinement
    small_candidates = [c for c in top_candidates if c[0].size <= 1200]
    if not small_candidates:
        # fallback: generate an arcsine of size 800
        lp_n = 800
        f_small = _arcsine_distribution(lp_n)
        lp_score = _max_convolution_score(f_small)
    else:
        f_small, lp_score = min(small_candidates, key=lambda x: x[1])
        lp_n = f_small.size

    print(f"[{time.time():.1f}] LP refinement start with n={lp_n}, score={lp_score:.6f}")

    # ------------------------------------------------------------------
    # Linear‑programming refinement on the small vector
    f_refined, refined_score = _lp_refine(f_small, deadline,
                                          max_iter=12, t0=0.35)
    print(f"[{time.time():.1f}] LP refinement finished; cheap score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # Upsample to target length (same as base_n if possible)
    target_n = max(base_n, lp_n)
    seq = _upsample(f_refined, target_n)
    seq = seq / seq.sum()                     # re‑normalise
    best_seq = seq.copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] Upsampled to n={target_n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Heavy gradient‑based refinements on the upsampled vector
    # (use the remaining time, but keep a small safety margin)
    heavy_deadline = deadline - 30.0
    if time.time() < heavy_deadline:
        # Exponentiated‑gradient
        seq1, sc1 = _exp_gradient_descent(best_seq, heavy_deadline,
                                          max_iters=150_000, eta0=0.9)
        if sc1 < best_score - 1e-14:
            best_seq, best_score = seq1, sc1
            print(f"[{time.time():.1f}] exp‑grad improved → {best_score:.6f}")

        # Adam
        seq2, sc2 = _adam_descent(best_seq, heavy_deadline,
                                  max_iters=150_000, step0=0.25)
        if sc2 < best_score - 1e-14:
            best_seq, best_score = seq2, sc2
            print(f"[{time.time():.1f}] Adam improved → {best_score:.6f}")

        # Smooth Adam
        seq3, sc3 = _smooth_adam_descent(best_seq, heavy_deadline,
                                         max_iters=120_000, eps0=1e-3, step0=0.2)
        if sc3 < best_score - 1e-14:
            best_seq, best_score = seq3, sc3
            print(f"[{time.time():.1f}] smooth‑Adam improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Long sub‑gradient descent
    subgrad_deadline = deadline - 20.0
    if time.time() < subgrad_deadline:
        seq4, sc4 = _subgradient_descent(best_seq, subgrad_deadline,
                                         max_iters=1_200_000, eta0=0.6)
        if sc4 < best_score - 1e-14:
            best_seq, best_score = seq4, sc4
            print(f"[{time.time():.1f}] sub‑gradient descent improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fine‑grained hill‑climbing (guided transfers + random jitter)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02  # up to 2 % of total mass in early swaps

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 12_000.0))

        # Occasionally use the exact sub‑gradient to pick a promising swap
        if iteration % 6 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional random jitter
        if iteration % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"[{time.time():.1f}] Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
