"""
Hybrid search for a step‑function with a small max self‑convolution.

The algorithm proceeds in three phases:
1.  A cheap grid‑search over several lengths and analytic families
    (uniform, symmetric Beta, mixtures with uniform, random Dirichlet)
    using a fast FFT based max‑convolution score.
2.  A smooth‑max mirror‑descent (exponentiated‑gradient) optimiser
    with an Adam‑style momentum schedule that drives the soft‑max
    temperature to a very large value while shrinking the learning rate.
    This refines the best candidate found by the grid search.
3.  A simulated‑annealing hill‑climber that performs O(n) mass‑transfer
    updates directly on the convolution (using an incremental update
    routine) to escape any remaining plateaus.

All work is performed on the probability simplex (∑a_i = 1, a_i ≥ 0) so
the objective 2·n·max(a*a) equals the value returned by
``evaluate_sequence`` (the denominator is 1 after normalisation).
The routine respects the 1000 s wall‑clock budget and prints progress
information.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_arcsine(n):
    """Discrete arcsine (Beta(0.5,0.5))"""
    return _distribution_beta_sym(n, 0.5)


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # contribution of removed mass
    conv[i:i + n] -= 2.0 * delta * a_old
    # contribution of added mass
    conv[j:j + n] += 2.0 * delta * a_old
    # diagonal terms (i,i) and (j,j)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

        if delta < eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        # temperature schedule – decays linearly to zero
        temp = max(1e-6, 0.5 * (1.0 - it / max_hc_iters))

        if new_M < M:
            accept = True
        else:
            # occasional uphill move
            accept = random.random() < math.exp(-(new_M - M) / temp)

        if accept:
            M = new_M
            cur_sc = 2.0 * n * M
            if cur_sc < best_score - 1e-12:
                best_score = cur_sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill it={it} new best {cur_sc:.6f}")
        else:
            a = a_snap
            conv = conv_snap

        # Periodic jitter to shake out stagnation
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
```
