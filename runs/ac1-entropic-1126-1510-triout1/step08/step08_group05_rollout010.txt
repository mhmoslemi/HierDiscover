def solve_lp(a, M):
    try:
        import scipy.sparse as sp
        from scipy.optimize import linprog
    except ImportError:
        return None

    n = len(a)
    # Build A: shape (2n-1, n)
    i_rep = np.repeat(np.arange(n), n)
    j_rep = np.tile(np.arange(n), n)
    rows = i_rep + j_rep
    data = a[i_rep]
    A = sp.coo_matrix((data, (rows, j_rep)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)   # minimize negative sum => maximize sum
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    b_opt = np.maximum(res.x, 0.0)
    return b_opt

"""
Propose a candidate step function that (approximately) minimises
``evaluate_sequence``.  The algorithm consists of:

1️⃣  A large pool of analytic families (uniform, triangular, cosine,
    Gaussian, Kaiser, arcsine, beta, power‑law, exponential, sine‑power)
    together with a few random Dirichlet draws.  The best of these cheap
    candidates seeds the optimisation.

2️⃣  Multi‑phase Adam optimisation on a smooth soft‑max approximation of the
    peak of the self‑convolution, with projection onto the probability simplex.

3️⃣  Exact sub‑gradient descent (the same update as in the reference code) to
    push the solution further.

4️⃣  If the current length is modest (≤ 1500) a **linear‑programming refinement**
    is performed: maximise the total mass of a vector `b` under the linear
    constraints ``a * b ≤ M`` (where ``*`` is discrete convolution and
    ``M`` is the current peak).  The LP is solved with SciPy’s HiGHS
    solver, the result is normalised and mixed with the current iterate.

5️⃣  A simulated‑annealing hill‑climber that works directly on the full
    self‑convolution using O(n) incremental updates.  Moves are guided by
    the active‑set sub‑gradient (mass is transferred from indices that
    increase the peak to those that decrease it), while a fraction of moves
    remains purely random to avoid trapping.

6️⃣  Optional diffusion smoothing (tiny Gaussian blur) as a final polish.

All phases respect the time budget (default 1000 s).  The function returns

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma_factor=0.25):
    """Discrete Gaussian centred in the middle of the window."""
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    x = np.arange(n, dtype=np.float64)
    vals = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_kaiser(n, beta=8.0):
    """Kaiser window (using the modified Bessel function I0)."""
    from numpy import i0
    centre = (n - 1) / 2.0
    x = np.arange(n, dtype=np.float64) - centre
    ratio = x / centre
    vals = i0(beta * np.sqrt(1 - ratio ** 2))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Incremental update of the self‑convolution (used by the SA phase)
# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_i = a[i]           # value before transfer
    a_j = a[j]           # value before transfer
    # Remove the contribution of the removed mass
    conv[i:i + n] -= 2.0 * delta * a
    # Add the contribution of the added mass
    conv[j:j + n] += 2.0 * delta * a
    # Quadratic self‑terms for the moved mass
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    # Update the vector itself
    a[i] = a_i - delta
    a[j] = a_j + delta


# ----------------------------------------------------------------------
# Linear‑programming refinement (requires SciPy)
# ----------------------------------------------------------------------
def _solve_lp_max_mass(a, M):
    """
    Solve   maximise   Σ b_j
    subject to a * b ≤ M   (convolution),   b ≥ 0.
    Returns the optimal b (may be None if the LP fails).
    """
    try:
        import scipy.sparse as sp
        from scipy.optimize import linprog
    except Exception:
        return None

    n = a.shape[0]
    # Build the convolution matrix A (shape (2n‑1, n)).
    # Row k contains a_i where i + j = k, i.e. coefficient for b_j is a_{k‑j}.
    i_rep = np.repeat(np.arange(n), n)            # i repeated n times
    j_rep = np.tile(np.arange(n), n)              # j tiled n times
    rows = i_rep + j_rep                          # convolution index k
    data = a[i_rep]                               # coefficient a_i
    A = sp.coo_matrix((data, (rows, j_rep)), shape=(2 * n - 1, n)).tocsr()

    # Linear programme: minimise –Σ b  (i.e. maximise Σ b)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return None
    b_opt = np.maximum(res.x, 0.0)
    return b_opt


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function that minimises ``evaluate_sequence``.

    The routine respects the supplied time budget and returns the best
    sequence (plain ``list[float]``) found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1. Scan a wide collection of analytic families + random draws.
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 5000, 8000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = [
            _distribution_uniform(n),
            _distribution_triangular(n),
            _distribution_cosine_window(n, coeff=0.5),
            _distribution_cosine_window(n, coeff=0.46),
            _distribution_cosine_window(n, coeff=0.42),
            _distribution_arcsine(n),
            _distribution_exponential(n, lam=0.003),
            _distribution_exponential(n, lam=0.02),
            _distribution_gaussian(n, sigma_factor=0.2),
            _distribution_gaussian(n, sigma_factor=0.35),
            _distribution_kaiser(n, beta=6.0),
            _distribution_kaiser(n, beta=10.0),
        ]

        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))
        for p in (0.5, 1.0, 1.5, 2.0, 3.0):
            candidates.append(_distribution_power_law(n, p))

        # sparse random seed (few spikes)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 250)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # few Dirichlet draws
        for _ in range(3):
            x = np.random.gamma(0.5, 1.0, size=n)
            x /= x.sum()
            candidates.append(x)

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. If a previous solution exists (global variable), use it as seed.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    if best_seq is None:
        # Fallback – shouldn't happen.
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam descent with a soft‑max (gradual sharpening).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    schedule = [
        (1.0,   2500),
        (5.0,   2500),
        (20.0,  5000),
        (100.0, 5000),
        (500.0, 4000),
        (2000.0,3000),
        (5000.0,2000)
    ]

    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_seq_adam = a.copy()
    best_score_adam = _max_convolution_score(a)

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if t % 250 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score_adam - 1e-12:
                    best_score_adam = cur
                    best_seq_adam = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

    a = best_seq_adam.copy()
    best_seq = a.copy()
    best_score = best_score_adam

    # ------------------------------------------------------------------
    # 4. Exact max‑sub‑gradient descent.
    # ------------------------------------------------------------------
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Sub‑grad it={it} new best {cur:.6f}")

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 5. Optional LP refinement (only for moderate lengths).
    # ------------------------------------------------------------------
    if n <= 1500 and time.time() < deadline:
        # Perform a few LP refinements; each solves a modest LP.
        lp_iters = 5
        for lp_it in range(lp_iters):
            if time.time() >= deadline:
                break
            conv = _fft_self_convolve(a)
            M = conv.max()
            b_opt = _solve_lp_max_mass(a, M)
            if b_opt is None:
                break
            S = b_opt.sum()
            if S <= 0:
                break
            g = b_opt / S                      # normalise to sum = 1
            # Mix a little towards g
            t_mix = 0.15
            a = (1.0 - t_mix) * a + t_mix * g
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()
            cur = _max_convolution_score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP it={lp_it} new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing hill‑climber (gradient‑guided).
    # ------------------------------------------------------------------
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    cur_score = 2.0 * n * M
    best_score = cur_score
    best_seq = a.copy()
    it = 0
    it_no_improve = 0
    T = cur_score * 0.05               # initial temperature
    decay = 0.9995
    max_delta_factor = 0.02            # up to 2 % of source mass per move

    while time.time() < deadline:
        it += 1
        # Decide whether to use a random move or a gradient‑guided move
        if random.random() < 0.2:          # pure random exploration
            source = np.random.choice(n, p=a)
            target = np.random.randint(n)
        else:
            # Compute an active‑set sub‑gradient (same as in the sub‑gradient phase)
            b = _fft_self_convolve(a)
            cur_max = b.max()
            eps_rel = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(b >= cur_max - eps_rel)
            w = np.zeros_like(b)
            w[active] = 1.0
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # source probability ∝ a_i * grad_i  (high influence)
            src_weights = a * grad
            if src_weights.sum() > 0:
                src_probs = src_weights / src_weights.sum()
                source = np.random.choice(n, p=src_probs)
            else:
                source = np.random.choice(n, p=a)

            # target probability ∝ 1 / (grad_i + ε)  (low influence)
            inv_grad = 1.0 / (grad + eps)
            tgt_probs = inv_grad / inv_grad.sum()
            target = np.random.choice(n, p=tgt_probs)

        if target == source:
            continue

        max_delta = min(a[source] * max_delta_factor, 0.05)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snapshot = a.copy()
        conv_snapshot = conv.copy()

        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        new_score = 2.0 * n * new_M

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            M = new_M
            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snapshot
            conv = conv_snapshot
            it_no_improve += 1

        T *= decay

        # Occasionally kick the system if we are stuck
        if it % 5000 == 0 and it_no_improve > 2000:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            cur_score = 2.0 * n * M
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

    # ------------------------------------------------------------------
    # 7. Optional diffusion smoothing (tiny Gaussian blur) – cheap final touch.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        a = best_seq.copy()
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, eps)
            a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Diffusion smoothing new best {cur:.6f}")

    final_score = 2.0 * len(best_seq) * _fft_self_convolve(np.asarray(best_seq)).max()
    print(f"Final best score ≈ {final_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
