def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n-1]

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

The objective is scale‑invariant, so we work on the simplex
∑a_i = 1.  The score for a normalized vector a of length n is

    score = 2·n·max(convolution(a, a))

We therefore (1) sample a few natural analytic families (arcsine,
symmetric Beta) for several lengths (up to ~2·10⁴) and keep the
candidate with the smallest cheap score, and (2) refine that
candidate with a fast mirror‑descent (exponentiated‑gradient) that
uses the exact sub‑gradient of the max‑convolution.  Finally a tiny
random hill‑climbing phase transfers a small amount of mass between
two coordinates to escape possible plateaus.

The whole search respects the global time budget (default 1000 s) and
returns the best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) for a normalised to sum=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of `max(a*a)` on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for indices where k = 2*i
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p) distribution, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid under/overflow
    log_w = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(log_w)
    s = w.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return w / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises evaluate_sequence."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a set of lengths – we go up to ~2·10⁴ to approach the
    #     continuous limit (4/π ≈ 1.273).
    lengths = [
        500, 800, 1200, 1800, 2700, 4050, 6075,
        9112, 13668, 20502   # ≈ 2·10⁴
    ]
    # add a few random lengths between 500 and 20000
    for _ in range(8):
        lengths.append(np.random.randint(500, 20503))
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # 2.  Broad sampling of analytic families.
    beta_ps = [0.30, 0.40, 0.50, 0.60, 0.80]   # includes the optimal p=0.5
    best_seq = None
    best_score = np.inf

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine (Beta(½,½))
        a = _arcsine_distribution(n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()

        # symmetric Beta with a few p‑values
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            sc = _max_convolution_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()

        # uniform (just as a sanity check)
        a = np.full(n, 1.0 / n)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_score, best_seq = sc, a.copy()

    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Refinement – exponentiated‑gradient (mirror descent) on the simplex.
    a = best_seq.copy()
    it = 0
    best_local_seq = a.copy()
    best_local_score = best_score
    eta0 = 1.0

    while time.time() < deadline:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        # multiplicative update (mirror‑descent)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_local_score - 1e-15:
            best_local_score = cur_score
            best_local_seq = a.copy()

    print(f"[{time.time():.1f}] after mirror‑descent best={best_local_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Fine‑grained hill climbing – tiny random mass transfers.
    a = best_local_seq.copy()
    cur_score = best_local_score
    n = a.shape[0]
    max_transfer = 0.02   # start with up to 2 % of the total mass

    hill_it = 0
    while time.time() < deadline:
        hill_it += 1
        # occasionally shrink the transferred amount
        if hill_it % 4000 == 0:
            max_transfer *= 0.8
            if max_transfer < 1e-6:
                max_transfer = 1e-6

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(max_transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_seq = a_new.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={hill_it} new best {best_local_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_local_score:.6f} (length={len(best_local_seq)})")
    return best_local_seq.tolist()
