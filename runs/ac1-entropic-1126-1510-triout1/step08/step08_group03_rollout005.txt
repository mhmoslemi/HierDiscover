Algorithm

   lengths (≈0.5‑2.5× the base, plus a few random sizes up to 15 000).

   coming from several analytic families (uniform, arcsine, triangular,
   sine, cosine, exponential) and many random Dirichlet draws.
   Additionally, for each length we *optimise* a symmetric Beta
   distribution by a tiny grid + ternary‑search, obtaining a near‑optimal
   shape quickly.

3. **Selection** – keep the top‑k (k≈30) cheapest candidates according
   to the surrogate score `2·n·max(a*a)` (identical to the true
   objective for a sum‑normalised vector).

4. **Heavy refinement** – the retained seeds are refined using several
   optimiser flavours (exponentiated‑gradient, Adam, smoothed‑max
   continuation and plain sub‑gradient descent).  The best few seeds
   receive a larger share of the remaining budget.

5. **Guided hill‑climbing** – a final local search moves tiny amounts
   of mass from indices with *positive* sub‑gradient (mass that raises
   the max‑convolution) to indices with *negative* sub‑gradient.
   The transfer magnitude decays over time; occasional uphill moves
   are allowed to escape plateaus.

All steps respect the global 1000 s budget.  The returned sequence is

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation method: start with smooth max (γ≈5) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score

def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()

def _optimize_beta_sym(n: int, deadline: float):
    """
    Quick optimisation of a symmetric Beta(p,p) distribution.
    Returns (best_a, best_score).  Uses a coarse geometric grid
    followed by a few ternary‑search refinements.
    """
    # coarse grid (log‑space)
    p_grid = np.geomspace(0.05, 10.0, num=30)
    best_score = np.inf
    best_a = None
    for p in p_grid:
        a = _beta_symmetric_distribution(n, p)
        score = _max_convolution_score(a)
        if score < best_score:
            best_score = score
            best_a = a
        if time.time() >= deadline:
            break
    if best_a is None:
        return None, np.inf

    # refine around the best grid point
    p_center = p_grid[np.argmin([_max_convolution_score(_beta_symmetric_distribution(n, p))
                                 for p in p_grid])]
    lo = max(p_center * 0.5, 0.01)
    hi = min(p_center * 2.0, 20.0)

    for _ in range(12):  # ternary steps
        if time.time() >= deadline:
            break
        p1 = lo + (hi - lo) / 3.0
        p2 = hi - (hi - lo) / 3.0
        a1 = _beta_symmetric_distribution(n, p1)
        a2 = _beta_symmetric_distribution(n, p2)
        s1 = _max_convolution_score(a1)
        s2 = _max_convolution_score(a2)
        if s1 < s2:
            hi = p2
            if s1 < best_score:
                best_score, best_a = s1, a1.copy()
        else:
            lo = p1
            if s2 < best_score:
                best_score, best_a = s2, a2.copy()
    return best_a, best_score

        # deterministic families
        families = [
            _arcsine_distribution,
            _triangular_distribution,
            lambda nn: _exponential_distribution(nn, lam=0.003),
            _sine_distribution,
            _cosine_distribution,
            lambda nn: np.full(nn, 1.0 / nn)
        ]
        for f in families:
            a = f(n)
            candidates.append((a, _max_convolution_score(a)))

        # symmetric Beta (grid + local refinement)
        a_opt, sc_opt = _optimize_beta_sym(n, deadline)
        if a_opt is not None:
            candidates.append((a_opt, sc_opt))

        # symmetric Beta with preset p values (quick extra)
        for p in beta_params:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # asymmetric Beta (sampling a few random (p,q))
        for _ in range(3):
            p = np.random.choice(beta_params)
            q = np.random.choice(beta_params)
            a = _beta_asymmetric_distribution(n, p, q)
            candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws
        for _ in range(4):
            a = _random_dirichlet(n)
            candidates.append((a, _max_convolution_score(a)))

        # mixtures of two families
        fam_names = ['arcsine', 'triangular', 'uniform', 'sine', 'cosine']
        for _ in range(2):
            f1 = np.random.choice(fam_names)
            f2 = np.random.choice(['beta_sym', 'beta_asym', 'exponential'])
            # generate first component
            if f1 == 'arcsine':
                a1 = _arcsine_distribution(n)
            elif f1 == 'triangular':
                a1 = _triangular_distribution(n)
            elif f1 == 'uniform':
                a1 = np.full(n, 1.0 / n)
            elif f1 == 'sine':
                a1 = _sine_distribution(n)
            else:  # cosine
                a1 = _cosine_distribution(n)

            # second component
            if f2 == 'beta_sym':
                p = np.random.choice(beta_params)
                a2 = _beta_symmetric_distribution(n, p)
            elif f2 == 'beta_asym':
                p = np.random.choice(beta_params)
                q = np.random.choice(beta_params)
                a2 = _beta_asymmetric_distribution(n, p, q)
            else:  # exponential
                lam = 10 ** np.random.uniform(-4, -2)
                a2 = _exponential_distribution(n, lam=lam)

            lam_mix = np.random.rand()
            a = lam_mix * a1 + (1.0 - lam_mix) * a2
            a = a / a.sum()
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 2. Include previously‑best sequence, if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 30
    top_seeds = [seq for seq, _ in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] cheap initial best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy optimisation (time‑sharing)
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        # out of time – just return the current best
        return best_seq.tolist()

    # allocate 80 % of the remaining budget to the top 5 seeds,
    # the rest to a quick pass over the remaining seeds.
    n_refine = min(5, len(top_seeds))
    per_refine = 0.8 * remaining_time / max(n_refine, 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # quick pass for all seeds (just a few exponentiated‑gradient steps)
        quick_deadline = time.time() + min(20.0, per_refine * 0.2)
        seq, sc = _exp_gradient_descent(seed_seq, quick_deadline,
                                        max_iters=30_000,
                                        eta0=0.9)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} quick‑exp → {sc:.6f}")

        # dedicated heavy optimisation for the first few elite seeds
        if idx < n_refine:
            seed_deadline = time.time() + per_refine

            seq, sc = _exp_gradient_descent(seed_seq, seed_deadline,
                                            max_iters=120_000,
                                            eta0=0.9)
            if sc < best_score - 1e-14:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time():.1f}] seed {idx+1} exp‑grad → {sc:.6f}")

            seq, sc = _adam_descent(seed_seq, seed_deadline,
                                    max_iters=120_000,
                                    step0=0.25)
            if sc < best_score - 1e-14:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time():.1f}] seed {idx+1} Adam → {sc:.6f}")

            seq, sc = _smooth_gradient_descent(seed_seq, seed_deadline,
                                                max_iters=180_000,
                                                eta0=0.7,
                                                gamma0=5.0,
                                                gamma_factor=1.001)
            if sc < best_score - 1e-14:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time():.1f}] seed {idx+1} smoothed → {sc:.6f}")

            seq, sc = _subgradient_descent(seed_seq, seed_deadline,
                                           max_iters=250_000,
                                           eta0=0.5)
            if sc < best_score - 1e-14:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time():.1f}] seed {idx+1} subgrad → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Guided hill‑climbing with sub‑gradient direction
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.025   # start with up to 2.5 % of total mass per move

    while time.time() < deadline:
        it += 1
        # geometric decay of transferred amount
        transfer = max_transfer * (0.5 ** (it / 12000.0))

        grad, _, _ = _subgrad(a)
        pos_idx = np.where(grad > 1e-12)[0]
        neg_idx = np.where(grad < -1e-12)[0]

        if pos_idx.size == 0 or neg_idx.size == 0:
            # fall back to random pair if gradient has vanished
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
        else:
            i = int(np.random.choice(pos_idx))
            j = int(np.random.choice(neg_idx))

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb it={it} new best {best_score:.6f}")
        else:
            # occasional uphill acceptance to escape plateaus
            if np.random.rand() < 0.001:
                a = a_new
                cur_score = new_score

        # occasional jitter
        if it % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
