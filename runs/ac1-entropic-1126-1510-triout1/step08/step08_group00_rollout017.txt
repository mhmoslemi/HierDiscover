def _smooth_obj_and_grad(a: np.ndarray, p: float = 8.0):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_pow = b**p
    S = b_pow.sum()
    if S == 0:
        val = 0.0
        grad = np.zeros_like(a)
        return val, grad
    val = S**(1.0/p)

"""
Search for a non‑negative step function minimizing ``evaluate_sequence``.

Algorithm outline
-----------------
1️⃣  Broad sampling:
   – a few analytic families (arcsine, triangular, exponential,
     symmetric / asymmetric beta, power‑law, sinusoidal, raised‑cosine)
   – random Dirichlet draws with several concentration parameters.
   All candidates are ℓ₁‑normalised (∑a=1) and scored with the cheap
   surrogate ``2·n·max(conv(a,a))``.

2️⃣  Keep the best K seeds (K≈10) and refine each seed with three
   optimisation schemes:
   – Adam on the sub‑gradient of the exact max‑convolution (as before).
   – Exponentiated‑gradient (KL‑mirror) descent on the same objective.
   – Adam on a smooth Lp‑approximation of the max (p=4).  This yields
     richer gradient information and often escapes shallow local minima.

3️⃣  Global polishing:
   – a Frank‑Wolfe style pairwise mass‑transfer move driven by the
     sub‑gradient (full or half transfer).
   – a light simulated‑annealing/jitter phase to hop out of plateaus.

All loops respect the overall 1000 s budget and the best sequence seen
(kept as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Surrogate score 2·n·max(conv(a,a)) (exact when sum(a)=1).
    Faster than the full ``evaluate_sequence``.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 500_000):
    """Adam on the true sub‑gradient (max‑convolution)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 400_000):
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _smooth_obj_and_grad(a: np.ndarray, p: float = 4.0):
    """
    Smooth Lp‑approximation of max(conv(a,a)) and its gradient.
    Returns ``(value, grad)`` where ``value`` = (∑ conv^p)^{1/p}.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                # length 2n‑1
    # avoid numerical issues for tiny values
    b_pow = b ** p
    S = b_pow.sum()
    if S == 0.0:
        return 0.0, np.zeros_like(a)
    val = S ** (1.0 / p)
    factor = S ** (1.0 / p - 1.0)            # = S^{(1‑p)/p}
    w = factor * (b ** (p - 1))              # length 2n‑1
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return val, grad


def _adam_smooth_descent(a0: np.ndarray, deadline: float,
                         p: float = 4.0,
                         step0: float = 0.15,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         max_iters: int = 300_000):
    """
    Adam on the smoothed Lp objective (p≥2).  The true evaluation score is
    still checked after each step, so the returned sequence is comparable to
    the other methods.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        _, grad = _smooth_obj_and_grad(a, p)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution with possibly different left/right shapes."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _sinusoidal_distribution(n: int, power: float = 1.0) -> np.ndarray:
    """Positive half‑sine raised to ``power``."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** power
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _raised_cosine_distribution(n: int, power: float = 1.0) -> np.ndarray:
    """(1‑cos) raised to ``power`` (a symmetric bell)."""
    i = np.arange(n, dtype=np.float64)
    vals = (1.0 - np.cos(2.0 * np.pi * (i + 0.5) / n)) ** power
    vals /= vals.sum()
    return vals

    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq, best_score = top_seeds[0][0].copy(), top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Refinement of each seed with three optimisers
    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # 5a) Adam on exact sub‑gradient
        seq_adam, sc_adam = _adam_descent(seed_seq, deadline)
        if sc_adam < best_score - 1e-14:
            best_seq, best_score = seq_adam.copy(), sc_adam
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam improved to {best_score:.6f}")

        # 5b) Exponentiated‑gradient descent
        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, deadline)
        if sc_eg < best_score - 1e-14:
            best_seq, best_score = seq_eg.copy(), sc_eg
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad improved to {best_score:.6f}")

        # 5c) Adam on smooth Lp surrogate (p=4)
        seq_sad, sc_sad = _adam_smooth_descent(seed_seq, deadline, p=4.0)
        if sc_sad < best_score - 1e-14:
            best_seq, best_score = seq_sad.copy(), sc_sad
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} SmoothAdam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Frank‑Wolfe mass‑transfer polishing
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 80_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # mass to move away from
        i_dst = int(np.argmin(grad))   # mass to move towards
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-13:
            a = a_full
            cur_score = score_full
            improved = True
        else:
            # try half transfer
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-13:
                a = a_half
                cur_score = score_half
                improved = True

        if improved:
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")
            continue

        # occasional jitter to escape tiny plateaus
        if fw_iter % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 7️⃣ Light simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.02
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 150_000:
        anneal_iter += 1
        # linear temperature schedule based on remaining time
        remaining = deadline - time.time()
        total = budget_s - 0.5
        t_frac = remaining / total
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
