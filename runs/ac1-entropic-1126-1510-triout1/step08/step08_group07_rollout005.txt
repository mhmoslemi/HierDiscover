"""
Propose a low‑value step‑function for ``evaluate_sequence``.

The search combines three stages:

1. **Diverse analytic seeds** – a catalogue of known families
   (uniform, arcsine, symmetric beta, exponential, alternating,
   triangular, sparse random, noisy perturbations) is generated for a
   fixed length ``n`` (≈ 4000).  The best seed w.r.t. the exact
   objective 2·n·max(conv) is kept.

2. **Smoothed ‑max gradient descent** – the non‑differentiable
   ``max(conv)`` is replaced by a soft‑max with a temperature

   The gradient of the smooth objective is the weighted

   previous state‑of‑the‑art code, followed by a short simulated‑annealing
   Metropolis walk.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    """Arcsine law on a discrete grid."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta distribution with shape p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    """Centered exponential decay."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    """Sparse alternating pattern."""
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangle (0 at the ends, max in the centre)."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs(idx - centre) / (centre + 1.0)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

    return cand

def _crosscorr_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the vector g_i = Σ_j a_j * w_{i+j}  (cross‑correlation).
    Implemented via convolution with the reversed weight vector.
    """
    # ``np.convolve`` automatically switches to FFT for large inputs.
    conv_rev = np.convolve(a, w[::-1])
    # The required values are the first ``len(a)`` entries of the reversed
    # convolution result.
    grad = conv_rev[::-1][: len(a)]
    return grad


# ----------------------------------------------------------------------
def _gradient_descent_opt(a_initial: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Smoothed‑max gradient descent with an annealing schedule on β.
    Returns the best vector seen and its exact objective value.
    """
    a = a_initial.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # 1) current self‑convolution
        conv = _fft_self_convolve(a)

        # 2) soft‑max temperature (β grows from ≈2 to ≈40)
        beta = 2.0 + 38.0 * it / max_iters

        # 3) soft‑max weights w_k = exp(β·conv_k) / Σ exp(β·conv)
        scaled = beta * conv
        max_scaled = scaled.max()
        exp_vals = np.exp(scaled - max_scaled)          # numerical stability
        w = exp_vals / exp_vals.sum()

        # 4) gradient of the smooth objective: ∇ = 2·(a ⋆ w)
        grad = 2.0 * _crosscorr_gradient(a, w)

        # 5) step size – normalised by the gradient norm and β
        norm = np.linalg.norm(grad)
        if norm == 0:
            step = 0.0
        else:
            step = (0.2 / (beta + 1.0)) / norm

        # 6) projected gradient step
        a_new = a - step * grad
        a_new = _project_onto_simplex(a_new, 1.0)

        cur_score = _objective(a_new)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[{time.time():.1f}] GD it={it} β={beta:.2f} score={best_score:.6f}")

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer based on the exact sub‑gradient
    of the active self‑convolution shifts (identical to the previous
    state‑of‑the‑art implementation).
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient: g_i = 2 * Σ_{k∈active} a_{k-i} (if index exists)
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy it={it} improved to {best_score:.6f}")
            continue

        # try a smaller transfer
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy (small) it={it} improved to {best_score:.6f}")
            continue

        # occasional random jitter
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk with exponential cooling (identical to the
    previous implementation).
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9997

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine builds many analytic
    seeds, refines the best with a smooth‑max gradient descent, then
    applies the aggressive greedy optimiser and a short annealing
    phase.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1) Length choice – a power of two gives fast FFTs.
    n = 4096
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 2) Generate a rich set of starting vectors.
    candidates = _initial_candidates(n, rng)

    # Optional: include a previously known solution, if present.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) Pick the best raw seed (according to the exact objective).
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand.astype(np.float64)
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx} score={sc:.6f}")

    if best_vec is None:
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Smoothed‑max gradient descent on the current best.
    if time.time() < deadline:
        best_vec, best_score = _gradient_descent_opt(best_vec, deadline, rng)

    # 5) A few random restarts (still within the budget).
    restarts = 3
    for r in range(restarts):
        if time.time() >= deadline:
            break
        pert = best_vec + rng.normal(scale=0.02, size=n)
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, 1.0)
        vec_r, sc_r = _gradient_descent_opt(pert, deadline, rng)
        if sc_r < best_score:
            best_score = sc_r
            best_vec = vec_r.copy()
            print(f"[{time.time():.1f}] restart {r} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Aggressive greedy refinement.
    if time.time() < deadline:
        best_vec, best_score = _greedy_local_search(best_vec, deadline, rng)

    # ------------------------------------------------------------------
    # 7) Final simulated‑annealing fine‑tuning (if time permits).
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)

    # ------------------------------------------------------------------
    # 8) Final sanity – ensure we are on the simplex.
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
