"""
Step‑function optimizer for the ``evaluate_sequence`` target.

The objective (after normalising the sum to 1) is
        F(a) = 2·n·max(conv(a,a))
where conv(a,a) is the full self‑convolution.
The optimisation proceeds in three stages:

1. **Rich seeding** – a wide variety of analytic families (uniform,
   arcsine, beta, exponential, power‑law at the edges, sine, alternating,
   block‑spikes, sparse random) are generated for several lengths.
   The best few are kept as starting points.

2. **Smoothed Adam** – a continuation method that minimises a
   log‑sum‑exp smooth approximation of the max‑convolution.
   Adam updates are projected onto the probability simplex.

3. **Simulated‑annealing refinement** – random mass‑transfer moves are
   accepted with Metropolis probability, allowing occasional uphill
   steps.  The temperature is cooled geometrically.  The current best
   vector is continuously updated.

All stages respect the global time budget (default 1000 s).
The best vector found is returned as a plain list of ``float``.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex   {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector whose sum is 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int,
                                 rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a fairly broad pool of start vectors for a given length.
    """
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    for lam in (0.001, 0.002, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    for gamma in (0.05, 0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40):
        cand.append(_distribution_powerlaw_edges(n, gamma))

    cand.append(_distribution_sine(n))

    for stride in (2, 3, 4):
        cand.append(_distribution_alternating(n, stride))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand

# Smoothed Adam descent (log‑sum‑exp approximation of the max)
def _smooth_adam(a: np.ndarray,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max approximation of the convolution objective."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # Alpha schedule – we go a bit further than the reference code.
    schedule = [
        (0.5, 2000, 0.20),
        (1.0, 2000, 0.15),
        (5.0, 2000, 0.12),
        (20.0, 2000, 0.10),
        (100.0, 4000, 0.08),
        (500.0, 3000, 0.06),
        (2000.0, 2000, 0.04),
        (5000.0, 1500, 0.03),
        (20000.0, 1200, 0.02),
        (50000.0,  800, 0.015),
        (200000.0, 600, 0.010),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + simplex projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score

# Simulated‑annealing refinement (allows uphill moves)
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves with Metropolis acceptance.
    The temperature is cooled geometrically.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective(cur)

    # Temperature schedule – start relatively high, finish near zero.
    T0 = max(0.01, cur_score * 0.08)
    T_final = 1e-7
    # allocate a generous number of iterations but stop when out of time
    max_iters = 200_000
    if max_iters <= 0:
        return best_seq, best_score
    cooling = (T_final / T0) ** (1.0 / max_iters)
    T = T0

    rng = np.random.default_rng(int(time.time() * 1e6) & 0xffffffff)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # pick a source with non‑trivial mass
        src = rng.integers(n)
        if cur[src] < 1e-14:
            continue
        # pick a distinct destination
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # transfer a random fraction of the source mass
        max_frac = min(0.5, cur[src])          # never move more than 50 % of src
        delta = cur[src] * rng.random() * max_frac
        if delta < 1e-16:
            continue

        new_vec = cur.copy()
        new_vec[src] -= delta
        new_vec[dst] += delta

        new_score = _objective(new_vec)
        diff = new_score - cur_score

        # Metropolis acceptance
        if diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-15)):
            cur = new_vec
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur.copy()
                print(f"[{time.time():.1f}] SA it={it} new best {best_score:.6f}")

        # geometric cooling
        T *= cooling

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best vector found as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths – longer vectors give finer discretisation.
    lengths = [2048, 4096, 8192, 16384]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        cand_list = _initial_candidates_extended(n, rng)
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] *new global best* length={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep a bounded pool of promising seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 3. Incorporate a previously discovered construction if present.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback in the extremely unlikely case the pool is empty.
    if overall_best_seq is None:
        n = 2048
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline.
    #   a) Smoothed Adam on a few top seeds.
    #   b) Simulated‑annealing refinement.
    seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]

    for seed_vec in seeds:
        if time.time() >= deadline - 5:
            break
        # a) smoothed Adam
        sub_deadline = deadline - 5.0
        seed_vec, overall_best_score = _smooth_adam(
            seed_vec.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        overall_best_seq = seed_vec.copy()

        # b) simulated annealing refinement
        if time.time() < deadline - 2:
            seed_vec, overall_best_score = _simulated_annealing(
                seed_vec.copy(),
                deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            overall_best_seq = seed_vec.copy()

        # occasional random perturbation of the current best to escape stalls
        if time.time() < deadline - 2:
            pert = overall_best_seq.copy()
            pert += rng.normal(scale=0.02, size=pert.shape)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0:
                pert = pert / pert.sum()
                seeds.append(pert)

        # keep seed list short
        if len(seeds) > 10:
            seeds = seeds[-10:]

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
