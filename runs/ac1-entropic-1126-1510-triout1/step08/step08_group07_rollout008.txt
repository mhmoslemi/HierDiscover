def gradient_smooth_max(a, alpha):
    conv = _fft_self_convolve(a) # length 2n-1
    max_val = conv.max()
    # compute softmax weights
    # Use stable softmax: w = np.exp(alpha * (conv - max_val))
    w = np.exp(alpha * (conv - max_val))
    Z = w.sum()
    s = w / Z  # softmax distribution
    # compute gradient = 2 * correlate(s, a)
    n = a.shape[0]
    # Use FFT
    L = 1 << ((3*n - 1) - 1).bit_length()  # size power of 2 >= 3n-1 perhaps.
    # compute FFT of s and a zero-padded.
    S = np.fft.rfft(s, n=L)
    A = np.fft.rfft(a, n=L)
    # cross-correlation via conj(S)*A
    corr = np.fft.irfft(np.conj(S) * A, n=L)
    # Extract values for lags 0..n-1: index from n-1 to n-1+n
    grad = 2 * corr[n-1:n-1+n]
    return grad

def gradient_descent(a, deadline):
    best = a.copy()
    best_score = _objective(best)
    alpha = 10.0
    lr = 0.5  # initial step size
    success = True
    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        max_b = conv.max()
        if max_b * 2 * len(a) < best_score - 1e-12:
            best = a.copy()
            best_score = 2 * len(a) * max_b
        # gradient via smooth max with current alpha
        # can increase alpha gradually
        w = np.exp(alpha * (conv - max_b))
        w_sum = w.sum()
        softmax = w / w_sum
        # correlation gradient
        n = len(a)
        L = 1 << ((3 * n - 2) - 1).bit_length()
        # compute corr
        S = np.fft.rfft(softmax, n=L)
        A = np.fft.rfft(a, n=L)
        corr = np.fft.irfft(np.conj(S) * A, n=L)
        grad = 2 * corr[n-1:n-1+n]
        # compute step size adaptively
        # attempt a step
        step = lr
        for _ in range(10):
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new, 1.0)
            # compute objective
            new_score = _objective(a_new)
            if new_score < best_score - 1e-12:
                a = a_new
                best = a_new.copy()
                best_score = new_score
                # enlarge step size for next iteration
                lr = step * 1.1
                break
            else:
                step *= 0.5
        else:
            # no improvement; reduce step size
            lr *= 0.5
            # optionally perturb
            if np.random.rand() < 0.01:
                # random jitter
                i = np.random.randint(n)
                j = np.random.randint(n-1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.05, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                a_tmp = _project_onto_simplex(a_tmp)
                tmp_score = _objective(a_tmp)
                if tmp_score < best_score - 1e-12:
                    a = a_tmp
                    best = a_tmp.copy()
                    best_score = tmp_score
        # gradually increase alpha to sharpen max
        if alpha < 200:
            alpha *= 1.001
        # Could re-evaluate if time leftover
    return best, best_score

def _smooth_gradient(a, alpha):
    conv = _fft_self_convolve(a)
    max_b = conv.max()
    # stable exp: subtract max_b for stability.
    w = np.exp(alpha * (conv - max_b))
    w_sum = w.sum()
    softmax = w / w_sum

    n = a.shape[0]
    L = 1 << ((3 * n - 2) - 1).bit_length()
    S_fft = np.fft.rfft(softmax, n=L)
    A_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(np.conj(S_fft) * A_fft, n=L)
    grad = 2 * corr[n-1:n-1+n]
    return grad, max_b

"""
Gradient‑descent + active‑set optimiser for the min‑max self‑convolution
objective.

Steps
-----
1.  Build a diverse pool of analytic distributions (uniform, arcsine,
    symmetric beta with a finely tuned shape parameter, exponential,
    sparse random).  The best one (according to the objective) is kept.

2.  Refine that seed with a smooth‑max projected gradient descent.
    The non‑smooth max‐convolution is approximated by a softmax
    (log‑sum‑exp) with a temperature that is slowly cooled, providing
    a usable gradient on the simplex.  A back‑tracking line‑search
    guarantees monotonic improvement.

3.  Run the aggressive greedy active‑set optimiser (the one used in the
    previous state‑of‑the‑art) to exploit the exact sub‑gradient of the
    max‑shift(s).

4.  Finish with a cheap simulated‑annealing walk if time permits.

The routine respects the global deadline (budget_s) and always returns a
non‑negative list whose sum is 1 (the scaling is irrelevant for the
scoring function).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta on {0,…,n‑1} with shape parameter p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _best_beta_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Fine‑grid search for the best symmetric beta shape."""
    best_vec = None
    best_score = float("inf")
    best_p = None

    # coarse log‑grid
    p_grid = np.logspace(np.log10(0.001), np.log10(2.0), num=200)
    for p in p_grid:
        vec = _distribution_beta_sym(n, float(p))
        sc = _objective(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec
            best_p = p

    # local refinements
    for _ in range(6):
        p = best_p * math.exp(rng.normal(scale=0.2))
        if p <= 0:
            continue
        vec = _distribution_beta_sym(n, float(p))
        sc = _objective(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec
            best_p = p

    # a few noisy perturbations around the best beta
    for _ in range(4):
        pert = best_vec + rng.normal(scale=0.015, size=n)
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, 1.0)
        sc = _objective(pert)
        if sc < best_score:
            best_score = sc
            best_vec = pert
            # keep best_p unchanged – it only matters for the pure beta

    return best_vec


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of starting vectors."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # beta family – let the dedicated routine polish it
    cand.append(_best_beta_candidate(n, rng))

    # exponentials with several decay rates
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # sparse random vectors (≈0.5 % non‑zeros)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    rnd[idx] = rng.random(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    # a few Gaussian‑noised copies of the current best beta
    base = cand[-2]  # the last added exponential‑type; replace if needed
    for _ in range(3):
        pert = base + rng.normal(scale=0.02, size=n)
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, 1.0)
        cand.append(pert)

    return cand

def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Aggressive pairwise mass transfer based on the exact active‑set sub‑gradient."""
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient of max_b
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # try a full‑mass move (capped)
        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # smaller move
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr

    return best_vec, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()

        T *= 0.9997

    return best_a, best_score

def _smooth_max_gradient(a: np.ndarray, alpha: float) -> tuple[np.ndarray, float]:
    """
    Returns (grad, current_max) where grad is the gradient of
    smooth_max( conv(a,a) ) w.r.t. a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()

    # soft‑max weights (stable)
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        # should never happen, but guard against numerical collapse
        soft = np.full_like(w, 1.0 / w.size)
    else:
        soft = w / w_sum

    # correlation grad = 2 * (soft ⋆ a)  (⋆ = cross‑correlation)
    L = 1 << ((3 * n - 2) - 1).bit_length()
    S_fft = np.fft.rfft(soft, n=L)
    A_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(np.conj(S_fft) * A_fft, n=L)
    grad = 2.0 * corr[n - 1:n - 1 + n]
    return grad, m


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a: np.ndarray,
                             deadline: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex using a smoothed
    approximation of the max‑convolution.  The smoothing parameter
    (alpha) is slowly increased so that the algorithm ends up minimising
    the true max.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    alpha = 10.0          # start with a fairly smooth approximation
    lr = 0.5              # initial step length
    max_alpha = 200.0

    while time.time() < deadline:
        grad, cur_max = _smooth_max_gradient(a, alpha)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()

        # ----- back‑tracking line‑search -----
        step = lr
        improved = False
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new, 1.0)
            new_score = _objective(a_new)
            if new_score < cur_score - 1e-12:
                a = a_new
                lr = step * 1.2     # slightly enlarge step for next iteration
                improved = True
                break
            step *= 0.5

        if not improved:
            lr *= 0.5
            # occasional random jitter to avoid stagnation
            if rng.random() < 0.012:
                i = rng.integers(n)
                if a[i] > 1e-12:
                    j = rng.integers(n - 1)
                    if j >= i:
                        j += 1
                    eps = min(a[i] * 0.05, 0.001)
                    a_tmp = a.copy()
                    a_tmp[i] -= eps
                    a_tmp[j] += eps
                    a_tmp = _project_onto_simplex(a_tmp, 1.0)
                    new_score = _objective(a_tmp)
                    if new_score < best_score - 1e-12:
                        a = a_tmp
                        best_score = new_score
                        best_vec = a_tmp.copy()

        # slowly cool the smoothness
        if alpha < max_alpha:
            alpha *= 1.001

    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # 1) choose a sensible length (large enough for a good shape but small enough for speed)
    n = 2600
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 2) generate a wide pool of seeds
    candidates = _initial_candidates(n, rng)

    # also try to reuse the previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0:
                prev = prev / prev.sum()
                candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) pick the best initial candidate (fast evaluation)
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
        # occasional progress report
        if idx % 5 == 0:
            print(f"[{time.time():.1f}] init cand {idx:3d} score={sc:.6f}")

    if best_vec is None:
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print("[fallback] using uniform distribution")

    print(f"[{time.time():.1f}] best initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) smooth‑max gradient descent refinement
    gd_deadline = deadline - 2.5          # keep a little room for the final phases
    best_vec, best_score = _smooth_gradient_descent(best_vec, gd_deadline, rng)
    print(f"[{time.time():.1f}] after smooth GD score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) aggressive greedy local optimisation (exact sub‑gradient)
    best_vec, best_score = _greedy_local_search(best_vec, deadline, rng)
    print(f"[{time.time():.1f}] after greedy optimisation score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) simulated‑annealing fine‑tuning (if any time left)
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)
        print(f"[{time.time():.1f}] after simulated annealing score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity – ensure we are on the simplex
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"[{time.time():.1f}] final reported score = {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
