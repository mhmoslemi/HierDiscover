"""
A multi‑phase optimizer for the step‑function bound.

The routine works in three stages:

1️⃣  Broad sampling – many analytic families (Beta, exponential, cosine,
    parabola, power‑law, …) are instantiated for a few different lengths.
    Random Dirichlet draws and random convex mixes of two families add
    stochastic diversity.  Only the top‑k candidates (by the cheap surrogate
    2·n·max(conv(a,a))) are retained.

2️⃣  Intensive refinement – each seed is improved by two first‑order
    methods that respect the simplex constraint:
        • Adam‑style adaptive‑moment descent.
        • Exponentiated‑gradient (KL‑mirror) descent.
    The best point seen is kept.

3️⃣  Polishing – a Frank‑Wolfe “mass‑transfer” step repeatedly moves mass
    from the component with the largest sub‑gradient to the one with the
    smallest, followed by a light simulated‑annealing phase.

The whole search respects a 1000 s wall‑time budget and always returns a
non‑negative list ready for ``evaluate_sequence``.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (real FFT, zero‑padded)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Surrogate score (exact when ∑a = 1)
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) – the cheap surrogate used for selection."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (used by all first‑order methods)
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a = 1).
    Returns (gradient, current max value, active indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# First‑order optimisation primitives ------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 500_000):
    """Adam‑style adaptive‑moment descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 400_000):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        # Avoid division by zero.
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

# Analytic families (all return a normalized vector, sum = 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _cosine_window_distribution(n: int) -> np.ndarray:
    """A Hann‑like window: zero at the ends, smooth interior."""
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    vals = (1.0 - np.cos(np.pi * x)) * 0.5
    vals[0] = 0.0
    vals[-1] = 0.0
    vals /= vals.sum()
    return vals


def _parabolic_distribution(n: int) -> np.ndarray:
    """A symmetric parabola that vanishes at the ends: 1‑x²."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0          # map i → [-1, 1]
    vals = 1.0 - x * x
    vals[vals < 0] = 0.0
    vals /= vals.sum()
    return vals

# Random helpers
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _random_mixture(n: int, families: list, rng: np.random.Generator) -> np.ndarray:
    """Pick two families at random, mix them with a random weight and normalise."""
    f1, _ = rng.choice(families)
    f2, _ = rng.choice(families)
    a1 = f1(n)
    a2 = f2(n)
    w = rng.random()
    mixed = w * a1 + (1.0 - w) * a2
    mixed /= mixed.sum()
    return mixed

                cur_score = new_score
