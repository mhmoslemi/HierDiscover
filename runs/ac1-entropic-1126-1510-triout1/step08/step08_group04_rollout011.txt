def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    best_seq = None
    best_score = float('inf')
    best_n = None

    # 1. Evaluate various initial families across lengths
    lengths = [500, 1000, 1500, 2000, 2500, 3000]
    for n in lengths:
        if time.time() >= deadline: break
        candidates = []
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.01))
        # take p Beta distribution
        for p in [0.4, 0.8, 1.2, 2.0, 4.0]:
            candidates.append(_distribution_beta_sym(n, p))
        # sine power
        for pw in [0.5, 1.0, 2.0, 4.0]:
            candidates.append(_distribution_sine_power(n, pw))
        for sigma in [0.2, 0.35, 0.5]:
            candidates.append(_distribution_gaussian(n, sigma))
        # random
        for alpha in [0.1, 0.5, 1.0, 2.0]:
            candidates.append(np.random.dirichlet(np.full(n, alpha)))
        # evaluate cheap
        for vec in candidates:
            if time.time() >= deadline: break
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} best {sc:.6f}")

    # If global variable height_sequence_1 exists, evaluate:
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)
        prev = np.maximum(prev, 1e-12)
        prev /= prev.sum()
        sc = _max_convolution_score(prev)
        if sc < best_score:
            best_seq = prev.copy()
            best_n = len(prev)
            best_score = sc
            print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    # Ensure best_seq not None
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)

    # Mirror descent refinement (optional)
    # We'll perform short smooth-max descent with fewer steps as we have LP.

    a = best_seq.copy()
    n = a.shape[0]
    # Mirror descent schedule with moderate steps
    schedule = [
        (1.0, 100.0, 1500, 0.1, 0.01),
        (100.0, 5000.0, 1500, 0.01, 0.001),
    ]
    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t
            conv = _fft_self_convolve(a)
            maxc = conv.max()

            w_sum = w.sum()

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            if it % 300 == 0:
                sc = _max_convolution_score(a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] mirror n={n} best {sc:.6f}")

    # LP refinement
    a = best_seq.copy()
    n = a.shape[0]
    # Set mixing factor
    t_mix = 0.2
    max_lp_iter = 30
    while time.time() < deadline and max_lp_iter > 0:
        conv = _fft_self_convolve(a)
        M = conv.max()
        # Build sparse convolution matrix
        # Preallocate arrays
        nnz = n * n
        row_idx = np.empty(nnz, dtype=np.int32)
        col_idx = np.empty(nnz, dtype=np.int32)
        data = np.empty(nnz, dtype=np.float64)
        pos = 0
        for i in range(n):
            # rows i ... i+n-1
            row_idx[pos:pos+n] = np.arange(i, i+n)
            col_idx[pos:pos+n] = np.arange(n)
            data[pos:pos+n] = a[i]
            pos += n
        T = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        c = -np.ones(n)  # maximize sum(g) -> minimize negative
        # Solve LP
        try:
            res = linprog(c, A_ub=T, b_ub=b_ub, bounds=(0, None), method='highs')
        except Exception as e:
            print(f"LP solve error: {e}")
            break
        if not res.success:
            print(f"LP not successful: {res.message}")
            break
        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g_norm = g / g_sum
        # mixing line search: try decreasing t to ensure progress
        improved = False
        for t_cur in (t_mix, t_mix*0.5, t_mix*0.25, t_mix*0.1, t_mix*0.05):
            new_a = (1 - t_cur) * a + t_cur * g_norm
            new_a = np.maximum(new_a, 1e-12)
            new_a /= new_a.sum()
            new_M = _fft_self_convolve(new_a).max()
            if new_M < M - 1e-12:
                a = new_a
                M = new_M
                improved = True
                best_score = 2 * n * M
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP iter {max_lp_iter} t={t_cur:.4f} new best {best_score:.6f}")
                break
        if not improved:
            # Reduce mixing factor, maybe break
            t_mix *= 0.5
            if t_mix < 1e-4:
                break
        max_lp_iter -= 1

    # Final hill climbing
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # locate maximal entry
        k = int(conv.argmax())
        # Determine pair (i,j) that gives max contribution among that k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high+1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        # Move mass from larger to random elsewhere
        source = best_i if a[best_i] > a[best_j] else best_j
        target = random.randint(0, n-1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < 1e-12:
            continue
        a_snapshot = a.copy()
        conv_snapshot = conv.copy()
        # Apply transfer efficiently (recompute conv each time? Use O(n) update)
        # For simplicity recompute conv from scratch after transfer (since n moderate)
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, 1e-12)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_seq = a.copy()
            best_score = 2 * n * M
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill iter {it} best {best_score:.6f}")
        else:
            a = a_snapshot
            conv = conv_snapshot
        if it % 5000 == 0:
            # occasional perturb
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter at hill iter {it}")

    final_score = 2 * n * conv.max()
    print(f"Final score ≈ {final_score:.6f} length={n}")
    return best_seq.tolist()

def subgradient_refine(a, deadline):
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        velocity = beta * velocity + (1 - beta) * grad
        eta = 0.15 / np.sqrt(it)  # may adjust adapt

"""
Propose a step‑function vector (non‑negative list) that minimises
``evaluate_sequence``.  The method combines:

1.   A quick scan of several analytic families (uniform, arcsine,
     triangular, exponential, Beta, sine‑power, Gaussian) for a few
     candidate lengths and keeps the best normalised vector.
2.   A short smooth‑max mirror‑descent on the simplex (exponentiated‑gradient)
     to further lower the max self‑convolution.
3.   A sub‑gradient descent with momentum (the same idea used in the
     reference code) for polishing.
4.   A **linear‑programming refinement**: for the current vector `f`,
     solve the LP
         maximise   Σ g_i
         subject to T(f)·g ≤ M ,   g_i ≥ 0
     where ``T(f)`` is the (dense) convolution matrix and ``M`` is the
     current maximum of ``f*f``.  The optimiser returns a feasible `g`
     that pushes mass to low‑impact positions; we mix it with `f`.
5.   A final cheap hill‑climber that repeatedly moves a small amount of
     mass from the pair that contributes most to the current maximum
     towards a random index.

The whole search respects a 1000 s wall‑clock budget; intermediate
results are printed and the best sequence found is returned as a plain
list of Python ``float`` objects.
"""

import time
import random
import math

import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _lp_refine(f, mix_factor, deadline):
    """One LP‑based refinement step. Returns potentially improved (f, M)."""
    n = f.shape[0]
    conv_f = _fft_self_convolve(f)
    M = conv_f.max()

    # Build the dense convolution matrix T such that T @ g = f * g
    # first column: f padded with n‑1 zeros; first row: f[0] followed by zeros
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    T = toeplitz(col, row)          # shape (2n‑1) × n

    # Linear program: maximise Σ g_i  <=> minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)          # objective
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # upper bound for each convolution entry
    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=(0, None), method='highs')
    except Exception as e:
        print(f"LP solver exception: {e}")
        return f, M

    if not res.success:
        # LP failed – fall back to the current vector
        # (the message may be useful for debugging)
        # print("LP failure:", res.message)
        return f, M

    g = res.x
    gsum = g.sum()
    if gsum < 1e-12:
        return f, M

    # Normalise g to have sum 1 (the same scale as f)
    g_norm = g / gsum

    # Try a few mixing ratios (starting from the supplied one)
    for t in (mix_factor, mix_factor * 0.5, mix_factor * 0.25,
              mix_factor * 0.1, mix_factor * 0.05):
        new_f = (1 - t) * f + t * g_norm
        new_f = np.maximum(new_f, 1e-12)
        new_f /= new_f.sum()
        new_M = _fft_self_convolve(new_f).max()
        if new_M < M - 1e-12:
            return new_f, new_M

    # No improvement – return the original vector
    return f, M

    best_score = float('inf')
    best_n = None

    # 1.  Quick scan of analytic families on a handful of lengths.
    lengths = [500, 1000, 1500, 2000, 2500, 3000]

        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.008))
        candidates.append(_distribution_exponential(n, lam=0.02))

            candidates.append(_distribution_beta_sym(n, p))

            candidates.append(_distribution_gaussian(n, sigma))
