def hill_climb(a_init, deadline):
    a = a_init.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    eps = 1e-12
    while time.time() < deadline:
        it += 1
        # find max index
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find best pair with max product
        best_i = i_start
        best_j = k - i_start
        max_prod = a[best_i] * a[best_j]
        # Iterate through i in range to find max product among pairs contributing to k
        # but we can also just pick the larger of a_i, a_j as source.
        for i in range(i_start, i_end+1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        # choose target far random (maybe weighted opposite side)
        target = random.randint(0, n-1)
        if target == source:
            continue
        # delta limited by source amount
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        # backup
        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            if it % 200 == 0:
                print(f"{time.time():.1f} hill it={it} new M={M:.6f} score={2*n*M:.6f}")
        else:
            a = a_snap
            conv = conv_snap

        # occasional jitter or re-evaluation
        if it % 5000 == 0:
            # jitter
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"{time.time():.1f} jitter after {it} iter.")
    return a

"""
Propose a step‑function vector with a tiny self‑convolution maximum.

The algorithm consists of three stages:

1. **Coarse search** – generate a handful of analytic families (uniform,
   arcsine, triangular, exponential, symmetric beta, sine‑power,
   Gaussian) and a few random Dirichlet samples for several lengths.
   The best vector (in the cheap score 2·n·max (a*a)) is kept.

2. **Smooth‑max Adam optimisation** – the chosen vector is re‑parameterised
   as soft‑max logits `z` (so `a = softmax(z)` guarantees non‑negativity
   and sum = 1).  A smoothed version of the max‑convolution, namely the
   log‑sum‑exp, is minimised with the Adam optimiser.  The smoothing
   parameter `α` is raised geometrically from 0.1 up to 1e6 while the
   learning rate is simultaneously decayed.  Gradients are obtained
   analytically using FFT‑based convolutions, and the current best
   vector is stored whenever the cheap score improves.

3. **O(n) hill‑climber** – starting from the Adam result we scan the
   convolution peak, identify the pair of indices that contributes most
   to it and move a small amount of mass from the larger of the pair to
   a random location.  The convolution array is updated in O(n) with an
   incremental routine.  Accepted moves strictly lower the peak; the
   procedure runs until the time budget expires.

All work respects the 1000 s wall‑clock limit; intermediate progress is
printed.  The returned list contains non‑negative numbers and is ready
to be evaluated with `evaluate_sequence`.
"""

import time
import random
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper – simplex projection not needed because we use a soft‑max
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution (a * b)."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Cheap surrogate for `evaluate_sequence` (assumes sum(a)=1):
    returns 2·n·max(a*a).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return vectors that sum to 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Mass‑transfer helper for the hill‑climber
# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Transfer `delta` mass from index i to j, updating both `a`
    and its full self‑convolution `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0.0 or i == j:
        return
    a_old = a.copy()
    # update convolution contributions that involve index i
    conv[i: i + n] -= 2.0 * delta * a_old
    # and contributions that involve index j
    conv[j: j + n] += 2.0 * delta * a_old
    # self‑pair (i,i) and (j,j) receive the quadratic term
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Starting from vector a0 (sum=1, non‑negative) perform a long
    Adam optimisation on the smooth‑max surrogate.
    Returns the best vector found before the deadline.
    """
    eps = 1e-12
    a = np.maximum(a0, eps)
    a = a / a.sum()
    z = np.log(a)                # logits – unconstrained parameters
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    it = 0

    best_a = a.copy()
    best_score = _max_convolution_score(best_a)

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (0.1, 10.0, 1500, 0.30, 0.030),
        (10.0, 100.0, 2500, 0.030, 0.0030),
        (100.0, 1000.0, 3000, 0.0030, 0.00030),
        (1000.0, 10000.0, 4000, 0.00030, 0.00003),
        (10000.0, 100000.0, 4000, 0.00003, 0.000003),
        (100000.0, 1_000_000.0, 4000, 0.000003, 0.0000003),
    ]

    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        if time.time() >= deadline:
            break
        for s in range(steps):
            if time.time() >= deadline:
                break
            it += 1

            # geometric interpolation for α and learning‑rate
            t = s / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            # ----- soft‑max parametrisation -----
            max_z = np.max(z)
            exp_z = np.exp(z - max_z)
            a = exp_z / exp_z.sum()

            # ----- smoothed max of self‑convolution -----
            conv = _fft_self_convolve(a)                # length 2n‑1
            maxc = conv.max()
            w_raw = np.exp(alpha * (conv - maxc))       # stable soft‑max
            w = w_raw / w_raw.sum()                     # shape 2n‑1

            # ----- gradient wrt a -----
            rev_a = a[::-1]
            conv_rev = _fft_convolve(rev_a, w)          # length 3n‑2
            grad_a = 2.0 * conv_rev[n - 1: n - 1 + n]    # size n

            # ----- chain rule for the logits -----
            grad_z = a * (grad_a - np.dot(a, grad_a))

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- bookkeeping -----
            if (it % 1000) == 0 or s == steps - 1:
                cur_score = _max_convolution_score(a)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={it} α={alpha:.1e} score={cur_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Simple hill‑climber that works directly on the convolution array
# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Local improvement phase: repeatedly move a tiny amount of mass from the
    index that participates most in the current convolution peak to a
    random location.  Accept only if the peak strictly decreases.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    eps = 1e-12

    while time.time() < deadline:
        it += 1
        # index of the maximal convolution entry
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) inside the window giving the largest product a[i]*a[j]
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a_save = a.copy()
        conv_save = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()

        if new_M < M - 1e-12:               # strictly better
            M = new_M
            if (it % 200) == 0:
                print(f"[{time.time():.1f}] hill iter={it} new M={M:.6e} score={2.0*n*M:.6f}")
        else:
            a = a_save
            conv = conv_save

        # occasional jitter to escape tiny plateaus
        if (it % 5000) == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    return a
