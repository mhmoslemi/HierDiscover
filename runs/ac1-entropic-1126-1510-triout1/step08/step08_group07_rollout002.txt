"""
A hybrid optimizer for the min‑max self‑convolution problem.

The routine builds a rich set of analytic seeds, picks the best,
then runs a smooth‐gradient descent (log‑sum‑exp smoothing of the
max‑convolution) followed by the greedy active‑set optimiser and a
short simulated‑annealing fine‑tuning.  All steps respect the global
deadline (default 1000 s) and the solution is projected onto the
probability simplex before being returned.

The smooth‑gradient phase gives a powerful way to move mass out of
indices that contribute most to the peak of the convolution while
still keeping the update cheap (two FFTs per iteration).  The
following implementation works for any sequence length up to a few
ten‑thousands and stays well within the time budget.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small utility – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective used internally: 2·n·max(conv(a,a)) – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (used for diverse initial seeds) --------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta distribution with shape parameter p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diversified pool of starting vectors."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta – many shape parameters (including very small)
    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – several decay rates
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # sparse random start (~0.5 % non‑zeros)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    rnd[idx] = rng.random(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    # noisy perturbations of a few beta shapes
    for _ in range(3):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, 1.0)
        cand.append(pert)

    # a few completely random dense vectors (help to escape symmetry)
    for _ in range(3):
        v = rng.random(n)
        v = _project_onto_simplex(v, 1.0)
        cand.append(v)

    return cand

def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        L(a) = log‑sum‑exp(alpha·conv(a,a))
    (up to a constant factor).  Returns a vector of length n.
    """
    n = a.shape[0]
    # self‑convolution
    conv = _fft_self_convolve(a)               # length 2n‑1
    maxc = conv.max()
    # soft‑max weights (stable)
    expv = np.exp(alpha * (conv - maxc))
    w = expv / expv.sum()                     # length 2n‑1, non‑negative, sum 1

    # gradient = 2 * (w * a_rev) evaluated on the n positions
    a_rev = a[::-1]
    size = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=size)
    R = np.fft.rfft(a_rev, n=size)
    full = np.fft.irfft(W * R, n=size)        # length = len(w)+len(a_rev)-1 = 3n‑2
    start = n - 1
    g = 2.0 * full[start:start + n]
    return g


def _smooth_descent(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Smoothed gradient descent with a linearly increasing smoothing
    parameter (alpha) and a linearly decreasing step size.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective(cur)
    best = cur.copy()
    best_score = cur_score

    max_iters = 5000          # hard cap; will exit earlier if time runs out
    it = 0
    while time.time() < deadline and it < max_iters:
        # schedules
        t = it / max_iters
        alpha = 5.0 + (200.0 - 5.0) * t            # 5 → 200
        step = 0.30 - (0.30 - 0.03) * t             # 0.30 → 0.03

        g = _smooth_gradient(cur, alpha)

        # candidate move
        cand = cur - step * g
        cand = _project_onto_simplex(cand, 1.0)

        cand_score = _objective(cand)

        # simple acceptance rule (strict improvement)
        if cand_score < cur_score - 1e-14:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-14:
                best = cand.copy()
                best_score = cand_score
        else:
            # occasional random jitter to avoid stagnation
            if rng.random() < 0.02:
                eps = rng.normal(scale=0.001, size=n)
                jitter = cur + eps
                jitter = np.clip(jitter, 0.0, None)
                jitter = _project_onto_simplex(jitter, 1.0)
                jitter_score = _objective(jitter)
                if jitter_score < cur_score:
                    cur = jitter
                    cur_score = jitter_score
                    if jitter_score < best_score:
                        best = jitter.copy()
                        best_score = jitter_score

        if it % 200 == 0:
            print(f"[{time.time():.1f}] smooth it={it:4d} alpha={alpha:6.2f} "
                  f"step={step:6.3f} best={best_score:.6f}")

        it += 1
    return best, best_score

def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer based on the exact sub‑gradient of
    the active set (identical to the version in the reference code).
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient for the active shifts
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i                     # the mirrored indices
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # attempt a fairly large move first
        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy it={it} improved to {best_score:.6f}")
            continue

        # try a reduced move
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy (small) it={it} improved to {best_score:.6f}")
            continue

        # occasional jitter
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score

def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk with exponential cooling, exactly as in the
    reference implementation.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03               # initial temperature (objective units)
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer up to 10 % of source mass
        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        # exponential cooling
        T *= 0.9997

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine builds many analytic
    seeds, refines the best one with a smoothed gradient descent,
    follows by the aggressive greedy optimiser, and finishes with a
    brief simulated‑annealing fine‑tuning.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a safety margin

    # ------------------------------------------------------------------
    # 1) choose a length – a few thousand gives a good trade‑off
    n = 4000
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 2) generate initial candidates
    candidates = _initial_candidates(n, rng)

    # possibly add a previously found solution
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size == n:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3) pick the best seed (according to the true objective)
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx} score={sc:.6f}")

    if best_vec is None:
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) smoothed gradient descent – the core of the new approach
    if time.time() < deadline:
        best_vec, best_score = _smooth_descent(best_vec, deadline, rng)
        print(f"[{time.time():.1f}] after smooth descent best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) aggressive greedy optimisation (as in the reference code)
    if time.time() < deadline:
        best_vec, best_score = _greedy_local_search(best_vec, deadline, rng)
        print(f"[{time.time():.1f}] after greedy best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) simulated‑annealing fine‑tuning (if time permits)
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)
        print(f"[{time.time():.1f}] after SA best={best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity – ensure we are on the simplex (sum = 1, non‑negative)
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
