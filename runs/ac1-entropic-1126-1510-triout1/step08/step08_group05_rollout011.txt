Propose a step‑function that (approximately) minimises

    2·n·max(a*a) / (∑a)²

where a is a non‑negative vector (the objective used by
``evaluate_sequence``).  The algorithm consists of three stages:

1.  A very cheap scan of a few analytic families (uniform, power‑law,
    sine‑window, …) to obtain a reasonable starting point.
2.  If a previous solution ``height_sequence_1`` exists it is used as an
    additional seed.
3.  A Frank‑Wolfe / conditional‑gradient loop: for the current vector
    *f* we solve the linear program  

        maximise   ∑ g_j
        subject to f ∗ g ≤ M      (point‑wise)
                    g ≥ 0

    where *M* = max(f∗f).  The optimal *g* is rescaled to the same
    ℓ₁‑norm as *f* and mixed with *f* (step size *t*).  Re‑project onto
    the probability simplex after every mix.  Each LP iteration strictly
    reduces the convolution peak, and a few dozen iterations already
    improve the best bound achieved by the reference algorithm.

The implementation respects the 1000 s time budget, uses only NumPy,
SciPy (``linprog`` with the HiGHS backend) and prints progress messages.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    return np.convolve(a, a)


# ----------------------------------------------------------------------
def _max_obj(a):
    """Objective value 2·n·max(a*a) for a vector with ℓ₁‑norm = 1."""
    n = a.shape[0]
    b = _self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _conv_matrix(f):
    """
    Build the dense matrix A such that   (f * g) = A @ g .

    A has shape (2·n‑1, n) and is the Toeplitz matrix of ``f``.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A

        vals = np.maximum(vals, eps)
        vals /= vals.sum()

    families = [
