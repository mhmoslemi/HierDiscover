   projected sub‑gradient method with a diminishing stepsize is run for
   the bulk of the remaining budget.

5. **Guided hill‑climbing & random restarts** – repeatedly move a tiny
   amount of mass from indices with a large positive sub‑gradient to
   those with a large negative sub‑gradient.  After a fixed number of
   moves a small Gaussian jitter is added and a short sub‑gradient
   refinement is performed; this helps to escape plateaus.

All operations are vectorised and use FFT‑based convolutions for speed.
The routine respects the 1000 s time budget (a safety margin is kept)

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) assuming sum(a)=1 (the exact objective)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k (counted twice)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using log‑sum‑exp:
    f_eps(a) = eps * log Σ_i exp(conv_i/eps).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_idx = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_idx] -= w[2 * diag_idx] * a[diag_idx]

    return grad

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 50_000,
                          eta0: float = 1.0):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _cheap_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _cheap_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 50_000,
                  step0: float = 0.4,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _cheap_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _cheap_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 40_000,
                         eps0: float = 5e-2,
                         step0: float = 0.35,
                         beta1: float = 0.9, beta2: float = 0.999):
    """Adam on the log‑sum‑exp smoothed objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _cheap_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # tighten smoothing occasionally
        if t % 25_000 == 0 and eps > 1e-6:
            eps *= 0.5

        cur = _cheap_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 600_000,
                         eta0: float = 0.6):
    """Plain projected sub‑gradient descent with diminishing steps."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _cheap_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _cheap_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families -------------------------------------------------------

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0            # map to [-1,1]
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    """Discrete Beta(a,b) on (0,1) – a,b>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow/underflow
    logw = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x)
    # replace -inf with a large negative number
    logw = np.where(np.isfinite(logw), logw, -1e300)
    w = np.exp(logw)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w.sum()
    return w


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α (α=1 → uniform on simplex)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} exp‑grad → {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                max_iters=80_000,
                                step0=0.4)
        if sc < best_score - 1e-15:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam → {sc:.6f}")

        # Smooth‑Adam
        seq, sc = _smooth_adam_descent(seed_seq, deadline,
                                       max_iters=60_000,
                                       eps0=5e-2,
                                       step0=0.35)
        if sc < best_score - 1e-15:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smooth‑Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient descent on the current best
    subgrad_deadline = deadline - 30.0          # reserve ~30 s for hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=600_000,
                                       eta0=0.6)
        if sc < best_score - 1e-15:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained hill‑climbing with occasional random restarts
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02                       # up to 2 % of total mass initially

    while time.time() < deadline:
        iteration += 1
        # geometric decay of transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        # choose source and destination indices
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _cheap_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter + short local sub‑gradient refinement
        if iteration % 5000 == 0 and np.random.rand() < 0.15:
            jitter_sigma = 5e-5 * (1.0 + iteration / 5000.0)
            jitter = np.random.randn(n) * jitter_sigma
            a_tmp = best_seq + jitter
            a_tmp = _project_onto_simplex(a_tmp, s=1.0)
            # short sub‑gradient run (few thousand iters) on the perturbed point
            sub_dead = time.time() + 2.0   # 2 s for this quick refinement
            a_tmp, _ = _subgradient_descent(a_tmp, sub_dead, max_iters=20_000, eta0=0.5)
            tmp_score = _cheap_score(a_tmp)
            if tmp_score < cur_score - 1e-15:
                a = a_tmp
                cur_score = tmp_score
                if tmp_score < best_score - 1e-14:
                    best_score, best_seq = tmp_score, a_tmp.copy()
                    print(f"[{time.time():.1f}] jitter‑refine iter={iteration} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # Return as plain Python list as required by ``evaluate_sequence``.
    return best_seq.tolist()
