def _smooth_subgrad(a, alpha):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    maxb = b.max()
    # compute weights
    # Use stable exponent: w = np.exp(alpha * (b - maxb))
    # subtract maxb to avoid overflow
    w = np.exp(alpha * (b - maxb))
    s = w.sum()
    if s == 0 or np.isinf(s):
        # fallback to binary weights of max indices
        active = np.flatnonzero(b >= maxb - 1e-12 * (abs(maxb)+1))
        w = np.zeros_like(b)
        w[active] = 1.0
        s = w.sum()
    w /= s
    # compute grad = 2 * conv_rev(a_rev, w)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    # note: diag correction not needed for weighted sum (sum of w_k b_k)
    return grad, maxb

def _smooth_descent(a0, deadline, alpha_start=1.0, alpha_end=40.0, step0=0.3, max_iters=200_000):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute alpha schedule
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        # compute grad
        grad, cur_max = _smooth_subgrad(a, alpha)

        eta = step0 / np.sqrt(it)

        a = _project_onto_simplex(a, s=1.0)

"""
Propose a step‑function with a lower upper‑bound.

Algorithm outline
-----------------
1.  Choose a few problem sizes (≈2 k – 9 k) and generate many *seed*
    sequences (analytic families + Dirichlet draws).  All seeds are
    ℓ₁‑normalised (∑a = 1) and scored by the cheap surrogate
    `2·n·max(conv(a,a))`.

2.  Keep the best ≈15 seeds and run a **smoothed‑gradient descent**
    on the simplex for each of them.  The smoothed objective is
    ``f(a)=∑_k w_k·b_k`` with `b = a*a` and weights
    `w_k ∝ exp(α·(b_k−max b))` (α grows from 1 to ≈40).  Its gradient
    is `∇f = 2·conv_rev(a_rev, w)`.  The step is projected back onto the
    simplex and the best point seen is remembered.

3.  A Frank‑Wolfe‑style *mass‑transfer* (move probability mass from the
    component with the largest gradient to the one with the smallest)
    polishes the best candidate.

4.  A final simulated‑annealing phase performs random small transfers
    with a slowly decreasing temperature, accepting occasional uphill
    moves.

The whole routine respects a 1000 s budget and always returns a list
of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Surrogate score 2·n·max(conv(a,a)); exact when the ℓ₁‑norm of ``a`` is 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_subgrad(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed objective
        f(a) = Σ_k w_k·b_k  with  w_k ∝ exp(α·(b_k‑max b)).
    Returns (grad, current_max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()

    # stable soft‑max weights (max b at 0 gives 1, others ≤ 1)
    w = np.exp(alpha * (b - cur_max))
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        # fall back to the binary sub‑gradient used in the original code
        active = np.flatnonzero(b >= cur_max - 1e-12 * (abs(cur_max) + 1.0))
        w = np.zeros_like(b)
        w[active] = 1.0
        w_sum = w.size  # will be normalised later
    w /= w_sum

    # grad = 2·conv_rev(a_rev, w)  (no diagonal correction needed)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]
    return grad, cur_max


def _smooth_descent(a0: np.ndarray, deadline: float,
                    alpha_start: float = 1.0,
                    alpha_end: float = 40.0,
                    step0: float = 0.25,
                    max_iters: int = 100_000):
    """
    Smoothed gradient descent on the simplex.
    Returns the best point seen and its surrogate score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linearly increase the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iters)

        grad, cur_max = _smooth_subgrad(a, alpha)

        # simple diminishing step
        eta = step0 / np.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()

        # occasional tiny jitter to avoid stagnation
        if t % 4000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = _project_onto_simplex(a + jitter, s=1.0)

    return best_a, best_score

def _subgrad(a: np.ndarray):
    """
    Original sub‑gradient (binary weight on the active set) used for
    comparison and for the Frank‑Wolfe polish.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families (all ℓ₁‑normalised)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _gaussian_like_distribution(n: int, sigma: float = None) -> np.ndarray:
    """bell‑shaped distribution centred in the middle of the interval."""
    if sigma is None:
        sigma = n / 5.0
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals
