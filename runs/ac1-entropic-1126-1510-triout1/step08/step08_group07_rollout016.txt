def _smooth_gradient_opt(a0, n, deadline, rng):
    """Run a homotopy gradient descent based on a smoothed max objective.
    Returns the best vector and its exact objective (2·n·max conv)."""
    a = a0.copy()
    best_a = a.copy()
    # compute initial objective
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    size = 1 << ((3 * n - 2) - 1).bit_length()   # power of two >= 3n-2
    alpha = 5.0
    alpha_max = 5000.0
    # schedule parameters
    factor = 1.8
    while time.time() < deadline and alpha <= alpha_max:
        # step size that scales with alpha
        lr = 0.5 / max(alpha, 1.0)
        # how many inner iterations at this alpha
        inner_iters = max(1000, int(2e6 / n))   # ~2000 for n~5000
        for it in range(inner_iters):
            if time.time() >= deadline:
                break
            # FFT of current a (size fixed)
            A = np.fft.rfft(a, n=size)
            # self‑convolution via A*A
            conv_full = np.fft.irfft(A * A, n=size)
            conv = conv_full[: 2 * n - 1]
            maxc = conv.max()
            cur_score = 2.0 * n * maxc
            # smoothed weights
            exp_terms = np.exp(alpha * (conv - maxc))
            w = exp_terms / exp_terms.sum()
            # FFT of weights
            W = np.fft.rfft(w, n=size)
            # correlation (gradient of smoothed objective)
            corr_full = np.fft.irfft(A * np.conj(W), n=size)
            g = 2.0 * corr_full[:n]
            # multiplicative update (mirror descent on simplex)
            a_new = a * np.exp(-lr * g)
            # protect against all zero
            sum_new = a_new.sum()
            if sum_new == 0.0 or not np.isfinite(sum_new):
                a_new = np.maximum(a + 1e-12 * rng.random(n), 1e-12)
                a_new /= a_new.sum()
            else:
                a_new /= sum_new
            # compute exact objective for the new candidate
            conv_new = _fft_self_convolve(a_new)
            new_score = 2.0 * n * conv_new.max()
            # keep best
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] n={n} alpha={alpha:.2f} it={it} improved to {best_score:.6f}")
            # accept if non‑worsening, otherwise keep old a
            if new_score <= cur_score:
                a = a_new
            # occasional jitter to escape local plateaus
            if it % 500 == 0:
                if rng.random() < 0.1:
                    idx = rng.integers(n)
                    eps = min(a[idx] * 0.05, 1e-3)
                    a[idx] = max(a[idx] - eps, 0.0)
                    a /= a.sum()
        alpha *= factor
    return best_a, best_score

"""
Search for a non‑negative step function (probability vector) that minimises
``evaluate_sequence``.  The method uses a smooth‑maximum (log‑sum‑exp)
approximation of the self‑convolution peak and performs a homotopy gradient
descent on the simplex.  Starting from an arcsine-shaped distribution (close
to the continuous optimum) for several sequence lengths, the smoothing
parameter α is increased while the vector is updated by a multiplicative
mirror‑descent step and re‑normalised.  Small random jitter is added occasionally.
The best vector found before the time budget expires is returned.
"""

import time
import math
import numpy as np


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _smooth_gradient_opt(a0: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Homotopy gradient descent on the smoothed max of the self‑convolution.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    # exact objective of the start vector
    cur_score = _objective(a)
    best_a = a.copy()
    best_score = cur_score

    # FFT size that accommodates both convolution (2n‑1) and correlation (3n‑2)
    size = 1 << ((3 * n - 2) - 1).bit_length()

    alpha = 5.0          # start smoothing
    alpha_max = 5000.0
    factor = 1.8         # growth of α each outer epoch

    while time.time() < deadline and alpha <= alpha_max:
        # learning rate scaling with α (large α → small steps)
        lr = 0.5 / max(alpha, 1.0)

        # a few thousand inner steps per α
        inner_iters = max(1000, int(2e6 / n))

        for it in range(inner_iters):
            if time.time() >= deadline:
                break

            # ----- FFT of current a -----
            A = np.fft.rfft(a, n=size)

            # ----- self‑convolution (full) -----
            conv_full = np.fft.irfft(A * A, n=size)
            conv = conv_full[: 2 * n - 1]
            maxc = conv.max()
            cur_score = 2.0 * n * maxc

            # ----- smoothed weight vector w (length 2n‑1) -----
            # w_k ∝ exp(α·(conv_k – maxc))  → numerically stable
            exp_terms = np.exp(alpha * (conv - maxc))
            w = exp_terms / exp_terms.sum()

            # ----- FFT of w -----
            W = np.fft.rfft(w, n=size)

            # ----- correlation A ⋆ w = A * conj(W) -----
            corr_full = np.fft.irfft(A * np.conj(W), n=size)
            g = 2.0 * corr_full[:n]          # gradient of the smoothed objective

            # ----- mirror‑descent update (multiplicative) -----
            a_new = a * np.exp(-lr * g)
            s = a_new.sum()
            if s <= 0.0 or not np.isfinite(s):
                # fall back to a tiny random perturbation if something went wrong
                a_new = np.maximum(a + 1e-12 * rng.random(n), 1e-12)
                a_new /= a_new.sum()
            else:
                a_new /= s

            # ----- exact objective of the candidate -----
            new_score = _objective(a_new)

            # keep the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] n={n} α={alpha:.2f} it={it} "
                      f"new best {best_score:.6f}")

            # accept non‑worsening moves (helps stability)
            if new_score <= cur_score:
                a = a_new

            # occasional jitter to escape plateaus
            if it % 500 == 0 and rng.random() < 0.1:
                idx = rng.integers(n)
                eps = min(a[idx] * 0.05, 1e-3)
                a[idx] = max(a[idx] - eps, 0.0)
                a /= a.sum()

        alpha *= factor

    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Run the optimisation described above and return the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    best_vec = None
    best_score = float("inf")

    # sequence lengths to try – longer vectors give more flexibility
    lengths = [3000, 5000, 8000, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- two initialisations: arcsine (structured) and random (exploratory) -----
        a_arc = _distribution_arcsine(n)
        a_arc = np.maximum(a_arc + 1e-6 * rng.random(n), 0.0)
        a_arc /= a_arc.sum()

        a_rand = rng.random(n)
        a_rand = np.maximum(a_rand, 1e-12)
        a_rand /= a_rand.sum()

        for start_id, a_start in enumerate([a_arc, a_rand]):
            if time.time() >= deadline:
                break
            a_opt, score = _smooth_gradient_opt(a_start, n, deadline, rng)
            if score < best_score:
                best_score = score
                best_vec = a_opt.copy()
                print(f"[{time.time():.1f}] global new best {best_score:.6f} "
                      f"len={n} start={start_id}")
                if best_score < 1.503:          # target reached – we can stop early
                    deadline = time.time()        # force exit loops

    # fallback in the unlikely case nothing was produced
    if best_vec is None:
        best_vec = _distribution_uniform(2000)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # final sanity: ensure we are on the simplex (sum = 1, non‑negative)
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
