"""
A time‑bounded optimiser for the “max‑convolution on the simplex” problem.

The algorithm works in three stages:

1. **Broad sampling** – many analytical families (arcsine, beta, exponential,
   Gaussian, Cauchy, power‑transformed arcsine) are generated with a variety of
   lengths.  Each candidate is ℓ₁‑normalised (sum = 1) and scored with the exact
   surrogate  ``2·n·max(conv(a,a))`` (identical to ``evaluate_sequence`` for a
   normalised vector).

2. **Active‑set LP refinement** – Starting from a seed we repeatedly:
      * compute the current maximum convolution value ``M`` and the tiny set of
        active indices where this maximum is attained,
      * solve a linear programme
            maximise   ∑ g_j
            subject to (a * g)_k ≤ M   for every active k,
                        g_j ≥ 0,
        which pushes mass onto coordinates that are “light” for the active
        constraints,
      * normalise the LP solution to the simplex and mix it with the current
        vector (convex combination with a small step size).  This is the
        Frank‑Wolfe / linear‑programming boost described in the reference
        literature and is very effective at lowering the ℓ∞‑norm of the
        self‑convolution.

3. **Local polishing** – a few short Adam / exponentiated‑gradient steps are
   applied to the best vector found to clean up any remaining sub‑gradient
   noise.

All phases respect a global budget of 1000 s; the function returns the best
sequence it discovered (a list of non‑negative floats).  The implementation
uses only ``numpy``, ``scipy`` and the provided ``evaluate_sequence``.
"""

import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import linprog

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a)) for a with sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide="ignore", invalid="ignore"):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) point on the simplex."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# sub‑gradient (used for Adam / SGDs) ---------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(conv(a,a)) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)
    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the simplex (uses the exact sub‑gradient)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _active_set_lp_step(a: np.ndarray, deadline: float,
                       max_iters: int = 12,
                       step_size: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the “maximise ∑g subject to (a∗g)≤M” LP on the active
    constraints and mix the solution back into ``a``.
    Returns (improved_a, score).  Stops early if no progress is possible.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # 1. current convolution and active set
        conv = _fft_self_convolve(best_a)
        M = conv.max()
        eps = max(1e-12 * M, 1e-14)
        active = np.flatnonzero(conv >= M - eps)
        if active.size == 0:
            break

        # 2. build the LP matrix for the active rows (dense – small)
        m = active.size
        A = np.zeros((m, n), dtype=np.float64)
        for row_idx, k in enumerate(active):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1, dtype=int)
            A[row_idx, js] = best_a[k - js]

        b = np.full(m, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)      # maximise sum(g)  <=> minimise -sum(g)
        bounds = [(0, None)] * n

        # time limit for the LP (few seconds at most)
        remaining = max(0.0, deadline - time.time())
        opts = {"presolve": True}
        if remaining > 0.1:
            opts["time_limit"] = remaining

        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=bounds,
                      method="highs",
                      options=opts)

        if not res.success:
            # LP failed – abort the inner loop
            break

        g0 = res.x
        sum_g0 = g0.sum()
        # If the LP does not increase the ℓ₁‑mass, we are at a (local) optimum
        if sum_g0 <= 1.0 + 1e-12:
            break

        # Normalise to the simplex
        g = g0 / sum_g0

        # 3. convex blend
        a_new = (1.0 - step_size) * best_a + step_size * g
        a_new = _project_onto_simplex(a_new, s=1.0)

        new_score = _max_convolution_score(a_new)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best_a = a_new
            # keep the same step size, continue
            it += 1
            # optional progress print
            print(f"[{time.time():.1f}] LP‑iter {it:<3d}  score={best_score:.6f}")
        else:
            # If not improved try a smaller step; if still no improvement → stop.
            step_small = step_size * 0.5
            a_tmp = (1.0 - step_small) * best_a + step_small * g
            a_tmp = _project_onto_simplex(a_tmp, s=1.0)
            tmp_score = _max_convolution_score(a_tmp)
            if tmp_score < best_score - 1e-14:
                best_score = tmp_score
                best_a = a_tmp
                step_size = step_small
                it += 1
                print(f"[{time.time():.1f}] LP‑iter {it:<3d} (small step) score={best_score:.6f}")
            else:
                # no improvement possible
                break
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the supplied time budget and
    returns the best vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # pick a sensible base length – either from a previous run or a default
    try:
        base_n = len(height_sequence_1)    # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # collection of lengths (avoid extremes that are too cheap or too costly)
    lengths = set()
    for factor in (0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * factor)
        if 800 <= n <= 18000:
            lengths.add(n)
    for _ in range(6):
        n = np.random.randint(800, min(18000, int(base_n * 2.5)) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # families & hyper‑parameter sweeps
    beta_ps = list(np.arange(0.05, 0.55, 0.05)) + list(np.arange(0.6, 2.1, 0.2)) + [2.5]
    beta_params = [0.3, 0.6, 1.0, 1.5, 2.0, 3.0, 5.0]
    power_exps = [0.3, 0.6, 0.9, 1.2, 1.5]
    dirichlet_alphas = [0.3, 0.7, 1.0, 2.0]

    candidates = []   # (array, score)

    # ------------------------------------------------------------------
    # 1.  broad sampling
    for n in lengths:
        if time.time() >= deadline:
            break

        # analytic families
        for gen, name in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.006), "exponential"),
            (lambda m: _gaussian_distribution(m, 0.12), "gaussian"),
            (lambda m: _cauchy_distribution(m, 0.12), "cauchy"),
        ):
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} {name:10s} score={candidates[-1][1]:.6f}")

        # symmetric beta sweep
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # unsymmetric beta
        for alpha in beta_params:
            for beta in beta_params:
                if time.time() >= deadline:
                    break
                a = _beta_distribution(n, alpha, beta)
                candidates.append((a, _max_convolution_score(a)))

        # power‑transformed arcsine / beta0.5 / beta1.0
        for base_func, base_name in (
            (_arcsine_distribution, "arcsine"),
            (lambda m: _beta_symmetric_distribution(m, 0.5), "beta0.5"),
            (lambda m: _beta_symmetric_distribution(m, 1.0), "beta1.0"),
        ):
            base = base_func(n)
            for pwr in power_exps:
                if time.time() >= deadline:
                    break
                a = np.power(base, pwr)
                a /= a.sum()
                candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws
        for alpha in dirichlet_alphas:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 2. include the previous best if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. keep a few elite seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(7, len(candidates))
    seeds = [c[0] for c in candidates[:top_k]]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    best_seq = seeds[0].copy()
    best_score = _max_convolution_score(best_seq)

    # ------------------------------------------------------------------
    # 4. active‑set LP refinement on each seed (most impactful)
    for idx, seed_seq in enumerate(seeds):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] LP‑refine seed {idx+1}/{len(seeds)} (n={seed_seq.size})")
        a_ref, sc_ref = _active_set_lp_step(seed_seq, deadline,
                                            max_iters=12,
                                            step_size=0.25)
        if sc_ref < best_score - 1e-14:
            best_score, best_seq = sc_ref, a_ref.copy()
            print(f"[{time.time():.1f}]  ** new global best after LP {best_score:.6f}")

        # short Adam polishing (only if time remains)
        if time.time() < deadline - 2.0:
            a_adam, sc_adam = _adam_descent(a_ref, deadline,
                                            max_iters=80_000,
                                            step0=0.15)
            if sc_adam < best_score - 1e-14:
                best_score, best_seq = sc_adam, a_adam.copy()
                print(f"[{time.time():.1f}]  ** new global best after Adam {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. final light polishing on the global best (if time permits)
    if time.time() < deadline - 5.0:
        a_final, sc_final = _adam_descent(best_seq, deadline,
                                          max_iters=120_000,
                                          step0=0.1)
        if sc_final < best_score - 1e-14:
            best_score, best_seq = sc_final, a_final.copy()
            print(f"[{time.time():.1f}] final Adam polish gave best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
