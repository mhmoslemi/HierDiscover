def _adam_softmax(seed_vec, sub_deadline, best_seq, best_score):
    n = seed_vec.shape[0]
    # ensure no zeros: add epsilon
    eps = 1e-12
    u = np.log(seed_vec + eps)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    ada_eps = 1e-8
    
    schedule = [...]
    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= sub_deadline:
                return best_seq, best_score
            t += 1
            # compute a from u
            a = np.exp(u - np.max(u))
            a = a / a.sum()
            # compute conv and gradient
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum
            # gradient wrt a
            conv_rev = _fft_convolve(a[::-1], w)
            grad_a = 2.0 * conv_rev[n-1:n-1+n]
            # gradient wrt u
            # grad_u = a * (grad_a - np.dot(grad_a, a))  (since softmax)
            dot = np.dot(grad_a, a)
            grad_u = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            u = u - lr * m_hat / (np.sqrt(v_hat) + ada_eps)
            # occasional evaluation with exact objective
            if t % 500 == 0:
                cur = 2.0 * n * b.max()
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha} step={t} new best {best_score:.6f}")
    # after schedule, compute final a
    final_a = np.exp(u - np.max(u))
    final_a = final_a / final_a.sum()
    cur = _objective(final_a)
    if cur < best_score:
        best_score = cur
        best_seq = final_a.copy()

"""
propose_candidate – enhanced search for the min‑max convolution step function.

Algorithm
---------
1. **Broad initialisation** – for several lengths (2k … 30k) we build a
   large pool of analytic families (uniform, arcsine, beta(p,p) with
   ultra‑small p, power‑law edges with γ up to 0.9, exponentials,
   sparse random Dirichlet draws, etc.).  Their exact objective
   ``2·n·max(conv(a,a))`` is evaluated and the best few are kept as seeds.

2. **Soft‑max Adam descent** – each seed is optimised by Adam on the
   *smooth‑max* approximation
        L_α(a)= (1/α)·log Σ exp(α·(a*a)_k).
   The variables are the logits `u` of a soft‑max (a_i = exp(u_i)/Σexp(u)),
   which automatically guarantees non‑negativity and sum‑to‑one.
   A schedule of increasing smoothing parameters α and decreasing learning
   rates drives the optimisation to a (near) exact optimum.  Whenever the
   true objective improves we store the new best vector.

3. **Fine‑grained local walk** – a cheap Metropolis‑type move swaps a
   random amount of mass between two indices and is accepted only if the
   true (unsmoothed) objective gets better.  This often removes the last
   remaining “bumps”.

The whole pipeline respects a 1000 s wall‑time budget; the best vector
found is returned as a plain Python list of floats (all entries are ≥0 and
the sum is 1, so ``evaluate_sequence`` will use the same value for the
objective).
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=s} – used only by fallback code."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (a is assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a probability vector that sums to 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) – clamp
    eps = 1e-15
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse Dirichlet‑like start (non‑zero at a few random indices)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k) + 1e-6   # avoid exact zero
    vals /= vals.sum()
    return vals


def _distribution_random_dirichlet(n: int, rng: np.random.Generator,
                                  concentration: float = 0.2) -> np.ndarray:
    """Fully dense Dirichlet sample – useful for exploring the interior."""
    a = rng.gamma(concentration, 1.0, size=n)
    a += 1e-12
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A generous pool of start vectors for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # beta family – many shape parameters, include ultra‑U‑shaped
    for p in (0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7,
              1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – wide range (very flat to very spiky)
    for lam in (0.0005, 0.001, 0.002, 0.004, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # power‑law edges – γ up to 0.9
    for gamma in (0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.5, 0.6, 0.7,
                  0.8, 0.9):
        cand.append(_distribution_powerlaw_edges(n, gamma))

    # two‑block spikes – smaller block fractions
    for block_frac in (0.02, 0.04, 0.07, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random sparse and dense Dirichlet starts
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.05))
    cand.append(_distribution_random_dirichlet(n, rng, concentration=0.2))
    cand.append(_distribution_random_dirichlet(n, rng, concentration=0.5))

    return cand


# ----------------------------------------------------------------------
def _adam_softmax(seed_vec: np.ndarray, deadline: float,
                  best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max objective using a soft‑max
    parametrisation (logits u, a = softmax(u)).  The schedule gradually
    increases the smoothing parameter α and reduces the learning rate.
    """
    n = seed_vec.shape[0]
    eps = 1e-12

    # initialise logits – keep a tiny epsilon to avoid log(0)
    u = np.log(np.maximum(seed_vec, eps))

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    t = 0
    beta1, beta2, ada_eps = 0.9, 0.999, 1e-8

    # schedule of (α, steps, learning‑rate)
    schedule = [
        (0.5, 2500, 0.30),
        (1.0, 2500, 0.25),
        (5.0, 2500, 0.20),
        (20.0, 2500, 0.15),
        (100.0, 2000, 0.12),
        (500.0, 1500, 0.10),
        (2000.0, 1200, 0.08),
        (5000.0, 800, 0.06),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                # early exit – return the best we have so far
                return best_seq, best_score

            # ---------------------------------------------------------
            # soft‑max projection
            a = np.exp(u - np.max(u))          # stabilised exp
            a /= a.sum()

            # conv and smooth‑max weights
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))     # shifted for stability
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient wrt a (as derived in the original code)
            conv_rev = _fft_convolve(a[::-1], w)
            grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

            # gradient wrt logits u (soft‑max Jacobian)
            dot = np.dot(grad_a, a)
            grad_u = a * (grad_a - dot)

            # Adam update on u
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_u
            v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            u = u - lr * m_hat / (np.sqrt(v_hat) + ada_eps)

            # occasional exact evaluation (every few hundred steps)
            if t % 500 == 0:
                cur_obj = 2.0 * n * maxb
                if cur_obj < best_score - 1e-13:
                    best_score = cur_obj
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    # final evaluation after the schedule finishes
    a = np.exp(u - np.max(u))
    a /= a.sum()
    cur_obj = _objective(a)
    if cur_obj < best_score:
        best_score = cur_obj
        best_seq = a.copy()
        print(f"[{time.time():.1f}] post‑schedule new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing: move a random fraction of mass from i → j if the
    true objective improves.  Repeated until time runs out."""
    n = a.shape[0]
    cur_seq = a.copy()
    cur_score = _objective(cur_seq)

    while time.time() < deadline:
        # pick a source that actually carries mass
        src_candidates = np.where(cur_seq > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # pick a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer – up to 30 % of the source mass
        delta = cur_seq[i] * np.random.rand() * 0.30
        if delta < 1e-14:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_score = _objective(new_seq)
        if new_score < cur_score - 1e-13:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point – returns a list of non‑negative heights of a step
    function that (hopefully) improves on the previous best bound.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # 1.  Scan a range of lengths with many analytic families.
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]
    all_candidates = []   # (score, vector)
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # analytic families
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            # guarantee strictly positive entries for the soft‑max stage
            cand = np.maximum(cand, 1e-12)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best length={n} score={sc:.6f}")
            all_candidates.append((sc, cand.copy()))

        # a few extra dense Dirichlet draws
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_dirichlet(n, rng, concentration=0.2)
            cand = np.maximum(cand, 1e-12)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (Dirichlet) length={n} score={sc:.6f}")
            all_candidates.append((sc, cand.copy()))

    # keep only a handful of promising seeds (memory & time bound)
    all_candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for _, vec in all_candidates[:6]]

    # ------------------------------------------------------------------
    # 2.  Adam+soft‑max descent on the best seeds.
    remaining = deadline - time.time()
    if remaining > 30 and top_seeds:
        # distribute the remaining budget equally among the seeds
        per_seed = max(1, remaining / len(top_seeds))
        for seed_vec in top_seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            overall_best_seq, overall_best_score = _adam_softmax(
                seed_vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            # keep the returned sequence as current best for the next seed
            # (helps propagate improvements)
            print(f"[{time.time():.1f}] after seed optimisation score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final fine‑grained local optimisation (pairwise swaps).
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after local swaps best score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
