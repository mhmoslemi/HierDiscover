def _hill_climb(a, deadline):
    cur_score = _max_convolution_score(a)
    best_seq = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.05 # start 5% maybe
    while time.time() < deadline:
        it += 1
        transfer = max_transfer * (0.5 ** (it / 10000.0))
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i: j+=1
        delta = np.random.rand() * min(transfer, a[i])
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill climb new best {best_score:.6f}")
        # jitter
        if it % 5000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a)
            cur_score = _max_convolution_score(a)
    return best_seq, best_score
```

"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Generate many seeds** – a variety of analytic families
   (arcsine, triangular, Hann, Blackman, Beta, exponential, sine,
   cosine, uniform) and random Dirichlet draws are normalised to the
   simplex.  A few mixtures of two families are also added.

2. **Select the most promising seeds** – keep the `top_k`
   candidates with the smallest cheap surrogate
   `2·n·max(a*a)` (exactly the objective when the sum of the vector
   equals 1).

3. **Intensive optimisation of each seed** – the allocated time for a
   seed is split among several gradient‑based optimisers that all keep
   the vector on the simplex:
   * exponentiated‑gradient (mirror descent)
   * Adam
   * continuation / smoothed‑max descent (soft‑max of the
     convolution with a growing temperature)
   * plain sub‑gradient descent

   The previous implementation used the same deadline for all four
   methods, so only the first one actually ran.  Here each method gets
   its own sub‑budget, allowing the full optimisation pipeline to be
   executed.

4. **Fine‑grained hill‑climbing** – after the seeds have been
   optimised, a simple random‑mass‑transfer hill‑climber (with a
   slowly decreasing transfer amount and occasional jitter) is run
   until the overall time budget expires.

All operations respect the global time limit (`budget_s`, default 1000 s)
and the vector is always projected onto the probability simplex,
ensuring that the final sequence is valid for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT‑based convolutions (real signals)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Cheap surrogate (exact when sum(a)=1)
# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (exact)
# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of `max(a*a)` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Smoothed max (soft‑max) gradient – continuation method
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()

    # soft‑max weights (stable)
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


# ----------------------------------------------------------------------
# Optimisers – each receives a *budget* (seconds) rather than a fixed
# iteration count.  They stop when the time budget is exhausted.
# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, budget: float,
                          max_iters: int = 200_000, eta0: float = 0.9):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    end = time.time() + budget
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < end and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a /= a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, budget: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    end = time.time() + budget
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < end and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray, budget: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation: start with a smooth max (γ≈5) and increase γ."""
    end = time.time() + budget
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < end and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, budget: float,
                         max_iters: int = 400_000, eta0: float = 0.7):
    """Plain sub‑gradient descent with diminishing step size."""
    end = time.time() + budget
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < end and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Random & analytic families
# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return vals / vals.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    vals = (a0
            - a1 * np.cos(2.0 * np.pi * i / (n - 1))
            + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Hill‑climbing (tiny random mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float):
    """Very simple random‑mass‑transfer local search."""
    cur_score = _max_convolution_score(a)
    best_seq = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.04  # start with up to 4 % of total mass
    while time.time() < deadline:
        it += 1
        # geometric decay of transfer amount
        transfer = max_transfer * (0.5 ** (it / 12000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

                q = np.random.choice(beta_params)

            else:   # exponential

            print(f"[{time.time():.1f}] sub‑grad → {sc:.6f}")

        best_seq, best_score = _hill_climb(best_seq, hill_deadline)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
