def _local_optimize_line_search(a, deadline, rng):
    best_a = a.copy()
    best_score = _objective(best_a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = np.where(b >= maxb - eps)[0]

        # compute gradient
        g = np.zeros_like(a)
        n = a.shape[0]
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # source = index with max gradient among positive a
        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # line search for delta in [0, a[src]]
        lo = 0.0
        hi = a[src]
        # if hi == 0, continue
        # Use golden-section search
        phi = (1 + np.sqrt(5)) / 2
        resphi = 2 - phi
        # Initialize points
        x1 = lo + resphi * (hi - lo)
        x2 = hi - resphi * (hi - lo)
        f1 = _objective(_transfer(a, src, dest, x1))
        f2 = _objective(_transfer(a, src, dest, x2))
        for _ in range(15):  # limited iterations
            if f1 < f2:
                hi = x2
                x2 = x1
                f2 = f1
                x1 = lo + resphi * (hi - lo)
                f1 = _objective(_transfer(a, src, dest, x1))
            else:
                lo = x1
                x1 = x2
                f1 = f2
                x2 = hi - resphi * (hi - lo)
                f2 = _objective(_transfer(a, src, dest, x2))
            if time.time() >= deadline:
                break

        # choose best among lo, hi, x1, x2
        candidates = [(lo, _objective(_transfer(a, src, dest, lo))),
                      (hi, _objective(_transfer(a, src, dest, hi))),
                      (x1, f1), (x2, f2)]
        delta_opt, best_local_score = min(candidates, key=lambda t: t[1])
        if best_local_score < best_score - 1e-12:
            a = _transfer(a, src, dest, delta_opt)
            best_a = a.copy()
            best_score = best_local_score

"""
Greedy line‑search optimiser for the self‑convolution max‑norm.

The routine builds a rich family of analytic seed vectors (uniform,
arcsine, symmetric beta, exponentials, alternating patterns,
random Dirichlet and sparse vectors).  The best seed (by the objective
2·n·max(conv(a,a))) is taken and refined by an aggressive pairwise
mass‑transfer method.  In each iteration the source index (largest
gradient) and destination index (smallest gradient) are identified.
Instead of a fixed step size the algorithm performs a cheap golden‑section
search on the amount of mass to move, guaranteeing the best possible
improvement along that direction.  After the greedy phase a few
low‑pass smoothing steps are applied, followed by short simulated‑
annealing walks to escape possible plateaus.

All operations respect the simplex (non‑negative entries, sum = 1) and
run under the user‑supplied time budget (default 1000 s).  The final
vector is returned as a plain Python list, ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Objective compatible with evaluate_sequence when sum(a)=1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid under/overflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    if w.sum() == 0.0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a

def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:

    cand = []

    for lam in (0.0003, 0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

        vec = rng.gamma(alpha, 1.0, size=n)
        vec = _project_onto_simplex(vec)
        cand.append(vec)

    # random sparse starts (≈0.5 % non‑zeros)
    k = max(2, n // 200)
    sparse = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    sparse[idx] = rng.random(k)
    sparse = _project_onto_simplex(sparse)
    cand.append(sparse)

    # noisy perturbations of a few beta shapes
    for _ in range(4):

        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)

    return cand

def _transfer(a: np.ndarray, src: int, dest: int, delta: float) -> np.ndarray:
    """Return a copy of a with `delta` moved from src to dest."""
    if delta <= 0.0 or src == dest:
        return a
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dest] += delta
    # numerical safety
    if a_new[src] < 0.0:
        a_new[src] = 0.0
    return a_new


# ----------------------------------------------------------------------
def _gradient_active(a: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """Gradient of max‑shift convolution w.r.t. a (sub‑gradient)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_shifts:
        i0 = max(0, k - (n - 1))
        i1 = min(k, n - 1)
        idx_i = slice(i0, i1 + 1)
        j_range = k - np.arange(i0, i1 + 1, dtype=np.int64)
        g[idx_i] += 2.0 * a[j_range]
    return g

def _best_delta_line_search(a: np.ndarray,
                            src: int,
                            dest: int,
                            max_delta: float,
                            deadline: float) -> tuple[float, float]:
    """
    Golden‑section search on the interval [0, max_delta] to find the
    delta giving the smallest objective after moving mass from src to dest.
    Returns (best_delta, best_score).  Stops early if time runs out.
    """
    # tolerance – we stop when interval is tiny
    tol = max_delta * 1e-8 + 1e-12
    lo, hi = 0.0, max_delta
    phi = (1 + np.sqrt(5.0)) * 0.5
    resphi = 2.0 - phi

    # initial points
    x1 = lo + resphi * (hi - lo)
    x2 = hi - resphi * (hi - lo)
    f1 = _objective(_transfer(a, src, dest, x1))
    f2 = _objective(_transfer(a, src, dest, x2))

    while hi - lo > tol and time.time() < deadline:
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + resphi * (hi - lo)
            f1 = _objective(_transfer(a, src, dest, x1))
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = hi - resphi * (hi - lo)
            f2 = _objective(_transfer(a, src, dest, x2))

    # pick the best among the end points and the two interior points
    candidates = [
        (lo, _objective(_transfer(a, src, dest, lo))),
        (hi, _objective(_transfer(a, src, dest, hi))),
        (x1, f1),
        (x2, f2),
    ]
    best_delta, best_score = min(candidates, key=lambda t: t[1])
    return best_delta, best_score

def _greedy_line_search_optimize(a: np.ndarray,
                                 deadline: float,
                                 rng: np.random.Generator,
                                 max_iters: int = 6000) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer where the optimal amount of mass
    (found by a cheap line search) is moved from the index with the
    largest gradient to the one with the smallest gradient.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(best_vec)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = np.where(b >= maxb - eps)[0]

        # sub‑gradient of the active set
        g = _gradient_active(a, active)

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # maximum transferable mass from src
        max_delta = a[src]
        if max_delta < 1e-14:
            break

        delta_opt, score_opt = _best_delta_line_search(
            a, src, dest, max_delta, deadline
        )

        if score_opt < best_score - 1e-14:
            a = _transfer(a, src, dest, delta_opt)
            best_vec = a.copy()
            best_score = score_opt
            # occasional progress report
            if it % 250 == 0:
                print(f"[{time.time():.1f}] greedy it={it} score={best_score:.6f}")
            continue

        # If no improvement, try a tiny random jitter every 500 iters
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = _transfer(a, i, j, eps)
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score


# ----------------------------------------------------------------------
def _smooth_step(a: np.ndarray) -> np.ndarray:
    """One low‑pass smoothing pass that preserves the total mass."""
    return (a + np.roll(a, 1) + np.roll(a, -1)) / 3.0


def _smoothing_phase(a: np.ndarray,
                     deadline: float,
                     max_passes: int = 12) -> tuple[np.ndarray, float]:
    """Apply a few smoothing passes, keeping any improvement."""
    best = a.copy()
    best_score = _objective(best)
    for p in range(max_passes):
        if time.time() >= deadline:
            break
        a_s = _smooth_step(best)
        scr = _objective(a_s)
        if scr < best_score - 1e-14:
            best = a_s
            best_score = scr
            print(f"[{time.time():.1f}] smoothing pass={p+1} improved to {best_score:.6f}")
        else:
            break
    return best, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis walk with exponential cooling – used as a final
    polishing step.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective(cur)
    best = cur.copy()
    best_score = cur_score
    T = 0.03

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        i = rng.integers(n)
        if cur[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = rng.random() * min(cur[i], 0.1 * cur[i])
        if delta <= 0.0:
            continue
        cand = _transfer(cur, i, j, delta)
        cand_score = _objective(cand)
        dE = cand_score - cur_score
        if dE < 0.0 or rng.random() < math.exp(-dE / T):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-14:
                best = cand.copy()
                best_score = cand_score
                print(f"[{time.time():.1f}] SA it={it} new best {best_score:.6f}")
        T *= 0.9997
        if T < 1e-5:
            break
    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine:

    1. builds a heterogeneous catalogue of analytic and random seeds,
       selecting the best.
    2. refines the seed with a greedy line‑search mass‑transfer optimiser,
    3. applies a few smoothing passes,
    4. finishes with a short simulated‑annealing polish,
    5. projects back onto the simplex and returns the best vector found
       within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Length choice – modestly larger than the reference implementation.
    n = 3400
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 1) generate candidates
    candidates = _initial_candidates(n, rng)

    # include previously discovered solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size == n:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) pick the best initial candidate
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx:02d} score={sc:.6f}")

    if best_vec is None:
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) aggressive greedy optimisation with optimal line‑search steps
    best_vec, best_score = _greedy_line_search_optimize(best_vec,
                                                        deadline,
                                                        rng)

    # ------------------------------------------------------------------
    # 4) smoothing passes
    best_vec, best_score = _smoothing_phase(best_vec, deadline)

    # ------------------------------------------------------------------
    # 5) short simulated‑annealing fine‑tuning (if time permits)
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)

    # ------------------------------------------------------------------
    # 6) occasional random restarts – try to beat the current best
    while time.time() < deadline - 5.0:
        # generate a fresh random Dirichlet start
        alpha = rng.uniform(0.1, 0.6)
        start = rng.gamma(alpha, 1.0, size=n)
        start = _project_onto_simplex(start)

        # a quick local optimisation (fewer iterations)
        start, scr = _greedy_line_search_optimize(start,
                                                  deadline,
                                                  rng,
                                                  max_iters=2500)
        if scr < best_score - 1e-12:
            best_vec, best_score = start, scr
            print(f"[{time.time():.1f}] restart improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity – project onto the simplex (eliminates tiny negatives)
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
