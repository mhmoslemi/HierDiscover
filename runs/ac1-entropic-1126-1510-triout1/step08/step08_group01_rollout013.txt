"""
Optimiser for the step‑function objective.

The algorithm works in three stages:

1. **Broad sampling** – generate a large collection of candidate
   probability vectors on a variety of lengths using analytic families
   (arcsine, beta, Gaussian, …) and random Dirichlet draws.  Every
   candidate is ℓ₁‑normalised and scored with the cheap surrogate
   `2·n·max(conv(a,a))` (exact for a normalised vector).

2. **Intensive refinement** – the best `k` candidates are polished with
   a long Adam mirror‑descent run (projected onto the simplex) and a
   short SGD pass.  The improvement is tracked after each iteration.

3. **Final polishing** – the globally best vector undergoes a Frank‑Wolfe
   “pairwise mass‑transfer” loop for many iterations, occasional jitter,
   and a final linear‑programming boost (Highs).  The routine respects a
   user‑supplied time budget (≈ 1000 s) and returns the best list of
   floats found.

All heavy lifting (convolutions, sub‑gradients) is performed by fast
real‑FFT calls.  The search never produces negative entries and the
output is already ℓ₁‑normalised, so `evaluate_sequence` can be called
directly on the returned list.
"""
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex.
    Returns (gradient, current_max, active_indices)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑pair counted once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

# analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑domain to avoid overflow
    with np.errstate(divide='ignore', invalid='ignore'):
        logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# optimisation primitives ------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 200_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _lp_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """One‑shot linear‑programming improvement (used only for modest n)."""
    n = a.shape[0]
    if n > 2000:                     # keep the LP cheap
        return a, _max_convolution_score(a)

    cur_max = _subgrad(a)[1]

    # Build sparse constraint matrix for (a * g) ≤ cur_max
    from scipy.sparse import csr_matrix
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, cur_max, dtype=np.float64)

    # maximise ∑ g_j  → minimise −∑ g_j
    c = -np.ones(n, dtype=np.float64)    # minimize c·g
    bounds = [(0, None)] * n

    from scipy.optimize import linprog
    remaining = max(0.0, deadline - time.time())
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'time_limit': remaining, 'presolve': True})
    if not res.success:
        return a, _max_convolution_score(a)

    g = res.x
    if g.sum() > 0.0:
        g = g / g.sum()
    else:
        g = a.copy()

    # Blend a small amount of g into a and polish with a quick SGD run
    t = 0.2
    a_new = (1 - t) * a + t * g
    a_new = _project_onto_simplex(a_new, s=1.0)
    a_ref, sc_ref = _sgd_subgrad(a_new, deadline, max_iters=60_000,
                                 step0=0.4)
    return a_ref, sc_ref


# ----------------------------------------------------------------------
def _frank_wolfe_mass_transfer(a: np.ndarray, cur_score: float,
                               deadline: float,
                               max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """Pairwise mass‑transfer loop (Frank‑Wolfe style)."""
    n = a.shape[0]
    best_a = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # attempt a full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        sc_full = _max_convolution_score(a_full)

        improved = False
        if sc_full < cur_score - 1e-12:
            a = a_full
            cur_score = sc_full
            improved = True
        else:
            # try half transfer
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            sc_half = _max_convolution_score(a_half)
            if sc_half < cur_score - 1e-12:
                a = a_half
                cur_score = sc_half
                improved = True

        if improved and cur_score < best_sc - 1e-14:
            best_sc, best_a = cur_score, a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[{time.time():.1f}] FW iter={it} new best {best_sc:.6f}")

        # occasional random jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_sc

"""
Optimiser for the step‑function objective.

Stages
-------
1. Broad sampling – generate a large pool of ℓ₁‑normalised candidates using
   analytic families (arcsine, beta, Gaussian, …) and random Dirichlet draws.
   Each candidate is scored with the cheap surrogate
   `2·n·max(conv(a,a))`, which equals `evaluate_sequence` for a normalised
   vector.

2. Intensive refinement – the best `k` seeds are polished with a long
   Adam mirror‑descent run (projected onto the simplex) and a short SGD
   pass.  Improvements are tracked after each iteration.

3. Final polishing – the current global best undergoes a Frank‑Wolfe style
   pairwise‑mass‑transfer loop, occasional jitter, a possible linear‑program
   boost (Highs), and one optional up‑sampling + short Adam refinement.

All convolutions and sub‑gradients are computed via real FFT for speed.
The routine respects a user‑supplied time budget (≈ 1000 s) and returns
the best list of floats found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑pair counted once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
# analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# optimisation primitives -----------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 200_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _lp_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Single linear‑programming step (used only for modest n)."""
    n = a.shape[0]
    if n > 2000:
        return a, _max_convolution_score(a)

    cur_max = _subgrad(a)[1]

    from scipy.sparse import csr_matrix
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, cur_max, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximise ∑g  → minimise -∑g
    bounds = [(0, None)] * n

    from scipy.optimize import linprog
    remaining = max(0.0, deadline - time.time())
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs',
                  options={'time_limit': remaining, 'presolve': True})
    if not res.success:
        return a, _max_convolution_score(a)

    g = res.x
    if g.sum() > 0.0:
        g = g / g.sum()
    else:
        g = a.copy()

    t = 0.2
    a_new = (1 - t) * a + t * g
    a_new = _project_onto_simplex(a_new, s=1.0)
    a_ref, sc_ref = _sgd_subgrad(a_new, deadline, max_iters=60_000, step0=0.4)
    return a_ref, sc_ref


def _frank_wolfe_mass_transfer(a: np.ndarray, cur_score: float,
                               deadline: float,
                               max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """Pairwise mass‑transfer loop (Frank‑Wolfe style)."""
    n = a.shape[0]
    best_a = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        sc_full = _max_convolution_score(a_full)

        improved = False
        if sc_full < cur_score - 1e-12:
            a, cur_score = a_full, sc_full
            improved = True
        else:
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            sc_half = _max_convolution_score(a_half)
            if sc_half < cur_score - 1e-12:
                a, cur_score = a_half, sc_half
                improved = True

        if improved and cur_score < best_sc - 1e-14:
            best_sc, best_a = cur_score, a.copy()
            if it % 5000 == 0:
                print(f"[{time.time():.1f}] FW iter={it} new best {best_sc:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget
    and returns the best vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1. pick a set of lengths to explore (2 k – 12 k is a sweet spot)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    lengths = set()
    for factor in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * factor)
        if 2000 <= n <= 12000:
            lengths.add(n)
    for _ in range(7):
        n = np.random.randint(2000, min(12000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # --------------------------------------------------------------
    # 2. generate a cheap pool of candidates
    candidates = []   # (vector, surrogate_score)
    beta_ps = list(np.arange(0.1, 3.01, 0.2))
    beta_params = [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    dirichlet_alphas = [0.1, 0.2, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families
        for gen, name in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exponential"),
            (lambda m: _gaussian_distribution(m, 0.15), "gaussian"),
            (lambda m: _cauchy_distribution(m, 0.15), "cauchy"),
        ):
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} {name:10s} "
                  f"score={candidates[-1][1]:.6f}")

        # symmetric beta family
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # a few random unsymmetric beta draws
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = np.random.choice(beta_params)
            beta = np.random.choice(beta_params)
            a = _beta_distribution(n, float(alpha), float(beta))
            candidates.append((a, _max_convolution_score(a)))

        # power‑transformed arcsine (helps shape the tails)
        base = _arcsine_distribution(n)
        for pwr in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            if time.time() >= deadline:
                break
            a = np.power(base, pwr)
            a /= a.sum()
            candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws
        for alpha in dirichlet_alphas:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # --------------------------------------------------------------
    # 3. inject the previous best construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, "
                  f"score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. keep the most promising seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Current best seed score = {seeds[0][1]:.6f}")

    best_seq = seeds[0][0].copy()
    best_score = seeds[0][1]

    # --------------------------------------------------------------
    # 5. short intensive refinement of each seed
    for idx, (seed_seq, _) in enumerate(seeds):
        if time.time() >= deadline:
            break
        # Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                 max_iters=300_000,
                                 step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} Adam -> {best_score:.6f}")

        # Exponentiated gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=150_000,
                                        eta0=0.8)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} ExpGrad -> {best_score:.6f}")

        # Simple SGD
        seq, sc = _sgd_subgrad(seed_seq, deadline,
                               max_iters=200_000,
                               step0=0.5)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} SGD -> {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. long‑run polishing of the current global best
    if time.time() < deadline:
        seq, sc = _adam_descent(best_seq, deadline,
                                 max_iters=800_000,
                                 step0=0.2)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] long Adam refinement -> {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Frank‑Wolfe pairwise mass‑transfer on the global best
    a = best_seq.copy()
    cur_score = best_score
    a, cur_score = _frank_wolfe_mass_transfer(a, cur_score, deadline,
                                             max_iters=150_000)
    if cur_score < best_score - 1e-14:
        best_seq, best_score = a.copy(), cur_score
        print(f"[{time.time():.1f}] FW final improvement -> {best_score:.6f}")

    # --------------------------------------------------------------
    # 8. optional linear‑programming boost
    if time.time() < deadline - 2.0:
        try:
            best_seq, best_score = _lp_refine(best_seq, deadline)
            print(f"[{time.time():.1f}] LP boost gave new best {best_score:.6f}")
        except Exception as e:
            print("LP boost failed:", e)

    # --------------------------------------------------------------
    # 9. optional up‑sampling + short Adam (if we still have time)
    if time.time() < deadline - 1.0 and len(best_seq) < 8000:
        try:
            n_old = len(best_seq)
            n_new = n_old * 2
            x_old = np.arange(n_old, dtype=np.float64)
            x_new = np.linspace(0, n_old - 1, n_new)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = a_interp / a_interp.sum()
            seq, sc = _adam_descent(a_interp, deadline,
                                    max_iters=200_000,
                                    step0=0.2)
            if sc < best_score - 1e-14:
                best_seq, best_score = seq, sc
                print(f"[{time.time():.1f}] up‑sampled Adam improvement -> {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 10. final reporting (the surrogate equals the true value for ℓ₁‑normalised vectors)
    print(f"Final surrogate best score: {best_score:.6f} (length={len(best_seq)})")
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"evaluate_sequence returns {true_val}")
    except Exception:
        pass

    return best_seq.tolist()
