def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, random, time, math

    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0  # safety

    def project(v):
        return _project_onto_simplex(v, 1.0)

    best_seq = None
    best_score = float('inf')
    best_n = None

    # Candidate lengths
    lengths = [2000, 4000, 8000, 15000, 30000]
    for n in lengths:
        if time.time() >= deadline:
            break
        # Candidate list
        cand = []
        # Uniform
        cand.append(np.full(n, 1.0 / n))
        # Linear decreasing (a_i ∝ n-i)
        lin = np.arange(n, 0, -1, dtype=np.float64)
        lin = lin / lin.sum()
        cand.append(lin)
        # Exponential left
        for lam in (0.0005, 0.001, 0.0025, 0.005, 0.01, 0.02):
            vec = np.exp(-lam * np.arange(n, dtype=np.float64))
            vec = vec / vec.sum()
            cand.append(vec)
        # Power law left
        for p in (0.5, 1.0, 1.5, 2.0, 3.0):
            idx = np.arange(1, n+1, dtype=np.float64)
            vec = 1.0 / (idx**p)
            vec = vec / vec.sum()
            cand.append(vec)
        # Random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            cand.append(vec)
        # Sorted Dirichlet to force decreasing
        for alpha in (0.2, 0.5, 1.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            vec = np.sort(vec)[::-1]  # decreasing
            vec = vec / vec.sum()
            cand.append(vec)

        # Evaluate
        for vec in cand:
            if time.time() >= deadline:
                break
            # Ensure positive epsilon
            eps = 1e-12
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] init length {n} new best {sc:.6f}")

    # Use previous global if better
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # Fallback
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, 1.0/n)
        best_n = n
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # -------------------------------------------------
    # 1) Smoothed max mirror descent (Softmax)
    a = best_seq.copy()
    n = a.shape[0]
    eps = 1e-12

    # Schedule: list of (alpha_start, alpha_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 30.0, 800, 0.1, 0.02),
        (30.0, 200.0, 1200, 0.02, 0.002),
        (200.0, 2000.0, 1200, 0.002, 0.0002),
    ]
    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            # geometric interpolation
            alpha = alpha0 * (alpha1/alpha0)**t
            lr = lr0 * (lr1/lr0)**t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # softmax weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smoothed max
            grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]

            # exponentiated gradient step
            a = a * np.exp(-lr * grad)
            a = project(a)
            a = np.maximum(a, eps)
            a = a / a.sum()

            # occasional check
            if it % 200 == 0 or it == steps - 1:
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-12:
                    best_score = cur
                    best_seq = a.copy()
                    best_n = n
                    print(f"[{time.time():.1f}] mirror it {it} new best {cur:.6f}")

    # -------------------------------------------------
    # 2) Simulated annealing hill climber
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    # temperature schedule
    T = 0.01
    decay = 0.99995
    iter_count = 0
    max_delta_abs = 0.02  # absolute bound
    while time.time() < deadline:
        iter_count += 1
        # Choose source proportional to a (larger chance for larger mass)
        # avoid zero mass
        prob = a / a.sum()
        src = np.searchsorted(np.cumsum(prob), random.random())
        # Ensure src has some mass
        if a[src] < eps:
            continue
        tgt = random.randint(0, n-1)
        if tgt == src:
            continue
        max_delta = min(a[src] * 0.5, max_delta_abs)
        if max_delta <= eps:
            continue
        delta = max_delta * random.random()
        # Apply transfer
        _apply_transfer(a, conv, src, tgt, delta)

        new_M = conv.max()
        improve = new_M < M
        if improve:
            M = new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] anneal iter {iter_count} new best {cur_score:.6f}")
        else:
            # Metropolis acceptance
            accept = random.random() < math.exp(-(new_M - M) / T)
            if not accept:
                # revert
                _apply_transfer(a, conv, tgt, src, delta)
            else:
                # accept worse
                M = new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
                    best_n = n
                    print(f"[{time.time():.1f}] anneal (worse) iter {iter_count} new best {cur_score:.6f}")

        # Annealing temperature decay
        T = max(T * decay, 1e-6)

        # Periodic full recompute to avoid drift
        if iter_count % 5000 == 0:
            conv = _fft_self_convolve(a)
            M = conv.max()
            # jitter
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied iter {iter_count}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()

"""
Propose a step‑function vector (non‑negative list of floats) with a low
self‑convolution peak.  The algorithm consists of three stages:

1.  **Broad initial scan** – generate many analytic families (uniform,
    left‑hand exponential, power‑law, linear decay) and random Dirichlet
    draws for several candidate lengths.  The best among them is kept.

2.  **Soft‑max mirror descent** – use an exponentiated‑gradient (mirror)
    update on the smooth maximum of the convolution (log‑sum‑exp
    smoothing).  The schedule gradually raises the smoothing parameter
    and shrinks the learning rate, while projecting onto the probability
    simplex after each step.

3.  **Simulated‑annealing hill‑climber** – modify the vector by moving a
    small amount of mass from one index to another.  The convolution is
    updated in O(n) with a custom incremental routine, and moves are
    accepted either if they improve the peak or with a Metropolis
    probability that slowly cools.  Periodic jitter and full FFT
    recomputation keep numerical drift under control.

The whole routine respects a 1000 s wall‑clock limit and returns the
best sequence found (as a plain Python list of floats).  It also
incorporates the previously discovered sequence `height_sequence_1`
when available.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    Returns the original values of a[i] and a[j] for possible reversion.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return a[i], a[j]
    a_i_old = a[i]
    a_j_old = a[j]

    # update convolution (each unordered pair appears twice)
    conv[i:i + n] -= 2.0 * delta * a_i_old
    conv[j:j + n] += 2.0 * delta * a_j_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta

    a[i] -= delta
    a[j] += delta
    return a_i_old, a_j_old


def _revert_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int,
                     delta: float, a_i_old: float, a_j_old: float):
    """Undo a transfer performed by `_apply_transfer`."""
    n = a.shape[0]
    # undo convolution updates
    conv[i:i + n] += 2.0 * delta * a_i_old
    conv[j:j + n] -= 2.0 * delta * a_j_old
    conv[2 * i] -= delta * delta
    conv[2 * j] -= delta * delta

    a[i] = a_i_old
    a[j] = a_j_old


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (finite vector) that minimises
    ``evaluate_sequence``.  The method mixes an analytic initial scan,
    soft‑max mirror descent, and a simulated‑annealing hill‑climber.
    It respects the 1000 s wall‑clock limit and returns the best sequence
    found.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad scan over several lengths and families (including asymmetric)
    lengths = [2000, 4000, 8000, 15000, 30000]
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # Uniform
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # Linear decreasing (a_i ∝ n‑i)
        lin = np.arange(n, 0, -1, dtype=np.float64)
        candidates.append(lin / lin.sum())

        # Left‑hand exponential families
        for lam in (5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2, 2e-2):
            v = np.exp(-lam * np.arange(n, dtype=np.float64))
            candidates.append(v / v.sum())

        # Left‑hand power‑law families
        for p in (0.5, 1.0, 1.5, 2.0, 3.0):
            idx = np.arange(1, n + 1, dtype=np.float64)
            v = 1.0 / (idx ** p)
            candidates.append(v / v.sum())

        # Random Dirichlet draws (different concentrations)
        for alpha in (0.2, 0.5, 1.0, 2.0):
            v = np.random.dirichlet(np.full(n, alpha))
            candidates.append(v)

        # Sorted Dirichlet (forced decreasing)
        for alpha in (0.2, 0.5, 1.0):
            v = np.random.dirichlet(np.full(n, alpha))
            v = np.sort(v)[::-1]                       # decreasing order
            candidates.append(v / v.sum())

        # Evaluate cheap candidates
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] init n={n} new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Compare with the previously discovered sequence (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if best_seq is None:
        n = 5000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Soft‑max mirror descent (exponentiated gradient)
    a = best_seq.copy()
    n = a.shape[0]

    # schedule: (alpha_start, alpha_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 30.0, 800, 0.10, 0.02),
        (30.0, 200.0, 1200, 0.02, 0.002),
        (200.0, 2000.0, 1200, 0.002, 0.0002),
    ]

    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # smooth‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smooth max (see original code)
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient step + projection onto simplex
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            # occasional check for improvement
            if (it % 200 == 0) or (it == steps - 1):
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-12:
                    best_score = cur
                    best_seq = a.copy()
                    best_n = n
                    print(f"[{time.time():.1f}] mirror it={it} new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing hill climber (O(n) mass transfers)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    temperature = 0.01
    decay = 0.99995
    max_delta_abs = 0.02                    # absolute bound on transferred mass
    iter_count = 0
    jitter_interval = 5000

    while time.time() < deadline:
        iter_count += 1

        # source index sampled proportionally to its mass (larger masses move more)
        prob = a / a.sum()
        src = np.searchsorted(np.cumsum(prob), random.random())
        if a[src] < eps:
            continue

        tgt = random.randint(0, n - 1)
        if tgt == src:
            continue

        max_delta = min(a[src] * 0.5, max_delta_abs)
        if max_delta <= eps:
            continue
        delta = max_delta * random.random()

        # perform the transfer and keep old values for possible revert
        old_src, old_tgt = _apply_transfer(a, conv, src, tgt, delta)

        new_M = conv.max()
        improvement = new_M < M

        if improvement:
            M = new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] anneal iter={iter_count} new best {cur_score:.6f}")
        else:
            # Metropolis acceptance
            accept = random.random() < math.exp(-(new_M - M) / temperature)
            if not accept:
                # revert the move
                _revert_transfer(a, conv, src, tgt, delta, old_src, old_tgt)
            else:
                # accept worse move
                M = new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
                    best_n = n
                    print(f"[{time.time():.1f}] anneal (worse) iter={iter_count} new best {cur_score:.6f}")

        # temperature cooling
        temperature = max(temperature * decay, 1e-6)

        # occasional jitter + full recomputation to curb drift
        if iter_count % jitter_interval == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at iter {iter_count}")

        # full recompute every few ten‑thousands of steps to avoid error buildup
        if iter_count % 20000 == 0:
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
