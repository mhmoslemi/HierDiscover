"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

* Global search – Differential Evolution (DE) on the probability simplex.
  DE explores many directions without gradients and works well for the
  piecewise‑linear “max‑convolution’’ objective.

* Local polishing – a few thousand iterations of projected sub‑gradient
  descent (exact sub‑gradient of the max‑convolution) and a short
  simulated‑annealing hill‑climber to escape shallow basins.

* Multiple lengths – the routine tries several sequence lengths around the
  length of the previously found best solution (or a sensible default) and
  keeps the overall best.

All steps respect the 1000 s wall‑clock budget.  The returned list is a
valid input for ``evaluate_sequence`` (non‑negative, any scale – the
objective is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) for a on the simplex (sum‑1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of ``max(a*a)`` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of a reversed with the active‑mask weights
    conv_rev = np.fft.irfft(
        np.fft.rfft(a[::-1], n=1 << ((a.size + w.size - 1) - 1).bit_length())
        * np.fft.rfft(w, n=1 << ((a.size + w.size - 1) - 1).bit_length()),
        n=1 << ((a.size + w.size - 1) - 1).bit_length(),
    )
    conv_rev = conv_rev[: a.size + w.size - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms where i=j (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 0.6):
    """Plain projected sub‑gradient descent (few thousand iterations)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Γ‑Dirichlet draw with α=1 (uniform on the simplex)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex vector onto a new length."""
    if seq.ndim != 1 or seq.size == 0:
        return np.full(new_n, 1.0 / new_n)
    old_n = seq.size
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq, left=0.0, right=0.0)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s <= 0.0:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq /= s
    return new_seq

    n = initial.size

    best_idx = np.argmin(scores)

    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(popsize):
            # pick three distinct donors

            idxs.remove(i)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.maximum(mutant, 0.0)                # enforce non‑negativity

def _anneal_hill_climb(a0: np.ndarray, deadline: float,
                      start_transfer: float = 0.05,
                      max_iter: int = 300_000):
    """Simulated‑annealing style random mass transfers."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # geometric decay of the allowed transfer amount
        transfer = start_transfer * (0.5 ** (it / max_iter))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

            if new_score < best_score:

            if np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
