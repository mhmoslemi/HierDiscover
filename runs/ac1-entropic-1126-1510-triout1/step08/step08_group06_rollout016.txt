def _hard_max_descent(a, deadline, best_seq, best_score, max_iters=5000):
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        # find active indices
        # tolerance relative: maybe maxb - b < 1e-12 or relative tolerance
        tol = maxb * 1e-12
        active = np.where(b >= maxb - tol)[0]
        # Build w indicator
        w = np.zeros_like(b)
        w[active] = 1.0 / len(active)  # equal weight
        # gradient of max is average gradient of active constraints
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n-1:n-1+n]

        # step size schedule: maybe η = 0.05 / sqrt(t)
        eta = 0.05 / np.sqrt(t)
        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, 1.0)
        cur_score = _objective(a_new)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time():.1f}] hard max iter={t} new best {best_score:.6f}")
        else:
            # maybe accept a_new if not worse to provide momentum? For simplicity, disregard
            a = a_new  # still update
    return best_seq, best_score

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0

    lengths = [2000, 4000, 8000, 12000, 20000]
    best_seq = None
    best_score = float('inf')
    top_candidates = [] # list of (score, array)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}")
        candidates = _initial_candidates_extended2(n, rng)
        # maybe add random mixtures: produce 5 mixture seeds per length
        for _ in range(5):
            f1 = rng.choice(candidates)
            f2 = rng.choice(candidates)
            alpha = rng.random()
            mix = alpha * f1 + (1-alpha) * f2
            mix = mix / mix.sum()
            candidates.append(mix)
        for cand in candidates:
            if time.time() >= deadline:
                break
            sc = _objective(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best length={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))
        # keep top few seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # fallback if none
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)

    # Main optimization
    # Use top candidates seeds (max 5)
    seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
    # allocate time for gradient descent
    remaining = deadline - time.time()
    if remaining > 30:
        per_seed_time = remaining / len(seeds)
        for seed_seq in seeds:
            sub_deadline = time.time() + per_seed_time
            # Adam descent
            best_seq, best_score = _smooth_adam(seed_seq.copy(),
                                                sub_deadline,
                                                best_seq=best_seq,
                                                best_score=best_score)
            # Hard max descent
            best_seq, best_score = _hard_max_descent(best_seq.copy(),
                                                     sub_deadline,
                                                     best_seq=best_seq,
                                                     best_score=best_score)

    # final greedy
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(best_seq.copy(),
                                                    deadline,
                                                    best_seq=best_seq,
                                                    best_score=best_score)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")

"""
Propose a non‑negative step function that minimizes the evaluation metric.

Algorithm Overview
------------------
1. **Rich initialisation** – For each chosen length a broad set of analytic
   families is generated: uniform, arcsine, triangular, beta (both symmetric
   and asymmetric), power‑law edges, sine, exponential, alternating,
   two‑block spikes, random‑sparse and random convex mixtures of two families.
   All candidates are normalised to ℓ₁‑norm 1 and their exact objective
   ``2·n·max(conv(a,a))`` is evaluated.  The best few are retained as seeds.

2. **Smoothed Adam optimisation** – Each seed is refined by Adam on the
   smooth‑max approximation of the max‑convolution (log‑sum‑exp).  The
   smoothing parameter α is increased through a schedule while the learning
   rate decays.  Every few hundred iterations the true (unsmoothed) objective
   is computed and the best vector seen so far is stored.

3. **Hard‑max sub‑gradient descent** – After the Adam phase we switch to the
   genuine max‑convolution.  The active set of indices where the convolution
   attains (or is within a tiny tolerance of) its maximum is identified.
   The sub‑gradient of the max is the average of the gradients of those
   active constraints (obtained by a single FFT‑based convolution).  A
   diminishing step size and Euclidean projection onto the simplex are used.

4. **Greedy pairwise swaps + tiny simulated annealing** – Finally a cheap
   Metropolis‑type walk repeatedly transfers a random amount of mass from a
   source index to a destination index.  Moves are accepted only if the true
   objective improves; a small probability of uphill moves (temperature decay)
   is added to escape shallow local minima.

All phases respect a global time budget (default 1000 s).  The best vector
found is returned as a plain Python list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric beta (Beta(p,q))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # protect against zeros
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    """Half‑sine shape, zero at the edges, positive inside."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of start vectors for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_sine(n))

    # beta – symmetric
    for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # beta – asymmetric grid
    beta_grid = (0.05, 0.2, 0.5, 1.0, 2.0, 5.0)
    for p in beta_grid:
        for q in beta_grid:
            cand.append(_distribution_beta_asym(n, p, q))

    # exponentials – wide range
    for lam in (0.001, 0.002, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # power‑law at the edges
    for gamma in (0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4):
        cand.append(_distribution_powerlaw_edges(n, gamma))

    # alternating patterns
    for stride in (2, 3, 4):
        cand.append(_distribution_alternating(n, stride))

    # two‑block spikes
    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random sparse start (~1 % non‑zeros)
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # random convex mixtures of two families (adds extra diversity)
    for _ in range(8):
        f1, f2 = rng.choice(cand, size=2, replace=False)
        alpha = rng.random()
        mix = alpha * f1 + (1 - alpha) * f2
        mix /= mix.sum()
        cand.append(mix)

    # final normalisation (just to be safe)
    cand = [c / c.sum() for c in cand]
    return cand


# ----------------------------------------------------------------------
# Smoothed Adam descent (log‑sum‑exp approximation of the max)
def _smooth_adam(a: np.ndarray, deadline: float,
                 best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the smooth max‑convolution objective."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    schedule = [
        (0.5, 2000, 0.20),
        (1.0, 2000, 0.15),
        (5.0, 2000, 0.12),
        (20.0, 2000, 0.10),
        (100.0, 1500, 0.08),
        (500.0, 1000, 0.06),
        (2000.0, 800, 0.04),
        (5000.0, 600, 0.03),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score

# Hard‑max sub‑gradient descent
def _hard_max_descent(a: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float,
                      max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent using the true max‑convolution (active‑set)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        b = _fft_self_convolve(a)
        maxb = b.max()

        # active set (indices where the max is attained up to a tiny tolerance)
        tol = maxb * 1e-12 + 1e-15
        active = np.where(b >= maxb - tol)[0]
        if active.size == 0:
            break

        # weight equally among active constraints
        w = np.zeros_like(b)
        w[active] = 1.0 / active.size

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # diminishing step size
        eta = 0.12 / np.sqrt(t)

        a_new = a - eta * grad
        a_new = _project_onto_simplex(a_new, 1.0)

        cur = _objective(a_new)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a_new.copy()
            a = a_new
            print(f"[{time.time():.1f}] hard‑max iter={t} new best {best_score:.6f}")
        else:
            a = a_new

    return best_seq, best_score

# Greedy pairwise swap / tiny simulated annealing fine‑tuning
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Random mass transfers that are only kept if the true objective improves."""
    n = a.shape[0]
    # temperature schedule for occasional uphill moves
    T0 = 1e-4
    step = 0
    while time.time() < deadline:
        step += 1
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 30 % of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
        else:
            # tiny probability to accept a worsening move (simulated annealing)
            T = T0 / (1 + step * 1e-3)
            if np.random.rand() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
    return best_seq, best_score
