def _grad_weighted_convolution(w, a):
    n = len(a)
    a_rev = a[::-1]
    size = 1 << ((len(w) + n - 1) - 1).bit_length()  # next power of two
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    # conv length = size
    # actual relevant length = len(w) + n - 1
    conv = conv[:len(w) + n - 1]
    # extract entries i+n-1 for i=0..n-1
    g = 2.0 * conv[n-1:n-1+n]
    return g
```

    eta0 = 0.2  # step size scaling

            s = a.sum()

"""
Propose a step‑function (non‑negative list of floats) that minimizes
the black‑box ``evaluate_sequence`` objective.

Algorithm
---------

1. **Broad initialisation** – generate many analytic families
   (uniform, arcsine, symmetric beta, exponential, alternating,
   sparse random).  The best of these candidates (by the exact
   objective) is kept as a start vector.

2. **Smooth mirror‑descent** – minimise the smoothed max of the
   self‑convolution using an exponentiated‑gradient (mirror descent)
   on the simplex.  At iteration *t* we compute
   ``w = exp(gamma * (conv - max(conv)))`` and the gradient
   ``g = 2 * (w * a_rev)`` (convolution).  The update
   ``a ← a * exp(-eta * g)`` is followed by renormalisation.
   ``gamma`` grows slowly (more focus on the active shifts) and
   ``eta`` decays as 1/√t.  Random jitter is injected occasionally.

3. **Greedy active‑set refinement** – the same pairwise mass‑transfer
   routine from the previous state‑of‑the‑art code is applied to the
   result of step 2.

4. **Simulated‑annealing fine‑tuning** – a cheap Metropolis walk with
   exponentially decaying temperature finishes the optimisation.

All stages respect a global deadline (default 1000 s).  The final
vector is projected onto the simplex (`sum = 1`) and returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _grad_weighted_convolution(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max:  g_i = 2 * Σ_k w_k * a_{k-i},
    computed via a single convolution (w * a_rev).
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # length of linear convolution
    out_len = len(w) + n - 1
    size = 1 << (out_len - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    conv = conv[:out_len]                     # keep only the linear part
    # g_i = 2 * conv[i + n - 1]   (i = 0 … n‑1)
    g = 2.0 * conv[n - 1 : n - 1 + n]
    return g


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective (2·n·max(conv(a,a))) – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta density with shape parameter p (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space for stability
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isnan(w)):
        # fallback to a very flat shape
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_sparse_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse random start (≈0.5 % non‑zeros)."""
    k = max(2, n // 200)
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    a[idx] = rng.random(k)
    total = a.sum()
    if total == 0:
        a[0] = 1.0
        total = 1.0
    a /= total
    return a


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a diverse list of starting vectors for length ``n``."""
    cand = []
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta – many shape parameters
    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – several decay rates
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # a few sparse random vectors
    for _ in range(3):
        cand.append(_distribution_sparse_random(n, rng))

    # noisy perturbations of a few beta shapes
    for _ in range(3):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, 1.0)
        cand.append(pert)

    return cand


# ----------------------------------------------------------------------
def _smooth_mirror_descent(a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Smoothed max minimisation by mirror descent (exponentiated gradient)
    on the probability simplex.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)

    gamma = 5.0          # initial smoothing factor
    eta0 = 0.20          # base step size
    it = 0

    while time.time() < deadline:
        it += 1

        # 1) self‑convolution and weight vector
        conv = _fft_self_convolve(a)
        M = conv.max()
        diff = conv - M                # ≤ 0
        w = np.exp(gamma * diff)       # w_i ∈ (0,1], 1 at active shifts

        # 2) gradient of the smoothed max
        g = _grad_weighted_convolution(w, a)

        # 3) mirror‑descent update (exponentiated gradient)
        eta = eta0 / math.sqrt(it)
        exp_arg = -eta * g
        # Clip to avoid overflow/underflow in the exponentials
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # 4) renormalise to the simplex
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = _distribution_uniform(n)
            s = a.sum()
        a = a / s

        # 5) track best solution
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] smooth descent it={it} new best {best_score:.6f}")

        # 6) schedule for gamma (focus more on active set)
        if it % 500 == 0:
            gamma = min(gamma * 1.10, 200.0)

        # 7) occasional jitter to escape plateaus
        if it % 2000 == 0:
            noise = rng.normal(scale=0.006, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_score

def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer based on the exact sub‑gradient.
    Very similar to the reference implementation but kept as a separate
    helper so that the smooth descent can be followed by this phase.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient: g_i = 2 * Σ_{k∈active} a_{k-i}
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # try a full‑mass move (capped to improve stability)
        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # try a smaller fraction if the full move fails
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # occasional jitter
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score

def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk with exponential cooling (as in the reference)."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03  # initial temperature (objective units)
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9997

    return best_a, best_score

            # if the length differs, resample linearly onto the current grid
            if len(prev) != n:
                x_old = np.linspace(0, 1, len(prev), endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                prev = np.interp(x_new, x_old, prev, left=0, right=0)
            if prev.sum() > 0:
                prev = prev / prev.sum()
                candidates.append(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2) pick the best initial candidate (fast exact evaluation)
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.asarray(cand, dtype=np.float64)
        if cand.sum() == 0:
            continue
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx} score={sc:.6f}")

    # fallback (should never happen)
    if best_vec is None:
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) smooth mirror‑descent optimisation
    best_vec, best_score = _smooth_mirror_descent(best_vec, deadline, rng)

    # --------------------------------------------------------------
    # 4) aggressive greedy refinement
    if time.time() < deadline:
        best_vec, best_score = _greedy_local_search(best_vec, deadline, rng)

    # --------------------------------------------------------------
    # 5) simulated‑annealing fine‑tuning (if time permits)
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)

    # final sanity – ensure we are exactly on the simplex
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
