"""
Propose a step‑function (non‑negative vector) that approximately minimises
``evaluate_sequence``.  The optimizer consists of:

1.  A broad sweep of analytic families (uniform, arcsine, beta, exponential,
    power‑law, sine‑power, cosine windows) for several lengths, plus a few
    random Dirichlet draws.  The best candidate seeds the continuous optimisation.

2.  Multi‑phase Adam descent on a smooth approximation of the max‑convolution.
    The soft‑max temperature (parameter ``alpha``) is raised gradually.
    After each Adam step the vector is projected onto the probability simplex
    and forced to be symmetric.

3.  Exact sub‑gradient descent on the true max‑convolution (still with a
    symmetric projection).  This runs until the time budget is exhausted.

4.  A short deterministic “mass‑transfer” descent: at each iteration the index
    with the largest sub‑gradient (most responsible for the peak) donates a
    small amount of mass to the index with the smallest sub‑gradient.
    Moves are accepted only if they lower the peak.

5.  A simulated‑annealing hill‑climber that works directly on the full
    self‑convolution using O(n) incremental updates.  The best solution seen
    across *all* stages is retained.

All phases respect the global time budget (default 1000 s) and the final
vector is returned as a plain list of Python floats (sum≈1, all entries ≥0).
"""

import time

import random

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_i_old = a[i]
    a_j_old = a[j]

    # Update convolution (linear terms)
    conv[i:i + n] -= 2.0 * delta * a_i_old
    conv[j:j + n] += 2.0 * delta * a_i_old

    # Quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta

    # Update the vector
    a[i] = a_i_old - delta
    a[j] = a_j_old + delta

            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it_sa} new best {new_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        T *= decay

        # occasional jitter to escape flat plateaus
        if it_sa % 5000 == 0 and it_no_improve > 2000:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at SA iter {it_sa}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
