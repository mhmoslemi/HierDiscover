"""
Propose a step‑function (probability vector) that minimises 
``evaluate_sequence``'s upper bound.

Approach
--------
1. **Initialisation** – try a wide range of lengths and a rich set of
   analytic families (uniform, arcsine, Beta(p,p), sine‑power, cosine
   window, exponential, Gaussian, triangular) together with random
   Dirichlet draws.  The best candidate (lowest 2·n·max(conv)) is kept.

2. **Adam‑based smooth‑max optimisation** – minimise the smooth
   approximation  
   `L = 2·n·logsumexp(alpha·conv)/alpha`  
   using Adam on the simplex (exponentiated gradient is replaced by a
   more robust Adam update).  ``alpha`` (smoothness) is increased
   geometrically while the learning‑rate is decayed.

3. **Symmetrisation** – after each update the vector is symmetrised
   (a ← (a+a[::-1])/2) because the optimal solution is known to be
   symmetric; this also reduces the effective dimensionality.

4. **Refined hill‑climbing** – with the current best vector we perform
   many cheap O(n) mass‑transfer moves.  A move transfers a small amount
   ``delta`` from a source index (chosen among the indices that
   contribute most to the current maximal convolution entry) to a
   random target.  The convolution array is updated incrementally,
   and the move is accepted only if the maximal convolution value
   drops.  The step size ``delta`` is slowly reduced.

5. **Tracking** – the best vector (lowest cheap score) is stored.
   Throughout the run progress is printed.  The function respects the
   1000 s wall‑clock budget.

The algorithm is deliberately deterministic given a seed, but several
random restarts are performed automatically.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Cheap objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_i = a[i]
    a_j = a[j]
    # Update the convolution
    # contribution of a_i to every offset
    conv[i:i + n] -= 2.0 * delta * a_i
    conv[j:j + n] += 2.0 * delta * a_j
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    # Update the masses
    a[i] -= delta
    a[j] += delta

        candidates.append(_distribution_exponential(n, lam=0.02))
        candidates.append(_distribution_cosine_window(n, coeff=0.5))

            vec = np.random.dirichlet(np.full(n, alpha))

                best_n = n

    # 4. Adam‑based smooth‑max optimisation (mirror descent variant).
    a = best_seq.copy()
    n = a.shape[0]
