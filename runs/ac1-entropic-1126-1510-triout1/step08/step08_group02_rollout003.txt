"""
search for a step‑function (probability vector) that minimizes
``evaluate_sequence``.  The algorithm:

1. initialise from the previous best solution (if any) or from a
   well‑behaved analytic distribution (arcsine).

2. repeatedly solve a small linear programme:
   maximise Σ b_j  subject to   a ∗ b ≤ M   and   b ≥ 0,
   where M = max(a∗a) and only the indices of the largest
   convolution values are kept as constraints.
   The optimiser b is mixed with the current a (convex combination)
   – this is the “LP‑refinement” step described in the editorial.

3. long Adam optimisation on the true objective (exact sub‑gradient of
   max(a∗a)).

4. projected sub‑gradient descent (many cheap iterations).

5. fine‑grained hill‑climbing: move a tiny amount of probability mass
   from the index with the largest positive sub‑gradient to the one
   with the most negative sub‑gradient, occasionally adding a small
   random jitter.

All steps respect the 1000 s budget and the final vector is returned
as a plain Python list (sum = 1, non‑negative).  The implementation
uses only NumPy, SciPy (linprog, sparse) and no external files.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # shift by max element for numerical stability
    v = v.astype(np.float64, copy=False)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the simplex."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best = cur, a.copy()
    return best, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimiser on the simplex using the exact sub‑gradient."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best = cur, a.copy()
    return best, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best = cur, a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, deadline: float,
               max_iters: int = 8,
               t_mix: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Perform up to ``max_iters`` linear‑programming refinement steps.
    In each step we solve:
        max Σ b_j  s.t. a * b ≤ M  ,  b ≥ 0
    over a small active set of constraints (the largest entries of a*a).
    The obtained b is mixed with the current a using factor ``t_mix``.
    Returns the best vector and its cheap score.
    """
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = a.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(best)
        M = conv.max()

        # Choose active constraints – the K largest values of conv
        K = max(20, int(0.01 * conv.size))
        # argpartition for speed (unsorted)
        act_idx = np.argpartition(conv, -K)[-K:]

        # Build sparse constraint matrix A (len(act_idx) × n)
        rows = []
        cols = []
        data = []
        for row, k in enumerate(act_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            # loop over i, coefficient a[i] for variable b_{k-i}
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(row)
                cols.append(j)
                data.append(best[i])
        A = sp.coo_matrix((data, (rows, cols)), shape=(len(act_idx), n))

        # Linear programme: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        b_ub = np.full(len(act_idx), M, dtype=np.float64)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                    bounds=bounds, method='highs',
                                    options={'presolve': True})
        if res.status != 0:
            # infeasible / unbounded – abort LP refinement
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-9:
            # no beneficial scaling found
            break

        # Normalise b to a probability vector
        b_norm = b / sum_b

        # Mix with current best (convex combination) and re‑project
        candidate = (1.0 - t_mix) * best + t_mix * b_norm
        candidate = _project_onto_simplex(candidate, s=1.0)
        cand_score = _max_convolution_score(candidate)

        if cand_score < best_score - 1e-12:
            best, best_score = candidate, cand_score
            # optionally tighten mixing factor for later iterations
            t_mix = max(0.02, t_mix * 0.9)
            print(f"[{time.time():.1f}] LP‑refine iter {it+1} → score {cand_score:.6f}")
        else:
            # if no progress, shrink mixing amount and try again
            t_mix *= 0.7
            if t_mix < 1e-4:
                break

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a probability vector (sum ≈ 1).  The routine adapts its strategy
    depending on the remaining time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 0. Initialise – try to reuse the previous best sequence.
    if 'height_sequence_1' in globals():
        try:
            a0 = np.asarray(height_sequence_1, dtype=np.float64)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                raise ValueError
            a0 = a0 / a0.sum()
            print(f"[{time.time():.1f}] loaded previous best, length={a0.size}")
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # fall back to a well‑behaved analytic shape
        n = 3400
        a0 = _arcsine_distribution(n)
        print(f"[{time.time():.1f}] no previous data – start with arcsine (n={n})")

    best_vec = a0.copy()
    best_score = _max_convolution_score(best_vec)
    print(f"[{time.time():.1f}] initial cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. LP‑based refinement – a few aggressive steps
    lp_deadline = deadline - 200.0          # reserve most of the time for later stages
    if time.time() < lp_deadline:
        best_vec, best_score = _lp_refine(best_vec, lp_deadline,
                                         max_iters=12, t_mix=0.18)
        print(f"[{time.time():.1f}] after LP‑refine cheap score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam optimisation (exact sub‑gradient) – long run
    adam_deadline = deadline - 80.0
    if time.time() < adam_deadline:
        vec, sc = _adam_descent(best_vec, adam_deadline,
                                max_iters=500_000, step0=0.28)
        if sc < best_score - 1e-12:
            best_vec, best_score = vec, sc
            print(f"[{time.time():.1f}] Adam descent improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent – very many cheap iterations
    subgrad_deadline = deadline - 30.0
    if time.time() < subgrad_deadline:
        vec, sc = _subgradient_descent(best_vec, subgrad_deadline,
                                       max_iters=2_500_000, eta0=0.7)
        if sc < best_score - 1e-12:
            best_vec, best_score = vec, sc
            print(f"[{time.time():.1f}] Sub‑gradient descent improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fine‑grained hill‑climbing (mass transfers + jitter)
    a = best_vec.copy()
    cur_score = best_score
    n = a.size
    it = 0
    max_transfer = 0.05           # up to 5 % of total mass initially

    while time.time() < deadline:
        it += 1
        # adapt transfer magnitude slowly
        transfer = max_transfer * (0.98 ** (it // 5000))

        # choose source/target indices
        if it % 7 == 0:
            grad, _, _ = _subgrad(a)
            i = int(np.argmax(grad))
            j = int(np.argmin(grad))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_vec = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter {it} new best {best_score:.6f}")

        # occasional random jitter to escape plateaus
        if it % 8_000 == 0 and np.random.rand() < 0.13:
            jitter = np.random.randn(n) * 5e-5
            a = best_vec + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_vec)})")
    # ``evaluate_sequence`` expects a plain list
    return best_vec.tolist()
