        # compute contributions weight for each i

        high = min(n-1, idx_max)
        # Use slice:

            source = np.random.choice(n, p=a)

"""
Propose a candidate step function that minimizes evaluate_sequence.
The algorithm combines:
  1. A fast sweep over many analytic families (including asymmetric Beta) to obtain a good seed.
  2. Adam optimisation on a soft‑max approximation of the max‑convolution with a gradual increase of the sharpness.
  3. Exact sub‑gradient descent for a few thousand steps.
  4. Optional diffusion smoothing.
  5. A targeted simulated‑annealing mass‑transfer hill‑climber that preferentially attacks the indices that
     currently realise the maximal convolution value.
All phases respect a global time budget (default 1000 s).  The best vector found is returned.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (return vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta(n, alpha, beta):
    """Asymmetric Beta distribution on positions i=0..n‑1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)
    w = np.exp(logw)
    if np.isnan(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals

def _transfer_mass(a, conv, i, j, delta):
    """
    Transfer a positive mass `delta` from index i to index j.
    Updates both the vector `a` and its full self‑convolution `conv`
    in O(n) time without allocating temporary copies.
    """
    if delta <= 0 or i == j:
        return
    n = a.shape[0]
    # The array `a` still contains the *old* values at i and j.
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

    random.seed(seed)

    best_score = float('inf')
    best_n = None

    lengths = [2000, 3000, 4000, 5000, 8000]

    for n in lengths:

            _distribution_arcsine(n),
            _distribution_triangular(n),

            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.shape[0]
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # Fallback – should never happen.
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam descent on a soft‑max surrogate.
    a = best_seq.copy()
    n = a.shape[0]

    # schedule: (softmax sharpness, number of steps)
    schedule = [
        (1.0,   1500),
        (5.0,   1500),
        (20.0,  3000),
        (100.0, 3000),
        (500.0, 3000),

    lr = 0.12

    v = np.zeros_like(a)
    t = 0

            t += 1

"""
Propose a candidate step function that (approximately) minimises the black‑box
``evaluate_sequence``.  The algorithm proceeds in several stages, each of
which respects a global time budget (default 1000 s).

1. **Analytic sweep** – a wide collection of families (uniform, arcsine,
   triangular, exponential, asymmetric Beta, sine‑power, cosine window,
   power‑law, sparse random seeds and a few Dirichlet draws) is evaluated for
   several lengths.  The best vector (normalised to sum = 1) seeds the later
   optimisation.

2. **Adam optimisation** – a smooth soft‑max surrogate of the max‑convolution
   ``max(b)`` is minimised.  The sharpness parameter ``alpha`` is raised
   gradually, while Adam’s adaptive steps keep the iterates inside the
   probability simplex.

3. **Exact sub‑gradient descent** – with the true max‑operator (active set of
   binding constraints) a few thousand iterations further decrease the
   objective.

4. **Diffusion smoothing** – a short 8‑pass blur (kernel [0.25,0.5,0.25])
   often reduces the peak of the self‑convolution without destroying structure.

5. **Targeted simulated‑annealing hill‑climber** – mass is repeatedly moved
   from indices that contribute most to the current maximal convolution value
   to relatively empty positions.  An O(n) incremental update of the full
   self‑convolution is used, and Metropolis acceptance with exponential cooling
   allows occasional uphill moves.

The best vector found is returned as a plain list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) assuming ``a`` is already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta(n, alpha, beta):
    """Asymmetric Beta distribution on positions i = 0 … n‑1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x)
    w = np.exp(logw)
    if np.isnan(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _transfer_mass(a, conv, i, j, delta):
    """
    Transfer a positive mass ``delta`` from index ``i`` to index ``j``.
    Updates both the vector ``a`` and its full self‑convolution ``conv`` in
    O(n) time without allocating temporary copies.
    """
    if delta <= 0 or i == j:
        return
    n = a.shape[0]
    # ``a`` still holds the *old* values at i and j.
    conv[i:i + n] -= 2.0 * delta * a
    conv[j:j + n] += 2.0 * delta * a
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta
