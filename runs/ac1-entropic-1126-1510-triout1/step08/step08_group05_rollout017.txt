        constraints = [a >= 0, cp.sum(a) == 1]

        # Use cp.sum(cp.multiply(coeff_vec, a)).

            coeff = np.zeros(n)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)

            # For each i, a0[i] * a[k-i]

            # Not directly align. We'll need expression sum_{i} a0[i]*a[k-i].
            # This is sum_{j} a0[k-j] * a[j] where j = k-i.
            # So coefficient for a[j] is a0[k-j] if 0 <= k-j < n.
            # So for each j in range(max(0, k - (n-1)) to min(n-1, k)):
            # coeff[j] = a0[k-j].
            coeff = np.zeros(n)
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end+1)
            coeff[idx] = a0[k - idx]
            constraints.append(2 * cp.sum(cp.multiply(coeff, a)) <= M + b0[k])

        obj = cp.Minimize(M)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, warm_start=True, abstol=1e-9, reltol=1e-9, max_iters=5000)
        if prob.status not in ["optimal", "optimal_inaccurate"]:
            break
        new_a = a.value
        # Clip small negatives to zero
        new_a = np.maximum(new_a, 0)
        new_a = new_a / new_a.sum()
        # compute new objective
        new_score = 2 * n * _fft_self_convolve(new_a).max()
        # if not improve break
        if new_score < best_score - 1e-12:
            best_a = new_a
            best_score = new_score
            a0 = new_a
        else:

"""
Propose a step‑function (non‑negative vector) that approximately minimises
``evaluate_sequence``.  The routine consists of:

1.  A broad search over several analytic families (uniform, beta, sine‑power,
    cosine window, exponential, power‑law) together with a few random Dirichlet
    draws and sparse vectors for several lengths.  The best of these cheap
    candidates is kept as a seed.

2.  A multi‑phase Adam optimiser on a smooth log‑sum‑exp approximation of the
    ``max`` convolution.  The softness parameter ``alpha`` is increased
    gradually while a Euclidean projection onto the probability simplex keeps
    the iterate feasible.

3.  Exact max‑sub‑gradient descent (the same update used in the reference
    implementation) to push the solution further.

4.  A short sequence of linear‑programming refinements.  At the current point
    the active (near‑maximal) convolution indices are identified; a linear
   isation of the quadratic convolution yields a set of linear constraints.
    A small LP (variables = the vector + a scalar ``M``) is solved with CVXPY
    to strictly decrease the bound on the maximum convolution value.

5.  Optional diffusion smoothing (a few passes with a 3‑tap kernel) and a
    final short simulated‑annealing pass.

All phases respect the global time budget (default 1000 s) and the best
sequence found so far is returned.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _random_dirichlet(n, concentration):
    """Draw a Dirichlet vector with the given concentration (scalar)."""
    x = np.random.gamma(concentration, 1.0, size=n)
    x /= x.sum()
    return x


def _sparse_random(n, k):
    """k‑sparse random vector (non‑negative, normalised)."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

            maxb0 = b0.max()
            # active set – indices whose convolution is within a tiny slack
            slack = maxb0 * 1e-8 + 1e-12
            active = np.where(b0 >= maxb0 - slack)[0]

            # Build LP
            a_var = cp.Variable(n)
            M_var = cp.Variable()
            constraints = [a_var >= 0,
                           cp.sum(a_var) == 1]

            # Linearised constraints: 2 * (a0 * a)[k] <= M + b0[k] for k in active
            for k in active:
                # coeff for a_i is a0[k-i] when the index is in range
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1, dtype=int)
                coeff = a0[k - idx]  # shape matches idx
                constraints.append(
                    2 * cp.sum(cp.multiply(coeff, a_var[idx])) <= M_var + b0[k]
                )

            obj = cp.Minimize(M_var)
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, warm_start=True,
                       abstol=1e-9, reltol=1e-9, max_iters=5000)

            if prob.status not in ["optimal", "optimal_inaccurate"]:
                # LP failed – break out
                break

            a_new = a_var.value
            if a_new is None:
                break
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            new_score = _max_convolution_score(a_new)

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[{time.time():.1f}] LP iter={lp_iters} new best {new_score:.6f}")
            else:
                # no improvement – stop LP refinement
                break

    # ------------------------------------------------------------------
    # 6. Optional diffusion smoothing (a few short passes).
    if time.time() < deadline:
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        a = best_seq.copy()
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, eps)
            a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Diffusion smoothing new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 7. Very short simulated‑annealing hill‑climber (O(n) mass transfers)
    # to mop up any remaining local irregularities.
    if time.time() < deadline:
        a = best_seq.copy()
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        best_score = cur_score
        best_seq = a.copy()
        T = cur_score * 0.05
        decay = 0.9997
        max_delta_factor = 0.015
        it = 0
        it_no_improve = 0
        while time.time() < deadline:
            it += 1
            # source weighted by current mass
            source = np.random.choice(n, p=a)
            target = np.random.randint(n)
            if target == source:
                continue
            max_delta = min(a[source] * max_delta_factor, 0.04)
            if max_delta <= eps:
                continue
            delta = max_delta * np.random.rand()
            if delta <= eps:
                continue

            # perform O(n) incremental update of convolution
            a_snap = a.copy()
            conv_snap = conv.copy()
            # transfer delta from source to target
            a[source] -= delta
            a[target] += delta
            # update convolution (the same formula as in the reference)
            conv[source:source + n] -= 2.0 * delta * a_snap
            conv[target:target + n] += 2.0 * delta * a_snap
            conv[2 * source] += delta * delta
            conv[2 * target] += delta * delta

            new_score = 2.0 * n * conv.max()
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
                if np.random.rand() < prob:
                    accept = True

            if accept:
                cur_score = new_score
                it_no_improve = 0
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
            else:
                a = a_snap
                conv = conv_snap
                it_no_improve += 1

            T *= decay
            if it % 5000 == 0 and it_no_improve > 3000:
                # jitter to escape flat regions
                jitter = np.random.randn(n) * 1e-5
                a += jitter
                a = np.maximum(a, eps)
                a = a / a.sum()
                conv = _fft_self_convolve(a)
                cur_score = 2.0 * n * conv.max()
                it_no_improve = 0
                print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
