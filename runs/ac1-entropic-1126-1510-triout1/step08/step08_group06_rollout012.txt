def _simulated_annealing(a, deadline, best_seq, best_score):
    n = len(a)
    temperature = 0.01  # base ratio
    # maybe schedule with exponential cooling
    while time.time() < deadline:
        # random move similar to earlier but with variable delta up to 0.5 of source
        src = np.random.choice(np.where(a > 1e-12)[0])
        dst = np.random.randint(n)
        if dst == src: continue
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-12: continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        cur = _objective(new_a)
        diff = cur - best_score
        if diff < -1e-13:
            a = new_a
            best_score = cur
            best_seq = new_a.copy()
            print(f"anneal improvement {best_score:.6f}")
        else:
            # accept with probability exp(-diff/temperature) (diff positive)
            if np.random.rand() < math.exp(-diff/temperature):
                a = new_a
        temperature *= 0.9999995  # gradually cool
    return best_seq, best_score

"""
Propose a low‑value step‑function for ``evaluate_sequence``.

Algorithm
---------
1. **Initial portfolio** – many analytic families (uniform, arcsine,
   beta, exponential, power‑law, sine, alternating, two‑block spikes,
   sparse random) are evaluated for a moderate length (2000).  The
   few best seeds are kept.

2. **Smoothed Adam** – a continuation on the log‑sum‑exp (smooth‑max)
   of the self‑convolution.  The smoothing parameter α is raised from
   0.5 up to 20 000 while Adam’s learning‑rate decays.  Exact
   objective values are recorded frequently.

3. **Linear‑programming refinement** – for a current vector *a* we
   solve the LP

        minimise   M
        subject to A·g ≤ M,
                   Σ g = 1,
                   g ≥ 0

   where *A* is the convolution matrix built from *a*
   ( (A·g)[k] = Σ_j a_j g_{k‑j} ).
   The optimum *g* is combined with *a* by a convex mixture
   (several mixing coefficients are tried) and the best improvement
   is kept.  This step is repeated a few times.

4. **Sub‑gradient polishing** – after the LP phase the smooth‑Adam
   optimiser is called once more with a very large α (≈2·10⁴) to
   converge to the true max‑convolution.

5. **Simulated‑annealing swaps** – a cheap Metropolis walk that moves
   a random amount of mass between two indices and accepts
   uphill moves with a slowly decreasing temperature.

All stages respect a global deadline (default 1000 s).  The best
vector found is returned as a plain ``list[float]``.  The code uses
only NumPy and SciPy (``scipy.optimize.linprog``) and stays well
within the 2‑CPU budget.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

# Smoothed Adam (log‑sum‑exp smooth max) -------------------------------
def _smooth_adam(a: np.ndarray, deadline: float,
                 best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max of the self‑convolution."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule – more steps and a larger final α than the reference code
    schedule = [
        (0.5, 3000, 0.20),
        (1.0, 3000, 0.15),
        (5.0, 3000, 0.12),
        (20.0, 3000, 0.10),
        (100.0, 2000, 0.08),
        (500.0, 1500, 0.06),
        (2000.0, 1200, 0.04),
        (5000.0, 800, 0.03),
        (20000.0, 500, 0.02),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 600 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score

# Linear‑programming refinement ----------------------------------------
def _lp_refine_step(a: np.ndarray, t_vals, linprog_func,
                    deadline: float, best_score: float):
    """
    One LP‑based refinement step.
    Returns (new_a, new_score) where new_a is the best convex
    combination of a and the LP solution g among the supplied mixing
    coefficients t_vals.  If the time budget is exhausted or the LP
    fails, the original a is returned unchanged.
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # build convolution matrix A where (A·g)[k] = Σ a_j g_{k‑j}
    A = np.empty((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a

    # LP: minimise M  subject to A·g ≤ M, Σg = 1, g ≥ 0
    c = np.concatenate([np.zeros(n), [1.0]])                     # n+1 variables
    A_ub = np.hstack([A, -np.ones((rows, 1))])                  # A·g - M ≤ 0
    b_ub = np.zeros(rows)
    A_eq = np.concatenate([np.ones(n), [0.0]])[np.newaxis, :]    # Σg = 1
    bounds = [(0, None)] * n + [(0, None)]

    # solve
    res = linprog_func(c, A_ub=A_ub, b_ub=b_ub,
                       A_eq=A_eq, b_eq=np.ones(1),
                       bounds=bounds, method='highs')
    if not res.success:
        return a, best_score

    g = res.x[:n]                     # already satisfies Σg = 1
    best_local = a.copy()
    best_local_score = best_score

    for t in t_vals:
        cand = (1.0 - t) * a + t * g
        cand = _project_onto_simplex(cand, 1.0)
        sc = _objective(cand)
        if sc < best_local_score - 1e-13:
            best_local_score = sc
            best_local = cand.copy()
            print(f"[{time.time():.1f}] LP mix t={t:.2f} → score {sc:.6f}")

    return best_local, best_local_score

# Greedy pairwise swaps (simulated‑annealing flavour) -----------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Random mass transfers with a slowly cooling temperature."""
    n = a.shape[0]
    temperature = 0.02
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.5          # up to 50 % of source mass
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        sc = _objective(new_a)
        if sc < best_score - 1e-13:
            a = new_a
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
            temperature = 0.02                     # reset temperature after improvement
        else:
            # Metropolis acceptance
            prob = math.exp(-(sc - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                a = new_a
        temperature *= 0.9999995                      # exponential cooling
    return best_seq, best_score

            best_score=overall_best_score)

    # ------------------------------------------------------------------
    # 5.  Simulated‑annealing pairwise swaps ---------------------------
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score)

    # ------------------------------------------------------------------
    # 6.  Optional up‑sampling to a larger length (if time permits)
    if time.time() < deadline and len(overall_best_seq) < 4000:
        try:
            target_n = 4000
            x_old = np.linspace(0.0, 1.0, len(overall_best_seq))
            x_new = np.linspace(0.0, 1.0, target_n)
            up = np.interp(x_new, x_old, overall_best_seq)
            up = up / up.sum()
            # a short polishing on the larger grid
            up, up_score = _smooth_adam(up, deadline,
                                         best_seq=overall_best_seq,
                                         best_score=overall_best_score)
            if up_score < overall_best_score - 1e-13:
                overall_best_seq = up
                overall_best_score = up_score
                print(f"[{time.time():.1f}] up‑sampled improvement {overall_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
