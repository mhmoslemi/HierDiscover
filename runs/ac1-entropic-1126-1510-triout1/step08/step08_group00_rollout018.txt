def hill_climb(a0, time_end):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    while time.time() < time_end:
        # randomness
        n = a.size
        i = np.random.randint(n)
        # choose j != i
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # transfer amount
        max_transfer = min(a[i], 1 - a[j])
        if max_transfer <= 0:
            continue
        # temperature schedule: compute proportion left
        t = (time_end - time.time()) / (time_end - start_time)   # fraction remaining
        # step scale
        delta = max_transfer * np.random.rand() * (0.5 + 0.5*t)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # project onto simplex not needed if we keep within bound 0, sum=1 preserved.
        # compute score
        new_score = _max_convolution_score(a_new)
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            # accept with some probability
            prob = np.exp(-(new_score - best_score)/ (1e-3 + 0.01*(1-t)))   # adjustable
            if np.random.rand() < prob:
                a = a_new
    return best_a, best_score

def hill_climb(a0, time_limit):
    start = time.time()
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    while time.time() < time_limit:
        # fraction remaining
        frac = (time_limit - time.time()) / (time_limit - start)   # 0 to 1
        # draw random pair
        i = np.random.randint(n)
        j = np.random.randint(n-1)

assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (step‑function heights) that minimises  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `*` is the ordinary (linear) convolution.  
Since `E` is homogeneous, we can restrict ourselves to vectors with  
\(\sum a = 1\); the denominator then disappears and the objective is simply  

\[
S(a)=2\,n\;\max (a*a).
\]

`max(a*a)` is the largest entry of the self‑convolution.  
A naïve uniform vector gives \(S=2\).  The best known construction reaches
\(S\approx1.51\) by breaking the symmetry of the uniform distribution and
spreading the mass so that no single convolution index receives many large
products.

The new search proceeds in three stages

1. **Broad random sampling** – many lengths (≈1 k–12 k) and many families
   (power‑law, Beta, exponential, Dirichlet, sparse random, etc.) are
   generated.  Every candidate is projected onto the simplex (sum = 1) and
   its cheap surrogate `S(a)=2·n·max(conv(a,a))` is evaluated with an FFT
   routine.

2. **Intensive local optimisation** – the best `k` seeds are refined by
   three complementary methods  

   * **Adam‑style adaptive gradient descent** on the simplex,  
   * **Exponentiated‑gradient (mirror) descent**, and  
   * **Plain hill‑climbing / simulated‑annealing** that repeatedly moves a
     small amount of mass from a high‑gradient coordinate to a low‑gradient
     one.

   The sub‑gradient of `max(conv)` is computed analytically (see `_subgrad`);
   all updates keep the vector on the simplex.

3. **Polishing** – a Frank‑Wolfe “mass transfer’’ step (using the sub‑gradient)
   and a final short simulated‑annealing phase are applied to the current
   best point.

All three phases respect the global 1000 s budget and constantly keep the
best vector seen.  The code is completely self‑contained, uses only `numpy`
(and optionally `scipy` for the `logsumexp` used in the simulated‑annealing
acceptance probability) and can be called as

```python
best = propose_candidate(seed=123, budget_s=1000)
```

which returns a plain Python list ready for `evaluate_sequence`.

"""
search for a non‑negative step function minimising
        2·n·max(conv(a,a))   (with Σa = 1)

Three phases:
  1️⃣ Broad random sampling of many analytic families & Dirichlet draws.
  2️⃣ Local refinement of the best seeds:
        – Adam‑style adaptive gradient descent on the simplex.
        – Exponentiated‑gradient (KL‑mirror) descent.
        – Simple hill‑climbing / simulated‑annealing moves.
  3️⃣ Frank‑Wolfe mass‑transfer polishing and a final annealing sweep.

All operations are performed with fast FFT‑based convolutions.
The routine respects the 1000 s budget and returns the best list found.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Surrogate score S(a)=2·n·max(conv(a,a)) for Σa = 1.
    Faster than evaluate_sequence().
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (Σa = 1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 500_000):
    """Adam‑style adaptive moment descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 400_000):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration alpha."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families (all already ℓ₁‑normalised)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (monotonically decreasing)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, end_time: float) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing / hill‑climbing on the simplex.
    Moves a random amount of mass from index i to j, preserving Σa = 1.
    """
    start = time.time()
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]

    while time.time() < end_time:
        # remaining fraction – used as a temperature proxy
        frac = (end_time - time.time()) / (end_time - start + 1e-12)

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        max_transfer = min(a[i], 1.0 - a[j])
        if max_transfer <= 0:
            continue

        # step size shrinks with time
        delta = max_transfer * np.random.rand() * (0.5 + 0.5 * frac)

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)

        if new_score < best_score - 1e-15:
            best_score = new_score
            best = a_new.copy()
            a = a_new
            continue

        # occasional Metropolis acceptance to escape local plateaus
        temp = 0.01 + 0.09 * frac
        if np.random.rand() < np.exp(-(new_score - best_score) / (temp + 1e-12)):
            a = a_new

    return best, best_score
