"""
Propose a step‑function (finite non‑negative vector) that minimises
``evaluate_sequence``.  The strategy improves on the reference
implementation in several ways:

1.  A broader set of analytic families (uniform, triangular,
    exponential, power‑law, symmetric Beta, sine‑power, cosine windows)
    is scanned for a variety of lengths (up to 20 000).  Random Dirichlet
    draws with many concentration parameters and sparse random seeds are
    added to the pool.

2.  The best 10 candidates are refined by a multi‑phase Adam optimiser
    that uses a soft‑max approximation of the ``max`` operator.
    The schedule is longer and the learning‑rate is larger than in the
    reference code.

3.  Exact max‑sub‑gradient descent with momentum follows the Adam phase.

4.  A *targeted* simulated‑annealing hill‑climber is applied.  Instead
    of moving mass uniformly at random, the move source is chosen
    proportionally to its contribution to the active (maximal) convolution
    entries – this directly attacks the peak.  The temperature schedule
    is slower and occasional jitter is injected when the walk stalls.

All steps respect a hard time budget (default 1000 s).  The routine
returns the best sequence (as a Python list of floats) found before the
deadline.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    (This coincides with the value returned by ``evaluate_sequence``.)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) density on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1.  Wide scan of analytic families + random draws for several lengths.
    lengths = [1024, 2048, 4096, 5000, 8000, 12000, 16000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- analytic families -------------------------------------------------
        family_fns = []

        family_fns.append(_distribution_uniform)

        family_fns.append(_distribution_triangular)

        for p in (0.5, 1.0, 1.5, 2.0, 3.0):
            family_fns.append(lambda n, p=p: _distribution_power_law(n, p))

        for lam in (0.001, 0.003, 0.008, 0.02, 0.05):
            family_fns.append(lambda n, lam=lam: _distribution_exponential(n, lam))

        for p in (0.2, 0.4, 0.6, 1.0, 2.0, 3.0):
            family_fns.append(lambda n, p=p: _distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            family_fns.append(lambda n, pw=pw: _distribution_sine_power(n, pw))

        for coeff in (0.5, 0.45, 0.42, 0.38):
            family_fns.append(lambda n, coeff=coeff: _distribution_cosine_window(n, coeff))

        # ---- generate candidates ------------------------------------------------
        for fn in family_fns:
            if time.time() >= deadline:
                break
            vec = fn(n)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} new init best {sc:.6f}")

        # ---- random Dirichlet draws (different concentrations) -----------------
        for conc in (0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 5.0):
            if time.time() >= deadline:
                break
            x = np.random.gamma(conc, 1.0, size=n)
            x = np.maximum(x, eps)
            x /= x.sum()
            sc = _max_convolution_score(x)
            if sc < best_score:
                best_score = sc
                best_seq = x.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} Dirichlet conc={conc} best {sc:.6f}")

        # ---- sparse random seed -------------------------------------------------
        if time.time() < deadline:
            k = max(2, n // 200)
            idx = np.random.choice(n, k, replace=False)
            sparse = np.zeros(n, dtype=np.float64)
            sparse[idx] = np.random.rand(k) + eps
            sparse /= sparse.sum()
            sc = _max_convolution_score(sparse)
            if sc < best_score:
                best_score = sc
                best_seq = sparse.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} sparse best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Keep a short list of the best candidates for deep optimisation.
    # (We already have the best overall – gather a few more for diversity.)
    top_candidates = [best_seq.copy()] if best_seq is not None else []

    # try to add a few random Dirichlet vectors of the best length found
    if best_n is not None:
        for _ in range(4):
            conc = np.random.choice([0.05, 0.2, 0.7, 1.5, 3.0])
            x = np.random.gamma(conc, 1.0, size=best_n)
            x = np.maximum(x, eps)
            x /= x.sum()
            top_candidates.append(x.copy())

    # incorporate the previous global solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            top_candidates.append(prev.copy())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Refine each top candidate with Adam (soft‑max) descent.
    for cand_id, a in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = a.shape[0]

        # schedule (more aggressive than the reference version)
        schedule = [
            (1.0,   1500),
            (5.0,   2500),
            (20.0,  4000),
            (100.0, 4000),
            (500.0, 3000),
            (2000.0,2500),
            (5000.0,2000)
        ]

        beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
        lr = 0.25
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0
        best_local = a.copy()
        best_local_score = _max_convolution_score(a)

        for alpha, steps in schedule:
            for _ in range(steps):
                if time.time() >= deadline:
                    break
                t += 1

                # smooth approximation of the max via soft‑max
                b = _fft_self_convolve(a)
                maxb = b.max()
                w = np.exp(alpha * (b - maxb))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(b, 1.0 / b.size)
                else:
                    w /= w_sum

                # gradient of the soft‑max objective
                grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

                # Adam update
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad * grad)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

                # projection onto the simplex and tiny regularisation
                a = _project_onto_simplex(a, 1.0)
                a = np.maximum(a, eps)
                a = a / a.sum()

                # occasional bookkeeping
                if t % 500 == 0:
                    cur = _max_convolution_score(a)
                    if cur < best_local_score - 1e-12:
                        best_local_score = cur
                        best_local = a.copy()
                        print(f"[{time.time():.1f}] cand{cand_id} Adam t={t} new best {cur:.6f}")

        # keep the best obtained from Adam
        if best_local_score < best_score - 1e-12:
            best_score = best_local_score
            best_seq = best_local.copy()
            best_n = n
            print(f"[{time.time():.1f}] cand{cand_id} Adam overall best {best_local_score:.6f}")

        # ------------------------------------------------------------------
        # 4. Exact max‑sub‑gradient descent with momentum (targeted)
        a = best_local.copy()
        vel = np.zeros_like(a)
        beta = 0.9
        step0 = 0.3
        it = 0
        while time.time() < deadline:
            it += 1
            b = _fft_self_convolve(a)
            cur_max = b.max()
            eps_rel = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(b >= cur_max - eps_rel)

            w = np.zeros_like(b)
            w[active] = 1.0
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

            vel = beta * vel + (1 - beta) * grad
            eta = step0 / np.sqrt(it)
            a = a - eta * vel
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            cur = _max_convolution_score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] cand{cand_id} subgrad it={it} new best {cur:.6f}")

            if time.time() + 0.5 > deadline:
                break

    # ------------------------------------------------------------------
    # 5. Targeted simulated‑annealing hill‑climber (mass‑transfer moves)
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_seq = a.copy()
    it = 0
    it_no_improve = 0
    T = cur_score * 0.05                # initial temperature
    decay = 0.9997                       # slower cooling
    max_delta_factor = 0.03              # up to 3 % of source mass per move

    while time.time() < deadline:
        it += 1

        # active set (indices attaining the current max, up to tolerance)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)

        # contribution of each position to the active peak (same as sub‑gradient)
        w = np.zeros_like(conv)
        w[active] = 1.0
        contrib = _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        contrib = np.maximum(contrib, 0.0)

        # choose source proportional to its contribution * current mass
        prob_src = contrib * a
        if prob_src.sum() > 0:
            prob_src /= prob_src.sum()
            source = np.random.choice(n, p=prob_src)
        else:
            source = np.random.choice(n, p=a)   # fallback

        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * max_delta_factor, 0.05)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        new_score = 2.0 * n * new_M

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Targeted SA iter={it} new best {new_score:.6f}")
        else:
            # reject – roll back
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        T *= decay

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and it_no_improve > 2000:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter injected at SA iter={it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
