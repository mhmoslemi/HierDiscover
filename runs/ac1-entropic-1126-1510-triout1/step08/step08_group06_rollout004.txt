"""
Propose a low‑objective step‑function.

The algorithm follows three stages:

1️⃣  **Broad initialization** – for several lengths we generate many analytic
    families (uniform, arcsine, beta(p,p), power‑law at the edges,
    sine, quadratic (edge‑heavy), random‑sparse).  The exact objective
    `2·n·max(conv(a,a))` is evaluated and the best few candidates are kept.

2️⃣  **Targeted sub‑gradient descent** – for each promising seed we minimise
    the true max‑convolution (not a smooth surrogate) by sub‑gradient steps.
    At every iteration we:
        • compute the self‑convolution,
        • locate the index where it attains its maximum,
        • build the sub‑gradient of that max entry,
        • take a step opposite to the gradient and project back onto the simplex.
    The learning‑rate decays slowly; exact objective values are logged
    whenever an improvement is found.

3️⃣  **Fine‑grained random mass transfers** – a cheap Metropolis walk that
    repeatedly moves a random amount of mass from a source index to a destination
    index.  A move is kept only if the true objective improves, which usually
    eliminates the last few tenths of a percent.

The whole search respects the global time budget (≈ 1000 s).  The best vector
found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_quadratic(n: int, p: float) -> np.ndarray:
    """Edge‑heavy quadratic shape: (i·(n‑i))^{p}  with p < 0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i)) ** p
    vals = np.where(np.isfinite(vals), vals, 0.0)
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                                density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _generate_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a fairly wide list of start vectors for length n."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))

    # beta family – sweep a wide range of p
    for p in np.linspace(0.05, 3.0, 15):
        cand.append(_distribution_beta_sym(n, float(p)))

    # power‑law at the edges – sweep γ up to almost 0.5
    for gamma in np.linspace(0.05, 0.49, 15):
        cand.append(_distribution_powerlaw_edges(n, float(gamma)))

    # quadratic edge‑heavy (negative exponent)
    for p in np.linspace(-0.45, -0.05, 10):
        cand.append(_distribution_quadratic(n, float(p)))

    # a few sparse random starts
    for _ in range(3):
        cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution objective."""
    n = a.shape[0]
    lr = 0.5                     # initial step size
    t = 0
    while time.time() < deadline:
        # compute full self‑convolution
        conv = _fft_self_convolve(a)
        maxb = conv.max()
        k = int(conv.argmax())   # index where the max is attained

        # build sub‑gradient of the max entry
        grad = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # gradient step + simplex projection
        a = a - lr * grad
        a = _project_onto_simplex(a, 1.0)

        # exact objective (fast because we already have maxb)
        cur = 2.0 * n * maxb
        if cur < best_score - 1e-15:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad iter {t} new best {best_score:.6f}")

        # simple learning‑rate schedule
        t += 1
        if t % 2000 == 0:
            lr = max(lr * 0.9, 1e-4)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Random mass transfers that are kept only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        # choose a source that actually carries mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = int(np.random.choice(src))
        # choose a distinct destination
        j = int(np.random.randint(n - 1))
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-15:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths – we keep them moderate so that many
    #    iterations are affordable.
    lengths = [3000, 5000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    # store a handful of promising seeds (objective, vector)
    top_candidates = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        cand_list = _generate_candidates(n, rng)

        # include the previous best if it exists and has the right length
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1:
                # either truncate or pad to current length
                if prev.size > n:
                    prev = prev[:n]
                elif prev.size < n:
                    prev = np.pad(prev, (0, n - prev.size), constant_values=0.0)
                prev = np.maximum(prev, 0.0)
                if prev.sum() > 0:
                    prev = prev / prev.sum()
                    cand_list.append(prev)
        except Exception:
            pass

        # evaluate candidates
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()            # safety normalisation
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (len={n}) score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the best few seeds globally (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 3. Ensure we have at least one seed.
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation: sub‑gradient descent + local swaps.
    #    We distribute the remaining time roughly equally over the top seeds.
    remaining = deadline - time.time()
    seeds_to_use = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
    # make sure the current best is also processed
    seeds_to_use.insert(0, overall_best_seq.copy())

    for seed_seq in seeds_to_use:
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget for this seed
        now = time.time()
        time_left = deadline - now
        # give each seed at most 30 % of the remaining time, but not less than 5 s
        sub_deadline = now + max(5.0, 0.30 * time_left)

        # sub‑gradient descent on the true objective
        best_seq, overall_best_score = _subgradient_descent(
            seed_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        overall_best_seq = best_seq.copy()

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random walk.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list of floats
    return overall_best_seq.tolist()
