def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = np.inf
    # small helper to update best
    def consider(a):
        nonlocal best_seq, best_score
        # ensure a non-zero sum
        if a.sum() == 0:
            return
        a = a / a.sum()
        score = _max_convolution_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best score={best_score:.6f}, len={len(a)}")
    # 1. families quick
    # Determine base lengths
    lengths = [800, 1200, 2000, 3000, 4000, 6000, 8000]
    for n in lengths:
        for gen in [_arcsine_distribution,
                    _triangular_distribution,
                    lambda m: _exponential_distribution(m, lam=0.005),
                    lambda m: _gaussian_distribution(m, sigma_factor=0.2),
                    lambda m: _cauchy_distribution(m, scale_factor=0.2)]:
            if time.time() >= deadline:
                break
            a = gen(n)
            consider(a)
    # also beta families
    for n in lengths:
        for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            consider(a)
            # unsymmetric
            for alpha in [0.5, 1.0, 2.0]:
                for beta in [0.5, 1.0, 2.0]:
                    a = _beta_distribution(n, alpha, beta)
                    consider(a)
    # incorporate previous best if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size>0:
            consider(prev)
    except Exception:
        pass

    # 2. refinement of current best via Adam for some time
    if best_seq is not None:
        # allow up to maybe 200 seconds for refinement
        refine_deadline = min(deadline, time.time() + 180)
        best_seq, best_score = _adam_descent(best_seq, refine_deadline, max_iters=500_000, step0=0.1)
        print(f"[{time.time():.1f}] After Adam refinement: score={best_score:.6f}")

    # 3. large random Dirichlet sampling
    # Decide length range
    large_min = 20000
    large_max = 50000
    alphas = [0.5, 1.0, 2.0, 5.0]
    while time.time() < deadline - 30:
        n = np.random.randint(large_min, large_max+1)
        alpha = np.random.choice(alphas)
        a = _random_dirichlet_alpha(n, alpha)
        consider(a)
        # optionally smooth with small Gaussian kernel (window)
        if np.random.rand() < 0.1:
            sigma = np.random.uniform(0.5, 3.0)
            # Gaussian kernel of length approx 6*sigma
            win_len = int(6 * sigma) | 1  # odd length
            idx = np.arange(win_len) - win_len//2
            kernel = np.exp(-0.5 * (idx / sigma) ** 2)
            kernel /= kernel.sum()
            a_smooth = np.convolve(a, kernel, mode='same')
            # ensure non-negative and renorm
            a_smooth = np.maximum(a_smooth, 0.0)
            if a_smooth.sum() > 0:
                consider(a_smooth)

    # 4. final short Adam on best (if time permits)
    if best_seq is not None and time.time() < deadline - 5:
        final_deadline = deadline
        best_seq, best_score = _adam_descent(best_seq, final_deadline, max_iters=300_000, step0=0.05)
        print(f"[{time.time():.1f}] Final Adam refinement: score={best_score:.6f}")

    # ensure normalization
    if best_seq is None:
        # fall back to uniform size 1000
        n = 1000
        best_seq = np.full(n, 1.0 / n)
        best_score = _max_convolution_score(best_seq)

    # project onto simplex (should already be)
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq.tolist()

"""
Optimiser for ``evaluate_sequence``

We use a three‑stage strategy that fits comfortably in the 1000 s budget:

1. **Broad sampling** – generate a large collection of analytic families
   (arcsine, triangular, exponential, Gaussian, Cauchy, symmetric/unsymmetric
   beta) on several moderate lengths (≈ 3‑8 k).  For each candidate we compute
   the exact surrogate  ``2·n·max(conv(a,a))``  (the same as the real score

2. **Heavy local optimisation** – the current best vector is refined by a

3. **Very‑large random probes** – finally we draw a few Dirichlet vectors on

   winner another short Adam polishing step.

All arithmetic is performed with real FFTs for speed, and every update is
projected onto the probability simplex to respect the problem’s invariances.
The routine respects the supplied time budget and returns a list of floats.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for ``evaluate_sequence`` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(conv(a,a)) on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide="ignore", invalid="ignore"):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a point on the simplex from Dirichlet(α,…,α)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.1,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam optimisation on the max‑convolution objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _fw_refine(a: np.ndarray, deadline: float,
               max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Frank‑Wolfe‑style pairwise mass transfer with a tiny line‑search.
    Transfers mass from the index with largest gradient (source) to the one
    with smallest gradient (destination).  Each iteration evaluates a few
    candidate step sizes (full, half, quarter) and keeps the best move.
    """
    cur_score = _max_convolution_score(a)
    n = a.shape[0]
    best_seq = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            break
        max_transfer = min(a[src], 1.0 - a[dst])
        if max_transfer < 1e-14:
            break

        # try a handful of step sizes
        candidates = [max_transfer, 0.5 * max_transfer,
                      0.25 * max_transfer, 0.75 * max_transfer]
        best_local = cur_score
        best_alpha = 0.0
        for alpha in candidates:
            a_tmp = a.copy()
            a_tmp[src] -= alpha
            a_tmp[dst] += alpha
            sc = _max_convolution_score(a_tmp)
            if sc < best_local:
                best_local = sc
                best_alpha = alpha

        if best_alpha > 0 and best_local < cur_score - 1e-12:
            a[src] -= best_alpha
            a[dst] += best_alpha
            cur_score = best_local
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] FW step {it:5d} new best {best_score:.6f}"
                )

        # occasional tiny jitter to escape plateaus
        if it % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s time budget and
    returns the best vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # helper to keep the globally best candidate
    best_seq = None
    best_score = np.inf

    def consider(v: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_score
        if v.size == 0:
            return
        v = v / v.sum()
        sc = _max_convolution_score(v)
        if sc < best_score - 1e-14:
            best_score = sc
            best_seq = v.copy()
            print(f"[{time.time():.1f}] NEW BEST ({tag}) score={best_score:.6f} n={len(v)}")

    # ------------------------------------------------------------------
    # 1. Broad analytical sampling (moderate lengths)
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break
        # basic families
        for gen, name in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exponential"),
            (lambda m: _gaussian_distribution(m, sigma_factor=0.2), "gaussian"),
            (lambda m: _cauchy_distribution(m, scale_factor=0.2), "cauchy"),
        ):
            consider(gen(n), f"{name}_n{n}")

        # symmetric beta sweep
        for p in (0.3, 0.5, 0.8, 1.2, 1.5, 2.0):
            consider(_beta_symmetric_distribution(n, p), f"beta_sym_{p}_n{n}")

        # unsymmetric beta sweep (small grid)
        for a in (0.5, 1.0, 1.5):
            for b in (0.5, 1.0, 1.5):
                consider(_beta_distribution(n, a, b), f"beta_{a}_{b}_n{n}")

    # ------------------------------------------------------------------
    # 2. Seed with any previously found sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            consider(prev, "prev")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Strong local refinement of the current best (if we have one)
    if best_seq is not None:
        # reserve a little time for the large‑scale random search later
        refine_deadline = min(deadline - 200, time.time() + 150)
        best_seq, best_score = _adam_descent(best_seq, refine_deadline,
                                             max_iters=800_000,
                                             step0=0.08)
        print(f"[{time.time():.1f}] after Adam   score={best_score:.6f}")

        # a few Frank‑Wolfe transfer steps
        fw_deadline = min(deadline - 120, time.time() + 80)
        best_seq, best_score = _fw_refine(best_seq, fw_deadline,
                                          max_iters=100_000)
        print(f"[{time.time():.1f}] after FW      score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Very‑large random Dirichlet probes (20k‑50k)
    large_min, large_max = 20000, 50000
    alphas = [0.5, 1.0, 2.0, 5.0]
    while time.time() < deadline - 30:
        n = np.random.randint(large_min, large_max + 1)
        alpha = np.random.choice(alphas)
        a = _random_dirichlet_alpha(n, alpha)
        consider(a, f"dirichlet_a{alpha}_n{n}")

        # occasional light Gaussian smoothing of the Dirichlet draw
        if np.random.rand() < 0.07:
            sigma = np.random.uniform(0.5, 2.5)
            win_len = int(6 * sigma) | 1
            x = np.arange(win_len) - win_len // 2
            kernel = np.exp(-0.5 * (x / sigma) ** 2)
            kernel /= kernel.sum()
            a_s = np.convolve(a, kernel, mode="same")
            a_s = np.maximum(a_s, 0.0)
            if a_s.sum() > 0:
                consider(a_s, f"smooth_dirichlet_a{alpha}_n{n}")

    # ------------------------------------------------------------------
    # 5. Final short Adam polishing (if any budget left)
    if best_seq is not None and time.time() < deadline - 5:
        best_seq, best_score = _adam_descent(best_seq, deadline,
                                             max_iters=300_000,
                                             step0=0.04)
        print(f"[{time.time():.1f}] final Adam   score={best_score:.6f}")

    # ------------------------------------------------------------------
    # safety fallback
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n)
        best_score = _max_convolution_score(best_seq)

    # Return a plain Python list as required by the evaluator
    return best_seq.tolist()
