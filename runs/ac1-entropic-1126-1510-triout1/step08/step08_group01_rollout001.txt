"""
Propose a step‑function minimizing `evaluate_sequence`.

The method works in three stages under a 1000 s budget:

1. **Broad sampling** – generate a rich set of analytic families
   (arcsine, triangular, exponential, Gaussian, Cauchy, uniform,
   symmetric Beta) on a variety of lengths.  Additionally create
   mixtures of a uniform distribution with arcsine and power‑transformed
   arcsine, and draw a few random Dirichlet points.

2. **Intensive refinement** – the best few seeds are polished by three
   mirror‑descent schemes (Adam, exponentiated‑gradient, simple
   sub‑gradient SGD).  Each optimiser respects the simplex constraint
   (non‑negative, unit ℓ₁‑norm).

3. **Frank‑Wolfe style pairwise mass transfer** – starting from the
   globally best vector we repeatedly move mass from the largest to the
   smallest gradient coordinate, accepting any improvement.  This cheap
   greedy loop often yields a final tiny reduction.

All vectors are kept ℓ₁‑normalised (sum = 1) for the cheap surrogate
score `2·n·max(conv(a,a))`; the returned sequence is finally rescaled
by `sqrt(2·n)` so that `evaluate_sequence` sees the same value as the
surrogate.  The routine respects the supplied time budget and always
returns a valid non‑negative list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(a*a) when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

# Analytic families (all return ℓ₁‑normalised vectors)
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Mirror‑descent variants (all keep ℓ₁‑norm = 1)
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 100_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        sc = _max_convolution_score(a)
        if sc < best_score - 1e-14:
            best_score, best = sc, a.copy()
    return best, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 80_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / len(a))
        sc = _max_convolution_score(a)
        if sc < best_score - 1e-14:
            best_score, best = sc, a.copy()
    return best, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 120_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        sc = _max_convolution_score(a)
        if sc < best_score - 1e-14:
            best_score, best = sc, a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _refine_sequence(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Apply a cascade of Adam → ExpGrad → SGD to polish a seed.
    Returns the best discovered version (still ℓ₁‑normalised).
    """
    # Adam
    seq_adam, sc_adam = _adam_descent(seq, deadline,
                                      max_iters=80_000,
                                      step0=0.25)
    best_seq, best_sc = seq_adam, sc_adam

    # ExpGrad
    seq_exp, sc_exp = _exp_gradient_descent(seq, deadline,
                                            max_iters=50_000,
                                            eta0=0.7)
    if sc_exp < best_sc - 1e-14:
        best_seq, best_sc = seq_exp, sc_exp

    # SGD
    seq_sgd, sc_sgd = _sgd_subgrad(seq, deadline,
                                   max_iters=80_000,
                                   step0=0.4)
    if sc_sgd < best_sc - 1e-14:
        best_seq, best_sc = seq_sgd, sc_sgd

    return best_seq, best_sc
