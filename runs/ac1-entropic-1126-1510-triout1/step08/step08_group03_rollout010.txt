def propose_candidate(seed=42, budget_s=1000):

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5

    # Determine base length
    try:
        base_n = len(height_sequence_1)
    except Exception:
        base_n = 4000

    # Determine lengths set
    lengths = set()
    for factor in [0.5, 0.7, 0.85, 1.0, 1.2, 1.4, 1.6, 2.0]:
        n = int(base_n * factor)
        if n >= 500:
            lengths.add(n)
    for _ in range(8):
        n = np.random.randint(500, min(base_n*3, 18000))
        lengths.add(n)
    lengths = sorted(lengths)[:10]   # limit to top e.g., 10 lengths to keep budget

    # Sampling
    sampling_deadline = time.time() + (budget_s * 0.2) - 5
    candidates = []
    beta_vals = np.logspace(-0.7, 0.7, 15)  # e.g., 0.2 to 5.0
    for n in lengths:
        if time.time() >= sampling_deadline:
            break

        # Add analytic families
        candidates.append(( _arcsine_distribution(n), _max_convolution_score(_arcsine_distribution(n)) ))
        candidates.append(( _triangular_distribution(n), _max_convolution_score(_triangular_distribution(n)) ))
        candidates.append(( _sine_distribution(n), _max_convolution_score(_sine_distribution(n)) ))
        candidates.append(( _cosine_distribution(n), _max_convolution_score(_cosine_distribution(n)) ))
        candidates.append(( np.full(n, 1.0/n), _max_convolution_score(np.full(n, 1.0/n)) ))

        # Semi-circle
        vals = np.sqrt( (np.arange(n, dtype=np.float64)+0.5) * (n - np.arange(n, dtype=np.float64) - 0.5) )
        semi = vals / vals.sum()
        candidates.append((semi, _max_convolution_score(semi)))

        # Exponential with random lambda
        for lam_exp in [0.001, 0.003, 0.01, 0.05]:
            cand = _exponential_distribution(n, lam=lam_exp)
            candidates.append((cand, _max_convolution_score(cand)))

        # Symmetric beta
        for p in beta_vals:
            cand = _beta_symmetric_distribution(n, p)
            candidates.append((cand, _max_convolution_score(cand)))
        # Asymmetric beta (random couples)
        for _ in range(3):
            p = np.random.choice(beta_vals)
            q = np.random.choice(beta_vals)
            cand = _beta_asymmetric_distribution(n, p, q)
            candidates.append((cand, _max_convolution_score(cand)))

        # Random Dirichlet (dense and sparse)
        for alpha in [0.5, 1.0, 2.0]:
            cand = np.random.gamma(alpha, 1.0, size=n)
            cand = cand / cand.sum()
            candidates.append((cand, _max_convolution_score(cand)))
        # Sparse Dirichlet: generate mask proportion.
        for sparsity in [0.05, 0.1, 0.2]:
            mask = np.random.rand(n) < sparsity
            if mask.sum() == 0:
                continue
            vals = np.random.gamma(0.5, 1.0, size=mask.sum())
            cand = np.zeros(n)
            cand[mask] = vals
            cand = cand / cand.sum()
            candidates.append((cand, _max_convolution_score(cand)))

        # Mixtures
        for _ in range(2):
            fams = [np.full(n, 1.0/n), _arcsine_distribution(n), _triangular_distribution(n), _sine_distribution(n), semi]
            a1 = np.random.choice(fams)
            a2 = np.random.choice(fams)
            lam_mix = np.random.rand()
            mix = lam_mix * a1 + (1 - lam_mix) * a2
            mix = mix / mix.sum()
            candidates.append((mix, _max_convolution_score(mix)))

    # Include prev sequence
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            prev_norm = prev / prev.sum()
            candidates.append((prev_norm, _max_convolution_score(prev_norm)))
    except Exception:
        pass

    # Keep top candidates
    candidates.sort(key=lambda x: x[1])
    top_k = min(30, len(candidates))
    top_seeds = [seq for seq,_ in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time() - start:.1f}s] initial cheap best={best_score:.6f}")

    # Optimization
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        return best_seq.tolist()

    per_seed_budget = remaining_time * 0.85 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # allocate budgets
        budget = per_seed_budget
        exp_budget = budget * 0.2
        adam_budget = budget * 0.3
        smooth_budget = budget * 0.5

        # Exp gradient descent
        seq, sc = _exp_gradient_descent(seed_seq, time.time() + exp_budget,
                                        max_iters=200_000, eta0=0.9)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start:.1f}s] seed {idx+1} exp -> {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, time.time() + adam_budget,
                                max_iters=200_000, step0=0.25)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start:.1f}s] seed {idx+1} adam -> {sc:.6f}")

        # Smoothed gradient
        seq, sc = _smooth_gradient_descent(seed_seq, time.time() + smooth_budget,
                                            max_iters=300_000, eta0=0.7,
                                            gamma0=5.0, gamma_factor=1.001)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time() - start:.1f}s] seed {idx+1} smooth -> {sc:.6f}")

    # Final hill climbing using fast convolution updates
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.03
    while time.time() < deadline:
        it += 1
        # gradually reduce transfer size
        transfer = max_transfer * (0.5 ** (it / 8000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time() - start:.1f}s] hill climb iter={it} new best {best_score:.6f}")

        if it % 4000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 1e-4
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()

Algorithm
---------

* **Sampling phase** (≈20 % of the time budget)
  – generate candidates from many analytic families:

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0.0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a) for a simplex‑normalised vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of ``max(a*a)`` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max

def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation method: start smooth, then increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _semicircle_distribution(n: int) -> np.ndarray:
    """w_i ∝ sqrt(i·(n‑i)) – a discrete semicircle."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sqrt(i * (n - i))
    return vals / vals.sum()


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α (α=1→uniform)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _sparse_dirichlet(n: int, sparsity: float = 0.1, alpha: float = 0.5) -> np.ndarray:
    """Sparse Dirichlet: keep only a fraction of entries."""
    mask = np.random.rand(n) < sparsity
    if mask.sum() == 0:
        return _random_dirichlet(n, alpha=1.0)
    vals = np.random.gamma(alpha, 1.0, size=mask.sum())
    a = np.zeros(n)
    a[mask] = vals
    return a / a.sum()
