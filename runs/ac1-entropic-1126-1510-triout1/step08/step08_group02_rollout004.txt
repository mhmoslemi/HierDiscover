"""
Hybrid optimisation for the step‑function problem.

The objective to minimise is

    E(a) = 2·n·max_k (a*a)_k / (sum_i a_i)^2 .

Because the objective is scale‑invariant we restrict ourselves to the
probability simplex (sum_i a_i = 1, a_i ≥ 0).  For a vector on the simplex
the cheap score 2·n·max_k (a*a)_k is exactly the value returned by
``evaluate_sequence``.

The search proceeds in several phases, all respecting the overall
time‑budget:

1. **Broad sampling** – generate candidates from many analytic families
   (arcsine, beta, triangular, exponential, Gaussian, Cauchy,
   semicircle, uniform, sine, cosine) and from Dirichlet draws with
   several concentration parameters.  Candidates are normalised and
   ranked by the cheap score.

2. **Heavy refinement** – the best ``top_k`` candidates are refined with
   a smoothed version of the max‑convolution using an Adam optimiser.
   The smoothing parameter (log‑sum‑exp temperature) is gradually reduced
   (continuation) so that the optimiser first sees a smooth landscape
   and finally the true max‑objective.

3. **Sub‑gradient descent** – a plain projected sub‑gradient method
   finishes the refinement on the current best vector.

4. **Guided hill‑climbing (simulated‑annealing style)** – small
   mass‑transfer moves are performed, guided by the exact sub‑gradient.
   Occasionally uphill moves are accepted according to a linear cooling
   schedule to escape shallow local minima.

5. **Upscaling (optional)** – if time permits, the best sequence is
   interpolated to a larger length and re‑optimised; larger supports
   often allow a lower supremum of the convolution.

The algorithm works completely on the simplex, so the returned list can
be fed directly to ``evaluate_sequence``.  Extensive use of FFT‑based
convolutions makes the per‑iteration cost essentially linearithmic in
the sequence length, allowing thousands of iterations well within the
1000 s budget.
"""

import time

# ----------------------------------------------------------------------
#  Utility functions

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.size
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b (length len(a)+len(b)-1) using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) (exact evaluation for sum(a)=1)."""
    n = a.size
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
#  Sub‑gradient and smooth gradient of the max‑convolution
# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.size
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)

    w = np.zeros_like(conv)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms i+j = 2*i (i.e. i=j)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using the log‑sum‑exp approximation:

        f_eps(a) = eps * log Σ_i exp(conv_i / eps)

    where conv = a*a.
    """
    n = a.size
    conv = _fft_self_convolve(a)                       # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)                    # unnormalised softmax weights
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (i=j terms counted once in the sum)
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]
    return grad

#  Optimisers

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the simplex (using exact sub‑gradient)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, 1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_anneal_descent(a0: np.ndarray, deadline: float,
                           max_iters: int = 300_000,
                           eps_start: float = 5e-2,
                           eps_decay: float = 0.9,
                           step0: float = 0.30,
                           beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smoothed max.
    The temperature ``eps`` is multiplied by ``eps_decay`` every
    ``decay_interval`` iterations, gradually turning the smooth objective
    into the true max.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    eps = eps_start
    decay_interval = 4000
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, 1.0)

        # evaluate exact max‑convolution (cheap because sum=1)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()

        if t % decay_interval == 0:
            eps = max(eps * eps_decay, 1e-7)
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, 1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _hill_climb(a0: np.ndarray, start_score: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Guided hill‑climbing with occasional simulated‑annealing moves.
    Moves consist of transferring a small amount of mass from an index
    with large positive sub‑gradient to an index with large negative
    sub‑gradient.
    """
    a = a0.copy()
    cur_score = start_score
    best_seq = a.copy()
    best_score = cur_score
    n = a.size
    start_time = time.time()
    iteration = 0
    max_transfer = 0.02                # initial fraction of total mass

    while time.time() < deadline:
        iteration += 1
        # decay the transfer magnitude geometrically
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        # every few steps we use the exact sub‑gradient to guide the move
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))          # most positive gradient -> reduce a_i
            j = int(np.argmin(g))          # most negative gradient -> increase a_j
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        # deterministic downhill move
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        else:
            # occasional uphill acceptance (simulated annealing)
            elapsed = time.time() - start_time
            total = deadline - start_time
            temperature = max(0.0, 1.0 - elapsed / total)   # linear cooling
            if temperature > 0 and np.random.rand() < np.exp((cur_score - new_score) / temperature):
                a = a_new
                cur_score = new_score

        # occasional jitter to escape flat plateaus
        if iteration % 5000 == 0 and np.random.rand() < 0.07:
            jitter = np.random.randn(n) * 4e-5
            a = _project_onto_simplex(a + jitter, 1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0          # map to [-1,1]
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw from Dirichlet(α,…,α) and normalise to sum = 1."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

        if n >= 500:
            lengths.add(n)
    max_len = int(min(14000, base_n * 3))
    for _ in range(9):
        lengths.add(np.random.randint(500, max_len + 1))
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1. Broad sampling
    candidates = []   # list of (array, cheap_score)

    beta_ps = [0.02, 0.04, 0.08, 0.12, 0.20, 0.35, 0.60, 1.0, 1.6, 2.5]
    exp_lams = [0.001, 0.003, 0.006, 0.010, 0.018]
    sigma_fracs = [0.14, 0.20, 0.30, 0.42]
    scale_fracs = [0.10, 0.18, 0.28, 0.40]
    dirichlet_alphas = [0.1, 0.3, 0.6, 1.0, 2.0]

    analytic_families = [
        (_arcsine_distribution, "arcsine"),
        (_triangular_distribution, "triangular"),
        (_uniform_distribution, "uniform"),
        (_sine_distribution, "sine"),
        (_cosine_distribution, "cosine"),
    ]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic analytic families
        for gen, name in analytic_families:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} {name:10s} score={score:.6f}")

        # Exponential tails
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} exponential(lam={lam:.3f}) score={score:.6f}")

        # Gaussian families
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} gaussian(sig={sf:.2f}) score={score:.6f}")

        # Cauchy families
        for sf in scale_fracs:
            a = _cauchy_distribution(n, scale_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} cauchy(scale={sf:.2f}) score={score:.6f}")

        # Semicircle
        a = _semicircle_distribution(n)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] n={n:4d} semicircle score={score:.6f}")

        # Symmetric beta
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} beta(p={p:5.2f}) score={score:.6f}")

        # Random Dirichlet draws
        for alpha in dirichlet_alphas:
            for _ in range(2):
                a = _random_dirichlet(n, alpha=alpha)
                score = _max_convolution_score(a)
                candidates.append((a, score))
                print(f"[{time.time():.1f}] n={n:4d} Dirichlet(α={alpha:.2f}) score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Include previously best sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(30, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Selected {len(top_candidates)} seeds for refinement; best cheap score={top_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy refinement (smooth‑anneal Adam, then plain Adam)
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # allocate a fraction of remaining time for refinement
    refinement_deadline = deadline - 50.0   # keep ~50 s for later phases
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= refinement_deadline:
            break

        # (a) smooth‑anneal Adam
        seq, sc = _smooth_anneal_descent(seed_seq, refinement_deadline,
                                         max_iters=200_000,
                                         eps_start=4e-2,
                                         eps_decay=0.85,
                                         step0=0.28)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smooth‑Adam → {sc:.6f}")

        # (b) plain Adam (helps fine‑tune around the smoothed optimum)
        seq, sc = _adam_descent(seed_seq, refinement_deadline,
                                max_iters=120_000,
                                step0=0.22)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient descent on the current best
    subgrad_deadline = deadline - 25.0
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=1_000_000,
                                       eta0=0.55)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Guided hill‑climbing (with simulated‑annealing acceptance)
    hill_deadline = deadline
    seq, sc = _hill_climb(best_seq, best_score, hill_deadline)
    if sc < best_score - 1e-14:
        best_score, best_seq = sc, seq.copy()
        print(f"[{time.time():.1f}] hill‑climb final best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Optional up‑scaling if time permits (try a longer support)
    if time.time() < deadline - 5.0:
        try:
            upscale_factor = 1.5
            n_old = best_seq.size
            n_new = int(min(14000, n_old * upscale_factor))
            # linear interpolation onto new grid
            old_idx = np.linspace(0, 1, n_old)
            new_idx = np.linspace(0, 1, n_new)
            a_interp = np.interp(new_idx, old_idx, best_seq)
            a_interp = a_interp / a_interp.sum()
            up_deadline = deadline
            seq, sc = _smooth_anneal_descent(a_interp, up_deadline,
                                             max_iters=150_000,
                                             eps_start=2e-2,
                                             eps_decay=0.88,
                                             step0=0.24)
            if sc < best_score - 1e-14:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] up‑scaled refinement → {sc:.6f}")
        except Exception as e:
            # up‑scaling failed – ignore
            print(f"[{time.time():.1f}] up‑scale exception: {e}")

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # Return plain Python list as required by ``evaluate_sequence``.
    return best_seq.tolist()
