"""
Propose a step‑function that (approximately) minimises the
`evaluate_sequence` score.

The algorithm proceeds in two stages:

1️⃣  **Differential‑Evolution (DE) global search**  
    A small population of probability vectors (summing to 1) is
    evolved on the simplex using classic DE operators (mutation,
    crossover and projection).  During the DE run we keep the best
    vector seen according to the exact surrogate
    `score(a)=2·n·max(conv(a,a))`, which equals the true evaluation
    for normalised vectors.

2️⃣  **Local polishing**  
    The best DE individual is refined with three complementary
    first‑order methods:
      – Adam on the true sub‑gradient of the max‑convolution,
      – Exponentiated‑gradient (mirror descent w.r.t. the KL‑divergence),
      – Simple projected sub‑gradient descent.
    Finally a short Frank‑Wolfe style mass‑transfer loop is performed.

All steps respect the global time budget (≈ 1000 s).  The returned
list sums to 1, hence it is safe for `evaluate_sequence` (the score
is invariant under positive scaling).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast O(n log n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(conv(a,a)) on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (the term a_i*a_i appears once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 800_000,
                  step0: float = 0.25,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        sc = _max_convolution_score(a)
        if sc < best_score - 1e-14:
            best_score, best = sc, a.copy()
    return best, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.9,
                          decay: float = 0.9999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a_sum
        sc = _max_convolution_score(a)
        if sc < best_score - 1e-14:
            best_score, best = sc, a.copy()
    return best, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 500_000,
                 step0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        sc = _max_convolution_score(a)
        if sc < best_score - 1e-14:
            best_score, best = sc, a.copy()
    return best, best_score


def _frank_wolfe(a: np.ndarray, cur_score: float,
                 deadline: float, max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """Simple pairwise mass‑transfer loop (already used in the reference)."""
    best = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta < 1e-14:
            break
        # try full transfer
        a_full = a.copy()
        a_full[i_src] -= delta
        a_full[i_dst] += delta
        sc_full = _max_convolution_score(a_full)
        improved = False
        if sc_full < cur_score - 1e-12:
            a, cur_score = a_full, sc_full
            improved = True
        else:
            # try half transfer
            half = delta * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            sc_half = _max_convolution_score(a_half)
            if sc_half < cur_score - 1e-12:
                a, cur_score = a_half, sc_half
                improved = True
        if improved and cur_score < best_sc - 1e-14:
            best_sc, best = cur_score, a.copy()
    return best, best_sc


# ----------------------------------------------------------------------
def _de_global_search(n: int, deadline: float,
                      popsize: int = 24,
                      F: float = 0.5,
                      CR: float = 0.9) -> tuple[np.ndarray, float]:
    """Differential‑Evolution on the probability simplex."""
    # initialise population with Dirichlet draws
    rng = np.random.default_rng()
    pop = [rng.gamma(0.5, 1.0, size=n) for _ in range(popsize)]
    pop = [p / p.sum() for p in pop]
    scores = [_max_convolution_score(p) for p in pop]

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        for i in range(popsize):
            # mutation: a + F*(b - c)
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # enforce non‑negativity then project to simplex
            mutant = np.maximum(mutant, 0.0)
            mutant = _project_onto_simplex(mutant, s=1.0)

            # crossover
            cross_mask = rng.random(n) < CR
            # ensure at least one component from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = _project_onto_simplex(trial, s=1.0)

            trial_score = _max_convolution_score(trial)
            if trial_score < scores[i] - 1e-14:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score - 1e-14:
                    best_score, best_vec = trial_score, trial.copy()
        # optional early stop when no improvement for a while
        if time.time() > deadline - 2.0:
            break
    return best_vec, best_score
