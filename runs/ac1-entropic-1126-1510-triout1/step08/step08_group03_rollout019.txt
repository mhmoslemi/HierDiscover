"""
Propose a step function that minimizes ``evaluate_sequence``.

Strategy
--------
1.   Build a diverse pool of candidate distributions (arcsine, Beta
     families, power‑law, exponential, mixtures and random Dirichlet draws)
     for several lengths (≈500–3000).  All candidates are normalised to
     sum = 1 and scored with the cheap proxy
     ``2·n·max(convolution(a,a))`` (identical to the true score for a
     normalised input).

2.   Keep the best *few* seeds and polish each of them intensively.
     Polishing consists of three stages
        • a few iterations of exponentiated‑gradient descent,
        • a short *linear‑program* refinement (maximise Σg subject to
          a∗g ≤ max(a∗a); this is the LP described in the paper),
        • a continuation of the smooth‑max (soft‑max) gradient descent
          with a slowly growing smoothness parameter γ.

3.   Finish with a fine‑grained hill‑climbing loop that repeatedly moves
     a tiny amount of mass between two randomly chosen positions.
     The amount of transferred mass decays geometrically, and occasional
     jitter is added to escape plateaus.

All stages respect the global time budget (default 1000 s).  The function
returns the best sequence discovered.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) – x is never exactly 0 or 1
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals /= vals.sum()

def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, expnt: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+½)^‑α + (n‑i‑½)^‑α."""
    i = np.arange(n, dtype=np.float64)
    left = (i + 0.5) ** (-expnt)
    right = (n - i - 0.5) ** (-expnt)
    vals = left + right
    vals /= vals.sum()
    return vals


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 120_000,
                          eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient of max(a*a)
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a /= a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best = cur, a.copy()
    return best, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 120_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best = cur, a.copy()
    return best, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best = cur, a.copy()
    return best, best_score


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Smooth‑max surrogate --------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    diagonal = w[2 * np.arange(n)]               # positions where k = 2*i
    grad -= a * diagonal
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 250_000,
                             eta0: float = 0.7,
                             gamma0: float = 1.0,
                             gamma_factor: float = 1.004):
    """Continuation method: start with a smooth max (γ≈1) and increase γ."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score, best = cur_score, a.copy()
        gamma *= gamma_factor
    return best, best_score

# Linear‑program refinement ---------------------------------------------
def _lp_refine(a0: np.ndarray, deadline: float,
               max_iters: int = 5, mixing: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Alternating LP refinement described in the paper:
    maximise Σ g   s.t.   a∗g ≤ max(a∗a),   g≥0.
    The obtained g is mixed back into a several times.
    """
    a = a0.copy()
    n = len(a)
    best = a.copy()
    best_score = _max_convolution_score(a)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        # current bound M = max(a*a)
        M = _fft_self_convolve(a).max()

        # build the (2n‑1)×n convolution matrix A (dense, but still modest)
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = a[i]

        # linear programme: maximise Σ g_i  ⇔  minimise −Σ g_i
        c = -np.ones(n)                       # minimise –sum(g)
        b_ub = np.full(rows, M)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break

        g0 = res.x
        if g0.sum() == 0:
            break
        g = g0 / g0.sum()                     # normalise to Σ g = 1

        # convex mix back into a
        a = (1.0 - mixing) * a + mixing * g
        a = _project_onto_simplex(a)          # safety

        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score, best = cur_score, a.copy()
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # Choose lengths (the search is cheap for moderate n)
    try:
        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        base_n = 2000

    lengths = set()
    for f in (0.6, 0.8, 1.0, 1.3, 1.6, 2.0):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(3000, base_n * 2) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # --------------------------------------------------------------
    # 1. Broad sampling of analytic families + random draws
    candidates = []   # (array, cheap_score)

    beta_ps = np.concatenate([np.linspace(0.05, 0.5, 6),
                             np.linspace(0.6, 2.0, 5),
                             np.array([3.0, 5.0])])
    power_exps = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic analytic families
        for fn in (_arcsine_distribution,
                   _triangular_distribution,
                   lambda n: _exponential_distribution(n, lam=0.004),
                   _sine_distribution,
                   _cosine_distribution,
                   lambda n: np.full(n, 1.0 / n)):
            a = fn(n)
            candidates.append((a, _max_convolution_score(a)))

        # symmetric Beta with a fine grid of p
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # asymmetric Beta (coarse grid)
        for p in (0.1, 0.3, 0.5, 1.0, 2.0):
            for q in (0.1, 0.5, 1.0, 2.0, 5.0):
                a = _beta_asymmetric_distribution(n, p, q)
                candidates.append((a, _max_convolution_score(a)))

        # power‑law families
        for e in power_exps:
            a = _power_law_distribution(n, e)
            candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws
        for _ in range(3):
            a = _random_dirichlet(n)
            candidates.append((a, _max_convolution_score(a)))

        # random mixtures of two families
        for _ in range(2):
            # choose two base families at random
            fam1 = np.random.choice(['arcsine', 'triangular', 'uniform', 'sine'])
            fam2 = np.random.choice(['beta_sym', 'beta_asym', 'exponential', 'power'])

            if fam1 == 'arcsine':
                a1 = _arcsine_distribution(n)
            elif fam1 == 'triangular':
                a1 = _triangular_distribution(n)
            elif fam1 == 'uniform':
                a1 = np.full(n, 1.0 / n)
            else:  # sine
                a1 = _sine_distribution(n)

            if fam2 == 'beta_sym':
                p = np.random.choice(beta_ps)
                a2 = _beta_symmetric_distribution(n, p)
            elif fam2 == 'beta_asym':
                p = np.random.choice(beta_ps)
                q = np.random.choice(beta_ps)
                a2 = _beta_asymmetric_distribution(n, p, q)
            elif fam2 == 'exponential':
                lam = 10 ** np.random.uniform(-4, -2)
                a2 = _exponential_distribution(n, lam=lam)
            else:  # power‑law
                e = np.random.choice(power_exps)
                a2 = _power_law_distribution(n, e)

            mix = np.random.rand()
            a = mix * a1 + (1.0 - mix) * a2
            a /= a.sum()
            candidates.append((a, _max_convolution_score(a)))

    # --------------------------------------------------------------
    # 2. Include the previous best sequence, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    top_seeds = [seq for seq, _ in candidates[:top_k]]

    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Heavy optimisation (time‑sharing)
    remaining = deadline - time.time()
    per_seed_budget = remaining * 0.85 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        seed_deadline = time.time() + per_seed_budget

        # a) exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, seed_deadline,
                                        max_iters=80_000, eta0=0.85)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} exp‑grad → {sc:.6f}")

        # b) linear‑program refinement (only for moderate lengths)
        if len(seq) <= 1500 and time.time() < seed_deadline:
            seq, sc = _lp_refine(seq, seed_deadline,
                                 max_iters=4, mixing=0.3)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{time.time():.1f}] seed {idx+1} LP‑refine → {sc:.6f}")

        # c) Adam
        seq, sc = _adam_descent(seed_seq, seed_deadline,
                                max_iters=80_000, step0=0.25)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} Adam → {sc:.6f}")

        # d) smooth‑max continuation
        seq, sc = _smooth_gradient_descent(seed_seq, seed_deadline,
                                            max_iters=200_000,
                                            eta0=0.7,
                                            gamma0=1.0,
                                            gamma_factor=1.004)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} smooth → {sc:.6f}")

        # e) plain sub‑gradient (final tidy‑up)
        seq, sc = _subgradient_descent(seed_seq, seed_deadline,
                                       max_iters=250_000, eta0=0.5)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} subgrad → {sc:.6f}")

    # --------------------------------------------------------------
    # 5. Fine‑grained hill climbing (tiny random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.03

    while time.time() < deadline:
        it += 1
        # Geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (it / 12_000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 5_000 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 3e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
