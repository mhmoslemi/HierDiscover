def lp_step(f, M, n):
    # Build matrix A (2n-1) x n
    rows = []
    cols = []
    data = []
    for i in range(n):
        # column i values: f[0..n-1] placed at rows i...i+n-1
        row_idx = np.arange(i, i+n)
        col_idx = np.full(n, i)
        rows.append(row_idx)
        cols.append(col_idx)
        data.append(f)   # broadcast
    # flatten
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.tile(f, n)  # or repeat f n times
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    # Solve LP
    c = -np.ones(n)  # maximize sum(g)
    b = np.full(2*n-1, M)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    g = res.x
    return g, res.success

"""
propose_candidate
-----------------

Search for a non‑negative step function (a probability vector) that minimises
``evaluate_sequence``.  The method consists of three phases:

1. **Initialisation** – start from the best known sequence (``height_sequence_1``)
   if it exists, otherwise use a smooth analytic family (arcsine).  The vector
   is normalised to have total mass ``sqrt(2·n)``; with this normalisation the
   objective reduces to ``max(conv(f,f))``.

2. **LP‑mixing refinement** – repeatedly solve a linear programme
   ``max   1ᵀg``  subject to  ``f * g ≤ M``  (convolution) and ``g ≥ 0``.
   The optimiser (SciPy HiGHS) finds a complementary shape ``g`` that can be
   mixed with the current ``f`` while keeping the maximal self‑convolution
   below the current bound ``M``.  A back‑tracking line‑search chooses the mixing
   weight ``t``.  A few such steps dramatically lower the peak of the
   convolution.

3. **Projected sub‑gradient descent** – with the remaining budget we run a
   long sub‑gradient descent on the exact max‑convolution (the same routine
   used in the reference implementation).  The step size is ``η_k = η₀/√k`` and
   each iterate is projected back onto the simplex of mass ``sqrt(2·n)``.

The final vector is re‑scaled to sum = 1 (the scaling does not change the
objective) and returned as a plain Python list.  All progress is printed.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex (mass = sqrt(2·n))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of a_rev with w
    conv_rev = np.fft.irfft(np.fft.rfft(a[::-1]) * np.fft.rfft(w, n=1 << ((2 * n - 1) - 1).bit_length()))
    conv_rev = conv_rev[: 2 * n - 1]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=a0.sum())
        cur = _fft_self_convolve(a).max()
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Construct the matrix A (shape (2n‑1)×n) such that (f * g) = A @ g.
    The matrix is dense but stored as a sparse CSC matrix.
    """
    n = f.shape[0]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    idx = 0
    for i in range(n):
        # column i contributes f[0..n‑1] to rows i..i+n‑1
        rows[idx : idx + n] = np.arange(i, i + n, dtype=np.int32)
        cols[idx : idx + n] = i
        data[idx : idx + n] = f
        idx += n

    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A.tocsc()


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float, mass: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   1ᵀg
        subject to A @ g ≤ M·1 , g ≥ 0
    where A represents convolution with f.  Returns the optimal g
    (not yet rescaled) or ``None`` if the LP fails.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)          # sparse CSC
    c = -np.ones(n)                    # maximise sum(g)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    res = opt.linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    g = res.x
    # In rare cases the LP may return a near‑zero solution; guard against that.
    if g.sum() < 1e-12:
        return None
    # Rescale g so that its total mass equals the target ``mass`` (sqrt(2n)).
    scale = mass / g.sum()
    return g * scale

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1) Determine the length n and initialise f.
    try:

    mass_target = np.sqrt(2.0 * n)

    outer_iter = 0
