"""
Propose a non‑negative step function (probability vector) that
minimises ``evaluate_sequence``.

Strategy
--------
1.  Build a diverse pool of candidate vectors using analytic families
    (arcsine, beta, triangular, Gaussian, exponential, …) and random
    Dirichlet draws for several lengths (≈ 500 … 10 000).
2.  Keep the best *k* seeds (cheap score = 2·n·max(a*a) with ∑a=1).
3.  Refine each seed with three first‑order methods:
        – exponentiated‑gradient (mirror descent on the simplex),
        – Adam on the Euclidean projection,
        – Adam on a log‑sum‑exp smoothing of the max‑convolution.
4.  Run a long projected sub‑gradient descent on the current best.
5.  Apply an aggressive simulated‑annealing / hill‑climbing stage:
        * repeatedly transfer a small amount of mass from a high‑gradient
          index to a low‑gradient index,
        * accept uphill moves with a temperature schedule,
        * slowly shrink the transfer magnitude.
All steps respect the overall time budget (default 1000 s).  The
returned list is a probability vector (sum = 1) which can be fed
directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full autocorrelation a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective 2·n·max(a*a) for a probability vector (sum=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0                # select the active entries of the max

    # convolution of a (reversed) with the selector w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (each term a_i*a_i counted once, not twice)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using a log‑sum‑exp approximation:
        f_eps(a) = eps * log Σ_i exp(conv_i / eps)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)           # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for the smoothing weights
    diag_idx = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_idx] -= w[2 * diag_idx] * a[diag_idx]

    return grad

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Mirror descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best = cur, a.copy()
    return best, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam with Euclidean projection onto the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best = cur, a.copy()
    return best, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 150_000,
                         eps0: float = 1e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smoothed objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score:

    return best, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best = cur, a.copy()
    return best, best_score

# Analytic families – all return a probability vector (sum=1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a probability vector that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and
    returns a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # a small safety margin

    # ---------------------------------------------------------------
    # Determine a baseline length (use previous best if available)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # Choose a set of lengths to explore (mix of nearby and far)
    lengths = set()
    for f in (0.7, 0.85, 1.0, 1.15, 1.3):
        n = int(base_n * f)
        if 400 <= n <= 12000:
            lengths.add(n)
    for _ in range(7):
        n = np.random.randint(400, min(15000, base_n * 4) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ---------------------------------------------------------------
    # 1. Broad sampling of analytic families and random draws
    candidates = []   # (vector, cheap_score)

    beta_ps = [0.02, 0.04, 0.08, 0.12, 0.20, 0.35, 0.55, 0.80, 1.2, 1.6, 2.0]
    exp_lams = [0.001, 0.003, 0.006, 0.012]
    sigma_fracs = [0.16, 0.25, 0.35, 0.45]
    scale_fracs = [0.12, 0.20, 0.30, 0.45]
    dirichlet_alphas = [0.3, 0.6, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_uniform_distribution, "uniform"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_semicircle_distribution, "semicircle"),
        ]
        for gen, name in fams:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} {name:10s} score={score:.6f}")

        # exponential tails
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} exponential(lam={lam:.3f}) score={score:.6f}")

        # Gaussian
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} gaussian(sig={sf:.2f}) score={score:.6f}")

        # Cauchy
        for sf in scale_fracs:
            a = _cauchy_distribution(n, scale_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} cauchy(scale={sf:.2f}) score={score:.6f}")

        # Symmetric beta
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} beta(p={p:4.2f}) score={score:.6f}")

        # Random Dirichlet draws
        for alpha in dirichlet_alphas:
            for _ in range(2):
                a = _random_dirichlet(n, alpha=alpha)
                score = _max_convolution_score(a)
                candidates.append((a, score))
                print(f"[{time.time():.1f}] n={n:5d} Dirichlet(α={alpha:.2f}) score={score:.6f}")

    # ---------------------------------------------------------------
    # 2. Insert the previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / s
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # ---------------------------------------------------------------
    # 3. Keep the best seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 30
    seeds = candidates[:top_k]
    print(f"Selected {len(seeds)} seeds for refinement; best cheap score={seeds[0][1]:.6f}")

    best_seq = seeds[0][0].copy()
    best_score = seeds[0][1]

    # ---------------------------------------------------------------
    # 4. Heavy refinement on each seed (exp‑grad, Adam, smooth‑Adam)
    for idx, (seed_vec, _) in enumerate(seeds):
        if time.time() >= deadline:
            break

        # exponentiated‑gradient
        seq, sc = _exp_gradient_descent(
            seed_vec, deadline,
            max_iters=120_000,
            eta0=0.9)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} exp‑grad → {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(
            seed_vec, deadline,
            max_iters=120_000,
            step0=0.25)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} Adam → {sc:.6f}")

        # smooth Adam
        seq, sc = _smooth_adam_descent(
            seed_vec, deadline,
            max_iters=80_000,
            eps0=1e-3,
            step0=0.22)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} smooth‑Adam → {sc:.6f}")

    # ---------------------------------------------------------------
    # 5. Long projected sub‑gradient descent on the current best
    sub_deadline = deadline - 35.0   # keep a margin for the final hill‑climb
    if time.time() < sub_deadline:
        seq, sc = _subgradient_descent(
            best_seq, sub_deadline,
            max_iters=2_000_000,
            eta0=0.6)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # ---------------------------------------------------------------
    # 6. Simulated‑annealing / hill‑climbing phase
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.06               # up to 6 % of total mass initially
    init_temp = 0.001 * cur_score    # temperature proportional to current score

    while time.time() < deadline:
        iteration += 1

        # decay of the transfer magnitude (slow exponential)
        transfer = max_transfer * (0.5 ** (iteration / 40000.0))
        temperature = init_temp * (0.995 ** (iteration / 500.0))

        # decide whether to use a gradient‑guided move
        if iteration % 5 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        # amount of mass to move
        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)

        # accept if improvement, else accept with probability exp(-Δ/T)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a_new.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")
        else:
            accept_prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if np.random.rand() < accept_prob:
                a = a_new
                cur_score = new_score

        # occasional random jitter to escape plateaus
        if iteration % 7000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={best_seq.size})")
    return best_seq.tolist()
