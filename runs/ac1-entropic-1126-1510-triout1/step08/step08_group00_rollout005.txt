"""
Algorithm to minimise 2·n·max(conv(a,a))/(∑a)²  (the value returned by
evaluate_sequence).  The steps are:

1️⃣  Broad sampling – generate many analytic families (arcsine with variable
    offset, beta, power‑law, sine, triangular, exponential) and a large
    collection of random Dirichlet draws (including very sparse draws).
    All candidates are ℓ₁‑normalised (∑a=1).  The cheap surrogate
    2·n·max(conv(a,a)) is used for ranking.

2️⃣  Intensive refinement – keep the top‑k candidates (k≈10) and improve
    each of them with several runs of projected Adam and exponentiated‑
    gradient descent, using a few different step‑size settings.

3️⃣  Aggressive Frank‑Wolfe mass‑transfer – repeatedly move mass from the
    index with the largest sub‑gradient to the index with the smallest
    sub‑gradient.  For each move a short line‑search (full, ¾, ½, ¼ of the
    transferable mass) is evaluated and the best transfer is kept.

4️⃣  Simulated‑annealing / random‑swap polishing – with a linear temperature
    schedule transfer small amounts of mass between random pairs and also
    try random swaps of two positions.  Deteriorations are accepted with
    Metropolis probability.

All operations stay on the simplex (a≥0, ∑a=1) so the true objective is
invariant under scaling; the final list can be passed directly to

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(a,a)) (exact when sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 200_000):
    """Projected Adam on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 150_000):
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _sparse_dirichlet(n: int, support: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet on a random support of given size."""
    if support >= n:
        return _random_dirichlet_alpha(n, alpha)
    idx = np.random.choice(n, size=support, replace=False)
    vals = np.random.gamma(alpha, 1.0, size=support)
    vals = vals / vals.sum()
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a

def _arcsine_distribution(n: int, offset: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + offset) * (n - i - offset))
    vals = np.where(np.isfinite(vals), vals, 0.0)
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _parabolic_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = i * (n - i - 1)
    vals = np.maximum(vals, 0)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals

            lengths.add(n)
    for _ in range(8):
        n = np.random.randint(1500, min(15000, base_n * 2) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring {len(lengths)} lengths (base ≈{base_n})")

    # --------------------------------------------------------------
    # 2️⃣ Broad sampling of analytic families and random draws
    candidates = []   # list of (array, surrogate-score)

    # parameter grids
    arcsine_offsets = [0.2, 0.4, 0.5, 0.8, 1.0, 1.5]
    beta_ps = np.concatenate(
        [np.arange(0.05, 0.25, 0.05), np.arange(0.3, 1.6, 0.2), [2.0]]
    )
    beta_asym_grid = [(0.3, 0.7), (0.5, 0.5), (0.7, 0.3), (0.2, 1.0), (1.0, 0.2),
                      (0.1, 0.9), (0.9, 0.1)]
    power_ps = [0.3, 0.5, 0.7, 0.9, 1.1, 1.4, 2.0]
    exponential_lams = [0.001, 0.005, 0.02, 0.05]
    dirichlet_alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    sparse_support_fractions = [0.01, 0.02, 0.05, 0.10]   # fraction of n

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families -------------------------------------------------
        for name, gen in [
            ("arcsine", lambda m: _arcsine_distribution(m, offset=0.5)),
            ("triangular", _triangular_distribution),
            ("sine", _sine_distribution),
            ("cosine", _cosine_distribution),
            ("parabolic", _parabolic_distribution),
        ]:
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} {name:12s}")

        # arcsine with different offsets
        for off in arcsine_offsets:
            a = _arcsine_distribution(n, offset=off)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} arcsine(off={off})")

        # exponential family
        for lam in exponential_lams:
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} exponential(lam={lam})")

        # symmetric beta
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))
        # asymmetric beta
        for a_alpha, a_beta in beta_asym_grid:
            a = _beta_asymmetric_distribution(n, a_alpha, a_beta)
            candidates.append((a, _max_convolution_score(a)))

        # power‑law
        for p in power_ps:
            a = _power_law_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws (dense)
        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

        # sparse Dirichlet draws
        for frac in sparse_support_fractions:
            support = max(2, int(frac * n))
            a = _sparse_dirichlet(n, support, alpha=0.5)
            a = a / a.sum()
            candidates.append((a, _max_convolution_score(a)))

    # ----------------------------------------------------------------------
    # 3️⃣ Include the previously known best (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(
                f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}"
            )
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # 4️⃣ Keep the top‑k seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    top_seeds = candidates[:top_k]
    best_seq, best_score = top_seeds[0][0].copy(), top_seeds[0][1]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5️⃣ Refinement of each seed (Adam + EG with a few different settings)
    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # Adam with moderate step
        seq_adam_a, sc_adam_a = _adam_descent(
            seed_seq, deadline, step0=0.20, max_iters=120_000
        )
        if sc_adam_a < best_score - 1e-14:
            best_score, best_seq = sc_adam_a, seq_adam_a.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam(0.2) → {best_score:.6f}")

        # Adam with larger step (more aggressive)
        seq_adam_b, sc_adam_b = _adam_descent(
            seed_seq, deadline, step0=0.45, max_iters=120_000
        )
        if sc_adam_b < best_score - 1e-14:
            best_score, best_seq = sc_adam_b, seq_adam_b.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam(0.45) → {best_score:.6f}")

        # Exponentiated‑gradient
        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, deadline, max_iters=100_000)
        if sc_eg < best_score - 1e-14:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} ExpGrad → {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 6️⃣ Frank‑Wolfe mass‑transfer with line‑search
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    max_fw_iters = 200_000
    while time.time() < deadline and fw_iter < max_fw_iters:
        fw_iter += 1
        grad, _, _ = _subgrad(a)

        i_src = int(np.argmax(grad))   # mass should leave here
        i_dst = int(np.argmin(grad))   # mass should go here
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try several fractions of the transferable mass
        best_local_seq = None
        best_local_score = cur_score
        for frac in (1.0, 0.75, 0.5, 0.25):
            alpha = max_transfer * frac
            a_cand = a.copy()
            a_cand[i_src] -= alpha
            a_cand[i_dst] += alpha
            sc = _max_convolution_score(a_cand)
            if sc < best_local_score - 1e-13:
                best_local_score = sc
                best_local_seq = a_cand
        if best_local_seq is not None:
            a = best_local_seq
            cur_score = best_local_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] FW iter={fw_iter} improved to {best_score:.6f}")
            continue

        # occasional random jitter to get out of a tiny plateau
        if fw_iter % 7_000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ----------------------------------------------------------------------
    # 7️⃣ Simulated‑annealing / random‑swap polishing
    a = best_seq.copy()
    cur_score = best_score
    anneal_iter = 0
    temp_start = 0.03
    temp_end = 0.001
    while time.time() < deadline and anneal_iter < 300_000:
        anneal_iter += 1
        # linear temperature schedule based on remaining fraction
        t_rem = max(0.0, (deadline - time.time()) / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * t_rem

        # propose a mass‑transfer move
        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(
                    f"[{time.time():.1f}] Anneal iter={anneal_iter} improved to {best_score:.6f}"
                )
            continue

        # Metropolis acceptance
        prob = math.exp(-(new_score - cur_score) / (temp + 1e-12))
        if np.random.rand() < prob:
            a = a_new
            cur_score = new_score

        # random swap move (occasionally)
        if anneal_iter % 10000 == 0 and np.random.rand() < 0.15:
            i_swap, j_swap = np.random.choice(a.size, 2, replace=False)
            a_sw = a.copy()
            a_sw[i_swap], a_sw[j_swap] = a_sw[j_swap], a_sw[i_swap]
            sc_sw = _max_convolution_score(a_sw)
            if sc_sw < cur_score - 1e-15:
                a = a_sw
                cur_score = sc_sw
                if cur_score < best_score - 1e-14:
                    best_score, best_seq = cur_score, a.copy()
                    print(
                        f"[{time.time():.1f}] Swap iter={anneal_iter} improved to {best_score:.6f}"
                    )

    # ----------------------------------------------------------------------
    print(
        f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})"
    )
    # ``evaluate_sequence`` works with any scaling; we keep the ℓ₁‑normalized form.
    return best_seq.tolist()
