    2·n·max(conv(p,p))

where p is the returned list normalised to sum = 1.
The algorithm builds a diverse set of analytic seeds, picks the best,

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using a real FFT."""
    n = a.shape[0]
    # length needed for a∗a is 2·n‑1; choose power‑of‑two for speed
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  – a is assumed to sum to 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _subgradient(p: np.ndarray, active_shifts: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of  max(conv(p,p))  w.r.t. p.
    For each active shift k we add 2·p_{k-i} to g_i.
    """
    n = p.shape[0]
    g = np.zeros_like(p)
    for k in active_shifts:
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        i = np.arange(i_lo, i_hi + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * p[j]
    return g

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid under/overflow for tiny p
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isnan(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a

    # ------------------------------------------------------------------
    # 3) Hybrid optimisation ------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # ----- convolution & active set -----
        conv = _fft_self_convolve(cur)
        M = conv.max()
        eps = max(M * 1e-12, 1e-14)
        active = np.where(conv >= M - eps)[0]

        # ----- sub‑gradient of the max‑conv -----
        g = _subgradient(cur, active)

        # ----- exponentiated‑gradient step -----
        # multiplicative update with a decaying learning rate
        eta = 0.5 / (1.0 + it / 5000.0)
        # shift gradient to avoid overflow/underflow
        g_shift = g - g.min()
        exp_factor = np.exp(-eta * g_shift)
        new = cur * exp_factor
        s = new.sum()
        if s == 0 or np.isnan(s):
            new = _distribution_uniform(n)
        else:
            new /= s

        new_score = _objective(new)

        # accept if we have improved the current point
        if new_score < cur_score:
            cur = new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_vec = cur.copy()
                print(f"[{time.time():.2f}] EG it={it} improved to {best_score:.6f}")

        # ----- occasional greedy mass‑transfer -----
        if it % 400 == 0:
            # source = index with large gradient that still has mass
            src_candidates = np.where(cur > 1e-12)[0]
            if src_candidates.size:
                src = src_candidates[np.argmax(g[src_candidates])]
                dst = int(np.argmin(g))
                if src != dst:
                    delta = min(cur[src], 0.5)      # move at most half of the source mass
                    trial = cur.copy()
                    trial[src] -= delta
                    trial[dst] += delta
                    trial_score = _objective(trial)
                    if trial_score < cur_score - 1e-14:
                        cur = trial
                        cur_score = trial_score
                        if trial_score < best_score - 1e-14:
                            best_score = trial_score
                            best_vec = cur.copy()
                            print(f"[{time.time():.2f}] greedy it={it} improved to {best_score:.6f}")
                    else:
                        # try a smaller move if the full one fails
                        delta2 = delta * 0.2
                        if delta2 > 1e-9:
                            trial2 = cur.copy()
                            trial2[src] -= delta2
                            trial2[dst] += delta2
                            s2 = _objective(trial2)
                            if s2 < cur_score - 1e-14:
                                cur = trial2
                                cur_score = s2
                                if s2 < best_score - 1e-14:
                                    best_score = s2
                                    best_vec = cur.copy()
                                    print(f"[{time.time():.2f}] small‑greedy it={it} improved to {best_score:.6f}")

        # ----- random jitter to escape plateaus -----
        if it % 1500 == 0:
            i = rng.integers(n)
            if cur[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps_move = min(cur[i] * 0.04, 0.001)
                jitter = cur.copy()
                jitter[i] -= eps_move
                jitter[j] += eps_move
                jitter_score = _objective(jitter)
                if jitter_score < cur_score - 1e-14:
                    cur = jitter
                    cur_score = jitter_score
                    if jitter_score < best_score - 1e-14:
                        best_score = jitter_score
                        best_vec = cur.copy()
                        print(f"[{time.time():.2f}] jitter it={it} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final sanity – project back onto simplex and compute final score
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f}  (length={len(best_vec)})")
    return best_vec.tolist()
