    t = 0

def _guided_swap_local(a, deadline, best_seq, best_score):

        max_val = conv.max()

        if k == 0:

            i_source = int(np.argmax(a))

            products = a[:n-k] * a[k:]
            i_source = int(np.argmax(products))
        # source index for mass transfer: choose among i_source and i_source+k

def _distribution_beta_general(n,p,q):
    x = (np.arange(n)+0.5)/n
    w = np.power(x, p-1) * np.power(1-x, q-1)
    w[np.isnan(w) | np.isinf(w)] = 0.0
    total = w.sum()
    if total == 0:
        w = np.full(n, 1.0/n)
    else:
        w /= total
    return w

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta(p,p) density on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_general(n: int, p: float, q: float) -> np.ndarray:
    """General Beta(p,q) density on a discrete grid (p,q>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    # protect against NaNs / infinities (possible for extreme parameters)
    w = np.where(np.isfinite(w), w, 0.0)
    total = w.sum()
    if total == 0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= total
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad set of start vectors for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_sine(n))

    # symmetric beta
    for p in (0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # general beta grid
    ps = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    qs = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    for p in ps:
        for q in qs:
            cand.append(_distribution_beta_general(n, p, q))

    # exponential
    for lam in (0.001, 0.002, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # power‑law at edges
    for gamma in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4):
        cand.append(_distribution_powerlaw_edges(n, gamma))

    # alternating
    for stride in (2, 3, 4):
        cand.append(_distribution_alternating(n, stride))

    # two block spikes
    for block_frac in (0.02, 0.05, 0.08):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random sparse
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # random dense (will be normalised)
    dense = rng.random(n)
    cand.append(dense)

    return cand

# Smoothed Adam descent (log‑sum‑exp approximation)
def _smooth_adam(a: np.ndarray, deadline: float,
                 best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth max‑convolution surrogate."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    schedule = [
        (0.5, 2000, 0.20),
        (1.0, 2000, 0.15),
        (5.0, 2000, 0.12),
        (20.0, 2000, 0.10),
        (100.0, 1500, 0.08),
        (500.0, 1000, 0.06),
        (2000.0, 800, 0.04),
        (5000.0, 600, 0.03),
        (20000.0, 400, 0.02),
        (50000.0, 200, 0.015),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # smooth‑max weights (log‑sum‑exp)
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of smooth‑max w.r.t. a
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact check
            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution
def _subgradient_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Pure sub‑gradient descent on  max(conv(a,a))  (projected onto simplex)."""
    n = a.shape[0]
    t = 0
    init_lr = 0.25  # relatively aggressive at start

    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        k = int(conv.argmax())

        # gradient of the peak convolution term
        g = np.zeros_like(a)
        if k < n:
            g[: n - k] += a[k:]
            g[k:] += a[: n - k]

        # diminishing step size
        lr = init_lr / math.sqrt(t)
        a = a - lr * g
        a = _project_onto_simplex(a, 1.0)

        if t % 200 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] sub‑grad iter {t} new best {best_score:.6f}")

    return best_seq, best_score

# Guided pairwise swap (targets the shift where the peak sits)
def _guided_swap_local(a: np.ndarray, deadline: float,
                       best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Mass transfers that focus on the most offending shift."""
    n = a.shape[0]
    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        # identify source index contributing the most to this peak
        if k == 0:
            src = int(np.argmax(a))
        else:
            prod = a[: n - k] * a[k:]
            src_local = int(np.argmax(prod))
            # pick the larger of the two partners
            if a[src_local] >= a[src_local + k]:
                src = src_local
            else:
                src = src_local + k

        # pick a destination distinct from src
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # move a modest fraction of the source mass
        delta = a[src] * np.random.rand() * 0.25
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] guided swap src={src}→dst={dst} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Cheap random pairwise swaps (fallback fine‑tuning)
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Random mass transfers accepted only if they strictly improve the true objective."""
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] random swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with minimal `evaluate_sequence` value.
    Returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # 1 s safety margin

    # ------------------------------------------------------------------
    # 1.  Pick several lengths – the objective is homogeneous, but a finer
    #     discretisation can reduce the discrete artefacts.
    lengths = [8000, 16000, 32000, 64000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2.  Scan analytic families for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating candidates for n={n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s <= 0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best n={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # retain only the very best few seeds so far (memory / time bound)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 3.  Include the previous construction if it exists.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0:
                prev = prev / prev.sum()
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous sequence improved score={sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Random mixtures of the best candidates.
    if len(top_candidates) >= 2:
        top_candidates.sort(key=lambda x: x[0])
        seeds = [seq for _, seq in top_candidates[:8]]
        for i in range(len(seeds)):
            for j in range(i, len(seeds)):
                if time.time() >= deadline:
                    break
                for _ in range(3):
                    lam = rng.random()
                    mix = lam * seeds[i] + (1.0 - lam) * seeds[j]
                    mix = mix / mix.sum()
                    sc = _objective(mix)
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = mix.copy()
                        print(f"[{time.time():.1f}] mix i={i} j={j} λ={lam:.3f} new best {sc:.6f}")
                    top_candidates.append((sc, mix.copy()))
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:20]

    # ------------------------------------------------------------------
    # 5.  Smoothed Adam descent on a few promising seeds.
    remaining_time = deadline - time.time()
    if remaining_time > 30 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]]
        for idx, seed_vec in enumerate(seeds):
            if time.time() >= deadline:
                break
            # allocate a slice of the remaining budget to each seed
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds) - idx)
            overall_best_seq, overall_best_score = _smooth_adam(
                seed_vec.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            # keep the best found so far
            if overall_best_seq is None:
                overall_best_seq = seed_vec.copy()
            overall_best_seq = overall_best_seq.copy()

    # ------------------------------------------------------------------
    # 6.  Sub‑gradient descent on the true max‑convolution.
    if time.time() < deadline:
        sub_deadline = deadline - 5.0
        overall_best_seq, overall_best_score = _subgradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 7.  Guided pairwise swaps targeting the peak shift.
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _guided_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 8.  Final cheap random swaps (fallback fine‑tuning).
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
