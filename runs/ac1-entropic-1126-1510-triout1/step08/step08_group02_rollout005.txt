def _smooth_obj_and_grad_for_softmax(x, eps):
    a = np.exp(x - np.max(x))
    a = a / a.sum()
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    # compute w = exp((conv - maxc)/eps)/sumexp
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w = w / w_sum
    # gradient w.r.t a:
    conv_rev = _fft_convolve(a[::-1], w)
    grad_a = 2.0 * conv_rev[n-1:n-1+n]  # n length
    # diagonal correction:
    even = (np.arange(conv.size) % 2 == 0)
    diag_idx = np.flatnonzero(even) // 2
    grad_a[diag_idx] -= w[2 * diag_idx] * a[diag_idx]
    # gradient w.r.t x:
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    # objective:
    val = eps * (np.log(w_sum) + maxc / eps)  # Since log sum exp = maxc/eps + log sum exp((conv-maxc)/eps)
    # Equivalent: val = eps * (maxc/eps + log(w_sum)) = maxc + eps*log(w_sum)
    # But compute as such for numeric stability.
    # Actually we can compute directly: val = maxc + eps*np.log(w_sum)
    # We'll use that.
    val = maxc + eps * np.log(w_sum)
    return val, grad_x

"""
Propose a step‑function (non‑negative vector on the simplex) that minimises
``evaluate_sequence``.  

The algorithm proceeds in three stages:

1. **Broad sampling** – generate many analytic families (arcsine, beta,
   triangular, exponential, Gaussian, Cauchy, semicircle, sine,
   cosine, power‑law, asymmetric beta) together with random Dirichlet draws
   for several lengths.  All candidates are normalised to sum = 1 and
   scored with the cheap objective `2·n·max(a*a)`.

2. **Heavy refinement** – for each of the best‑scoring seeds we minimise a
   smoothed version of the max‑convolution using a log‑sum‑exp surrogate.
   The surrogate is optimised by SciPy's L‑BFGS‑B on the *softmax*
   parametrisation (i.e. a = softmax(x)).  A schedule of decreasing
   smoothing parameters `eps` is used, each stage performing a short L‑BFGS
   run (≈30 iterations).  If SciPy is unavailable we fall back to the
   existing Adam/exp‑gradient refinements.

3. **Final local search** – a long projected sub‑gradient descent followed
   by a tiny mass‑transfer hill‑climbing loop further reduces the
   max‑convolution.  The best sequence found is returned as a plain Python
   list.

All operations respect the supplied time budget (default 1000 s) and
print progress information.
"""

import time
import numpy as np

# optional SciPy for L‑BFGS‑B

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of the smoothed max using log‑sum‑exp."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    maxc = conv.max()
    w_raw = np.exp((conv - maxc) / eps)
    w_sum = w_raw.sum()
    w = w_raw / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    even = (np.arange(conv.size) % 2 == 0)
    diag_idx = np.flatnonzero(even) // 2
    grad[diag_idx] -= w[2 * diag_idx] * a[diag_idx]

    return grad


def _smooth_obj_and_grad(a: np.ndarray, eps: float):
    """Return (value, grad_a, maxc, w_sum) for the smooth surrogate."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w_raw = np.exp((conv - maxc) / eps)
    w_sum = w_raw.sum()
    w = w_raw / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    even = (np.arange(conv.size) % 2 == 0)
    diag_idx = np.flatnonzero(even) // 2
    grad_a[diag_idx] -= w[2 * diag_idx] * a[diag_idx]

    # smoothed max value
    val = maxc + eps * np.log(w_sum)
    return val, grad_a, maxc, w_sum

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+0.5)^p .  p may be negative, zero or positive."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    if p == 0.0:
        vals = np.ones_like(idx)
    else:
        vals = idx ** p
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()

def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

# Optimisation helpers ----------------------------------------------------
def _lbfgs_optimize(seed_a: np.ndarray, deadline: float,
                    eps_schedule=None, max_iter_per_eps: int = 30):
    """
    Refine ``seed_a`` by minimising the log‑sum‑exp smoothed max‑convolution
    with L‑BFGS‑B on the softmax parametrisation.
    Returns the best vector and its cheap score.
    """
    if eps_schedule is None:
        eps_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]

    # initialise logits (softmax)
    x = np.log(seed_a + 1e-12)

    best_a = seed_a.copy()
    best_score = _max_convolution_score(best_a)

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        # closure captures current eps
        def fun(logits):
            a = np.exp(logits - np.max(logits))
            a = a / a.sum()
            val, grad_a, _, _ = _smooth_obj_and_grad(a, eps)
            # chain rule: ∂val/∂logits = a * (grad_a - ⟨a,grad_a⟩)
            grad = a * (grad_a - np.dot(a, grad_a))
            return val, grad

        opts = {
            'maxiter': max_iter_per_eps,
            'ftol': 1e-12,
        }
        res = minimize(fun, x, method='L-BFGS-B', jac=True,
                       options=opts)
        x = res.x  # start next eps from here

        a = np.exp(x - np.max(x))
        a = a / a.sum()
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.1f}] L‑BFGS eps={eps:.1e} → {cur_score:.6f}")

    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimiser on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 0.6):
    """Plain projected sub‑gradient descent with a diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a probability vector (sum = 1).  The routine respects the supplied
    time budget and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Base length (fallback to 3500)
    try:
        base_n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # Lengths to explore: deterministic list + a few random ones
    lengths = {base_n, max(500, int(base_n * 0.8)), int(base_n * 0.9),
               int(base_n * 1.0), int(base_n * 1.1), int(base_n * 1.2)}
    for _ in range(5):
        lo = 500
        hi = min(15000, base_n * 3)
        lengths.add(np.random.randint(lo, hi + 1))
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1. Broad sampling
    candidates = []   # (array, cheap_score)

    beta_ps = [0.01, 0.02, 0.04, 0.06, 0.08, 0.12, 0.18, 0.25,
               0.35, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0]

    exp_lams = [0.001, 0.003, 0.006, 0.01]
    sigma_fracs = [0.16, 0.25, 0.33, 0.45]
    scale_fracs = [0.12, 0.20, 0.30, 0.45]
    dirichlet_alphas = [0.3, 0.6, 1.0]
    power_ps = [-2.0, -1.5, -1.0, -0.5, 0.0, 0.2, 0.5, 1.0, 1.5, 2.0]
    beta_asym_params = [(0.5, 2.0), (0.5, 1.0), (1.0, 2.0),
                        (2.0, 0.5), (3.0, 1.0), (1.0, 3.0)]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Analytic families -------------------------------------------------
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_uniform_distribution, "uniform"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_semicircle_distribution, "semicircle"),
        ]
        for gen, name in fams:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} {name:10s} score={score:.6f}")

        # Exponential families
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} exponential(lam={lam:.3f}) score={score:.6f}")

        # Gaussian families
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} gaussian(sig={sf:.2f}) score={score:.6f}")

        # Cauchy families
        for sf in scale_fracs:
            a = _cauchy_distribution(n, scale_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} cauchy(scale={sf:.2f}) score={score:.6f}")

        # Symmetric beta
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} beta(p={p:4.2f}) score={score:.6f}")

        # Power‑law families
        for p in power_ps:
            a = _power_law_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} power(p={p:4.1f}) score={score:.6f}")

        # Asymmetric beta families
        for alpha, beta in beta_asym_params:
            a = _beta_asymmetric_distribution(n, alpha, beta)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} beta(α={alpha:.1f},β={beta:.1f}) score={score:.6f}")

        # Random Dirichlet draws
        for alpha in dirichlet_alphas:
            for _ in range(2):
                a = _random_dirichlet(n, alpha=alpha)
                score = _max_convolution_score(a)
                candidates.append((a, score))
                print(f"[{time.time():.1f}] n={n:5d} Dirichlet(α={alpha:.2f}) score={score:.6f}")

        # Random convex combos of two families (to increase diversity)
        if time.time() < deadline:
            # pick two distinct families at random
            fam_indices = np.random.choice(len(fams), size=2, replace=False)
            a1 = fams[fam_indices[0]][0](n)
            a2 = fams[fam_indices[1]][0](n)
            w = np.random.rand()
            a = w * a1 + (1.0 - w) * a2
            a = a / a.sum()
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} combo score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Include the previously best sequence, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 30
    top_seeds = candidates[:top_k]
    print(f"Selected {len(top_seeds)} seeds for refinement; best cheap score={top_seeds[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy refinement (L‑BFGS or fallbacks)
    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]

    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # L‑BFGS smoothing (if SciPy available)
        if _SCIPY_AVAILABLE:
            seq, sc = _lbfgs_optimize(seed_seq, deadline,
                                      eps_schedule=[0.05, 0.02, 0.01, 0.005,
                                                    0.002, 0.001, 5e-4, 2e-4, 1e-4],
                                      max_iter_per_eps=30)
            if sc < best_score - 1e-14:
                best_score, best_seq = sc, seq.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} L‑BFGS → {sc:.6f}")

        # Exponentiated‑gradient (quick sanity)
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=60_000, eta0=0.9)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} exp‑grad → {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                max_iters=60_000, step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Long projected sub‑gradient descent on the current best
    subgrad_deadline = deadline - 30.0   # reserve ~30 s for hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=1_000_000, eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained hill‑climbing (mass transfer moves)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02  # up to 2 % of total mass initially

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 12000.0))

        # Choose source & destination indices
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional random jitter to escape tiny plateaus
        if iteration % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
