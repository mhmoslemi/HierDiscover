def propose_candidate(seed: int=42, budget_s: int=1000):
    import time, random, numpy as np, math

    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # margin

    best_seq=None
    best_score=inf
    best_n=0

    # Candidate generation from analytic families and param sweeps
    lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500, 5000]
    for n in lengths:
        if time.time() >= deadline: break

        # Always include uniform base
        cand_list = [_distribution_uniform(n)]

        # arcsine
        cand_list.append(_distribution_arcsine(n))

        # triangular
        cand_list.append(_distribution_triangular(n))

        # sine power
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            cand_list.append(_distribution_sine_power(n, pw))

        # Gaussian sigma
        for sigma in (0.15,0.25,0.35):
            cand_list.append(_distribution_gaussian(n, sigma))

        # exponential
        for lam in (0.001,0.003,0.008,0.02):
            cand_list.append(_distribution_exponential(n, lam))

        # Beta symmetric p sweep
        for p in np.geomspace(0.05, 1.0, num=15):
            cand_list.append(_distribution_beta_sym(n, p))

        # power law distribution
        for alpha in np.geomspace(0.2, 5.0, num=10):
            vals = 1.0 / ((np.arange(n)+0.5)*(n - (np.arange(n)+0.5)))**alpha
            vals = np.maximum(vals, 0)
            vals /= vals.sum()
            cand_list.append(vals)

        # random Dirichlet draws
        for alpha in (0.2,0.5,1.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            cand_list.append(vec)

        # Evaluate each candidate cheap
        for vec in cand_list:
            if time.time() >= deadline: break
            vec = np.maximum(vec, 1e-12)
            vec = vec / vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # incorporate previous sequence if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size>0:
            prev = np.maximum(prev, 1e-12)

            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # if still none choose uniform at n=4000
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq/=best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    # Now refine with Adam mirror descent
    a = best_seq.copy()
    n = a.shape[0]

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    it = 0
    # schedule for alpha (smoothness)
    # start alpha small then grow to large.
    alpha_start = 0.5
    alpha_end = 20000.0

    # learning rate schedule: start at lr0 then decay
    lr_start = 0.1
    lr_end = 0.001

    while time.time() < deadline:
        it += 1
        # compute conv
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # temperature schedule (alpha)
        t_frac = it / (1e5)  # approximate max iteration
        if t_frac > 1:
            t_frac = 1.0
        alpha = alpha_start * (alpha_end / alpha_start) ** t_frac
        # compute weights
        # subtract maxc for stability
        w = np.exp(alpha * (conv - maxc))
        sumw = w.sum()

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n-1:n-1+n]

        max_prod = a[best_i] * a[best_j]
        # simple loop to check best pair among those
        # We can skip for speed; just use best_i,best_j as initial
        # Could randomly choose source from top few indices
        if random.random() < 0.5:
            source = best_i
        else:
            source = best_j
        # choose target random
        target = random.randint(0, n-1)
        if target == source:
            continue
        delta = min(a[source] * 0.5, 0.02) * random.random()
        if delta <= 1e-12:
            continue
        # apply transfer in O(n)
        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
        else:
            a = a_snap
            conv = conv_snap

        if it_hc % 2000 == 0:

max‑self‑convolution value.

Algorithm
---------

1. **Broad analytic search**

   Starting from the best candidate we minimise the smooth approximation
   of the maximum convolution:

        w_k ∝ exp(α·(conv_k – max(conv))).

   The gradient of the smooth maximum is `2 * (a ⊛ w)`.

   projected back onto the probability simplex.  The temperature `α`

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-16
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, alpha):
    """w[i] ∝ ((i+0.5)*(n-i-0.5))^{-α}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = 1.0 / ((i * (n - i)) ** alpha)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # Update convolution entries that involve i
    conv[i:i + n] -= 2.0 * delta * a_old
    # Update entries that involve j
    conv[j:j + n] += 2.0 * delta * a_old
    # Update the diagonal terms (i+i) and (j+j)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    # Transfer the mass
    a[i] -= delta
    a[j] += delta
