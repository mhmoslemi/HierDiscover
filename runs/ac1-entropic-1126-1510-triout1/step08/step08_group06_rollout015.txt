def _smooth_adam_custom(a, deadline, best_seq, best_score):
    n = a.shape[0]
    # parameters
    beta1,beta2,eps = 0.9,0.999,1e-8
    m=np.zeros_like(a)
    v=np.zeros_like(a)
    t=0

    schedule = [
        (0.5, 2000, 0.15),
        (1.0, 2000, 0.12),
        (5.0, 2000, 0.10),
        (20.0, 2000, 0.08),
        (100.0, 2000, 0.06),
        (500.0, 2000, 0.04),
        (2000.0, 2000, 0.03),
        (5000.0, 2000, 0.02),
        (20000.0, 1000, 0.015),
        (50000.0, 600, 0.01),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t+=1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # compute softmax weights
            # to avoid overflow for large alpha compute with subtract max.
            if alpha > 0:
                w_exp = np.exp(alpha * (b - maxb))
                w_sum = w_exp.sum()
                if w_sum == 0.0:
                    w = np.full_like(b, 1.0 / b.size)
                else:
                    w = w_exp / w_sum
            else:
                w = np.full_like(b, 1.0 / b.size)  # for alpha=0
            # gradient as convolution of a_rev with w
            grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # projection onto simplex
            a = _project_onto_simplex(a, 1.0)
            # periodic evaluation
            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")
    return best_seq, best_score

def _pairwise_swap_local(a, deadline, best_seq, best_score):
    n = a.shape[0]
    while time.time() < deadline:
        # choose source with some mass
        src = np.random.choice(np.where(a > 1e-12)[0])
        # dest random distinct
        dest = np.random.randint(n-1)
        if dest >= src:
            dest += 1
        # delta up to a[src] (maybe 50% for more effective)
        delta = a[src] * np.random.rand() * 0.5
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dest] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {src}->{dest} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

"""
Improved search for a non‑negative step function with a low
evaluate_sequence() score.

Main ideas
----------
* Start from the analytically good arcsine shape (β(½,½)).
* Refine the shape with a long Adam optimisation that minimises a
  smoothed version of the max‑convolution (log‑sum‑exp with a schedule
  of increasing smoothness parameters α).
* Gradually increase the discretisation length by interpolating the
  current best vector (2000 → 4000 → 8000 → 16000 → 32000).  After each
  up‑sampling the Adam optimiser runs again – the shape is kept but
  allowed to adapt to the finer grid.
* Finish with a cheap Metropolis‑type local search that moves a random
  amount of mass between two positions only when the true (unsmoothed)
  objective improves.
* All operations respect the simplex constraint (non‑negative, sum = 1)
  via Euclidean projection onto the simplex.
* The whole procedure is time‑bounded (default 1000 s).  The best
  sequence found so far is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.02) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a (which lives on a centred grid) to length new_n."""
    old_n = a.shape[0]
    x_old = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s <= 0:
        a_new.fill(1.0 / new_n)
    else:
        a_new /= s
    return a_new


# ----------------------------------------------------------------------
def _smooth_adam(a: np.ndarray, deadline: float,
                 best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth approximation of the max‑convolution.
    A fairly long schedule of increasing smoothness (α) and decreasing learning‑rate
    is used.  The best vector seen so far (across all calls) is kept.
    """
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # (α, steps, learning‑rate) – extended compared to the reference code
    schedule = [
        (0.5, 2000, 0.15),
        (1.0, 2000, 0.12),
        (5.0, 2000, 0.10),
        (20.0, 2000, 0.08),
        (100.0, 2000, 0.06),
        (500.0, 2000, 0.04),
        (2000.0, 2000, 0.03),
        (5000.0, 2000, 0.02),
        (20000.0, 1000, 0.015),
        (50000.0, 600, 0.01),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # --- smooth‑max weights (log‑sum‑exp) ---
            w_exp = np.exp(alpha * (b - maxb))
            w_sum = w_exp.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w = w_exp / w_sum

            # gradient of the smooth max (a↔w cross‑correlation)
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

            # Adam update + simplex projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis‑type local walk that moves a random amount of mass."""
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # move up to 50 % of the source mass
        delta = a[i] * np.random.rand() * 0.5
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector found as a plain Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # a small safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  Start from the arcsine distribution at a modest size.
    n = 2000
    a = _distribution_arcsine(n)

    # 2.  Optimise at this resolution.
    a, best_score = _smooth_adam(a, deadline, a, best_score)
    best_seq = a.copy()
    print(f"[{time.time():.1f}] start length={n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Gradually upscale, re‑optimising after each interpolation.
    max_n = 32000
    while n < max_n and time.time() < deadline - 5.0:
        new_n = min(max_n, n * 2)
        a = _interpolate(a, new_n)

        # add a tiny random jitter to avoid getting stuck on the same exact point
        a += 0.005 * rng.random(new_n)
        a = np.maximum(a, 0.0)
        a /= a.sum()

        a, best_score = _smooth_adam(a, deadline, best_seq, best_score)
        if best_score < _objective(best_seq):
            best_seq = a.copy()
        n = new_n
        print(f"[{time.time():.1f}] upscaled to {n}, best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  A few extra random starts at the final resolution – they sometimes
    #    discover a better basin of attraction.
    if time.time() < deadline:
        for _ in range(3):
            rand_a = _distribution_random_sparse(n, rng, density=0.02)
            rand_a, sc = _smooth_adam(rand_a, deadline, best_seq, best_score)
            if sc < best_score:
                best_score = sc
                best_seq = rand_a.copy()
                print(f"[{time.time():.1f}] random start improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final cheap Metropolis‑type fine‑tuning.
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(best_seq.copy(),
                                                    deadline,
                                                    best_seq,
                                                    best_score)

    # ------------------------------------------------------------------
    # 6.  Fallback – should never be needed.
    if best_seq is None:
        best_seq = _distribution_uniform(2000)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
