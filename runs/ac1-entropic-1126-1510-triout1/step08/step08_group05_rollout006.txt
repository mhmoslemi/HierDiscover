"""
Propose a step‑function (non‑negative vector) that minimises the black‑box
``evaluate_sequence``.  The optimisation consists of several stages:

1. **Broad initialisation** – a pool of analytic windows (uniform,
   arcsine, triangular, exponential, beta, sine‑power, cosine) together with a
   few random Dirichlet draws and the optional previous best (`height_sequence_1`)
   are normalised to a sum of 1.  The best of these cheap candidates seeds the
   subsequent optimisation.

2. **Adam descent on a softened max** – a smooth log‑sum‑exp (controlled by a
   temperature‑like ``alpha``) approximates the peak of the self‑convolution.
   Adam with momentum descends on this surrogate and projects onto the
   probability simplex after each step.

3. **Exact max‑sub‑gradient descent** – the true objective ``max(a*a)`` is
   tackled by sub‑gradient updates (with momentum) while staying on the simplex.

4. **Optional diffusion smoothing** – a few cheap convolution‑with‑a‑small‑kernel
   passes that occasionally lower the peak.

5. **Focused simulated‑annealing (SA) hill‑climber** – works directly on the
   full self‑convolution using an O(n) incremental update.  The move generator
   is *aware* of the current maximal convolution index: the source index is
   sampled proportionally to its contribution to the peak, while the target is
   chosen uniformly.  Temperature, step‑size and a periodic jitter are
   adapted, and the global best is never overwritten by a worse intermediate
   solution (a bug in the reference implementation).

The function respects the given time budget (default 1000 s) and returns the
best sequence found (a plain ``list[float]``).  All helper functions are
defined at module level and no closures are used.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """
    Fast evaluation of the objective for a vector that is already normalised
    to sum = 1.  Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff – coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part of the update (the effect of moving delta on the convolution)
    conv[i:i + a.shape[0]] -= 2.0 * delta * a_old
    conv[j:j + a.shape[0]] += 2.0 * delta * a_old
    # quadratic self‑terms (a_i^2 and a_j^2)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

            prev = _normalise(prev)
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # Fallback – should never happen.
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam descent on a smooth approximation of the max
    a = best_seq.copy()
    n = a.shape[0]

    schedule = [
        (1.0,   1500),
        (5.0,   1500),
        (20.0,  3000),
        (100.0, 3000),
        (500.0, 3000),
        (2000.0,2000),
        (5000.0,1000)
    ]

    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_seq_adam = a.copy()
    best_score_adam = _max_convolution_score(a)

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1

            # smooth approximation of max via soft‑max
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth surrogate
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # projection onto the simplex
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if t % 250 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score_adam - 1e-12:
                    best_score_adam = cur
                    best_seq_adam = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

    # adopt Adam result
    a = best_seq_adam.copy()
    best_seq = a.copy()
    best_score = best_score_adam

    # ------------------------------------------------------------------
    # 4. Exact max‑sub‑gradient refinement (with momentum)
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cur:.6f}")

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 5. Optional diffusion smoothing (few short passes)
    if time.time() < deadline:
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        a = best_seq.copy()
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, eps)
            a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Diffusion new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 6. Focused simulated‑annealing hill‑climber (no reset of best_score)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    cur_M = conv.max()
    cur_score = 2.0 * n * cur_M
    # keep the global best found so far
    if cur_score < best_score - 1e-12:
        best_score = cur_score
        best_seq = a.copy()
        print(f"[{time.time():.1f}] SA start improves best to {cur_score:.6f}")

    it = 0
    it_no_improve = 0
    T = cur_score * 0.05               # initial temperature ≈5 % of current score
    decay = 0.9995                     # exponential cooling
    max_delta_factor = 0.02            # up to 2 % of source mass per move

    while time.time() < deadline:
        it += 1

        # --- identify the current peak and contributions ---
        k_max = int(np.argmax(conv))
        start_i = max(0, k_max - (n - 1))
        end_i = min(k_max, n - 1)
        contrib = np.zeros(n, dtype=np.float64)
        # contributions only from indices that actually appear in the peak
        idx_range = np.arange(start_i, end_i + 1, dtype=np.int64)
        contrib[idx_range] = a[idx_range] * a[k_max - idx_range]

        # source selection: 70 % contribution‑biased, else mass‑biased
        if contrib.sum() > 0 and random.random() < 0.7:
            src_probs = contrib / contrib.sum()
        else:
            src_probs = a / a.sum()
        source = np.random.choice(n, p=src_probs)

        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * max_delta_factor, 0.05)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        # --- try the move (incremental update) ---
        a_snap = a.copy()
        conv_snap = conv.copy()

        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        new_score = 2.0 * n * new_M

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-16))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            cur_M = new_M
            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        T *= decay

        # periodic jitter to escape stagnation
        if it % 5000 == 0 and it_no_improve > 2000:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            cur_M = conv.max()
            cur_score = 2.0 * n * cur_M
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

        # Re‑compute the full convolution occasionally to avoid drift
        if it % 10000 == 0:
            conv = _fft_self_convolve(a)

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
