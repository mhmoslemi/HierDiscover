def solve_lp_for_g(f, M):
    n = len(f)
    m = 2*n - 1
    # Build A
    diags_data = [np.full(n, f[d]) for d in range(n)]
    offsets = list(range(n))
    A = scipy.sparse.diags(diags_data, offsets, shape=(m, n), format='csr')
    # Bounds for g
    bounds = [(0, None)]*n
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    # b_ub = M * np.ones(m)
    b_ub = np.full(m, M)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if not res.success:
        # fallback - maybe raise error
        return None
    g = res.x
    return g
```

def _solve_lp_max_sum(f, M):

    return res.x

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise a feasible probability vector (the previous best if it
   exists, otherwise the arcsine distribution).
2. Apply an alternating LP‑based refinement:
   * Let M = max(f ∗ f).
   * Solve the linear programme
         maximise   Σ g_j
         subject to  f ∗ g ≤ M ,   g ≥ 0 .
     The optimum g tries to pack as much mass as possible while keeping
     the convolution with the current f bounded by M.
   * Normalise g to a probability vector and mix it with f.
     A short line‑search over a few mixing coefficients is performed and
     the mixture with the smallest max‑convolution is kept.
   * Repeat – the bound M is monotone decreasing and the distribution
     moves towards one whose self‑convolution is (almost) flat.
3. With the remaining time polish the result by a projected sub‑gradient
   descent on the exact max‑convolution objective and a tiny random‑mass
   hill‑climber.
The whole routine respects a 1000 s wall‑clock budget and always returns a
list of non‑negative numbers (the format required by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Try to import SciPy – it is needed for the LP step.  If unavailable we
# fall back to a pure‑gradient approach (the previous code) which still
# returns a feasible sequence.
try:
    import scipy.optimize
    import scipy.sparse
    _SCIPY_AVAILABLE = True
except Exception:         # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families – only the arcsine law is needed as a default seed.
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


# ----------------------------------------------------------------------
# Sub‑gradient utilities (used for the final polishing stage).
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# ----- Alternating LP improvement (requires SciPy) --------------------
def _build_conv_matrix(f: np.ndarray):
    """Return the (2n‑1)×n matrix A with rows A[i]·g = (f∗g)[i]."""
    n = f.shape[0]
    m = 2 * n - 1
    # each diagonal d (0 ≤ d < n) contains the constant value f[d]
    diags = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0,1,…,n‑1
    return scipy.sparse.diags(diags, offsets, shape=(m, n), format='csr')


def _solve_lp_max_sum(f: np.ndarray, M: float):
    """Solve  max Σ g_j   s.t.  f∗g ≤ M , g ≥ 0 .  Return the primal solution."""
    n = f.shape[0]
    A = _build_conv_matrix(f)                 # (2n‑1) × n
    m = 2 * n - 1
    b_ub = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)         # minimise –Σg
    bounds = [(0.0, None)] * n

    # HiGHS is fast for large sparse LPs.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _alternating_optim(f0: np.ndarray, deadline: float,
                       max_iters: int = 12):
    """Alternating LP+mixing scheme that monotonically reduces max(f*f)."""
    f = f0.copy()
    n = f.shape[0]
    conv = _fft_self_convolve(f)
    best_f = f.copy()
    best_score = 2.0 * n * conv.max()
    t_grid = [0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0]   # mixing candidates

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = conv.max()
        g = _solve_lp_max_sum(f, M)
        if g is None:
            print("[alternating] LP failed – terminating.")
            break
        sum_g = g.sum()
        if sum_g <= 1e-12:
            print("[alternating] LP returned zero mass – terminating.")
            break
        g_norm = g / sum_g                     # now Σg = 1

        # try several mixing coefficients, keep the best
        best_local = None
        best_local_score = float('inf')
        for t in t_grid:
            mix = (1.0 - t) * f + t * g_norm
            # (Σmix = 1 automatically)
            s = _max_convolution_score(mix)
            if s < best_local_score:
                best_local_score = s
                best_local = mix
        # accept the best mixture
        f = best_local
        conv = _fft_self_convolve(f)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            print(f"[alternating] iter {it+1} improved score {best_score:.6f}")

    return best_f, best_score
