"""
Greedy‑enhanced search for a non‑negative step function that minimises
``evaluate_sequence``.  The method builds on the previous pipeline
(analytic families → Adam → sub‑gradient) and adds a deterministic
“max‑reduction’’ phase that repeatedly transfers mass from indices that
contribute most to the current maximal convolution value to indices that
contribute least.  The transfer is performed with an O(n) incremental
convolution update, guaranteeing fast convergence.  After the greedy
phase a short simulated‑annealing refinement is executed.  All phases
honour a user‑supplied time budget (default 1000 s) and keep the best
sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


def _distribution_logistic(n, k=0.08):
    """Smooth bell‑shaped logistic distribution centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + np.exp(-k * (i - centre)))
    vals = vals * (1.0 - vals)   # make it symmetric around centre
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()                     # linear part uses the *old* a
    # linear part of convolution change
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _greedy_max_reduction(a, conv, deadline):
    """
    Deterministic greedy phase that repeatedly moves a small amount of mass
    from indices that contribute most to the current maximal convolution value
    to indices that contribute least.  The update is performed with the
    O(n) incremental convolution routine.
    """
    n = a.shape[0]
    eps = 1e-12
    while time.time() < deadline:
        M = conv.max()
        # indices that are essentially maximal
        tol = max(eps, 1e-9 * M)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            break
        w = np.zeros_like(conv)
        w[active] = 1.0
        # gradient of the active max w.r.t. a (up to factor 2)
        grad_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * grad_rev[n - 1:n - 1 + n]

        # candidate source / target indices (extremes of the gradient)
        src_candidates = np.argsort(-grad)[:5]   # largest contribution
        tgt_candidates = np.argsort(grad)[:5]    # smallest contribution

        improved = False
        for src in src_candidates:
            if a[src] < eps:
                continue
            # we will try to move at most 20 % of the source mass,
            # but never more than 0.05 (absolute) to keep steps safe.
            max_delta = min(a[src] * 0.2, 0.05)
            if max_delta < eps:
                continue
            for tgt in tgt_candidates:
                if src == tgt:
                    continue
                delta = max_delta
                # backup the current state
                a_snap = a.copy()
                conv_snap = conv.copy()
                _apply_transfer(a, conv, src, tgt, delta)
                newM = conv.max()
                if newM <= M + 1e-14:
                    improved = True
                    break
                # revert – the move was not beneficial
                a[:] = a_snap
                conv[:] = conv_snap
            if improved:
                break
        if not improved:
            # No improving elementary transfer found – exit.
            break
    return a, conv

    ``evaluate_sequence``.  The routine mixes an analytic‑family scan,

    max‑reduction phase and a final simulated‑annealing hill‑climber.
    The best sequence encountered within the time budget is returned.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Scan a diverse pool of analytic families (different lengths).
    lengths = [2000, 3000, 4000, 5000, 8000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break

        cand = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.003),
            _distribution_exponential(n, lam=0.02),
            _distribution_logistic(n, k=0.08),

        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            cand.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            cand.append(_distribution_sine_power(n, pw))

        for c in (0.5, 0.46, 0.42):
            cand.append(_distribution_cosine_window(n, c))

        for p in (0.5, 1.0, 1.5, 2.0):
            cand.append(_distribution_power_law(n, p))

        # Sparse random seed (few spikes)
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        cand.append(sparse)

        # A handful of Dirichlet / Gamma draws
        for _ in range(5):
            x = np.random.gamma(0.5, 1.0, size=n)
            x /= x.sum()
            cand.append(x)

        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} new init best {sc:.6f}")

    # 2. Include the previous global solution if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # Fallback – should never happen.
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to keep the global best up‑to‑date
    def _maybe_update(seq):
        nonlocal best_seq, best_score, best_n
        sc = _max_convolution_score(seq)
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = seq.copy()
            best_n = len(seq)
            print(f"[{time.time():.1f}] new global best {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam descent with a progressively sharpening soft‑max.
    a = best_seq.copy()
    n = a.shape[0]

    schedule = [
        (1.0,   1500),
        (5.0,   1500),
        (20.0,  3000),
        (100.0, 3000),
        (500.0, 3000),
        (2000.0,2000),
        (5000.0,1500),
        (15000.0,1000)
    ]

    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1

"""
Greedy‑enhanced search for a non‑negative step function that minimises
``evaluate_sequence``.

The pipeline combines the previous approach (analytic families → Adam →
sub‑gradient) with two new ingredients:

1. **Deterministic greedy max‑reduction** – mass is repeatedly transferred
   from indices that contribute most to the current maximal
   self‑convolution value to indices that contribute least.  The transfer
   updates the full convolution in O(n) time, so many cheap improving
   moves can be performed.

2. **Optional up‑sampling** – after the first optimisation stage the best
   distribution is interpolated to a larger length (up to ≈20 000).  A short
   greedy refinement on the longer vector often yields a lower peak
   because the support becomes smoother.

All phases respect a user‑supplied time budget (default 1000 s) and the
best sequence encountered is returned.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


def _distribution_logistic(n, k=0.08):
    """Smooth bell‑shaped logistic distribution centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + np.exp(-k * (i - centre)))
    vals = vals * (1.0 - vals)   # make it symmetric
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

def _greedy_max_reduction(a, conv, deadline):
    """
    Deterministic greedy phase that repeatedly moves a small amount of mass
    from indices that contribute most to the current maximal convolution value
    to indices that contribute least.  The update uses the incremental
    convolution routine, so each step is O(n).
    """
    n = a.shape[0]
    eps = 1e-12
    while time.time() < deadline:
        M = conv.max()
        tol = max(eps, 1e-9 * M)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            break
        w = np.zeros_like(conv)
        w[active] = 1.0

        # gradient of the active max (up to factor 2)
        grad_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * grad_rev[n - 1:n - 1 + n]

        src_candidates = np.argsort(-grad)[:5]   # biggest contributors
        tgt_candidates = np.argsort(grad)[:5]    # smallest contributors

        improved = False
        for src in src_candidates:
            if a[src] < eps:
                continue
            max_delta = min(a[src] * 0.2, 0.05)   # modest move
            if max_delta < eps:
                continue
            for tgt in tgt_candidates:
                if src == tgt:
                    continue
                delta = max_delta
                a_snap = a.copy()
                conv_snap = conv.copy()
                _apply_transfer(a, conv, src, tgt, delta)
                newM = conv.max()
                if newM <= M + 1e-14:
                    improved = True
                    break
                # revert
                a[:] = a_snap
                conv[:] = conv_snap
            if improved:
                break
        if not improved:
            break
    return a, conv

            if T > 0:
                prob = math.exp(-(new_score - cur_score) / T)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            cur_score = new_score
            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        T *= decay

        if it % 4000 == 0 and it_no_improve > 1200:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
