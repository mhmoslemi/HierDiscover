"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The algorithm tries many diverse seeds,
optimises each with a projected sub‑gradient descent on the
maximum self‑convolution, then refines with the greedy active‑set
method and a short simulated‑annealing walk.

The optimisation works on the simplex (sum = 1) because the
objective 2·n·max(conv(a,a))/sum(a)² is scale‑invariant.
The sub‑gradient of the active (max) convolution is obtained
analytically, a step is taken along the direction that reduces
the peak, and the vector is projected back onto the simplex.
Multiple restarts are performed until the time budget runs out.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta on {0,…,n‑1} with shape parameter p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a

    cand.append(_distribution_arcsine(n))

    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # Random sparse vectors (≈0.5 % non‑zeros)
    k = max(2, n // 200)
    sparse = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    sparse[idx] = rng.random(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Random Dirichlet draws with varied concentration
    for _ in range(5):
        conc = rng.uniform(0.05, 5.0)               # single concentration for all components
        vec = rng.dirichlet(np.full(n, conc))
        cand.append(vec)

    # Noisy perturbations of a few beta shapes
    for _ in range(3):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert)
        cand.append(pert)

    return cand


# ----------------------------------------------------------------------
def _subgradient_descent(p: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the maximal self‑convolution.
    Returns the best vector seen and its objective value.
    """
    n = p.shape[0]
    best = p.copy()
    best_score = _objective(p)
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(p)
        M = conv.max()
        eps = M * 1e-12 + 1e-12
        active = np.where(conv >= M - eps)[0]

        # sub‑gradient of the max convolution
        g = np.zeros(n, dtype=np.float64)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * p[j]

        # direction that reduces the active set value
        d = -(g - g.mean())                     # sum(d) == 0
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-15:                      # degenerate – add tiny jitter
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            delta = min(p[i], 1e-4)
            if delta > 0:
                p_tmp = p.copy()
                p_tmp[i] -= delta
                p_tmp[j] += delta
                p = _project_onto_simplex(p_tmp)
            continue

        # step size decays roughly like 1/√t
        step = step0 / (1.0 + 0.001 * it)

        p_new = p + step * d
        p_new = _project_onto_simplex(p_new)

        new_score = _objective(p_new)
        if new_score < best_score - 1e-14:
            best_score = new_score
            best = p_new.copy()
            # occasionally print progress
            if it % 200 == 0:
                print(f"[{time.time():.1f}] subgrad it={it} best={best_score:.6f}")

        # accept if the move does not increase the objective
        if new_score <= best_score + 1e-12:
            p = p_new

        # occasional random jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.3:
            noise = rng.normal(scale=0.001, size=n)
            p_tmp = p + noise
            p_tmp = np.clip(p_tmp, 0, None)
            p_tmp = _project_onto_simplex(p_tmp)
            scr = _objective(p_tmp)
            if scr < best_score:
                p = p_tmp
                best = p_tmp.copy()
                best_score = scr
                print(f"[{time.time():.1f}] subgrad jitter it={it} new best {best_score:.6f}")

    return best, best_score

def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer based on the exact sub‑gradient
    of the active convolution shift(s).  Returns the best vector seen.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient for active shifts
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        delta = min(a[src], 0.5)                 # try moving a decent chunk
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # smaller fraction if full move fails
        delta *= 0.2
        if delta < 1e-9:
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # occasional jitter
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps_move = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps_move
                a_tmp[j] += eps_move
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
    return best_vec, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑type random walk with exponential cooling.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    T = 0.03

    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9997

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Generate a candidate step‑function within the given time budget.
    The routine builds many analytic and random seeds, optimises each
    with sub‑gradient descent, refines with the greedy active‑set
    optimiser and a short simulated‑annealing walk, and returns the
    best vector found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5     # safety margin

    # Length: larger than the previous 3300 gives us more freedom.
    n = 5000
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 1) Generate seeds
    candidates = _initial_candidates(n, rng)

    # include previously discovered solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size == n:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) Quick scan of candidates (evaluate with the cheap objective)
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx} score={sc:.6f}")

    if best_vec is None:
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Main optimisation loop – multiple restarts until time runs out
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:      # not enough time for a full restart
            break

        # pick a random initial point (occasionally from the pool)
        if rng.random() < 0.3:
            init = best_vec.copy()
        else:
            # random Dirichlet with random concentration
            conc = rng.uniform(0.05, 5.0)
            init = rng.dirichlet(np.full(n, conc))
        init = _project_onto_simplex(init)

        # allocate a slice of the remaining budget for this restart
        slice_deadline = time.time() + min(6.0, remaining * 0.4)

        # sub‑gradient descent (fast, captures most improvement)
        vec, sc = _subgradient_descent(init, slice_deadline, rng)

        # aggressive greedy refinement (uses remaining time of the slice)
        if time.time() < slice_deadline:
            vec, sc = _greedy_local_search(vec, slice_deadline, rng)

        # short SA fine‑tuning
        if time.time() < slice_deadline:
            vec, sc = _simulated_annealing(vec, slice_deadline, rng)

        if sc < best_score - 1e-14:
            best_score = sc
            best_vec = vec.copy()
            print(f"[{time.time():.1f}] NEW GLOBAL BEST {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final fine‑tuning with greedy + SA on the remaining budget
    if time.time() < deadline:
        best_vec, best_score = _greedy_local_search(best_vec, deadline, rng)
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)

    # ------------------------------------------------------------------
    # 5) Ensure feasibility and return
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
