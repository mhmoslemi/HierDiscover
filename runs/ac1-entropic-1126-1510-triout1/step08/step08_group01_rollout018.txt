"""
Hybrid optimizer for minimizing 2*n*max(a*a) / (sum a)^2.

The algorithm:
1) Generate a diverse set of seed distributions (uniform, arcsine,
   beta families, Dirichlet draws) of various lengths.
2) Keep the best few seeds according to the cheap surrogate
   `2*n*max(conv(a,a))` (i.e. `_max_convolution_score`).
3) For each seed run two complementary local searches:
   * Adam optimisation of a smooth quadratic surrogate that forces the
     self‑convolution to be close to the uniform distribution (target
     `1/(2*n-1)`).  The gradient of this surrogate is cheap via a single
     FFT.
   * Standard sub‑gradient Adam on the true `max` objective.
4) The overall best vector is finally polished by a simulated‑annealing
   random mass‑transfer routine that explores non‑convex directions that
   Adam may miss.
All steps respect the time budget and the result is returned as a plain
`list[float]`.
"""
import time
import numpy as np
from numpy.fft import rfft, irfft

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for evaluate_sequence when sum(a)=1: 2·n·max(a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of the max‑convolution objective on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (when i=j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

def _smooth_target_gradient(a: np.ndarray, target: np.ndarray) -> np.ndarray:
    """
    Gradient of  J(a) = ||conv(a,a) - target||_2^2 .
    Returns 4 * (a_rev * (conv - target)) restricted to length n.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    diff = conv - target
    # convolution of reversed a with diff
    rev = _fft_convolve(a[::-1], diff)
    grad = 4.0 * rev[n - 1 : n - 1 + n]
    return grad

def _adam_optimize_target(a0: np.ndarray, target: np.ndarray,
                          deadline: float,
                          max_iters: int = 200_000,
                          step0: float = 0.05,
                          beta1: float = 0.9,
                          beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimiser for the smooth L2‑target objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    cur_score = _max_convolution_score(a)   # use true objective for tracking
    best_score = cur_score
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_target_gradient(a, target)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _adam_subgrad(a0: np.ndarray,
                  deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimisation directly on the max‑convolution objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _annealing_refine(a0: np.ndarray,
                      deadline: float,
                      init_temp: float = 0.01,
                      decay: float = 0.9995,
                      min_temp: float = 1e-7,
                      batch: int = 1) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing mass‑transfer polish."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    temp = init_temp
    n = a.shape[0]
    while time.time() < deadline and temp > min_temp:
        # propose a batch of independent moves
        for _ in range(batch):
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # maximal feasible transfer from i to j
            max_transfer = min(a[i], 1.0 - a[j])
            if max_transfer <= 0:
                continue
            # transfer scaled by temperature
            delta = max_transfer * min(1.0, np.random.rand() * temp * 10)
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            new_score = _max_convolution_score(a_new)
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # Metropolis rule
                prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
                if np.random.rand() < prob:
                    accept = True
            if accept:
                a = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        temp *= decay
    return best_a, best_score

def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of candidate vectors of length n."""
    seeds = []
    # uniform
    seeds.append(np.full(n, 1.0 / n))
    # arcsine
    seeds.append(_arcsine_distribution(n))
    # power‑transformed arcsine
    base = _arcsine_distribution(n)
    for p in [0.5, 0.8, 1.2, 1.5, 2.0]:
        a = np.power(base, p)
        a /= a.sum()
        seeds.append(a)
    # symmetric beta
    for p in [0.2, 0.5, 0.8, 1.0, 1.5, 2.0]:
        a = _beta_symmetric_distribution(n, p)
        seeds.append(a)
    # unsymmetric beta grid
    for alpha in [0.3, 0.5, 0.8, 1.0, 1.5]:
        for beta in [0.3, 0.5, 0.8, 1.0, 1.5]:
            a = _beta_distribution(n, alpha, beta)
            seeds.append(a)
    # random Dirichlet draws
    for alpha in [0.1, 0.2, 0.5, 1.0]:
        a = rng.dirichlet(np.full(n, alpha))
        seeds.append(a)
    return seeds

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

Hybrid optimizer for the step‑function bound.

The algorithm works in three stages under a strict time budget:

1. **Broad sampling** – a large collection of analytic families (arcsine,
   symmetric/unsymmetric beta, power‑transformed arcsine, random Dirichlet)
   yields many feasible vectors.  All vectors are ℓ₁‑normalised (sum = 1)
   and scored with the cheap surrogate `2·n·max(conv(a,a))`.

2. **Two‑phase refinement** – each of the best few seeds is first polished
   by Adam on a smooth quadratic surrogate that forces the self‑convolution
   to be close to the uniform distribution (target `(2·n‑1)⁻¹`).  The result
   is then fed to a second Adam run that directly minimises the true
   `max`‑objective via the exact sub‑gradient.  Both inner loops respect
   the global deadline.

3. **Simulated‑annealing polish** – starting from the current global best,
   random mass‑transfer moves are proposed.  Moves are accepted if they
   improve the objective or with a Metropolis probability that decays with
   a simple exponential cooling schedule.  The annealing stage can escape
   shallow local minima that gradient descent may miss.

All steps protect against invalid vectors by projecting onto the simplex.
The returned list is ready for `evaluate_sequence`; scaling does not affect
the score because the objective is homogeneous of degree 0.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)

            accept = False
            if new_score < cur_score:
                accept = True
            else:
                prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
                if np.random.rand() < prob:
                    accept = True
            if accept:
                a = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        temp *= decay

    return best_a, best_score

# analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _generate_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse list of candidate vectors of length n."""
    seeds = []

    # 1. uniform
    seeds.append(np.full(n, 1.0 / n))

    # 2. arcsine and power transforms
    arcs = _arcsine_distribution(n)
    seeds.append(arcs)
    for p in (0.5, 0.8, 1.2, 1.5, 2.0):
        a = np.power(arcs, p)
        a /= a.sum()
        seeds.append(a)

    # 3. symmetric beta with several shape parameters
    for p in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0):
        seeds.append(_beta_symmetric_distribution(n, p))

    # 4. unsymmetric beta grid
    for alpha in (0.3, 0.5, 0.8, 1.0, 1.5):
        for beta in (0.3, 0.5, 0.8, 1.0, 1.5):
            seeds.append(_beta_distribution(n, alpha, beta))

    # 5. random Dirichlet draws (low‑α concentrate mass, high‑α approach uniform)
    for alpha in (0.1, 0.2, 0.5, 1.0, 2.0):
        seeds.append(rng.dirichlet(np.full(n, alpha)))

    return seeds

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget and
    returns the best vector discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a base length.  If a previous solution exists we reuse its size.
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # A handful of lengths – the optimum may prefer a slightly different size.
    lengths = set()
    for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * f)
        if 500 <= n <= 20000:
            lengths.add(n)
    for _ in range(5):
        n = rng.integers(500, min(20000, base_n * 3) + 1)
        lengths.add(int(n))
    lengths = sorted(lengths)
    print(f"[{time.time():.1f}] Exploring lengths: {lengths}")

    # ------------------------------------------------------------------
    # 1. Broad sampling
    all_cands = []   # (vector, surrogate_score)
    for n in lengths:
        if time.time() >= deadline:
            break
        seeds = _generate_seeds(n, rng)
        for a in seeds:
            score = _max_convolution_score(a)
            all_cands.append((a, score))
        print(f"[{time.time():.1f}] n={n:<5d} generated {len(seeds)} seeds, best among them {min(s for _, s in all_cands[-len(seeds):]):.6f}")

    # ------------------------------------------------------------------
    # Keep a modest number of top seeds for expensive local optimisation.
    all_cands.sort(key=lambda x: x[1])
    top_k = min(8, len(all_cands))
    top = all_cands[:top_k]
    best_seq, best_score = top[0][0].copy(), top[0][1]
    print(f"Total candidates {len(all_cands)} – retaining top {top_k} (best surrogate {best_score:.6f})")

    # ------------------------------------------------------------------
    # 2. Two‑phase refinement of each top seed
    for idx, (seed_vec, _) in enumerate(top):
        if time.time() >= deadline:
            break
        n = seed_vec.shape[0]
        # smooth L2‑target (uniform convolution) – helps flatten the profile
        target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
        seq_s, sc_s = _adam_optimize_target(seed_vec, target, deadline,
                                            max_iters=120_000,
                                            step0=0.03)
        # direct max‑objective Adam
        seq_m, sc_m = _adam_subgrad(seq_s, deadline,
                                    max_iters=100_000,
                                    step0=0.15)
        if sc_m < best_score:
            best_score, best_seq = sc_m, seq_m.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} refined -> new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing final polish (if time permits)
    if time.time() < deadline - 5.0:
        print(f"[{time.time():.1f}] Starting annealing polish (score {best_score:.6f})")
        seq_a, sc_a = _annealing_refine(best_seq, deadline,
                                        init_temp=0.02,
                                        decay=0.998,
                                        batch=2)
        if sc_a < best_score:
            best_score, best_seq = sc_a, seq_a.copy()
            print(f"[{time.time():.1f}] annealing gave new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final surrogate score {best_score:.6f} (length={len(best_seq)})")
    # The vector is already ℓ₁‑normalised; `evaluate_sequence` is homogeneous,
    # so any positive scaling would give the same score.
    return best_seq.tolist()
