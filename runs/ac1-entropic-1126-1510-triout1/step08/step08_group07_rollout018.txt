"""
A fast “smooth‐max + exponential‐gradient” optimizer for the
self‑convolution min‑max problem.

The algorithm works in three stages:

1. **Rich start‑up** – a set of analytic families (uniform,
   arcsine, symmetric beta, exponential, alternating, random Dirichlet)
   is sampled for a moderate length (4096).  An inexpensive grid search
   over the beta shape parameter finds a refined beta distribution.

2. **Smooth‑max exponential‑gradient descent** – the non‑smooth objective
   `max(conv(a,a))` is replaced by a soft‐max
   `Φβ(a)= (1/β)·log Σ exp(β·conv_k)`.  Its gradient can be written
   analytically and computed with a single FFT+correlation per
   iteration.  An exponentiated‑gradient step (multiplicative update)
   keeps the vector on the simplex.  β is increased slowly, turning the
   smooth surrogate into the true max.

3. **Active‑set greedy polishing** – the exact sub‑gradient of the
   true max (obtained from the active convolution shifts) is used to
   move mass from the worst to the best index, as in the previous
   state‑of‑the‑art code.  This cleans up the remaining tiny
   imperfections.

All stages respect a global time budget (default 1000 s).  The final
vector is projected onto the probability simplex and returned as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape p>0 (density ∝ x^{p‑1}(1‑x)^{p‑1})."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) by clipping x into (0,1)
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    """Random point on the simplex via Dirichlet(α)."""
    v = rng.gamma(alpha, 1.0, size=n)
    if v.sum() == 0:
        v = np.ones(n, dtype=np.float64)
    return v / v.sum()


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse catalogue of start vectors for length n."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta – wide range of shape parameters
    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – several decay rates
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # random Dirichlet draws
    for _ in range(5):
        cand.append(_distribution_dirichlet(n, rng, alpha=rng.uniform(0.05, 2.0)))

    return cand

    best_score = float("inf")

        np.logspace(np.log10(0.005), np.log10(0.2), num=30, endpoint=False),

def _exp_gradient_descent(a0: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator):
    """Smooth‑max exponential‑gradient descent on the simplex."""
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    eta = 0.5                     # initial step size
    beta = 30.0                   # smooth‑max temperature
    it = 0
    eps = 1e-30

    while time.time() < deadline and it < 8000:
        # 1) self‑convolution
        conv = _fft_self_convolve(a)

        # 2) soft‑max weights (smooth approximation of the max)
        m = conv.max()
        w = np.exp(beta * (conv - m))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 3) gradient of the smooth objective
        size = 1 << ((3 * n - 2) - 1).bit_length()
        Af = np.fft.rfft(a, n=size)
        Wf = np.fft.rfft(w, n=size)
        # correlation = ifft( conj(A) * W )
        corr = np.fft.irfft(np.conj(Af) * Wf, n=size)
        g = 2.0 * corr[:n]       # sub‑gradient of Φβ

        # 4) exponentiated‑gradient update (multiplicative)
        log_a = np.log(a + eps) - eta * g
        # stabilise the exponentials
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        # 5) schedule
        eta *= 0.9995
        if it % 200 == 0 and beta < 200:
            beta *= 1.06

        # 6) keep best
        if it % 50 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_a = a.copy()
                print(f"[{time.time():.1f}] exp‑grad it={it} new best {best_score:.6f}")

        it += 1

    return best_a, best_score

def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer using the exact sub‑gradient of the
    active convolution shifts (the exact algorithm from the reference).
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = np.where(b >= maxb - eps)[0]

        # sub‑gradient for the active set
        g = np.zeros_like(a)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        # source (largest gradient that actually holds mass)
        src_cand = np.where(a > 1e-12)[0]
        if src_cand.size == 0:
            break
        src = src_cand[np.argmax(g[src_cand])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # try to move a substantial amount of mass
        delta = min(a[src], 0.5)
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        news = _objective(a_new)

        if news < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = news
            continue

        # smaller step if the big one failed
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        news = _objective(a_new)
        if news < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = news
            continue

        # occasional random jitter to escape plateaus
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                news = _objective(a_tmp)
                if news < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = news
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises ``evaluate_sequence``.  The routine builds many analytic
    seeds, refines the most promising one with a smooth–max
    exponential‑gradient optimiser, and finishes with an active‑set
    greedy polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # length – a power of two gives fast FFTs; 4096 works well in practice
    n = 4096
    if n < 2:
        n = 2

    # --------------------------------------------------------------
    # 1) build a diverse set of start vectors
    candidates = _initial_candidates(n, rng)

    # include the previously best construction if it is present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore
        if prev.ndim == 1 and prev.size == n:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2) fine grid search over the Beta‑shape parameter
    best_beta, _ = _grid_search_beta(n, deadline, rng)
    candidates.append(_distribution_beta_sym(n, best_beta))

    # --------------------------------------------------------------
    # 3) pick the best initial candidate (plain objective)
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx} score={sc:.6f}")

    if best_vec is None:                       # very unlikely
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)

    # --------------------------------------------------------------
    # 4) smooth‑max exponential‑gradient descent
    if time.time() < deadline:
        best_vec, best_score = _exp_gradient_descent(best_vec, deadline, rng)

    # --------------------------------------------------------------
    # 5) aggressive greedy polishing (exact max)
    if time.time() < deadline:
        best_vec, best_score = _greedy_local_search(best_vec, deadline, rng)

    # --------------------------------------------------------------
    # final sanity – ensure we are on the simplex
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
