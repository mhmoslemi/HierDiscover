def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution a*b using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Return the (unnormalised) objective 2·n·max(a∗a) for sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – all return vectors with L1‑sum = 1.
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Discrete symmetric Beta(p,p) on the grid {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / n
    else:
        w /= s
    return w


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer `delta` mass from a[i] to a[j] (i≠j) and update the full
    self‑convolution `conv` in O(n) time.
    """
    if delta <= 0.0 or i == j:
        return
    # old values needed for the O(n) update
    a_i_old = a[i]
    a_j_old = a[j]

    # Update convolution: (a*a)[k] = Σ_l a[l] a[k‑l]
    # effect of decreasing a[i] by delta:
    conv[i:i + a.size] -= 2.0 * delta * a  # subtract all products a[i]*a[·] twice
    # effect of increasing a[j] by delta:
    conv[j:j + a.size] += 2.0 * delta * a
    # diagonal terms (i,i) and (j,j) need +delta² because they were sub‑added/added twice
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta

    a[i] = a_i_old - delta
    a[j] = a_j_old + delta


# ----------------------------------------------------------------------
def _optimize_one_seed(a0, deadline):
    """
    Perform the three optimisation phases on a single seed vector.
    Returns (best_vector, best_score) found before reaching `deadline`.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-15

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max mirror descent (exponentiated gradient)
    a_phase = a.copy()
    schedule = [
        # (alpha_start, alpha_end, steps, lr_start, lr_end)
        (1.0, 1e3, 2000, 0.15, 0.015),
        (1e3, 1e5, 3000, 0.015, 0.0015),
    ]

    best_vec = a_phase.copy()
    best_score = _max_convolution_score(best_vec)

    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_vec, best_score
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a_phase)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum

            # gradient of ⟨w, a*a⟩ w.r.t. a : 2·(a⋆w)
            grad = 2.0 * _fft_convolve(a_phase[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient update
            a_phase = a_phase * np.exp(-lr * grad)
            a_phase = _project_onto_simplex(a_phase, 1.0)
            a_phase = np.maximum(a_phase, eps)
            a_phase /= a_phase.sum()

            if (it & 511) == 0:  # periodic check
                cur_sc = _max_convolution_score(a_phase)
                if cur_sc < best_score - 1e-12:
                    best_score = cur_sc
                    best_vec = a_phase.copy()
                    # early exit if we already beat the target
                    if best_score < 1.503:
                        return best_vec, best_score

    # ------------------------------------------------------------------
    # Phase 2 – true ℓ∞ sub‑gradient with back‑tracking line search
    a_phase = best_vec.copy()
    velocity = np.zeros_like(a_phase)
    beta = 0.9
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a_phase)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)

        # indicator of active constraints
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a_phase[::-1], w)[n - 1 : n - 1 + n]

        # Nesterov‑type momentum
        velocity = beta * velocity + (1.0 - beta) * grad

        # simple back‑tracking: try a step, if objective does not improve, shrink.
        step = 0.2 / math.sqrt(it)  # initial guess
        for _ in range(5):
            a_new = a_phase - step * velocity
            a_new = _project_onto_simplex(a_new, 1.0)
            a_new = np.maximum(a_new, eps)
            a_new /= a_new.sum()
            new_sc = _max_convolution_score(a_new)
            if new_sc < best_score - 1e-12:
                a_phase = a_new
                best_score = new_sc
                best_vec = a_new.copy()
                break
            step *= 0.5
        # terminate early if target reached
        if best_score < 1.503:
            return best_vec, best_score
        # occasional jitter to escape flat regions
        if (it & 511) == 0:
            jitter = np.random.randn(n) * 1e-5
            a_phase += jitter
            a_phase = np.maximum(a_phase, eps)
            a_phase /= a_phase.sum()
    # ------------------------------------------------------------------
    # Phase 3 – O(n) hill‑climber on the convolution
    a_hc = best_vec.copy()
    conv = _fft_self_convolve(a_hc)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # locate a maximal entry of the convolution
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair with maximal product a[i]*a[j] among those generating entry k
        best_i, best_j = i_start, k - i_start
        max_prod = a_hc[best_i] * a_hc[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a_hc[i] * a_hc[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # source is the larger of the two contributors
        source = best_i if a_hc[best_i] >= a_hc[best_j] else best_j
        target = random.randrange(n)
        if target == source:
            continue

        # shift a modest amount of mass
        max_delta = min(a_hc[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a_snap = a_hc.copy()
        conv_snap = conv.copy()
        _apply_transfer(a_hc, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-13:
            M = new_M
            cur_sc = 2.0 * n * M
            if cur_sc < best_score - 1e-12:
                best_score = cur_sc
                best_vec = a_hc.copy()
                if best_score < 1.503:
                    return best_vec, best_score
        else:
            # revert – move was not improving
            a_hc = a_snap
            conv = conv_snap

        # small random jitter every few thousand steps
        if (it % 5000) == 0:
            jitter = np.random.randn(n) * 1e-5
            a_hc += jitter
            a_hc = np.maximum(a_hc, eps)
            a_hc /= a_hc.sum()
            conv = _fft_self_convolve(a_hc)
            M = conv.max()
    return best_vec, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  A mixture of analytic families, soft‑max
    mirror descent, sub‑gradient steps and an O(n) hill‑climber is used.
    The routine respects the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    eps = 1e-15
    best_overall_seq = None
    best_overall_score = float("inf")
    best_overall_n = None

    # ------------------------------------------------------------------
    # 1. Length sweep + cheap screening
    lengths = [2000, 4000, 8000, 15000, 25000, 40000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # generate a pool of seeds
        pool = []

        # analytic families
        pool.append(_distribution_uniform(n))
        pool.append(_distribution_arcsine(n))
        for p in (0.3, 0.45, 0.55, 0.8, 1.2):
            pool.append(_distribution_beta_sym(n, p))
        for lam in (0.004, 0.02, 0.08):
            pool.append(_distribution_exponential(n, lam))
        for sigma in (0.2, 0.35, 0.5):
            pool.append(_distribution_gaussian(n, sigma))

        # random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            pool.append(np.random.dirichlet(np.full(n, alpha)))

        # sparse random start
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 250)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        pool.append(sparse)

        # if a previous best is available, throw it in as well
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                prev = np.maximum(prev, eps)
                prev = prev / prev.sum()
                if prev.size == n:
                    pool.append(prev)
        except Exception:
            pass

        # cheap evaluation and keep the 3 best seeds
        scored = []
        for vec in pool:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            scored.append((sc, vec))
        scored.sort(key=lambda x: x[0])
        seeds = [vec for _, vec in scored[:3]]

        # ------------------------------------------------------------------
        # 2. Optimise each seed
        for seed_vec in seeds:
            if time.time() >= deadline:
                break
            cand_vec, cand_score = _optimize_one_seed(seed_vec, deadline)
            if cand_score < best_overall_score - 1e-12:
                best_overall_score = cand_score
                best_overall_seq = cand_vec.copy()
                best_overall_n = n
                print(f"[{time.time():.1f}] length={n} new global best {cand_score:.6f}")
                if best_overall_score < 1.503:
                    print("Target reached – exiting early.")
                    return best_overall_seq.tolist()

    # ------------------------------------------------------------------
    # 3. If nothing beat the target, still return the best we have.
    if best_overall_seq is None:
        # extremely unlikely – fallback to a simple arcsine distribution
        n = 4000
        best_overall_seq = _distribution_arcsine(n)
        best_overall_seq /= best_overall_seq.sum()
        print("Fallback used.")
    else:
        print(f"Final best score ≈ {best_overall_score:.6f} (length={best_overall_n})")
    return best_overall_seq.tolist()
