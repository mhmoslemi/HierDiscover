"""optimize_step_function.py
Search for a non‑negative step‑function that minimises the evaluation
function

    evaluate_sequence(seq) = 2·n·max(convolve(seq,seq)) / (sum(seq)^2)

The objective is invariant to scaling, therefore we work with vectors that
are normalised to sum = 1 and minimise 2·n·max(conv).  The algorithm
consists of three layers:

1. **Broad initialisation** – analytic families (uniform, arcsine, beta,
   exponential, power‑law, sine, alternating, two‑block, random dense and
   random sparse) are generated for a series of lengths.  Every candidate
   is normalised to the simplex.

2. **Smoothed Adam optimisation** – a continuation on a log‑sum‑exp
   smooth approximation of the max‑convolution.  The smoothing parameter
   α is increased logarithmically while Adam’s learning rate is reduced.
   After each α‑segment the best vector seen so far is recorded.

3. **Greedy mass‑transfer refinement** – random moves that transfer a
   fraction of mass from a source index to a destination index are kept
   only when the true (unsmoothed) objective improves.  This cheap
   Metropolis walk finishes the optimisation.

All steps respect a global time budget (default 1000 s).  The best vector
found is returned as a plain Python list of floats.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    # next power‑of‑two at least 2·n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑γ}   (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_dense(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dense random vector – iid exponential(1) for positivity."""
    vals = rng.exponential(scale=1.0, size=n)
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01):
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a fairly broad set of starting vectors for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_beta_sym(n, 0.5))
    cand.append(_distribution_exponential(n, 0.003))
    cand.append(_distribution_powerlaw_edges(n, 0.15))
    cand.append(_distribution_alternating(n, stride=3))
    cand.append(_distribution_two_blocks(n, block_frac=0.08))

    # random dense / sparse
    cand.append(_distribution_random_dense(n, rng))
    cand.append(_distribution_random_sparse(n, rng, density=0.02))

    return cand

def _smooth_adam(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth max‑convolution objective."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule of (α, steps, learning_rate)
    schedule = [
        (0.5, 2000, 0.20),
        (1.0, 2000, 0.15),
        (5.0, 2000, 0.12),
        (20.0, 2000, 0.10),
        (100.0, 1500, 0.08),
        (500.0, 1000, 0.06),
        (2000.0, 800, 0.04),
        (5000.0, 600, 0.03),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of smooth‑max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact check
            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] α={alpha:.0f} step={t} "
                        f"new best {best_score:.6f}"
                    )
    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Random mass transfers that are kept only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # pick a destination distinct from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 30 % of source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} "
                f"new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimise_one_seed(
    seed: np.ndarray,
    deadline: float,
    global_best_seq: np.ndarray,
    global_best_score: float,
) -> tuple[np.ndarray, float]:
    """Run Adam + greedy refinement on a single seed."""
    # Adam phase
    seq, score = _smooth_adam(seed.copy(), deadline, global_best_seq, global_best_score)
    # Greedy fine‑tuning
    if time.time() < deadline:
        seq, score = _pairwise_swap_local(
            seq.copy(), deadline, seq, score
        )
    return seq, score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths – longer vectors give finer discretisation.
    #    We try several increasing sizes; the loop will stop early if we run out of time.
    lengths = [8000, 12000, 20000, 30000, 40000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length and keep a few promising seeds.
    candidate_seeds: list[np.ndarray] = []

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating candidates for length {n}")

        cand_list = _initial_candidates(n, rng)
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = _project_onto_simplex(cand, 1.0)  # guarantee sum = 1
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] new global best length={n} "
                    f"score={sc:.6f}"
                )
            candidate_seeds.append(cand)

        # keep a manageable number of seeds – the best 8 so far
        candidate_seeds.sort(key=_objective)
        candidate_seeds = candidate_seeds[:8]

    # ------------------------------------------------------------------
    # 3. If a previously discovered construction is available, use it.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = _project_onto_simplex(prev, 1.0)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            candidate_seeds.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. If nothing was generated, fall back to a uniform vector.
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation loop – run Adam + greedy refinement on every seed.
    while time.time() < deadline and candidate_seeds:
        # allocate a fair time slice for the next seed
        remaining = deadline - time.time()
        per_seed = remaining / len(candidate_seeds)
        seq, sc = _optimise_one_seed(
            candidate_seeds.pop(0),
            time.time() + per_seed,
            overall_best_seq,
            overall_best_score,
        )
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(
                f"[{time.time():.1f}] improved global best to {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
