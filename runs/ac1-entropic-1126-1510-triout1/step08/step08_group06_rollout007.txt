"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the full (linear) convolution.
The search consists of three stages

1. **Broad initialization** – many analytic families (uniform,
   arcsine, symmetric beta, power‑law edges, exponential, sine,
   alternating, two‑block spikes, sparse random, Dirichlet draws)
   are generated for a few lengths.  The exact objective is evaluated
   and the best few vectors are kept as seeds.

2. **Extended smoothed Adam descent** – a continuation on the
   log‑sum‑exp smooth approximation of the max‑convolution.
   The smoothing parameter `α` is increased up to 2·10⁵ and the
   learning rate is reduced accordingly.  After each Adam step the
   vector is projected onto the probability simplex and symmetrised
   (a[i]=a[n‑1‑i]), keeping the scale invariant property of the
   original objective.

3. **Simulated‑annealing pairwise swaps** – a cheap Metropolis walk
   that repeatedly transfers a random amount of mass from a source
   index to a destination index.  Moves are accepted if they improve
   the true objective, otherwise they are accepted with a
   temperature‑controlled probability.  This finishes the optimisation.

The whole routine respects a global time budget (default 1000 s) and
returns the best vector found as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.nonzero(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) / (sum(a)**2).
    The vector is assumed non‑negative; no extra normalisation is performed.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    maxb = b.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * maxb / (s * s)


def _objective_sum1(a: np.ndarray) -> float:
    """Same as _objective but assumes sum(a)=1 (used during optimisation)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid, robust for p<1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space to avoid overflow when p<1
    logw = (p - 1.0) * (np.log(x) + np.log(1.0 - x))
    # shift to avoid huge exponents
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0.0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    vals = np.power(denom, -gamma, where=denom > 0, out=np.zeros_like(denom))
    if vals.sum() == 0.0:
        return _distribution_uniform(n)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0.0:
        return _distribution_uniform(n)
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           concentration: float = 0.2) -> np.ndarray:
    """Draw a Dirichlet vector with the given concentration."""
    a = rng.gamma(concentration, 1.0, size=n)
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _make_symmetric(v: np.ndarray) -> np.ndarray:
    """Enforce exact mirror symmetry (v[i]=v[n-1-i])."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a reasonably diverse pool of start vectors for length n."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))

    # Beta family – wide range of shape parameters (including p<0.05)
    for p in (0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Power‑law at the edges
    for gamma in (0.01, 0.05, 0.1, 0.2, 0.3, 0.4, 0.45, 0.48):
        cand.append(_distribution_powerlaw_edges(n, gamma))

    # Exponential decays – a few lambdas
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # Alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # Two‑block spikes
    for block_frac in (0.03, 0.05, 0.08, 0.12):
        cand.append(_distribution_two_blocks(n, block_frac))

    # Random sparse starts
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws (different concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, concentration=conc))

    # Symmetrise everything (keeps the sum unchanged)
    cand = [_make_symmetric(v) for v in cand]

    # Normalise to sum = 1 (objective is scale‑invariant)
    cand = [v / v.sum() for v in cand]
    return cand


# ----------------------------------------------------------------------
def _smooth_adam(a: np.ndarray,
                 deadline: float,
                 best_seq: np.ndarray,
                 best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smoothed max objective.
    The schedule is extended up to very large smoothing factors.
    The vector is symmetrised after every update.
    """
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule: (α, steps, learning_rate)
    schedule = [
        (0.5, 2000, 0.20),
        (1.0, 2000, 0.15),
        (5.0, 2000, 0.12),
        (20.0, 2000, 0.10),
        (100.0, 2000, 0.08),
        (500.0, 1500, 0.06),
        (2000.0, 1200, 0.04),
        (5000.0, 1000, 0.03),
        (20000.0, 800, 0.02),
        (50000.0, 600, 0.015),
        (100000.0, 500, 0.012),
        (200000.0, 400, 0.010),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            # self‑convolution
            b = _fft_self_convolve(a)
            maxb = b.max()
            # smooth‑max weights (log‑sum‑exp)
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of smooth‑max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # enforce symmetry and simplex feasibility
            a = _make_symmetric(a)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 500 == 0:
                cur = _objective_sum1(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis walk that transfers a random amount of mass from a
    source index to a destination index.  Moves that improve the true
    objective are always accepted; otherwise they are accepted with a
    probability exp(-(Δ)/T) where the temperature T decreases linearly
    with the remaining time.
    """
    n = a.shape[0]
    cur_seq = a.copy()
    cur_score = best_score
    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        # linear temperature from 1e-3 down to 0
        elapsed = time.time() - start
        T = max(1e-6, 1e-3 * (1.0 - elapsed / total))

        # pick a source that holds non‑negligible mass
        src_idxs = np.where(cur_seq > 1e-14)[0]
        if src_idxs.size == 0:
            break
        src = np.random.choice(src_idxs)

        # distinct destination
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # transfer a random fraction (up to 40 % of source mass)
        delta = cur_seq[src] * np.random.rand() * 0.40
        if delta < 1e-14:
            continue

        new_seq = cur_seq.copy()
        new_seq[src] -= delta
        new_seq[dst] += delta

        # re‑symmetrise and re‑project (keeps sum=1)
        new_seq = _make_symmetric(new_seq)
        new_seq = _project_onto_simplex(new_seq, 1.0)

        new_score = _objective_sum1(new_seq)

        Δ = new_score - cur_score
        if Δ < -1e-15:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{time.time():.1f}] SA improve Δ={Δ:.6g} new best {best_score:.6f}")
        else:
            if np.random.rand() < math.exp(-Δ / T):
                cur_seq = new_seq
                cur_score = new_score

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function obeys the global time limit
    (default 1000 s) and returns the best vector found as a Python list.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a handful of lengths – we keep them modest to stay fast.
    lengths = [5000, 10000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2.  Scan analytic families for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            # already normalised, just evaluate
            sc = _objective_sum1(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best length={n} score={sc:.6f}")

            top_candidates.append((sc, cand.copy()))

        # keep only a few elite seeds for the next phase
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3.  Include a previously discovered solution if present.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = _make_symmetric(prev)
            prev = prev / prev.sum()
            sc = _objective_sum1(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Fallback – uniform if nothing else survived.
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective_sum1(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Main optimisation pipeline.
    # pick up to three seeds (including the current best)
    seeds = [overall_best_seq] + [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:2]]

    # 5a – smoothed Adam descent for each seed (time split)
    for seed_seq in seeds:
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
        print(f"[{time.time():.1f}] Adam start (len={len(seed_seq)})")
        overall_best_seq, overall_best_score = _smooth_adam(
            seed_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after Adam best score={overall_best_score:.6f}")

    # 5b – final simulated‑annealing pairwise swaps
    if time.time() < deadline:
        print(f"[{time.time():.1f}] starting simulated annealing")
        overall_best_seq, overall_best_score = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] final best score: {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result as a plain list of Python floats.
    return overall_best_seq.tolist()
