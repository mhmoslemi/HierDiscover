Propose a step‑function vector that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the ordinary (non‑reversed) convolution.
The algorithm combines three phases:

1. **Diverse initialization** – many analytic families (uniform,
   arcsine, triangular, exponential, beta, sine‑power, Gaussian) as
   well as random Dirichlet draws are generated for several lengths.
   The best among them (or the previously‑found `height_sequence_1`)
   is kept.

2. **LP‑driven boost** – for the current best vector `f` we solve a
   short linear program

        maximise   Σ g_j
        subject to   (f * g)_k ≤ M   for all k,
                     g ≥ 0,

   where `M = max(conv(f,f))`.  The LP finds a feasible `g` with a
   larger ℓ₁‑norm; mixing a little of the normalised `g` into `f`
   strictly reduces the bound.  Only the constraints that are close
   to being active (within 0.1 % of `M`) are used, keeping the LP
   tiny.

3. **Fine‑grained hill‑climbing** – a fast O(n) mass‑transfer
   local search (with occasional jitter) monotonically reduces the
   maximal self‑convolution until the wall‑clock budget expires.

All steps respect the 1000 s budget, use only NumPy / SciPy, and

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution via real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _objective(a):
    """Return 2·n·max(conv(a,a)) for a already normalised (Σa = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a non‑negative vector with Σ=1
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j,
    updating both the vector `a` and its full self‑convolution `conv`
    in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    # keep a copy of the old a for the update formulas
    a_old = a[i]
    # update convolution entries that involve i or j
    # conv[k] = Σ a[p] a[k-p]
    # Changing a[i] -> a[i]-delta, a[j] -> a[j]+delta
    # The effect on conv is linear in delta and quadratic in delta.
    # We use the exact update derived in the original code.
    # Subtract contributions of pairs (i,·) and add contributions of (j,·).
    conv[i : i + n] -= 2.0 * delta * a_old
    conv[j : j + n] += 2.0 * delta * a_old
    # diagonal terms (i,i) and (j,j) need the quadratic correction
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    # finally adjust the vector itself
    a[i] -= delta
    a[j] += delta

def _active_constraint_indices(conv, M, rel_tol=0.001, max_cnt=80):
    """
    Return up to `max_cnt` indices where conv is within `rel_tol`
    of the current maximum `M`.  The indices are the ones with the
    largest values.
    """
    mask = conv >= M * (1.0 - rel_tol)
    idx = np.flatnonzero(mask)
    if idx.size > max_cnt:
        # keep the `max_cnt` largest entries
        order = np.argsort(-conv[idx])
        idx = idx[order[:max_cnt]]
    return idx


def _build_lp_matrix(f, active_idx):
    """
    Build the (sparse) matrix A such that   (f * g)[k] = (A g)_k
    for each active shift `k` in `active_idx`.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idx):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1, dtype=np.int32)
        ks = k - js
        rows.extend([row] * len(js))
        cols.extend(js.tolist())
        data.extend(f[ks].tolist())
    A = sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    return A


def _lp_boost(f, M):
    """
    Solve a small linear program that finds a vector g with maximal ℓ₁ norm
    under the constraints (f * g)_k ≤ M for all k.
    Returns the normalised vector g (Σg = 1) or None if the LP fails.
    """
    if not _SCIPY_AVAILABLE:            # pragma: no cover
        return None

    conv_f = _fft_self_convolve(f)
    active = _active_constraint_indices(conv_f, M, rel_tol=0.001, max_cnt=80)
    if active.size == 0:
        return None

    A = _build_lp_matrix(f, active)
    b = np.full(active.shape[0], M, dtype=np.float64)

    c = -np.ones(f.shape[0], dtype=np.float64)          # maximise Σg
    bounds = [(0, None)] * f.shape[0]

    # `highs` works with sparse matrices and is fast for our tiny LPs.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    if g.sum() <= 0:
        return None
    # Normalise to the simplex
    g = np.maximum(g, 0.0)
    g = g / g.sum()
    return g

        # move mass away from the larger of the two indices
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # eligible transfer size – keep changes modest
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        conv_snapshot = conv.copy()

        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        new_score = 2.0 * n * new_M

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_vec = a.copy()
            M = new_M
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill iter={it} new best {new_score:.6f}")
        else:
            # revert – the move was not beneficial
            a = a_snapshot
            conv = conv_snapshot

        # occasional jitter to shake out of plateaux
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_vec.tolist()
```
