"""

Algorithm (improved version of the previous baseline)

   probability simplex.  Families: arcsine, beta (dense p‑grid),

   methods (exponentiated‑gradient, Adam and smooth‑Adam) with

   the sub‑gradient, with occasional random jitter to escape plateaus.

6. **Optional LP‑style boost** – a single extra pass that solves a
   linear programme (using SciPy if available) to increase the total
   mass of a feasible vector and mixes it back into the best

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a) under the assumption sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """Gradient of the log‑sum‑exp smooth approximation of max(conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w /= w.sum()

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.9):
    """Exponentiated‑gradient (mirror descent w.r.t. KL) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam optimiser on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eps0: float = 1e-3,
                         step0: float = 0.2,
                         beta1: float = 0.9, beta2: float = 0.999):
    """Adam on the log‑sum‑exp smoothed objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    eps = eps0
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)

        a = a - eta * m_hat

        if t % 40_000 == 0:

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 0.6):
    """Projected sub‑gradient descent with diminishing steps."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:

        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families (all return a probability vector, i.e. sum = 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget and prints progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Lengths to explore.  We pick a spread from ~500 up to ~12 000.
    lengths = set()
    base_vals = [800, 1200, 2000, 3000, 5000, 8000, 12000]
    for n in base_vals:
        if n >= 500:
            lengths.add(n)
    # add a few random lengths for extra diversity
    for _ in range(7):
        n = np.random.randint(500, 12001)
        lengths.add(n)
    lengths = sorted(lengths)

    candidates = []   # list of (array, cheap_score)

    # ------------------------------------------------------------------
    # Parameter grids for the analytic families
    beta_ps = np.concatenate([
        np.linspace(0.02, 0.5, 9),
        np.linspace(0.6, 2.0, 8),
        np.array([2.5, 3.0])
    ])
    exp_lams = [0.0005, 0.001, 0.003, 0.006, 0.01]
    sigma_fracs = [0.12, 0.18, 0.24, 0.30, 0.36, 0.45]
    scale_fracs = [0.12, 0.18, 0.24, 0.30, 0.36, 0.45]
    dirichlet_alphas = [0.03, 0.1, 0.3, 0.6, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- analytic families -------------------------------------------------
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_uniform_distribution, "uniform"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
        ]
        for gen, name in fams:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} {name:10s} score={score:.6f}")

        # ----- exponential -------------------------------------------------------
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} exponential(lam={lam:.4f}) score={score:.6f}")

        # ----- Gaussian -----------------------------------------------------------
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} gaussian(sig={sf:.2f}) score={score:.6f}")

        # ----- Cauchy -------------------------------------------------------------
        for sf in scale_fracs:
            a = _cauchy_distribution(n, scale_frac=sf)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} cauchy(scale={sf:.2f}) score={score:.6f}")

        # ----- semicircle ---------------------------------------------------------
        a = _semicircle_distribution(n)
        score = _max_convolution_score(a)
        candidates.append((a, score))
        print(f"[{time.time():.1f}] n={n:5d} semicircle score={score:.6f}")

        # ----- symmetric beta -----------------------------------------------------
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} beta(p={p:.3f}) score={score:.6f}")

        # ----- random Dirichlet draws --------------------------------------------
        for alpha in dirichlet_alphas:
            for _ in range(3):
                if time.time() >= deadline:
                    break
                a = _random_dirichlet(n, alpha=alpha)
                score = _max_convolution_score(a)
                candidates.append((a, score))
                print(f"[{time.time():.1f}] n={n:5d} Dirichlet(α={alpha:.2f}) score={score:.6f}")

    # ------------------------------------------------------------------
    # If a previous best sequence exists, add it as a candidate.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best added, score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep a larger pool of seeds (top 40)
    candidates.sort(key=lambda x: x[1])
    top_k = 40
    top_candidates = candidates[:top_k]
    print(f"Selected {len(top_candidates)} seeds for refinement; best cheap score={top_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # Heavy refinement on the seeds
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        # Exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=150_000,
                                        eta0=0.9)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} exp‑grad → {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                max_iters=150_000,
                                step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam → {sc:.6f}")

        # Smooth Adam
        seq, sc = _smooth_adam_descent(seed_seq, deadline,
                                       max_iters=120_000,
                                       eps0=1e-3,
                                       step0=0.2)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smooth‑Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # Long sub‑gradient descent on the current best
    subgrad_deadline = deadline - 30.0   # keep ~30 s for final hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=2_000_000,
                                       eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # ------------------------------------------------------------------
    # Fine‑grained hill‑climbing (guided transfers + random jitter)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02   # up to 2 % of the total mass at start

    while time.time() < deadline:
        iteration += 1
        # geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 12_000.0))

        # pick source & destination indices
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter
        if iteration % 5_000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Optional LP‑style boost (used only if SciPy is available)
    try:
        from scipy.optimize import linprog
        import scipy.sparse as sp

        def _lp_boost(seq: np.ndarray, remaining: float) -> np.ndarray:
            """One LP step that tries to increase total mass while respecting conv bound."""
            n = seq.shape[0]
            M = _fft_self_convolve(seq).max()

            # Build sparse matrix A where A[k,i] = seq[i] if i ≤ k ≤ i+n-1 else 0
            # This is a Toeplitz matrix with constant diagonals equal to seq.
            col_data = [np.full(n, seq[i]) for i in range(n)]
            A = sp.diags(col_data, offsets=range(n), shape=(2 * n - 1, n), format='csc')

            # Linear program: maximise sum(g)  ⇒ minimise -sum(g)
            c = -np.ones(n)

            bounds = [(0, None) for _ in range(n)]
            res = linprog(c, A_ub=A, b_ub=M * np.ones(2 * n - 1), bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': remaining})
            if not res.success:
                return seq
            g0 = res.x
            # Scale to the target sum (chosen as sqrt(2n))
            target = np.sqrt(2.0 * n)
            scale = target / g0.sum()
            g = g0 * scale
            # Small convex mix back into the current sequence
            t = 0.3
            new_seq = (1 - t) * seq + t * g
            return _project_onto_simplex(new_seq, s=1.0)

        # run a single boost if time permits
        if time.time() < deadline - 5.0:
            remaining = deadline - time.time()
            boosted = _lp_boost(best_seq, remaining)
            boosted_score = _max_convolution_score(boosted)
            if boosted_score < best_score:
                best_score, best_seq = boosted_score, boosted.copy()
                print(f"[{time.time():.1f}] LP‑boost improved score to {best_score:.6f}")
    except Exception:
        # SciPy not available or LP failed – ignore.
        pass

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
