"""
Propose a non‑negative step function with a low value of ``evaluate_sequence``.

Algorithm
---------

1. **Broad sampling** – Generate many candidates from analytic families
   (arcsine, beta, triangular, exponential, Gaussian, Cauchy, semicircle,
   sine, cosine, uniform) and from Dirichlet draws with a wide range of
   concentration parameters.  All candidates are normalised to sum = 1 and
   scored by the cheap objective
   ``2·n·max(a*a)`` (identical to ``evaluate_sequence`` for sum = 1).

2. **Select seeds** – Keep the ``top_k`` candidates with the smallest cheap
   scores.

3. **Local optimisation** – For every seed run a *SLSQP* optimisation
   (gradient supplied via the exact sub‑gradient of the max‑convolution).
   This respects the simplex constraints and usually finds a much better
   local minimum than pure gradient descent.

4. **Heavier refinement** – On the best seed so far run a short
   exponentiated‑gradient descent, Adam and a plain sub‑gradient descent.
   These are cheap and can still improve the result.

5. **Simulated‑annealing hill‑climb** – Finally perform a long
   guided‑mass‑transfer hill‑climb that occasionally accepts worsening
   moves (temperature‑controlled).  This helps escape tiny plateaus and
   pushes the cheap score just a little further.

The routine respects the supplied time budget (default 1000 s) and always
returns a feasible sequence (non‑negative, sum > 0).  The returned list can
be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 600_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families -------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _slsqp_refine(a0: np.ndarray, deadline: float, max_iter: int = 120):
    """Local optimisation on the simplex using SLSQP (gradient supplied)."""
    n = a0.shape[0]

    def fun(x):
        return _max_convolution_score(x)

    def jac(x):
        g, _, _ = _subgrad(x)
        return 2.0 * n * g

    # equality: sum(x) = 1
    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - 1.0,
        'jac': lambda x: np.ones_like(x)
    })
    bounds = [(0.0, None)] * n

    if time.time() >= deadline:
        return a0, fun(a0)

    res = minimize(fun, a0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x_opt = res.x if res.success else a0
    # projection for safety
    x_opt = _project_onto_simplex(x_opt, s=1.0)
    return x_opt, fun(x_opt)


# ----------------------------------------------------------------------
def _hill_climb_sa(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Guided mass‑transfer hill‑climbing with simulated‑annealing acceptance.
    """
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02                      # 2 % of total mass at start
    temperature = cur_score * 0.02          # initial temperature

    while time.time() < deadline:
        iteration += 1
        # geometric decay of step size
        transfer = max_transfer * (0.5 ** (iteration / 10_000.0))
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        accept = False
        if new_score < cur_score - 1e-15:
            accept = True
        else:
            # probabilistic acceptance of worse moves
            if temperature > 0:
                prob = np.exp(-(new_score - cur_score) / temperature)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA hill‑climb iter={iteration} new best {best_score:.6f}")

        # temperature decay
        temperature = max(temperature * 0.9999, 1e-12)

        # occasional jitter to re‑explore
        if iteration % 5_000 == 0 and np.random.rand() < 0.08:
            jitter = np.random.randn(n) * 5e-5
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score

            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_uniform_distribution, "uniform"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_semicircle_distribution, "semicircle"),
        ]
        for gen, name in fams:
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:4d} {name:10s} cheap={candidates[-1][1]:.6f}")

        # exponential -------------------------------------------------------
        for lam in exp_lams:
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))
        # Gaussian -----------------------------------------------------------
        for sf in sigma_fracs:
            a = _gaussian_distribution(n, sigma_frac=sf)
            candidates.append((a, _max_convolution_score(a)))
        # Cauchy -------------------------------------------------------------
        for sf in scale_fracs:
            a = _cauchy_distribution(n, scale_frac=sf)
            candidates.append((a, _max_convolution_score(a)))
        # Beta ---------------------------------------------------------------
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))
        # Dirichlet -----------------------------------------------------------
        for alpha in dirichlet_alphas:
            for _ in range(2):
                a = _random_dirichlet(n, alpha=alpha)
                candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 2. Add the previous best sequence (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0:
                prev = prev / s
            else:
                prev = _uniform_distribution(prev.size)
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] previous best added, cheap={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep top‑k seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 12
    top_seeds = candidates[:top_k]
    print(f"Selected {len(top_seeds)} seeds for refinement, best cheap={top_seeds[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Local SLSQP optimisation on every seed
    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]

    for idx, (seed_seq, cheap_sc) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] SLSQP on seed {idx+1}/{len(top_seeds)} start cheap={cheap_sc:.6f}")
        seq, sc = _slsqp_refine(seed_seq, deadline, max_iter=120)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}]   -> new best cheap={sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Heavier refinement – run a few fast gradient methods on the current best
    if time.time() < deadline:
        seq, sc = _exp_gradient_descent(best_seq, deadline, max_iters=60000)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] exp‑grad refinement → {sc:.6f}")

    if time.time() < deadline:
        seq, sc = _adam_descent(best_seq, deadline, max_iters=60000)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] Adam refinement → {sc:.6f}")

    if time.time() < deadline:
        seq, sc = _subgradient_descent(best_seq, deadline, max_iters=400000)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient refinement → {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Final simulated‑annealing hill‑climb
    if time.time() < deadline:
        seq, sc = _hill_climb_sa(best_seq, deadline)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] SA hill‑climb final → {sc:.6f}")

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
