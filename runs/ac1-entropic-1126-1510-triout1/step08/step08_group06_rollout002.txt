"""
Propose a non‑negative step‑function that minimizes
    2·n·max(conv(a,a))  (with sum(a)=1)

The optimizer consists of three stages:

1. **Broad seeding** – generate many analytic families (uniform,
   arcsine, beta, power‑law at the edges, exponential, sine, block‑spikes,
   sparse random) for several lengths.  Keep the best few vectors.

2. **Exact sub‑gradient descent** – repeatedly move in the (negative)
   sub‑gradient of the true max‑convolution (no smooth approximation),
   with momentum and simplex projection.  Adaptive step‑size and occasional
   random perturbations avoid stagnation.

3. **Local greedy swaps** – a cheap Metropolis–type walk that transfers a
   random amount of mass between two indices and keeps the move only if
   the true objective improves.

All stages respect a global time budget (default 1000 s).  The best vector
found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _generate_seeds(rng: np.random.Generator,
                    lengths: list[int],
                    topk: int = 12) -> list[np.ndarray]:
    """
    Produce a broad pool of start vectors for the given lengths.
    Returns the `topk` best vectors (according to the exact objective).
    """
    candidates = []

    for n in lengths:
        # classic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_sine(n))

        # beta family – many shape parameters
        for p in np.linspace(0.05, 5.0, 20):
            candidates.append(_distribution_beta_sym(n, p))

        # power‑law at the edges
        for gamma in np.linspace(0.02, 0.49, 20):
            candidates.append(_distribution_powerlaw_edges(n, gamma))

        # exponentials – wide range
        for lam in np.linspace(0.001, 0.02, 10):
            candidates.append(_distribution_exponential(n, lam))

        # two‑block spikes
        for block_frac in (0.03, 0.07, 0.12):
            candidates.append(_distribution_two_blocks(n, block_frac))

        # sparse random
        candidates.append(_distribution_random_sparse(n, rng, density=0.02))

    # Evaluate and keep the best `topk`
    scored = []
    for cand in candidates:
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        sc = _objective(cand)
        scored.append((sc, cand))
    scored.sort(key=lambda x: x[0])
    return [seq for _, seq in scored[:topk]]


# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses momentum, adaptive step‑size and occasional random perturbations.
    """
    n = a.shape[0]
    step = 0.5          # initial step size
    mu = 0.9            # momentum factor
    vel = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # ----- current objective
        b = _fft_self_convolve(a)
        maxb = b.max()
        cur_score = 2.0 * n * maxb

        # ----- update global best
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad new best {best_score:.6f} len={n}")

        # ----- sub‑gradient of the (smoothed) max
        tol = maxb * 1e-12 + 1e-15
        max_idx = np.where(b >= maxb - tol)[0]
        w = np.zeros_like(b)
        w[max_idx] = 1.0 / len(max_idx)          # average over ties
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # ----- momentum update + projection
        vel = mu * vel - step * grad
        a_candidate = _project_onto_simplex(a + vel, 1.0)

        # ----- evaluate candidate
        b_cand = _fft_self_convolve(a_candidate)
        cand_score = 2.0 * n * b_cand.max()

        if cand_score < cur_score:
            a = a_candidate
            # successful step – slowly increase step size
            step = min(step * 1.05, 1.0)
        else:
            # reject – shrink step and discard momentum
            step *= 0.5
            vel[:] = 0.0

            # very small step → inject a tiny random move
            if step < 1e-6:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    delta = a[i] * rng.random() * 0.01
                    a[i] -= delta
                    a[j] += delta
                a = _project_onto_simplex(a, 1.0)
                step = 0.5

        it += 1
        if it % 2000 == 0:
            print(f"[{time.time():.1f}] subgrad iter {it}, cur {cur_score:.6f}, step {step:.3g}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap random mass transfers that are kept only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0. Load a previously known construction if available
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        init = np.maximum(init, 0.0)
        if init.ndim == 1 and init.size > 0:
            init = init / init.sum()
        else:
            init = None
    except Exception:
        init = None

    # ------------------------------------------------------------------
    # 1. Generate a diversified pool of seed vectors
    lengths = [2000, 3000, 4000, 5000, 8000]
    seeds = _generate_seeds(rng, lengths, topk=12)

    # incorporate the previous construction and a uniform fallback
    if init is not None:
        seeds.append(init)
    else:
        seeds.append(_distribution_uniform(4000))

    # keep only a handful of the most promising seeds
    seeds = sorted(seeds, key=_objective)[:4]

    # ------------------------------------------------------------------
    # 2. Run sub‑gradient descent on each seed, sharing the remaining budget
    overall_best_seq = None
    overall_best_score = float("inf")
    for idx, seed_seq in enumerate(seeds):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds) - idx)
        print(f"[{time.time():.1f}] subgrad start seed {idx+1}/{len(seeds)} len={seed_seq.size}")

        best_seq, overall_best_score = _subgradient_descent(
            seed_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq if overall_best_seq is not None else seed_seq,
            best_score=overall_best_score,
            rng=rng
        )
        overall_best_seq = best_seq.copy()

    # ------------------------------------------------------------------
    # 3. Final local fine‑tuning with cheap random swaps
    if time.time() < deadline and overall_best_seq is not None:
        print(f"[{time.time():.1f}] final greedy swaps (len={len(overall_best_seq)})")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng
        )

    # ------------------------------------------------------------------
    # 4. Report and return
    if overall_best_seq is None:
        overall_best_seq = _distribution_uniform(2000)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
