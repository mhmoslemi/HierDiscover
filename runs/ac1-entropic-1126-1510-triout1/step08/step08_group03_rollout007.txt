"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Broad sampling** – generate a diverse set of candidate vectors from
   analytic families (arcsine, beta, power‑law, triangular, sine,
   cosine, exponential, uniform) and random Dirichlet draws.
   All candidates are normalised to the simplex and scored by the cheap
   surrogate ``2·n·max(a*a)`` (identical to the true objective when the
   sum is 1).

2. **Selection** – keep the ``top_k`` cheapest candidates.

3. **Heavy refinement** – each seed is optimised by mirror‑descent
   (exponentiated‑gradient) on the simplex.  We run the optimiser for a
   time slice that is proportional to the remaining budget.

4. **Final annealing** – the current global best undergoes a simulated‑
   annealing hill‑climb that accepts occasional uphill moves.  The
   temperature decreases linearly to allow exploration early on and a
   fine‑grained exploitation later.

All steps are time‑aware, respect the 1000 s budget and never use
closures or lambda expressions.  The returned list satisfies the
non‑negativity and sum‑invariance requirements of ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of ``max(a*a)`` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for indices where k = 2*i
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 0.5):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _powerlaw_distribution(n: int, alpha: float) -> np.ndarray:
    """w_i ∝ ((i+0.5)*(n-i-0.5))^{-alpha/2}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = (i * (n - i + 0.5)) ** (-alpha / 2.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _anneal_improve(a0: np.ndarray, deadline: float):
    """Simulated‑annealing fine‑tuning of a probability vector."""
    cur = a0.copy()
    cur_score = _max_convolution_score(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]

    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling
        elapsed = time.time() - start
        total = deadline - start
        T = max(0.0, 0.001 * (1.0 - elapsed / total))

        # pick a source index with some mass
        i = np.random.randint(n)
        if cur[i] <= 0.0:
            continue
        # pick a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the source mass
        max_frac = 0.025 * (1.0 - elapsed / total) + 0.001   # shrink later
        delta = np.random.rand() * min(cur[i], max_frac * cur[i])

        a_new = cur.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        diff = new_score - cur_score

        if diff < 0 or np.random.rand() < np.exp(-diff / (T + 1e-12)):
            cur = a_new
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] anneal new best {best_score:.6f}")

        # occasional jitter to shake the system
        if it % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            cur = best + jitter
            cur = _project_onto_simplex(cur, s=1.0)
            cur_score = _max_convolution_score(cur)

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # tiny safety margin

    # ------------------------------------------------------------------
    # Length selection (use previous best if available)
    try:
        base_n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        base_n = 3000

    lengths = set()
    for f in (0.8, 1.0, 1.2, 1.5):
        n = int(base_n * f)
        if 500 <= n <= 8000:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(8000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1. Broad sampling of analytic families and random draws
    candidates = []  # list of (array, cheap_score) – scores filled later
    beta_params = np.linspace(0.15, 1.5, 7)      # include strong U‑shapes

    for n in lengths:
        if time.time() >= deadline:
            break

        # core analytic families
        candidates.append((_arcsine_distribution(n), None))
        candidates.append((_triangular_distribution(n), None))
        candidates.append((_exponential_distribution(n, lam=0.004), None))
        candidates.append((_sine_distribution(n), None))
        candidates.append((_cosine_distribution(n), None))
        candidates.append((np.full(n, 1.0 / n), None))

        # symmetric Beta family
        for p in beta_params:
            candidates.append((_beta_symmetric_distribution(n, p), None))

        # asymmetric Beta family (full grid – cheap because n is moderate)
        for p in beta_params:
            for q in beta_params:
                candidates.append((_beta_asymmetric_distribution(n, p, q), None))

        # power‑law family (generalised arcsine)
        for alpha in np.linspace(0.5, 2.0, 5):
            candidates.append((_powerlaw_distribution(n, alpha), None))

        # few random Dirichlet points
        for _ in range(3):
            candidates.append((_random_dirichlet(n), None))

        # random mixtures of two families
        families = ['arcsine', 'triangular', 'uniform', 'sine',
                    'cosine', 'powerlaw']
        for _ in range(2):
            f1, f2 = np.random.choice(families, size=2, replace=False)

            # first component
            if f1 == 'arcsine':
                a1 = _arcsine_distribution(n)
            elif f1 == 'triangular':
                a1 = _triangular_distribution(n)
            elif f1 == 'uniform':
                a1 = np.full(n, 1.0 / n)
            elif f1 == 'sine':
                a1 = _sine_distribution(n)
            elif f1 == 'cosine':
                a1 = _cosine_distribution(n)
            else:   # powerlaw
                a1 = _powerlaw_distribution(n, np.random.uniform(0.5, 2.0))

            # second component
            if f2 == 'arcsine':
                a2 = _arcsine_distribution(n)
            elif f2 == 'triangular':
                a2 = _triangular_distribution(n)
            elif f2 == 'uniform':
                a2 = np.full(n, 1.0 / n)
            elif f2 == 'sine':
                a2 = _sine_distribution(n)
            elif f2 == 'cosine':
                a2 = _cosine_distribution(n)
            else:
                a2 = _powerlaw_distribution(n, np.random.uniform(0.5, 2.0))

            mix = np.random.rand()
            a = mix * a1 + (1.0 - mix) * a2
            a = a / a.sum()
            candidates.append((a, None))

    # ------------------------------------------------------------------
    # 2. Include previous best sequence (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, None))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Compute cheap scores and keep the top‑k seeds
    scored = [(seq, _max_convolution_score(seq)) for seq, _ in candidates]
    scored.sort(key=lambda x: x[1])
    top_k = 10
    top_seeds = [seq for seq, _ in scored[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Heavy optimisation of each seed (mirror‑descent)
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_seq.tolist()
    per_seed_budget = remaining * 0.6 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        seed_deadline = time.time() + per_seed_budget

        # exponentiated‑gradient descent
        seq, sc = _exp_gradient_descent(seed_seq, seed_deadline,
                                        max_iters=200_000,
                                        eta0=0.9)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} exp‑grad → {best_score:.6f}")

        # optional tidy‑up with sub‑gradient descent (uses any leftover time)
        if time.time() < seed_deadline:
            seq2, sc2 = _subgradient_descent(seq, seed_deadline,
                                             max_iters=200_000,
                                             eta0=0.5)
            if sc2 < best_score - 1e-14:
                best_seq, best_score = seq2.copy(), sc2
                print(f"[{time.time():.1f}] seed {idx+1} sub‑grad → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final simulated‑annealing fine‑tuning on the global best
    leftover = deadline - time.time()
    if leftover > 5.0:
        best_seq, best_score = _anneal_improve(best_seq, deadline)
        print(f"[{time.time():.1f}] after annealing best cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # Normalise one final time and return
    best_seq = best_seq / best_seq.sum()
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
