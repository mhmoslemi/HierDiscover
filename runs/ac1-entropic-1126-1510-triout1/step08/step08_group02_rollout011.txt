def _lp_refine(a, deadline, max_iters=10, tol=1e-8):
    # Input a normalized to sum 1
    n = a.shape[0]
    while time.time() < deadline and max_iters > 0:
        max_conv = _fft_self_convolve(a).max()
        # Build constraint matrix
        # shape (2*n-1, n)
        # Efficient building: using broadcasting
        k = np.arange(2*n-1)[:, None]
        j = np.arange(n)[None, :]
        diff = k - j
        mask = (diff >= 0) & (diff < n)
        A = np.zeros((2*n-1, n), dtype=np.float64)
        A[mask] = a[diff[mask]]
        # solve LP
        c = -np.ones(n)  # minimize -sum(b)
        b_ub = np.full(2*n-1, max_conv, dtype=np.float64)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={"presolve": True, "time_limit": max(0, deadline - time.time())})
        if not res.success:
            break
        b = res.x
        s = b.sum()
        if s <= 1 + tol:
            break   # no improvement
        a = b / s
        max_iters -= 1
        # maybe recompute cheap score for printing
        cheap = 2*n*_fft_self_convolve(a).max()
        print(f"[{time.time():.1f}] LP refined: sum(b)={s:.6f}, cheap={cheap:.6f}")
    return a
```

def _lp_refine(a, deadline, max_iter=12, tol=1e-6):
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute M
        conv = _fft_self_convolve(a)
        M = conv.max()
        # Build constraint matrix A_ub (2n-1 x n)
        # Use dense matrix for simplicity then convert to csr
        k = np.arange(2*n-1)[:, None]
        j = np.arange(n)[None, :]
        diff = k - j
        mask = (diff >= 0) & (diff < n)
        A = np.zeros((2*n-1, n), dtype=np.float64)
        A[mask] = a[diff[mask]]
        # Solve LP
        c = -np.ones(n)  # maximize sum(b)
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        # Convert A to csr for efficiency
        A_csr = csr_matrix(A)
        try:
            res = linprog(c, A_ub=A_csr, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            print(f"[{time.time():.1f}] LP solve failed: {e}")
            break
        if not res.success:
            print(f"[{time.time():.1f}] LP not successful: {res.message}")
            break
        b = res.x
        s = b.sum()
        if s <= 1 + tol:
            # no improvement
            break
        a = b / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it+1}: sum(b)={s:.6f}, cheap={cur_score:.6f}")
    return best_a

Algorithm

   historic base size.  All candidates are normalised to sum = 1 and

   final objective for a normalised sequence).

3. **Heavy refinement** – for every seed run three first‑order
   optimiser schemes (exponentiated‑gradient, Adam and a smooth‑Adam

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using log‑sum‑exp:
    f_eps(a) = eps * log Σ_i exp(conv_i/eps).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]
    return grad


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eps0: float = 1e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999):
    """Adam optimiser on the log‑sum‑exp smoothing of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        if t % 50_000 == 0:
            eps = eps * 0.5 if eps > 1e-6 else eps
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.6):

    best_score = _max_convolution_score(a)

        it += 1

# Analytic families -------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    w[np.isnan(w)] = 0.0
    total = w.sum()
    if total == 0:
        return np.full_like(w, 1.0 / n)
    return w / total


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α (α=1 → uniform on simplex)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

def _lp_refine(a: np.ndarray, deadline: float,
               max_iter: int = 12, tol: float = 1e-6) -> np.ndarray:
    """
    Iterative LP refinement: maximise sum(b) subject to a⋆b ≤ M,
    b≥0.  After each solve we set a←b/∑b which reduces the maximal
    convolution.  Stops when no improvement or time runs out.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception as e:
        print(f"[{time.time():.1f}] LP refinement unavailable: {e}")
        return a

    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        M = conv.max()

        # Build dense constraint matrix (2n‑1) × n, then convert to CSR.
        k = np.arange(2 * n - 1)[:, None]
        j = np.arange(n)[None, :]
        diff = k - j
        mask = (diff >= 0) & (diff < n)
        A_dense = np.zeros((2 * n - 1, n), dtype=np.float64)
        A_dense[mask] = a[diff[mask]]
        A = csr_matrix(A_dense)

        c = -np.ones(n)                      # maximise sum(b)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs')
        except Exception as e:
            print(f"[{time.time():.1f}] LP solve failed: {e}")
            break

        if not res.success:
            print(f"[{time.time():.1f}] LP not successful: {res.message}")
            break

        b = res.x
        s = b.sum()
        if s <= 1 + tol:
            break          # no improvement possible

        a = b / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.1f}] LP iter {it+1}: sum(b)={s:.6f}, cheap={cur_score:.6f}")

    return best_a

                                       step0=0.2)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smooth‑Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient descent on the current best
    sub_deadline = deadline - 20.0          # keep time for later stages
    if time.time() < sub_deadline:
        seq, sc = _subgradient_descent(best_seq, sub_deadline,
                                       max_iters=800_000,
                                       eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Linear‑programming refinement
    lp_deadline = deadline - 5.0
    if time.time() < lp_deadline:
        seq = best_seq.copy()
        seq = _lp_refine(seq, lp_deadline,
                         max_iter=12, tol=1e-6)
        sc = _max_convolution_score(seq)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] LP refinement → {sc:.6f}")

    # ------------------------------------------------------------------
    # 7. Fine‑grained hill‑climbing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.03                       # up to 3 % of total mass

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        if iteration % 4000 == 0 and np.random.rand() < 0.08:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
