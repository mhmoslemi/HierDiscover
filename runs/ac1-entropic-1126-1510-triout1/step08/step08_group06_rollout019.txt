def _subgradient_descent(a, deadline, best_seq, best_score, lr0=0.5, max_iter=20000, eval_interval=200):
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        # conv and argmax
        b = _fft_self_convolve(a)
        idx = int(b.argmax())
        # gradient for max
        ids = idx - np.arange(n)
        mask = (ids >= 0) & (ids < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ids[mask]]
        # step size schedule (1/sqrt)
        lr = lr0 / math.sqrt(t)
        # exponentiated update
        a = a * np.exp(-lr * grad)
        # renormalize to simplex (simple scaling)
        sum_a = a.sum()
        if sum_a == 0.0:

        if t % eval_interval == 0:

                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

"""
Search for a non‑negative step function that minimises
``evaluate_sequence`` (i.e. 2·n·max(conv(a,a)) for a normalised vector).

Algorithm
---------

1. **Rich analytic seed pool** – several classic families (uniform,
   arcsine, symmetric/asymmetric Beta, power‑law, exponential, random
   sparse, two‑block spikes) are generated for a handful of lengths.
   Their exact objective is evaluated and the best few are retained.

2. **Exact sub‑gradient descent** – for a selected seed we repeatedly
   compute the *true* maximal convolution index, obtain the sub‑gradient
   (the reversed vector shifted to that index), and perform a mirror‑
   descent update ``a_i ← a_i·exp(-η·grad_i)`` followed by a simple
   normalisation to the simplex.  The step size η decays like 1/√t.
   Because the gradient is taken from the exact max (not a smooth
   approximation) the method drives the true objective down efficiently.

3. **Greedy pairwise swaps** – a cheap Metropolis‑type walk that moves a
   random amount of mass from a source index to a destination index.
   Every move is kept only if the *exact* objective improves, polishing
   the final vector.

All stages respect a global time budget (default 1000 s).  The best
vector found is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric beta (Beta(alpha,beta))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p , monotone increasing."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    """a_i ∝ exp(lam·i) , monotone increasing."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of seeds for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta
    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    # asymmetric beta – random sampling on a log‑scale
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)    # 0.01 … 10
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # monotone power‑law
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # monotone exponential
    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes
    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # sparse random start (~1 % non‑zeros)
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution objective ----------
def _subgradient_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         lr0: float = 0.5, max_iter: int = 20000,
                         eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true ``max(conv)`` objective.
    Uses the sub‑gradient coming from the current maximal convolution index.
    """
    n = a.shape[0]
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1

        # exact convolution and argmax
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # sub‑gradient: 2*a shifted by idx
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]

        # step size schedule η_t = lr0 / sqrt(t)
        eta = lr0 / math.sqrt(t)

        # mirror‑descent (exponentiated gradient) update
        a = a * np.exp(-eta * grad)

        # re‑normalise to the simplex (simple scaling – cheap & sufficient)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer (tiny simulated‑annealing step) -------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score
