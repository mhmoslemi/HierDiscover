"""
Hybrid global+local optimizer for the step‑function minimisation problem.

The pipeline follows three stages:

1️⃣ **Broad sampling** – many analytic families (arcsine, triangular,
   Gaussian, raised‑cosine, beta, power‑law, exponential) and a variety
   of random constructions (Dirichlet draws, sparse spikes, Sidon‑set
   seeds, Gaussian mixtures) are generated for several lengths.
   Each candidate is ℓ₁‑normalised (∑a = 1) and scored with the cheap
   surrogate 2·n·max(conv(a,a)), which equals ``evaluate_sequence`` for a
   normalised sequence.

2️⃣ **Intensive refinement** – the top‑k seeds are optimised on the
   simplex by two complementary first‑order methods:
   * Adam‑style adaptive‑moment descent,
   * Exponentiated‑gradient (KL‑mirror) descent.
   Both use the exact sub‑gradient of the max‑convolution (computed
   via FFT).  The best point ever seen is retained.

3️⃣ **Local polishing** – a Frank‑Wolfe style mass‑transfer loop moves
   probability mass from the component with largest sub‑gradient to the
   component with smallest one, followed by a simulated‑annealing jitter
   phase that performs random transfers with Metropolis acceptance.

All loops respect a global 1000 s deadline.  The function returns a
Python list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate: 2·n·max(conv(a,a)) for a ∈ simplex (∑a = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 300_000):
    """Adam‑style descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 250_000):
    """Exponentiated‑gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _raised_cosine_distribution(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / ((n - 1) / 2.0))) ** power
    vals[~np.isfinite(vals)] = 0.0
    vals /= vals.sum()
    return vals


def _sparse_random_distribution(n: int, k: int) -> np.ndarray:
    """k randomly placed spikes (k ≤ n)."""
    if k >= n:
        return _random_dirichlet_alpha(n, 1.0)
    idx = np.random.choice(n, size=k, replace=False)
    weights = np.random.rand(k) + 1e-12
    a = np.zeros(n, dtype=np.float64)
    a[idx] = weights
    a /= a.sum()
    return a


def _sidon_seed(n: int) -> np.ndarray:
    """Construct a rough Sidon‑set based sparse seed."""
    # pick a prime p ≈ √n
    p = int(np.sqrt(n))
    # simple primality test (fast for tiny numbers)
    def is_prime(x):
        if x < 2:
            return False
        for d in range(2, int(np.sqrt(x)) + 1):
            if x % d == 0:
                return False
        return True
    while not is_prime(p):
        p += 1
    positions = []
    for i in range(p):
        pos = i * p + (i * i) % p
        if pos < n:
            positions.append(pos)
    if not positions:
        return _random_dirichlet_alpha(n, 0.2)
    weights = np.random.rand(len(positions)) + 1e-12
    a = np.zeros(n, dtype=np.float64)
    a[positions] = weights
    a /= a.sum()
    return a


def _gaussian_mixture_distribution(n: int, comps: int = 3) -> np.ndarray:
    """Random mixture of Gaussians."""
    i = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    comp_weights = np.random.rand(comps)
    comp_weights /= comp_weights.sum()
    for w in comp_weights:
        centre = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(0.02 * n, 0.15 * n)
        a += w * np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a += 1e-12
    a /= a.sum()
    return a

            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} EG improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Frank‑Wolfe style mass‑transfer polishing
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 70_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # move mass away from large grad
        i_dst = int(np.argmin(grad))   # move mass toward small grad
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # Try full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-13:
            a, cur_score = a_full, score_full
            improved = True
        else:
            # Try half transfer
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-13:
                a, cur_score = a_half, score_half
                improved = True

        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional random jitter to shake out plateaus
        if fw_iter % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 7️⃣ Simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.025
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 110_000:
        anneal_iter += 1
        # linear temperature schedule based on remaining fraction
        t_frac = (deadline - time.time()) / (budget_s - 0.5)
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a, cur_score = a_new, new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a, cur_score = a_new, new_score

    # ------------------------------------------------------------------
    # 8️⃣ Final quick Adam run from the best point (fine‑tuning)
    if time.time() < deadline:
        seq_final, sc_final = _adam_descent(best_seq, deadline, max_iters=120_000)
        if sc_final < best_score - 1e-14:
            best_score, best_seq = sc_final, seq_final.copy()
            print(f"[{time.time():.1f}] Final Adam fine‑tune improved → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9️⃣ Report & return
    print(f"Final surrogate score: {best_score:.6f} (length={len(best_seq)})")
    try:
        # sanity check with the official evaluator (scale‑invariant)
        real_score = evaluate_sequence(best_seq.tolist())
        print(f"evaluate_sequence reports: {real_score:.6f}")
    except Exception:
        pass
    return best_seq.tolist()
