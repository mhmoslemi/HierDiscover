"""
PROPOSE_CANDIDATE (v2)

This implementation improves on the previous version by
1. Exhaustively scanning symmetric Beta distributions (a continuous
   family that includes the arcsine law) with a fine grid of parameters.
2. Adding a cheap local “parameter hill‑climb” that perturbs the
   distribution parameters (Beta shape or mixture weights) and accepts
   any move that reduces the cheap surrogate 2·n·max(a*a).
3. Refining the best seed with a continuation‑smooth gradient descent
   (soft‑max of the convolution) and a final stochastic hill‑climb.
4. An optional linear‑programming “tighten‑step’’ that tries to
   increase the total mass under the current max‑convolution bound,
   rescales the solution to the simplex and mixes it back with the
   current iterate.  The LP uses `scipy.optimize.linprog` with the
   HiGHS solver and a sparse Toeplitz convolution matrix.

All steps obey the global time budget (default 1000 s).  The returned
list is already normalised (its sum is 1), but the objective is
scale‑invariant, so any positive scaling would give the same score.
"""

import time
import numpy as np
from scipy import sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of `max(a*a)` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    # soft‑max weights, stabilised by subtracting the maximum
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation method: start with smooth max (γ≈5) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score

def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on {0,…,n‑1}, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _uniform_distribution(n: int) -> np.ndarray:
    """Flat distribution on length n."""
    return np.full(n, 1.0 / n)

def _hill_climb(a: np.ndarray, best_score: float, deadline: float) -> tuple:
    """Tiny random mass‑transfer hill‑climb."""
    n = a.shape[0]
    cur_score = best_score
    it = 0
    max_transfer = 0.03  # start with up to 3 % of total mass
    while time.time() < deadline:
        it += 1
        transfer = max_transfer * (0.5 ** (it / 12_000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 5_000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a)
            cur_score = _max_convolution_score(a)

    return a, best_score

def _lp_tighten_step(f: np.ndarray, deadline: float) -> tuple:
    """
    Perform one LP‑based tightening step as described in the paper:
    given f (sum = 1) compute M = max(f*f), solve
        maximise 1·g   s.t.   conv(f,g) ≤ M , g ≥ 0 .
    If the LP yields a larger total mass we rescale g to sum 1,
    mix it back with f (t = 0.5) and return the improved vector.
    """
    n = f.shape[0]
    L = 2 * n - 1
    # Build sparse Toeplitz matrix C_f with entries C_f[k,i] = f[k-i] (if 0≤k-i<n)
    diags = []
    offs = []
    for d in range(n):
        length = min(L - d, n)
        diags.append(np.full(length, f[d]))
        offs.append(d)
    C_f = sp.diags(diags, offs, shape=(L, n), format='csc')

    M = _fft_self_convolve(f).max()
    # Linear program: maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=C_f, b_ub=np.full(L, M), bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if not res.success:
        return f, _max_convolution_score(f)

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1.0 + 1e-12:
        # No improvement possible
        return f, _max_convolution_score(f)

    # Rescale to the simplex
    g_norm = g / sum_g

    # Try a few mixing fractions and keep the best
    best_local = f
    best_score = _max_convolution_score(f)
    for t in (0.2, 0.4, 0.6, 0.8):
        h = (1 - t) * f + t * g_norm
        h = _project_onto_simplex(h)
        sc = _max_convolution_score(h)
        if sc < best_score:
            best_local, best_score = h, sc
    return best_local, best_score
