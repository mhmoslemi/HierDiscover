"""
Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – start from the previously found best sequence
   (global ``height_sequence_1``) if it exists, otherwise from an
   arcsine distribution.  A few alternative lengths are examined
   (±20 % of the base length) and the cheapest start is kept.

2. **Heavy first‑order optimisation** – the chosen seed is refined by a
   cascade of projected first‑order methods that all respect the simplex
   constraint:
        * exponentiated‑gradient descent,
        * Adam,
        * continuation with a soft‑max (smooth‑max) of the convolution,
        * plain sub‑gradient descent.
   Each method runs until the global deadline is reached or a large
   iteration limit is exhausted.

3. **Gradient‑guided hill‑climbing** – after the heavy passes we repeatedly
   move a small amount of mass from the index with the *largest*
   sub‑gradient (i.e. most harmful for the max‑convolution) to the index
   with the *smallest* sub‑gradient.  The transferred amount decays
   geometrically, giving fine‑grained local improvement.  Occasionally
   a tiny random jitter is added to escape plateaus.

All steps are timed so that the function always returns before the
``budget_s`` limit (the last few seconds are kept as a safety margin).
The returned list is a valid input for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of `max(a*a)` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.9):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 400_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.002):
    """Continuation method: smooth max → hard max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        # ----- soft‑max gradient ------------------------------------
        n = a.shape[0]
        b = _fft_self_convolve(a)
        b_max = b.max()
        w = np.exp(gamma * (b - b_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # diagonal correction (k = 2*i)
        diag_w = w[2 * np.arange(n)]
        grad -= a * diag_w
        # ----- update ---------------------------------------------
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step‑size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _arc_distribution(n: int) -> np.ndarray:
    """Arcsine (Beta(½,½)) distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals = np.where(np.isfinite(vals), vals, 0.0)
    vals /= vals.sum()
    return vals


def _interpolate(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    vals = np.interp(x_new, x_old, old)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0:
        vals = np.full(new_n, 1.0 / new_n)
    else:
        vals /= s
    return vals

        best_seq, best_score = seq2, sc2
        print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

    # 3. smooth‑max continuation
    seq3, sc3 = _smooth_gradient_descent(best_seq, deadline,
                                         max_iters=300_000,
                                         eta0=0.7, gamma0=5.0,
                                         gamma_factor=1.003)
    if sc3 < best_score - 1e-14:
        best_seq, best_score = seq3, sc3
        print(f"[{time.time():.1f}] smooth‑max improved to {best_score:.6f}")

    # 4. plain sub‑gradient (tidy‑up)
    seq4, sc4 = _subgradient_descent(best_seq, deadline,
                                     max_iters=400_000, eta0=0.5)
    if sc4 < best_score - 1e-14:
        best_seq, best_score = seq4, sc4
        print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _gradient_guided_hill_climb(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Fine‑grained hill‑climbing informed by the sub‑gradient.
    Mass is transferred from the index with the largest gradient to the
    index with the smallest gradient; the transferred amount decays
    geometrically.
    """
    a = seq.copy()
    cur_score = _max_convolution_score(a)
    best_seq = a.copy()
    best_score = cur_score
    n = a.shape[0]

    it = 0
    max_transfer = 0.05          # start with 5 % of total mass (absolute)
    while time.time() < deadline:
        it += 1
        # decay of the allowed transfer (smoothly)
        transfer = max_transfer * (0.5 ** (it / 8000.0))

        grad, _, _ = _subgrad(a)
        i = int(np.argmax(grad))   # most harmful
        j = int(np.argmin(grad))   # most helpful
        if i == j:
            continue

        delta = min(transfer, a[i])
        # add a tiny random factor to avoid deterministic stagnation
        delta *= 0.5 + 0.5 * np.random.rand()
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq, best_score = a.copy(), new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 3e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the time budget (default 1000 s) and returns the
    best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 3.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Determine a base length from the previously best sequence (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base_n = prev.shape[0]
        prev = prev / prev.sum()
    except Exception:
        prev = None
        base_n = 3400                     # a reasonable default

    # ------------------------------------------------------------------
    # 2. Choose a handful of lengths to explore
    length_factors = [0.85, 1.0, 1.15]
    lengths = sorted({max(500, int(base_n * f)) for f in length_factors})

    best_overall_seq = None
    best_overall_score = np.inf

    # ------------------------------------------------------------------
    # 3. For each length: build a seed (interpolated prev or arcsine) and
    #    run the heavy optimisation pipeline.
    for n in lengths:
        if time.time() >= deadline:
            break

        if prev is not None:
            seed_seq = _interpolate(prev, n)
        else:
            seed_seq = _arc_distribution(n)

        # normalise explicitly (project onto simplex)
        seed_seq = _project_onto_simplex(seed_seq, s=1.0)

        # heavy optimisation
        opt_seq, opt_score = _run_heavy_optim(seed_seq, deadline)

        print(f"[{time.time():.1f}] length={n} heavy‑opt score={opt_score:.6f}")

        if opt_score < best_overall_score - 1e-14:
            best_overall_seq = opt_seq.copy()
            best_overall_score = opt_score

    # ------------------------------------------------------------------
    # 4. If we never obtained a sequence (very unlikely) fall back to arcsine
    if best_overall_seq is None:
        best_overall_seq = _arc_distribution(base_n)
        best_overall_seq = _project_onto_simplex(best_overall_seq, s=1.0)
        best_overall_score = _max_convolution_score(best_overall_seq)

    # ------------------------------------------------------------------
    # 5. Final gradient‑guided hill‑climbing to polish the solution
    final_seq = _gradient_guided_hill_climb(best_overall_seq, deadline)
    final_score = _max_convolution_score(final_seq)

    print(f"[{time.time():.1f}] final cheap score: {final_score:.6f} (len={len(final_seq)})")
    # Return as plain Python list
    return final_seq.tolist()
