# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Adapted from Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _cheap_score(a: np.ndarray) -> float:
    """Cheap surrogate: 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of `max(a*a)` on the simplex.
    Returns (grad, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient = 2 * conv_rev(a[::-1], w) restricted to the original length
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms where k = 2*i
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """
    Gradient of the soft‑max approximation of max(a*a).
    `γ` controls smoothness (larger → closer to the true max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    # Soft‑max weights (stable)
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation – start smooth and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _cheap_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _cheap_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 150_000,
                         eta0: float = 0.9):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _cheap_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        # clip to avoid overflow in exp
        grad_clip = np.clip(grad, -50, 50)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad_clip)
        total = a.sum()
        if total > 0:
            a = a / total
        else:   # should never happen
            a = np.full_like(a, 1.0 / a.size)
        cur = _cheap_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 0.6):
    """Plain projected sub‑gradient descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _cheap_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _cheap_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Sample a uniform point on the simplex (α=1 Dirichlet)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _powerlaw_distribution(n: int, p: float = 1.0) -> np.ndarray:
    """Symmetric power‑law: w_i ∝ (i+0.5)^(-p) + (n-i-0.5)^(-p)"""
    idx = np.arange(n, dtype=np.float64) + 0.5
    left = idx ** (-p)
    right = left[::-1]
    vals = left + right
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()

def _simulated_annealing(a: np.ndarray, deadline: float):
    """
    Random mass‑transfer hill‑climb with simulated‑annealing acceptance.
    The total mass is conserved, hence the sum stays 1.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _cheap_score(a)
    cur = a.copy()
    cur_score = best_score
    start = time.time()
    # Initial temperature based on typical score fluctuations
    T0 = 0.02
    while time.time() < deadline:
        t = (time.time() - start) / (deadline - start + 1e-12)
        T = T0 * (1.0 - t)  # linear cooling
        # propose a transfer
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        max_delta = cur[i] * 0.12  # up to 12 % of source mass
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # occasionally renormalise to avoid drift
        if abs(new_seq.sum() - 1.0) > 1e-12:
            new_seq = new_seq / new_seq.sum()
        new_score = _cheap_score(new_seq)
        if new_score < cur_score:
            cur, cur_score = new_seq, new_score
            if new_score < best_score:
                best, best_score = new_seq.copy(), new_score
                print(f"[{time.time():.1f}] anneal new best {best_score:.6f}")
        else:
            # uphill acceptance
            prob = math.exp((cur_score - new_score) / (T + 1e-12))
            if np.random.rand() < prob:
                cur, cur_score = new_seq, new_score
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point for the optimisation challenge.
    Returns a list of non‑negative heights (sum ≈ 1) that minimises
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Base length – try to use the length of the previously best sequence
    try:

    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # Choose a diverse set of lengths to explore

    for f in (0.8, 1.0, 1.2, 1.5, 2.0, 2.5):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    max_len = min(15000, base_n * 4)
    for _ in range(6):
        n = np.random.randint(500, max_len + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1. Broad sampling
    candidates = []   # list of (array, cheap_score)

    beta_params = [0.3, 0.6, 0.9, 1.2, 2.0, 3.0, 5.0]
    power_ps = [0.8, 1.0, 1.2, 1.5, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Analytic families
        for gen, tag in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.003), "exp_small"),
            (lambda m: _exponential_distribution(m, lam=0.02), "exp_big"),
            (lambda m: np.full(m, 1.0 / m), "uniform"),
        ):
            a = gen(n)
            candidates.append((a, _cheap_score(a)))

        # Power‑law family
        for p in power_ps:
            a = _powerlaw_distribution(n, p)
            candidates.append((a, _cheap_score(a)))

        # Symmetric Beta
        for p in beta_params:
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _cheap_score(a)))

        # Random Dirichlet draws
        for _ in range(3):
            a = _random_dirichlet(n)
            candidates.append((a, _cheap_score(a)))

        # Random mixtures of two families
        for _ in range(2):
            fam1 = np.random.choice(['arcsine', 'triangular', 'uniform', 'exp_small'])
            fam2 = np.random.choice(['beta_sym', 'powerlaw', 'exp_big'])
            if fam1 == 'arcsine':
                a1 = _arcsine_distribution(n)
            elif fam1 == 'triangular':
                a1 = _triangular_distribution(n)
            elif fam1 == 'uniform':
                a1 = np.full(n, 1.0 / n)
            else:  # exp_small
                a1 = _exponential_distribution(n, lam=0.003)

            if fam2 == 'beta_sym':
                p = np.random.choice(beta_params)
                a2 = _beta_symmetric_distribution(n, p)
            elif fam2 == 'powerlaw':
                p = np.random.choice(power_ps)
                a2 = _powerlaw_distribution(n, p)
            else:  # exp_big
                lam = 10 ** np.random.uniform(-4, -2)
                a2 = _exponential_distribution(n, lam=lam)

            mix = np.random.rand()
            a = mix * a1 + (1.0 - mix) * a2
            a /= a.sum()
            candidates.append((a, _cheap_score(a)))

    # ------------------------------------------------------------------
    # Include previously found best sequence, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _cheap_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Keep the top‑k seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = [seq for seq, _ in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _cheap_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Heavy optimisation on each seed (time‑sharing)
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        return best_seq.tolist()
    per_seed_budget = remaining_time * 0.65 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        seed_deadline = time.time() + per_seed_budget

        # 3a) Smooth‑max continuation
        seq, sc = _smooth_gradient_descent(seed_seq, seed_deadline,
                                           max_iters=120_000,
                                           eta0=0.7,
                                           gamma0=5.0,
                                           gamma_factor=1.0015)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} smooth → {sc:.6f}")

        # 3b) Exponentiated gradient descent
        seq, sc = _exp_gradient_descent(seed_seq, seed_deadline,
                                        max_iters=80_000,
                                        eta0=0.9)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} exp‑grad → {sc:.6f}")

        # 3c) Sub‑gradient tidy‑up
        seq, sc = _subgradient_descent(seed_seq, seed_deadline,
                                       max_iters=150_000,
                                       eta0=0.6)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} subgrad → {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Global simulated‑annealing hill‑climb with the remaining budget
    if time.time() < deadline:
        seq, sc = _simulated_annealing(best_seq, deadline)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] final anneal best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanitisation – ensure non‑negative and normalised
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total == 0:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / total

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
