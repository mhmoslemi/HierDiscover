def delta_effect(a, src, dest):
    n = len(a)
    size = 2*n-1
    delta = np.zeros(size, dtype=np.float64)
    # contributions from dest
    idx = np.arange(dest, dest+n)
    delta[idx] = 2*a[idx-dest]
    # contributions from src (subtract)
    idx2 = np.arange(src, src+n)
    delta[idx2] -= 2*a[idx2-src]
    return delta
```

"""
Improved search for the minimum of
    2·n·max(conv(a,a))   (with sum(a)=1, a_i ≥ 0)

The algorithm

* builds a large, diverse pool of analytic and random initial vectors;
* keeps the best few candidates;
* refines each of them with an **enhanced greedy active‑set** step that
  evaluates many (source,destination) transfers per iteration and performs
  a short line‑search on the transfer amount; the convolution is updated
  cheaply using the known linear effect of a mass move;
* finishes with a longer simulated‑annealing walk;
* respects the 1000 s budget and returns the best vector found.

All helpers are top‑level; the final vector is projected onto the simplex
so that it can be safely passed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Assumes sum(a)=1. Returns 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# -------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_powerlaw(n: int, gamma: float) -> np.ndarray:
    """U‑shaped power‑law: w_i ∝ i^{‑γ} + (n‑i+1)^{‑γ}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-gamma) + (n - i + 1) ** (-gamma)
    w /= w.sum()
    return w


# -------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse set of start vectors."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric beta – very small shape parameters as well
    for p in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02,
              0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # Power‑law shapes
    for gamma in (0.3, 0.5, 0.8, 1.0, 1.5):
        cand.append(_distribution_powerlaw(n, gamma))

    # Exponentials
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # Alternating patterns
    for stride in (2, 3, 4, 5, 6):
        cand.append(_distribution_alternating(n, stride))

    # Random Dirichlet draws with various concentrations
    for conc in (0.001, 0.01, 0.05, 0.2, 0.5, 1.0, 5.0):
        a = rng.gamma(conc, size=n)
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a.sum()
        cand.append(a)

    # Sparse random start (~0.5 % non‑zeros)
    k = max(2, n // 200)
    sparse = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    sparse[idx] = rng.random(k)
    if sparse.sum() == 0:
        sparse[0] = 1.0
    sparse /= sparse.sum()
    cand.append(sparse)

    # Noisy perturbations of a few beta shapes
    for _ in range(4):
        p = rng.choice([0.001, 0.003, 0.005, 0.01, 0.05, 0.1])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert)
        cand.append(pert)

    return cand


# -------------------------------------------------------------
def _greedy_optimize(a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive active‑set improvement.
    At each iteration a small set of (src,dest) pairs is examined;
    for each pair a few transfer amounts (full, ½, ⅕, ⅕) are tested using
    the exact linear effect on the convolution.  The best improving move
    is applied.
    """
    a = a.astype(np.float64)
    a = _project_onto_simplex(a)
    n = len(a)
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()
    best_vec = a.copy()

    while time.time() < deadline:
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient of max(b) at active shifts
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            g[i] += 2.0 * a[k - i]

        src_mask = a > 1e-12
        if not np.any(src_mask):
            break
        src_indices = np.where(src_mask)[0]

        # top‑K source indices by gradient (restricted to those with mass)
        K_src = min(5, src_indices.size)
        top_src = src_indices[np.argpartition(-g[src_indices], K_src - 1)[:K_src]]

        # K destination indices with smallest gradient (any index allowed)
        K_dest = min(5, n)
        dest_candidates = np.argpartition(g, K_dest - 1)[:K_dest]

        improved = False

        for src in top_src:
            max_delta = a[src]                     # we may move at most this much
            if max_delta < 1e-12:
                continue
            for dest in dest_candidates:
                if src == dest:
                    continue

                # linear effect of moving mass from src to dest
                delta_arr = np.zeros_like(b)
                delta_arr[dest:dest + n] += 2.0 * a      # +2*a[k‑dest]
                delta_arr[src:src + n] -= 2.0 * a        # –2*a[k‑src]

                # try a few transfer amounts
                for alpha in (max_delta,
                              max_delta * 0.5,
                              max_delta * 0.2,
                              max_delta * 0.1,
                              max_delta * 0.02):
                    new_max = np.max(b + alpha * delta_arr)
                    new_score = 2.0 * n * new_max
                    if new_score < best_score - 1e-14:
                        # Apply the move
                        a[src] -= alpha
                        a[dest] += alpha
                        a = _project_onto_simplex(a)   # clean up tiny drift
                        b = _fft_self_convolve(a)      # refresh convolution
                        best_score = new_score
                        best_vec = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # occasional jitter to escape plateaus
            if rng.random() < 0.02:
                i = rng.integers(n)
                if a[i] > 1e-12:
                    j = rng.integers(n - 1)
                    if j >= i:
                        j += 1
                    eps = min(a[i] * 0.05, 0.001)
                    a[i] -= eps
                    a[j] += eps
                    a = _project_onto_simplex(a)
                    b = _fft_self_convolve(a)
                    cur_score = 2.0 * n * b.max()
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_vec = a.copy()
                continue
            # no improving move found → stop
            break

    return best_vec, best_score


# -------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk with a slow exponential cooling schedule."""
    n = len(a)
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.05   # start temperature (objective units)
    while time.time() < deadline and T > 1e-6:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9999   # gentle cooling

    return best_a, best_score


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (probability vector)
    that minimises ``evaluate_sequence``.
    Returns the best vector found before the deadline.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # length – a few thousand gives a good trade‑off between expressiveness and speed
    n = 5000
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 1) generate a broad catalogue of start vectors
    candidates = _initial_candidates(n, rng)

    # include the previous best solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size == n:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) select a few promising seeds (lowest objective)
    scored = []
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        scored.append((sc, cand))
        print(f"[{time.time():.1f}] init cand {idx:3d} score={sc:.6f}")

    if not scored:
        # safety fallback – uniform distribution
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
    else:
        scored.sort(key=lambda x: x[0])
        top_k = min(5, len(scored))
        best_vec, best_score = scored[0][1].copy(), scored[0][0]
        seeds = [vec for _, vec in scored[:top_k]]

    # ------------------------------------------------------------------
    # 3) optimise each seed, sharing the remaining budget
    global_best = best_vec.copy()
    global_best_score = best_score

    for s_idx, seed_vec in enumerate(seeds):
        if time.time() >= deadline:
            break
        # Allocate a fair share of the remaining time to this seed
        remaining = deadline - time.time()
        seeds_left = len(seeds) - s_idx
        seed_deadline = time.time() + remaining / seeds_left

        print(f"[{time.time():.1f}] Optimising seed {s_idx} (budget ≈ {seed_deadline - time.time():.1f}s)")

        # Greedy active‑set refinement
        vec, sc = _greedy_optimize(seed_vec.copy(), seed_deadline, rng)

        # Simulated‑annealing fine‑tuning (if time permits)
        if time.time() < seed_deadline:
            vec, sc = _simulated_annealing(vec, seed_deadline, rng)

        if sc < global_best_score - 1e-12:
            global_best_score = sc
            global_best = vec.copy()
            print(f"[{time.time():.1f}] New global best {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity – ensure we are on the simplex
    global_best = _project_onto_simplex(global_best, 1.0)
    final_score = _objective(global_best)
    print(f"Final best score: {final_score:.6f} (length={len(global_best)})")
    return global_best.tolist()
