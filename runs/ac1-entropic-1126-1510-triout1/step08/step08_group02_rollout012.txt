# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a) for a on the simplex (sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max:
        f_ε(a) = eps * log Σ_i exp((a*a)_i / eps)
    Returns ∇_a f_ε.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    # convolution of a[::-1] with the weight vector w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (each a_i appears twice in the full sum)
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

# L‑BFGS optimisation on the smooth max (homotopy) ----------------------
def _softmax(u: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a vector on the simplex)."""
    mx = np.max(u)
    e = np.exp(u - mx)
    s = e.sum()
    return e / s


def _lbfgs_obj(u: np.ndarray, eps: float):
    """
    Returns (f, g) where

        f = eps·log Σ exp((a*a)_k/eps)
        g = ∇_u f      (u → a = softmax(u))
    """
    a = _softmax(u)                # a lies on the simplex
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    f = maxc + eps * np.log(w.sum())

    grad_a = _smooth_grad(a, eps)      # ∇_a f
    dot = np.dot(grad_a, a)
    grad_u = a * (grad_a - dot)        # chain rule through softmax
    return f, grad_u


def _optimize_smooth(a0: np.ndarray, deadline: float):
    """
    Homotopic optimisation: start from a0, run a sequence of L‑BFGS
    minimisations for decreasing values of ε.
    Returns the best sequence and its cheap score.
    """
    try:
        from scipy.optimize import minimize
    except Exception as e:
        print("SciPy unavailable – falling back to previous routine.")
        return a0, _max_convolution_score(a0)

    u = np.log(np.maximum(a0, 1e-16))
    n = a0.shape[0]
    bounds = [(-30.0, 30.0)] * n

    best_a = a0.copy()
    best_score = _max_convolution_score(best_a)

    # ε schedule – from very smooth to almost the true max.
    eps_schedule = [
        0.5, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4,
        1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6
    ]

    for eps in eps_schedule:
        if time.time() >= deadline:
            break

        # L‑BFGS – we limit the number of iterations per ε to keep the total run‑time low.
        options = {"maxiter": 140, "disp": False, "gtol": 1e-12}
        res = minimize(
            _lbfgs_obj,
            u,
            args=(eps,),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
        # Re‑build a from the optimiser output and enforce exact symmetry.
        a = _softmax(res.x)
        a = (a + a[::-1]) / 2.0          # keep the sequence symmetric
        a /= a.sum()
        cur_score = _max_convolution_score(a)

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[{time.time():.1f}] eps={eps:.1e} → new best cheap score {best_score:.6f}")

        # Use this a as the start for the next (tighter) ε.
        u = np.log(np.maximum(a, 1e-16))

    return best_a, best_score

# Very long projected sub‑gradient descent (fine polishing) ---------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_000_000,
                         eta0: float = 0.6):
    """Projected sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

        iteration += 1
        # Geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 8000.0))

        # Choose donor i (large positive gradient) and receiver j (most negative)
        if iteration % 9 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to shake out tiny plateaus
        if iteration % 6000 == 0 and np.random.rand() < 0.13:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
