def anneal(a, total_time_allowed):
    start = time.time()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_score = 2*n*conv.max()
    best_a = a.copy()
    best_score = cur_score
    T0 = cur_score * 0.05 # maybe 5% of score
    Tmin = cur_score * 1e-5
    while time.time() - start < total_time_allowed:
        # schedule
        elapsed = time.time() - start
        frac = elapsed / total_time_allowed
        T = T0 * (Tmin / T0) ** frac

        # propose move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j: continue
        # delta draws from small proportion of a[i]
        max_delta = min(a[i] * 0.5, 0.05) # adjust
        delta = max_delta * np.random.rand()
        if delta < 1e-12: continue

        a_new = a.copy()
        conv_new = conv.copy()
        _apply_transfer(a_new, conv_new, i, j, delta)
        new_score = 2*n*conv_new.max()
        # acceptance
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score) / T):
            a = a_new
            conv = conv_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # maybe occasional reinject jitter
```

"""
Hybrid search for a step‑function with very small max‑self‑convolution.

The algorithm works in three phases:

1. **Broad initialization** – generate many candidate vectors of several
   lengths using analytic families (uniform, arcsine, triangular, exponential,
   β‑distribution, sine‑power, cosine‑windows, Gaussian) and random Dirichlet
   draws.  Each candidate is normalised to L1‑norm = 1 and evaluated by the fast
   FFT based “max‑convolution” score 2·n·max(a*a).  The best few candidates are
   kept.

2. **Simulated‑annealing hill‑climbing** – for every retained candidate we run
   a temperature‑controlled random walk that repeatedly transfers a small mass
   `δ` from one index to another, updating the full self‑convolution in O(n)
   time (via the incremental routine `_apply_transfer`).  Moves that decrease
   the maximal convolution are always accepted; worsening moves are accepted
   with probability `exp(-(Δscore)/T)`.  The temperature T is annealed from a
   relatively high value to near‑zero while the step size shrinks, allowing the
   search to escape local minima and then converge.

3. **Final polishing** – after the time budget is exhausted the best sequence
   found is returned as a plain Python list of floats.  The implementation
   respects the 1000 s wall‑clock limit, avoids filesystem/network I/O and
   contains no closures or lambdas.

The returned vector works directly with the supplied `evaluate_sequence`
function (its L1‑norm is 1, all entries are non‑negative and far below the
clipping threshold 1000).
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Score for a (already normalised to sum = 1): 2·n·max(a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (return vectors with L1=1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to j, updating both `a`
    and the full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

def _random_dirichlet(n, alpha):
    """Draw a Dirichlet vector with concentration `alpha`."""
    vec = np.random.dirichlet(np.full(n, alpha))
    # guard against pathological near‑zero values
    eps = 1e-12
    vec = np.maximum(vec, eps)
    vec /= vec.sum()
    return vec

def _anneal_hill_climb(a, time_budget):
    """
    Simulated‑annealing hill‑climber that works directly on the convolution.
    `a` must already be normalised to sum = 1.  Returns the best vector found
    and its score (2·n·max(b)).
    """
    eps = 1e-12
    n = a.shape[0]
    a = np.maximum(a, eps)
    a /= a.sum()

    conv = _fft_self_convolve(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    start = time.time()
    # temperature schedule: from 5 % of the current score down to 0.001 %
    T_start = cur_score * 0.05
    T_end = cur_score * 1e-5

    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget

        # linear temperature decay
        T = T_start * (1 - frac) + T_end * frac

        # pick a source index (must have some mass) and a target index
        i = np.random.randint(n)
        if a[i] < eps:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # step size: large at the beginning, shrinking later
        max_delta = min(a[i] * 0.5, 0.02) * (1 - frac) + a[i] * 0.005 * frac
        delta = max_delta * np.random.rand()
        if delta < eps:
            continue

        # apply the transfer (in‑place, O(n))
        _apply_transfer(a, conv, i, j, delta)
        new_score = 2.0 * n * conv.max()

        # Metropolis acceptance
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            prob = math.exp((cur_score - new_score) / (T + 1e-30))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert the move
            _apply_transfer(a, conv, j, i, delta)

        # occasional tiny jitter to shake the system out of flat plateaus
        if random.random() < 0.001:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            cur_score = 2.0 * n * conv.max()

    return best_a, best_score

        # analytic families
        cand.append(_distribution_uniform(n))
        cand.append(_distribution_arcsine(n))
        cand.append(_distribution_triangular(n))
        cand.append(_distribution_exponential(n, lam=0.002))
        cand.append(_distribution_exponential(n, lam=0.02))

        for p in (0.2, 0.4, 0.8, 1.5, 3.0):
            cand.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            cand.append(_distribution_sine_power(n, pw))

        for coeff in (0.5, 0.54, 0.42):  # Hann, Hamming, Blackman‑like
            cand.append(_distribution_cosine_window(n, coeff))

        for sigma in (0.15, 0.25, 0.35):
            cand.append(_distribution_gaussian(n, sigma))

        # random Dirichlet draws with various concentrations
        for alpha in (0.1, 0.3, 0.7, 1.0, 2.0):
            cand.append(_random_dirichlet(n, alpha))

        # very sparse start (few spikes)
        for k in (2, 4, 8):
            s = np.zeros(n, dtype=np.float64)
            idx = np.random.choice(n, k, replace=False)
            s[idx] = np.random.rand(k) + 0.1
            s /= s.sum()
            cand.append(s)

        # ----- evaluate cheap candidates ------------------------------------
        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            # Keep a sliding list of the ~10 best for later refinement
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] n={n} new global best {sc:.6f}")

            # store locally if among top candidates
            if len(top_candidates) < 12:
                top_candidates.append((sc, vec.copy()))
                top_candidates.sort(key=lambda x: x[0])
            else:
                if sc < top_candidates[-1][0]:
                    top_candidates[-1] = (sc, vec.copy())
                    top_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Include a previously discovered sequence if present.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:12]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. If nothing survived, fall back to a uniform vector.
    if best_seq is None:
        n_fallback = 4096
        best_seq = _distribution_uniform(n_fallback)
        best_seq = best_seq / best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine the retained top candidates with simulated‑annealing hill‑climb.
    remaining = deadline - time.time()
    if remaining > 5.0 and len(top_candidates) > 0:
        # allocate equal time slots
        per_candidate = max(5.0, remaining / len(top_candidates))
        for idx, (init_sc, vec) in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] annealing candidate {idx+1}/{len(top_candidates)} "
                  f"(length={vec.size})")
            a_opt, sc_opt = _anneal_hill_climb(vec, per_candidate)
            if sc_opt < best_score:
                best_score = sc_opt
                best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] new global best after anneal {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity check with the official evaluator (optional, cheap)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] final evaluate_sequence value = {final_val:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Return as a plain Python list of floats (values already non‑negative).
    return best_seq.tolist()
