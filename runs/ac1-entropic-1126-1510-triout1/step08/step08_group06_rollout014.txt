def _block_average(a, K):
    n = len(a)
    block_size = n // K
    residual = n % K
    # For simplicity common block sizes: first residual blocks get size+1 else size.
    start = 0
    for b in range(K):
        size = block_size + (1 if b < residual else 0)
        end = start + size
        mean = a[start:end].mean()
        a[start:end] = mean
        start = end
    return a

"""
Propose a good non‑negative step‑function sequence.

The algorithm consists of three stages:

1. **Rich seeding** – for several lengths (up to 30 000) a large
   collection of analytic families is generated (uniform, power‑law
   edges, symmetric Beta, exponential, triangular, half‑sine,
   semicircle, random block‑constant, sparse random).  Their exact
   objective 2·n·max(conv(a,a)) (with sum(a)=1) is evaluated and the
   best few are kept.

2. **Projected smoothed‑Adam optimisation** – each top seed is refined
   by Adam on a log‑sum‑exp smooth approximation of the max‑convolution.
   After every Adam update the vector is projected onto the probability
   simplex and, optionally, averaged inside a small number of equal‑size
   blocks (piecewise‑constant constraint).  The smoothing parameter α
   is increased along a schedule; this yields a fast descent on the
   true non‑smooth objective.

3. **Exact greedy local search** – a Metropolis‑type walk that repeatedly
   moves a random amount of mass from a source index to a destination
   index, keeping the move only if the *exact* objective improves.
   This finishes the optimisation.

All stages respect a global time budget (default 1000 s).  The best
vector found is returned as a plain Python list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_semicircle(n: int) -> np.ndarray:
    """Half‑semicircle: sqrt((i+0.5)*(n-i-0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sqrt(i * (n - i))
    vals = np.where(np.isfinite(vals), vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_block_constant(n: int, rng: np.random.Generator,
                                        num_blocks: int) -> np.ndarray:
    """Piecewise‑constant distribution with a small number of blocks."""
    # decide exact block sizes (as equal as possible)
    base = n // num_blocks
    remainder = n % num_blocks
    block_sizes = np.full(num_blocks, base, dtype=int)
    block_sizes[:remainder] += 1
    heights = rng.random(num_blocks) + 1e-6   # keep strictly positive
    a = np.repeat(heights, block_sizes)
    if a.size != n:                           # safety pad / trim
        a = np.pad(a, (0, n - a.size), constant_values=0)[:n]
    a /= a.sum()
    return a

# Greedy pairwise swaps (exact objective)
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Random mass transfers that are kept only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector as a plain Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to explore (larger lengths give finer discretisation)
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray, int]] = []   # (score, seq, length)

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best length={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy(), n))

        # keep a modest number of best seeds globally
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 3. Include previous construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy(), prev.size))
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:30]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never be needed.
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline.
    # 5a – Smoothed Adam (full‑vector) on a few top seeds
    remaining_time = deadline - time.time()
    if remaining_time > 30:
        seeds = [seq for _, seq, _ in sorted(top_candidates, key=lambda x: x[0])[:3]]
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            overall_best_seq, overall_best_score = _smooth_adam(
                seed_seq.copy(),
                sub_deadline,
                overall_best_seq,
                overall_best_score,
            )

    # 5b – Block‑Adam (piecewise‑constant) on the same top seeds
    if time.time() < deadline:
        # take up to three seeds (might be the same as above)
        block_seeds = [seq for _, seq, n in sorted(top_candidates, key=lambda x: x[0])[:3]]
        for seed_seq in block_seeds:
            if time.time() >= deadline:
                break
            n = seed_seq.shape[0]
            K = max(2, n // 2000)                     # ≈ 1 block per 2000 points
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(block_seeds))
            overall_best_seq, overall_best_score = _smooth_adam_block(
                seed_seq.copy(),
                K,
                sub_deadline,
                overall_best_seq,
                overall_best_score,
            )

    # 5c – final greedy pairwise swaps / tiny simulated annealing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
