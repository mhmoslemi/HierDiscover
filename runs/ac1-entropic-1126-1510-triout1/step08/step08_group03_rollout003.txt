"""
Propose a step‑function that minimises ``evaluate_sequence``.

Strategy
--------
1. **Diverse seeds** – generate many candidate probability vectors using a
   large family of analytic forms (Beta, power‑law, exponential, arcsine,
   uniform) and random Dirichlet draws.  Parameters are sampled densely
   (continuous‐valued p,q,α,β,λ) to explore the shape space more thoroughly
   than the previous baseline.

2. **Cheap pre‑selection** – compute the scale‑invariant surrogate
   ``2·n·max(a*a)`` (the exact objective for a normalised vector) and keep
   only the best ``top_k`` candidates (we use a very small ``top_k`` to
   devote most of the time budget to a few promising seeds).

3. **Heavy optimisation** – for each seed run several gradient‑based
   optimisers that respect the simplex constraint:
      * exponentiated‑gradient descent,
      * Adam,
      * continuation (soft‑max) gradient descent,
      * plain sub‑gradient descent.
   The per‑seed time budget is a large fraction of the remaining time,
   allowing many more iterations than the reference implementation.

4. **Fine‑grained stochastic hill‑climbing** – after the deterministic
   passes we perform many tiny random mass‑transfer moves (and occasional
   jitter) while always staying on the simplex.  This stage exploits any
   remaining time budget.

All operations are performed with NumPy and FFT‑based convolutions,
which are fast enough for vectors of length up to ~15000.  The returned
sequence is normalised to unit sum (scaling does not affect the final
score because ``evaluate_sequence`` is homogeneous of degree 0).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of ``max(a*a)`` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with weight vector w
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for terms i=j (k=2i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.3,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    # soft‑max weights
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (once for k = 2*i)
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 2.0,
                             gamma_factor: float = 1.005):
    """Continuation: start with smooth max (γ≈2) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score

def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _random_dirichlet(n: int, concentration: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet point with given concentration parameter."""
    x = np.random.gamma(concentration, 1.0, size=n)
    s = x.sum()
    return x / s if s != 0 else np.full(n, 1.0 / n)

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w.sum()
    return w


# ----------------------------------------------------------------------
def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _powerlaw_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^α · (n-i)^β, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** alpha * (n - i) ** beta
    # guard against overflow / underflow
    w = np.where(np.isfinite(w), w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / s
    return w

    candidates.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [seq for seq, _ in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy optimisation on each seed (time‑sharing)
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        return best_seq.tolist()
    per_seed_budget = remaining_time * 0.85 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        seed_deadline = time.time() + per_seed_budget

        # Exponentiated‑gradient descent
        seq, sc = _exp_gradient_descent(seed_seq, seed_deadline,
                                        max_iters=200_000, eta0=0.95)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} exp‑grad → {sc:.6f}")

        # Adam descent
        seq, sc = _adam_descent(seed_seq, seed_deadline,
                                 max_iters=200_000, step0=0.3)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} Adam → {sc:.6f}")

        # Smoothed‑max continuation
        seq, sc = _smooth_gradient_descent(seed_seq, seed_deadline,
                                            max_iters=250_000,
                                            eta0=0.8,
                                            gamma0=2.0,
                                            gamma_factor=1.005)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} smooth → {sc:.6f}")

        # Plain sub‑gradient descent
        seq, sc = _subgradient_descent(seed_seq, seed_deadline,
                                       max_iters=300_000, eta0=0.6)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} subgrad → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Fine‑grained hill climbing (tiny random transfers, optional SA)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.025               # up to 2.5 % of total mass initially
    # Simulated‑annealing schedule
    init_temp = 4e-4
    final_temp = 2e-6
    total_hc_iters = 200_000

    while time.time() < deadline:
        it += 1
        # geometric decay of transferred amount & temperature
        transfer = max_transfer * (0.5 ** (it / 12000.0))
        temp = init_temp * ((final_temp / init_temp) ** (it / total_hc_iters))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        # accept if improvement or with Metropolis probability
        if new_score < cur_score - 1e-15:
            accept = True
        else:
            # occasional uphill moves
            accept = np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-12))

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to shake out plateaus
        if it % 7_000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 6e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

        # prevent infinite loop if we are out of time
        if it >= total_hc_iters and time.time() + 1.0 >= deadline:
            break

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # Return a normalised vector – scaling does not affect the true objective.
    best_seq = best_seq / best_seq.sum()
    return best_seq.tolist()
