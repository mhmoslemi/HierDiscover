def _self_convolution_fft(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n-1]

def _cross_convolution(a, w):
    """Compute g_i = sum_k w_k a_{k-i} for i=0..n-1 via FFT."""
    n = a.shape[0]
    len_w = w.shape[0]
    size = 1 << ((n + len_w - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)
    # conv length = size; full conv length = n+len_w-1; us slice [n-1:n-1+n]
    start = len_w - 1  # n-1? Let's compute len_w = 2n-1, start = n-1
    start = n - 1
    return conv[start:start + n]

    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    lr = 0.1  # may tune
    t=0
    best_a = a0.copy()
    best_obj = _objective(a0)
    while time.time() < deadline:
        # compute a from z
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()
        # Compute self conv
        b = _self_convolution_fft(a)
        maxb = b.max()
        # compute w
        # For numerical stability, subtract max in exponent
        beta_b = beta * b
        max_beta_b = beta_b.max()
        exp_beta_b = np.exp(beta_b - max_beta_b)
        w = exp_beta_b / exp_beta_b.sum()

        grad_a = 2.0 * _cross_convolution(a, w)  # shape n
        # compute grad_z
        grad_z = _grad_z_from_a(a, grad_a)
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # Optionally early stop if improvement small.
    return best_a, best_obj

"""
Propose a step‑function (non‑negative vector with unit sum) that minimizes
the evaluation function

    2·n·max(conv(a,a))

where conv(a,a) is the full self‑convolution.
The algorithm works in three stages:

1.  **Diverse analytic initialisation** – families of distributions
    (uniform, arcsine, symmetric beta, exponential, alternating,
    sparse random and noisy perturbations) are sampled for a fixed length
    `n`.  The best of these candidates (according to the exact objective)
    is taken as the starting point.

2.  **Deterministic smoothed optimisation** – the non‑convex objective
    `max(conv)` is approximated by a smooth `log‑sum‑exp` surrogate

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) .

    For a given smoothing parameter `β` the gradient w.r.t. the
    soft‑max parametrisation `a_i = exp(z_i)/Σ_j exp(z_j)` can be written
    analytically and is evaluated by two FFT‑based convolutions.
    An Adam optimiser performs projected gradient steps on `z`.
    The smoothing parameter is increased geometrically
    (β = 0.5, 1, 2, 5, 10, 20, 40, 80) so that the algorithm first moves
    quickly on a smooth landscape and then refines the exact max.

3.  **Greedy active‑set polishing** – a fast pairwise mass‑transfer
    local search (the same routine used in the reference solution) is
    run for the remaining time to exploit the exact sub‑gradient of the
    true max‑convolution.

All work is done on the simplex (non‑negative, unit sum) and respects
the global 1000 s budget.  The final vector is returned as a plain list
of floats.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _cross_convolution(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k w_k a_{k-i}  (i = 0..n‑1) using FFT.
    `w` has length 2·n‑1 (the weight vector from the smooth max).
    """
    n = a.shape[0]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()   # ≥ 3·n‑2
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    conv_full = np.fft.irfft(A * W, n=size)
    start = n - 1
    return conv_full[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  – assumes Σa = 1."""
    n = a.shape[0]
    b = _self_convolution_fft(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse catalogue of starting vectors."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta – a broad range of shape parameters
    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – several decay rates
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # sparse random start (≈0.5 % non‑zeros)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    rnd[idx] = rng.random(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    # noisy perturbations of a few beta shapes
    for _ in range(3):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert)
        cand.append(pert)

    return cand

def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass‑transfer based on the exact active‑set
    sub‑gradient of max(conv).  Returns the best vector seen.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _self_convolution_fft(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient: g_i = 2 Σ_{k∈active} a_{k-i}
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # try a smaller transfer
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            continue

        # occasional jitter
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr

    return best_vec, best_score

def _optimize_z(z0: np.ndarray,
                beta: float,
                deadline: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth surrogate
        f_β(a) = (1/β)·log Σ exp(β·(a*a)_k)
    where a = softmax(z).  Returns the best affine‐feasible vector.
    """
    n = len(z0)
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # learning rate scaled with β (large β ⇒ smaller steps)
    lr = 0.2 / (beta + 1.0)

    best_a = None
    best_score = float('inf')
    t = 0

    while time.time() < deadline:
        # ----- a = softmax(z) -----
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # occasional exact objective evaluation
        if t % 20 == 0:
            obj = _objective(a)
            if obj < best_score:
                best_score = obj
                best_a = a.copy()
                print(f"[{time.time():.1f}] β={beta:.2f} it={t} obj={obj:.6f}")

        # ----- smooth max weights -----
        b = _self_convolution_fft(a)                     # length 2n‑1
        beta_b = beta * b
        max_beta_b = beta_b.max()
        exp_beta_b = np.exp(beta_b - max_beta_b)         # stabilise
        w = exp_beta_b / exp_beta_b.sum()                # Σ w = 1

        # ----- gradient w.r.t. a -----
        grad_a = 2.0 * _cross_convolution(a, w)           # length n

        # ----- gradient w.r.t. z (soft‑max chain rule) -----
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z -= step

    # If we never stored a better point, fall back to the final one
    if best_a is None:
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)
        best_a = exp_z / exp_z.sum()
        best_score = _objective(best_a)

    return best_a, best_score
