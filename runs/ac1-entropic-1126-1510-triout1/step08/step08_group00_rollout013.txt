def _smooth_adam_descent(a0, deadline,
                         step0=0.2,
                         beta1=0.9,
                         beta2=0.999,
                         lam_start=0.5,
                         lam_end=30.0,
                         max_iters=200_000):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        lam = lam_start + (lam_end - lam_start) * min(1.0, (t-1) / max_iters)
        # compute weighted gradient
        conv = _fft_self_convolve(a)
        maxb = conv.max()
        # compute weights w = exp(lam * conv) normalized
        exp_lam = np.exp(lam * (conv - maxb))  # shift to avoid overflow
        # It may create underflow for large negative; okay.
        w_sum = exp_lam.sum()
        if w_sum == 0:
            w = np.full_like(exp_lam, 1.0 / exp_lam.size)
        else:
            w = exp_lam / w_sum
        # compute weighted gradient
        # use conv_rev approach
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[len(a)-1: len(a)-1+len(a)]
        # diagonal correction
        diag_w = w[::2]  # size n
        grad -= a * diag_w
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

"""
Propose a low‑score non‑negative step function.

This version augments the previous approach with a *smooth* Adam optimizer
that minimizes a soft‑max (log‑sum‑exp) surrogate of the max‑convolution.
The surrogate is gradually sharpened (λ increased) so the optimizer can
pass the non‑smooth ridge that often traps plain sub‑gradient methods.
After the smooth phase we fall back to the classic Adam / exponentiated‑
gradient refinements, followed by Frank‑Wolfe mass‑transfer polishing and
a short simulated‑annealing jitter stage.  The whole search respects a
1000 s budget and always returns a list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(a,a)) – exact when ∑a = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _weighted_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum  Σ_i w_i * (a*a)_i  w.r.t. a,
    assuming w ≥ 0, Σw = 1.
    """
    n = a.shape[0]
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (terms a_j^2 appear only once in the weighted sum)
    diag_w = w[::2]                     # w_{2j}
    grad -= a * diag_w
    return grad


def _smooth_grad_and_val(a: np.ndarray, lam: float):
    """
    Compute smooth surrogate (soft‑max) value and its gradient.
    The surrogate is Σ_i w_i b_i where b = a*a and
    w_i = exp(lam * b_i) / Σ_j exp(lam * b_j).
    """
    b = _fft_self_convolve(a)
    maxb = b.max()
    # soft‑max weights (shifted by maxb for numerical stability)
    exp_lam = np.exp(lam * (b - maxb))
    w_sum = exp_lam.sum()
    if w_sum == 0.0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w = exp_lam / w_sum

    val = np.dot(w, b)                 # weighted sum = smooth approximation of max
    grad = _weighted_grad(a, w)        # gradient of the weighted sum
    return grad, val, maxb

def _smooth_adam_descent(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.18,
    beta1: float = 0.9,
    beta2: float = 0.999,
    lam_start: float = 0.5,
    lam_end: float = 30.0,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth surrogate.
    Returns the best point (in terms of the true max‑convolution score)
    seen before the deadline.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linearly increase λ from lam_start to lam_end
        lam = lam_start + (lam_end - lam_start) * min(1.0, (t - 1) / max_iters)

        grad, _, _ = _smooth_grad_and_val(a, lam)

        # Adam updates (the gradient is for the *max* part only – scale is absorbed
        # by the step size)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)

        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _adam_descent(
    a0: np.ndarray,
    deadline: float,
    step0: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Plain Adam on the non‑smooth surrogate (max‑convolution)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    eta0: float = 0.6,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (KL‑mirror) descent on the max‑convolution surrogate."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ``alpha``."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families (all return a *normalized* vector that sums to 1)

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    vals /= vals.sum()
    return vals


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    vals /= vals.sum()
    return vals


def _sin_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals /= vals.sum()
    return vals

        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:

            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
