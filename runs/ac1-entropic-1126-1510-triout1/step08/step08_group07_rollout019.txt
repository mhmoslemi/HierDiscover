"""
A robust optimiser for the “min‑max self‑convolution” task.

Strategy
--------
1. Build a diverse pool of analytic seeds (uniform, arcsine, beta,
   exponential, alternating) and a number of random mixtures that are
   deliberately asymmetric (different mass on the left and on the right).
   All vectors are normalised to a sum of 1.

2. Fast exploration: repeatedly draw a random mixture, evaluate the
   objective with a single FFT‑based convolution and keep the best
   candidate seen so far.  This phase uses the majority of the time
   budget.

3. Refinement: starting from the best seed, run the aggressive greedy
   active‑set method from the reference implementation and finish with
   a short simulated‑annealing walk.  Both stages respect the deadline.

The whole routine is pure NumPy / SciPy, works on a single thread and
returns a plain Python list that can be fed directly to
`evaluate_sequence`.  The implementation is deliberately modular so
that each component can be unit‑tested independently.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Objective for a probability vector (sum = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (p = q) on [0,1]."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """General (asymmetric) beta density on a discretised unit interval."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Random / mixed candidates --------------------------------------------
def _random_mixture_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build an asymmetric mixture of three components:
        – uniform,
        – a left‑skewed beta (or exponential) density,
        – its mirror image (right‑skewed).
    The three weights are drawn from a Dirichlet distribution.
    """
    # three mixture weights that sum to 1
    w = rng.dirichlet([1.0, 1.0, 1.0])
    w_uniform, w_left, w_right = w

    # uniform component
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # choose a left‑hand shape at random
    if rng.random() < 0.6:       # beta dominates
        p = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        q = 10 ** rng.uniform(-3, 1)
        left = _distribution_beta(n, p, q)
    else:                         # exponential tail
        lam = 10 ** rng.uniform(-4, -0.5)   # 0.0001 … 0.3
        left = _distribution_exponential(n, lam)

    right = left[::-1]          # mirror of the left part

    a = w_uniform * uniform + w_left * left + w_right * right
    # protect against tiny negatives caused by floating‑point noise
    a = np.maximum(a, 0.0)
    # final simplex projection (cheap, keeps sum == 1)
    return _project_onto_simplex(a, 1.0)


def _random_sparse_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """A very sparse random vector (few non‑zeros)."""
    k = max(2, n // 200)                     # ~0.5 % non‑zero entries
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.random(k)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a

# Initial candidate catalogue -------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # a small random sparse vector – useful as a “break‑symmetry” start
    cand.append(_random_sparse_candidate(n, rng))

    # noisy perturbations of a few beta shapes
    for _ in range(3):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert)
        cand.append(pert)

    return cand

# Greedy active‑set optimiser (unchanged, but kept here for completeness)
def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient for the active set
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        delta = min(a[src], 0.5)                     # try moving a decent chunk
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-15:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy it={it} improved to {best_score:.6f}")
            continue

        # smaller fraction if the full move does not help
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-15:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy (small) it={it} improved to {best_score:.6f}")
            continue

        # occasional jitter to escape plateaux
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-15:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score

# Simulated‑annealing fine‑tuning (unchanged)
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9997

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine consists of:
      * generating a rich pool of analytic seeds,
      * an intensive random‑mixture exploration phase,
      * aggressive greedy local optimisation,
      * a concluding simulated‑annealing polish.
    The final vector is projected onto the probability simplex and returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # spare margin for termination

    # length – large enough to give flexibility but still fast for many FFTs
    n = 3300
    if n < 2:
        n = 2

    # ---------------------- 1) analytic seeds -------------------------
    candidates = _initial_candidates(n, rng)

    # incorporate a previously discovered best vector if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx:02d} score={sc:.6f}")

    # ---------------------- 2) random‑mixture search -----------------
    # reserve a safe chunk of the budget for the final optimisation steps
    refinement_budget = 200.0   # seconds
    while time.time() < deadline - refinement_budget:
        # 70 % mixtures, 30 % sparse random vectors
        if rng.random() < 0.7:
            cand = _random_mixture_candidate(n, rng)
        else:
            cand = _random_sparse_candidate(n, rng)
        cand = cand / cand.sum()
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] random mix score={sc:.6f}")

    if best_vec is None:                     # extreme fallback
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ---------------------- 3) greedy refinement ----------------------
    best_vec, best_score = _greedy_local_search(best_vec, deadline, rng)

    # ---------------------- 4) simulated‑annealing --------------------
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)

    # final sanity – enforce the simplex constraint exactly
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
