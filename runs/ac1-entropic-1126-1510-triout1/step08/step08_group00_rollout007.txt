def _adam_optimize_smooth(a0, gamma, deadline, step0=0.15, max_iters=200000):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t+=1
        grad = _smooth_gradient(a, gamma)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        # bias correction
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        eta = step0/(np.sqrt(v_hat)+1e-8)
        a = a - eta*m_hat
        a = _project_onto_simplex(a, 1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _exp_gradient_optimize_smooth(a0, gamma, deadline, eta0=0.6, max_iters=200000):
    a = a0.copy()

        it+=1
        grad = _smooth_gradient(a, gamma)
        eta = eta0/np.sqrt(it)
        a = a * np.exp(-eta*grad)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0/a.size)

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()

    Every candidate is ℓ₁‑normalised (∑a=1) and scored with the cheap
    surrogate 2·n·max(conv(a,a)).

2️⃣  Refinement of the best seeds with Adam‑ and exponentiated‑gradient
    descent on a *smooth* approximation of the max‑convolution.
    The smooth objective is
          fγ(a)=2·n· (1/γ)·log ∑ₖ exp(γ·(a∗a)ₖ)                (γ≫1)
    Its gradient is   ∇fγ(a)=2·conv_rev(w)   with
          wₖ ∝ exp(γ·(a∗a)ₖ) ,   Σₖ wₖ = 1 .
    A continuation schedule (γ = 5 → 20 → 80 → 300 → 1200) allows
    coarse‑to‑fine optimisation.  All iterates are projected onto the
    probability simplex.

3️⃣  Final polishing – a Frank‑Wolfe‑style “mass‑transfer’’ step that moves
    probability mass from the component with the largest gradient to the
    one with the smallest – followed by a tiny simulated‑annealing jitter.

The routine respects the 1000 s budget, always returns a list of
non‑negative floats, and prints a few progress messages.
"""

import time
import itertools
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate score 2·n·max(conv(a,a)) (exact when Σa = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the smooth surrogate
        fγ(a) = 2·n· (1/γ)·log Σₖ exp(γ·(a∗a)ₖ)
    which is   ∇fγ(a) = 2·conv_rev(w)   with   wₖ ∝ exp(γ·bₖ).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                     # length 2n‑1
    # stable softmax on γ·b
    m = gamma * b
    m_max = m.max()
    w = np.exp(m - m_max)
    w /= w.sum()
    # conv_rev = a_rev * w   (then take centre slice)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize_smooth(a0: np.ndarray, gamma: float,
                          deadline: float,
                          step0: float = 0.15,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          max_iters: int = 300_000):
    """
    Adam‑style adaptive moment descent on the smooth surrogate.
    Returns the best point seen and its surrogate score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_gradient(a, gamma)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, 1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_optimize_smooth(a0: np.ndarray, gamma: float,
                                 deadline: float,
                                 eta0: float = 0.6,
                                 max_iters: int = 250_000):
    """
    Exponentiated‑gradient descent on the smooth surrogate.
    Returns the best point seen and its surrogate score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _uniform_distribution(n: int) -> np.ndarray:
    """Flat probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the 1000 s time budget and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # 0️⃣ Determine a base length (fallback to 4000)
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # Choose a handful of lengths around the base (not too many, not too big)
    lengths = set()
    for scale in (0.85, 1.0, 1.15):
        n = int(base_n * scale)
        if 1500 <= n <= 12000:
            lengths.add(n)
    # a few random lengths for diversity
    for _ in range(5):
        n = np.random.randint(1500, min(12000, base_n * 2) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    print(f"[{time.time():.1f}] Exploring {len(lengths)} lengths "
          f"(base≈{base_n})")

    # --------------------------------------------------------------
    # 1️⃣ Broad sampling (families + simple mixtures)

    family_fns = [
        ("arcsine", _arcsine_distribution),
        ("triangular", _triangular_distribution),
        ("exponential", lambda n: _exponential_distribution(n, lam=0.005)),
        ("beta_sym_0.5", lambda n: _beta_symmetric_distribution(n, p=0.5)),
        ("beta_asym_0.3_0.7",
         lambda n: _beta_asymmetric_distribution(n, 0.3, 0.7)),
        ("powerlaw_1.0", lambda n: _power_law_distribution(n, p=1.0)),
        ("uniform", _uniform_distribution),
    ]

    candidates = []   # list of (seq ndarray, surrogate score)

    # parameters for random Dirichlet draws
    dirichlet_alphas = [0.2, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families -------------------------------------------------
        for name, gen in family_fns:
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} fam={name}")

        # simple convex combos of two families (few fixed mixing ratios) -----------
        for (name1, gen1), (name2, gen2) in itertools.combinations(family_fns, 2):
            # try a few mixing weights
            for w in (0.2, 0.5, 0.8):
                a = w * gen1(n) + (1.0 - w) * gen2(n)
                a /= a.sum()
                candidates.append((a, _max_convolution_score(a)))
                # keep the output quiet – only count as a candidate

        # random Dirichlet draws -------------------------------------------------
        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ----------------------------------------------------------------------
    # 2️⃣ If a previous best exists, add it to the pool (helps with restarts)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # 3️⃣ Keep the top‑k seeds for refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")

    best_seq, best_score = top_seeds[0][0].copy(), top_seeds[0][1]
    print(f"Initial best surrogate score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4️⃣ Refinement of each seed (smooth Adam + smooth EG + continuation)

    gamma_schedule = [5, 20, 80, 300, 1200]   # increasingly sharp approximations

    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # start from the seed; run the γ‑continuation
        a = seed_seq.copy()
        cur_best_a = a.copy()
        cur_best_sc = _max_convolution_score(a)

        for gamma in gamma_schedule:
            # Adam on the smooth surrogate
            a, sc = _adam_optimize_smooth(a, gamma, deadline,
                                          step0=0.12, max_iters=150_000)
            if sc < cur_best_sc - 1e-14:
                cur_best_sc, cur_best_a = sc, a.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{top_k} "
                      f"Adam γ={gamma} improved to {cur_best_sc:.6f}")

            # Exponentiated‑gradient on the same γ
            a, sc = _exp_gradient_optimize_smooth(a, gamma, deadline,
                                                  eta0=0.5, max_iters=120_000)
            if sc < cur_best_sc - 1e-14:
                cur_best_sc, cur_best_a = sc, a.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{top_k} "
                      f"ExpGrad γ={gamma} improved to {cur_best_sc:.6f}")

        # update global best
        if cur_best_sc < best_score - 1e-14:
            best_score, best_seq = cur_best_sc, cur_best_a.copy()
            print(f"[{time.time():.1f}] GLOBAL new best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5️⃣ Frank‑Wolfe style mass‑transfer polishing (uses the true max‑conv gradient)

    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 100_000:
        fw_iter += 1
        # true sub‑gradient of max(conv) – cheap since we already have a helper
        grad, _, _ = _subgrad(a)   # re‑use the previous sub‑grad implementation
        i_src = int(np.argmax(grad))    # where moving mass away reduces the max
        i_dst = int(np.argmin(grad))    # where adding mass helps most

        if i_src == i_dst:
            break

        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break

        # try a full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-13:
            a, cur_score = a_full, score_full
            improved = True
        else:
            # try half‑transfer as a fallback
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-13:
                a, cur_score = a_half, score_half
                improved = True

        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if fw_iter % 4000 == 0 and np.random.rand() < 0.18:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, 1.0)
            cur_score = _max_convolution_score(a)

    # ----------------------------------------------------------------------
    # 6️⃣ Light simulated‑annealing / jitter phase

    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.025
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 150_000:
        anneal_iter += 1
        # linear temperature schedule (more aggressive near the end)
        t_frac = (deadline - time.time()) / (budget_s - 0.5)
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a, cur_score = a_new, new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} "
                      f"new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a, cur_score = a_new, new_score

    # ----------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
