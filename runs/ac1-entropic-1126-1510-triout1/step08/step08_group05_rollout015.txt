"""
propose_candidate
-----------------

Search for a non‑negative step function (finite vector) that minimises the
black‑box ``evaluate_sequence`` objective

    2 * n * max(a * a) / (sum(a)**2)

(the denominator is invariant under scaling, therefore we keep the vector
normalised to sum = 1).

The search consists of three stages:

1. **Broad initial scan** – analytic families (uniform, arcsine, beta,
   exponential, power‑law, sine‑power, cosine windows, a simple U‑shape,
   a few random Dirichlet draws).  For families with a free parameter we
   first test a coarse grid and then perform a tiny stochastic refinement
   of the parameter.

2. **Refinement** – two complementary descent schemes on the exact
   non‑smooth objective:
      * *Mirror descent* on the simplex using the sub‑gradient of
        ``max(a*a)`` (exponential update followed by Euclidean projection
        onto the probability simplex).
      * *Sub‑gradient descent with momentum* (the same update that appears
        in the reference implementation).

   Both phases run until the time budget expires, continuously keeping the
   best vector found.

3. **Final polishing** – a short diffusion smoothing pass and a fast
   simulated‑annealing hill‑climber that moves a tiny amount of mass from
   a source index to a target index using an O(n) incremental update of the
   self‑convolution.

All heavy work is performed with FFT‑based convolutions (O(n log n)).
The routine respects the supplied time budget (default 1000 s) and returns

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow_two(x):
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow_two(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    """
    Return 2·n·max(a*a) for a vector a that is already normalised to sum = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)

# Analytic families – all return a vector with sum = 1
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log to avoid underflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

    if n == 1:

    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

    idx = np.arange(1, n + 1, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / (i + 1) + 1.0 / (n - i)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part of the convolution
    conv[i:i + len(a)] -= 2.0 * delta * a_old
    conv[j:j + len(a)] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

        conv = _fft_self_convolve(a_cur)
        cur_score = 2.0 * n * conv.max()
        best_sa_score = cur_score
        best_sa_seq = a_cur.copy()

        T = cur_score * 0.05       # initial temperature (≈5 % of current score)
        decay = 0.9998
        max_delta_factor = 0.02    # move at most 2 % of the source mass each step
        SA_iter = 0

        while time.time() < deadline:
            SA_iter += 1
            src = np.random.choice(n, p=a_cur)
            tgt = np.random.randint(n)
            if src == tgt:
                continue

            max_delta = min(a_cur[src] * max_delta_factor, 0.04)
            if max_delta <= eps:
                continue
            delta = max_delta * np.random.rand()
            if delta <= eps:
                continue

            a_snapshot = a_cur.copy()
            conv_snapshot = conv.copy()

            _apply_transfer(a_cur, conv, src, tgt, delta)

            new_score = 2.0 * n * conv.max()
            if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / T):
                cur_score = new_score
                if new_score < best_sa_score - 1e-12:
                    best_sa_score = new_score
                    best_sa_seq = a_cur.copy()
                    best_seq = best_sa_seq.copy()
                    best_score = best_sa_score
                    print(f"[{time.time():.1f}] SA iter={SA_iter} new best {new_score:.6f}")
            else:
                a_cur = a_snapshot
                conv = conv_snapshot

            T *= decay
            # occasional jitter when stuck
            if SA_iter % 5000 == 0 and T < 1e-9:
                jitter = np.random.randn(n) * 1e-5
                a_cur += jitter
                a_cur = np.maximum(a_cur, eps)
                a_cur /= a_cur.sum()
                conv = _fft_self_convolve(a_cur)
                cur_score = 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 6.  Final mixing with the uniform distribution (tiny convex combos).
    if time.time() < deadline:
        uniform = _distribution_uniform(n)
        for t in np.linspace(0.0, 0.5, 11):   # t is weight on the current best
            cand = t * best_seq + (1 - t) * uniform
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            sc = _max_convolution_score(cand)
            if sc < best_score - 1e-12:
                best_score, best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] Mixing t={t:.2f} new best {sc:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score ≈ {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
