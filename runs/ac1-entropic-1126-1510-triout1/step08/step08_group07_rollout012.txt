"""
A two‑phase stochastic optimiser for the self‑convolution supremum problem.

Phase 1 – randomised construction search.
    * Generates many analytic families (uniform, arcsine, symmetric / asymmetric
      beta, exponential, alternating, sparse) and random mixtures of two.
    * The parameters (shape exponents, decay, stride, mixture weight) are drawn
      from wide log‑uniform ranges.
    * Each candidate is projected onto the probability simplex and evaluated
      with a fast FFT‑based convolution.  The best ~5 candidates are kept.

Phase 2 – local refinement.
    * For each of the retained candidates a greedy active‑set move (source →
      destination of mass) is applied, identical to the method that achieved
      1.510197 in the reference solution, but with a slightly higher jitter
      frequency.
    * After the greedy phase a short simulated‑annealing walk fine‑tunes the
      solution.
    * The best refined vector is returned as a plain Python list.

The whole routine respects the 1000 s time‑budget and stops early if the
target bound 1.503 is reached.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta with shape parameters a,b (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """~0.5 % non‑zero entries, random weights."""
    k = max(2, n // 200)
    v = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    v[idx] = rng.random(k)
    v /= v.sum()
    return v


# ----------------------------------------------------------------------
def _generate_base_distribution(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a single analytic (non‑mixture) distribution."""
    typ = rng.integers(0, 7)
    if typ == 0:
        return _distribution_uniform(n)
    if typ == 1:
        return _distribution_arcsine(n)
    if typ == 2:
        # symmetric beta, log‑uniform shape in [1e‑3, 4]
        p = 10 ** rng.uniform(-3, 0.6)
        return _distribution_beta_sym(n, p)
    if typ == 3:
        # asymmetric beta
        a = 10 ** rng.uniform(-3, 0.6)
        b = 10 ** rng.uniform(-3, 0.6)
        return _distribution_beta_asym(n, a, b)
    if typ == 4:
        lam = 10 ** rng.uniform(-4, -1)   # lam ∈ [1e‑4, 1e‑1]
        return _distribution_exponential(n, lam)
    if typ == 5:
        stride = rng.integers(2, 6)
        return _distribution_alternating(n, stride)
    # typ == 6
    return _distribution_sparse(n, rng)


def _generate_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """Random mixture / perturbation of analytic families."""
    # 30 % of the time produce an explicit mixture of two bases
    if rng.random() < 0.30:
        d1 = _generate_base_distribution(rng, n)
        d2 = _generate_base_distribution(rng, n)
        w = rng.random()
        cand = w * d1 + (1.0 - w) * d2
    else:
        cand = _generate_base_distribution(rng, n)

    # occasional small Gaussian perturbation (adds diversity)
    if rng.random() < 0.10:
        cand = cand + rng.normal(scale=0.005, size=n)
        cand = np.maximum(cand, 0.0)

    return _project_onto_simplex(cand, 1.0)


# ----------------------------------------------------------------------
def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer based on the exact sub‑gradient of the
    active convolution shift(s).  Returns the best vector seen before the
    deadline.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = np.where(b >= maxb - eps)[0]

        # sub‑gradient
        g = np.zeros_like(a)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        dest = int(np.argmin(g))

        if src == dest:
            break

        # attempt a full transfer (capped for safety)
        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if best_score < target:
                break
            continue

        # try a reduced move if full transfer failed
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if best_score < target:
                break
            continue

        # jitter more frequently than the reference (every 200 it.)
        if it % 200 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    if best_score < target:
                        break

    return best_vec, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         target: float = 1.503) -> tuple[np.ndarray, float]:
    """Cheap Metropolis walk with exponential cooling, terminating early if target met."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    T = 0.03

    while time.time() < deadline and T > 1e-6:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer up to 10 % of source mass
        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0.0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                if best_score < target:
                    break

        T *= 0.9997

    return best_a, best_score


# ----------------------------------------------------------------------
def _refine_candidate(vec: np.ndarray,
                      time_budget: float,
                      rng: np.random.Generator,
                      target: float = 1.503) -> tuple[np.ndarray, float]:
    """
    Run greedy optimisation followed by a short SA, constrained by a time budget.
    """
    deadline = time.time() + time_budget
    vec, sc = _greedy_local_search(vec, deadline, rng, target)
    if sc > target and time.time() < deadline:
        vec, sc = _simulated_annealing(vec, deadline, rng, target)
    return vec, sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Stochastic search for a step‑function (probability vector) that minimises
    ``evaluate_sequence``.  The routine creates many analytic seeds,
    keeps the best few, and refines each with a greedy active‑set optimiser
    plus simulated‑annealing.  The final vector is projected onto the simplex
    and returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.4  # tiny safety margin

    # Choose a length that balances accuracy and speed.
    n = 4000
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # Phase 1 – random sampling of analytic / mixed families
    top_K = 6                     # keep the best few for refinement
    top = []                      # list of (score, vector)
    best_score = float('inf')
    best_vec = None

    # leave ~30 % of the budget for the refinement phase
    refinement_reserve = 0.30 * budget_s

    while time.time() < deadline - refinement_reserve:
        cand = _generate_candidate(rng, n)
        sc = _objective(cand)

        # track global best
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.2f}] new global best (sampling)  score={sc:.6f}")

        # maintain a small elite set
        if len(top) < top_K:
            top.append((sc, cand.copy()))
        else:
            # replace the worst in the elite if current is better
            worst_idx = max(range(len(top)), key=lambda i: top[i][0])
            if sc < top[worst_idx][0]:
                top[worst_idx] = (sc, cand.copy())

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    # Ensure the elite set contains the current overall best
    if best_vec is not None:
        worst_idx = max(range(len(top)), key=lambda i: top[i][0]) if top else None
        if worst_idx is not None and best_score < top[worst_idx][0]:
            top[worst_idx] = (best_score, best_vec.copy())

    # ------------------------------------------------------------------
    # Phase 2 – refinement of the elite candidates
    time_left = deadline - time.time()
    if time_left <= 0:
        # budget exhausted – just return the best we have
        best_vec = _project_onto_simplex(best_vec, 1.0)
        return best_vec.tolist()

    # allocate time roughly equally (give a bit more to the current best)
    allocations = []
    total = sum(1.0 for _ in top)
    # give 1.5× weight to the best elite entry
    weights = [1.5 if i == 0 else 1.0 for i in range(len(top))]
    weight_sum = sum(weights)
    for w in weights:
        allocations.append(time_left * (w / weight_sum))

    # sort elite by score (ascending) so index 0 is the strongest
    top.sort(key=lambda x: x[0])

    refined_best_score = best_score
    refined_best_vec = best_vec

    for (sc, vec), t_alloc in zip(top, allocations):
        if time.time() >= deadline:
            break
        if t_alloc < 0.5:   # not worth a full refinement
            continue
        vec_ref, sc_ref = _refine_candidate(vec, t_alloc, rng, target=1.503)
        if sc_ref < refined_best_score:
            refined_best_score = sc_ref
            refined_best_vec = vec_ref.copy()
            print(f"[{time.time():.2f}] new global best (refinement) score={sc_ref:.6f}")

        if refined_best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Final sanity – ensure we are on the simplex
    if refined_best_vec is None:
        refined_best_vec = _distribution_uniform(n)
    refined_best_vec = _project_onto_simplex(refined_best_vec, 1.0)

    final_score = _objective(refined_best_vec)
    print(f"Final best score after all phases: {final_score:.6f} (length={len(refined_best_vec)})")
    return refined_best_vec.tolist()
