"""
Propose a step‑function vector with a very small max self‑convolution.

The search proceeds in three stages:

1. **Broad initial scan** – a collection of analytic families (uniform,
   arcsine, triangular, exponential, symmetric Beta, sine‑power, Gaussian)
   together with random Dirichlet draws are generated for a few candidate
   lengths.  The best candidate (lowest 2·n·max(a*a) with sum(a)=1) is kept.

2. **Linear‑programming (Frank–Wolfe) refinement** – given a current
   distribution `f`, we solve the LP  

        maximise   Σ g_i
        subject to (f * g)_k ≤ M   (∀k)
                   g_i ≥ 0

   where `M = max(f*f)`.  The LP returns an extreme point `g`; we mix it
   with `f` using a diminishing step size `t = 2/(iter+2)`.  This strictly
   reduces `M` and quickly drives the shape towards the optimal one.
   The LP is built with a sparse Toeplitz convolution matrix and solved
   by SciPy’s HiGHS interface.

3. **Local hill‑climber** – a lightweight O(n) mass‑transfer hill‑climber
   works directly on the full self‑convolution to polish the solution
   until the time budget (1000 s) expires.

All vectors are kept on the simplex (non‑negative, sum = 1) and the
algorithm respects the 1000 s wall‑clock limit.  The best sequence found
is returned as a plain Python list of floats.
"""

from __future__ import annotations

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Return the evaluation objective 2·n·max(a*a) with sum(a)=1."""
    a = a / a.sum()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _dist_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dist_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _dist_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _dist_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _dist_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _dist_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _dist_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _dist_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _make_lp_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix for the constraints (f * g)_k ≤ M.
    Entry (k, j) = f_{k‑j} if 0 ≤ k‑j < n else 0.
    """
    n = f.shape[0]
    # Row indices: for each offset d (0..n-1) we add rows d + j, j=0..n-1
    d = np.arange(n, dtype=np.int64)[:, None]            # (n,1)
    j = np.arange(n, dtype=np.int64)[None, :]            # (1,n)
    rows = (d + j).ravel()                               # length n*n
    cols = np.tile(np.arange(n, dtype=np.int64), n)      # repeated n times
    data = np.repeat(f, n)                               # each f[d] repeats n times
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Transfer `delta` mass from index i to index j, updating the vector `a`
    and its full self‑convolution `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    # old copy of a needed for the update formula
    a_i_old = a[i]
    a_j_old = a[j]

    # update convolution: increase contributions involving j, decrease those with i
    # For each k, conv[k] changes by 2*delta*(a_{k-i} - a_{k-j}) plus delta^2 at the
    # diagonal positions 2*i and 2*j.
    conv[i : i + n] -= 2.0 * delta * a
    conv[j : j + n] += 2.0 * delta * a
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta

    # finally move the mass
    a[i] -= delta
    a[j] += delta

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine combines analytic initialisation, a linear‑programming Frank‑Wolfe
    refinement, and a final O(n) hill‑climber.  It respects the 1000 s
    wall‑clock budget.
    """
    # ------------------------------------------------------------------
    # basic setup
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0               # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad scan over a handful of lengths
    scan_lengths = [1500, 2000, 2500, 3000]
    for n in scan_lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # Analytic families
        candidates.append(_dist_uniform(n))
        candidates.append(_dist_arcsine(n))
        candidates.append(_dist_triangular(n))
        candidates.append(_dist_exponential(n, lam=0.004))
        candidates.append(_dist_exponential(n, lam=0.02))

        for p in (0.2, 0.3, 0.5, 0.8, 1.0, 2.0, 4.0):
            candidates.append(_dist_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_dist_sine_power(n, pw))

        for sigma in (0.15, 0.2, 0.3, 0.4):
            candidates.append(_dist_gaussian(n, sigma))

        # Random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            vec = np.random.dirichlet(np.full(n, alpha))
            candidates.append(vec)

        # Sparse random start (k = n//200)
        k = max(2, n // 200)
        sparse = np.zeros(n, dtype=np.float64)
        idxs = np.random.choice(n, k, replace=False)
        sparse[idxs] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # Evaluate cheaply (FFT based)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Optionally incorporate a globally supplied previous best
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        best_n = 2000
        best_seq = _dist_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Linear‑programming refinement (Frank–Wolfe style)
    a = best_seq.copy()
    n = a.shape[0]
    max_lp_iters = 15                     # small number – each LP is cheap
    lp_iter = 0

    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1
        conv_a = _fft_self_convolve(a)
        M = conv_a.max()

        # Build the Toeplitz constraint matrix for (a * g) ≤ M
        A_ub = _make_lp_matrix(a)
        b_ub = np.full(2 * n - 1, M * (1.0 + 1e-12))
        c = -np.ones(n)                   # maximize sum(g) → minimize -sum(g)

        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True, "time_limit": max(1, deadline - time.time())},
        )
        if not res.success:
            print(f"[{time.time():.1f}] LP failed at iter {lp_iter}, stopping LP phase.")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g                     # re‑normalise to sum = 1

        # Diminishing step size (standard Frank–Wolfe schedule)
        t = 2.0 / (lp_iter + 2.0)

        a_new = (1.0 - t) * a + t * g
        conv_new = _fft_self_convolve(a_new)
        M_new = conv_new.max()
        if M_new < M - 1e-15:            # descent observed
            a = a_new
            cur_score = 2.0 * n * M_new
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP iter={lp_iter}, t={t:.4f} new best {cur_score:.6f}")
        else:
            # If no improvement, shrink t aggressively and retry once
            t = t * 0.5
            a_new = (1.0 - t) * a + t * g
            conv_new = _fft_self_convolve(a_new)
            M_new = conv_new.max()
            if M_new < M - 1e-15:
                a = a_new
                cur_score = 2.0 * n * M_new
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] LP iter={lp_iter} (after shrink) new best {cur_score:.6f}")
            else:
                # No progress – exit the LP loop early
                print(f"[{time.time():.1f}] LP iter={lp_iter} no progress, breaking.")
                break

        # Early exit guard
        if time.time() + 0.5 >= deadline:
            break

    # ------------------------------------------------------------------
    # 4. Final hill‑climber (O(n) mass transfers)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iter = 0
    while time.time() < deadline:
        hill_iter += 1

        # Identify the convolution index attaining the current maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Find the pair (i,j) that actually gives the maximal product at position k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # Choose a source (the larger of the two) and a random target
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # Transfer a modest amount of mass
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a_snapshot = a.copy()
        conv_snapshot = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill_iter={hill_iter} new best {cur_score:.6f}")
        else:
            # revert
            a = a_snapshot
            conv = conv_snapshot

        # occasional jitter to escape stagnation
        if hill_iter % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill_iter={hill_iter}")

        if time.time() + 0.5 >= deadline:
            break

    final_score = 2.0 * n * conv.max()
    print(f"[{time.time():.1f}] final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
