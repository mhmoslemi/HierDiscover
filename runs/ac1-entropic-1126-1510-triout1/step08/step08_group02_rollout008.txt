   * mixtures w·arcsine+(1‑w)·uniform  (w∈{0.2,0.4,0.6,0.8}).

   quantity ``2·n·max(a*a)`` (identical to ``evaluate_sequence`` for sum = 1).

   cheap scores.

3. **Heavy refinement** – For each seed run three first‑order schemes,

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using log‑sum‑exp:
    f_eps(a) = eps * log Σ_i exp(conv_i/eps).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    maxc = conv.max()
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

    best_score = _max_convolution_score(a)

    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        if t % 50_000 == 0 and eps > 1e-7:
            eps *= 0.5

        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.6):
    """Plain projected sub‑gradient descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic / parametric families

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on the discrete grid."""
    x = (np.arange(n) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    return w / w.sum()


def _beta_distribution(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on the discrete grid."""
    x = (np.arange(n) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if w.sum() == 0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    return w / w.sum()


def _powerlaw_distribution(n: int, beta: float) -> np.ndarray:
    """U‑shaped power‑law a_i ∝ [(i+½)(n‑i‑½)]^{-beta}."""
    i = np.arange(n, dtype=np.float64)
    left = i + 0.5
    right = n - i - 0.5
    vals = (left * right) ** (-beta)
    return vals / vals.sum()


def _mixture_arcsine_uniform(n: int, w: float) -> np.ndarray:
    """Convex mixture w·arcsine + (1‑w)·uniform."""
    a = _arcsine_distribution(n)
    u = _uniform_distribution(n)
    mix = w * a + (1.0 - w) * u
    return mix / mix.sum()


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns a probability vector (sum = 1).  The routine respects the
    supplied time budget and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Determine a sensible base length – fallback to 3500 if unknown.
    try:
        base_n = len(height_sequence_1)       # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # --------------------------------------------------------------
    # Lengths to explore
    lengths = set()
    for f in (0.8, 0.9, 1.0, 1.1, 1.2, 1.5, 2.0):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    # a few random lengths up to 3×base_n (capped at 18000)
    for _ in range(5):
        n = np.random.randint(500, min(18000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # --------------------------------------------------------------
    # 1. Broad sampling
    candidates = []   # tuple (array, cheap_score)

    # Parameter grids
    beta_grid = [0.3, 0.4, 0.5, 0.6, 0.8, 0.9]
    beta_full_grid = [(p, q) for p in [0.3, 0.5, 0.8] for q in [0.3, 0.5, 0.8]]
    w_grid = [0.2, 0.4, 0.6, 0.8]
    dirichlet_alphas = [0.2, 0.5, 0.8, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # --- Fixed analytic families ------------------------------------------------
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_uniform_distribution, "uniform"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_semicircle_distribution, "semicircle"),
        ]
        for gen, name in fams:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} {name:12s} score={score:.6f}")

        # --- Symmetric Beta(p,p) ----------------------------------------------------
        for p in beta_grid:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} beta_sym(p={p:.2f})   score={score:.6f}")

        # --- Full Beta(p,q) ---------------------------------------------------------
        for p, q in beta_full_grid:
            a = _beta_distribution(n, p, q)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} beta(p={p:.2f},q={q:.2f}) score={score:.6f}")

        # --- Power‑law U‑shape -------------------------------------------------------
        for beta in beta_grid:
            a = _powerlaw_distribution(n, beta)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} powerlaw(beta={beta:.2f}) score={score:.6f}")

        # --- Mixture arcsine / uniform -----------------------------------------------
        for w in w_grid:
            a = _mixture_arcsine_uniform(n, w)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} mix_arc_uni(w={w:.2f})   score={score:.6f}")

        # --- Random Dirichlet draws --------------------------------------------------
        for alpha in dirichlet_alphas:
            for _ in range(2):
                if time.time() >= deadline:
                    break
                a = _random_dirichlet(n, alpha=alpha)
                score = _max_convolution_score(a)
                candidates.append((a, score))
                print(f"[{time.time():.1f}] n={n:5d} Dirichlet(α={alpha:.2f})   score={score:.6f}")

    # --------------------------------------------------------------
    # 2. Add the historical best (if present) as a seed
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Keep the top‑k candidates (more seeds than baseline)
    candidates.sort(key=lambda x: x[1])
    top_k = 30
    top_candidates = candidates[:top_k]
    print(f"Selected {len(top_candidates)} seeds for refinement; best cheap score={top_candidates[0][1]:.6f}")

    # --------------------------------------------------------------
    # 4. Heavy refinement on each seed
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        # a) Exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=150_000,
                                        eta0=0.9)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} exp‑grad → {sc:.6f}")

        # b) Adam on exact sub‑gradient
        seq, sc = _adam_descent(seed_seq, deadline,
                                max_iters=150_000,
                                step0=0.25)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam → {sc:.6f}")

        # c) Adam on smoothed max
        seq, sc = _smooth_adam_descent(seed_seq, deadline,
                                       max_iters=120_000,
                                       eps0=1e-3,
                                       step0=0.20)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smooth‑Adam → {sc:.6f}")

    # --------------------------------------------------------------
    # 5. Long sub‑gradient descent on the current best
    subgrad_deadline = deadline - 30.0   # reserve ~30 s for hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=2_000_000,
                                       eta0=0.6)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # --------------------------------------------------------------
    # 6. Guided hill‑climbing with simulated annealing
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02                     # 2 % of total mass initially
    start_hill = time.time()
    init_temp = 1e-4
    final_temp = 1e-8

    while time.time() < deadline:
        iteration += 1
        # geometric decay of transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 8_000.0))

        # choose source and destination indices
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        # temperature schedule based on elapsed fraction
        frac = (time.time() - start_hill) / (deadline - start_hill + 1e-12)
        temp = init_temp * (1 - frac) + final_temp * frac

        accept = False
        if new_score < cur_score:
            accept = True
        elif np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional random jitter to shake the system
        if iteration % 4_000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
