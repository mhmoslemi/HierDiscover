"""
Optimise a non‑negative step function for the ``evaluate_sequence`` metric.

The optimiser works in three stages:

1. **Fast analytic seeding** – generate a handful of parametric families
   (arcsine, beta‑type, power‑transformed arcsine, a uniform‑arcsine mixture
   and a few random Dirichlet draws).  Every seed is ℓ₁‑normalised
   (sum = 1) and scored with the exact surrogate
   ``2·n·max(conv(a,a))`` (identical to the true evaluation when the sum is 1).

2. **Heavy‑weight local search** – the best few seeds are polished with a
   standard Adam optimiser (projected onto the simplex).  This reduces the
   peak of the autocorrelation dramatically and provides a high‑quality
   starting point for the final stage.

3. **Exact pairwise mass‑transfer with line‑search** – repeatedly move mass from
   the index with the largest sub‑gradient (worst contributor to the max‑convolution)
   to the index with the smallest sub‑gradient.  The optimal transfer amount
   is found by a cheap ternary search on the true max‑convolution (the update
   can be evaluated analytically in O(n) using the current convolution).  The
   loop runs until the time budget expires or no improvement is possible.
   Occasionally a tiny random jitter is added to escape plateaus.

The whole routine respects the supplied time budget (≈ 1000 s) and returns the
best vector (as a plain list of floats).  All numbers are non‑negative and the
vector is automatically ℓ₁‑normalised before returning, satisfying the checks
in ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (sum(a)=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _shift_convolve(a: np.ndarray, shift: int) -> np.ndarray:
    """
    Return the convolution a * δ_{shift} (i.e. a shifted by ``shift``) as a
    length‑2·n‑1 vector.
    """
    n = a.shape[0]
    out = np.zeros(2 * n - 1, dtype=a.dtype)
    out[shift:shift + n] = a
    return out


def _max_b_with_alpha(b: np.ndarray, c: np.ndarray,
                      d_idx: np.ndarray, d_val: np.ndarray,
                      alpha: float) -> float:
    """Compute max(b + 2·α·c + α²·d) efficiently."""
    arr = b + 2.0 * alpha * c
    if d_idx.size:
        arr[d_idx] += (alpha * alpha) * d_val
    return arr.max()


def _ternary_search_alpha(b: np.ndarray, c: np.ndarray,
                          d_idx: np.ndarray, d_val: np.ndarray,
                          lo: float, hi: float,
                          iters: int = 25) -> float:
    """Find α∈[lo,hi] minimizing max_b_with_alpha using ternary search."""
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = _max_b_with_alpha(b, c, d_idx, d_val, m1)
        f2 = _max_b_with_alpha(b, c, d_idx, d_val, m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (symmetric)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_transform(base: np.ndarray, p: float) -> np.ndarray:
    """Raise a non‑negative vector to a power and renormalise."""
    a = np.power(base, p)
    s = a.sum()
    if s == 0:
        return base.copy()
    return a / s


def _product_power_distribution(n: int, alpha: float) -> np.ndarray:
    """Law a_i ∝ ((i+½)*(n‑i‑½))^{‑α}."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    # use logs for stability when α is large
    a = np.exp(-alpha * np.log(prod))
    a /= a.sum()
    return a


def _uniform_distribution(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)
