def _exponentiated_gradient(a, deadline, start_time):
    n = a.shape[0]
    best_a = a.copy()
    best_score = _objective(a)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = np.where(b >= maxb - eps)[0]

        # Compute gradient
        g = np.zeros_like(a)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            g[i] += 2 * a[j]

        # Avoid zero gradient everywhere: add small epsilon
        g = g + 1e-12

        # schedule eta decreasing linearly from eta0 to eta_min
        elapsed = time.time() - start_time
        total = deadline - start_time
        eta0 = 5.0
        eta_min = 0.05
        eta = eta0 * (1 - elapsed/total) + eta_min * (elapsed/total)

        # multiplicative update
        exp_factor = np.exp(-eta * g)
        a = a * exp_factor
        # symmetrize and project onto simplex
        a = (a + a[::-1]) * 0.5
        a = _project_onto_simplex(a, 1.0)

        score = _objective(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{time.time():.1f}] EG iter={iteration} score={score:.6f}")

        # optional early stop if no improvement for many iters
        if iteration % 500 == 0 and time.time() > deadline:
            break

    return best_a, best_score
```

"""
Propose a step‑function (non‑negative sequence with sum=1) that minimizes
``evaluate_sequence``.  The method combines three stages:

1. **Rich initialization** – a catalogue of analytic families (uniform,
   arcsine, symmetric beta, exponential, alternating, random Dirichlet,
   sparse spikes) is generated and the best candidate (lowest max‑self‑convolution)
   is kept.

2. **Exponentiated‑gradient descent** – the L∞‑objective is smoothed by
   using the exact sub‑gradient of the active convolution shifts.
   A mirror‑descent (multiplicative) update
   `a_i ← a_i·exp(-η·g_i)` is applied, followed by projection onto the
   probability simplex and explicit symmetrisation.  The step size η is
   gradually reduced.

3. **Greedy active‑set optimisation** – a fast pairwise mass‑transfer
   routine (as in the reference code) finishes the search, and a short
   simulated‑annealing walk provides fine‑tuning.

All stages respect the global time budget (default 1000 s).  The returned
list satisfies the requirements of ``evaluate_sequence`` and works for any
length up to several ten‑thousands of elements.
"""

import time
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force a vector to be symmetric: v ← (v+v[::-1])/2."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a))  – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex (concentration = alpha)."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse set of seed vectors for length n."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta – wide range of shape parameters (including extreme U‑shapes)
    for p in (0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05,
              0.07, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – several decay rates
    for lam in (0.0002, 0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4, 5, 6):
        cand.append(_distribution_alternating(n, stride))

    # random Dirichlet draws (moderate concentration)
    for _ in range(6):
        alpha = rng.uniform(0.05, 0.5)
        cand.append(_random_dirichlet(n, rng, alpha))

    # sparse random start (~0.3 % non‑zeros)
    k = max(2, n // 300)
    rnd = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    rnd[idx] = rng.random(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    # noisy perturbations of a few beta shapes
    for _ in range(4):
        p = rng.choice([0.01, 0.03, 0.07, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.01, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, 1.0)
        cand.append(pert)

    return cand

def _exponentiated_gradient(a: np.ndarray,
                           deadline: float,
                           start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) update on the simplex using the exact
    sub‑gradient of the active convolution shifts.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = np.where(b >= maxb - eps)[0]

        # sub‑gradient g_i = 2 * Σ_{k∈active} a_{k-i}
        g = np.zeros_like(a)
        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        # ensure positivity (avoid zero exponent)
        g += 1e-12

        # linear schedule for η (large → small)
        elapsed = time.time() - start_time
        total = deadline - start_time
        eta0 = 5.0
        eta_min = 0.05
        eta = eta0 * (1.0 - elapsed / total) + eta_min * (elapsed / total)

        # multiplicative mirror‑descent step
        factor = np.exp(-eta * g)
        a = a * factor

        # enforce symmetry, non‑negativity and the simplex constraint
        a = _symmetrize(a)
        a = _project_onto_simplex(a, 1.0)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] EG it={it} improved to {best_score:.6f}")

    return best, best_score

            best_score = new_score

        # try a smaller transfer if the full move fails
        delta *= 0.2
        if delta < 1e-9:

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        a_new = _symmetrize(a_new)
        a_new = _project_onto_simplex(a_new, 1.0)
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new

def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small‑step Metropolis walk with exponential cooling.  The proposal
    respects symmetry: a transfer from i to j is mirrored with a transfer
    from n‑1‑i to n‑1‑j.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score
    T = 0.03

    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n // 2 + 1)               # only need one side because of symmetry
        if cur_a[i] <= 1e-12:
            continue
        # pick a partner j on the opposite side (ensure i≠j)
        j = rng.integers(n // 2 + 1, high=n)
        if j == i:
            continue

        # transfer a fraction of the source mass
        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # mirror the move
        mi = n - 1 - i
        mj = n - 1 - j
        new_a[mi] -= delta
        new_a[mj] += delta

        new_a = _project_onto_simplex(new_a, 1.0)

        new_score = _objective(new_a)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        T *= 0.9997

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a length – larger lengths give finer discretisation.
    #  8000 works well within the time budget on a single CPU.
    n = 8000
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 1) generate a broad set of starting vectors
    candidates = _initial_candidates(n, rng)

    # include the previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1:
            prev = prev / prev.sum()
            prev = _symmetrize(prev)
            candidates.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) pick the best initial candidate (lowest objective)
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        cand = _symmetrize(cand)
        cand = _project_onto_simplex(cand, 1.0)
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx} score={sc:.6f}")

    if best_vec is None:
        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) exponentiated‑gradient phase (≈55 % of budget)
    phase1_deadline = start_time + 0.55 * (deadline - start_time)
    best_vec, best_score = _exponentiated_gradient(best_vec, phase1_deadline, start_time)

    # ------------------------------------------------------------------
    # 4) aggressive greedy optimisation (≈30 % of budget)
    phase2_deadline = phase1_deadline + 0.30 * (deadline - start_time)
    best_vec, best_score = _greedy_local_search(best_vec, phase2_deadline, rng)

    # ------------------------------------------------------------------
    # 5) simulated‑annealing fine‑tuning (remaining time)
    if time.time() < deadline:
        best_vec, best_score = _simulated_annealing(best_vec, deadline, rng)

    # ------------------------------------------------------------------
    # final sanity – enforce symmetry and simplex feasibility
    best_vec = _symmetrize(best_vec)
    best_vec = _project_onto_simplex(best_vec, 1.0)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
