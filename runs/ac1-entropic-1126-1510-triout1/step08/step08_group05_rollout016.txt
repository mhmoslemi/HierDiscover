"""
Propose a candidate step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The search consists of three stages:

1. **Broad initialization** – many analytic families (uniform, exponential,
   power‑law, symmetric/asymmetric beta, random Dirichlet draws) are sampled
   for several lengths and the best one (according to the
   2·n·max (a*a) score) is kept.

2. **Iterated linear‑programming refinement** – for a fixed length ``n`` the
   current vector ``a`` (∑a=1) defines a bound ``M = max(a*a)``.
   We solve the LP

        maximise   ∑b_j
        subject to  (a * b)_k ≤ M   for all k
                    b_j ≥ 0

   The solution ``b`` tells us where additional mass can be placed without
   increasing the current worst convolution value.  If ``∑b > 1`` we
   normalise ``b`` to a probability vector ``g`` and mix it with ``a``.
   The mixing factor is chosen by a tiny line‑search; the procedure is
   repeated until no improvement is possible or time runs out.

3. **Fast simulated‑annealing fine‑tuning** – a cheap O(n) mass‑transfer
   move is repeatedly applied directly to the full self‑convolution.
   The move is accepted if it lowers the score or with a temperature‑controlled
   probability.  This clean‑up phase usually squeezes a few extra digits.

The whole routine respects a 1000 s wall‑clock budget and returns the best
vector found (as a plain Python list).  The code only uses NumPy, SciPy
and the already‑imported ``evaluate_sequence`` function.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector normalised to Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Guard against log(0)
    eps = 1e-30
    w = np.exp((p - 1) * np.log(x + eps) + (q - 1) * np.log(1 - x + eps))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sparse_random(n, k_frac=0.01):
    """k non‑zero entries drawn uniformly, normalised."""
    a = np.zeros(n, dtype=np.float64)
    k = max(2, int(k_frac * n))
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    if delta <= 0 or i == j:
        return
    # keep a copy of the old a for the linear part of the update
    a_old = a.copy()
    # remove contributions of the transferred mass from the old source location
    conv[i:i + a.size] -= 2.0 * delta * a_old
    # add contributions to the new target location
    conv[j:j + a.size] += 2.0 * delta * a_old
    # update the quadratic self‑terms (a_i^2 part)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    # move the mass
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _build_conv_matrix(a):
    """
    Build the (2n‑1)×n matrix A such that (A @ b) equals np.convolve(a, b)
    (full mode).  The matrix is dense but small enough for n ≲ 3000.
    """
    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n - 1)])
    row = np.concatenate([a[:1], np.zeros(n - 1)])
    return toeplitz(col, row)

                cur_score = 2.0 * n * conv.max()
                it_no_improve = 0
                print(f"[{time.time():.1f}] n={n} jitter applied at SA iter {it}")

        # ------------------------------------------------------------------
        # Record the best result for this length
        final_score = 2.0 * n * conv.max()
        print(f"[{time.time():.1f}] length={n} final best ≈ {final_score:.6f}")
        if final_score < best_score:
            best_score = final_score
            best_seq = best_local_seq.copy()
            best_n = n

    # ----------------------------------------------------------------------
    # 4. If a previously computed solution exists, give it a quick try
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Safety net – should never happen
    if best_seq is None:
        n = 1024
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    print(f"=== DONE === best length={best_n} best score≈{best_score:.6f}")
    return best_seq.tolist()
