"""
Propose a high‑quality step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.
The algorithm builds a rich set of analytic seeds (uniform, arcsine,
Beta(p,p), power‑law edges, exponentials, etc.) and explores a fine grid of
shape parameters (p down to 0.01, γ up to 0.49).  For the most promising
seeds it runs three optimisation phases:

1. **Smoothed Adam** on a log‑sum‑exp approximation of the max‑convolution.
   The smoothing parameter α is increased gradually while the learning‑rate
   decays.  The iterate is always projected onto the probability simplex.

2. **Projected sub‑gradient descent** on the true (non‑smoothed) max‑convolution.
   The gradient is 2·a reversed and shifted at the active (max‑valued) shift(s).

3. **Simulated‑annealing pairwise swaps** that transfer a random amount of
   mass from one index to another, accepting both improving and, with a
   decreasing temperature, worsening moves.

All phases respect a global deadline (default 1000 s).  The best vector
found is returned as a plain Python list of floats.

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) under the constraint sum(a)=1.
    a must be non‑negative and already normalised.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a normalised vector)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_powerlaw_edges(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-γ}  (γ∈(0,0.5))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = i * (n - i)
    denom = np.where(denom <= 0, 1e-12, denom)
    vals = denom ** (-gamma)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    """Half‑sine shape, zero at the edges, positive inside."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.clip(vals, 0, None)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_dense(n: int, rng: np.random.Generator) -> np.ndarray:
    """Completely dense random vector."""
    vals = rng.random(n)
    if vals.sum() == 0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a wide family of start vectors for length *n*."""
    cand = []

    # classical families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_sine(n))

    # Beta(p,p) – really fine‑grained grid, including very small p
    for p in (0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2,
              0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Exponential – broad range of decay rates
    for lam in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential(n, lam))

    # Power‑law edges – approach the arcsine exponent from below
    for gamma in (0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35,
                  0.4, 0.45, 0.48, 0.49):
        cand.append(_distribution_powerlaw_edges(n, gamma))

    # Alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # Two‑block spikes
    for block_frac in (0.03, 0.05, 0.07, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # Sparse and dense random starts
    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    cand.append(_distribution_random_sparse(n, rng, density=0.03))
    cand.append(_distribution_random_dense(n, rng))

    # Small random convex combinations of two families
    # (pick two random seeds and mix with a random weight)
    for _ in range(4):
        a = rng.choice(cand)
        b = rng.choice(cand)
        w = rng.random()
        mixed = w * a + (1 - w) * b
        mixed = mixed / mixed.sum()
        cand.append(mixed)

    return cand

def _smooth_adam(a: np.ndarray, deadline: float,
                 best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) objective."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule – we go a little farther than before
    schedule = [
        (0.5, 1500, 0.20),
        (1.0, 1500, 0.15),
        (5.0, 1500, 0.12),
        (20.0, 1500, 0.10),
        (100.0, 1000, 0.08),
        (500.0, 800, 0.06),
        (2000.0, 600, 0.04),
        (5000.0, 500, 0.03),
        (10000.0, 400, 0.025),
    ]

    for alpha, steps, lr in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth‑max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 400 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected (sub)‑gradient descent on the true max‑convolution objective.
    Uses a diminishing step size.
    """
    n = a.shape[0]
    step = 0.5          # initial step size on the simplex
    decay = 0.995       # geometric decay
    min_step = 1e-8

    while time.time() < deadline and step > min_step:
        b = _fft_self_convolve(a)
        M = b.max()
        # indices attaining the maximum (within a tiny tolerance)
        active = np.where(b >= M - 1e-12)[0]
        w = np.zeros_like(b)
        w[active] = 1.0 / len(active)

        # gradient of max‑conv (sub‑gradient)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        a_new = a - step * grad
        a_new = _project_onto_simplex(a_new, 1.0)

        cur = _objective(a_new)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a_new.copy()
            print(f"[{time.time():.1f}] subgrad new best {best_score:.6f}")

        a = a_new
        step *= decay

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style random mass transfers.
    Accepts improvements always; accepts worsening moves with a temperature
    that decays exponentially.
    """
    n = a.shape[0]
    T0 = 0.02               # initial temperature
    T = T0
    decay = 0.9999

    while time.time() < deadline:
        # choose a source index that currently carries mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to 50 % of the source mass
        delta = a[i] * np.random.rand() * 0.5
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
            T = T0                     # boost temperature after a true improvement
        else:
            # possibly accept a worsening move
            prob = math.exp(-(cur - best_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                a = new_a
        T *= decay
        if T < 1e-9:
            T = 1e-9

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point – search for a non‑negative step function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best vector found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Coarse scan on modest lengths (fast) to spot good shape families.
    short_lengths = [2000, 4000, 8000]
    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    for n in short_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning short length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best length={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep a limited number of elite seeds
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2. Optional use of a previously discovered construction.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. If nothing found (should not happen), fall back to uniform.
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation on larger, more expressive lengths.
    long_lengths = [15000, 25000, 40000]

    for n in long_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] optimisation on length {n}")

        # generate a fresh pool of candidates for this size
        local_pool: list[tuple[float, np.ndarray]] = []
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            local_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best length={n} score={sc:.6f}")

        # keep the very best seeds (including the best seen so far)
        local_pool.sort(key=lambda x: x[0])
        seeds = [seq for _, seq in local_pool[:3]]
        if overall_best_seq is not None and not any(np.allclose(overall_best_seq, s) for s in seeds):
            seeds.append(overall_best_seq.copy())

        # ------------------------------------------------------------------
        # 4a – smoothed Adam descent for each seed
        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + min(25.0, deadline - time.time())
            best_seq, best_score = _smooth_adam(seed_seq.copy(),
                                                sub_deadline,
                                                best_seq=overall_best_seq,
                                                best_score=overall_best_score)
            if best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after Adam length={n} new best {best_score:.6f}")

            # ------------------------------------------------------------------
            # 4b – sub‑gradient descent on the true objective
            sub_deadline = time.time() + min(15.0, deadline - time.time())
            best_seq, best_score = _subgradient_descent(best_seq.copy(),
                                                        sub_deadline,
                                                        best_seq=overall_best_seq,
                                                        best_score=overall_best_score)
            if best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after sub‑grad length={n} new best {best_score:.6f}")

            # ------------------------------------------------------------------
            # 4c – final simulated‑annealing pairwise swaps
            best_seq, best_score = _pairwise_swap_local(best_seq.copy(),
                                                        deadline,
                                                        best_seq=overall_best_seq,
                                                        best_score=overall_best_score)
            if best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after swaps length={n} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
