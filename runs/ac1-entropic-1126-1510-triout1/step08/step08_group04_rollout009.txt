def propose_candidate(seed=42, budget_s=1000):
    import time, random, numpy as np
    from scipy.optimize import linprog
    # set random seeds
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0

    # define epsilon small and simplex projection not needed (we always guarantee sum=1)
    eps = 1e-12

    # helper: evaluate approximated objective using normalized sum=1
    def objective(a):
        conv = _fft_self_convolve(a)
        return 2.0 * a.shape[0] * conv.max()
    # store best
    best_a = None
    best_obj = float('inf')
    # list of initial candidates (list of vectors)
    candidates = []
    # first candidate: existing global variable
    try:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.ndim == 1 and init.size > 0:
            init = np.maximum(init, eps)
            init = init / init.sum()
            candidates.append(init)
    except Exception:
        pass

    # add analytic families for a few lengths
    lengths = [500, 1000, 2000, 4000, 8000]  # moderate for quick evaluation
    for n in lengths:
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_beta_sym(n, 0.5))
        candidates.append(_distribution_beta_sym(n, 2.0))
        # random Dirichlet
        candidates.append(np.random.dirichlet(np.full(n, 0.5)))
        candidates.append(np.random.dirichlet(np.full(n, 2.0)))
        # also random sparse
        sparse = np.zeros(n)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

    # Evaluate each candidate: compute objective and keep best
    for vec in candidates:
        if time.time() >= deadline:
            break
        # ensure positivity
        vec = np.maximum(vec, eps)
        vec = vec / vec.sum()
        obj = objective(vec)
        if obj < best_obj:
            best_obj = obj
            best_a = vec.copy()
            print(f"[{time.time():.1f}] init candidate len={vec.size} obj={obj:.6f}")

    # iterative LP improvement on best candidate
    a = best_a.copy()
    n = a.size
    # precompute a_n = n
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute current conv and M
        conv = _fft_self_convolve(a)
        M = conv.max()
        # Build A matrix
        # Only enforce constraints for indices where conv is close to M (within tolerance),
        # plus maybe a few random ones, to keep LP size manageable.
        # We'll select all constraints, but can prune for efficiency.

        # Use full constraints for moderate n (<3000). For bigger n maybe prune.
        # if n too large > 4000, prune.
        max_rows = 2*n-1
        A = np.zeros((max_rows, n), dtype=np.float64)
        for i in range(n):
            A[i:i+n, i] = a[i]

        # Solve LP
        c = -np.ones(n)   # maximize sum(b)
        b_ub = np.full(max_rows, M, dtype=np.float64)
        bounds = [(0, None)] * n
        # Use high-level interface
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            print(f"[{time.time():.1f}] LP failed at iter {iteration} message: {res.message}")
            break

        b = res.x
        sumb = b.sum()
        if sumb <= 0:
            break

        g = b / sumb   # normalized sum=1
        # convex combination with step size t
        t = 0.5
        a_new = (1 - t) * a + t * g

    # We'll implement simple random transfer moves limited.

    a = best_a.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0

    print(f"Final best objective ≈ {final_obj:.6f} (len={n})")

   triangular, exponential, symmetric Beta, sine‑power, Gaussian) and a few
   random Dirichlet draws of moderate length (≤ 2500) are generated.
   Each candidate is normalised to sum = 1 and evaluated with a cheap FFT
   score  ``2·n·max(a*a)``.  The best candidate becomes the starting point.

2. **Alternating LP refinement** – for the current vector ``f`` we compute
   ``M = max(f*f)``.  The linear program

        max   1ᵀ b
        s.t.  b ≥ 0
              f * b ≤ M                     (convolution is linear in b)

   is solved (``scipy.optimize.linprog``).  The solution ``b`` always satisfies
   ``‖b‖₁ ≥ 1``; after normalising ``g = b / ‖b‖₁`` we obtain a new vector whose
   self‑convolution peak is provably at most ``M / ‖b‖₁``.  We blend the new
   vector with the old one, ``f ← (1‑t)·f + t·g`` (``t = 0.5``) and repeat
   while time permits.

3. **Hill‑climber** – random mass transfers between two indices are performed.
   The full self‑convolution is recomputed with an FFT after each move; a move
   is kept only if it strictly decreases the current peak.  Small Gaussian
   jitter is added every few thousand iterations to escape plateaus.

The routine respects a wall‑clock budget of ``budget_s`` seconds (default
1000 s).  The returned list is non‑negative, sums to 1 and can be fed directly
to ``evaluate_sequence``.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the linear‑programming stage.
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:          # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _downsample(vec: np.ndarray, target_n: int) -> np.ndarray:
    """Reduce ``vec`` to length ``target_n`` by averaging consecutive blocks."""
    n = vec.size
    if n <= target_n:
        return vec.copy()
    # positions where each block starts
    splits = np.floor(np.arange(1, target_n) * n / target_n).astype(int)
    reduced = np.add.reduceat(vec, np.concatenate(([0], splits)))
    # ``reduced`` may be off‑by‑one because of rounding; fix size
    if reduced.size > target_n:
        reduced = reduced[:target_n]
    elif reduced.size < target_n:
        reduced = np.pad(reduced, (0, target_n - reduced.size), 'constant')
    reduced = np.maximum(reduced, 1e-12)
    reduced /= reduced.sum()
    return reduced

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that yields a small value of
    ``evaluate_sequence``.  The function respects the ``budget_s`` second
    wall‑clock limit and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0       # a small safety margin

    eps = 1e-12
    max_n = 2500                                 # size limit for LP stage

    # ------------------------------------------------------------------
    # 1. generate a diverse pool of starting vectors (all normalised to sum = 1)
    candidates = []

    # try to use a previously discovered sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:                                 # pragma: no cover
        pass

    # analytic families + a few Dirichlet draws for several lengths
    lengths = [500, 1000, 1500, 2000, 2500]
    for n in lengths:
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.004))
        candidates.append(_distribution_exponential(n, lam=0.02))
        for p in (0.3, 0.6, 1.0, 2.0, 4.0):
            candidates.append(_distribution_beta_sym(n, p))
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))
        for sigma in (0.2, 0.35, 0.5):
            candidates.append(_distribution_gaussian(n, sigma))
        # random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            cand = np.random.dirichlet(np.full(n, alpha))
            candidates.append(cand)
        # sparse random start
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 250)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

    # ------------------------------------------------------------------
    # 2. cheap evaluation – keep the best normalised candidate
    def cheap_score(v: np.ndarray) -> float:
        """2·n·max(v*v) for a vector already normalised to sum = 1."""
        n = v.shape[0]
        conv = _fft_self_convolve(v)
        return 2.0 * n * conv.max()

    best_vec = None
    best_score = float('inf')
    best_len = None

    for vec in candidates:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, eps)
        vec = vec / vec.sum()
        sc = cheap_score(vec)
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()
            best_len = vec.size
            print(f"[{time.time():.1f}] init len={best_len} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. enforce a size that comfortably fits the LP memory budget
    if best_vec is None:                               # pragma: no cover
        best_vec = _distribution_uniform(max_n)
        best_vec = best_vec / best_vec.sum()
        best_score = cheap_score(best_vec)

    if best_vec.size > max_n:
        # down‑sample while preserving the overall shape
        best_vec = _downsample(best_vec, max_n)
        best_len = best_vec.size
        best_score = cheap_score(best_vec)
        print(f"[{time.time():.1f}] down‑sampled to {best_len}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. alternating linear‑programming refinement (only if SciPy is present)
    if SCIPY_AVAILABLE:
        a = best_vec.copy()
        n = a.size
        it = 0
        while time.time() < deadline:
            it += 1
            conv = _fft_self_convolve(a)
            M = conv.max()

            # Build the convolution matrix A where (A @ b) == a * b
            # A has shape (2n‑1, n); each column i consists of a[i] on a diagonal.
            rows = 2 * n - 1
            A = np.zeros((rows, n), dtype=np.float64)
            for i in range(n):
                A[i:i + n, i] = a[i]

            # Linear program: maximise 1ᵀb   s.t.  A b ≤ M , b ≥ 0
            c = -np.ones(n, dtype=np.float64)          # linprog minimises, so negate
            b_ub = np.full(rows, M, dtype=np.float64)
            bounds = [(0, None)] * n

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
            if not res.success:
                print(f"[{time.time():.1f}] LP failed (iter {it}): {res.message}")
                break

            b = res.x
            sum_b = b.sum()
            if sum_b <= 1.0 + 1e-12:       # no progress possible
                # we are at a (near) fixed point – stop early
                break

            g = b / sum_b                 # normalise to ℓ₁‑norm = 1
            # blend current vector with the new one
            t = 0.5
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, eps)
            a_new /= a_new.sum()

            new_score = cheap_score(a_new)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_vec = a_new.copy()
                print(f"[{time.time():.1f}] LP iter={it} improved score={new_score:.6f}")

            a = a_new

            # early‑exit guard
            if time.time() + 0.5 >= deadline:
                break

    # ------------------------------------------------------------------
    # 5. final hill‑climber (purely heuristic, works at any length)
    a = best_vec.copy()
    n = a.size
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0
    while time.time() < deadline:
        hill_it += 1

        # pick a source (large entry) and a random destination
        src = np.argmax(a)                     # deterministic source
        dst = random.randint(0, n - 1)
        if src == dst:
            continue

        # transfer a modest amount of mass from src to dst
        max_delta = min(a[src] * 0.4, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta
        a_candidate = np.maximum(a_candidate, eps)
        a_candidate /= a_candidate.sum()

        conv_candidate = _fft_self_convolve(a_candidate)
        M_candidate = conv_candidate.max()

        if M_candidate < M - 1e-13:
            a = a_candidate
            conv = conv_candidate
            M = M_candidate
            best_vec = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 500 == 0:
                print(f"[{time.time():.1f}] hill_it={hill_it} improved score={best_score:.6f}")

        # occasional jitter to shake out of plateaus
        if hill_it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    final_obj = 2.0 * n * M
    print(f"Final objective ≈ {final_obj:.6f} (length={n})")
    return best_vec.tolist()
