"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Length selection** – try a few lengths around the previous best
   (or a default of 3500).  The objective 2·n·max(a*a) is asymptotically
   independent of n, but a different discretisation can give a slightly
   lower max‑convolution.

2. **Parametric mixture model** – a probability vector is built as a
   convex combination of three elementary families:
       * arcsine  (Beta(½,½))
       * symmetric beta  (Beta(p,p))
       * exponential   (exp(−λ·|i−centre|))

   The three mixture weights (soft‑max of three free numbers) and the
   two shape parameters (p, λ) are optimised with a cheap surrogate
   ``_max_convolution_score`` (the exact objective after normalising
   ∑a=1).  Optimisation is performed by *differential evolution* (small
   population, a few generations) which is fast enough to explore many
   lengths.

3. **Local refinement** – the best mixture found for each length is
   further optimised on the simplex by a continuation method that
   smooths the max‑operator with a soft‑max (log‑sum‑exp).  The method
   starts with a small smoothing parameter γ and slowly increases it,
   using projected gradient descent with a diminishing step size.

4. **Fine‑grained hill climbing** – after the smooth descent a cheap
   random mass‑transfer hill‑climber with geometric decay of the step
   size is applied.  Occasionally a tiny Gaussian jitter is added to
   escape plateaus.

All steps respect the global time budget (default 1000 s).  The
returned list already satisfies the constraints of ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) for a simplex‑normalised vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all already normalised to sum 1)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on the grid (i+0.5)/n, p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow/underflow
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw, where=np.isfinite(logw), out=np.zeros_like(logw))
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return w / s


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    # soft‑max weights, stabilised by shifting by b_max
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 0.5,
                             gamma_factor: float = 1.001):
    """Continuation: start with a smooth max (small γ) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score


# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, score: float, deadline: float) -> tuple[np.ndarray, float]:
    """Tiny random mass transfers with geometric decay."""
    n = a.shape[0]
    cur_score = score
    best_seq = a.copy()
    best_score = score
    it = 0
    max_transfer = 0.03  # up‑to 3 % of total mass initially

    while time.time() < deadline:
        it += 1
        # decay the allowed transfer size
        transfer = max_transfer * (0.5 ** (it / 12_000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 5_000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_seq, best_score

def _optimize_mixture_for_n(n: int, deadline: float):

    # objective for DE – cheap surrogate (already assumes sum=1)

        p = np.exp(logp)
        lam = np.exp(loglam)

             w[1] * _beta_symmetric_distribution(n, p) +
             w[2] * _exponential_distribution(n, lam))
        return _max_convolution_score(a)

    # bounds – log‑weights are unrestricted, p∈[0.05,5], λ∈[3e‑4,0.12]
    bounds = [(-5, 5), (-5, 5), (-5, 5), (-3, 1.6), (-9.5, -2.3)]

    popsize = 5

    # run DE

                                    maxiter=max_iter,
                                    popsize=popsize,
                                    polish=False,
                                    seed=np.random.randint(0, 2**32 - 1),
                                    updating='deferred',
                                    workers=1)

    # reconstruct the mixture vector
    logw0, logw1, logw2, logp, loglam = result.x
    w_raw = np.exp([logw0, logw1, logw2])
    w = w_raw / w_raw.sum()
    p = np.exp(logp)

         w[1] * _beta_symmetric_distribution(n, p) +
         w[2] * _exponential_distribution(n, lam))
