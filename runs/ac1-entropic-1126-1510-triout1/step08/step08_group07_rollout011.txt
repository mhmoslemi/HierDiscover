def _lp_optimize(a: np.ndarray):
    n = len(a)
    # Build A_ub: shape (2n-1, n+1)
    # We'll use scipy.sparse.lil_matrix for incremental building.

    from scipy.sparse import lil_matrix

    rows = 2*n - 1
    A = lil_matrix((rows, n+1), dtype=np.float64)
    # for each row k, fill columns j where 0 <= k-j < n => j in [max(0, k-(n-1)), min(k, n-1)]
    for k in range(rows):
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # for j from i_low to i_high inclusive: coefficient a_{k-j}
        # j corresponds to column index.
        # a_{k - j} = a_{i}, where i = k - j.
        # we can fill row k for each j.
        for j in range(i_low, i_high + 1):
            i = k - j
            A[k, j] = a[i]
        A[k, n] = -1.0  # coefficient for t

    # b_ub = zeros
    b_ub = np.zeros(rows, dtype=np.float64)

    # equality constraint sum(g) = 1
    A_eq = np.zeros((1, n+1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # bounds
    bounds = [(0, None)] * n + [(0, None)]

    res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),  # minimize t
                  A_ub=A, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})

    if not res.success:
        return None, None
    g = res.x[:n]
    t = res.x[-1]
    return g, t

"""
Greedy‑pairwise optimiser with richer initialisation and a fast
gradient computation.  The algorithm works in three phases:

1.  Initialise a probability vector of length *n* (chosen adaptively
    according to the time budget) from a diverse catalogue of
    analytic families and a few random Dirichlet draws.  The best
    candidate w.r.t. the objective 2·n·max(conv(a,a)) is kept.

2.  Aggressive greedy local search.  In each iteration the full
    self‑convolution is computed with a real FFT, the active shifts
    attaining the maximal value are identified and the exact
    sub‑gradient (the sum of the corresponding gradients) is formed.
    The algorithm tries to move mass from the indices with the largest
    gradient to the indices with the smallest gradient.  Several
    source/destination pairs (up to *K* = 10) are examined and the
    transfer giving the largest improvement is performed.  Full‑mass
    moves are attempted first; if they do not improve, a reduced
    transfer is tried.  Random jitter is injected periodically to
    escape plateaux.

3.  Short simulated‑annealing fine‑tuning (Metropolis walk) while
    respecting the simplex constraint.

All phases respect a global deadline (the given budget).  The final
vector is projected onto the simplex (non‑negative, sum = 1) and
returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # optional, not used in the main loop

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(a,a))  (assumes sum(a)=1).  This is the exact
    quantity evaluated by ``evaluate_sequence`` up to the harmless
    clipping performed there.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.05) -> np.ndarray:
    """Sparse random draw – Gamma(alpha,1) i.i.d. and then normalise."""
    vec = rng.gamma(shape=alpha, scale=1.0, size=n)
    vec = np.maximum(vec, 0.0)
    if vec.sum() == 0:
        vec[0] = 1.0
    return vec / vec.sum()


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Produce a diverse set of starting vectors for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta – wide range of shape parameters
    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07,
              0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – several decay rates
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # random sparse start (~0.5 % non‑zeros)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    rnd[idx] = rng.random(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    # Dirichlet draws – give very spiky vectors
    for _ in range(4):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.02))

    # Noisy perturbations of a few beta shapes
    for _ in range(3):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert)
        cand.append(pert)

    return cand


# ----------------------------------------------------------------------
def _gradient_and_active(a: np.ndarray):
    """
    Compute the self‑convolution, the active shifts (those attaining the
    maximal value up to numerical tolerance) and the sub‑gradient
    g_i = 2·Σ_{k∈active} a_{k‑i} (only where the index exists).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    max_b = b.max()
    eps = max_b * 1e-12 + 1e-12
    active = np.where(b >= max_b - eps)[0]

    g = np.zeros_like(a)
    for k in active:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i = np.arange(i_low, i_high + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * a[j]
    return b, max_b, active, g


# ----------------------------------------------------------------------
def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer based on the exact sub‑gradient of the
    active convolution shift(s).  The best vector seen is returned.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0
    max_transfer = 0.5      # initial cap for a single move

    while time.time() < deadline:
        it += 1
        b, max_b, active, g = _gradient_and_active(a)

        # source indices – must carry some mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break

        # pick up to K best sources and K worst destinations
        K = min(10, src_candidates.size)
        src_sorted = src_candidates[np.argsort(-g[src_candidates])][:K]
        dest_sorted = np.argsort(g)[:K]

        improved = False
        # try every source/destination pair (up to K² attempts)
        for src in src_sorted:
            for dest in dest_sorted:
                if src == dest:
                    continue
                delta = min(a[src], max_transfer)
                if delta < 1e-12:
                    continue

                # --- full transfer -------------------------------------------------
                a_new = a.copy()
                a_new[src] -= delta
                a_new[dest] += delta
                new_score = _objective(a_new)
                if new_score < best_score - 1e-14:
                    a = a_new
                    best_vec = a.copy()
                    best_score = new_score
                    improved = True
                    break

                # --- reduced transfer (20 % of the full amount) ------------------
                delta2 = delta * 0.2
                if delta2 < 1e-12:
                    continue
                a_new2 = a.copy()
                a_new2[src] -= delta2
                a_new2[dest] += delta2
                new_score2 = _objective(a_new2)
                if new_score2 < best_score - 1e-14:
                    a = a_new2
                    best_vec = a.copy()
                    best_score = new_score2
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # no improvement with the current transfer size → shrink it
            max_transfer *= 0.5
            if max_transfer < 1e-12:
                # occasional random jitter to kick the system out of a flat region
                if it % 500 == 0:
                    i = rng.integers(n)
                    if a[i] > 1e-12:
                        j = rng.integers(n - 1)
                        if j >= i:
                            j += 1
                        eps = min(a[i] * 0.04, 0.001)
                        a_tmp = a.copy()
                        a_tmp[i] -= eps
                        a_tmp[j] += eps
                        scr = _objective(a_tmp)
                        if scr < best_score - 1e-14:
                            a = a_tmp
                            best_vec = a.copy()
                            best_score = scr
                            print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")
                else:
                    # nothing useful can be done – exit the loop
                    break

        if it % 500 == 0:
            print(f"[{time.time():.1f}] greedy it={it} best={best_score:.6f}")

    return best_vec, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑type random walk with exponential cooling.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03                        # initial temperature (objective units)
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer up to 10 % of source mass
        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        # exponential cooling
        T *= 0.9997

    return best_a, best_score
