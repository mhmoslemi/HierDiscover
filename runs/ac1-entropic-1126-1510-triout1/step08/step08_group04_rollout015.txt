def propose_candidate(seed=0, budget_s=1000):
    import time, random, numpy as np, math
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0
    best_seq = None
    best_score = float('inf')
    best_n = None

    lengths = [2000, 4000, 8000, 12000, 20000, 40000]   # maybe adjust
    # also small lengths for LP refine later

    # Candidate generation loops
    for n in lengths:
        if time.time() >= deadline:
            break

        # generate families
        cand = []
        # uniform
        cand.append(_distribution_uniform(n))
        # arcsine
        cand.append(_distribution_arcsine(n))
        # triangular
        cand.append(_distribution_triangular(n))
        # cosine windows
        for coeff in [0.25, 0.5, 0.75]:
            cand.append(_distribution_cosine_window(n, coeff))
        # sine powers
        for pw in [0.5, 1.0, 2.0, 4.0, 8.0, 12.0]:
            cand.append(_distribution_sine_power(n, pw))
        # gaussian sigmas
        for sigma in [0.15, 0.2, 0.25, 0.3]:
            cand.append(_distribution_gaussian(n, sigma))
        # symmetric exponential tails
        for lam in [0.01, 0.03, 0.06, 0.12]:
            cand.append(_distribution_exponential(n, lam))
        # beta symmetric
        for p in np.linspace(0.3, 3.0, 14):
            cand.append(_distribution_beta_sym(n, p))
        # random Dirichlet draws
        for alpha in [0.2, 0.5, 1.0, 2.0]:
            vec = np.random.dirichlet(np.full(n, alpha))
            cand.append(vec)

        # evaluate
        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, 1e-12)
            vec = vec / vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.2f}] n={n} new best score {sc:.6f}")

    # Incorporate previous global height_sequence_1 if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.2f}] previous best {sc:.6f}")
    except Exception:
        pass

    # If no candidate found (unlikely)
    if best_seq is None:
        n = 5000
        best_seq = _distribution_uniform(n)
        best_seq = best_seq / best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"Fallback uniform best {best_score}")

    # ***** Refinement stages *****
    # 1. Mirror descent (like before) but limited steps
    a = best_seq.copy()
    n = a.shape[0]
    eps = 1e-12
    # schedule parameters (few steps)
    schedule = [
        (1.0, 100.0, 2000, 0.05, 0.005),
        (100.0, 2000.0, 2000, 0.005, 0.0005),
    ]
    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w = w / w.sum()
            # compute gradient via convolution of a reversed with w
            grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
            a = a * np.exp(-lr * grad)
            # project onto simplex
            a = np.maximum(a, eps)
            a = a / a.sum()
            # evaluate
            if it % 500 == 0 or it == steps - 1:
                cur_sc = _max_convolution_score(a)
                if cur_sc < best_score:
                    best_score = cur_sc
                    best_seq = a.copy()
                    best_n = n
                    print(f"[{time.time():.2f}] mirror iter={it} new best {cur_sc:.6f}")

    # 2. Linear programming refinement (small n)
    # Use current best if n <= 3000 to reduce matrix size; else downsample using block-averaging.
    if best_n <= 3000:
        try:
            from scipy.optimize import linprog
            import scipy.sparse as sp
            a = best_seq.copy()
            n = a.shape[0]
            conv = _fft_self_convolve(a)
            M = conv.max()
            rows = []
            cols = []
            data = []
            # Build full constraint matrix
            for j in range(n):
                a_j = a[j]
                # rows i = j ... j+n-1
                i_start = j
                i_end = j + n
                rows.extend(range(i_start, i_end))
                cols.extend(range(n))
                data.extend([a_j] * n)  # length n
            A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
            c = -np.ones(n)  # maximize sum
            res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=(0, None), method='highs')
            if res.success:
                g = res.x
                g_sum = g.sum()
                if g_sum > 0:
                    g = g / g_sum
                    sc_g = _max_convolution_score(g)
                    if sc_g < best_score:
                        best_score = sc_g
                        best_seq = g.copy()
                        best_n = n
                        print(f"[{time.time():.2f}] LP refined best {sc_g:.6f}")
        except Exception as e:
            print("LP refinement failed:", e)

    # 3. Simple hill climber: transfer mass to reduce max
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # Choose source index with largest contribution to max slot
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Find source index with largest a_i among pairs producing max
        # We'll compute contributions a[i]*a[j]
        best_src = i_start
        best_j = k - i_start
        max_prod = a[best_src] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_src = i
                best_j = j
        source = best_src if a[best_src] >= a[best_j] else best_j
        # pick random target
        target = random.randrange(n)
        if target == source:
            continue
        # propose transfer
        delta = min(a[source] * 0.5, 0.02) * random.random()
        if delta <= 0:
            continue
        # incremental update
        a_old = a.copy()
        conv_old = conv.copy()
        # O(n) update function
        # Since we have limited time, we reuse _apply_transfer from earlier code
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            if it % 200 == 0:
                print(f"[{time.time():.2f}] hill iter={it} new best {best_score:.6f}")
        else:
            a = a_old
            conv = conv_old

        # occasional jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")

"""
Propose a step‑function (non‑negative vector) with a small
self‑convolution maximum.  The search consists of three phases:

1.  Broad exploration – generate many analytic families and random
    Dirichlet draws for several lengths (up to 40 000).  Keep the best
    normalised candidate (sum = 1).

2.  Refine the winner – a short mirror‑descent run (exponentiated‑gradient
    on the simplex) followed by an optional linear‑programming step
    (small lengths only) that maximises the total mass under the
    current max‑convolution bound.

3.  Light hill‑climber – repeatedly transfer a tiny amount of mass
    from the most “responsible’’ index to a random target, updating the
    full self‑convolution in O(n).  The loop stops when the time budget
    (≈ 1000 s) is exhausted.

The function returns the best sequence found as a plain Python list of
floats.
"""

import time
import math
import random
import numpy as np

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a probability vector (sum = 1).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray, conv: np.ndarray,
                    i: int, j: int, delta: float) -> None:
    """
    Transfer a mass `delta` from index i to index j, updating both the
    vector `a` and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)

    # ------------------------------------------------------------------

    a = best_seq.copy()
