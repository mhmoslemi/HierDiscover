"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1️⃣  Generate a diverse set of candidate sequences for several lengths
    (analytic families + Dirichlet draws) and keep the best few
    (≈30) according to the cheap surrogate 2·n·max(conv(a,a)).

2️⃣  Refine every seed with two first‑order schemes:
    • Adam‑style adaptive moment descent on the simplex.
    • Exponentiated‑gradient (KL‑mirror) descent.
    The best point seen is retained.

3️⃣  One “global” refinement:
    Starting from the current best sequence solve a non‑negative
    least‑squares problem  A g ≈ M·1  where A is the convolution matrix
    ( (A g)_k = (a * g)_k ) and M = max(a * a).  The solution g is
    re‑normalised and mixed with a.  A few mixing steps (t=0.5,0.25)
    usually lower the max‑convolution noticeably.

4️⃣  Frank–Wolfe style mass‑transfer moves followed by a light
    simulated‑annealing phase to polish the result.

All operations respect the 1000 s budget and the sequence is
returned with ℓ₁‑norm 1, which makes the surrogate equal to the
true ``evaluate_sequence`` value.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import lsq_linear

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate 2·n·max(conv(a,a)) (exact when ∑a=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (self‑product terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 300_000):
    """Adam‑style adaptive moment descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6,
                          max_iters: int = 250_000):
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _conv_matrix_dense(a: np.ndarray) -> np.ndarray:
    """Return the dense convolution matrix A where (A·g)_k = (a*g)_k."""
    n = a.shape[0]
    m = 2 * n - 1
    # first column = a padded with zeros, first row = [a[0], 0, ..., 0]
    c = np.concatenate([a, np.zeros(m - n, dtype=np.float64)])
    r = np.zeros(n, dtype=np.float64)
    r[0] = a[0]
    return toeplitz(c, r)  # shape (m, n)


def _nnls_refine(a: np.ndarray) -> np.ndarray:
    """
    One step of NNLS‑based refinement:
    find non‑negative g that tries to keep (a·g) ≤ M where M = max(a*a),
    then mix a←(1‑t)a + t g.
    """
    n = a.shape[0]
    m = 2 * n - 1
    # current maximal convolution value
    M = _fft_self_convolve(a).max()
    if M <= 0:
        return a

    # build dense A
    A = _conv_matrix_dense(a)                # (2n‑1) × n
    y = np.full(m, M, dtype=np.float64)

    # solve non‑negative least squares (LSQ)
    res = lsq_linear(A, y, bounds=(0, np.inf), lsmr_tol='auto',
                     max_iter=2000, verbose=0)
    g = res.x
    if g.sum() == 0:
        return a
    g = g / g.sum()                           # re‑normalise

    # try a couple of mixing factors
    for t_mix in (0.5, 0.25):
        a_new = _project_onto_simplex(a * (1.0 - t_mix) + g * t_mix, s=1.0)
        if _max_convolution_score(a_new) < _max_convolution_score(a) - 1e-12:
            return a_new
    return a   # no improvement

        cur_score = best_score
        it_nnls = 0
        while time.time() < deadline - 2.0 and it_nnls < 8:
            it_nnls += 1
            a_new = _nnls_refine(a)
            new_score = _max_convolution_score(a_new)
            if new_score < cur_score - 1e-12:
                a, cur_score = a_new, new_score
                print(f"[{time.time():.1f}] NNLS iter {it_nnls} improved to {cur_score:.6f}")
                if cur_score < best_score - 1e-14:
                    best_score, best_seq = cur_score, a.copy()
            else:
                break

    # ----------------------------------------------------------------------
    # 7️⃣ Frank‑Wolfe style mass‑transfer polishing
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline - 1.0 and fw_iter < 60_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # move mass *away* from largest gradient
        i_dst = int(np.argmin(grad))   # move mass *toward* smallest gradient
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # full transfer trial
        a_candidate = a.copy()
        a_candidate[i_src] -= max_transfer
        a_candidate[i_dst] += max_transfer
        score_candidate = _max_convolution_score(a_candidate)

        if score_candidate < cur_score - 1e-13:
            a, cur_score = a_candidate, score_candidate
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] FW iter {fw_iter} new best {best_score:.6f}")
            continue

        # half transfer trial
        half = max_transfer * 0.5
        a_candidate = a.copy()
        a_candidate[i_src] -= half
        a_candidate[i_dst] += half
        score_candidate = _max_convolution_score(a_candidate)
        if score_candidate < cur_score - 1e-13:
            a, cur_score = a_candidate, score_candidate
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] FW half iter {fw_iter} new best {best_score:.6f}")
            continue

        # occasional jitter to escape tiny plateaus
        if fw_iter % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(a.size) * 2e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ----------------------------------------------------------------------
    # 8️⃣ Light simulated‑annealing / jitter phase
    a = best_seq.copy()
    cur_score = best_score
    anneal_iter = 0
    temp_start, temp_end = 0.02, 0.001
    while time.time() < deadline - 0.1 and anneal_iter < 100_000:
        anneal_iter += 1
        # linear temperature schedule
        t_frac = (deadline - time.time()) / (budget_s - 0.5)
        temp = temp_end + (temp_start - temp_end) * t_frac

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a, cur_score = a_new, new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter {anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a, cur_score = a_new, new_score

    # ----------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
