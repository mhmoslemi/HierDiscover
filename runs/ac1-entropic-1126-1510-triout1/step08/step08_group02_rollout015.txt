"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – build a few analytic U‑shaped candidates
   (arcsine, symmetric Beta, triangular, Gaussian, semicircle) for a
   chosen length *n* (taken from the previous best if available, else
   3000).  The candidate with the smallest exact objective
   ``2·n·max(a*a)/sum(a)^2`` is kept.

2. **Linear‑programming improvement** – for the current sequence ``f``
   compute ``M = max(f*f)``.  Solve the LP

        maximise   sum(g)
        subject to  f * g ≤ M     (element‑wise convolution)
                    g ≥ 0

   The LP is built with a sparse Toeplitz matrix (SciPy) and solved with
   HiGHS.  The solution ``g`` can carry substantially more total mass
   while staying under the same convolution bound.  We mix the old and
   new sequences, trying several mixing rates ``t``∈{0.5,0.3,0.2,0.1,0.05},
   and keep the mixture that yields the greatest reduction of the true
   objective.  The process is repeated until no improvement is observed
   or the time budget runs out (typically a handful of iterations).

3. **Fine‑grained hill‑climbing** – with the remaining time perform
   inexpensive random mass‑transfer moves that keep the total mass
   constant.  Moves are accepted only when they lower the exact objective.
   Occasionally a tiny random jitter is added to escape plateaux.

The algorithm respects the 1000 s budget, works entirely on the
non‑negative simplex (the sum may grow, which is harmless because the
objective is homogeneous), and always returns a feasible list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` (no clipping)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * M / (s * s)

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

# Linear‑programming helpers -------------------------------------------
def _conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build a sparse matrix A such that A @ g = f * g (full convolution).
    Shape of A is (2·n‑1) × n.
    """
    n = f.shape[0]
    # each diagonal offset d contains the constant f[d]
    diags = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise sum(g)   subject to   f * g ≤ M , g ≥ 0.
    Returns the optimal g (or None if the solver fails).
    """
    if not _SCIPY_OK:                      # pragma: no cover
        return None
    n = f.shape[0]
    A = _conv_matrix(f)                   # (2n‑1) × n sparse
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)     # maximise → minimise –sum(g)

    # HiGHS (via scipy.optimize.linprog) handles sparse A_ub efficiently.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        # tiny negative values caused by numeric noise are clipped
        return np.clip(res.x, 0.0, None)
    # if the LP fails, we simply give up on this iteration
    print("LP solve failed:", res.message)
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget and always returns a feasible
    sequence (list of floats).  It may use SciPy's linear programming facilities
    when available; otherwise it falls back to a purely gradient based approach.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working length n
    try:
        n = int(len(height_sequence_1))      # type: ignore[name-defined]
        if n < 800:
            n = 800
    except Exception:
        n = 3000

    print(f"[{time.time():.1f}] Working length n = {n}")

    # ------------------------------------------------------------------
    # 2. Build a modest set of analytic candidates and keep the best
    cand_funcs = [
        (_arcsine_distribution, "arcsine", {}),
        (_triangular_distribution, "triangular", {}),
        (_uniform_distribution, "uniform", {}),
        (_semicircle_distribution, "semicircle", {}),
    ]

    beta_ps = [0.05, 0.10, 0.20, 0.30, 0.40, 0.50, 0.70, 1.0, 1.5, 2.0]
    for p in beta_ps:
        cand_funcs.append((_beta_symmetric_distribution, f"beta(p={p})",
                           {"p": p}))
    sigma_fracs = [0.15, 0.25, 0.35]
    for sf in sigma_fracs:
        cand_funcs.append((_gaussian_distribution, f"gauss(sig={sf})",
                           {"sigma_frac": sf}))

    best_seq = None
    best_score = np.inf

    for gen, name, kw in cand_funcs:
        if time.time() >= deadline:
            break
        a = gen(n, **kw) if kw else gen(n)
        score = _objective(a)
        if score < best_score:
            best_score, best_seq = score, a.copy()
        print(f"[{time.time():.1f}] init {name:15s} score={score:.6f}")

    # Include previous best if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size == n:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0:
                prev = prev / prev.sum()
                sc = _objective(prev)
                if sc < best_score:
                    best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    if best_seq is None:                       # pragma: no cover
        # Fallback – uniform distribution
        best_seq = _uniform_distribution(n)
        best_score = _objective(best_seq)

    print(f"[{time.time():.1f}] Starting LP refinements, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Linear‑programming improvement loop
    max_lp_iters = 12
    lp_iter = 0
    while time.time() < deadline and lp_iter < max_lp_iters:
        lp_iter += 1
        M = _fft_self_convolve(best_seq).max()
        g = _solve_lp_max_sum(best_seq, M)
        if g is None:
            break

        sum_f = best_seq.sum()
        sum_g = g.sum()
        # If LP cannot increase total mass, stop
        if sum_g <= sum_f * (1.0 + 1e-8):
            print(f"[{time.time():.1f}] LP iteration {lp_iter}: no mass increase")
            break

        # Try several mixing rates and keep the best improvement
        best_local_seq = best_seq
        best_local_score = best_score
        for t in (0.5, 0.3, 0.2, 0.1, 0.05):
            cand = (1.0 - t) * best_seq + t * g
            cand = np.clip(cand, 0.0, None)  # ensure non‑negative
            sc = _objective(cand)
            if sc < best_local_score:
                best_local_score, best_local_seq = sc, cand.copy()

        if best_local_score < best_score - 1e-12:
            best_seq, best_score = best_local_seq, best_local_score
            print(f"[{time.time():.1f}] LP iter {lp_iter} → score={best_score:.6f} (mass {best_seq.sum():.4f})")
        else:
            # No improvement for any mixing factor → stop
            print(f"[{time.time():.1f}] LP iter {lp_iter}: mixing did not improve")
            break

    # ------------------------------------------------------------------
    # 4. Fine‑grained hill‑climbing with constant total mass
    a = best_seq.copy()
    n = a.shape[0]
    cur_score = best_score
    it = 0
    max_iters = 300_000

    while time.time() < deadline and it < max_iters:
        it += 1
        # geometric decay of transfer magnitude
        transfer = 0.02 * (0.5 ** (it / 8000.0))

        # pick source and destination indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _objective(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb it={it} new best {best_score:.6f}")

        # occasional jitter
        if it % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, s=a.sum())
            cur_score = _objective(a)

    # ------------------------------------------------------------------
    # 5. Final safety: clip values into the range accepted by evaluate_sequence
    final_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"[{time.time():.1f}] Finished – final objective ≈ {best_score:.6f}")
    return final_seq.tolist()
