def lp_step(a, t=0.4):
    # Input a: numpy array sum=1
    # returns new a after mixing - if fails returns original.
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    # Build sparse matrix
    rows = np.concatenate([np.arange(j, j+n) for j in range(n)])
    cols = np.concatenate([np.full(n, j, dtype=np.int64) for j in range(n)])
    data = np.tile(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Solve LP
    c = -np.ones(n)   # maximize sum(b)
    bounds = [(0, None)] * n
    b_ub = M * np.ones(2*n-1)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return a
    b_opt = res.x
    if b_opt.sum() == 0:
        return a
    # normalize b to sum 1
    b_norm = b_opt / b_opt.sum()
    # mix
    new_a = (1 - t) * a + t * b_norm
    new_a = _project_onto_simplex(new_a, 1.0)
    new_a = np.maximum(new_a, 1e-12)
    new_a /= new_a.sum()
    return new_a
```

def subgrad_descent(a, max_iters, deadline):
    n = a.shape[0]
    vel = np.zeros_like(a)
    beta = 0.9
    best_local = a.copy()
    best_sc = _max_convolution_score(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        it+=1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        eps_rel = 1e-12 * (abs(maxc) + 1.0)
        active = np.flatnonzero(conv >= maxc - eps_rel)
        w = np.zeros_like(conv)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n-1:n-1+n]  # length n
        vel = beta * vel + (1-beta) * grad
        eta = 0.1 / np.sqrt(it)   # tuned
        a = a - eta * vel
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        sc = _max_convolution_score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_local = a.copy()
    return best_local, best_sc

def hill_climb(a, best_seq, best_sc, deadline):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        # find pair i,j generating max product a[i]*a[j] for that k
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        i_best = i_start
        j_best = k - i_start
        max_prod = a[i_best] * a[j_best]
        for i in range(i_start, i_end+1):
            j = k - i
            prod = a[i]*a[j]
            if prod > max_prod:
                max_prod = prod
                i_best, j_best = i, j
        source = i_best if a[i_best] >= a[j_best] else j_best
        target = np.random.randint(n)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta < 1e-12:
            continue
        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_sc = 2.0 * n * M
            best_seq = a.copy()
        else:
            a = a_snap
            conv = conv_snap
        # schedule maybe every 5000 iter jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return best_seq, best_sc
```

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    The lower the better.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    # keep a copy of the old a only for the update formulas
    a_old = a.copy()
    # contribution of the moved mass to the convolution
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _lp_improve(a, t=0.4):
    """
    One linear‑programming improvement step.
    Given a (sum=1, non‑negative), compute the maximal feasible b under
    the current maximal self‑convolution M, then mix a ← (1‑t)a + t·b.
    """
    if not _SCIPY_AVAILABLE:                     # pragma: no cover
        return a

    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()

    # Build the convolution matrix A (shape (2n‑1, n))
    # A[i, j] = a[i‑j] if 0≤i‑j<n else 0.
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.concatenate([np.full(n, j, dtype=np.int64) for j in range(n)])
    data = np.tile(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)               # maximise sum(b)  ==  minimise -sum(b)
    bounds = [(0, None)] * n
    b_ub = M * np.ones(2 * n - 1)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:             # pragma: no cover
        return a

    if not res.success:
        return a

    b_opt = res.x
    if b_opt.sum() == 0:
        return a

    # normalise to sum‑1 (the simplex we work on)
    b_norm = b_opt / b_opt.sum()
    new_a = (1.0 - t) * a + t * b_norm
    new_a = _project_onto_simplex(new_a, 1.0)
    new_a = np.maximum(new_a, 1e-12)
    new_a /= new_a.sum()
    return new_a


# ----------------------------------------------------------------------
def _subgradient_descent(a, max_iters, deadline):
    """
    Momentum‑based sub‑gradient descent (projected onto the simplex).
    Returns the best vector seen during the run and its score.
    """
    n = a.shape[0]
    vel = np.zeros_like(a)
    beta = 0.9
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        eps_rel = 1e-12 * (abs(maxc) + 1.0)
        active = np.flatnonzero(conv >= maxc - eps_rel)

        # sub‑gradient of the max entry
        w = np.zeros_like(conv)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]    # length n

        # momentum (Adam‑like with only first moment)
        vel = beta * vel + (1.0 - beta) * grad
        eta = 0.15 / np.sqrt(it)                 # decreasing stepsize
        a = a - eta * vel
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        sc = _max_convolution_score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
def _hill_climb(a, best_a, best_sc, deadline):
    """
    Simple stochastic hill‑climber that moves small amounts of mass.
    Works directly on the convolution array for O(n) updates.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1

        # locate a maximal convolution entry and the contributing pair
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < 1e-12:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_sc = 2.0 * n * M
            best_a = a.copy()
        else:
            a = a_snap
            conv = conv_snap

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    The routine mixes analytic seeds, a few LP improvement steps,
    momentum sub‑gradient descent and a final hill‑climber.
    It respects the 1000 s wall‑clock limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_len = None

    # ------------------------------------------------------------------
    # 1. Generate a few well‑chosen initial candidates for several lengths
    lengths = [800, 1500, 2500, 3500]          # moderate sizes (memory‑friendly)

    for n in lengths:
        if time.time() >= deadline:
            break

        # a collection of deterministic families
        candidates = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.004),
            _distribution_exponential(n, lam=0.02),
            _distribution_gaussian(n, sigma=0.3),
        ]

        # random Dirichlet draws with a few concentration parameters
        for alpha in (0.2, 0.5, 1.0, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate cheap candidates and keep the best one for this length
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_len = n
                print(f"[{time.time():.1f}] length={n} init new best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. If a previous global result exists, try it as a seed
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_len = len(prev)
                print(f"[{time.time():.1f}] previous global new best {sc:.6f}")
    except Exception:                     # pragma: no cover
        pass

    # ------------------------------------------------------------------
    # 3. Core optimisation loop – start from the best seed found so far
    a = best_seq.copy()
    n = a.shape[0]

    # ---- 3a. Linear‑programming improvement cycles ----
    lp_steps = 5
    for step in range(lp_steps):
        if time.time() >= deadline:
            break
        a = _lp_improve(a, t=0.4)
        cur_sc = _max_convolution_score(a)
        if cur_sc < best_score - 1e-12:
            best_score = cur_sc
            best_seq = a.copy()
            best_len = n
            print(f"[{time.time():.1f}] after LP step {step+1} new best {cur_sc:.6f}")

    # ---- 3b. Sub‑gradient descent (momentum) ----
    max_subgrad_iters = 30000
    a, sub_sc = _subgradient_descent(a, max_subgrad_iters, deadline)
    if sub_sc < best_score - 1e-12:
        best_score = sub_sc
        best_seq = a.copy()
        best_len = n
        print(f"[{time.time():.1f}] after sub‑grad descent new best {sub_sc:.6f}")

    # ---- 3c. Final hill‑climber ----
    a, hill_sc = _hill_climb(a, best_seq, best_score, deadline)
    if hill_sc < best_score - 1e-12:
        best_score = hill_sc
        best_seq = a.copy()
        best_len = n
        print(f"[{time.time():.1f}] after hill‑climb new best {hill_sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Safety finalisation – ensure a valid output
    if best_seq is None:
        best_seq = _distribution_uniform(1000)
        best_seq = best_seq / best_seq.sum()

    # Clip to the allowed range (the evaluator does this anyway)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq = np.maximum(best_seq, 0.0)

    final_obj = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_obj:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
