The algorithm follows four stages:

  2. Refine each candidate with a smooth‑max mirror‑descent that uses an

  3. A vanilla sub‑gradient descent with Nesterov momentum finalises the shape.
  4. A deterministic hill‑climber (plus occasional jitter) works on the

import time, math, random
import numpy as np

def _project_onto_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """Transfer `delta` from index i to j, updating `a` and its convolution `conv`
    in O(n) time. The implementation follows the exact algebraic update
    described in the analysis."""
    if delta <= 0 or i == j:
        return
    n = a.shape[0]
    a_old = a.copy()
    # linear part:   conv[k]  += 2*delta*(a_old shifted by j) - 2*delta*(a_old shifted by i)
    conv[i : i + n] -= 2.0 * delta * a_old
    conv[j : j + n] += 2.0 * delta * a_old
    # quadratic part
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    ij = i + j
    if ij < conv.size:
        conv[ij] -= 2.0 * delta * delta
    # finally update the vector a
    a[i] -= delta
    a[j] += delta

"""
Propose a step‑function vector that minimises ``evaluate_sequence``.
The algorithm proceeds in four phases and respects a 1000 s wall‑clock
budget:

1. **Broad cheap scan** – several analytic families (uniform,
   arcsine, triangular, exponential, beta, sine‑power, Gaussian) and
   many random Dirichlet draws are generated for a handful of lengths.
   The best few candidates are kept.

2. **Smooth‑max mirror‑descent** – the best candidate is refined with an
   Adam optimiser on the log‑softmax parametrisation.  A smooth
   approximation of the max (soft‑max) is used; the temperature (α) is
   annealed while the learning rate is decayed.

3. **Sub‑gradient descent** – a classical sub‑gradient method with
   Nesterov momentum further polishes the shape.

4. **Hill‑climber (simulated annealing)** – an O(n) incremental update
   of the full self‑convolution is employed.  Mass is transferred between
   indices; moves that improve the maximal convolution are always kept,
   while worsening moves are accepted with a temperature‑dependent
   probability.  Periodic jitter and full recomputation avoid drift.

All helper routines are top‑level (no closures) and the final vector is
projected onto the simplex and returned as a plain list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with Σ = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Transfer a mass ``delta`` from index ``i`` to ``j`` while updating
    both the vector ``a`` and its full self‑convolution ``conv`` in O(n)
    time.  The formula implements the exact algebraic change caused by
    the rank‑one update a ← a + δ(e_j − e_i).
    """
    if delta <= 0 or i == j:
        return
    n = a.shape[0]
    a_old = a.copy()

    # linear contribution:   ±2·δ·(a shifted by i / j)
    conv[i : i + n] -= 2.0 * delta * a_old
    conv[j : j + n] += 2.0 * delta * a_old

    # quadratic contribution
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    ij = i + j
    if ij < conv.size:
        conv[ij] -= 2.0 * delta * delta

    # finally update the probability vector
    a[i] -= delta
    a[j] += delta

        a = np.maximum(a, eps)
        a = a / a.sum()

        cur_sc = _max_convolution_score(a)
        if cur_sc < best_score - 1e-12:
            best_score = cur_sc
            best_seq = a.copy()
            best_n = n
            print(f"[{time.time():.1f}] subgrad iter={it} new best {cur_sc:.6f}")

        if time.time() + 0.5 > deadline:
            break

    # --------------------------------------------------------------
    # 6.  Final hill‑climber (simulated annealing) with O(n) updates.
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    T = M * 0.5               # initial temperature
    decay = 0.9995

    while time.time() < deadline:
        it += 1

        # pick a source index proportionally to its mass
        source = np.random.choice(n, p=a / a.sum())
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # propose a transfer (up to 5 % of source mass, capped at 0.02)
        max_delta = min(a[source] * 0.05, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        conv_snapshot = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        dM = new_M - M
        accept = False
        if dM < -1e-14:
            accept = True
        else:
            prob = math.exp(-dM / max(T, 1e-16))
            if random.random() < prob:
                accept = True

        if accept:
            M = new_M
            cur_sc = 2.0 * n * M
            if cur_sc < best_score - 1e-12:
                best_score = cur_sc
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] SA hill it={it} new best {cur_sc:.6f}")
        else:
            a = a_snapshot
            conv = conv_snapshot

        T *= decay

        if it % 5000 == 0:
            # occasional jitter & full recompute to keep numerical drift low
            jitter = np.random.randn(n) * 1e-6
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
```
