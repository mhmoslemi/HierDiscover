"""
Propose a step‑function vector (non‑negative list of floats) that minimises
the evaluation function `evaluate_sequence`.  The search proceeds in four
phases and respects a 1000 s wall‑clock budget.

1. **Broad scan** – Analytic families (uniform, arcsine, triangular,
   exponential, symmetric beta, sine‑power, cosine window, Gaussian) and
   random Dirichlet draws are generated for several lengths (up to 20000).
   The cheap objective  `2·n·max(conv(a,a))` (with sum(a)=1) is evaluated
   via FFT.  The best few candidates are retained.

2. **Mirror‑descent** – A soft‑max (exponential‑weights) mirror descent
   with a geometric temperature schedule refines the simplex vector.
   This is the same idea as the reference implementation but with a
   longer schedule and finer learning‑rate annealing.

3. **Adam sub‑gradient** – The true ∞‑norm of the self‑convolution is
   optimised using an Adam‑style adaptive step.  The gradient is the
   convolution of `a` with the indicator of the active (maximal) entries.

4. **Simulated‑annealing hill‑climbing** – Mass is transferred from the
   dominant pair of indices that realise the current maximal convolution
   entry to a random target.  Moves that increase the maximum are accepted
   with probability `exp(-(ΔM)/T)`.  The temperature `T` decays geometrically.
   The convolution is updated incrementally in O(n) via the helper
   `_apply_transfer`.  Periodic full recomputation of the convolution
   prevents drift.

The best vector seen during the whole run is returned as a plain Python
list; it automatically satisfies the non‑negativity, sum>0.01 and
`max ≤ 1000` constraints of `evaluate_sequence`.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

            M = conv.max()
            eps_rel = 1e-12 * (abs(M) + 1.0)
            active = np.flatnonzero(conv >= M - eps_rel)
            w = np.zeros_like(conv)
            w[active] = 1.0
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
            # Adam moments
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.15 / np.sqrt(t)          # decreasing base step
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()
            # occasional check
            if t % 300 == 0:
                cur_sc = _max_convolution_score(a)
                if cur_sc < best_score - 1e-12:
                    nonlocal best_seq, best_score, best_n
                    best_seq = a.copy()
                    best_score = cur_sc
                    best_n = n
                    print(f"[{time.time():.1f}] adam n={n} iter={t} new best {cur_sc:.6f}")
            # early exit guard
            if time.time() + 0.5 > budget:
                break
        return a

    # --------------------------------------------------------------
    # 4. Simulated‑annealing hill‑climber (O(n) updates)
    def _hill_climb_anneal(a, n, budget):
        """Transfer mass from the dominant pair to random sites with annealing."""
        conv = _fft_self_convolve(a)
        M = conv.max()
        T = M * 0.02                      # initial temperature (relative)
        it = 0
        while time.time() < budget:
            it += 1
            # locate the maximal convolution entry and the most contributing pair
            k = int(conv.argmax())
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            best_i, best_j = i_start, k - i_start
            max_prod = a[best_i] * a[best_j]
            for i in range(i_start, i_end + 1):
                j = k - i
                prod = a[i] * a[j]
                if prod > max_prod:
                    max_prod = prod
                    best_i, best_j = i, j
            source = best_i if a[best_i] >= a[best_j] else best_j
            # pick a random target distinct from source
            target = random.randint(0, n - 1)
            if target == source:
                continue
            # propose a transfer amount (scale with temperature)
            max_delta = min(a[source] * 0.5, 0.03)
            delta = max_delta * random.random()
            if delta < eps:
                continue
            a_snap = a.copy()
            conv_snap = conv.copy()
            _apply_transfer(a, conv, source, target, delta)
            new_M = conv.max()
            if new_M < M - 1e-14:
                M = new_M
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-12:
                    nonlocal best_seq, best_score, best_n
                    best_seq = a.copy()
                    best_score = cur_score
                    best_n = n
                    print(f"[{time.time():.1f}] anneal n={n} iter={it} new best {cur_score:.6f}")
            else:
                # accept with Metropolis probability
                dE = new_M - M
                prob = math.exp(-dE / (T + 1e-30))
                if random.random() < prob:
                    M = new_M
                else:
                    a = a_snap
                    conv = conv_snap
            # geometric cooling
            T *= 0.99996
            # periodic full recomputation to curb drift
            if it % 4000 == 0:
                conv = _fft_self_convolve(a)
                M = conv.max()
                jitter = np.random.randn(n) * 1e-6
                a = np.maximum(a + jitter, eps)
                a = a / a.sum()
                conv = _fft_self_convolve(a)
                M = conv.max()
                print(f"[{time.time():.1f}] anneal n={n} iter={it} T={T:.2e}")
        return a

    # --------------------------------------------------------------
    # 5. Run refinement pipelines on the best few candidates
    # keep only distinct lengths to avoid doing the same work twice
    processed_lengths = set()
    for sc, vec in top_candidates[:5]:   # at most five seeds
        if time.time() >= deadline:
            break
        n = vec.shape[0]
        if n in processed_lengths:
            continue
        processed_lengths.add(n)

        # start from the candidate
        a = vec.copy()
        # mirror descent
        a = _mirror_descent(a, n, deadline)

        # Adam sub‑gradient
        a = _adam_subgrad(a, n, deadline)

        # final annealed hill‑climbing
        a = _hill_climb_anneal(a, n, deadline)

        # one final cheap check
        final_sc = _max_convolution_score(a)
        if final_sc < best_score - 1e-12:
            best_seq = a.copy()
            best_score = final_sc
            best_n = n
            print(f"[{time.time():.1f}] final check n={n} new best {final_sc:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best vector discovered (as plain Python list)
    if best_seq is None:
        # fallback to a uniform vector of moderate size
        best_n = 4096
        best_seq = np.full(best_n, 1.0 / best_n, dtype=np.float64)
        best_score = _max_convolution_score(best_seq)

    print(f"Best objective ≈ {best_score:.6f} achieved with length={best_n}")
    return best_seq.tolist()
