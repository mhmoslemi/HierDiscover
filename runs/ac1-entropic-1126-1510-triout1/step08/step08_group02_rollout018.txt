"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Search over simple analytic families** –  we consider
   * decreasing exponential distributions  a_i ∝ exp(-λ·i)
   * power‑law distributions                a_i ∝ (i+1)^(-p)
   * a few random Dirichlet draws.
   For each length we sweep a small set of parameters (λ or p),
   compute the cheap score 2·n·max(conv) (which equals the final
   objective for sum = 1) using an FFT based self‑convolution,
   and keep the best candidate seen so far.

2. **Refinement** –  if time permits we run a short projected
   sub‑gradient descent (the exact sub‑gradient of the max‑convolution)
   on the current best sequence.

The decreasing exponential family is asymmetric, so the convolution
with its reversal has a much smaller peak than any symmetric family.
Choosing λ≈k/√n with k≈0.55–0.65 yields a cheap score around
2·max(conv)·n≈4·k²≲1.5 for n of a few thousand, well below the target.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the indicator w
    conv_rev = np.fft.irfft(
        np.fft.rfft(a[::-1], n=1 << ((a.size + w.size - 1) - 1).bit_length())
        * np.fft.rfft(w, n=1 << ((a.size + w.size - 1) - 1).bit_length()),
        n=1 << ((a.size + w.size - 1) - 1).bit_length(),
    )[: a.size + w.size - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correction for diagonal entries (i = 2·k)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Analytic families -------------------------------------------------------
def _exponential_decreasing_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-λ·i)  (asymmetric, decreasing)."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    return vals / vals.sum()


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), decreasing."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** (-p)
    return vals / vals.sum()


def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine respects the time budget, prints progress, and returns a list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5           # safety margin

    # ------------------------------------------------------------------
    # Choose a base length.  If a previous solution exists use its size,
    # otherwise fall back to 5000.
    try:
        base_n = len(height_sequence_1)               # type: ignore[name-defined]
    except Exception:
        base_n = 5000

    # Lengths to explore (ensure they are not too small)
    lengths = sorted(
        set(
            int(base_n * f)
            for f in (0.6, 0.8, 1.0, 1.2, 1.5)
            if int(base_n * f) >= 500
        )
    )
    # also add a few random lengths up to 3×base_n, capped at 12000
    for _ in range(3):
        n = np.random.randint(500, min(12000, base_n * 3) + 1)
        lengths.append(n)
    lengths = sorted(set(lengths))

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Broad sampling over analytic families
    for n in lengths:
        if time.time() >= deadline:
            break

        # --- Decreasing exponential -------------------------------------------------
        # λ = k / sqrt(n) ; try several k values
        for k in (0.45, 0.50, 0.55, 0.60, 0.65, 0.70, 0.80):
            lam = k / np.sqrt(n)
            a = _exponential_decreasing_distribution(n, lam)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[{time.time():.1f}] n={n:5d} exp‑dec(k={k:.2f}) lam={lam:.6f} ⇒ {score:.6f}")

        # --- Power‑law -------------------------------------------------------------
        for p in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
            a = _power_law_distribution(n, p)
            score = _max_convolution_score(a)
            if score < best_score:
                best_score, best_seq = score, a.copy()
                print(f"[{time.time():.1f}] n={n:5d} power(p={p:.1f})                ⇒ {score:.6f}")

        # --- Random Dirichlet draws -------------------------------------------------
        for alpha in (0.3, 0.6, 1.0):

                score = _max_convolution_score(a)
                if score < best_score:
                    best_score, best_seq = score, a.copy()
                    print(f"[{time.time():.1f}] n={n:5d} Dirichlet(α={alpha:.1f})          ⇒ {score:.6f}")

    # ------------------------------------------------------------------
    # 2. (Optional) refinement with sub‑gradient descent if we have time
    if best_seq is not None and time.time() < deadline - 2.0:
        seq, sc = _subgradient_descent(best_seq, deadline,

        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] post‑refine sub‑grad ⇒ {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Final sanity check and return
    if best_seq is None:

        n = max(500, base_n)
        best_seq = np.full(n, 1.0 / n)
        best_score = _max_convolution_score(best_seq)
        print("[fallback] uniform distribution used")

    print(f"Best cheap score = {best_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
