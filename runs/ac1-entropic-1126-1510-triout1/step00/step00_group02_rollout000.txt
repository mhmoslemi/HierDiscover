"""Propose candidate sequence using simulated annealing + projected gradient descent.

We start from a uniform or known good sequence and iterate:
- compute convolution, find index of maximal value.
- take a projected gradient step decreasing that max.
- occasionally perform random pairwise mass swaps.
- accept moves according to Metropolis criterion with linear cooling.
The best sequence encountered is returned.
"""

import numpy as np
import time

def project_simplex(v):
    """Project a vector onto the probability simplex {x >=0, sum x =1}."""
    # using algorithm from https://arxiv.org/abs/1309.1541
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all entries are zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def gradient_step(seq, eta):
    """One projected gradient step decreasing max convolution entry."""
    n = len(seq)
    # compute convolution
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    # gradient of conv_k w.r.t. seq
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]
    # step opposite to gradient
    new_seq = seq - eta * g
    # project back onto simplex (nonnegative sum=1)
    new_seq = project_simplex(new_seq)
    return new_seq

def random_pair_move(seq, sigma):
    """Swap a small amount of mass between two randomly chosen positions."""
    n = len(seq)
    i, j = np.random.choice(n, size=2, replace=False)
    # draw a zero‑mean proposal
    delta = (np.random.rand() - 0.5) * sigma
    # keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])   # take from j
    else:
        delta = -min(-delta, seq[i]) # give to j, take from i
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Return a sequence of non‑negative heights that (hopefully) has a low score.

    The optimizer combines a projected gradient descent on the maximal
    convolution entry with a simple simulated‑annealing scheme that swaps
    mass between two coordinates.  The algorithm works on the probability
    simplex (sum of entries = 1) which removes the trivial scaling freedom
    of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a safety margin

    # Determine the length of the target sequence.
    try:
        # height_sequence_1 is expected to be defined globally.
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
        # Normalise to a probability simplex.
        base_seq = base_seq / np.maximum(np.sum(base_seq), 1e-12)
    except Exception:
        # Fallback: a uniform vector of moderate size.
        n = 3000
        base_seq = np.full(n, 1.0 / n, dtype=float)

    # Ensure the starting point lies on the simplex.
    current = project_simplex(base_seq)
    best = current.copy()
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = float("inf")
    current_score = best_score

    # Annealing schedule parameters
    temp_initial = 1.0
    temp_final = 1e-4

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # cooling factor (linear in time)
        t = (time.time() - start_time) / (deadline - start_time)
        temperature = temp_initial * (1 - t) + temp_final * t

        # Choose move type
        if np.random.rand() < 0.6:
            # Projected gradient move – step size decays with sqrt(iter)
            eta = 0.5 / np.sqrt(iteration)
            candidate = gradient_step(current, eta)
        else:
            # Random pairwise swap – sigma shrinks slowly

        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:

            continue

        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()

            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

    # Return as a plain Python list (what the external harness expects)
    return best.tolist()

"""
Propose a candidate step‑function sequence using a hybrid of
projected gradient descent (targeting the maximal convolution entry)
and simulated‑annealing‑style random pairwise swaps.

The algorithm works on the probability simplex (all entries ≥0, sum = 1),
which eliminates the trivial scaling freedom of ``evaluate_sequence``.
It starts from the known construction ``height_sequence_1`` when
available, otherwise from a uniform vector, and iteratively improves
the score until the time budget is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # Find rho = max { j | u_j - cssv_j / (j+1) > 0 }
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:   # all entries are zero or negative → uniform point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: enforce exact sum 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


# ----------------------------------------------------------------------
# Gradient step that tries to reduce the current maximal convolution value
# ----------------------------------------------------------------------
def _gradient_step(seq: np.ndarray, eta: float) -> np.ndarray:
    """
    One projected gradient step.
    - Compute the index k where (seq * seq) is maximal.
    - Gradient of that entry w.r.t. seq is g_i = 2 * seq_{k-i} (if defined).
    - Move opposite to the gradient with step size ``eta`` and project back.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)

    g = np.zeros_like(seq)
    g[mask] = 2.0 * seq[j[mask]]

    new_seq = seq - eta * g
    new_seq = _project_simplex(new_seq)
    return new_seq


# ----------------------------------------------------------------------
# Random pairwise mass‑swap move (preserves the sum exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """
    Choose two distinct coordinates i, j and transfer a small amount of mass
    (drawn from N(0, sigma²) and clipped to keep non‑negativity).
    The overall sum stays exactly 1, so no projection is required.
    """
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # Zero‑mean proposal
    delta = (np.random.rand() - 0.5) * sigma

    # Clip to keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])        # take from j
    else:
        delta = -min(-delta, seq[i])      # take from i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that strives
    to minimise ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds, using a mixture of
    projected gradient descent and simulated‑annealing moves.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 2.0)   # leave a tiny safety margin

    # ------------------------------------------------------------------
    # Initialise the search vector
    # ------------------------------------------------------------------
    try:
        # ``height_sequence_1`` is expected to be defined globally.
        base = np.array(height_sequence_1, dtype=float)
        n = base.size
        # Normalise onto the simplex (sum = 1)
        base = base / max(base.sum(), 1e-12)
    except Exception:
        # Fallback: a uniform vector of a reasonable size
        n = 3000
        base = np.full(n, 1.0 / n, dtype=float)

    current = _project_simplex(base)
    best = current.copy()

    # Initial scores
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = float("inf")
    current_score = best_score

    # Annealing schedule parameters
    temp_start = 1.0
    temp_end = 1e-4

    # Simple stagnation detector – restart after a long non‑improving stretch
    last_improve_iter = 0

    it = 0
    while time.time() < deadline:
        it += 1

        # Linear cooling between temp_start and temp_end
        elapsed_frac = (time.time() - start) / (deadline - start)
        temperature = temp_start * (1.0 - elapsed_frac) + temp_end * elapsed_frac

        # ------------------------------------------------------------------
        # Choose a move type
        # ------------------------------------------------------------------
        if np.random.rand() < 0.6:
            # Projected gradient step – step size decays with sqrt(it)
            eta = 0.5 / np.sqrt(it)
            candidate = _gradient_step(current, eta)
        else:
            # Random pairwise swap – sigma also shrinks slowly
            sigma = 0.2 / np.sqrt(it)
            candidate = _random_pair_move(current, sigma)

        # ------------------------------------------------------------------
        # Evaluate the candidate (heavy operation)
        # ------------------------------------------------------------------
        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:
            # Should never happen because we keep entries non‑negative,
            # but guard against accidental NaNs etc.
            continue

        # Keep the best ever seen
        if cand_score < best_score:
            best_score = cand_score
            best = candidate.copy()
            last_improve_iter = it
            print(f"[Iter {it:6d}] New best score: {best_score:.6f}")

        # Metropolis acceptance rule
        if cand_score < current_score:
            accept = True
        else:
            # Probability exp(-(Δ)/T).  Guard against division by zero.
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            current = candidate
            current_score = cand_score

        # ------------------------------------------------------------------
        # Optional restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if it - last_improve_iter > 5000:
            # Inject a modest random perturbation around the current best
            perturb = np.random.rand(n)
            perturb = perturb / perturb.sum()   # make it a simplex point
            # Blend a tiny fraction of the random point into the best one
            current = _project_simplex(0.99 * best + 0.01 * perturb)
            current_score = evaluate_sequence(current.tolist())
            last_improve_iter = it

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    return best.tolist()
