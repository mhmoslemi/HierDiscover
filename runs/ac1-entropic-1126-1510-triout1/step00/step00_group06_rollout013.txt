Procedure `project_simplex(v)`:

```
def project_simplex(v):
    # v: numpy array may be negative, sum need to be 1.
    n = len(v)
    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    start_time = time.time()
    deadline = start_time + budget_s - 5

    # Determine length
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 1000

    # Initialize a as uniform
    a = np.full(n, 1.0 / n)

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print("Initial score:", best_score)

    # parameters
    beta = 5.0
    eta = 0.5
    iter = 0
    while time.time() < deadline:
        # Compute conv and softmax weights
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # Softmax
        # to avoid overflow:
        scaled = beta * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / np.sum(exp_scaled)  # shape length 2n-1

        # gradient: 2 * correlation of w and a
        # correlation w (length 2n-1) with a (length n) -> result length n (valid)
        grad = 2 * np.correlate(w, a, mode='valid')

        # gradient step
        a_new = a - eta * grad
        # project onto simplex (sum to 1)
        a_new = project_simplex(a_new)

        # Evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"Iter {iter}, new best {best_score:.6f}")
        # Accept step if improvement else maybe keep anyway?
        if score <= best_score + 1e-12: # if not worse
            a = a_new
        else:
            # maybe reduce eta
            eta *= 0.9
            # optionally revert a to previous best? Keep current a
        # gradually increase beta to focus on max later

            a += np.random.randn(n) * 0.001
            a = np.maximum(a, 0)
            a = project_simplex(a)
        iter += 1
    return best_a.tolist()

def project_simplex(v):
    # v: numpy array
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The algorithm starts from a uniform distribution (the trivial sequence that gives
the known bound 2.0).  A smooth approximation of the supremum
`max_k (a*a)_k` is built by a soft‑max:
    Lβ(a) = (1/β)·log( Σ_k exp(β·(a*a)_k) ).
For β>0 the gradient of Lβ w.r.t. the coefficients a_i can be written in
closed form as a correlation between the soft‑max weights and a itself.
A projected gradient descent step is taken:
    a ← a – η·∇Lβ(a)   (η = step size)
followed by a projection onto the probability simplex so that the coefficients
remain non‑negative and sum to 1 (the objective is scale‑invariant, therefore a
fixed L1‑norm is convenient).

β is slowly increased (the approximation becomes tighter) while η is
adapted: when a step does not improve the true objective
`evaluate_sequence` the step size is reduced.  Occasionally a small random
perturbation is added to avoid stagnation.

The loop runs until the time budget (≈ 1000 s) expires and the best sequence
found is returned.
"""

import time
import numpy as np

def _project_simplex(v):
    """Project a vector onto the simplex {x ≥ 0, sum(x)=1}."""
    # based on algorithm of Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive → return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Return a list of non‑negative heights with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate time budget in seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0    # a small safety margin

    # length of the sequence – use the known construction if available
    n = len(globals().get('height_sequence_1', []))
    if n == 0:               # fallback
        n = 2000

    # initialise on the simplex (sum = 1, all entries ≥0)
    a = np.full(n, 1.0 / n)

    # best solution seen so far
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    curr_score = best_score
    print(f"[init] n={n}, score={best_score:.6f}")

    # hyper‑parameters (will be tuned on‑the‑fly)
    beta = 5.0          # soft‑max temperature (higher → closer to true max)
    eta = 0.8           # learning rate
    iteration = 0

    while time.time() < deadline:
        # --- convolution and soft‑max weights ---------------------------------
        conv = np.convolve(a, a)                 # length 2n-1
        # soft‑max on β·conv, numerically stable
        scaled = beta * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w_sum = w.sum()
        if w_sum == 0:
            # should never happen, but guard against it
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # --- gradient of the smoothed max (log‑sum‑exp) ----------------------
        # ∂L/∂a_i = 2 * Σ_k w_k * a_{k-i}
        grad = 2.0 * np.correlate(w, a, mode='valid')   # shape (n,)

        # --- gradient step with projection onto the simplex ------------------
        a_candidate = a - eta * grad
        a_candidate = _project_simplex(a_candidate)

        # evaluate true objective
        try:
            cand_score = evaluate_sequence(a_candidate.tolist())
        except Exception:
            cand_score = np.inf

        # --- acceptance -------------------------------------------------------
        if cand_score < curr_score:
            # step was successful
            a = a_candidate
            curr_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                print(f"[iter {iteration}] new best score = {best_score:.6f}")
        else:
            # reject – shrink step size
            eta *= 0.9

        # --- schedule updates -------------------------------------------------
        if iteration % 30 == 0 and beta < 200:
            beta *= 1.05          # tighten soft‑max
        if iteration % 40 == 0 and eta < 1.5:
            eta = min(eta * 1.02, 1.0)   # gently increase if things are stable

        # occasional random jitter to escape plateaus
        if iteration % 250 == 0 and iteration > 0:
            jitter = np.random.randn(n) * 0.003
            a = _project_simplex(a + jitter)

        # early exit if we already beat the target
        if best_score <= 1.505:      # a tiny margin below the target 1.503
            print("[early stop] target reached")
            break

        iteration += 1

    print(f"[final] best score = {best_score:.6f} after {iteration} iterations")
    # return as plain list (the evaluation harness expects a Python list)
    return best_a.tolist()
