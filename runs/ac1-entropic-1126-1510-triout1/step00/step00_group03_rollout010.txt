Approach
--------
1. The evaluation metric is scale‑invariant:
   evaluate_seq(a) = 2·n·max(conv(a,a)) / (sum(a)**2).
   Hence we can normalise the sequence to any convenient sum, e.g. 1.

2. For a non‑negative vector `x` of length `n` with sum 1 define
      M(x) = max_k (x * x)[k]   (normal convolution).
   Uniform `x_i = 1/n` gives `M = 1/n` and a score of exactly 2.
   By concentrating the mass towards the beginning (a decreasing power‑law)

   convolution) becomes smaller while all other shifts stay small.
   Theory shows that for `x_i ∝ (i+1)^{-p}` with `0<p<1` the asymptotic

   scores around 1.3 – 1.5, well below the current bound 2.0.

3. We therefore search a one‑dimensional parameter `p` (using a golden‑section
   line search) on the power‑law family, keep the best sequence, and then
   perform a quick hill‑climbing phase that moves a tiny amount of mass from
   one index to another while maintaining the total sum.  Each improvement
   is kept.

The whole routine respects the 1000 s budget, prints progress and finally
returns the best sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------------------- #
# Helper functions (all top‑level – no closures or lambdas as required)
# --------------------------------------------------------------------------- #

def power_law_sequence(n: int, p: float) -> np.ndarray:
    """
    Return a decreasing power‑law vector of length n, normalised to sum 1.
    a_i ∝ (i+1)^{-p}.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)               # safety fallback
    a /= a_sum
    return a


def evaluate_vec(vec: np.ndarray) -> float:
    """
    Wrapper around the provided evaluate_sequence (available in the global
    namespace).  The function accepts a numpy array and returns the float score.
    """
    # evaluate_sequence expects a plain Python list
    return evaluate_sequence(vec.tolist())


def golden_section_search(n: int, left: float, right: float,
                         max_iters: int, time_limit: float):
    """
    1‑D optimisation of the exponent `p` in the power‑law family.
    Returns (best_p, best_score, best_vec).
    """
    phi = (np.sqrt(5) - 1) / 2          # 0.618...
    a = left
    b = right
    # interior points
    p1 = a + (1 - phi) * (b - a)
    p2 = a + phi * (b - a)

    v1 = evaluate_vec(power_law_sequence(n, p1))
    v2 = evaluate_vec(power_law_sequence(n, p2))

    best_score = min(v1, v2)
    best_p = p1 if v1 <= v2 else p2
    best_vec = power_law_sequence(n, best_p)

    it = 0
    while it < max_iters and time.time() < time_limit:
        if v1 < v2:
            b = p2
            p2 = p1
            v2 = v1
            p1 = a + (1 - phi) * (b - a)
            v1 = evaluate_vec(power_law_sequence(n, p1))
            if v1 < best_score:
                best_score = v1
                best_p = p1
                best_vec = power_law_sequence(n, best_p)
        else:
            a = p1
            p1 = p2
            v1 = v2
            p2 = a + phi * (b - a)
            v2 = evaluate_vec(power_law_sequence(n, p2))
            if v2 < best_score:
                best_score = v2
                best_p = p2
                best_vec = power_law_sequence(n, best_p)
        it += 1

    return best_p, best_score, best_vec


def hill_climb(vec: np.ndarray, time_deadline: float,
               max_no_improve: int = 2000) -> np.ndarray:
    """
    Small stochastic hill‑climbing that moves a tiny amount of probability mass
    between two randomly chosen positions while preserving the total sum.
    Accepts only strictly improving moves.
    """
    n = len(vec)
    curr = vec.copy()
    curr_score = evaluate_vec(curr)
    no_improve = 0

    while time.time() < time_deadline and no_improve < max_no_improve:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # take at most 5 % of the source entry (to keep steps tiny)
        delta = np.random.rand() * 0.05 * curr[i]
        if delta <= 0:
            continue
        cand = curr.copy()
        cand[i] = max(0.0, cand[i] - delta)
        cand[j] = cand[j] + delta          # sum is preserved
        cand_score = evaluate_vec(cand)
        if cand_score < curr_score:
            curr = cand
            curr_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
    return curr


# --------------------------------------------------------------------------- #
# Main entry point required by the competition framework
# --------------------------------------------------------------------------- #

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a sequence of non‑negative heights that (hopefully) improves upon the
    current best known bound (2.0).  The routine performs:

    * a golden‑section search over the exponent `p` of a decreasing power‑law,
      using the existing global `height_sequence_1` to obtain the length,
    * a short hill‑climbing refinement,
    * and finally returns the best list it discovered.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------- #
    # Determine the length to work with.
    # ------------------------------------------------------------------- #
    try:
        n = len(height_sequence_1)               # use the known construction length
    except Exception:
        n = 2000                                 # fallback length if not defined

    deadline = time.time() + budget_s - 5.0      # keep a small safety margin

    # ------------------------------------------------------------------- #
    # 1) Exponent search on the power‑law family
    # ------------------------------------------------------------------- #
    # we restrict the exponent to (0.05, 0.95) – values outside this range
    # either give something close to uniform (p≈0) or a too spiky front‑loaded
    # distribution that hurts the early convolution entries.
    left, right = 0.05, 0.95
    max_iters = 30                               # more than enough for convergence

    best_p, best_score, best_vec = golden_section_search(
        n, left, right, max_iters, deadline)

    print(f"[search] best exponent after line search: p = {best_p:.5f}, "
          f"score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # 2) Local refinement via hill climbing
    # ------------------------------------------------------------------- #
    # allocate a second time slice for hill climbing (if time permits)
    hill_deadline = min(deadline, time.time() +  (budget_s * 0.3))
    refined_vec = hill_climb(best_vec, hill_deadline)

    refined_score = evaluate_vec(refined_vec)
    if refined_score < best_score:
        best_vec = refined_vec
        best_score = refined_score
        print(f"[hill] improved after hill‑climb: score = {best_score:.6f}")

    # ------------------------------------------------------------------- #
    # Final output
    # ------------------------------------------------------------------- #
    print(f"[final] best score obtained: {best_score:.6f} "
          f"with length {n}")
    # The evaluate_sequence function will clamp negatives / outsized values,
    # but we already guarantee the vector is non‑negative and well inside limits.
    return best_vec.tolist()
