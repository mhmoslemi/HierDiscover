"""
A simple simulated‑annealing / greedy optimizer for the step‑function problem.

The goal is to minimise
    2 * n * max(conv(a, a)) / (sum(a))**2
with a ≥ 0 and sum(a) > 0.  The objective is scale‑invariant, so we keep
the sequence normalised (sum = 1) during the search.  The algorithm works
as follows:

1. Choose a length n (taken from the existing best construction when
   available, otherwise 1600) and initialise a random probability vector.
2. Repeatedly pick the index r where the self‑convolution is maximal.
   Reduce the weight at an index i that contributes heavily to this peak
   and move a small amount of mass to a random index j.
   The move is accepted if it lowers the maximal convolution value,
   otherwise it is accepted with a tiny probability (simulated‑annealing).
3. The step size gradually shrinks; after a number of iterations a
   completely new random start is performed (a restart) to avoid getting
   stuck.
4. The best sequence seen so far (lowest evaluation score) is returned.

All heavy lifting is done with FFT based convolution, which gives an
O(n log n) update per iteration and easily fits into the 1000 s time
budget.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def convolve_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Return the linear convolution of ``a`` with itself using a real FFT.
    The result length is 2*len(a)-1.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    # rfft works on real data and returns the positive frequency terms
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)
    # Only the first (2n‑1) entries belong to the linear convolution
    return conv[: 2 * n - 1]


def max_convolution(a: np.ndarray) -> float:
    """Compute max(conv(a, a)).  a is assumed non‑negative."""
    return convolve_self_fft(a).max()


def objective_from_max(n: int, max_conv: float) -> float:
    """Objective value (2 * n * max_conv) assumes sum(a)=1."""
    return 2.0 * n * max_conv


def _local_search(
    a: np.ndarray, time_budget: float, verbose: bool = False
) -> np.ndarray:
    """
    Greedy / simulated annealing improvement starting from a normalised
    vector ``a`` (sum(a)=1).  The function runs for at most ``time_budget``
    seconds and returns the best vector found.
    """
    deadline = time.time() + time_budget
    n = a.shape[0]

    # best solution seen so far
    best_a = a.copy()
    best_max = max_convolution(best_a)

    # step size is a fraction of the current source entry; it shrinks over time
    step_fraction = 0.08  # initial magnitude
    min_step_frac = 0.002
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        conv = convolve_self_fft(a)
        cur_max = conv.max()
        r = int(conv.argmax())

        # indices that actually contribute to the peak at r
        i_start = max(0, r - (n - 1))
        i_end = min(n - 1, r)
        idxs = np.arange(i_start, i_end + 1)
        partners = r - idxs
        mask = (partners >= 0) & (partners < n)
        idxs = idxs[mask]
        partners = partners[mask]

        # contribution weights a[i] * a[partner]
        weights = a[idxs] * a[partners]
        if weights.sum() > 0 and random.random() < 0.85:
            # sample a source index proportionally to its contribution
            probs = weights / weights.sum()
            i = np.random.choice(idxs, p=probs)
        else:
            i = random.randrange(n)

        # choose a destination index different from i
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        # step size (absolute amount moved)
        delta = step_fraction * a[i]
        # never remove more than half of the source entry
        delta = min(delta, 0.5 * a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = max_convolution(a_new)

        # Accept if strictly better, otherwise accept with tiny probability
        if new_max < cur_max or random.random() < 0.001:
            a = a_new
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        # decay the step size every few hundred iterations
        if iteration % 300 == 0:
            step_fraction = max(min_step_frac, step_fraction * 0.93)

        # occasional logging
        if verbose and iteration % 1000 == 0:
            obj = objective_from_max(n, cur_max)
            best_obj = objective_from_max(n, best_max)
            print(
                f"[Local search] it={iteration:,} obj={obj:.6f} "
                f"best={best_obj:.6f} step={step_fraction:.5f}"
            )

        # If we've made no improvement for a long time, break early
        if no_improve > 3000:
            break

    return best_a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights (the step function) that
    minimises the provided ``evaluate_sequence`` score.

    The routine uses a simple stochastic local search based on FFT
    convolution and runs for at most ``budget_s`` seconds.
    """
    # deterministic randomness
    np.random.seed(seed)
    random.seed(seed)

    # total time budget (leave a small safety margin)
    end_time = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose a working length.  Prefer the length of the known best
    # construction (height_sequence_1) when it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(
        height_sequence_1, (list, tuple, np.ndarray)
    ):
        base_len = len(height_sequence_1)
    else:
        base_len = 1600  # sensible default

    # Clamp the length to a range that is fast enough for the search
    n = max(500, min(2500, base_len))

    best_seq = None
    best_score = float("inf")
    restarts = 4  # number of independent random starts

    for r in range(restarts):
        if time.time() >= end_time:
            break

        # ------------------------------------------------------------------
        # Initialise a starting vector (normalised to sum = 1)
        # ------------------------------------------------------------------
        if r == 0 and "height_sequence_1" in globals():
            a0 = np.array(height_sequence_1, dtype=float)
            if a0.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
            # If the stored sequence is longer/shorter than n, truncate/pad
            if a0.shape[0] > n:
                a0 = a0[:n]
            elif a0.shape[0] < n:
                a0 = np.pad(a0, (0, n - a0.shape[0]), constant_values=0.0)
        else:
            # Random Dirichlet (exponential) start – gives a diverse profile
            a0 = np.random.exponential(scale=1.0, size=n)
            a0 = a0 / a0.sum()

        # ------------------------------------------------------------------
        # Allocate time for this restart (share the remaining budget)
        # ------------------------------------------------------------------
        remaining = end_time - time.time()
        # keep a small fraction for final evaluation/printing
        alloc = max(5.0, 0.3 * remaining)

        # Run the local optimisation
        a_opt = _local_search(a0, alloc, verbose=False)

        # Evaluate with the provided (slow) checker only for the final candidate
        seq = a_opt.tolist()
        try:
            score = evaluate_sequence(seq)
        except Exception:
            # any unexpected problem – treat as terrible
            score = float("inf")

        if score < best_score:
            best_score = score
            best_seq = seq
            print(
                f"[{r+1}/{restarts}] New best score {best_score:.6f} (len={n})"
            )
        else:
            if r == 0:
                print(
                    f"[{r+1}/{restarts}] First run score {score:.6f} (no improvement)"
                )

    # Fallback – uniform distribution (should never happen)
    if best_seq is None:
        best_seq = [1.0] * n

    return best_seq
