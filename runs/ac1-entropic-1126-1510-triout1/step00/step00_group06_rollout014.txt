import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generate power-law increasing sequences a_i = (i+1)^p, normalize to sum=1,
    evaluate them and return the one with the smallest evaluation score.
    The search explores several lengths (n) and exponents (p) within the time
    budget.  The algorithm is deterministic (apart from the optional seed) and
    runs far below the 1000‑second limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # leave a tiny margin

    best_seq = None
    best_score = float('inf')

    # candidate lengths – we try both short and long sequences.
    lengths = [200, 500, 1000, 2000, 3130, 4000, 5000]

    # exponents to shape the distribution; p=0 gives uniform,
    # larger p places more mass near the right hand side → smaller max‑convolution.
    exponents = [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0]

    for n in lengths:
        # early exit if we run out of time
        if time.time() > deadline:
            break
        # pre‑compute integer indices (1‑based) for speed
        idx = np.arange(1, n + 1, dtype=np.float64)

        for p in exponents:
            if time.time() > deadline:
                break

            raw = idx ** p
            total = raw.sum()
            # Normalise to sum = 1 (any positive sum works because the metric is scale‑invariant)
            seq = (raw / total).tolist()

            # The evaluation function will clip values >1000, but our normalisation guarantees
            # the maximum entry is ~= (p+1)/n < 0.01 for all candidates, so clipping never occurs.
            score = evaluate_sequence(seq)
            # In case the checker returns np.inf (should not happen) we ignore the candidate.
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq
                print(f"New best: n={n}, p={p:.1f}, score={score:.6f}")

    # If for some reason no candidate was accepted, fall back to a short uniform sequence.
    if best_seq is None:
        n = 100
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback sequence, score={best_score}")

    print(f"Final best score: {best_score:.6f} with length {len(best_seq)}")
    return best_seq

import time
import numpy as np

# evaluate_sequence is provided in the execution environment.

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value step‑function by trying simple power‑law shapes.
    --------------------------------------------------------------------
    For a chosen length `n` we build sequences of the form

        a_i = (i+1)^p      for i = 0 … n‑1

    and renormalise them so that their sum equals 1.
    (The evaluation metric is scale‑invariant, therefore any positive normalisation
    works.)  Larger exponents push most of the mass towards the right end of the
    array; this dramatically reduces the peak of the self‑convolution while keeping
    the L1‑norm fixed.  For each pair (n, p) we evaluate the sequence and keep the
    one with the smallest score.  The search explores a handful of lengths and
    exponents and finishes well before the 1000 s budget.

    The returned list contains non‑negative floats and respects all security checks
    performed by `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # candidate lengths – short, medium and very long vectors.
    # ------------------------------------------------------------------
    lengths = [200, 500, 1000, 2000, 3130, 4000, 5000]

    # exponents that shape the distribution.
    # p = 0  -> uniform (score = 2)
    # p > 0  -> increasingly skewed to the right, lowering the max‑convolution.
    exponents = [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0]

    for n in lengths:
        if time.time() > deadline:
            break
        # pre‑compute the integer indices (1‑based) as float – vectorised power is fast.
        idx = np.arange(1, n + 1, dtype=np.float64)

        for p in exponents:
            if time.time() > deadline:
                break

            # ------- build a raw power‑law vector ---------------------------------
            raw = idx ** p                     # a_i ∝ (i+1)^p
            total = raw.sum()
            seq = (raw / total).tolist()       # normalise to L1‑norm = 1

            # All entries are ≤ (p+1)/n  (< 0.01 for the ranges we test),
            # therefore the clipping inside `evaluate_sequence` never alters the data.

            score = evaluate_sequence(seq)

            # keep the best (lowest) score
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = seq
                print(f"New best → n={n}, p={p:.1f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # safeguard: if for some reason nothing was accepted, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 100
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback sequence, score={best_score}")

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
