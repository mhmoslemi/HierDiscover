"""
Propose a candidate step‑function using a fast successive linear‑programming
update. Starting from the current best sequence we repeatedly

1. compute the current peak M = max(conv(a,a)),
2. solve a linear programme that maximises the L1‑norm of a new vector g
   while keeping conv(a,g) ≤ M,
3. normalise g to the same total mass,
4. blend a and g with a set of step sizes and keep the one that yields the
   lowest evaluation score.

The loop runs until the time budget expires. Random perturbations and occasional
re‑initialisation from a Dirichlet sample help to escape local minima.
"""

import time
import numpy as np
from scipy.optimize import linprog

def solve_convolution_lp(f, rhs):
    """Linear programme:
       maximise sum(b) s.t.   (f * b)[k] ≤ rhs  for all k,
       b ≥ 0.
    """
    n = len(f)
    # coefficients for the linear objective (maximise sum(b) → minimise -sum(b))
    c = -np.ones(n)

    # Build the convolution constraints: for each k the row contains f_i for which
    # i + j = k → coefficient for b_j is f_i.
    # This is a (2n‑1)×n matrix.
    A_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i ranges over all indices such that j = k - i is in [0,n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For each such i, coefficient for variable b_{k-i} is f_i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        A_rows.append(row)
        b_rows.append(rhs)

    # Non‑negativity constraints:  -b ≤ 0
    A_ub = np.vstack([A_rows, -np.eye(n)])
    b_ub = np.hstack([b_rows, np.zeros(n)])

    # Solve with the HiGHS solver – it is fast for the modest problem sizes we use.
    # We set a modest time limit to avoid pathological runs.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        return res.x
    return None


def improve_sequence(seq, max_iter=30, time_deadline=None):
    """
    Perform a few LP‑driven updates on `seq` and return the best improvement
    discovered. `seq` is assumed to be a NumPy array with non‑negative entries
    and unit L1‑norm.
    """

    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())
    for _ in range(max_iter):
        # Current peak of the autocorrelation
        conv = np.convolve(best, best)
        M = conv.max()

        # Linear programme that tries to increase the L1‑norm while keeping
        # the cross‑correlation bounded by the current peak.
        g = solve_convolution_lp(best.tolist(), M)
        if g is None:
            # Simple stochastic perturbation – keep us away from a degenerate LP.
            i = np.random.randint(n)
            best[i] = max(best[i] + np.random.randn() * 0.005, 0.0)
            # Re‑normalise to keep the L1‑norm equal to one.
            best /= best.sum()
            continue

        # Normalise the returned vector.
        g = np.asarray(g, dtype=float)
        g /= g.sum()

        # Try several mixing coefficients.
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = (1.0 - t) * best + t * g
            # Keep the sum exactly one – the scaling does not affect the objective.
            cand /= cand.sum()
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:

            best /= best.sum()

        # Time check to avoid overrunning the budget.
        if time_deadline is not None and time.time() > time_deadline:

"""
Propose a candidate step‑function using a fast LP‑driven descent.

The algorithm works as follows:

1. Start from the publicly known construction ``height_sequence_1`` (if it
   exists) or from a uniform vector.
2. Repeatedly:
   * Compute the current peak ``M = max(conv(a, a))``.
   * Solve a linear programme that finds a non‑negative vector ``g`` with the
     largest possible L¹‑norm while keeping ``conv(a, g) ≤ M``.
   * Normalise ``g`` to have unit L¹‑norm and try a few convex combinations
     ``(1‑t)·a + t·g`` (with several values of ``t``).  Keep the combination that
     gives the smallest evaluation score.
3. Occasionally restart from a random Dirichlet sample or apply a tiny random
   jitter to escape shallow local minima.
4. Stop when the allotted time (≈ 1000 s) is exhausted and return the best
   sequence found.

The optimisation target is the value returned by ``evaluate_sequence``:
``2·n·max(conv(a,a)) / (sum(a)²)``.  Scaling the vector does not change this
value, therefore we keep the L¹‑norm equal to 1 throughout the search.
"""

import time
import numpy as np
from scipy.optimize import linprog

def solve_convolution_lp(f_seq, rhs):
    """
    Linear programme:
        maximise   sum(b)
        subject to conv(f_seq, b)[k] ≤ rhs   for every k = 0 … 2·n‑2
                    b ≥ 0
    Returns the optimal vector ``b`` as a NumPy array or ``None`` on failure.
    """
    n = len(f_seq)
    # objective – we minimise –sum(b) to obtain a maximisation of sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.
    # Row k contains the coefficients f_i for those i for which a variable b_j
    # with j = k‑i participates.
    rows = []
    bounds = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i ranges over admissible indices; j = k‑i is then in [0, n‑1]
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        bounds.append(rhs)

    # Non‑negativity constraints:  -b ≤ 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([bounds, np.zeros(n)])

    # Use the HiGHS solver – very fast for the modest sizes we encounter.
    # A short time limit prevents pathological runs.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.success:

    return None

        # Test a handful of mixing coefficients.
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            cand = (1.0 - t) * best + t * g
            cand /= cand.sum()
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best = cand
                improved = True
                break   # keep the first (largest) t that yields improvement

        if not improved:
            # Tiny random perturbation if nothing helped.
            i = np.random.randint(n)
            best[i] = max(best[i] + np.random.randn() * 0.001, 0.0)
            best /= best.sum()

        if deadline is not None and time.time() > deadline:
            break

    return best, best_score


def propose_candidate(seed=0, budget_s=1000, **_):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``. The function respects the 1000 s time budget and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # Length of the construction – use the length of the known solution if
    # available, otherwise fall back to a reasonably large size.
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 1600

    # Initialise from the public construction or from a uniform vector.
    try:
        base = np.array(height_sequence_1, dtype=float)
        base = np.maximum(base, 0.0)
        base /= base.sum()
    except NameError:
        base = np.full(n, 1.0 / n)

    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score:.6f}")

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # With a modest probability start from a fresh random Dirichlet vector.
        if np.random.rand() < 0.15:
            cur = np.random.rand(n)
            cur /= cur.sum()
        else:
            cur = best_seq.copy()

        # Apply a short burst of LP‑driven refinement.
        cur, cur_score = improve_sequence(cur, max_iter=25, deadline=deadline)

        # Keep the globally best solution.
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[{iteration:04d}] New global best: {best_score:.6f}")

        # Every few iterations try a tiny random jitter of the global best.
        if iteration % 30 == 0 and np.random.rand() < 0.5:
            jitter = best_seq.copy()
            i = np.random.randint(n)
            jitter[i] = max(jitter[i] + np.random.randn() * 0.01, 0.0)
            jitter /= jitter.sum()
            jitter_score = evaluate_sequence(jitter.tolist())
            if jitter_score < best_score:
                best_score = jitter_score
                best_seq = jitter
                print(f"[{iteration:04d}] Jitter improvement: {best_score:.6f}")

    print(f"Finished. Best score found: {best_score:.6f}")
    return best_seq.tolist()
