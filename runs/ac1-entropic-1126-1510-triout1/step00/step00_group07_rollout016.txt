```python
def gradient_step(a, lr):
    n = len(a)
    # Compute convolution
    b = np.convolve(a, a)
    M = np.max(b)
    kmax = np.argmax(b)
    # Compute gradient of M w.r.t a
    g = np.zeros_like(a)
    for i in range(n):
        j = kmax - i
        if 0 <= j < n:
            g[i] = 2 * a[j]  # derivative of b_kmax w.r.t a_i
    # Gradient of f (assuming sum=1)
    df = 2 * n * g - 4 * n * M * np.ones_like(a)  # because S=1
    # Project onto sum=0 subspace: subtract mean
    d_proj = -df  # descent direction
    d_proj = d_proj - np.mean(d_proj)  # ensure sum 0
    # Update
    a_new = a + lr * d_proj
    # Clip
    a_new[a_new < 0] = 0
    # Renormalize
    s = np.sum(a_new)
    if s == 0:
        # reinitialize
        a_new = np.random.rand(n)
        a_new /= a_new.sum()
    else:
        a_new /= s
    return a_new
```

from scipy import sparse

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    rows = 2*n - 1 + n
    A_data = []
    A_rows = []
    A_cols = []
    # for each k in 0..2n-2:
    row_idx = 0
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_data.append(f_seq[i])
                A_rows.append(row_idx)
                A_cols.append(j)
        row_idx += 1
    # non-negativity constraints: -b <= 0
    for i in range(n):
        A_data.append(-1.0)
        A_rows.append(row_idx)
        A_cols.append(i)
        row_idx += 1
    A_ub = sparse.csr_matrix((A_data, (A_rows, A_cols)), shape=(rows, n))
    b_ub = np.concatenate([np.full(2*n-1, rhs), np.zeros(n)])
    c = -np.ones(n)   # maximize sum(b) => minimize -sum(b)
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 5})
    if result.success:
        return result.x
    return None
```

"""
Simple iterative improvement using the LP‑direction from the reference
implementation, but with an adaptive line‑search for the mixing weight.
The algorithm repeatedly:
  1. Normalises the current sequence to have L1‑norm √(2 n) (the scale used
     by the evaluation function).
  2. Solves the linear programme that finds a new non‑negative sequence *g*
     with a larger L1‑norm while keeping the convolution bound ‖f∗g‖∞
     below the current maximum.
  3. Performs a cheap ternary‑search on the mixing coefficient t∈[0,1] for
     the convex combination  (1‑t)·f + t·g  and keeps the best point.
  4. If the LP fails, a small random perturbation is applied.
The process is run many times (within the 1000 s budget) from random
re‑starts as well as from the supplied state (``height_sequence_1``).  The
best sequence found is returned.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def _solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise   sum(b_j)
        subject to   (f * b)(k) <= rhs  for all k,
                     b_j >= 0.
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # Build the inequality matrix A_ub * b <= b_ub.
    # First (2*n-1) rows: convolution constraints.
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # (f * b)_k = sum_i f_i * b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        rows.append(row)
        rhs_vec.append(rhs)

    # Non‑negativity constraints: -b_j <= 0.
    rows.extend(-np.eye(n))
    rhs_vec.extend([0.0] * n)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs_vec, dtype=float)

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # Highs is fast and robust; give a modest time limit per solve.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  method='highs',
                  options={'time_limit': 8.0, 'disp': False})
    if res.success:
        return res.x
    return None


def _line_search(cur, g, max_evals=7):
    """
    Ternary search on the mixing coefficient t∈[0,1] for the combination
    (1‑t)*cur + t*g.  Returns the best sequence and its score.
    """
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    left, right = 0.0, 1.0
    # one extra evaluation at t = 0 (already done) and t = 1 (full g)
    for _ in range(max_evals):
        t1 = left + (right - left) / 3.0
        t2 = right - (right - left) / 3.0

        seq1 = (1.0 - t1) * cur + t1 * g
        seq2 = (1.0 - t2) * cur + t2 * g

        score1 = evaluate_sequence(seq1.tolist())
        score2 = evaluate_sequence(seq2.tolist())

        if score1 < best_score:
            best_score = score1
            best_seq = seq1.copy()
        if score2 < best_score:
            best_score = score2
            best_seq = seq2.copy()

        # Keep the interval containing the smaller value
        if score1 < score2:
            right = t2
        else:
            left = t1

    # also try the endpoint t = 1 (pure g)
    g_score = evaluate_sequence(g.tolist())
    if g_score < best_score:
        best_score = g_score
        best_seq = g.copy()

    return best_seq, best_score


def _improve_once(sequence):
    """
    One improvement round: solve the LP direction and perform a line‑search.
    Returns a possibly better sequence (or the original one if nothing improved).
    """
    n = len(sequence)
    s = np.sum(sequence)
    if s < 1e-12:
        return sequence  # degenerate – should never happen

    # Normalise to the standard L1 norm √(2 n) used by the LP.
    target_norm = np.sqrt(2.0 * n)
    f_norm = sequence * (target_norm / s)

    # Current convolution bound (the quantity that the LP must respect).
    rhs = np.max(np.convolve(f_norm, f_norm))

    # Solve the LP for a better direction.
    g_raw = _solve_convolution_lp(f_norm, rhs)
    if g_raw is None:
        return None

    # Normalise the new vector to the same L1 norm.
    g_norm = g_raw * (target_norm / np.sum(g_raw))

    # Line‑search on the mixing weight t.
    new_seq, new_score = _line_search(f_norm, g_norm)
    return new_seq

    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The routine runs for at most ``budget_s`` seconds,
    returning the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a safety margin

    # Use a couple of different lengths – short lengths give cheap trials,
    # long lengths (the current best size) are used for final refinement.
    lengths = [400, 800, 1600, 3937]   # the last one is the known state‑of‑the‑art size

    best_seq = None
    best_score = float("inf")

    # Helper to evaluate safely.
    def _score(seq):
        return evaluate_sequence(seq.tolist())

    # Main search loop.
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Choose an initial point.
        # ------------------------------------------------------------------
        n = int(np.random.choice(lengths))
        if np.random.rand() < 0.4 and 'height_sequence_1' in globals():
            # start from the known construction (trim / pad as needed)
            seq = np.array(height_sequence_1, dtype=float)
            if len(seq) > n:
                seq = seq[:n]
            elif len(seq) < n:

        else:

            seq = np.random.rand(n)
            seq /= seq.sum()
            seq *= np.sqrt(2.0 * n)   # bring to the standard normalisation

        cur_seq = seq.copy()
        cur_score = _score(cur_seq)

        # Track the best from this start.

        # ------------------------------------------------------------------
        inner_deadline = time.time() + 8.0   # a few seconds per start point

            # Attempt one LP‑based improvement.
            new_seq = _improve_once(cur_seq)
            if new_seq is None:

                idx = np.random.randint(len(cur_seq))
                delta = (np.random.rand() - 0.5) * 0.02 * cur_seq[idx]
                cur_seq[idx] = max(0.0, cur_seq[idx] + delta)

                cur_seq *= np.sqrt(2.0 * len(cur_seq)) / np.sum(cur_seq)
                cur_score = _score(cur_seq)

            cand_score = _score(new_seq)
            if cand_score < cur_score:
                cur_seq = new_seq
                cur_score = cand_score
                # update global best if needed
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = new_seq.copy()
                    print(f"[{time.time():.1f}] New global best: {best_score:.6f} (len={len(best_seq)})")
            else:

                break

        if np.random.rand() < 0.15:
            continue

    # Fallback in the unlikely case no sequence was found.
    if best_seq is None:
        best_seq = np.ones(4000, dtype=float) * (np.sqrt(2.0 * 4000) / 4000)

    # Return as a plain Python list (as required by the evaluator).
    return best_seq.tolist()
