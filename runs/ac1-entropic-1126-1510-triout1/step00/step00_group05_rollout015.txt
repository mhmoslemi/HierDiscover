def gradient_step(a):
    b = np.convolve(a, a)
    k = np.argmax(b)
    grad = np.zeros_like(a)
    # compute grad_i = 2 * a[k - i] if within bound
    for i in range(len(a)):
        j = k - i
        if 0 <= j < len(a):
            grad[i] = 2 * a[j]
    sum_grad = np.sum(grad)
    d = -grad + sum_grad / len(a)  # keep sum constant
    # compute maximum step size to keep a_i+η*d_i>=0
    max_eta = np.min([-a_i/d_i if d_i<0 else np.inf for a_i,d_i in zip(a,d) if d_i<0], initial=np.inf)

    eta = min(0.1 * max_eta, some_max)  # maybe adaptive
    a_new = a + eta * d
    a_new = np.clip(a_new, 0, 1000)
    return a_new

```
def gradient_step(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    grad = np.zeros(n)
    # gradient_i = 2 * a_{k - i}
    # compute vectorized
    idx = np.arange(n)
    # compute j = k - idx
    j = k - idx
    valid = (j >= 0) & (j < n)
    grad[valid] = 2 * seq[j[valid]]
    sum_grad = np.sum(grad)
    # ensure sum_grad nonzero
    d = -grad + sum_grad / n
    # compute max feasible step size to keep nonneg
    # We need η such that seq + η d >= 0
    # For each i where d_i < 0, η <= -seq_i / d_i
    neg = d < 0
    if np.any(neg):
        max_eta = np.min(-seq[neg] / d[neg])
        eta = 0.5 * max_eta  # take half to stay safely positive
    else:
        eta = 0.1  # arbitrary small step
    # Optionally limit step size
    eta = min(eta, 0.1 * np.mean(seq) if np.mean(seq) > 0 else eta)
    new_seq = seq + eta * d
    # Clip to [0,1000]
    new_seq = np.clip(new_seq, 0, 1000)
    return new_seq
```

        for j in range(j_start, j_end + 1):
            i = k - j
            A[k, j] = f_seq[i]
    # Convert to CSR
    A_csr = A.tocsr()
    b = np.full(2 * n - 1, rhs)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_csr, b_ub=b, bounds=bounds, method='highs')
        if result.success:
            return result.x
    except Exception as e:
        print("LP solve exception", e)
    return None

"""
Hybrid optimizer for the step‑function bound.

The algorithm starts from a known construction (the global ``height_sequence_1``
if it exists) and repeatedly performs three kinds of moves:

1. **LP direction** – solve a linear program that finds a sequence `g`
   with a larger total sum while keeping the convolution with the current
   sequence `f` below the present maximum value `M`.  This direction is the
   core of the state‑of‑the‑art method.

2. **Line‑search mixing** – instead of using a fixed tiny step, we search for
   the optimal mixing coefficient `t∈[0,1]` for the convex combination
   `h = (1‑t)·f + t·g`.  The optimal `t` is found by a cheap golden‑section
   search using the provided ``evaluate_sequence`` routine.

3. **Random local perturbations** – occasional Gaussian noise and a short
   projected gradient step help the search escape flat plateaus.

All moves are evaluated and the best sequence seen so far is kept.
The routine respects the 1000 s time budget, prints progress, and finally
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _normalize(seq: np.ndarray) -> np.ndarray:
    """Scale the sequence so that its L1‑norm equals sqrt(2·n)."""
    n = len(seq)
    s = np.sum(seq)
    if s == 0:
        return seq.copy()
    factor = np.sqrt(2.0 * n) / s
    return seq * factor


def _max_convolution(seq: np.ndarray) -> float:
    """Maximum entry of the autocorrelation of ``seq``."""
    return np.max(np.convolve(seq, seq))


def _solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Linear program:
        maximise   sum(b_j)
        subject to  (f_seq * b)_k <= rhs  for all k,
                    b_j >= 0
    Returns the optimal ``b`` as a 1‑D NumPy array or ``None`` on failure.
    """
    n = len(f_seq)
    # Objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build A_ub (2n‑1 constraints).  For each k we have
    #   sum_j f_seq[k‑j] * b_j   (with out‑of‑range indices ignored)  <= rhs
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    for k in range(rows):
        # columns j such that i = k‑j is in [0, n)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        j = np.arange(j_start, j_end + 1, dtype=int)
        i = k - j
        A[k, j] = f_seq[i]

    b = np.full(rows, rhs, dtype=np.float64)
    bounds = [(0, None)] * n

    # The default HiGHS solver is fast and robust.
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if res.success:
            return res.x
    except Exception as e:               # pragma: no cover – defensive
        print("LP exception:", e)
    return None


def _line_search_opt_t(f: np.ndarray, g: np.ndarray,
                       deadline: float, max_iter: int = 20) -> float:
    """
    Golden‑section search for the mixing weight t ∈ [0,1] that minimises
    ``evaluate_sequence((1‑t)·f + t·g)``.  Stops early if the time budget is
    exhausted and returns the best t found so far.
    """
    lo, hi = 0.0, 1.0
    best_t = 0.0
    best_val = float("inf")

    phi = (1 + np.sqrt(5)) / 2.0
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        t1 = hi - (hi - lo) / phi
        t2 = lo + (hi - lo) / phi

        seq1 = (1 - t1) * f + t1 * g
        seq2 = (1 - t2) * f + t2 * g

        val1 = evaluate_sequence(seq1.tolist())
        val2 = evaluate_sequence(seq2.tolist())

        if val1 < val2:
            hi = t2
            if val1 < best_val:
                best_val = val1
                best_t = t1
        else:
            lo = t1
            if val2 < best_val:
                best_val = val2
                best_t = t2

    # If the search never improved (unlikely) fall back to the midpoint.
    return best_t if best_val < float("inf") else (lo + hi) / 2.0


def _gradient_step(seq: np.ndarray) -> np.ndarray:
    """
    One projected sub‑gradient step that (approximately) reduces the current
    maximal convolution while keeping the L1‑norm unchanged.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    # Gradient of max_b = sum_i a_i a_{k‑i} w.r.t. a_i is 2·a_{k‑i}
    grad = np.zeros_like(seq)
    idx = np.arange(n)
    j = k - idx
    mask = (j >= 0) & (j < n)
    grad[mask] = 2.0 * seq[j[mask]]

    # Make the step sum‑preserving: d = -grad + (mean(grad))·1
    avg = grad.mean()
    direction = -grad + avg

    # Largest step size that keeps all components non‑negative
    neg = direction < 0
    if np.any(neg):
        max_eta = np.min(-seq[neg] / direction[neg])
        eta = 0.5 * max_eta                # stay safely inside the feasible region
    else:
        eta = 0.01                          # arbitrary small step when direction points outward

    new_seq = seq + eta * direction
    # Clip to the allowed range (the evaluate function will also clip)
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    return new_seq


def _random_perturb(seq: np.ndarray, scale: float = 0.05) -> np.ndarray:
    """
    Apply a small Gaussian perturbation (relative to the mean value) and clip.
    """
    if seq.mean() == 0:
        sigma = scale
    else:
        sigma = scale * seq.mean()
    pert = seq + np.random.randn(len(seq)) * sigma
    pert = np.clip(pert, 0.0, 1000.0)
    return pert


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Starting point -------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # Use the previously discovered construction if it is present.
        init_seq = np.array(list(height_sequence_1), dtype=np.float64)
        print(f"Starting from provided height_sequence_1 (length {len(init_seq)})")
    except NameError:  # pragma: no cover – fallback when the variable is absent
        length = np.random.randint(200, 1000)
        init_seq = np.random.rand(length).astype(np.float64) * 10.0
        print(f"No global sequence found – random init of length {length}")

    # Ensure a reasonable sum (the evaluation routine requires sum > 0.01)
    if init_seq.sum() < 0.1:
        init_seq += 1.0

    best_seq = init_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    current_seq = best_seq.copy()

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop ----------------------------------------
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0   # spare a couple of seconds for cleanup
    it = 0
    while time.time() < deadline:
        it += 1
        n = len(current_seq)

        # Normalise the current candidate (the score is scale‑invariant)
        f_norm = _normalize(current_seq)
        M = _max_convolution(f_norm)

        # --------------------------------------------------------------
        # 2a.  LP direction (performed roughly every 3‑4 iterations)
        # --------------------------------------------------------------
        if np.random.rand() < 0.30:   # ~30 % of the iterations try the LP step
            g_raw = _solve_convolution_lp(f_norm, M)
            if g_raw is not None and g_raw.sum() > 0:
                g_norm = _normalize(g_raw)

                # Try the pure LP candidate first
                cand_score = evaluate_sequence(g_norm.tolist())
                if cand_score < best_score:
                    best_seq = g_norm.copy()
                    best_score = cand_score
                    current_seq = g_norm.copy()
                    print(f"[{it}] LP‑only improvement: score={best_score:.6f}")
                    continue  # go to next iteration

                # ------------------------------------------------------
                # 2b.  Line‑search mixing with the LP direction
                # ------------------------------------------------------
                t_opt = _line_search_opt_t(f_norm, g_norm, deadline, max_iter=15)
                mixed = (1.0 - t_opt) * f_norm + t_opt * g_norm
                mixed_score = evaluate_sequence(mixed.tolist())

                if mixed_score < best_score:
                    best_seq = mixed.copy()
                    best_score = mixed_score
                    current_seq = mixed.copy()
                    print(f"[{it}] LP‑mix (t={t_opt:.3f}) improvement: score={best_score:.6f}")
                    continue
                else:
                    # keep the mixed candidate as a new starting point even if
                    # it is not a new global best – it may help escape plateaus.
                    current_seq = mixed.copy()
                    continue

        # --------------------------------------------------------------
        # 2c.  Random local search (fallback when LP is not used)
        # --------------------------------------------------------------
        if np.random.rand() < 0.5:
            # Small Gaussian noise
            cand = _random_perturb(current_seq, scale=0.03)
        else:
            # Gradient‑descent step aimed at reducing the current maximum convolution
            cand = _gradient_step(current_seq)

        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            current_seq = cand.copy()
            print(f"[{it}] Local move improvement: score={best_score:.6f}")
        else:
            # Occasionally accept a worse move to keep the search diverse
            if np.random.rand() < 0.02:
                current_seq = cand.copy()

        # --------------------------------------------------------------
        # 2d.  Periodic restart from the best known sequence (adds diversification)
        # --------------------------------------------------------------
        if it % 500 == 0:
            restart = best_seq * (1.0 + np.random.randn(n) * 0.01)
            restart = np.clip(restart, 0.0, 1000.0)
            current_seq = restart
            print(f"[{it}] Restarted from best sequence (score={best_score:.6f})")

    # ------------------------------------------------------------------
    # 3.  Return the best sequence found --------------------------------
    # ------------------------------------------------------------------
    print(f"Finished after {it} iterations – best score: {best_score:.6f}")
    return best_seq.tolist()
