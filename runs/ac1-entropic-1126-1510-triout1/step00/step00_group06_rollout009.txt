    simulated‑annealing / hill‑climbing loop that mixes in directions obtained

    The loop runs until the time budget expires and returns the best

    """

    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    # Helper to update best
    def update(seq, score):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best score: {best_score:.6f} (len={len(seq)})")

    # 1. Gather initial candidates
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 2000, 3000]
    for n in lengths:

        score = evaluate_sequence(seq.tolist())
        update(seq, score)

        i = np.arange(n)
        tri = np.minimum(i, n-1-i) + 1

        seq = seq / seq.sum()

        update(seq, score)

            update(seq, score)

"""
Propose a step‑function sequence that minimises `evaluate_sequence`.

The algorithm works in three phases:

1. **Diverse initial candidates** – uniform, triangular, sine‑squared,
   Gaussian, geometric, and symmetric exponential shapes are generated for
   several lengths.  Every candidate is evaluated and the best is kept.

2. **Local optimisation loop** – a simulated‑annealing / hill‑climbing
   routine runs until the time budget expires.  At each iteration one of
   the following moves is performed:
        * small random perturbation of a single coordinate,
        * smoothing with a 3‑point moving‑average,
        * replacement by a freshly drawn shape (Gaussian / exponential /
          geometric),
        * occasional LP‑guided direction (the “good direction” from the
          state‑of‑the‑art) with a line‑search on the mixing factor.
   Accept‑if‑better moves are always kept; worsening moves may be accepted
   according to a decreasing temperature schedule.

3. **Return** the best sequence found (as a plain Python list of floats).

All numbers are non‑negative, normalised to sum to 1 (scaling does not affect
the objective because the evaluation function is homogeneous).  The LP
sub‑problem uses `scipy.optimize.linprog` with a sparse constraint matrix
to keep memory usage modest.

The routine respects the 1000 s budget; if it finishes early it returns the
best sequence discovered so far.
"""

import time
import numpy as np
from scipy import optimize, sparse

# ----------------------------------------------------------------------
# Linear‑programming sub‑problem: maximise Σ b_i  subject to
#   convolution(f, b) ≤ rhs  and  b ≥ 0.
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """Return a feasible vector `b` (numpy array) or None.
    The LP is:
        maximise  Σ b_i
        subject to  (f * b)_k ≤ rhs   for all k,
                    b_i ≥ 0.
    """
    n = f_seq.size
    # Build convolution constraints (sparse)
    row_inds = []
    col_inds = []
    data = []

    # For each output index k of the convolution (0 … 2n‑2)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals                # j = k - i
        row_inds.extend([k] * i_vals.size)
        col_inds.extend(j_vals.tolist())
        data.extend(f_seq[i_vals].tolist())

    # Non‑negativity constraints: -I * b ≤ 0
    row_start = 2 * n - 1
    row_inds.extend(range(row_start, row_start + n))
    col_inds.extend(range(n))
    data.extend([-1.0] * n)

    A_ub = sparse.coo_matrix((data, (row_inds, col_inds)), shape=(3 * n - 1, n)).tocsr()
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

    c = -np.ones(n)               # maximise sum ⇒ minimise −sum
    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": 5.0},
        )
    except Exception:
        return None

    if res.success:
        # Numerical noise may produce tiny negatives; clip to zero.
        return np.maximum(0.0, res.x)
    return None

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # Helper to update the current best
    def update_best(seq: np.ndarray, score: float):
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New best: {best_score:.6f} (len={len(seq)})")

    # ------------------------------------------------------------------
    # 1.  Generate a variety of initial shapes
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 2000, 3000]

    for n in lengths:
        # uniform
        seq = np.ones(n) / n
        update_best(seq, evaluate_sequence(seq.tolist()))

        # symmetric triangular (peak in the middle)
        i = np.arange(n)
        tri = np.minimum(i, n - 1 - i) + 1
        seq = tri.astype(float)
        seq /= seq.sum()
        update_best(seq, evaluate_sequence(seq.tolist()))

        # sine‑squared
        seq = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        seq /= seq.sum()
        update_best(seq, evaluate_sequence(seq.tolist()))

        # Gaussian families
        for sigma_factor in [0.06, 0.09, 0.12, 0.18]:
            sigma = sigma_factor * n
            x = np.arange(n) - (n - 1) / 2.0
            seq = np.exp(-(x ** 2) / (2.0 * sigma ** 2))
            seq /= seq.sum()
            update_best(seq, evaluate_sequence(seq.tolist()))

        # Geometric decreasing (one‑sided)
        for alpha in [0.0005, 0.001, 0.002, 0.004]:
            r = np.exp(-alpha)
            seq = r ** np.arange(n)
            seq /= seq.sum()
            update_best(seq, evaluate_sequence(seq.tolist()))

        # Symmetric exponential about the centre
        for alpha in [0.0005, 0.001, 0.002, 0.004]:
            x = np.abs(np.arange(n) - (n - 1) / 2.0)
            seq = np.exp(-alpha * x)
            seq /= seq.sum()
            update_best(seq, evaluate_sequence(seq.tolist()))

        # "theoretically balanced" geometric shape (derived from r^{n-1}=1/n)
        if n > 2:
            alpha_opt = np.log(n) / (n - 1)
            r_opt = np.exp(-alpha_opt)
            seq = r_opt ** np.arange(n)
            seq /= seq.sum()
            update_best(seq, evaluate_sequence(seq.tolist()))

    # If the global `height_sequence_1` exists, try it as a start point.
    try:
        if "height_sequence_1" in globals():
            seq = np.array(height_sequence_1, dtype=float)
            seq = np.maximum(0.0, seq)          # safety
            if seq.sum() > 0:
                seq = seq / seq.sum()
                update_best(seq, evaluate_sequence(seq.tolist()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Main iterative improvement (simulated annealing / hill climbing)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback: random initialisation
        n = 800
        best_seq = np.random.rand(n)
        best_seq /= best_seq.sum()
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback random start: score {best_score:.6f}")

    curr_seq = best_seq.copy()
    curr_score = best_score

    # temperature for simulated annealing
    T0 = 0.02
    decay = 0.9997
    iteration = 0
    lp_every = 40        # attempt LP move only every `lp_every` iterations

    while time.time() < deadline:
        iteration += 1
        n = len(curr_seq)
        move_type = np.random.rand()

        new_seq = None

        # --------------------------------------------------------------
        # Small random perturbation of a single coordinate
        # --------------------------------------------------------------
        if move_type < 0.25:
            idx = np.random.randint(n)
            delta = np.random.randn() * 0.02 * max(curr_seq[idx], 1e-3)
            new_seq = curr_seq.copy()
            new_seq[idx] = max(0.0, new_seq[idx] + delta)
            s = new_seq.sum()
            if s > 0:
                new_seq /= s

        # --------------------------------------------------------------
        # Smoothing with a 3‑point kernel (preserves non‑negativity)
        # --------------------------------------------------------------
        elif move_type < 0.5:
            kernel = np.array([0.25, 0.5, 0.25])
            new_seq = np.convolve(curr_seq, kernel, mode="same")
            s = new_seq.sum()
            if s > 0:
                new_seq /= s

        # --------------------------------------------------------------
        # Randomly draw a fresh analytic shape
        # --------------------------------------------------------------
        elif move_type < 0.75:
            shape = np.random.choice(["gauss", "exp", "geom"])
            if shape == "gauss":
                sigma_factor = np.random.uniform(0.05, 0.22)
                sigma = sigma_factor * n
                x = np.arange(n) - (n - 1) / 2.0
                cand = np.exp(-(x ** 2) / (2.0 * sigma ** 2))
            elif shape == "exp":
                alpha = np.random.uniform(0.0003, 0.006)
                x = np.abs(np.arange(n) - (n - 1) / 2.0)
                cand = np.exp(-alpha * x)
            else:   # geometric
                alpha = np.random.uniform(0.0003, 0.006)
                r = np.exp(-alpha)
                cand = r ** np.arange(n)
            cand /= cand.sum()
            new_seq = cand

        # --------------------------------------------------------------
        # LP‑guided direction (performed sparsely)
        # --------------------------------------------------------------
        else:
            # Only attempt LP every few iterations to keep cost low.
            if iteration % lp_every == 0 and n <= 1800:
                # Normalise as required by the LP routine.
                sum_seq = curr_seq.sum()
                norm_factor = np.sqrt(2.0 * n) / sum_seq
                f_norm = curr_seq * norm_factor
                rhs = np.max(np.convolve(f_norm, f_norm))

                g = solve_convolution_lp(f_norm, rhs)
                if g is not None:
                    # Renormalise g to the same L1‐scale used in the algorithm.
                    sum_g = g.sum()
                    if sum_g > 0:
                        g_norm = g * np.sqrt(2.0 * n) / sum_g
                        # Simple line‑search on the mixing factor t.
                        t = 0.08
                        while t > 1e-5:
                            cand = (1 - t) * curr_seq + t * g_norm
                            cand = np.maximum(0.0, cand)
                            cand /= cand.sum()
                            cand_score = evaluate_sequence(cand.tolist())
                            if cand_score < curr_score:
                                new_seq = cand
                                curr_score = cand_score
                                break
                            t *= 0.5
        # --------------------------------------------------------------
        # If nothing was generated (unlikely), skip this iteration.
        # --------------------------------------------------------------
        if new_seq is None:
            continue

        # --------------------------------------------------------------
        # Evaluate the candidate and decide whether to accept it.
        # --------------------------------------------------------------
        try:
            cand_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue   # safety

        # Acceptance rule
        if cand_score < curr_score:
            curr_seq = new_seq
            curr_score = cand_score
            if cand_score < best_score:
                best_seq = new_seq.copy()
                best_score = cand_score
                print(f"[{iteration}] New global best: {best_score:.6f}")
        else:
            # simulated‑annealing acceptance of worse moves
            T = T0 * (decay ** iteration)
            prob = np.exp(-(cand_score - curr_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                curr_seq = new_seq
                curr_score = cand_score

    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    return best_seq.tolist()
