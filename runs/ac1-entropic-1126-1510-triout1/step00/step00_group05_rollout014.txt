"""
Propose a candidate sequence for the evaluation function.

The idea is to use a *geometric* (exponentially decaying) sequence.
If a_i ∝ r^i with 0<r<1, the self‑convolution (a⋆a)_k is much flatter
than the triangular shape obtained from a uniform sequence.
For a fixed length `n`, the maximum convolution value occurs near the
center (k = n‑1) and is approximately

    max(b) ≈ n·r^{n‑1} / (∑_{j=0}^{n‑1} r^j)^2 .

Choosing the decay rate by the parameter `c` with

    r = exp(-c/(n‑1))

gives `r^{n‑1} = exp(-c)`.  The evaluation value becomes

    2·n·max(b) ≈ 2·c²·exp(-c) / (1‑exp(-c))² .

This expression decreases quickly as `c` grows (for moderate `c`
the diagonal term a_i² is negligible), so we can obtain values far
below the current bound 2.0.  We simply scan a few values of `c`,
build the corresponding normalized geometric sequence, evaluate it,
and keep the best one.  The whole search finishes well within the
time budget.

The returned sequence contains only non‑negative floats, respects the
1000‑value cap (all entries are ≤1), and is ready for `evaluate_sequence`.
"""

import math
import time
import numpy as np


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Return a sequence that minimises `evaluate_sequence`.

    The function searches over a family of exponential sequences
    a_i ∝ exp(-c·i/(n‑1)) for several values of the decay constant `c`.
    The best sequence (lowest evaluation score) is returned.
    """
    # Length of the construction – we use the same length as the reference.
    n = 1607

    deadline = time.time() + budget_s - 0.1  # leave a tiny margin

    best_score = float("inf")
    best_seq = None

    # Candidate decay parameters.  The range 1.0–12.0 is ample to drive the
    # score well below the target 1.503.  Step 0.1 gives a fine grid without
    # costing much time.
    cs = np.linspace(1.0, 12.0, 111)

    for c in cs:
        if time.time() > deadline:
            break

        # r is the per‑step multiplier of the geometric progression.
        r = math.exp(-c / (n - 1))

        # Build the raw (unnormalised) geometric sequence.
        # Using float64 prevents premature under‑flow for the largest `c`.
        raw = np.power(r, np.arange(n, dtype=np.float64))

        # Normalise so that the sum equals 1 (any positive scale is
        # irrelevant because the evaluation function is homogeneous).
        total = raw.sum()
        seq = (raw / total).tolist()

        # Evaluate; any exception (e.g. overflow) is ignored – the candidate
        # is simply discarded.
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"New best – c={c:.3f}, score={score:.6f}")

    # If, for any unexpected reason, no candidate succeeded, fall back to a
    # uniform sequence (which yields a score of exactly 2.0).
    if best_seq is None:
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence:", best_score)

    print("Final best score:", best_score)
    return best_seq
