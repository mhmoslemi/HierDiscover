def project_simplex(v):
    """Projects a vector onto the probability simplex (nonnegative, sum=1)."""
    n = len(v)
    # shift to ensure sum <= 1 if negative etc.
    # Algorithm: sort v decreasing, find rho
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all non-positive, return uniform? Actually projection onto simplex of all negative vector returns uniform? Usually result is zero vector with sum=0? But we need sum=1, not possible. So fallback to uniform.
        w = np.full(n, 1.0/n)
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Ensure sum is exactly 1 (might have rounding small diff)
    w_sum = np.sum(w)
    if w_sum == 0:
        w = np.full(n, 1.0/n)
    else:
        w = w / w_sum
    return w

def project_onto_simplex(v):
    # v as numpy array
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # Find rho
    rho_arr = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho_arr.size == 0:
        # return uniform
        return np.full(n, 1.0/n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    # Normalize to sum exactly 1 (due to numeric)
    w_sum = w.sum()
    if w_sum == 0:
        return np.full(n, 1.0/n)
    w /= w_sum
    return w

"""
Propose a low‑value step‑function.

The objective is 2·n·max(conv(a,a))/sum(a)^2.
The function is scale‑invariant, so we keep the coefficients on the unit‑simplex
( non‑negative, sum = 1 ).  The algorithm works in three stages:

1. **Parameterized initialisation**
   – Generate families of symmetric “U‑shaped’’ sequences
     a_i ∝ ((i+δ)*(n‑1‑i+δ))^p  with p∈[‑0.9,‑0.2] and a small shift δ>0.
   – Try several lengths (500 … 3000) and keep the best candidate
     according to the provided ``evaluate_sequence`` function.

2. **Projected gradient descent**
   – Re‑normalise the current sequence to sum 1.
   – Compute the linear convolution using an FFT (O(n log n)).
   – Identify the index k* where the convolution attains its maximum.
   – The gradient of the maximal term w.r.t. a is
          ∂b_{k*}/∂a_i = 2·a_{k*‑i}   (zero when the index is out of range).
   – Take a step opposite to the gradient, then project back onto the
     probability simplex (non‑negative, sum = 1).  The step size decays
     slowly; occasional random perturbations help escaping shallow minima.

3. **Time‑bounded search**
   – The whole routine runs until the supplied budget (default 1000 s) is
     exhausted, constantly updating the best sequence found.
   – The returned list contains plain Python floats and satisfies the
     security checks of ``evaluate_sequence``.

The method is simple, requires only NumPy (no external LP/SDP solvers) and
behaves well within the time limit.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the probability simplex (non‑negative, sum=1)
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        # All entries are non‑positive – return the uniform distribution
        return np.full(n, 1.0 / n, dtype=float)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Normalise numerically (should already sum to 1)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n, dtype=float)
    return w / w_sum


# ----------------------------------------------------------------------
# Helper: linear convolution using FFT (real inputs, non‑circular)
def linear_convolution(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    L = int(2 ** math.ceil(math.log2(2 * n - 1)))  # next power of two
    A = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values; clip to zero
    conv_full[conv_full < 0] = 0.0
    return conv_full


# ----------------------------------------------------------------------
# Generate a symmetric “U‑shaped’’ family based on a power law
def generate_u_shape(n: int, p: float, delta: float = 0.5) -> np.ndarray:
    """
    Produce a symmetric non‑negative sequence of length n:
        a_i ∝ ((i+δ)*(n‑1‑i+δ))**p   with p<0 (U‑shape) or p>0 (hump).
    The result is normalised to sum 1.
    """
    i = np.arange(n, dtype=float)
    v = (i + delta) * (n - 1 - i + delta)
    # Prevent zero for p<0
    eps = 1e-12
    v = np.maximum(v, eps)
    a = np.power(v, p)
    total = a.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=float)
    return a / total


# ----------------------------------------------------------------------
# Core optimisation: projected gradient descent from a starting vector
def improve_by_gradient(a0: np.ndarray,
                        time_deadline: float,
                        max_iters: int = 10_000,
                        init_step: float = 0.15) -> (np.ndarray, float):
    """
    Reduce max(conv(a,a)) while keeping a on the simplex.
    Returns the best sequence seen and its (internal) objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # compute initial objective
    conv = linear_convolution(a)
    best_M = conv.max()
    best_f = 2.0 * n * best_M / (a.sum() ** 2)   # sum is 1 after projection
    it = 0
    step = init_step

    while time.time() < time_deadline and it < max_iters:
        # convolution and location of the current maximum
        conv = linear_convolution(a)
        k_max = int(conv.argmax())
        # gradient of the maximal term (2 * a reversed around k_max)
        grad = np.zeros_like(a)
        # indices i that contribute: 0 ≤ k_max‑i < n  ⇔ i ∈ [k_max‑(n‑1), k_max]
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k_max - idx]

        # simple step with decaying size
        step_size = step / (1.0 + 0.0005 * it)
        a_candidate = a - step_size * grad
        a_candidate = project_onto_simplex(a_candidate)

        # evaluate
        conv_cand = linear_convolution(a_candidate)
        M_cand = conv_cand.max()
        f_cand = 2.0 * n * M_cand / (a_candidate.sum() ** 2)

        # accept if improvement; otherwise keep a (occasionally allow random jump)
        if f_cand < best_f:
            best_f = f_cand
            best_M = M_cand
            best_a = a_candidate.copy()
            a = a_candidate
            # occasional aggressive shrink of step after a good move
            step = max(step * 0.98, 1e-4)
        else:
            # rarely accept worse move to avoid stagnation
            if random.random() < 0.001:
                a = a_candidate
            # otherwise keep current a unchanged
            # slowly increase step size to encourage movement
            step = min(step * 1.003, 0.5)

        # small random perturbation every few hundred iterations
        if it % 500 == 0 and random.random() < 0.2:
            pert = np.random.randn(n) * 0.005
            a = project_onto_simplex(a + pert)

        it += 1

    return best_a, best_f

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a small parametric sweep followed by a projected gradient descent.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    # leave a safety margin for final return
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. PARAMETRIC INITIALISATION
    lengths = [600, 900, 1200, 1500, 1800, 2100, 2400]  # reasonable sizes
    exponents = [-0.9, -0.7, -0.5, -0.4, -0.3, -0.2]   # U‑shape strength
    delta = 0.5

    print("=== Parametric initialisation ===")
    for n in lengths:
        for p in exponents:
            seq = generate_u_shape(n, p, delta)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f"n={n:4d}, p={p:5.2f} → score={score:.6f}")
            # early break if time is almost up
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # also compare with the plain uniform sequence (baseline = 2.0)
    for n in lengths:
        seq = np.full(n, 1.0 / n, dtype=float)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"Uniform n={n:4d} → score={score:.6f}")
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2. REFINEMENT BY PROJECTED GRADIENT DESCENT
    if best_seq is None:
        # fallback: a small uniform vector
        best_seq = [1.0]

    a = np.array(best_seq, dtype=float)
    # normalise to sum = 1 (the objective is scale‑invariant)
    a = a / a.sum()
    # run the descent while time permits
    print("\n=== Gradient descent refinement ===")
    a_opt, f_opt = improve_by_gradient(a, deadline)
    if f_opt < best_score:
        best_score = f_opt
        best_seq = a_opt.tolist()
        print(f"After descent: score={f_opt:.6f}")

    # ------------------------------------------------------------------
    # 3. OPTIONAL random‑restart improvements (lightweight)
    # Try a few quick random restarts if there is still time left.
    while time.time() < deadline:
        # random length in same range
        n = random.choice(lengths)
        # random Dirichlet‑like start (adds diversity)
        raw = np.random.rand(n)
        raw = raw / raw.sum()
        a_candidate, f_candidate = improve_by_gradient(raw, deadline, max_iters=2000)
        if f_candidate < best_score:
            best_score = f_candidate
            best_seq = a_candidate.tolist()
            print(f"Random restart n={n} → improved score={f_candidate:.6f}")
        # exit condition if we're very close to theoretical lower bound
        if best_score <= 1.51:   # a little margin below the target
            break

    # ------------------------------------------------------------------
    # Final safety clipping (the evaluation code will also clip)
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    # final report
    elapsed = time.time() - start_time
    print(f"\nSearch finished in {elapsed:.2f}s – best score = {best_score:.6f}")
    return best_seq
