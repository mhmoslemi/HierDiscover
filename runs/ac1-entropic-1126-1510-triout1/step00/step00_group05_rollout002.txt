import numpy as np, time, math
from scipy.optimize import linprog

def convolve_max(seq):
    n = len(seq)
    size = 1 << (2*n-1).bit_length()
    f = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f*f, size)[:2*n-1]
    return np.max(conv)

def solve_lp_cross(f_seq, M):
    n = len(f_seq)
    # objective - sum(b) to maximize sum(b)
    c = -np.ones(n)
    # Build convolution constraints
    # For k in 0..2n-2, row_j = f_seq[i] where i+j = k -> j = k - i
    # Compute using Toeplitz matrix: each row is reversed f_seq with shift.
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # For each i, j = k - i
        # row[j] = f_i
        # So fill row[k - i] = f_i
        # Could vectorize: col indices = k - np.arange(i_min,i_max+1), values = f_i
        indices = k - np.arange(i_min, i_max+1)
        row[indices] = f_seq[i_min:i_max+1]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([np.full(2*n - 1, M), np.zeros(n)])
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5 # buffer
    n = 600  # moderate length
    # initial sequence: uniform + small random perturbation
    sum_target = math.sqrt(2 * n)
    init = np.random.rand(n)
    init = init / np.sum(init) * sum_target
    best_seq = init.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    f = best_seq.copy()
    t = 0.15
    while time.time() < deadline:
        # compute max conv
        M = convolve_max(f)
        # solve LP
        g = solve_lp_cross(f, M)
        if g is None:
            # random perturbation
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01)
            continue
        # normalize g to sum_target
        sum_g = np.sum(g)
        if sum_g <= 1e-12:
            continue
        g = g * (sum_target / sum_g)
        # evaluate max of g
        M_g = convolve_max(g)
        # if g better, replace
        if M_g < M:
            f = g
            cur_score = evaluate_sequence(f.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = f.copy()
            # maybe increase t after success?
            continue
        # mixing
        f_new = (1 - t) * f + t * g
        # renormalize to sum_target (since mixing preserves sum if both sum to sum_target)
        # but due to rounding may drift: renormalize anyway
        f_new = f_new * (sum_target / np.sum(f_new))
        cur_score = evaluate_sequence(f_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f_new.copy()
            f = f_new
            # optionally increase t a bit
            t = min(0.3, t * 1.05)
        else:
            # maybe decrease t
            t = max(0.02, t * 0.9)
            # occasionally random shuffle to escape
            if np.random.rand() < 0.1:
                # random perturb many entries
                perturb = np.random.randn(n) * 0.01
                f = np.maximum(0, f + perturb)
                f = f * (sum_target / np.sum(f))
    return best_seq.tolist()

"""
Propose a better step‐function (non‑negative sequence) by iteratively
reducing the maximum value of its self‑convolution while keeping the
ℓ₁‑norm fixed (the evaluation function equals that maximum after the
normalisation sum = sqrt(2·n)).  The algorithm starts from the provided
height_sequence_1 (or a random uniform start), solves a linear programme
to find a direction that can increase the sum without increasing the
convolution bound, mixes the current sequence toward that direction,
and finally performs small random refinements.  It uses FFT for fast
convolution and SciPy’s high‑speed simplex/HiGHS LP solver.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _conv_max(seq: np.ndarray) -> float:
    """Fast linear convolution via FFT and return its maximum entry."""
    n = seq.size
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    fft = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft * fft, size)[: 2 * n - 1]
    return float(conv.max())


def _solve_lp_cross(f_seq: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   sum(b)
        subject to conv(f_seq, b) <= bound  (for all k)
                    b >= 0
    Returns the optimal b (as a NumPy array) or None on failure.
    """
    n = f_seq.size
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # build the convolution constraints (2n‑1 rows)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        # i + j = k  →  j = k – i
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill the appropriate positions
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    A_ub = np.vstack([rows, -np.eye(n, dtype=float)])
    b_ub = np.hstack([np.full(2 * n - 1, bound, dtype=float), np.zeros(n, dtype=float)])

    # HiGHS simplex (fast for dense problems). 5‑second time limit per call.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs", options={"time_limit": 5.0})
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a sequence with a low evaluation score (the lower the better)."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # leave a little safety margin

    # ------------------------------------------------------------------
    # Initialise a feasible sequence.
    # Prefer the known construction if it exists; otherwise start uniform.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, (list, np.ndarray)):
        f = np.asarray(height_sequence_1, dtype=float)
        n = f.size
        target_sum = math.sqrt(2.0 * n)
        if target_sum > 0:
            f = f * (target_sum / f.sum())
    else:
        n = 600                     # moderate size – fast LP but enough degrees of freedom
        target_sum = math.sqrt(2.0 * n)
        f = np.random.rand(n)
        f = f * (target_sum / f.sum())

    # make sure we satisfy the evaluator's clipping rules (they are harmless here)
    f = np.clip(f, 0.0, 1000.0)

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial_score={best_score:.6f}")

    # mixing coefficient (how far we step towards the LP direction)
    t = 0.15
    it = 0

    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) compute current convolution bound M = max(f*f)
        # --------------------------------------------------------------
        M = _conv_max(f)

        # --------------------------------------------------------------
        # 2) solve the LP to obtain a direction g that can increase sum
        #    without breaking the bound.
        # --------------------------------------------------------------
        g_raw = _solve_lp_cross(f, M)

        if g_raw is None:
            # LP failed – perform a tiny random perturbation and continue.
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01)
            f = f * (target_sum / f.sum())
            continue

        # --------------------------------------------------------------
        # 3) normalise the LP solution to the required ℓ₁‑norm.
        # --------------------------------------------------------------
        sum_g = g_raw.sum()
        if sum_g <= 1e-12:
            continue
        g = g_raw * (target_sum / sum_g)

        # --------------------------------------------------------------
        # 4) evaluate the new candidate.
        #    If g itself improves the bound we replace f immediately.
        # --------------------------------------------------------------
        M_g = _conv_max(g)
        if M_g < M:
            f = g
            cur_score = evaluate_sequence(f.tolist())
            if cur_score < best_score:

            # keep the same mixing step size – we already moved to a better point
            continue

        # 5) otherwise mix a fraction t towards g and keep the sum fixed.
        # --------------------------------------------------------------
        f_new = (1.0 - t) * f + t * g
        # re‑normalise (tiny drift may appear due to rounding)
        f_new = f_new * (target_sum / f_new.sum())

        cur_score = evaluate_sequence(f_new.tolist())
        if cur_score < best_score:

            f = f_new
            # be a bit more aggressive after a successful step
            t = min(0.35, t * 1.07)
            print(f"[{it}] Mix improvement: score={best_score:.6f}, t={t:.3f}")
            # early exit if we have already crossed the target
            if best_score < 1.55:

                pass
        else:
            # no improvement – step back a little
            t = max(0.02, t * 0.85)
            # occasional random shake‑up to escape plateaus
            if np.random.rand() < 0.08:
                perturb = np.random.randn(n) * 0.02 * target_sum
                f = np.maximum(0.0, f + perturb)
                f = f * (target_sum / f.sum())

        # --------------------------------------------------------------
        # 6) occasional tiny hill‑climbing refinement (cheaper than LP).
        # --------------------------------------------------------------
        if it % 20 == 0:
            # try a few random local tweaks
            for _ in range(5):
                idx = np.random.randint(n)
                delta = np.random.randn() * 0.03 * target_sum
                cand = f.copy()
                cand[idx] = max(0.0, cand[idx] + delta)

                s = evaluate_sequence(cand.tolist())
                if s < best_score:
                    best_score, best_seq = s, cand.copy()
                    f = cand
                    print(f"[{it}] Local tweak: score={best_score:.6f}")

    print(f"[final] best_score={best_score:.6f} after {it} iterations")
    return best_seq.tolist()
