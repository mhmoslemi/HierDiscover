    # grad is gradient (positive for larger contributions)
    # Update: x_i_new = x_i * exp(-step * grad_i)

        return np.full_like(x, 1.0/len(x))
    return new / sum_new

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (probability vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)^2)

where `conv` is the full linear convolution.  The objective is invariant to
scaling, therefore we work with a probability vector (sum = 1).  The algorithm
runs many short random restarts; each restart performs a mirror‑descent style
update that directly reduces the largest entry of the self‑convolution.
The update is multiplicative (x_i ← x_i·exp(−η·∇_i)) which keeps the vector
on the simplex after a simple renormalisation.  Gradient information is taken
from all convolution entries whose value is within 0.1 % of the current
maximum – this gives a sub‑gradient of the ``max``‑objective.

The whole search respects the supplied time budget (default 1000 s) and
continually records the best sequence seen.  The returned list can be fed to
`evaluate_sequence`; it already satisfies all security checks (non‑negative,
finite, sum > 0.01).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Small helper returning the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Compute the full linear convolution of a real vector with itself using FFT.
    Returns an array of length 2*len(x)-1 (the non‑negative part only).
    """
    # Real FFT is faster than complex FFT for real data.
    X = np.fft.rfft(x, n=fft_size)
    conv = np.fft.irfft(X * X, n=fft_size)
    # Numerical noise can produce tiny negative numbers – clip to zero.
    conv = np.maximum(conv, 0.0)
    return conv[: 2 * len(x) - 1]


def _max_convolution_score(x: np.ndarray, fft_size: int) -> float:
    """Return the value of the objective for a probability vector x."""
    conv = _full_convolution(x, fft_size)
    return 2.0 * len(x) * np.max(conv)


def _gradient_from_peak(x: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the ``max(conv)`` function.
    All convolution entries whose value is ≥ 0.999·max are considered
    (this gives a useful direction even when the maximum is attained at several
    positions).
    """
    n = len(x)
    max_val = np.max(conv)
    # Select the “active’’ convolution indices.
    active = np.where(conv >= max_val * 0.999)[0]
    grad = np.zeros_like(x)

    # For each active offset k we add the contribution 2·x_{k-i} to
    # ∂c_k/∂x_i where the pair (i, k‑i) is valid.
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1)
        j_range = k - i_range
        grad[i_range] += 2.0 * x[j_range]

    return grad


def _mirror_step(x: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative (mirror‑descent) update on the simplex:
        x_new_i ∝ x_i·exp(−η·grad_i)
    followed by normalisation to sum to one.
    """
    # All entries stay non‑negative because the exponential is positive.
    new = x * np.exp(-eta * grad)
    s = new.sum()
    if s <= 0.0:                     # numerical safeguard – fall back to uniform.
        return np.full_like(x, 1.0 / len(x))
    return new / s

# Main search entry point
# ----------------------------------------------------------------------

            # -----------------------------------------------------------------
            conv = _full_convolution(x, fft_size)
            cur_score = 2.0 * n * np.max(conv)

            if cur_score < best_score:
                best_score = cur_score
                best_x = x.copy()
                no_improve_iters = 0
                print(
                    f"[{time.time():.1f}] New global best = {best_score:.6f} "
                    f"(restart {restart_counter}, it {it})"
                )
            else:
                no_improve_iters += 1

            # -----------------------------------------------------------------
            # Compute a sub‑gradient from the peak(s) of the convolution.
            # -----------------------------------------------------------------
            grad = _gradient_from_peak(x, conv)

            # -----------------------------------------------------------------
            # Propose a multiplicative update and decide whether to keep it.
            # -----------------------------------------------------------------
            new_x = _mirror_step(x, grad, eta)

            # Fast evaluation of the candidate (reuse FFT plan).
            new_conv = _full_convolution(new_x, fft_size)
            new_score = 2.0 * n * np.max(new_conv)

            if new_score <= cur_score:
                # Accept the move – optionally increase the step size a bit.
                x = new_x
                eta = min(eta * 1.05, 5.0)
                no_improve_iters = 0
            else:
                # Reject – shrink the step size.
                eta = max(eta * 0.6, 1e-4)

            # Occasionally inject a tiny random jitter to keep the dynamics lively.
            if no_improve_iters >= 200:
                jitter = 0.002 * np.random.randn(n)
                x = np.clip(x + jitter, 0.0, None)
                if x.sum() == 0.0:
                    x = np.full(n, 1.0 / n)
                else:
                    x /= x.sum()
                eta = max(eta, 0.001)
                no_improve_iters = 0

        # End of inner loop – start the next random restart.
        # Print a brief status line.
        print(
            f"Restart {restart_counter} finished. Current best = {best_score:.6f}"
        )

    # --------------------------------------------------------------
    # Final conversion – the evaluator works with any scaling, keeping the
    # vector as‑is (sum = 1) is safe.
    # --------------------------------------------------------------
    final_seq = best_x.tolist()
    print(f"Search finished. Best score = {best_score:.6f}")
    return final_seq
