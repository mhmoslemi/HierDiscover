def improve_sequence(seq, total_iter, lr):
    best_score = evaluate_sequence(seq)
    best_seq = seq.copy()
    n = len(seq)
    target_sum = math.sqrt(2*n)
    seq = np.array(seq, dtype=float)
    # Normalize sum
    seq = seq * (target_sum / np.sum(seq))
    for it in range(total_iter):
        b = np.convolve(seq, seq)
        max_idx = int(np.argmax(b))
        grad = np.zeros_like(seq)
        # compute w_i = seq[max_idx - i] if valid
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                grad[i] = seq[j]
        # direction to reduce max conv: move opposite grad with sum constraint
        direction = -grad
        # enforce zero sum: subtract mean
        direction = direction - np.mean(direction)
        # step size adapt
        step = lr / (np.sqrt(it+1) + 1)  # decreasing
        new_seq = seq + step * direction
        # clip to >=0
        new_seq = np.maximum(new_seq, 0)
        # renorm sum
        sum_new = np.sum(new_seq)
        if sum_new == 0:
            continue
        new_seq = new_seq * (target_sum / sum_new)
        # evaluate
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
        # also accept if improvement else keep old?
        if score <= best_score * 1.000001: # allow slight improvement
            seq = new_seq
        else:
            # if not better, maybe reduce step or random pert
            # keep old seq
            pass
    return best_seq.tolist()

"""
Propose a better step‑function sequence.

The objective to minimise is

    E(a) = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the full discrete convolution.
Uniform heights give `E = 2`.  By using a slightly non‑uniform shape
the peak of the self‑convolution can be reduced.

The search consists of three stages:

1. **Parametric families** – we try two simple families of non‑negative
   vectors whose shape is controlled by a single scalar:
        * geometric decay  a_i ∝ r**i ,   0 < r < 1
        * decreasing power  a_i ∝ (n‑i)**p ,  p ≥ 0
   For each family a short golden‑section search (30 iterations) is run
   and the best candidate is kept.

2. **Local hill‑climbing** – starting from the best candidate obtained
   above we repeatedly move a small amount of mass from one position to
   another (keeping the total sum constant).  Whenever the move improves
   the evaluation we accept it.  This inexpensive stochastic descent can
   discover small refinements that the parametric search cannot reach.

3. **Result** – the best sequence found before the time budget expires is
   returned.  All entries are kept non‑negative; the final scaling does
   not matter because the evaluation is homogeneous, but we keep the sum
   close to `sqrt(2*n)` to stay well away from the clipping limits of the
   verifier.

The whole routine respects the 1000 s budget, prints progress
information and works with any length (the default is the length of the
global reference sequence `height_sequence_1` when it exists).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _normalise_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale `seq` so that its L1‑sum equals ``target_sum``."""
    s = seq.sum()
    if s == 0:
        return seq
    return seq * (target_sum / s)


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the supplied ``evaluate_sequence``."""
    # ``evaluate_sequence`` expects a plain list of floats.
    return evaluate_sequence(seq.tolist())


def _geometric_sequence(r: float, n: int, target_sum: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i, normalised to ``target_sum``."""
    base = np.power(r, np.arange(n, dtype=np.float64))
    return _normalise_to_target(base, target_sum)


def _power_sequence(p: float, n: int, target_sum: float) -> np.ndarray:
    """Decreasing power a_i ∝ (n-i)**p, normalised to ``target_sum``."""
    # use 1‑based decreasing indices (largest weight at the first element)
    idx = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, ..., 1
    base = np.power(idx, p)
    return _normalise_to_target(base, target_sum)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The function runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Determine the length to work with.
    # Prefer the existing reference length; fall back to a moderate size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # noqa: F821  (the variable is supplied)
    except Exception:
        n = 4000  # a safe default when the reference sequence is missing

    # Target sum – any constant works because the score is scale‑invariant.
    target_sum = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1.  Uniform baseline (reference point)
    # ------------------------------------------------------------------
    uniform_seq = np.full(n, target_sum / n, dtype=np.float64)
    best_seq = uniform_seq.copy()
    best_score = _evaluate(best_seq)
    print(f"[init] uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Search geometric decay a_i ∝ r**i
    # ------------------------------------------------------------------
    phi = (math.sqrt(5.0) - 1.0) / 2.0      # golden‑section factor
    lo_r, hi_r = 0.90, 0.999999           # search interval for r

    for _ in range(30):
        r1 = hi_r - (hi_r - lo_r) * phi
        r2 = lo_r + (hi_r - lo_r) * phi

        seq1 = _geometric_sequence(r1, n, target_sum)
        seq2 = _geometric_sequence(r2, n, target_sum)

        score1 = _evaluate(seq1)
        score2 = _evaluate(seq2)

        if score1 < score2:
            hi_r = r2
            if score1 < best_score:
                best_score, best_seq = score1, seq1.copy()
        else:
            lo_r = r1
            if score2 < best_score:
                best_score, best_seq = score2, seq2.copy()

    print(f"[geom] best r ∈ [{lo_r:.6f}, {hi_r:.6f}], score = {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    lo_p, hi_p = 0.0, 5.0                     # exponent range

        else:

        new_seq[j] += delta

        # non‑negativity is already guaranteed by construction
        new_score = _evaluate(new_seq)

        # accept only if we improve
        if new_score < cur_score:
            cur_seq = new_seq
            cur_score = new_score
            if cur_score < best_score:
                best_score, best_seq = cur_score, cur_seq.copy()
                print(f"[hill] improvement: score = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    # Convert to plain list of Python floats for the evaluator
    return best_seq.tolist()
