"""
Gradient‑soft‑max descent for minimizing the convolution‑max objective.

The algorithm treats the objective  max_k (a*a)[k]  (with a ≥ 0,
Σa = 1) as a smooth “log‑sum‑exp” function.  Its gradient can be written
as a correlation between the soft‑max weights of the convolution and the
current vector a.  We perform projected gradient steps, keep the L1‑norm
fixed by projecting the gradient onto the hyper‑plane Σa = 1, and enforce
non‑negativity by clipping.  The step size is adapted on‑the‑fly and a
tiny amount of random shaking is added occasionally to escape shallow
local minima.

The loop runs until the time budget is exhausted (or until a score below
the target 1.503 is found).  The best sequence seen is returned as a
plain Python list, ready to be evaluated by `evaluate_sequence`.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Total time budget in seconds.
    **kwargs : dict
        Ignored, kept for compatibility.

    Returns
    -------
    list[float]
        The best non‑negative sequence found within the allocated time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # 1. Problem size – a few hundred to a thousand entries works well.
    # ------------------------------------------------------------------
    n = 800                       # length of the sequence (adjustable)

    # ------------------------------------------------------------------
    # 2. Initial candidate – Dirichlet (i.e. random positive) and normalised.
    # ------------------------------------------------------------------
    a = np.random.rand(n)
    a /= a.sum()                  # enforce Σ a = 1 (scale does not affect score)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Hyper‑parameters for the smooth‑max optimisation.
    # ------------------------------------------------------------------
    beta = 30.0                    # smooth‑max sharpness (larger → closer to true max)
    eta = 0.15                     # initial step size
    momentum = 0.0                 # optional momentum term (starts at 0)

    # ------------------------------------------------------------------
    # 4. Main optimisation loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- convolution and smooth‑max weights ---------------------------------
        conv = np.convolve(a, a)                # linear convolution a*a
        max_conv = conv.max()
        # soft‑max (stable: shift by the maximum before exponentiation)
        shifted = conv - max_conv
        weights = np.exp(beta * shifted)
        weights /= weights.sum()                # p_k – probability‑like weights

        # ---- gradient of the smooth‑max objective (correlation) -----------------
        # grad_j = 2 * Σ_k p_k * a_{k-j}
        grad = 2.0 * np.correlate(weights, a, mode='valid')

        # ---- project gradient to the hyper‑plane Σ a = 1 (zero‑mean) -------------
        grad -= grad.mean()

        # ---- momentum update (optional, helps acceleration) --------------------
        velocity = momentum * momentum - eta * grad   # simple Nesterov‑style form
        # Update the sequence, keep it non‑negative, then renormalise.
        a_new = a + velocity
        a_new = np.maximum(a_new, 0.0)

        # Renormalise to keep Σ a = 1 (required for the homogeneous objective).
        if a_new.sum() == 0.0:
            a_new = np.random.rand(n)
        a_new /= a_new.sum()

        # ---- evaluate candidate -------------------------------------------------
        cur_score = evaluate_sequence(a_new.tolist())

        # ---- keep the best ever ------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_new.copy()
            print(f"[{time.time():.1f}s] new best score = {best_score:.6f}")

        # ---- acceptance rule (always accept improvements) ----------------------
        if cur_score < best_score * 1.001:   # tiny tolerance for numerical noise
            a = a_new
        else:
            # occasional random jump to avoid stagnation
            if np.random.rand() < 0.003:
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.02   # small random shake
                a[idx] = max(a[idx] + delta, 0.0)
                a /= a.sum()

        # ---- adapt step size ----------------------------------------------------
        if cur_score < best_score:
            eta = min(eta * 1.02, 5.0)
        else:
            eta = max(eta * 0.98, 1e-5)

        # ---- early exit if target reached --------------------------------------
        if best_score < 1.503:
            print(f"[stop] target reached with score {best_score:.6f}")
            break

    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()
