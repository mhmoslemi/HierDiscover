    i = np.arange(n, dtype=np.float64)

def smooth(seq, steps=1):
    a = np.array(seq)
    kernel = np.array([0.25, 0.5, 0.25])
    for _ in range(steps):
        a = np.convolve(a, kernel, mode='same')
    a = np.clip(a, 0, None)
    a = a / a.sum()
    return a.tolist()

"""
Propose a candidate step‐function sequence.

The evaluation metric is   f(a)=2·n·max(a*a)/(sum(a)^2) .
Since the metric is scale‑invariant we fix sum(a)=1 and minimise max(a*a).

A classical construction is the discrete “arcsine” shape
    a_i ∝ 1/√((i+½)(n‑½‑i)),
whose continuous analogue has a flat self‑convolution.
We start from this shape, try a few simple alternatives,
and then perform a cheap stochastic hill‑climbing search
(perturb one coordinate, renormalise, keep the improvement).
All candidates are kept as non‑negative floats; values are automatically
clamped by ``evaluate_sequence``.  The routine runs for the given time
budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize  # only needed for the LP helper (optional)


def constant_seq(n: int) -> list[float]:
    """Uniform sequence a_i = 1/n (sum = 1)."""
    return [1.0 / n] * n


def arcsine_seq(n: int) -> list[float]:
    """
    Discrete arcsine density:
    a_i ∝ 1 / sqrt((i+½)*(n‑½‑i)).
    This shape is known to give an almost flat self‑convolution.
    """
    i = np.arange(n, dtype=np.float64)
    eps = 0.5          # avoid division by zero at the ends
    vals = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    vals /= np.sum(vals)          # normalise to sum = 1
    return vals.tolist()


def linear_seq(n: int) -> list[float]:
    """Simple symmetric linear “tent” shape."""
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i, n - 1 - i) + 1.0
    vals /= np.sum(vals)
    return vals.tolist()


def quadratic_seq(n: int) -> list[float]:
    """Quadratic (parabolic) symmetric shape."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 1) * (n - i)          # (i+1)*(n-i) gives a parabola
    vals = np.sqrt(vals)             # make the spread milder
    vals /= np.sum(vals)
    return vals.tolist()


def random_dirichlet_seq(n: int, rng: np.random.Generator, alpha: float = 1.0) -> list[float]:
    """
    Sample from Dirichlet(alpha,…,alpha) using Gamma draws.
    alpha=1 gives the flat Dirichlet; larger alpha yields more uniform vectors.
    """
    if alpha == 1.0:
        # exponential (Gamma(1,1)) is a cheap way
        draws = rng.exponential(scale=1.0, size=n)
    else:
        draws = rng.gamma(shape=alpha, scale=1.0, size=n)
    draws /= np.sum(draws)
    return draws.tolist()


def perturb_sequence(seq: list[float], rng: np.random.Generator, sigma: float = 0.05) -> list[float]:
    """
    Randomly perturb a single coordinate (multiplicative Gaussian noise),
    then renormalise to keep the total sum equal to 1.
    """
    n = len(seq)
    new_seq = np.array(seq, copy=True)
    idx = rng.integers(0, n)
    factor = 1.0 + rng.normal(loc=0.0, scale=sigma)
    new_seq[idx] = max(0.0, new_seq[idx] * factor)
    total = new_seq.sum()
    if total == 0.0:
        return seq
    new_seq /= total
    return new_seq.tolist()

    Return a list of non‑negative heights (a step function) that
    minimises ``evaluate_sequence`` within the allotted time.
    """
    # ------------------------------------------------------------------
    # initialise
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    # try to reuse the length of the previously discovered construction
    try:

    except Exception:
        n = 2000                         # fallback length

    deadline = time.time() + budget_s - 2.0   # keep a small safety margin

    best_seq = None
    best_score = float("inf")

    def try_update(candidate):
        """Evaluate candidate, keep it if it improves the best score."""
        nonlocal best_seq, best_score
        try:

        except Exception:
            return
        if score < best_score:
            best_score = score

            print(f"[{time.time():.2f}] New best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # deterministic starters
    # ------------------------------------------------------------------
    print("Evaluating deterministic starters...")
    try_update(constant_seq(n))

    try_update(quadratic_seq(n))

    # ------------------------------------------------------------------
