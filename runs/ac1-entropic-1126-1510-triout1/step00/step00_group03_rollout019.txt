"""Simple stochastic search for a low‐evaluation step‑function.

The evaluation metric is scale‑invariant, therefore we keep the sum of the
coefficients equal to one.  The algorithm repeatedly draws random
vectors of modest length, normalises them, evaluates the objective and
keeps the best one seen.  After the random phase a tiny local optimiser
perturbs the current best vector (moving a small amount of mass between
two coordinates) and accepts the move only if it improves the score.
This cheap strategy is enough to beat the trivial uniform construction
(which scores exactly 2.0) and usually finds a sequence with a score
≈ 1.78 (the optimal 2‑element case).

The function `propose_candidate` is the entry point required by the
framework – it must return a plain list of non‑negative numbers.
"""
import time
import numpy as np

def _evaluate(seq):
    """Wrapper that guards against unexpected errors."""
    try:
        return evaluate_sequence(seq)      # built‑in, already imported
    except Exception:                     # pragma: no cover
        return np.inf

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **_):
    """
    Search for a coefficient sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds.  The function will stop a little before the
        limit to guarantee a return value.
    **_ :
        Ignored additional arguments (kept for compatibility).

    Returns
    -------
    list[float]
        A list of non‑negative heights.  The best sequence found within the
        time budget is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 0.5)  # safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise with the analytically optimal 2‑element sequence.
    # ------------------------------------------------------------------
    best_seq = [2.0 / 3.0, 1.0 / 3.0]        # sum == 1, known optimum for n=2
    best_score = _evaluate(best_seq)

    # ------------------------------------------------------------------
    # 2.  Random sampling phase (very cheap, works for n up to ~30).
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # random length between 2 and 30 (inclusive)
        n = np.random.randint(2, 31)

        # generate a random non‑negative vector and normalise its sum to 1
        seq = np.random.rand(n)
        seq_sum = seq.sum()
        if seq_sum == 0.0:                     # extremely unlikely
            continue
        seq /= seq_sum

        score = _evaluate(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"[+] New best (random)  n={n}  score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 3.  Tiny local refinement around the current best.
        # ------------------------------------------------------------------
        # perform a few cheap perturbations only when we still have time
        if time.time() > deadline - 0.2:
            break

        cur = np.array(best_seq, copy=True)
        # propose a move: shift a small amount of mass from i→j
        i, j = np.random.choice(len(cur), 2, replace=False)
        # step size scales with the magnitude of the source entry
        max_delta = 0.1 * cur[i]
        delta = (np.random.rand() - 0.5) * max_delta
        # keep entries non‑negative
        delta = np.clip(delta, -cur[j], cur[i])
        cur[i] -= delta
        cur[j] += delta

        # re‑evaluate the candidate
        cand_score = _evaluate(cur.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cur.tolist()
            print(f"[+] New best (local)   n={len(cur)}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Final report
    # ------------------------------------------------------------------
    print(f"[final] Best sequence length {len(best_seq)} with score {best_score:.6f}")
    return best_seq
