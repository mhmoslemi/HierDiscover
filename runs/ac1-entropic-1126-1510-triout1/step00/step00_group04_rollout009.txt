def project_to_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]  # find last positive
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    B = np.convolve(a, a)
    Bmax = np.max(B)

    w = np.exp(gamma * (B - Bmax))

    a_new = a - step * grad

Algorithm
---------

The objective can be written as  

    R(a) = 2·n·max_k (a∗a)_k / (∑a_i)^2 .

# Helper: projection onto the simplex {x ≥ 0, sum(x)=1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive: return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# FFT based linear convolution (full length)
def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    n = a.shape[0] + b.shape[0] - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    conv = np.fft.irfft(A * B, N)[:n]
    return conv


# --------------------------------------------------------------
# Linear correlation a⋆w (full length, using FFT)
def _corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    # correlation a⋆w = ifft( fft(a) * conj(fft(w)) )
    n = a.shape[0] + w.shape[0] - 1
    N = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    corr = np.fft.irfft(A * np.conj(W), N)[:n]
    return corr


# --------------------------------------------------------------
def _smooth_gradient_step(a: np.ndarray,
                         beta: float,
                         step: float) -> np.ndarray:
    """
    One iteration of the smoothed gradient descent:

    * compute the full convolution B = a∗a
    * build soft‑max weights W that concentrate on the largest entries of B
    * compute the gradient ∇ = 2·corr(a,W)/sum(W)
    * take a step and project onto the simplex
    * symmetrise the result (a←(a+a[::-1])/2)
    """
    n = a.shape[0]

    # 1) convolution B = a * a
    B = _conv_fft(a, a)                     # length 2n‑1
    Bmax = B.max()
    if Bmax <= 0.0:                         # should never happen, but guard
        return a

    # 2) soft‑max weights, scaled by the relative gap to the maximum
    #    (the factor 1/Bmax makes the exponent size roughly independent of n)
    W = np.exp(beta * (B - Bmax) / Bmax)    # each entry ≤ 1, max entry = 1
    sW = W.sum()
    if sW == 0.0:
        return a

    # 3) gradient = 2 * corr(a, W) / ΣW   (only the first n components are needed)
    corr_full = _corr_fft(a, W)             # length 3n‑2
    grad = 2.0 * corr_full[:n] / sW

    # 4) gradient descent step
    a_next = a - step * grad

    # 5) projection onto the simplex (keeps sum = 1, non‑negative)
    a_proj = _project_simplex(a_next)

    # 6) enforce symmetry – improves convergence and reduces search space
    a_sym = (a_proj + a_proj[::-1]) * 0.5
    # re‑project because averaging can break exact sum‑to‑1
    a_final = _project_simplex(a_sym)

    return a_final


# --------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: int = 1000,
                     **kwargs) -> list:
    """
    Search for a step‑function height sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.  The function stops a few seconds before the
        deadline to guarantee a return value.
    kwargs : dict
        Ignored – present for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best height sequence found within the time budget.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the working length.  Use the length of the provided SOTA
    # construction if it exists, otherwise fall back to a moderate size.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Initialise the current iterate.
    # With 50 % probability start from the known construction,
    # otherwise start from a fresh random point.
    if np.random.rand() < 0.5 and 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a = a / a.sum()
    else:
        a = np.random.rand(n)
        a = a / a.sum()

    best_a = a.copy()
    try:
        best_score = evaluate_sequence(best_a.tolist())
    except Exception:                     # safety net
        best_score = float('inf')

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    step0 = 0.12               # initial step size (decays as 1/√t)
    beta0 = 5.0                # initial soft‑max sharpness
    beta_max = 20.0
    beta_inc = 0.02            # incremental increase per iteration
    max_no_improve = 2500      # restart after this many stagnant steps

    # Timing control
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a small safety margin

    it = 0
    no_improve = 0
    beta = beta0

    while time.time() < deadline:
        # gradually increase β to focus more on the true max
        beta = min(beta_max, beta + beta_inc)

        # step size decays like η₀ / sqrt(t)
        step = step0 / np.sqrt(it + 1.0)

        # one gradient step
        a = _smooth_gradient_step(a, beta, step)

        # evaluate the current candidate
        try:
            cur_score = evaluate_sequence(a.tolist())
        except Exception:
            cur_score = float('inf')

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[iter {it:05d}] new best = {best_score:.6f}, beta={beta:.2f}")
        else:
            no_improve += 1

        # occasional random restart to escape possible local minima
        if no_improve >= max_no_improve:
            a = np.random.rand(n)
            a = a / a.sum()
            no_improve = 0
            print(f"[iter {it:05d}] restart (no improvement)")

        it += 1

    print(f"[done] iterations={it}, best score={best_score:.6f}")
    # Convert to plain Python list as required by the caller
    return best_a.tolist()
