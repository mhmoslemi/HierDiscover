def gradient_step(a, step):
    # compute convolution
    b = np.convolve(a, a)
    max_b = b.max()
    # find active indices
    active = np.where(b >= max_b - 1e-12)[0]
    # compute gradient
    grad = np.zeros_like(a)
    n = len(a)
    for k in active:
        # for i in [max(0, k-(n-1)), min(n-1, k)]:
        start = max(0, k-(n-1))
        end = min(n-1, k)
        # i runs start..end:
        # grad[i] += 2*a[k-i]
        # we can vectorize: slice of a from k-start down to ...
        # Actually we can use convolution of indicator? Use loops.
        for i in range(start, end+1):
            grad[i] += 2 * a[k-i]
    # move opposite direction
    a_new = a - step * grad
    # project onto nonnegative simplex (maintain sum=1)
    a_new = np.maximum(a_new, 0)
    # renormalize to sum=1
    total = a_new.sum()
    if total < 1e-12:
        # avoid zero: set uniform
        a_new = np.ones_like(a) / len(a)
    else:
        a_new /= total
    return a_new

    deadline = time.time() + budget_s - 0.1   # tiny safety margin

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------ #
    # 2. Grid‑search a handful of polynomial exponents.
    #    The shape a_i = (i+1)^p is normalised to sum‑to‑one, keeping
    #    all entries tiny and well inside the allowed range.
    # ------------------------------------------------------------------ #
    exponents = list(range(1, 11))          # 1 … 10
    for p in exponents:
        if time.time() > deadline:
            break

        i = np.arange(1, n + 1, dtype=np.float64)
        seq = i ** p                         # polynomial growth
        seq = seq / seq.sum()                # normalise → sum(seq) == 1

        # evaluate the candidate (evaluate_sequence is provided by the runner)
        score = evaluate_sequence(seq.tolist())
        print(f"p = {p:2d} → score = {score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # Fallback – should never happen, but guard against an empty best_seq.
    if best_seq is None:
        best_seq = np.ones(n, dtype=np.float64) / n
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback uniform sequence score = {best_score:.6f}")

    # ------------------------------------------------------------------ #
    # 3. Light stochastic refinement: add tiny Gaussian noise,
    #    keep the candidate non‑negative, renormalise to keep the scale,
    #    and accept it if it improves the score.
    # ------------------------------------------------------------------ #
    while time.time() < deadline:
        # generate a perturbed copy of the current best sequence
        noise = np.random.normal(scale=0.01, size=best_seq.shape)
        cand = best_seq + noise
        cand = np.maximum(cand, 0.0)                     # enforce non‑negative

        s = cand.sum()
        if s < 1e-12:                                   # avoid division by zero
            continue
        cand = cand / s                                  # renormalise to sum == 1

        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"Improved score to {best_score:.6f}")

    print(f"Best sequence found with score {best_score:.6f}")
    return best_seq.tolist()
