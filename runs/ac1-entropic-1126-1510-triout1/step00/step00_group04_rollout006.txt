    best_seq = None
    best_val = float('inf')
    n = 4000  # maybe adjust based on time.

    # parameters
    beta = 5.0
    eta = 0.2  # step size maybe adjust after iterations

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute convolution
        b = np.convolve(a, a)
        max_b = b.max()
        # early compute temperatures
        # use softmax weights
        # compute w = exp(beta * (b - max_b))
        # Might overflow if beta large; but exponent <= 1
        w = np.exp(beta * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(b) / len(b)
        else:
            w = w / w_sum

        # compute gradient with conv of a and reversed w
        # w_rev = w[::-1]
        # conv = np.convolve(a, w_rev)
        # g_i = 2 * conv[ (len(w)-1) - i ]  # indices
        w_rev = w[::-1]
        conv = np.convolve(a, w_rev)
        L = len(w)  # = 2*n -1
        # compute gradient for i = 0..n-1
        # index in conv for i: idx = L-1 - i
        indices = (L - 1) - np.arange(n)
        g = 2.0 * conv[indices]

        # update a with step
        a_new = a - eta * g
        # ensure nonnegative
        a_new = np.maximum(0.0, a_new)

        # renormalize to sum=1 (if sum zero, reinitialize)
        sum_a_new = a_new.sum()
        if sum_a_new < 1e-12:
            # reinitialize random
            a_new = np.random.rand(n)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / sum_a_new

        a = a_new

        # optionally adapt eta
        if iteration % 100 == 0:
            # evaluate current objective
            cur_val = evaluate_sequence(a.tolist())
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

"""
A fast and robust search for a step‑function with a low value of
``evaluate_sequence``.

The search is based on the following observations.

* For a non‑negative sequence a with Σa = 1 the evaluation reduces to
  `2·n·max(conv(a,a))`.  Hence we only have to make the maximal entry of the
  self‑convolution as small as possible.

* If a is chosen as a discretised *arcsine* law,
      a_i ∝ ((i+½)(n‑i‑½))^{‑α}   (0 < α ≤ 1),
  the distribution of the sum of two independent draws becomes almost
  uniform, i.e. `max(conv(a,a))` is close to the average value `1/(2n‑1)`.
  This already brings the objective close to 1 (the theoretical lower bound).

* Starting from such a parametrised candidate we run a few steps of a
  projected gradient descent on a smooth approximation of the max–norm.
  The gradient of the smoothed max is computed efficiently with two
  convolutions, so each iteration is cheap (≈ O(n log n)).

The routine explores several lengths `n` and several exponents `α`,
keeps the best candidate seen, and finally refines the best one until the
budget expires.  All numbers are kept non‑negative and are re‑normalised
to have unit sum, which guarantees that the security checks inside
`evaluate_sequence` never reject the sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper utilities
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a normalised sequence (sum=1).
    Returns 2 * n * max(conv(a, a)).
    """
    n = a.shape[0]
    max_b = np.max(np.convolve(a, a))
    return 2.0 * n * max_b


def _gradient_descent(a: np.ndarray, stop_time: float) -> np.ndarray:
    """
    Very short projected gradient descent on a smooth approximation of
    max(conv(a,a)).  The sequence is kept non‑negative and re‑normalised
    after each step.
    """
    # parameters – they are deliberately conservative because we only have
    # a few seconds per refinement.
    eta = 0.2          # step size
    beta = 5.0         # smoothing parameter for the soft‑max
    best_a = a.copy()
    best_val = _objective(best_a)

    while time.time() < stop_time:
        # convolution of a with itself
        conv_aa = np.convolve(a, a)
        max_b = conv_aa.max()

        # smooth (soft‑max) weights over the convolution entries
        w = np.exp(beta * (conv_aa - max_b))   # w_i ≤ 1, w_max = 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient of the smoothed max: g_i = 2 * Σ_k w_k * a_{k-i}
        w_rev = w[::-1]
        conv = np.convolve(a, w_rev)          # length = n + (2n‑1) – 1 = 3n‑2
        L = w.size                            # = 2n‑1
        idx = (L - 1) - np.arange(a.size)     # indices that give Σ_k w_k a_{k-i}
        grad = 2.0 * conv[idx]

        # projected gradient step
        a = a - eta * grad
        a = np.maximum(a, 0.0)                # non‑negative
        s = a.sum()
        if s == 0.0:                          # avoid degenerate case
            a = np.random.rand(a.size)
            s = a.sum()
        a /= s                                 # renormalise to sum = 1

        # keep the best seen
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # aggressive step‑size reduction after a good improvement
            eta = max(0.01, eta * 0.9)

        # slow decay of the step size to guarantee convergence
        eta = max(0.001, eta * 0.995)

    return best_a


# ----------------------------------------------------------------------
# main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    The method:
      1. Enumerate a few lengths `n` and exponents `α` for the
         family   a_i ∝ ((i+½)(n‑i‑½))^{‑α}.
      2. Keep the best candidate (according to the fast objective).
      3. Refine the best candidate with a short gradient‑descent routine.
    The function returns the best sequence found before the time budget runs
    out.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1. explore a small family of analytically motivated candidates
    # ------------------------------------------------------------------
    lengths = [500, 1000, 2000, 4000, 5000]   # larger n -> lower theoretical bound
    alphas = np.linspace(0.20, 1.00, 9)       # from near‑uniform (α≈0) to heavy edges

    for n in lengths:
        if time.time() >= deadline:
            break
        i = np.arange(n, dtype=np.float64)
        # denominator used for the arcsine‑type law
        base = (i + 0.5) * (n - i - 0.5)      # strictly positive for all i

        for alpha in alphas:
            if time.time() >= deadline:
                break

            # a_i ∝ base^{‑α}
            raw = np.power(base, -alpha, dtype=np.float64)
            a = raw / raw.sum()               # enforce Σa = 1

            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[init] n={n:5d} α={alpha:4.2f} → val={val:.6f}")

            # ------------------------------------------------------------------
            # 2. a quick local refinement (≈5 s or until the global deadline)
            # ------------------------------------------------------------------
            sub_deadline = min(deadline, time.time() + 5.0)
            a_refined = _gradient_descent(a, sub_deadline)
            val_ref = _objective(a_refined)
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_refined.copy()
                print(f"[ref ] n={n:5d} α={alpha:4.2f} → val={val_ref:.6f}")

    # ----------------------------------------------------------------------
    # 3. final long‑run refinement of the globally best sequence
    # ----------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print(f"[final] Starting long refinement (remaining time ≈ "
              f"{deadline - time.time():.1f}s)")
        best_seq = _gradient_descent(best_seq, deadline)
        best_val = _objective(best_seq)
        print(f"[final] refined value = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Convert the numpy array to plain Python list as required by the runner
    # ----------------------------------------------------------------------
    print(f"[done] best objective = {best_val:.6f}")
    return best_seq.tolist()
