            best_seq = seq
            print(f"New best score {best_score:.6f} n={n} alpha={alpha:.3f}")

        a[j] -= delta

"""
Propose a sequence that improves the evaluation score.

The idea is based on the known near‑optimal shape a_i ≈ 1/√(i+1) which gives
a theoretical evaluation of π/2 ≈ 1.571 (much better than the current
upper bound of 2.0).  Starting from this analytic form we perform a very fast
hill‑climbing search: a random coordinate is perturbed and the new sequence
is kept only when the evaluation score improves.  Because the evaluation
function is homogeneous (scaling the whole vector does not change the
score) we keep the values in the natural range (all below 1) – the security
checks therefore never clip them.

The algorithm runs for the allotted time (default ≈ 1000 s) and returns the
best sequence found.
"""

import time
import math
import numpy as np

def _initial_sequence(n: int) -> np.ndarray:
    """Return the classic 1/√(i+1) sequence of length *n*."""
    # i starts at 0 -> denominator i+1 to avoid division by zero
    return 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))


def _perturb(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Produce a new sequence by perturbing a single random coordinate.
    The perturbation is multiplicative: a_i <- a_i * (1 + delta) with
    delta drawn from U[-step, step].  Values are kept non‑negative.
    """
    new_seq = seq.copy()
    i = np.random.randint(len(seq))
    delta = (np.random.rand() * 2.0 - 1.0) * step   # uniform in [-step, step]
    new_val = new_seq[i] * (1.0 + delta)
    # keep non‑negative (the security checks also enforce this)
    new_seq[i] = max(0.0, new_val)
    return new_seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a sequence of non‑negative heights that (hopefully) beats the
    current best evaluation score.  The function respects the 1000 second
    time budget and returns the best sequence it discovered.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (the harness will call us with the remaining
        time; we stop a little earlier to be safe).

    Returns
    -------
    list[float]
        The sequence with the lowest evaluation score found.
    """
    np.random.seed(seed)
    # Choose a length that is comfortably large but still fast to handle.
    # 8000 gives a good trade‑off (≈ 2 × √n ≈ 180 in sum, max‑conv ≈ π).
    n = 8000

    # Start from the analytical near‑optimal shape.
    current_seq = _initial_sequence(n)

    # Initial evaluation.
    best_score = evaluate_sequence(current_seq.tolist())
    best_seq = current_seq.copy()
    print(f"Initial score (n={n}): {best_score:.6f}")

    # Hill‑climbing parameters.
    step = 0.05               # maximal relative change of a coordinate
    temperature = 1.0         # used for a simple simulated‑annealing acceptance rule
    cooling_rate = 0.9999     # geometric cooling

    deadline = time.time() + budget_s - 0.5   # a safety margin
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # Propose a small perturbation.
        candidate = _perturb(current_seq, step)

        # Evaluate the candidate.
        try:
            cand_score = evaluate_sequence(candidate.tolist())
        except Exception:
            # Any numerical mishap – just skip this candidate.
            continue

        # Accept if it improves, otherwise accept with a tiny probability
        # that slowly vanishes (simulated annealing).
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(cand_score - best_score) / max(temperature, 1e-9))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current_seq = candidate
            if cand_score < best_score:
                best_score = cand_score
                best_seq = candidate.copy()
                print(f"Iter {iteration:6d}: new best {best_score:.6f}")

        # Cool down the temperature a little each iteration.
        temperature *= cooling_rate

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    # Convert to a plain Python list as required by the evaluator.
    return best_seq.tolist()
