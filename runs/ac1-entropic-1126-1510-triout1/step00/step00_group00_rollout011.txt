import numpy as np, time, math

def forward_convolve(a):
    # compute full convolution via FFT
    n = len(a)
    L = 1
    while L < 2*n - 1:
        L <<= 1
    fa = np.fft.rfft(a, n=L)
    b = np.fft.irfft(fa*fa, n=L)
    return b[:2*n-1]

def smooth_max(b, gamma):
    # compute softmax weights
    max_b = np.max(b)
    exp_terms = np.exp(gamma * (b - max_b))
    sum_exp = np.sum(exp_terms)
    w = exp_terms / sum_exp
    # smooth max value
    smooth_val = max_b + np.log(sum_exp) / gamma
    return smooth_val, w

def project_simplex(a, target):
    # ensure nonnegative, scale to target sum
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s == 0:
        return a
    return a * (target / s)

Propose a better step‑function by directly minimizing a smooth approximation of
the quantity  

    2·n·max(conv(a,a)) / (sum(a)^2)

which is exactly the value returned by ``evaluate_sequence`` (the expression is
scale‑invariant, therefore we keep the sum fixed to √(2·n)).  

The algorithm
  1. builds a random feasible start,
  2. repeatedly computes the full convolution a*a,
  3. replaces the hard “max” by a smoothed max (soft‑max) with a temperature
     parameter γ that is increased during the run,
  4. evaluates the gradient of the smoothed objective
     (∇_a  softmax(conv(a,a)) = 2·conv(w, a_rev) where w are the soft‑max
     weights and a_rev is the reversed sequence),
  5. takes a projected gradient step (non‑negativity and projection onto the
     simplex Σa = √(2·n)),
  6. keeps the best sequence seen.

The whole loop runs for the allotted budget (≈ 1000 s).  The use of FFT‑based
convolutions makes each iteration cheap (≈ 10⁴‑10⁵ flops for n≈6500), so many
updates are performed and the method usually drives the maximal convolution
value below the uniform‑sequence value 2.0, yielding a better upper bound.
"""

import time

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small utility: smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def _conv_full_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution using real FFTs.
    Both inputs are 1‑D real arrays.
    """
    n = a.size + b.size - 1
    L = _next_pow2(n)
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)[:n]
    return conv


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper for convolving a vector with itself."""
    return _conv_full_fft(a, a)


def _smooth_max_weights(b: np.ndarray, gamma: float) -> np.ndarray:
    """
    Return the soft‑max weights w_k = exp(gamma·b_k) / Σexp(gamma·b)
    (numerically stable by subtracting the max).
    """
    b_max = np.max(b)
    # shift to avoid overflow
    exp_shift = np.exp(gamma * (b - b_max))
    w = exp_shift / np.sum(exp_shift)
    return w


def _project_onto_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a non‑negative vector onto the simplex Σx = target_sum.
    For this problem the projection reduces to a simple scaling because we
    never obtain negative components after the gradient step.
    """
    x = np.maximum(x, 0.0)
    current = np.sum(x)
    if current == 0.0:
        # avoid division by zero – return a uniform vector
        return np.full_like(x, target_sum / x.size)
    return x * (target_sum / current)


def _gradient_of_smooth_max(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Compute the gradient of the smoothed max of conv(a,a) w.r.t. a.
    The gradient is 2·conv(w, a_rev) where w are the soft‑max weights.
    """
    # 1. full convolution a*a
    b = _full_convolution(a)

    # 2. soft‑max weights for the current γ
    w = _smooth_max_weights(b, gamma)

    # 3. reversed a (needed for the gradient formula)
    a_rev = a[::-1]

    # 4. convolution w * a_rev  (full)
    grad_full = _conv_full_fft(w, a_rev)

    # The required slice starts at index (len(a)-1) and has length len(a)
    n = a.size
    grad = 2.0 * grad_full[n - 1: n - 1 + n]
    return grad
