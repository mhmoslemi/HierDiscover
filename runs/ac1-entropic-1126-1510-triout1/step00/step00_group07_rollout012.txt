def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    # Use existing height_sequence_1 if defined
    try:
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
        # ensure non-negative and clip
        base_seq = np.clip(base_seq, 0, None)
    except NameError:
        n = 2000
        base_seq = None

    # target sum = sqrt(2*n)
    target_sum = math.sqrt(2.0 * n)

    best_seq = None
    best_score = float('inf')

    # function to normalize sequence to target sum
    def normalize(seq):
        s = seq.sum()
        if s == 0:
            return seq
        return seq * (target_sum / s)

    # initial candidates list
    candidates = []
    # uniform
    uniform = np.full(n, target_sum / n, dtype=float)
    candidates.append(uniform.copy())
    # random Dirichlet
    for _ in range(3):
        rand = rng.random(n)
        seq = rand / rand.sum() * target_sum
        candidates.append(seq)

    # include previous best if available
    if base_seq is not None:
        candidates.append(normalize(np.array(base_seq, dtype=float)))

    # main loop: for each candidate run gradient descent
    for init_seq in candidates:
        seq = init_seq.copy()
        # small noise to avoid degenerate
        seq = np.clip(seq + rng.normal(scale=1e-3, size=n), 0, None)
        seq = normalize(seq)

        # inner optimization
        alpha = 1.0
        lr = 0.5
        # schedule
        for it in range(2000):
            if time.time() >= deadline:
                break
            # convolution
            b = np.convolve(seq, seq)
            # compute max evaluation
            # no need to compute objective directly because evaluate_sequence does clipping etc.
            # but we can compute max_b for scheduling
            max_b = b.max()
            # evaluate after scaling (should be same as max_b due to normalization)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                # print progress
                print(f"New best score {best_score:.6f} at iteration {it}")
            # smoothed weights
            shift = b.max()
            exp_term = np.exp(alpha * (b - shift))
            w = exp_term / exp_term.sum()
            # gradient = 2 * alpha * (conv of w with reversed seq)
            conv_rev = np.convolve(w, seq[::-1])
            grad = 2.0 * alpha * conv_rev[n-1: n-1+n]  # length n
            # update using mirror descent (exponential) for positivity
            seq = seq * np.exp(-lr * grad)
            seq = normalize(seq)
            # adapt alpha and lr
            if (it+1) % 200 == 0:
                alpha *= 1.5
                lr *= 0.7
        if time.time() >= deadline:
            break

    # fallback: if we have not found any sequence, return uniform
    if best_seq is None:
        best_seq = uniform

    return best_seq.tolist()

"""
propose_candidate
~~~~~~~~~~~~~~~~~

Search for a sequence of non‑negative heights that minimizes the provided
``evaluate_sequence``.  The optimisation treats the problem as the minimisation
of the ℓ∞‑norm of the self‑convolution of a vector under an ℓ1‑normalisation.
The algorithm proceeds by many short restarts, each consisting of a simple
projected “sub‑gradient” descent:

*   Initialise a probability vector (sum = sqrt(2 n)) – either the existing
    ``height_sequence_1`` or random Dirichlet draws.
*   Repeatedly compute the convolution, locate the index of its current maximum
    and move mass away from those entries via an exponential (mirror) descent
    step that respects non‑negativity.
*   Renormalise after every step, keep the best sequence seen and occasionally
    inject tiny random perturbations to escape plateaus.

The method uses only NumPy and plain Python, runs for at most ``budget_s``
seconds and returns the best sequence found (as a plain ``list[float]``).
"""

import time
import math
import numpy as np

def _normalise_to_target(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``seq`` so that its ℓ1‑sum equals ``target_sum`` (preserving non‑negativity)."""
    s = seq.sum()
    if s == 0.0:
        # avoid division by zero – fall back to a uniform vector
        return np.full_like(seq, target_sum / len(seq))
    return seq * (target_sum / s)


def _subgradient_max_conv(seq: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution functional.
    ``conv`` is the full linear convolution ``seq * seq`` (length 2n‑1).
    Returns a non‑negative gradient vector of length n.
    """
    n = len(seq)
    k = int(conv.argmax())                     # index of the current maximal entry
    grad = np.zeros_like(seq)
    # derivative of that entry w.r.t. each component:
    # ∂(sum_i seq_i * seq_{k-i}) / ∂seq_j = 2 * seq_{k-j}  (if the index is inside the array)
    for j in range(n):
        i = k - j
        if 0 <= i < n:
            grad[j] = 2.0 * seq[i]
    return grad

        conv = np.convolve(seq, seq)

    best_score = float('inf')

    # ------------------------------------------------------------------
    # Main optimisation loop – each seed gets a short but intensive descent.
    # ------------------------------------------------------------------
    for seed_idx, seq in enumerate(seeds):
        # small random jitter to avoid starting exactly at a saddle point
        seq = np.clip(seq + rng.normal(scale=1e-4, size=n), 0.0, None)
        seq = _normalise_to_target(seq, target_sum)

        # plain exponential‑mirror descent parameters
        lr_initial = 0.5          # larger values move faster at the start
        lr_decay = 0.995          # geometric decay per iteration
        lr = lr_initial

        # run a bounded number of inner iterations or until the time budget expires
        max_inner = 3000
        for it in range(max_inner):
            if time.time() >= deadline:
                break

            # convolution and objective value
            conv = np.convolve(seq, seq)                 # length 2n‑1
            cur_score = float(conv.max())

            # keep the best ever seen
            if cur_score < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                print(f"[{time.time():.1f}] New best score {best_score:.6f} "
                      f"(seed {seed_idx}, iter {it})")

            # Sub‑gradient step w.r.t. the current max entry
            grad = _subgradient_max_conv(seq, conv)

            # Exponential (mirror) update that respects non‑negativity
            seq = seq * np.exp(-lr * grad)

            # Re‑normalise to the required ℓ1‑sum
            seq = _normalise_to_target(seq, target_sum)

            # slowly decay the learning rate
            lr *= lr_decay

            # Every few hundred steps inject a tiny random jitter – helps to
            # escape tiny plateaus without destroying progress.
            if (it + 1) % 250 == 0:
                jitter = rng.normal(scale=1e-5, size=n)
                seq = np.clip(seq + jitter, 0.0, None)
                seq = _normalise_to_target(seq, target_sum)

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return the plain uniform construction.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform

    # The evaluator will clip any value >1000, but all our numbers are far
    # below that bound, so the returned list is safe.
    return best_seq.tolist()
