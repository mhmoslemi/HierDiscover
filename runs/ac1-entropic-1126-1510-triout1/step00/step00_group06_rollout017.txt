```python
def project_onto_simplex(v):
    # ensure v is array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=314159, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    # Trying multiple lengths
    lengths = [500, 800, 1200, 2000, 3000]  # maybe adjust
    while time.time() < deadline:
        # Pick length random
        n = np.random.choice(lengths)
        # Initialize random simplex vector
        # Dirichlet with concentration = 0.5 => many small entries
        conc = np.random.uniform(0.3, 1.0) # random concentration
        vec = np.random.gamma(conc, 1.0, size=n)
        vec = vec / np.sum(vec)
        # Simulated annealing loop within remaining time for this restart
        # give e.g., 1 sec per restart? or dynamic.
        restart_deadline = min(deadline, time.time() + 2.0) # allocate 2 secs each
        cur_seq = vec
        cur_score = evaluate_sequence(cur_seq.tolist())
        # Keep local best same as global best
        # temperature schedule parameters
        start_temp = 0.5
        end_temp = 1e-4
        # parameters
        it = 0
        while time.time() < restart_deadline:
            n = len(cur_seq)
            # compute convolution
            conv = np.convolve(cur_seq, cur_seq)
            k_max = int(np.argmax(conv))
            # contributions
            # compute products only where indices valid: i in 0..n-1, j = k_max - i must be in range.
            i_vals = np.arange(n)
            j_vals = k_max - i_vals
            mask = (j_vals >= 0) & (j_vals < n)
            # contributions for valid i
            contrib = np.zeros(n)
            valid_i = i_vals[mask]
            contrib[valid_i] = cur_seq[valid_i] * cur_seq[j_vals[mask]]
            # find index with max contribution
            i_max = int(np.argmax(contrib))
            # choose a random index j != i_max
            j = i_max
            while j == i_max:
                j = np.random.randint(n)
            # step size based on iteration
            fraction = 0.2  # base fraction of mass moved
            # can decay over iterations
            frac = fraction * (1 - it / 5000)
            eps = frac * cur_seq[i_max]
            if eps <= 0:
                # no mass to move, break
                break
            # propose new sequence
            new_seq = cur_seq.copy()
            new_seq[i_max] -= eps
            new_seq[j] += eps
            # ensure nonnegative
            if new_seq[i_max] < 0:
                new_seq[i_max] = 0.0
            # evaluate
            new_score = evaluate_sequence(new_seq.tolist())
            # compute temperature
            total_iters = 5000
            temp = start_temp * (1 - it / total_iters) + end_temp * (it / total_iters)
            # acceptance
            accept = False
            if new_score < cur_score:
                accept = True
            else:
                if temp > 0:
                    prob = np.exp(-(new_score - cur_score) / temp)
                    if np.random.rand() < prob:
                        accept = True
            if accept:
                cur_seq = new_seq
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.tolist()
            it += 1
        # optional use LP improvement on best_seq for few iterations

        if time.time() < deadline and best_seq is not None:

                    best_seq = new_seq
                seq = new_seq
    # Return best_seq (list of floats)
    if best_seq is None:
        # fallback: uniform sequence
        n = 1000

"""
Propose a good step‑function sequence.

The algorithm is a hybrid of:
1. Random/simplex initialization (Dirichlet draws) for many lengths.
2. A fast simulated‑annealing / greedy mass‑transfer heuristic that
   repeatedly moves probability mass from indices that contribute most
   to the current maximal convolution entry to indices that contribute
   the least.  The move size shrinks with the iteration count and a
   temperature schedule allows occasional uphill moves.
3. Occasionally improve the current best sequence with the linear‑program
   direction used by the previous state‑of‑the‑art code (get_good_direction_to_move_into).
4. Keep the best sequence seen before the time limit expires.

All sequences are normalised to sum 1 (scaling does not affect the
evaluation score).  The routine returns a plain Python list of floats.
"""

import numpy as np
import time
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex (sum = 1, ≥0)
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x | x_i ≥ 0, Σ_i x_i = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n + 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Linear‑program direction from the previous approach (re‑used here)
def solve_convolution_lp(f_sequence, rhs):
    """Solve LP: maximise Σ b_j  s.t.  (f * b)_k ≤ rhs , b ≥ 0."""
    n = len(f_sequence)
    c = -np.ones(n)                     # maximise sum(b) → minimise -sum(b)
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)
    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])
    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub,
                          method='highs',
                          options={'time_limit': 5.0, 'disp': False})
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(sequence):
    """LP step: produce a new sequence that keeps the maximal convolution bound."""
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return None
    # normalise to sqrt(2n) as required by the original method
    normalized = [x * np.sqrt(2 * n) / sum_seq for x in sequence]
    rhs = np.max(np.convolve(normalized, normalized))
    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    g_norm = [x * np.sqrt(2 * n) / sum_g for x in g]
    t = 0.02                     # modest mixing factor
    new_seq = [(1 - t) * a + t * b for a, b in zip(sequence, g_norm)]
    # re‑normalise to sum 1 (scaling does not affect the score)
    new_seq = np.array(new_seq)
    new_seq = new_seq / np.sum(new_seq)
    return new_seq.tolist()

# ----------------------------------------------------------------------
def simulated_annealing(initial_seq, time_budget):
    """
    Greedy + SA local optimisation on the simplex.
    Returns (best_seq, best_score) found within time_budget seconds.
    """
    start = time.time()
    cur_seq = np.array(initial_seq, dtype=np.float64)
    cur_score = evaluate_sequence(cur_seq.tolist())
    best_seq = cur_seq.copy()
    best_score = cur_score

    n = len(cur_seq)
    # SA schedule parameters
    max_iters = 20000
    start_temp = 0.5
    end_temp = 1e-4
    base_fraction = 0.25   # fraction of mass moved from worst contributor

    for it in range(max_iters):
        if time.time() - start > time_budget:
            break

        # convolution and index of the maximal entry
        conv = np.convolve(cur_seq, cur_seq)
        k_max = int(np.argmax(conv))

        # contributions of each i to this maximal convolution value
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)
        contrib = np.zeros(n)
        valid_i = i_idx[mask]
        contrib[valid_i] = cur_seq[valid_i] * cur_seq[j_idx[mask]]

        # pick the worst and the best contributing indices
        i_worst = int(valid_i[np.argmax(contrib[valid_i])])
        i_best = int(valid_i[np.argmin(contrib[valid_i])])

        # step size shrinks with iteration count
        frac = base_fraction * (1.0 - it / max_iters)
        eps = frac * cur_seq[i_worst]          # amount of mass to move
        if eps <= 0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i_worst] -= eps
        new_seq[i_best] += eps

        # evaluate new candidate
        new_score = evaluate_sequence(new_seq.tolist())
        # temperature for this iteration
        temp = start_temp * (1 - it / max_iters) + end_temp * (it / max_iters)

        # accept if better, otherwise with Metropolis probability
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if temp > 0:
                prob = np.exp(-(new_score - cur_score) / temp)
                if np.random.rand() < prob:
                    accept = True
        if accept:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
    return best_seq.tolist(), best_score


# ----------------------------------------------------------------------
def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Main entry point for the challenge.
    Uses several random restarts, a fast SA mass‑transfer heuristic,
    and occasional LP refinements.  Returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # a few seconds safety margin
    best_seq = None
    best_score = float('inf')

    # allow the previous best construction as a possible start
    use_sota = np.random.rand() < 0.5
    if use_sota and 'height_sequence_1' in globals():
        init = np.array(height_sequence_1, dtype=np.float64)
        init = init / np.sum(init)    # normalise to sum 1
        best_seq = init.tolist()
        best_score = evaluate_sequence(best_seq)
        print(f"Starting from SOTA sequence – initial score {best_score:.6f}")

    # set of lengths to explore (larger lengths give a larger search space)
    lengths = [400, 800, 1200, 2000, 3000]

    # main loop: keep launching short SA runs until time runs out
    while time.time() < deadline:
        # pick random length and Dirichlet concentration
        n = int(np.random.choice(lengths))
        concentration = np.random.uniform(0.2, 0.9)   # lower → spikier initial vectors
        # random simplex point
        vec = np.random.gamma(concentration, 1.0, size=n)
        vec = vec / np.sum(vec)

        # allocate a short budget for this restart (≈2 s or remaining time)
        remaining = deadline - time.time()
        run_budget = min(2.0, remaining)

        # run SA on this starting point
        cand_seq, cand_score = simulated_annealing(vec, run_budget)

        # keep the best found so far
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq
            print(f"New best after SA: score={best_score:.6f}, n={n}")

        # optional LP refinements (a few iterations, cheap)
        # if we still have time, try to polish the current best
        extra_time = deadline - time.time()
        if extra_time > 3.0 and best_seq is not None:
            seq = best_seq
            for _ in range(3):
                new_seq = get_good_direction_to_move_into(seq)
                if new_seq is None:
                    break
                new_score = evaluate_sequence(new_seq)
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq
                    print(f"LP refinement improved score to {best_score:.6f}")
                seq = new_seq

    # fallback – if everything failed, return a uniform sequence
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence.")

    print(f"Finished search – best score {best_score:.6f} with length {len(best_seq)}")
    return best_seq
