def fft_convolve(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n-1).bit_length()

    Y = np.fft.rfft(y, size)
    result = np.fft.irfft(X * Y, size)[:n]
    return result

import numpy as np
import time

def fft_convolve(x, y):
    """Return the linear convolution of two 1‑D arrays using FFT.

    The result length is len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    res = np.fft.irfft(X * Y, size)[:n]
    return res

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence() by iteratively flattening the self‑convolution.
    The method uses a smooth approximation of the max (log‑sum‑exp)
    and exponentiated‑gradient updates on the probability simplex.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep a safety margin

    # choose a working length – use the known construction if present,
    # otherwise fall back to a moderate size.
    n = 2000
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)
        except Exception:
            pass

    # Initialise the best solution with the uniform vector (its value is 2.0).
    uniform = np.full(n, 1.0 / n)
    best_seq = uniform.tolist()
    best_score = evaluate_sequence(best_seq)

    # hyper‑parameters – they are slowly annealed during the run.
    lr = 0.2          # step size for the exponentiated update
    beta = 0.5        # smoothing factor for the soft‑max (low → smooth)
    it = 0
    restart_counter = 0

    # main outer loop – repeatedly start from a fresh random point.
    while time.time() < deadline:
        # start from a random point on the simplex
        a = np.random.rand(n)
        a = a / a.sum()

        # inner optimisation loop for a single start point
        # we limit the number of inner iterations heuristically so that
        # we can restart many times and explore the landscape.
        inner_iters = 0
        while time.time() < deadline and inner_iters < 5000:
            # ---------------------------------------------------------
            # 1. compute the convolution a*a  (length 2n‑1)
            conv = fft_convolve(a, a)
            # 2. maximum of the convolution – the quantity we try to shrink
            M = conv.max()

            # ---------------------------------------------------------
            # 3. smooth approximation of the max (log‑sum‑exp)
            #    w_k ∝ exp(beta * (conv_k – M)),  sum w = 1
            #    when beta is large the weight concentrates on the true
            #    maximisers, when beta is small we get a smoother landscape.
            #    We shift by M for numerical stability.
            w = np.exp(beta * (conv - M))
            ws = w.sum()
            if ws == 0 or not np.isfinite(ws):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / ws

            # ---------------------------------------------------------
            # 4. gradient of the smoothed objective:
            #    ∂/∂a_i  Σ_k w_k (a ⋆ a)_k = 2 Σ_k w_k a_{k‑i}
            #    The sum can be written as a convolution of a with w reversed.
            w_rev = w[::-1]
            grad_full = fft_convolve(a, w_rev)          # length 3n‑2
            idx = 2 * n - 2 - np.arange(n)              # positions that give the needed shift
            grad = 2.0 * grad_full[idx]

            # ---------------------------------------------------------
            # 5. exponentiated‑gradient step (keeps the vector non‑negative)
            #    a_i ← a_i * exp(‑η * grad_i)  then re‑normalise.
            #    Clip the gradient to avoid overflow in the exponential.
            grad = np.clip(grad, -10.0, 10.0)
            a = a * np.exp(-lr * grad)
            s = a.sum()
            if s == 0 or not np.isfinite(s):
                a = np.random.rand(n)
                s = a.sum()
            a = a / s

            # ---------------------------------------------------------
            # 6. evaluate and keep the best solution seen so far
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] new best {best_score:.6f} (iter {it})")
                # if we already beat 1.6 we can tighten the step‑size
                if best_score < 1.6:
                    lr = max(lr * 0.9, 0.01)
                    beta = min(beta * 1.1, 5.0)

            # ---------------------------------------------------------
            # 7. schedule the annealing of hyper‑parameters
            it += 1
            inner_iters += 1
            if it % 1000 == 0:
                lr = max(lr * 0.95, 0.01)
                beta = min(beta * 1.02, 10.0)

        restart_counter += 1
        # occasionally print progress
        if restart_counter % 5 == 0:
            print(f"[{time.time():.1f}] restart {restart_counter}, best {best_score:.6f}")

    print(f"Finished – best value {best_score:.6f}")
    return best_seq

def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:

    The returned array has length len(x)+len(y)-1.
    """

    Y = np.fft.rfft(y, size)

            w_rev = w[::-1]
            grad_full = _fft_convolve(a, w_rev)               # length 3n‑2
            idx = 2 * n - 2 - np.arange(n)                    # pick the needed slice
            grad = 2.0 * grad_full[idx]

            # 5) exponentiated‑gradient step (keeps all entries ≥ 0).
            #    Clip the raw gradient to avoid overflow in the exponential.
            grad = np.clip(grad, -10.0, 10.0)
            a = a * np.exp(-lr * grad)

            # 6) re‑project onto the simplex (non‑negative & sum = 1).
            s = a.sum()
            if s == 0 or not np.isfinite(s):
                a = np.random.rand(n)
                s = a.sum()
            a /= s

            # 7) evaluate the candidate; keep the best ever seen.
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.tolist()
                print(f"[{time.time():.1f}] new best {best_score:.6f} (it {it})")
                # Tighten parameters once we are below a nice threshold.
                if best_score < 1.6:
                    lr = max(lr * 0.9, 0.01)
                    beta = min(beta * 1.1, 5.0)

            # 8) simple annealing of hyper‑parameters.
            it += 1
            inner_iters += 1
            if it % 1000 == 0:
                lr = max(lr * 0.95, 0.01)
                beta = min(beta * 1.02, 10.0)

        # ------------------------------------------------------------------
        # End of one restart.
        restart_counter += 1
        if restart_counter % 5 == 0:
            print(f"[{time.time():.1f}] restart {restart_counter}, best {best_score:.6f}")

    print(f"Finished – best value {best_score:.6f}")
    return best_seq
