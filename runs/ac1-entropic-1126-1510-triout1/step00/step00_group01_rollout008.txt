    # ensure v is numpy array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # Find rho = max i where condition holds
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected sub‑gradient descent for minimizing
      2 * n * max(conv(a,a)) / (sum(a)**2)

The function is scale‑invariant, so we keep the ℓ1‑norm fixed:
      sum(a) = sqrt(2*n)

We minimise a smooth approximation of the max‑operator:
      L(a) = (1/β)·log( Σ_k exp(β·(a*a)_k) )
Its sub‑gradient is
      g_i = 2· Σ_k exp(β·(a*a)_k)·a_{k-i}
      (indices out of range are ignored)

The iteration is
    a ← a – η·g                     (η decreasing)
    a ← proj_{simplex}(a)           (non‑negative, fixed ℓ1‑norm)

Convolutions are performed with FFT (real‑valued) for speed.
A small amount of random noise is added periodically to avoid
stagnation.  The best sequence found before the time budget
expires is returned.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def _conv_self_fft(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a*a using FFT of given size."""
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv_full = np.fft.irfft(B, n=size)[: 2 * len(a) - 1]
    return conv_full


def _conv_cross_fft(u: np.ndarray, v: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution u*v using FFT of given size."""
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    C = np.fft.irfft(U * V, n=size)
    # length = len(u)+len(v)-1
    return C[: len(u) + len(v) - 1]


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that minimises
    the provided evaluate_sequence function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the caller deducts a small safety margin).

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Parameters (feel free to tweak if you have spare time)
    # ------------------------------------------------------------------
    n = 5000                         # length of the sequence
    S = math.sqrt(2.0 * n)           # fixed ℓ1‑norm (sum of coefficients)
    beta = 5.0                       # smooth‑max temperature
    base_step = 0.5 * S / n          # initial step size factor
    noise_interval = 5000           # iterations between tiny random perturbations
    max_iters = 2_000_000            # hard limit (the time limit will stop earlier)
    # ------------------------------------------------------------------

    # Pre‑compute the FFT size large enough for all convolutions:
    # we need size ≥ (len(exp_beta)+len(a)-1) = (2n-1)+n-1 = 3n-2
    fft_size = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # Initial point – random positive vector scaled to the correct sum.
    # ------------------------------------------------------------------
    a = np.random.rand(n)
    a = a / a.sum() * S

    best_score = float("inf")
    best_a = a.copy()

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a tiny safety margin

    iteration = 0
    last_report = start_time

    while time.time() < deadline and iteration < max_iters:
        # --------------------------------------------------------------
        # 1) current self‑convolution and objective value
        # --------------------------------------------------------------
        conv_full = _conv_self_fft(a, fft_size)      # length 2n-1
        max_b = conv_full.max()
        # With the ℓ1‑norm already fixed (S² = 2n) the evaluation equals max_b.
        score = max_b

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[{iteration:08d}] New best score: {best_score:.6f}")

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        exp_beta = np.exp(beta * conv_full)          # positive weights

        # --------------------------------------------------------------
        # 3) step size (decreasing) and projected update
        # --------------------------------------------------------------
        eta = base_step / math.sqrt(iteration + 1.0)
        a_new = a - eta * g
