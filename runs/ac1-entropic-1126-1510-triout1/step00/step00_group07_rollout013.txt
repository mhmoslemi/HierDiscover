def conv_fft(x, y):
    N = len(x) + len(y) - 1
    M = 1 << (N-1).bit_length()
    X = np.fft.rfft(x, n=M)
    Y = np.fft.rfft(y, n=M)
    z = np.fft.irfft(X*Y, n=M)[:N]
    return z
```

import numpy as np
import time

def linear_convolution_fft(x, y):
    """Linear convolution using real FFT."""
    n = len(x)
    m = len(y)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:size]
    return conv

def propose_candidate(seed=0, budget_s=1000):
    """
    Heuristic search for a non‑negative step‑function that minimizes
    ``evaluate_sequence``.  The algorithm repeatedly builds a random
    candidate, then performs a simple mass‑redistribution loop that
    reduces the largest convolution entry.  The loop moves mass from the
    pair of indices that currently realise the maximal convolution value
    to an index that does not participate in that value, keeping the
    total sum constant.  Random restarts, decaying step size and occasional
    random perturbations are used to escape local minima.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # keep the best solution found
    best_seq = None
    best_score = float('inf')
    # Choose a few candidate lengths up to about 4000
    possible_lengths = [256, 512, 1024, 2048, 3937]  # include known length
    # Main loop
    while time.time() < deadline:
        # pick length
        n = int(np.random.choice(possible_lengths))
        # initialise a random non‑negative vector, biasing edges a little
        a = np.random.exponential(scale=1.0, size=n)
        # optional edge bias: multiply by (1/(i+1)+1/(n-i))
        idx = np.arange(n)
        bias = 1.0/(idx+1) + 1.0/(n-idx)
        a = a * bias
        # normalise to sum 1 (scale invariant for the objective)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum
        # keep a copy of the current best for this restart
        cur_best_score = evaluate_sequence(a.tolist())
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_seq = a.tolist()
        step = 0.5   # initial fraction of mass moved each iteration
        # inner optimisation loop
        for it in range(800):
            if time.time() >= deadline:
                break
            # compute full convolution
            b = linear_convolution_fft(a, a)
            # find index of maximal entry
            k_max = int(np.argmax(b))
            max_val = b[k_max]
            # compute contributions of each a_i to this maximal entry
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contrib = a[i_range] * a[j_range]
            # index with largest contribution (the one we want to reduce)
            idx_high = i_range[int(np.argmax(contrib))]
            # pick an index that does NOT participate in the maximal sum
            # i.e. any index outside [i_start, i_end]
            if i_start > 0:
                low_candidates = np.arange(0, i_start)
            else:
                low_candidates = np.arange(i_end + 1, n)
            if low_candidates.size == 0:
                # fallback: choose a random index different from idx_high
                low_candidates = np.arange(n)
                low_candidates = low_candidates[low_candidates != idx_high]
            idx_low = int(np.random.choice(low_candidates))
            # move a small amount of mass from high to low
            delta = step * a[idx_high]
            if delta <= 0:
                # nothing to move, break
                break
            a[idx_high] -= delta
            a[idx_low] += delta
            # re‑normalise to keep sum exactly 1 (helps numerical stability)
            a = a / a.sum()
            # evaluate every few iterations
            if it % 20 == 0:

                if score < best_score:

    # if nothing found (should not happen) fallback to a simple uniform vector

import numpy as np
import time

# ------------------------------------------------------------
# Helper: fast linear convolution using real FFT
# ------------------------------------------------------------
def conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two 1‑D real arrays using an FFT of length
    a power of two.  The result length is len(x)+len(y)-1.
    """
    n = len(x)
    m = len(y)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=N)
    Y = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(X * Y, n=N)[:size]
    return conv

# Main entry point – the search routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Heuristic search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.
    
    The algorithm works as follows:
    
    1.  Repeatedly initialise a random candidate vector `a` (length from a
        small set, normalised to sum 1).  The initial draw is biased
        towards larger values at the two ends – this helps to break the
        strong centre peak of the uniform distribution.
    
    2.  Inside a short inner loop a *mass‑redistribution* step is applied.
        The index `k_max` where the convolution `b = a * a` attains its
        maximum is located.  Every entry `a_i` contributes a term
        `a_i * a_{k_max-i}` to this maximal value (when the index is
        inside the admissible range).  The algorithm moves a small
        fraction of mass from the index with the **largest contribution**
        to the index with the **smallest overall weight** (outside the
        contributing range whenever possible).  After the transfer the
        vector is re‑normalised – the objective is scale‑invariant, so the
        normalisation merely stabilises the numerics.
    
    3.  Every few iterations the current candidate is evaluated with
        ``evaluate_sequence``.  If it improves the best score seen so far,
        the best vector is stored and a tiny progress message is printed.
    
    4.  The inner loop uses a decaying step size; occasional random
        perturbations (Gaussian noise) are injected to avoid stagnation.
        Several independent restarts are performed until the allotted
        time (``budget_s``) expires.
    
    The method is simple, requires only NumPy and the provided
    ``evaluate_sequence`` function, and typically drives the maximal
    convolution entry well below the trivial bound of 2.0 for sequences
    of a few thousand entries.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # leave a tiny margin

    # best solution seen so far
    best_seq: list[float] | None = None
    best_score = float('inf')

    # lengths that are cheap enough to handle but large enough to improve the bound
    possible_lengths = [256, 512, 1024, 2048, 3937]

    # ------------------------------------------------------------------
    # Main outer loop – keep trying new starts until time runs out
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1.  Choose a length and create a biased random start
        # --------------------------------------------------------------
        n = int(np.random.choice(possible_lengths))

        # exponential draw gives a reasonably diverse shape
        a = np.random.exponential(scale=1.0, size=n)

        # bias the ends slightly (edges a little larger than the centre)
        idx = np.arange(n)
        bias = 1.0 / (idx + 1) + 1.0 / (n - idx)   # larger at both ends
        a *= bias

        # normalise – the objective is homogeneous, any positive scaling works
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a = a / a_sum

        # store the current best for this restart (may be useful for early exit)
        cur_best_score = evaluate_sequence(a.tolist())
        if cur_best_score < best_score:
            best_score = cur_best_score
            best_seq = a.tolist()
            print(f"Init improvement: score={best_score:.6f}  (n={n})")

        # ------------------------------------------------------------------
        # 2.  Inner optimisation – mass‑redistribution loop
        # ------------------------------------------------------------------
        step = 0.5                # fraction of mass moved each iteration
        for it in range(800):     # a few hundred cheap iterations are enough
            if time.time() >= deadline:
                break

            # convolution a*a (full linear convolution)
            b = conv_fft(a, a)

            # index of the current maximal convolution entry
            k_max = int(np.argmax(b))

            # --------------------------------------------------------------
            #   Contributions of the individual a_i to that maximal value
            # --------------------------------------------------------------
            i_start = max(0, k_max - (n - 1))
            i_end   = min(n - 1, k_max)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k_max - i_range
            contrib = a[i_range] * a[j_range]

            # index that contributes the most (the "heavy" side)
            idx_high = int(i_range[int(np.argmax(contrib))])

            # choose a "light" side – preferably an index that does NOT
            # participate in this maximal sum (contribution = 0)
            if i_start > 0:
                low_candidates = np.arange(0, i_start)
            else:
                low_candidates = np.arange(i_end + 1, n)
            if low_candidates.size == 0:
                # fallback: pick the globally smallest entry (different from idx_high)
                low_candidates = np.argsort(a)
                low_candidates = low_candidates[low_candidates != idx_high]
            idx_low = int(np.random.choice(low_candidates))

            # --------------------------------------------------------------
            #   Move a fraction of mass from the heavy to the light index
            # --------------------------------------------------------------
            delta = step * a[idx_high]
            if delta <= 0.0:
                break
            a[idx_high] -= delta
            a[idx_low] += delta

            # small numerical drift may appear – enforce a proper normalisation
            a = a / a.sum()

            # --------------------------------------------------------------
            #   Periodically evaluate and adapt the step size
            # --------------------------------------------------------------
            if it % 20 == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.tolist()
                    print(f"New best {best_score:.6f} (n={n}, it={it})")
                # adapt step – if we are not improving, shrink it slowly
                if score >= cur_best_score:
                    step *= 0.94
                else:
                    cur_best_score = score
                # early exit if we are already close to the target constant
                if best_score < 1.55:
                    print("Target vicinity reached – returning early.")
                    return best_seq

            # --------------------------------------------------------------
            #   Occasionally inject a tiny random perturbation to escape plateaus
            # --------------------------------------------------------------
            if it % 100 == 0 and np.random.rand() < 0.1:
                a += np.random.randn(n) * 0.005
                a = np.clip(a, 0, None)
                a = a / a.sum()

    # ----------------------------------------------------------------------
    # If nothing better than the fallback was found, return a simple uniform
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 100
    return best_seq
