def simplex_projection(v, s=1.0):
    """Euclidean projection of v onto the simplex {x>=0, sum x = s}."""
    n = len(v)
    if s < 0:
        raise ValueError("s must be non-negative")
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # find rho
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all zero?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_max_convolution(a, k):
    n = len(a)
    # vector of indices j = k - i
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    # For each i where mask true set gradient accordingly
    # start with 2*a[j]
    g[mask] = 2 * a[j[mask]]
    # For positions where i == j (i.e., 2*i == k)
    eq_mask = mask & (i == j)
    if np.any(eq_mask):
        # Replace 2*a[i] with a[i]
        g[eq_mask] = a[i[eq_mask]]
    return g

Note that for positions not mask (i.e., k-i out of range), gradient remains zero, i.e., a_i has no effect on conv at shift where i not pairing.

"""
search for a non‑negative step‑function (list of heights) that minimizes
evaluate_sequence().  The objective after normalising the L1‑norm to
sqrt(2·n) is simply the maximum entry of the self‑convolution.
A simple projected‑gradient descent is used:  at each iteration the
gradient of the currently maximal convolution entry is taken, a step is
applied and the vector is projected back onto the simplex
{a_i ≥ 0 , sum a_i = sqrt(2·n)}.  Several different initialisations
(constant, geometric, random) are tried within the time budget and the
best result is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x) = s}
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of size s."""
    if s < 0:
        raise ValueError("Target simplex size must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond].max() - 1          # zero‑based index
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the objective (equivalent to evaluate_sequence)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)**2).  Handles degenerate cases."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-8:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Gradient of the *active* convolution term (sub‑gradient of the max)
def _max_convolution_gradient(a: np.ndarray) -> (np.ndarray, int):
    """Return the sub‑gradient of max(conv(a,a)) and the index where the max occurs."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))
    i = np.arange(n, dtype=int)
    j = k - i
    mask = (j >= 0) & (j < n)

    grad = np.zeros_like(a)
    # generic case: each ordered pair contributes twice
    grad[mask] = 2.0 * a[j[mask]]
    # diagonal entries (i == j) appear only once
    diag = mask & (i == j)
    if np.any(diag):
        grad[diag] = a[i[diag]]
    return grad, k


# ----------------------------------------------------------------------
# One optimisation run (projected gradient descent) starting from `start`
def _optimise(start: np.ndarray,
              deadline: float,
              verbose: bool = False) -> (np.ndarray, float):
    """Return a locally improved sequence and its objective."""
    n = start.shape[0]
    target_sum = np.sqrt(2.0 * n)

    # normalise the start vector onto the simplex
    a = start.astype(np.float64)
    a = _simplex_projection(a, target_sum)

    best_a = a.copy()
    best_val = _objective(a)

    # initial step size proportional to the typical entry size
    step = 0.5 * target_sum / n   # ≈ 0.5·average height

    if verbose:
        print(f"  start objective = {best_val:.8f}")

    while time.time() < deadline:
        grad, _ = _max_convolution_gradient(a)

        # tentative update
        a_new = a - step * grad
        a_new = _simplex_projection(a_new, target_sum)

        val = _objective(a_new)

        if val < best_val - 1e-12:            # improvement
            best_val = val
            best_a = a_new.copy()
            a = a_new
            step *= 1.1                       # be a little more aggressive
            if verbose:
                print(f"    improved to {best_val:.8f} (step={step:.6f})")
        else:
            # no progress – shrink step and occasionally jitter to escape plateaux
            step *= 0.5
            if step < 1e-12:
                break
            if np.random.rand() < 0.05:
                jitter = np.random.randn(n) * step * 0.5
                a = _simplex_projection(a + jitter, target_sum)

    return best_a, best_val
