def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Determine n
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000
    target_sum = math.sqrt(2.0 * n)
    best_seq = None
    best_score = float('inf')
    # families
    exponents = [0, 0.5, 1, 1.5, 2, 2.5, 3, 4]
    for p in exponents:
        raw = np.arange(1, n+1, dtype=np.float64) ** p
        scale = target_sum / raw.sum()
        seq = (raw * scale).tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f'Init p={p}, score={score}')
    # also try reversed shape (decreasing)
    for p in exponents:
        raw = np.arange(n, 0, -1, dtype=np.float64) ** p
        scale = target_sum / raw.sum()
        seq = (raw * scale).tolist()
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f'Init reversed p={p}, score={score}')
    # random family: raw = (i+1)**p * (n-i)**q
    for p in [1,2,3]:
        for q in [0,1]:
            raw = (np.arange(1, n+1) ** p) * ((np.arange(n,0,-1)) ** q)
            scale = target_sum / raw.sum()
            seq = (raw * scale).tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f'Init p={p},q={q}, score={score}')
    # do local hill climbing to improve further
    seq = np.array(best_seq, dtype=np.float64)
    # precompute gradient function
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(seq, seq)
        max_idx = np.argmax(conv)
        # compute gradient of max element
        grad = np.zeros_like(seq)
        # Use vectorized approach: for i where j = max_idx - i in range
        indices = np.arange(len(seq))
        j = max_idx - indices
        mask = (j >= 0) & (j < len(seq))
        grad[mask] = 2.0 * seq[j[mask]]
        # move opposite direction
        eta = 0.02  # step size (relative)
        new_seq = seq - eta * grad
        # enforce non-negativity
        new_seq = np.maximum(new_seq, 0.0)
        # renormalize sum
        s = new_seq.sum()
        if s == 0:
            continue
        new_seq *= target_sum / s
        # compute score
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.tolist()
            seq = new_seq
            print(f'Iter improvement: score={score:.6f}')
        else:
            # maybe random perturb to escape
            if np.random.rand() < 0.01:
                # random local jitter on random index
                idx = np.random.randint(n)
                new_seq[idx] += (np.random.rand() - 0.5) * 0.1 * new_seq[idx]
                new_seq = np.maximum(new_seq, 0.0)
                new_seq *= target_sum / new_seq.sum()
                seq = new_seq
    return best_seq

"""
Optimiser for the step‑function evaluation metric.

The evaluation 2 * n * max(conv) / (sum(a)^2) is scale invariant:
for any positive factor α, both the numerator (quadratic in a) and the
denominator (quadratic in the sum) scale by α² and cancel.
Hence we may normalise the sequence so that sum(a) = √(2·n) – this is
exactly the normalisation used by the reference implementation.

The maximum of the (non‑circular) convolution of a sequence with itself
tends to be attained at the “central” index (shift n‑1) and equals
∑ a_i·a_{n‑1‑i}.  For a monotone increasing shape a_i = (i+1)^p
the integral limit of the normalised value is

    score(p) = 2·(p+1)²·B(p+1, p+1),

where B is the Beta function.  This quantity decreases quickly with p:
p=1 → 4/3 ≈ 1.33, p=2 → 0.60, p=3 → 0.23, …  Thus even a simple
polynomial law yields scores far below the target 1.503.

The algorithm:
1. Choose the length n (use the provided `height_sequence_1` if
   available, otherwise a default ~5000).
2. Scan a small set of exponents p (including fractional values),
   build the raw sequence a_i = (i+1)^p, scale it to sum = √(2·n) and
   keep the best sequence.
3. Optionally scan the reversed (decreasing) shape.
4. Perform a cheap hill‑climbing step that reduces the currently
   highest convolution entry – the gradient of the max entry is
   g_i = 2·a_{k‑i} (zero when the partner index is out of range).
   Move a little opposite to this gradient, re‑normalise, and accept
   the move if the score improves.
5. Return the best sequence found before the time budget expires.

The method is deterministic apart from a tiny random jitter used to
escape plateaus and runs comfortably within the 1000 s limit.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Return a sequence of non‑negative heights with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for the occasional jitter.
    budget_s : float
        Total time budget in seconds (the function will stop ~5 s
        before the deadline to avoid a forced termination).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Determine the length of the sequence.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000                            # fallback length

    target_sum = math.sqrt(2.0 * n)          # the normalisation used by SOTA
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 2. Scan a few simple power‑law shapes (increasing and decreasing).
    # ------------------------------------------------------------------
    exponents = [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0]  # include 0 → uniform for sanity

    for p in exponents:
        # increasing shape a_i = (i+1)^p
        raw = np.arange(1, n + 1, dtype=np.float64) ** p
        seq = (raw * (target_sum / raw.sum())).tolist()
        sc = evaluate_sequence(seq)    # evaluate_sequence is supplied by the framework
        print(f'init p={p:4.1f}  score={sc:.6f}')
        if sc < best_score:
            best_score, best_seq = sc, seq

    for p in exponents:
        # decreasing shape a_i = (n-i)^p
        raw = np.arange(n, 0, -1, dtype=np.float64) ** p
        seq = (raw * (target_sum / raw.sum())).tolist()
        sc = evaluate_sequence(seq)
        print(f'init rev p={p:4.1f}  score={sc:.6f}')
        if sc < best_score:
            best_score, best_seq = sc, seq

    # ------------------------------------------------------------------
    # 3. Simple hill‑climbing on the worst convolution entry.
    #    This step is cheap (one convolution per iteration) and can
    #    reduce the score a little further.
    # ------------------------------------------------------------------
    seq_arr = np.array(best_seq, dtype=np.float64)

    # a tiny constant step size; it is scaled automatically by the
    # normalisation, so we use an absolute step that works well for
    # the typical magnitude of the entries (~O(1/√n)).
    step_factor = 0.01 * target_sum

    while time.time() < deadline:
        # compute the convolution and locate the maximal entry
        conv = np.convolve(seq_arr, seq_arr)
        k_max = int(np.argmax(conv))

        # gradient of the max entry: d/d a_i  (a*a)[k_max] = 2 * a_{k_max-i}
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        mask = (j_idx >= 0) & (j_idx < n)

        grad = np.zeros_like(seq_arr)
        grad[mask] = 2.0 * seq_arr[j_idx[mask]]

        # take a small step opposite to the gradient
        new_seq = seq_arr - step_factor * grad
        new_seq = np.maximum(new_seq, 0.0)          # enforce non‑negativity
        s = new_seq.sum()
        if s <= 0.0:                               # degenerate – give up this step
            break
        new_seq *= (target_sum / s)                # re‑normalise to the required sum

        # evaluate the new candidate
        new_score = evaluate_sequence(new_seq.tolist())

        if new_score < best_score:
            best_score, best_seq = new_score, new_seq.tolist()
            seq_arr = new_seq
            print(f'iter improvement: score={new_score:.6f}')
            # occasionally increase step size after a successful move
            step_factor *= 1.02
        else:
            # if no improvement, shrink the step size and occasionally apply a random jitter
            step_factor *= 0.95
            if np.random.rand() < 0.02:
                idx = np.random.randint(n)
                jitter = (np.random.rand() - 0.5) * 0.1 * seq_arr[idx] + np.random.randn() * 1e-4
                seq_arr[idx] = max(0.0, seq_arr[idx] + jitter)
                seq_arr *= target_sum / seq_arr.sum()
    # ------------------------------------------------------------------
    # Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f'final best score = {best_score:.6f}')
    return best_seq
