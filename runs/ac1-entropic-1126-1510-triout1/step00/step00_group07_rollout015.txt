def build_sidon_set(n, m_target, max_trials=200):
    positions = list(range(n))
    for _ in range(max_trials):
        random.shuffle(positions)
        s = []
        sum_set = set()
        for p in positions:
            # compute sums with existing + self
            conflict = False
            for q in s:
                if p + q in sum_set:
                    conflict = True
                    break
            if not conflict and (p + p) not in sum_set:
                # add p
                s.append(p)
                # add new sums
                sum_set.add(p + p)
                for q in s[:-1]:
                    sum_set.add(p + q)
                # if reached target, break
                if len(s) == m_target:
                    return s
        # try again
    return None

"""
Propose a step‑function (list of non‑negative heights) with a low
evaluate_sequence() score.

The idea:
* The objective 2 n·max(conv)/sum² is minimised when the auto‑convolution
  of the sequence is as flat as possible.
* A classic construction for a flat auto‑convolution is to take a
  Sidon (B₂) set – a set of indices whose pairwise sums are all distinct.
  If the large heights are placed on a Sidon set and the remaining
  positions contain a common small height, then each entry of the
  convolution consists of at most one “large–large’’ product, at most
  |S| large–small products and at most n small–small products.
* For a given sequence length n and Sidon‑set size m the optimal ratio
  between the large and the small height is obtained analytically:
        r = a / b = m·(n+m) / ( 2·(n‑m) – m² )
  (large = r·small).  This is the stationary point of the exact objective.
* We try to find the largest possible Sidon set (up to the theoretical
  limit √(2 n) ≈ 88 for n≈4000) by a simple randomized greedy algorithm.
  For each feasible size m we build the sequence with the optimal ratio,
  evaluate it and keep the best one.  A small local‑search around the
  optimal ratio refines the solution.
* If time permits we finish with a cheap simulated‑annealing style
  neighbourhood search that moves a tiny amount of mass between two
  positions while preserving non‑negativity.

The whole procedure respects the 1000‑second budget, uses only numpy,
random and the provided evaluate_sequence, and returns the best sequence
found.
"""

import time
import random
import math
import numpy as np

def _find_sidon_set(n, m_target, max_trials=200):
    """Return a list of m_target indices forming a Sidon set (all pairwise sums unique).
    If not found after max_trials attempts, return None."""
    for _ in range(max_trials):
        # Random order of candidates
        candidates = list(range(n))
        random.shuffle(candidates)
        sidon = []
        sums = set()
        for p in candidates:
            # quick reject if self‑sum already used
            if (p + p) in sums:
                continue
            conflict = False
            for q in sidon:
                if (p + q) in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # accept p
            for q in sidon:
                sums.add(p + q)
            sums.add(p + p)
            sidon.append(p)
            if len(sidon) >= m_target:
                return sidon
    return None


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Return a list of heights (non‑negative floats) with a low evaluate_sequence score."""
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # a little margin for safety

    # ------------------------------------------------------------------
    # 1. Determine the length of the problem (use the provided sequence if any)
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        base_seq = list(height_sequence_1)  # fallback if everything else fails
    except Exception:
        n = 4000                     # reasonable default
        base_seq = [1.0] * n

    # initial best candidate (uniform, score = 2.0)
    best_seq = base_seq.copy()
    best_score = float(evaluate_sequence(best_seq))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Theoretical maximal Sidon size for given n (from 2(n‑m) > m²)
    max_possible_m = int(math.sqrt(2 * n + 1)) - 1
    max_possible_m = min(max_possible_m, n)  # safety
    # we start a few steps below the theoretical maximum to keep search cheap
    start_m = max_possible_m
    low_m = max(2, int(math.sqrt(n)) - 5)

    # ------------------------------------------------------------------
    # 3. Search for large Sidon sets, decreasing size until we find a good one
    for m in range(start_m, low_m - 1, -1):
        if time.time() > deadline:
            break
        # try a handful of random builds
        sidon = _find_sidon_set(n, m, max_trials=150)
        if sidon is None:
            continue
        # --------------------------------------------------------------
        # 4. Compute the optimal large/small ratio for this m
        D = 2 * (n - m) - m * m
        if D <= 0:

        # r = a / b where a is the large height, b the small height
        r_opt = (m * (n + m)) / D

        # Two equivalent parametrisations; we use small=1, large=r_opt
        small = 1.0
        large = r_opt

        # --------------------------------------------------------------
        # 5. Build the candidate sequence and evaluate
        seq = [small] * n
        for idx in sidon:
            seq[idx] = large

        score = evaluate_sequence(seq)
        print(f"[sidon] m={m}, r={large:.4f}, score={score:.6f}")
        if score < best_score:
            best_score, best_seq = score, seq.copy()

        # --------------------------------------------------------------

        for factor in (0.90, 0.95, 1.05, 1.10):
            if time.time() > deadline:

            lg = large * factor
            seq_adj = [small] * n

                seq_adj[idx] = lg

            if sc < best_score:

        # --------------------------------------------------------------

        if best_score <= 1.503:

    # ------------------------------------------------------------------

    print(f"[final] best score={best_score:.6f}")
    return best_seq
