"""
Gradient‑based local search for minimizing

    2·n·max(conv(a,a)) / (sum(a))²

where `a` is a non‑negative step‑function (list of heights).
The algorithm:

*   Samples many lengths and many random Dirichlet initialisations
    (highly non‑uniform seeds encourage asymmetry).
*   Performs a simple sub‑gradient descent on the objective.
    At each iteration the index `k` where the convolution attains its
    maximum is found; the gradient of that particular convolution entry
    w.r.t. the coefficients is known analytically and is used to move
    the vector in a direction that reduces the maximal value.
*   After each step the vector is projected onto the probability simplex
    (non‑negative and normalised to sum = 1) – scaling does not change the
    objective, but the projection keeps the iterates feasible.
*   The best sequence seen (lowest evaluation score) is stored.
*   The search runs until the time budget finishes and finally returns
    the best sequence discovered.

This method explores a large variety of shapes (asymmetric, “U‑shaped”,
spiky, etc.) without relying on the LP routine used in the reference
implementation, and typically finds scores well below the previous bound
of 2.0 within a few seconds.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _objective_value(seq: np.ndarray) -> float:
    """
    Compute the exact objective returned by ``evaluate_sequence`` (without
    the safety clamping).  This is cheaper than calling the external
    function many times during the search.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_conv = conv.max()
    # sum(seq) is positive (the simplex projection guarantees it)
    return 2.0 * n * max_conv / (seq.sum() ** 2)


def _max_conv_index(seq: np.ndarray) -> int:
    """Return an index k where the convolution attains its maximal value."""
    conv = np.convolve(seq, seq)
    return int(conv.argmax())


def _subgradient_max_conv(seq: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry w.r.t. ``seq``.
    For a given shift ``k`` we have
        (seq*seq)[k] = Σ_i seq[i] * seq[k-i] .
    The derivative w.r.t. seq[i] is
        2·seq[k-i]  when i ≠ k-i,
        seq[i]      when i = k-i.
    """
    n = len(seq)
    grad = np.zeros_like(seq)

    # i ranges over indices that actually appear in the sum for this k
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)

    i_range = np.arange(i_start, i_end + 1)
    j_range = k_max - i_range                     # j = k_max - i

    # mask where i == j (only possible when k_max is even)
    eq_mask = (i_range == j_range)

    # generic case: i != j  → derivative 2·seq[j]
    grad[i_range] = 2.0 * seq[j_range]
    # correct the diagonal entries
    grad[i_range[eq_mask]] = seq[i_range[eq_mask]]

    return grad


def _project_onto_simplex(x: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { y | y ≥ 0 , Σ y = 1 }.
    Implements the algorithm from
    Duchi et al., “Efficient Projections onto the l1‑Ball for
    Learning in High Dimensions”, ICML 2008.
    """
    if x.sum() == 1.0 and np.all(x >= 0):
        return x
    # shift to make the whole vector non‑negative after projection
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(x)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(x - theta, 0)
    # In the extremely unlikely case of numerical issues, renormalise
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(x, 1.0 / len(x))
    else:
        w /= w_sum
    return w

# ----------------------------------------------------------------------

    try:
        candidate = list(height_sequence_1)  # type: ignore[name‑defined]
        score = _objective_value(np.array(candidate, dtype=float))
        best_seq = candidate
        best_score = score
        print(f"Baseline from height_sequence_1: score = {best_score:.6f}")
    except Exception:
        # No baseline available – start from scratch.
        pass

    # Lengths explored – we keep them modest to stay fast while still
    # allowing the algorithm to discover the benefit of larger n.
    lengths = [30, 50, 80, 120, 200, 400, 800, 1600, 3200]

    # ------------------------------------------------------------------
    # Main stochastic search loop.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --------------------------------------------------------------
        # Pick a random length and a random Dirichlet concentration.
        # Small concentration (<1) yields spiky, highly asymmetric seeds,
        # which are essential for breaking symmetry.
        # --------------------------------------------------------------
        n = int(np.random.choice(lengths))
        concentration = float(np.random.choice([0.2, 0.5, 0.8, 1.0, 2.0]))
        seq = np.random.dirichlet(np.ones(n) * concentration)

        # --------------------------------------------------------------
        # Sub‑gradient descent on the maximum convolution entry.
        # The learning rate decays with the iteration number.
        # --------------------------------------------------------------
        for it in range(max_inner_iters):
            # Periodically check the time budget.
            if time.time() > deadline:
                break

            # Current objective value.
            current_score = _objective_value(seq)

            # Record a new best sequence.
            if current_score < best_score:
                best_score = current_score
                best_seq = seq.copy()
                print(
                    f"[{time.time():.1f}] New best = {best_score:.6f} "
                    f"(n={n}, it={it})"
                )

            # Identify the most offending shift and its sub‑gradient.
            k_max = _max_conv_index(seq)
            grad = _subgradient_max_conv(seq, k_max)

            # Simple diminishing step size.
            lr = 0.02 / (it + 1)

            # Gradient step, projection onto the simplex.
            seq = seq - lr * grad
            seq = np.maximum(seq, 0.0)
            seq = _project_onto_simplex(seq)

            # Occasionally inject a tiny random perturbation to escape
            # shallow plateaus.
            if (it + 1) % 25 == 0:
                noise = np.random.randn(n) * 0.004
                seq = np.maximum(seq + noise, 0.0)
                seq = _project_onto_simplex(seq)

        # End of inner optimisation for this random start.

    # ------------------------------------------------------------------
    # Fallback: if no sequence was ever accepted (should never happen),
    # return a simple uniform vector of length 10.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(10, 1.0 / 10.0)

    # Convert to a plain Python list of floats before returning.
    return [float(v) for v in best_seq]
