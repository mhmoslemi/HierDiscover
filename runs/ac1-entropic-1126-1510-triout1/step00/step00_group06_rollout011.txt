"""
Propose a good step‑function (a list of non‑negative heights).

The algorithm is a simple stochastic search:
* Randomly pick a support size k (500–1500 works well for n≈3100).
* Choose k distinct positions in the interval [0, n) uniformly at random.
* Initialise the heights on the support with equal weight (so the total sum is 1).
* Perform a few cheap gradient‑descent steps that move mass away from the
  index where the auto‑convolution attains its maximum.
* Keep the best sequence seen so far.

Because the objective is scale‑invariant, fixing the sum to 1 is
convenient and avoids the clipping performed inside `evaluate_sequence`.
The search is time‑bounded; it returns the best sequence found before the
deadline (≈ budget_s seconds).
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: one gradient descent refinement step
# ------------------------------------------------------------
def _refine_sequence(a: np.ndarray, max_iters: int = 30, lr: float = 0.5):
    """
    Perform a few projected gradient–descent steps on a non‑negative vector a.
    The objective is exactly the one computed by ``evaluate_sequence``.
    Returns the best vector seen during the refinement and its score.
    """
    n = a.shape[0]
    # make sure the vector sums to 1 (any positive constant works)
    if a.sum() == 0:
        return a, float("inf")
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    cur_a = a.copy()
    cur_score = best_score

    for _ in range(max_iters):
        # convolution and its maximum
        conv = np.convolve(cur_a, cur_a)
        Bmax = conv.max()
        k0 = conv.argmax()
        S = cur_a.sum()
        if S == 0:
            break

        # derivative of the maximal convolution w.r.t. each entry
        i = np.arange(n)
        j = k0 - i
        valid = (j >= 0) & (j < n)

        dB = np.zeros_like(cur_a)
        # ordered pairs (i,j) with i!=j contribute twice, i==j once
        dB[valid] = 2.0 * cur_a[j[valid]]
        same = valid & (i == j)
        dB[same] = cur_a[i[same]]          # correct the double‑count on the diagonal

        # gradient of the full objective  f = 2n * Bmax / S^2
        C = 2.0 * n
        grad = C / (S ** 3) * (S * dB - 2.0 * Bmax)

        # simple projected gradient step
        new_a = cur_a - lr * grad
        new_a[new_a < 0] = 0.0

        # renormalise to keep the scale fixed (objective is scale invariant)
        new_sum = new_a.sum()
        if new_sum > 0:
            new_a = new_a / new_sum

        # evaluate
        try:
            new_score = evaluate_sequence(new_a.tolist())
        except Exception:
            break

        # accept if we improved the current point
        if new_score < cur_score:
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()

    return best_a, best_score
