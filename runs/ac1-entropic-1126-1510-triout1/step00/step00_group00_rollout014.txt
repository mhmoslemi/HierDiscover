def project_simplex(v):
    # ensure nonnegativity and sum 1
    # Using algorithm: sort v descending, compute cumulative sum, find rho, compute theta.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_proj(v):
    # v: numpy array
    # Duchi et al. algorithm
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that yields a
low value of the supplied `evaluate_sequence` function.

The algorithm combines three simple ideas:

1. **Projected sub‑gradient descent** –  
   From the current sequence we identify the convolution index `k_max`
   where the maximum occurs. The sub‑gradient of `max(conv)` w.r.t. the
   coefficients is `g_i = 2 * a_{k_max-i}` (zero when the index is out of
   range). A small step in the opposite direction and a projection onto the
   probability simplex (non‑negative, sum = 1) reduces the peak of the
   convolution.

2. **Simulated annealing** –  
   The sub‑gradient step is accepted if it improves the score.
   Otherwise it may be accepted with a probability that depends on a
   temperature that cools linearly with the remaining time.  This helps
   escaping shallow local minima.

3. **Periodic LP refinement** –  
   Every few thousand iterations a linear program is solved:

        minimise   t
        subject to Σ_i a_i * g_{k-i} ≤ t   for all k
                    Σ_j g_j = Σ_i a_i
                    g_j ≥ 0

   The LP finds a new sequence `g` with the smallest possible maximal
   convolution under the current sum constraint.  If this LP solution
   improves the best score it replaces the current sequence.

The search runs until the allotted time (`budget_s`) is exhausted or a
target value (< 1.503) is reached.  The best sequence found is returned
as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0, Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise exactly to 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


def _solve_min_max_lp(f: np.ndarray) -> np.ndarray | None:
    """
    Solve the linear program:

        minimise   t
        subject to Σ_i f_i * g_{k-i} ≤ t   for all convolution indices k
                    Σ_j g_j = Σ_i f_i
                    g_j ≥ 0,  t ≥ 0

    Returns the optimal `g` (as a numpy array) or `None` if the LP fails.
    """
    n = f.shape[0]
    # Variables are (g_0,…,g_{n-1}, t)
    c = np.zeros(n + 1)
    c[-1] = 1.0                     # minimise t

    # Inequality constraints: one per convolution index (2n‑1 of them)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n + 1)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f[i]      # coefficient of g_j
        row[-1] = -1.0              # -t
        rows.append(row)
    A_ub = np.array(rows, dtype=float)
    b_ub = np.zeros(2 * n - 1, dtype=float)

    # Equality: keep the total mass unchanged
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([f.sum()], dtype=float)

    # Bounds: g_j ≥ 0, t ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Solve with HiGHS; limit the wall‑clock time to a few seconds
    options = {"disp": False, "time_limit": 5.0}
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs",
            options=options,
        )
    except Exception:          # pragma: no cover – defensive
        return None

    if not res.success:
        return None
    g = res.x[:n]
    return g
