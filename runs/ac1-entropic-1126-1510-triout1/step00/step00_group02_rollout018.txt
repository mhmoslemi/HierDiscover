def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))
```

"""
Propose a low evaluation sequence.

This implementation searches over short sequences (length up to 20) using a
combination of random restarts, simple hill‑climbing and a gradient‑like
update that pushes weight away from the convolution entry that attains the
current maximum.  The best sequence found is returned.  The search is
limited by the provided time budget (default 1000 s).  A handcrafted
Sidon‑set construction of length 7 (ratio 1.75) is also used as a reliable
starting point.
"""
import time
import random
import numpy as np

# evaluate_sequence is assumed to be imported in the execution environment.


def random_dirichlet(n):
    """Sample a random point on the simplex of dimension n."""
    x = np.random.gamma(1.0, size=n)
    s = x.sum()
    if s == 0:
        x = np.ones(n) / n
    else:
        x = x / s
    return x.tolist()


def sidon_len4():
    """
    Uniform Sidon set of size 4 on positions {0,1,4,6}.
    Length = 7, each weight = 1/4.
    """
    seq = [0.0] * 7
    for p in (0, 1, 4, 6):
        seq[p] = 0.25
    return seq


def gradient_step(seq, step):
    """
    One sub‑gradient step aiming to reduce the current maximal convolution
    entry.  The step size is a small positive number.
    """
    a = np.array(seq, dtype=float)
    n = a.size
    conv = np.convolve(a, a)
    k = int(conv.argmax())
    # sub‑gradient of max(conv) w.r.t. a_i is 2 * a_{k-i} when the index is valid
    grad = np.zeros_like(a)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    # take a step opposite to the gradient
    a -= step * grad
    # enforce non‑negativity
    a = np.maximum(a, 0.0)
    # renormalise to keep sum constant (scale invariance of the score)
    total = a.sum()
    if total < 1e-12:
        # fall back to uniform if everything vanished
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a.tolist()


def hill_climb(seq, iters=200, step=0.03):
    """
    Simple hill‑climbing: at each iteration move a small amount of mass
    from a random index to another random index.  Keep changes that improve
    the score.
    """
    best_seq = seq[:]
    best_score = evaluate_sequence(best_seq)
    n = len(best_seq)
    for _ in range(iters):
        i, j = random.sample(range(n), 2)
        # move a fraction of the mass from i to j
        amount = best_seq[i] * random.uniform(0.0, 0.3)
        if amount == 0.0:
            continue
        cand = best_seq[:]
        cand[i] -= amount
        cand[j] += amount
        # keep non‑negative (i should be non‑negative already)
        if cand[i] < 0.0:
            continue
        score = evaluate_sequence(cand)
        if score < best_score:
            best_seq, best_score = cand, score
    return best_seq


def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a sequence with low evaluation score.
    The function runs for at most `budget_s` seconds and returns the best
    sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # small safety margin
    best_seq = None
    best_score = float('inf')

    # deterministic Sidon‑set seed (ratio 1.75) – ensures we beat the initial 2.0
    candidate = sidon_len4()
    score = evaluate_sequence(candidate)
    if score < best_score:
        best_seq, best_score = candidate, score
        print(f"[init] Sidon seed score = {best_score:.6f}")

    # try various lengths; short lengths give the best chances for a low score
    lengths = [2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 15, 20]
    while time.time() < deadline:
        n = random.choice(lengths)

        # start from a random point on the simplex
        seq = random_dirichlet(n)

        # a few gradient descent steps to push mass away from the worst convolution entry
        for _ in range(8):
            seq = gradient_step(seq, step=0.04)

        # refinement with hill climbing
        seq = hill_climb(seq, iters=250, step=0.03)

        # final evaluation
        try:
            cur_score = evaluate_sequence(seq)
        except Exception:
            continue

        if cur_score < best_score:
            best_seq, best_score = seq, cur_score
            print(f"[update] new best (len={n}) score = {best_score:.6f}")

        # Occasionally re‑seed from the good Sidon construction to explore its neighbourhood
        if random.random() < 0.05:
            seq = sidon_len4()
            seq = hill_climb(seq, iters=150, step=0.02)
            cur_score = evaluate_sequence(seq)
            if cur_score < best_score:
                best_seq, best_score = seq, cur_score
                print(f"[sidon‑refine] improved score = {best_score:.6f}")

    if best_seq is None:
        # fallback: return the deterministic Sidon seed
        best_seq = sidon_len4()
        best_score = evaluate_sequence(best_seq)

    print(f"Best score after search: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
