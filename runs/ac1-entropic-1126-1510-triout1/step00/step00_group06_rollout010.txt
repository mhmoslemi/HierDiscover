def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    # Create sparse matrix
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        # j ranges
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f_seq[i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    b_ub = np.full(2*n - 1, rhs, dtype=float)
    bounds = [(0, None)] * n
    c = -np.ones(n)
    # solve
    result = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    else:
        return None

def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Fast convolution using real FFT (returns length len(x)+len(y)-1)."""
    n = len(x) + len(y) - 1
    # next power of two for efficient FFT
    L = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, L)
    fy = np.fft.rfft(y, L)
    conv = np.fft.irfft(fx * fy, L)[:n]
    return conv


def build_toeplitz_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n Toeplitz matrix T such that T·b = conv(f,b).
    Each column j contains the vector f shifted down by j rows.
    """
    n = len(f)
    rows = 2 * n - 1
    T = np.zeros((rows, n), dtype=float)
    for j in range(n):
        T[j:j + n, j] = f
    return T


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve   max  sum(b)   s.t.   conv(f_seq, b) ≤ rhs ,  b ≥ 0.

    Returns the optimal b as a 1‑D ndarray, or ``None`` if the LP fails.
    """
    n = len(f_seq)
    # Toeplitz matrix for the convolution constraints
    A_ub = build_toeplitz_matrix(f_seq)
    b_ub = np.full(A_ub.shape[0], rhs, dtype=float)

    # objective: maximize sum(b)  →  minimize -sum(b)
    c = -np.ones(n, dtype=float)

    # bounds for each variable: b_i ≥ 0
    bounds = [(0, None)] * n

    # use the default 'highs' method (fast and reliable)
    try:
        result = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                 bounds=bounds, method='highs')
    except Exception:
        return None

    if result.success:
        return result.x
    return None


def max_allowed_step(conv_ff: np.ndarray, conv_fg: np.ndarray,
                     conv_gg: np.ndarray, M: float) -> float:
    """
    For the mixture   f_new = (1‑t)·f + t·g
    we need   max_k conv(f_new,f_new)[k] ≤ M.

    For each k the inequality becomes:
        a2·t² + a1·t + a0 ≤ 0,
    with
        a0 = conv_ff[k] - M      (≤0),
        a1 = 2·(conv_fg[k] - conv_ff[k]),
        a2 = conv_ff[k] + conv_gg[k] - 2·conv_fg[k].

    The function returns the largest t∈[0,1] satisfying all inequalities.
    """
    t_max = 1.0
    a0 = conv_ff - M                     # ≤0
    a1 = 2.0 * (conv_fg - conv_ff)
    a2 = conv_ff + conv_gg - 2.0 * conv_fg

    # iterate over all k (vectorised root finding is cumbersome; a loop is fine)
    for k in range(len(conv_ff)):
        a0k = a0[k]
        a1k = a1[k]
        a2k = a2[k]

        # a0k ≤ 0 by construction
        if abs(a2k) < 1e-14:                     # effectively linear
            if a1k > 0:
                t_root = -a0k / a1k
                if t_root < t_max:
                    t_max = max(0.0, min(t_root, t_max))
            # a1k ≤ 0 already satisfies the inequality for all t≥0
            continue

        # quadratic case: solve a2·t² + a1·t + a0 = 0
        disc = a1k * a1k - 4.0 * a2k * a0k
        if disc <= 0:
            # No real crossing → inequality holds for all t≥0
            continue
        sqrt_disc = math.sqrt(disc)
        # two roots
        t1 = (-a1k - sqrt_disc) / (2.0 * a2k)
        t2 = (-a1k + sqrt_disc) / (2.0 * a2k)

        # we need the smallest positive root (first time the expression becomes >0)
        candidates = [t for t in (t1, t2) if t > 1e-12]
        if not candidates:
            continue
        t_root = min(candidates)
        if t_root < t_max:
            t_max = t_root
            if t_max <= 1e-12:
                break
    return t_max


def random_perturb(seq: np.ndarray) -> np.ndarray:
    """
    Small random redistribution of mass: move a tiny amount from one index
    to another while keeping the total sum unchanged.
    """
    n = len(seq)
    i, j = np.random.choice(n, size=2, replace=False)
    if seq[i] <= 0:
        return seq
    eps = np.random.rand() * 0.02 * seq[i]   # up to 2 % of the source entry
    seq[i] = max(0.0, seq[i] - eps)
    seq[j] += eps
    return seq


def maybe_shuffle(seq: np.ndarray, prob: float = 0.02) -> np.ndarray:
    """With a small probability, randomly permute the sequence."""
    if np.random.rand() < prob:
        np.random.shuffle(seq)
    return seq

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative sequence) that yields a low
    ``evaluate_sequence`` score.

    Returns the best sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # initialise the current sequence
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        seq = np.array(height_sequence_1, dtype=float)
    else:
        # fallback: random start, length 800‑1200
        length = np.random.randint(800, 1200)
        seq = np.random.rand(length)

    # normalise a bit (scaling does not affect the score, but avoids huge numbers)
    seq = np.clip(seq, 0.0, 1000.0)

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f'Initial score = {best_score:.6f}, length = {len(seq)}')

    # ------------------------------------------------------------------
    # iterative improvement
    # ------------------------------------------------------------------
    it = 0
    no_improve = 0
    while time.time() < deadline:
        n = len(seq)

        # current convolution and peak
        conv_ff = convolve_fft(seq, seq)
        M = conv_ff.max()

        # --------------------------------------------------------------
        # 1) Linear‑programming step: find a direction g that increases the sum
        # --------------------------------------------------------------
        g = solve_convolution_lp(seq, float(M))
        if g is None:
            # LP failed – apply a tiny random move
            seq = random_perturb(seq)
            continue

        # --------------------------------------------------------------
        # 2) Determine the maximal safe mixing factor t
        # --------------------------------------------------------------
        conv_fg = convolve_fft(seq, g)
        conv_gg = convolve_fft(g, g)

        t = max_allowed_step(conv_ff, conv_fg, conv_gg, float(M))

        # if the admissible step is negligible, perform a random jitter
        if t < 1e-6:
            seq = random_perturb(seq)
            continue

        # --------------------------------------------------------------
        # 3) Mix and evaluate
        # --------------------------------------------------------------
        new_seq = (1.0 - t) * seq + t * g

        # occasional random permutation to break possible symmetry
        new_seq = maybe_shuffle(new_seq, prob=0.01)

        # evaluate the new candidate
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # safety net – discard this candidate
            seq = random_perturb(seq)
            continue

        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            no_improve = 0
            print(f'Iter {it:04d}: improved score = {best_score:.6f}, '
                  f't = {t:.4f}, length = {n}')
        else:
            no_improve += 1

        # accept the new sequence as the current state (even if not better)
        seq = new_seq

        # ------------------------------------------------------------------
        # 4) Restart strategy if stuck for many iterations
        # ------------------------------------------------------------------
        if no_improve >= 250:
            # random restart with a different length
            new_len = np.random.randint(600, 1500)
            seq = np.random.rand(new_len)
            seq = np.clip(seq, 0.0, 1000.0)
            no_improve = 0
            print(f'-- restart: new length = {new_len}')
        it += 1

        # early exit if we already beat the target (optional)
        if best_score < 1.503:
            print('Target reached, terminating early.')
            break

    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f'Finished. Best score = {final_score:.6f}, length = {len(final_seq)}')
    return final_seq
