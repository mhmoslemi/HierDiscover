def project_simplex(v):
    # based on https://gist.github.com/daien/1272551
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative list) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the ordinary discrete convolution.
The optimizer works on the probability simplex (sum=1, a_i≥0) because the
objective is scale‑invariant.  A smooth approximation of the maximum
(using a soft‑max with a temperature α) is minimised by projected
gradient descent with momentum.  The gradient is obtained analytically
and the required convolutions are performed with an FFT for speed.
Multiple random restarts with different lengths and initial shapes are
performed until the time budget expires.  The best sequence seen so far
is returned.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: fast (real) convolution using FFT (O(N log N))
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution of real vectors x and y using FFT."""
    n = x.size + y.size - 1
    # next power of two for efficient FFT
    N = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    conv = np.fft.irfft(X * Y, N)[:n]
    return conv


# --------------------------------------------------------------
# Helper: projection onto the probability simplex (sum = 1, all ≥ 0)
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x | x_i ≥ 0, sum x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, return a uniform point
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
