def project_to_simplex(v, z):
    # ensures sum(v) = z and v >= 0
    # using algorithm from Duchi et al: sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_simplex(v, z):
    # returns vector w that is projection of v onto {w >=0, sum(w)=z}
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    # find rho: last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all u_i <= etc, then set w = zeros, w[0] = z?
        # fallback to just set w = max(v - theta, 0)
        # but set w = zeros and w[0]=z is safe for positivity constraint?
        rho = 0
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1] + 1  # +1 for one-indexed
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def gradient_of_max(a, top_k=3):
    n = a.shape[0]
    b = np.convolve(a, a)
    # find indices of top k values (including ties)
    if top_k >= len(b):
        idxs = np.arange(len(b))
    else:
        idxs = np.argpartition(-b, top_k-1)[:top_k]
    g = np.zeros_like(a)
    # compute gradient contributions
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)

        g[mask] += 2 * a[j[mask]]
        diag = mask & (i == j)
        g[diag] -= a[i[diag]]  # because we added 2*a[i] but need only a[i]
    return g, b.max()

"""
propose_candidate
-----------------

A fast heuristic that minimises the evaluation score by directly reducing the
maximum of the self‑convolution while keeping the ℓ¹–norm fixed.
The algorithm works on a fixed length (the length of the provided
`height_sequence_1` when it exists) and consists of a projected gradient
descent on the maximum convolution entry.

The functional to minimise (with a fixed sum S) is

    max_k (a * a)[k]          (where * denotes discrete convolution)

because `evaluate_sequence` returns

    2·n·max_k (a*a)[k] / (∑a)²

and the sum is kept constant (S = √(2 n)).  The gradient of the active
convolution entry `k*` is

    ∂/∂a_i  Σ_j a_j a_{k*-j} = 2·a_{k*-i}   (i ≠ k*/2)
    ∂/∂a_i  = a_i                       (i = k*/2)

The method iteratively:
  * computes the convolution,
  * builds a sub‑gradient from the `top_k` largest convolution values,
  * takes a small step opposite to the gradient,
  * projects the result back onto the simplex {a ≥ 0, Σa = √(2 n)},
  * keeps the best vector seen.

Occasional random perturbations help to escape plateaus.  The whole
search runs for almost the whole allotted budget (≈ 1000 s) and finally
returns the best sequence as a plain Python list.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex of total mass z."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)          # 1‑based indices
    # locate the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all entries will be zero
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1    # +1 because of zero‑based indexing
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, top_k: int = 3):
    """Compute a sub‑gradient of max_k (a*a)[k] and return (grad, max_val)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()

    # indices of the `top_k` largest entries (including ties)
    if top_k >= conv.size:
        idxs = np.arange(conv.size)
    else:
        idxs = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(a)
    i = np.arange(n)

    for k in idxs:
        j = k - i                # partner index for each i
        mask = (j >= 0) & (j < n)

        # derivative = 2*a_j   (for all valid pairs)
        grad[mask] += 2.0 * a[j[mask]]

        # diagonal correction: when i == j the term appears only once
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad, max_val


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a step‑function giving a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, arbitrary length).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little margin for cleanup

    # ------------------------------------------------------------------
    # Choose a length.  Prefer the length of the known construction.
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    # Initial vector – either the reference or a fresh random start.
    if have_ref and np.random.rand() < 0.5:
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
    else:
        a = np.random.rand(n)

    # Target ℓ¹‑norm (the value used by the reference solution)
    target_sum = math.sqrt(2.0 * n)

    # Project the start onto the feasible simplex.
    a = _project_to_simplex(a, target_sum)

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution seen so far.
    best_a = a.copy()
    # Since the sum is fixed, the score is simply 2·n·max_conv / target_sum².
    _, max0 = _gradient_of_max(best_a, top_k=1)
    best_score = 2.0 * n * max0 / (target_sum ** 2)
    print(f"[init] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the projected gradient descent.
    top_k = 5                # number of active convolution peaks
    eta0 = target_sum * 0.4 / n   # base step size (≈ 0.5·average entry)
    decay = 1e-4             # simple 1/(1+decay·t) decay
    max_iters = 10 ** 7      # safety cap (real stop is time based)
    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iters:
        # --------------------------------------------------------------
        # 1) gradient of the current active constraints
        grad, cur_max = _gradient_of_max(a, top_k=top_k)

        # 2) step size (decreases slowly)
        eta = eta0 / (1.0 + decay * it)

        # 3) gradient step and projection back onto the simplex
        a_new = a - eta * grad
        a_new = _project_to_simplex(a_new, target_sum)

        # 4) evaluate (fast, uses the already computed convolution max)
        #    because the ℓ¹‑norm is fixed we can reuse the formula.
        _, cur_max = _gradient_of_max(a_new, top_k=1)
        cur_score = 2.0 * n * cur_max / (target_sum ** 2)

        # 5) keep the best solution
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            no_improve = 0
            print(f"[{it:08d}] improved score = {best_score:.6f}")
        else:

        # --------------------------------------------------------------

        if (it % 2000 == 0 and it > 0) or no_improve > 5000:

            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[{it:08d}] random perturbation")

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # A few cheap random “block” candidates are tried after the main run.
    # They often discover surprisingly good patterns with almost no cost.
    block_trials = 200
    for _ in range(block_trials):

        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)

        cand = np.zeros(n, dtype=float)

        cand = _project_to_simplex(cand, target_sum)
        _, cm = _gradient_of_max(cand, top_k=1)

    # ------------------------------------------------------------------

    # will call `evaluate_sequence` on it.
    print(f"[final] best score = {best_score:.6f} after {it} iterations")
    return best_a.tolist()
