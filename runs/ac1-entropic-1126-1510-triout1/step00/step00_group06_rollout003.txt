```python
import numpy as np
import time

def project_simplex(v):
    # Implements Euclidean projection onto the simplex sum(v)=1, v>=0
    n = len(v)
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def gradient_step(seq, eta):
    n = len(seq)
    b = np.convolve(seq, seq)
    k_max = int(np.argmax(b))
    # gradient vector g: for i where j = k_max - i in [0, n-1]
    rev = np.zeros_like(seq)
    # rev[i] = seq[k_max-i] if valid else 0
    low = max(0, k_max - (n-1))
    high = min(k_max, n-1)
    # i runs from low to high inclusive
    # i index, j = k_max - i
    # rev[i] = seq[j]
    rev[low:high+1] = seq[k_max - np.arange(low, high+1)]
    # do step
    new_seq = seq - eta * rev
    # projection onto simplex
    new_seq = project_simplex(new_seq)
    return new_seq

"""
Propose a candidate step‑function sequence that minimises the
`evaluate_sequence` metric.

The algorithm is simple but very effective for large n:

*   The metric equals 2·n·max(b) with b = a*a (convolution) when the
    sequence is normalised to sum(a)=1.  For a uniform a the maximum
    of b is twice its average, giving the value 2.  Randomly
    distributed entries break the perfect correlation of the uniform
    case, making the convolution essentially flat; the maximum then
    is close to the average, yielding a value close to 1.

*   We therefore generate many random Dirichlet‑distributed vectors
    of moderate length (1500–5000), normalise them, evaluate the
    metric and keep the best one.

*   To squeeze a little more performance we apply a cheap projected‑
    gradient descent on promising candidates.  The gradient of
    max(b) w.r.t. a_i is a_{k_max-i} where k_max is the index of the
    current maximum.  After a gradient step the vector is projected
    back onto the probability simplex (non‑negative, sum = 1).

*   The search runs until the allocated time (default 1000 s) is about
    to expire and finally returns the best sequence found.

The whole procedure respects the security checks in `evaluate_sequence`,
uses only standard scientific libraries and never performs I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector v onto the simplex {x >= 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        # All entries are zero – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: a single gradient‑descent step on max(b) with simplex projection
# ----------------------------------------------------------------------
def _gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One projected‑gradient step that tries to reduce max(convolve(a,a)).
    a is assumed to sum to 1.
    """
    n = a.shape[0]
    # convolution and index of the current maximum
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # gradient of the maximal convolution entry
    grad = np.zeros_like(a)
    # valid i satisfy 0 <= k_max-i < n  -> i in [low, high]
    low = max(0, k_max - (n - 1))
    high = min(k_max, n - 1)
    if low <= high:
        idx = np.arange(low, high + 1)
        # grad[i] = a[k_max - i]
        grad[idx] = a[k_max - idx]

    # gradient descent step followed by projection onto the simplex
    a_new = a - eta * grad
    a_new = _project_onto_simplex(a_new)
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000.0, **kwargs) -> list[float]:

    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs for up to `budget_s` seconds
    and returns the best sequence found so far.
    """
    # time budget (a safety margin of a few seconds is kept)
    deadline = time.time() + budget_s - 5.0

    # start from the provided global solution if it exists
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        try:
            cand = list(height_sequence_1)
            sc = evaluate_sequence(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"[init] baseline score = {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Random Dirichlet generation (alpha controls concentration)
    # ------------------------------------------------------------------
    def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
        """Return a random point on the simplex using Gamma(alpha,1)."""
        # Gamma can return zeros when alpha is very small; guard against that
        g = np.random.gamma(shape=alpha, scale=1.0, size=n)
        total = g.sum()
        if total <= 0.0:
            g = np.ones(n)
            total = n
        return g / total

    it = 0
    while time.time() < deadline:
        # ---- sample a random length and concentration parameter ----
        n = np.random.randint(1500, 5000)          # length of the candidate
        alpha = np.random.uniform(0.5, 2.0)        # Dirichlet concentration

        # ---- generate a random candidate and evaluate it ----
        a = _random_dirichlet(n, alpha)
        try:

        except Exception:

        if score < best_score:

            best_seq = a.tolist()

        n = a.shape[0]
        eta = 0.2
        for step in range(80):

            eta *= 0.985
        best_seq = a.tolist()
        # One final evaluation (mostly for logging)
        try:
            final_score = evaluate_sequence(best_seq)
            print(f"[final] Polished score = {final_score:.6f} (n={n})")
        except Exception:
            pass

    # Return the best sequence (as a plain Python list of floats)
    return best_seq
