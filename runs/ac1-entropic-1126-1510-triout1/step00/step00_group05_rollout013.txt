def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum -> minimize -sum
    # Build A_ub, b_ub
    # We'll preallocate matrix of shape (2*n-1 + n, n) = (3n-1, n)
    rows = []
    # for each k from 0 to 2n-2
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        # for i in i_min..i_max: a_i corresponds to f_seq[i] and we need row[j] = f_seq[i] where j = k - i
        # That means row[j] = f_seq[i]
        # For each i compute j = k - i, set row[j] = f_seq[i]
        # Equivalent to row = zeros; for i in range(i_min,i_max+1):
        #   j = k - i
        #   row[j] = f_seq[i]
        # So O(n^2).
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
    # Add nonnegativity constraints: -I * b <= 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([np.full(len(rows), rhs), np.zeros(n)])
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'time_limit': 5.0})
    if res.success:
        return res.x
    return None

"""
Propose a candidate step‑function (height sequence) that beats the current bound
(≈2.0).  The algorithm starts from a simple uniform distribution, then repeatedly
solves a linear program that finds a direction g whose convolution with the
current sequence f is bounded by the current maximum.  By mixing f and a scaled
version of g we strictly reduce the peak of the self‑convolution.  The mixing
step uses the optimal convex combination derived from the quadratic bound
(1‑t)²‖f*f‖∞ + t²‖g*g‖∞ + 2t(1‑t)‖f*g‖∞.  Several restarts and occasional
random swaps avoid being trapped in a local optimum.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """
    Linear program: maximise sum(b)  subject to  conv(f_seq, b) ≤ rhs  and  b ≥ 0.
    Returns the optimal b (numpy array) or None if the solver fails.
    """
    n = len(f_seq)
    # objective: maximise sum(b) → minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # Build the inequality matrix for conv(f, b) ≤ rhs.
    # For each output index k of the convolution we have
    #   sum_{i+j=k} f[i] * b[j] ≤ rhs.
    # Row k contains f[i] at column j = k-i (if the index is inside 0..n-1).
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # i runs over indices of f that appear in this convolution entry.
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    # Non‑negativity constraints  -I·b ≤ 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([np.full(len(rows), rhs, dtype=float), np.zeros(n, dtype=float)])

    # Solve.  Use the HiGHS solver which is fast for moderate sized LPs.
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": time_limit},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


def _proj_to_simplex(v, target_sum):
    """
    Project a (possibly negative) vector onto the simplex
    {x ≥ 0, Σx = target_sum}.  Uses the O(n log n) algorithm.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    # Find the rho where u_i - (cumsum - target)/i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed=42, budget_s=950, **kwargs):
    """
    Search for a height sequence with a low evaluation score.
    The routine performs several independent restarts; each restart runs a
    simple LP‑guided descent that mixes the current vector with a new direction
    obtained from a linear program.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a tiny margin

    # --------------------------------------------------------------------
    # Helper: compute the evaluation score using the provided evaluate_sequence.
    # --------------------------------------------------------------------
    def _score(seq):
        # evaluate_sequence is assumed to be already imported in the environment
        return evaluate_sequence(seq.tolist())

    # --------------------------------------------------------------------
    # Hyper‑parameters (can be tweaked if more time is available)
    # --------------------------------------------------------------------
    n = 800                      # length of the sequence (moderate size)
    target_sum = math.sqrt(2.0 * n)
    max_lp_time = 6.0            # seconds per LP solve
    mix_candidates = [0.2, 0.4, 0.5, 0.6, 0.8]   # t‑values for mixing
    random_swap_prob = 0.15      # probability to perform a random swap instead of LP

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------------
    # Perform a few independent restarts; each tries to improve its own copy.
    # --------------------------------------------------------------------
    restarts = 3
    restart_budget = (deadline - time.time()) / restarts

    for r in range(restarts):
        # initialise: uniform plus a light random perturbation
        f = np.full(n, target_sum / n, dtype=float)
        f += 0.001 * target_sum * (np.random.rand(n) - 0.5)
        f = _proj_to_simplex(f, target_sum)

        # remember best for this restart
        cur_score = _score(f)
        if cur_score < best_score:
            best_score, best_seq = cur_score, f.copy()

        # inner optimisation loop
        while time.time() < start_time + (r + 1) * restart_budget:
            # Occasionally do a cheap random swap to escape plateaus
            if np.random.rand() < random_swap_prob:
                i, j = np.random.choice(n, size=2, replace=False)
                delta = min(f[i], 0.01 * target_sum)
                f[i] -= delta
                f[j] += delta
                f = _proj_to_simplex(f, target_sum)
                cur_score = _score(f)
                if cur_score < best_score:
                    best_score, best_seq = cur_score, f.copy()
                continue

            # ---- 1. evaluate current max convolution --------------------
            conv_f = np.convolve(f, f)
            M_f = conv_f.max()

            # ---- 2. solve the LP to obtain a promising direction -----
            g = _solve_convolution_lp(f, M_f, time_limit=max_lp_time)
            if g is None:
                # fall back to a tiny random perturbation
                idx = np.random.randint(n)
                f[idx] = max(0.0, f[idx] + np.random.randn() * 0.01 * target_sum)
                f = _proj_to_simplex(f, target_sum)
                continue

            # ---- 3. scale g so that it has the required L1 norm ------
            sum_g = g.sum()
            if sum_g <= 0:
                # degenerate LP – skip
                continue
            g = g * (target_sum / sum_g)

            # ---- 4. pre‑compute the three needed convolution maxima ----
            M_g = np.convolve(g, g).max()
            M_fg = np.convolve(f, g).max()

            # ---- 5. try a few mixing coefficients t --------------------
            improved = False
            for t in mix_candidates:
                new_f = (1.0 - t) * f + t * g
                # mixing of two vectors with the same L1 norm automatically keeps the sum
                new_score = _score(new_f)
                if new_score < cur_score - 1e-8:
                    f = new_f
                    cur_score = new_score
                    improved = True
                    if new_score < best_score:
                        best_score, best_seq = new_score, new_f.copy()
                    break

            # if none of the fixed t‑values helped, try the analytically optimal t
            if not improved:
                denom = M_f + M_g - 2.0 * M_fg
                if denom > 1e-12:
                    t_opt = (M_f - M_fg) / denom
                    t_opt = max(0.0, min(1.0, t_opt))
                else:
                    t_opt = 0.5
                if 0.0 < t_opt < 1.0:
                    new_f = (1.0 - t_opt) * f + t_opt * g
                    new_score = _score(new_f)
                    if new_score < cur_score - 1e-8:
                        f = new_f
                        cur_score = new_score
                        if new_score < best_score:
                            best_score, best_seq = new_score, new_f.copy()
                        continue

            # If still no improvement, do a tiny random perturbation to shake things.
            if not improved:
                idx = np.random.randint(n)
                delta = 0.005 * target_sum * (np.random.rand() - 0.5)
                f[idx] = max(0.0, f[idx] + delta)
                f = _proj_to_simplex(f, target_sum)

        # end of one restart
        # print progress for this restart
        print(
            f"Restart {r+1}/{restarts} finished.  Best score so far = {best_score:.6f}"
        )
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list of floats)
    # --------------------------------------------------------------------
    print(f"Overall best score = {best_score:.6f}")
    return best_seq.tolist()
