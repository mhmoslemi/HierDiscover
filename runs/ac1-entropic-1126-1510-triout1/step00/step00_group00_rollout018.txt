"""
Propose a low‑value step sequence.

The search consists of two stages.
1. A parametrised “arc‑sine” shape   a_i ∝ 1/√((i+½+ε)(n‑i‑½+ε))
   is examined for many values of ε.  After a simple normalisation the
   sequence is fed to ``evaluate_sequence`` – the ε giving the smallest
   score is kept.
2. Starting from the best ε we run a cheap stochastic hill‑climb:
   a small amount of mass is moved from one index to another while
   keeping the total sum unchanged.  Whenever a move improves the
   score it is accepted; otherwise it may be accepted with a tiny
   probability that slowly vanishes (simulated‑annealing style).
   The step size shrinks as the score improves.

Both stages respect the non‑negativity, clipping and scaling rules of
``evaluate_sequence`` and finish well before the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: build the arc‑sine family for a given epsilon.
# ----------------------------------------------------------------------
def _arcseq(n: int, eps: float) -> np.ndarray:
    """
    Return a non‑negative sequence of length ``n``.
    a_i = 1 / sqrt( (i+0.5+eps) * (n-i-0.5+eps) )
    The sequence is normalised so that its L¹‑norm is 1
    (the scaling does not affect ``evaluate_sequence`` but keeps the
    numbers well‑behaved).
    """
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5 + eps) * (n - i - 0.5 + eps))
    return w / w.sum()                     # sum == 1


# ----------------------------------------------------------------------
# The required entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low ``evaluate_sequence`` value.

    The function is deterministic apart from the random seed.
    It returns a Python ``list`` of floats.
    """
    np.random.seed(seed)

    # length of the problem – the reference construction is always present
    n = len(height_sequence_1)

    # Leave a small safety margin for the last prints etc.
    deadline = time.time() + max(0.0, budget_s - 5.0)

    # --------------------------------------------------------------
    # 1. Coarse sweep over epsilon
    # --------------------------------------------------------------
    eps_grid = [
        0.0, 0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
        80.0, 160.0, 320.0, 640.0, 1280.0, 2560.0, 5000.0
    ]

    best_seq = None
    best_score = float("inf")
    best_eps = None

    for eps in eps_grid:
        cand = _arcseq(n, eps)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            best_eps = eps
        print(f"[eps={eps:6.1f}] score = {score:.6f}")

    print(f"--> best eps after grid = {best_eps:.1f}, score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Local stochastic improvement (hill‑climb / simulated annealing)
    # --------------------------------------------------------------
    cur_seq = best_seq.copy()
    step_factor = 0.15                     # max proportion of source mass to move
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # pick a source index that currently carries some mass
        positive = np.where(cur_seq > 1e-12)[0]
        if positive.size == 0:
            break
        src = np.random.choice(positive)
        dst = np.random.randint(n)
        if src == dst:
            continue

        new_seq[src] -= delta
        new_seq[dst] += delta

        new_score = evaluate_sequence(new_seq.tolist())

        # Acceptance rule
        # ------------------------------------------------------------------

    print(f"Search finished after {iteration} iterations – best score: {best_score:.6f}")
    # ``evaluate_sequence`` works with a plain list, so we return that.
