    # Additional initial from height_sequence_1 (maybe different)
    seq1 = np.array(base_seq, dtype=float)
    score1 = compute_score(seq1)
    if score1 < best_score:

    # Local search
    a = best_seq.copy()
    cur_score = best_score

    # Simulated annealing parameters
    temperature = 1.0
    min_temp = 1e-4
    # step size fraction
    step_frac = 0.01   # relative to max element

    while time.time() < deadline:
        # choose two indices i != j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        delta = step_frac * a[i] * np.random.rand()
        if delta == 0:
            continue
        # propose new a
        a_new = a.copy()
        a_new[i] = max(0.0, a_new[i] - delta)
        a_new[j] = a_new[j] + delta
        # evaluate quickly
        cur_candidate_score = compute_score(a_new)

        # accept if better or with probability
        if cur_candidate_score < cur_score or np.random.rand() < np.exp(-(cur_candidate_score - cur_score) / max(temperature, 1e-12)):

                best_seq = a.copy()

        # cooling
        temperature *= 0.9995  # gradually cool
        if temperature < min_temp:
            temperature = min_temp

    # after loop, return best_seq (list)
    return best_seq.tolist()

"""
Simple simulated‑annealing / gradient style optimizer for the
step‑function bound.

The evaluation metric is

    M = 2 * n * max(conv(a,a)) / (sum(a)^2)

which is invariant under a global scaling of the vector.
The algorithm keeps the ℓ1‑norm (sum) equal to 1, works with the
full (linear) convolution computed by FFT (O(n log n)) and repeatedly
moves a tiny amount of mass from an index that contributes most to the
current maximal convolution entry to an index that contributes little.
A temperature schedule allows occasional uphill moves, and a small
random jitter is applied every few hundred iterations to escape plateaus.

The routine works with any length (defaults to the length of the
provided `height_sequence_1`).  It returns the best sequence it has
found before the time budget expires.
"""

import time
import numpy as np


def _fft_max_conv(seq: np.ndarray, size: int):
    """
    Compute the linear convolution of `seq` with itself using FFT,
    return (max_value, argmax_index, full convolution array).
    """
    # real FFT of zero‑padded vector
    fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(seq) - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _score(seq: np.ndarray, size: int) -> float:
    """
    Fast replica of ``evaluate_sequence`` (clipping, sum check,
    max of the self‑convolution, then the formula).
    The sequence is assumed to be non‑negative and already normalised
    (its ℓ1‑norm is not zero).
    """
    a = np.clip(seq, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    max_b, _, _ = _fft_max_conv(a, size)
    n = len(a)
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    bound returned by ``evaluate_sequence``.

    The method starts from a uniform vector (or the provided global
    ``height_sequence_1`` if it exists) and performs a large number
    of cheap gradient‑like moves while keeping the total mass constant.
    A simple temperature schedule and occasional random jitter help
    to avoid local minima.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # initialise the sequence -------------------------------------------------
    try:
        # the historical construction is available globally
        init = np.array(height_sequence_1, dtype=float)
        n = init.size
    except NameError:
        # fall back to a plain uniform vector of moderate size
        n = 4000
        init = np.full(n, 1.0 / n)

    # normalise to ℓ1‑norm = 1 (scaling does not affect the score)
    init = np.maximum(init, 0.0)
    init = init / init.sum()

    uniform = np.full(n, 1.0 / n)

    # pre‑compute the FFT length (next power of two ≥ 2n−1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # pick the better of the two simple starts
    cur = init if _score(init, fft_len) < _score(uniform, fft_len) else uniform.copy()
    cur_score = _score(cur, fft_len)

    best_seq = cur.copy()
    best_score = cur_score

    print(f"[init] n={n}  start_score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation parameters -------------------------------------------------
    temperature = 1.0
    min_temp = 1e-4
    iteration = 0
    # step fraction is scaled with temperature (large early, tiny later)
    base_step = 0.2

    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- current convolution information ----------------------------
        max_val, k_max, conv = _fft_max_conv(cur, fft_len)

        # contributions of each index to the maximal entry
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = cur[mask] * cur[partner[mask]]

        # source: index with the largest contribution to the peak
        i_src = int(contrib.argmax())

        # destination: any index with a relatively small contribution
        low_thr = np.percentile(contrib, 20)
        low_cands = np.where(contrib <= low_thr)[0]
        if low_cands.size == 0:
            # fallback – pick a random different index
            i_dst = int(np.random.randint(n))
            while i_dst == i_src:
                i_dst = int(np.random.randint(n))
        else:
            i_dst = int(np.random.choice(low_cands))

        # how much mass to move (scaled with temperature)
        step_frac = max(1e-4, temperature * base_step)
        delta = step_frac * cur[i_src]
        if delta <= 0.0:
            continue

        # propose a new sequence (mass transfer while preserving total sum)
        new_seq = cur.copy()
        new_seq[i_src] = max(0.0, new_seq[i_src] - delta)
        new_seq[i_dst] += delta
        # renormalise to avoid drift (doesn't affect the score)
        total = new_seq.sum()
        if total == 0.0:
            continue
        new_seq /= total

        new_score = _score(new_seq, fft_len)

        # accept if better or with Metropolis probability
        if new_score < cur_score or np.random.rand() < np.exp(
            -(new_score - cur_score) / max(temperature, 1e-12)
        ):
            cur = new_seq
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(
                    f"[iter {iteration:06d}] new best  score={best_score:.6f}  "
                    f"temp={temperature:.4f}"
                )

        # temperature cooling
        temperature = max(min_temp, temperature * 0.9995)

        # occasional random jitter to shake the system
        if iteration % 500 == 0:
            jitter = np.random.randn(n) * 0.001
            cur += jitter
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()
            cur_score = _score(cur, fft_len)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(
                    f"[jitter {iteration:06d}] improved after jitter  score={best_score:.6f}"
                )

    print(f"[done] iterations={iteration}  best_score={best_score:.6f}")
    # convert to plain python list as required by the outer harness
    return best_seq.tolist()
