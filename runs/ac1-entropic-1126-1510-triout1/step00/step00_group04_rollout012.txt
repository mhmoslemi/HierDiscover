def simplex_projection(v, z=1):
    """Project v onto simplex {x >= 0, sum x = z}."""
    # From https etc..
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v, target):
    """Projection of v onto the simplex {x >= 0, sum x = target}."""
    if target <= 0:
        return np.zeros_like(v)
    # shift v by subtracting some offset to meet sum constraint.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values will be zero
        theta = 0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

```
def subgradient_optimize(a, max_iter, deadline):
    n = len(a)
    target = np.sqrt(2*n)
    best_a = a.copy()
    best_score = np.max(np.convolve(a, a))
    avg = target / n
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        # compute gradient
        idx = np.where(conv >= M - 1e-12)[0]
        grad = np.zeros_like(a)
        for k in idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            grad[i_range] += a[j_range]
        # step size
        eta = avg / np.sqrt(it + 1) * 0.5  # heuristic
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        # project onto simplex
        a_sum = a.sum()
        if a_sum > 0:
            a = project_to_simplex(a, target)
        else:
            a = np.full_like(a, target / n)
        cur_score = np.max(np.convolve(a, a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # early stop if improvement tiny?
    return best_a, best_score

"""
Propose a candidate step‑function sequence with a low evaluation score.

The algorithm combines three ideas:

1.   Randomly initialise many sequences of moderate length (20–500) with
     simple parametric shapes (exponential, power‑law, linear).  Each
     sequence is normalised so that its L1‑norm equals sqrt(2·n) – this
     makes the score directly equal to the maximal convolution value.

2.   Refine every initial candidate with a cheap sub‑gradient descent.
     The objective is  max_k (a*a)[k]  under the simplex constraint
     Σa_i = sqrt(2·n) and a_i ≥ 0.  At each iteration the gradient of the
     maximal convolution entry is assembled (only the indices that are
     currently tight contribute) and the vector is moved opposite to that
     gradient.  After the step the vector is projected back onto the
     simplex.  A diminishing step‑size (∝ 1/√iteration) provides stability.

3.   Keep the best sequence seen so far and periodically report progress.
     The whole search stops when the allotted wall‑clock time (budget_s)
     is exhausted.  The returned sequence is a Python list of non‑negative
     floats; the evaluation routine will clamp values automatically.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (scaled to a given sum)
def project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project the vector v onto the simplex {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimiser for a fixed length
def subgradient_optimize(
    a0: np.ndarray,
    max_iter: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run a simple sub‑gradient descent on the sequence a0.
    Returns the best sequence found and its score (max convolution value).
    """
    a = a0.copy()
    n = len(a)
    target_sum = np.sqrt(2.0 * n)
    avg = target_sum / n
    best_a = a.copy()
    best_score = np.max(np.convolve(a, a))

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        # indices where the max is attained (within numerical tolerance)
        K = np.where(conv >= M - 1e-12)[0]

        # gradient of the maximal entry (sub‑gradient)
        grad = np.zeros_like(a)
        for k in K:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=int)
            j_range = k - i_range
            grad[i_range] += a[j_range]

        # diminishing step size
        eta = 0.5 * avg / np.sqrt(it + 1.0)

        a = a - eta * grad
        a = np.maximum(a, 0.0)
        # project back onto the simplex (keeps Σa_i fixed)
        a = project_to_simplex(a, target_sum)

        cur_score = np.max(np.convolve(a, a))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """Search for a low‑score sequence within the given time budget."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # tiny safety margin

    # start from the already discovered sequence if it exists
    best_seq = None
    best_score = np.inf
    try:
        if "height_sequence_1" in globals():
            seq0 = np.asarray(height_sequence_1, dtype=float)
            n0 = len(seq0)
            if n0 > 0:
                seq0 = seq0 * (np.sqrt(2.0 * n0) / seq0.sum())
                cur = np.max(np.convolve(seq0, seq0))
                best_seq = seq0.tolist()
                best_score = cur
                print(
                    f"Initialised from height_sequence_1 (n={n0}) → score {best_score:.6f}"
                )
    except Exception as e:  # pragma: no cover
        print("Failed to read height_sequence_1:", e)

    # ------------------------------------------------------------------
    # Main stochastic search loop
    lengths = [20, 30, 50, 80, 120, 200, 300, 500]
    shape_kinds = ["exp", "power", "linear"]

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) random initialisation
        n = int(np.random.choice(lengths))
        shape = np.random.choice(shape_kinds)

        if shape == "exp":
            q = np.random.uniform(0.3, 0.99)           # decay factor
            a = np.power(q, np.arange(n, dtype=float))
        elif shape == "power":
            p = np.random.uniform(0.1, 2.0)            # power law exponent
            a = 1.0 / np.power(np.arange(1, n + 1, dtype=float), p)
        else:  # linear decreasing from a random max value
            mx = np.random.uniform(0.5, 2.0)
            a = np.linspace(mx, 0.0, n, dtype=float)

        # tiny random noise helps avoid perfect symmetry
        a += np.random.rand(n) * 0.01
        a = np.maximum(a, 0.0)

        # normalise the L1‑norm to sqrt(2·n) (the scale used by the evaluator)
        target = np.sqrt(2.0 * n)
        if a.sum() == 0:
            a = np.full(n, target / n, dtype=float)
        else:
            a = a * (target / a.sum())

        # quick score (max convolution) – no need to call evaluate_sequence yet
        cur_score = np.max(np.convolve(a, a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            print(
                f"New baseline from random start: n={n}, shape={shape}, score={best_score:.6f}"
            )

        # ------------------------------------------------------------------
        # 2) local refinement with sub‑gradient descent
        max_iter_local = 200
        a_opt, score_opt = subgradient_optimize(a, max_iter_local, deadline)
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.tolist()
            print(
                f"Improved after sub‑gradient: n={n}, shape={shape}, score={best_score:.6f}"
            )

        # occasionally break early to respect the deadline
        if time.time() >= deadline:
            break

    # ----------------------------------------------------------------------
    # Fallback – should never be needed but kept for safety
    if best_seq is None:
        n = 50
        a = np.linspace(1.0, 0.0, n, dtype=float)
        a = a * (np.sqrt(2.0 * n) / a.sum())
        best_seq = a.tolist()
        best_score = evaluate_sequence(best_seq)

    # final verification (the evaluator does the safety checks)
    final_score = evaluate_sequence(best_seq)
    print(
        f"Returning best sequence (length={len(best_seq)}) with final score {final_score:.6f}"
    )
    return best_seq
