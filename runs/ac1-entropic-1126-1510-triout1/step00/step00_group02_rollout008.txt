"""
Propose a step‐function (a non‑negative list of heights) whose
self‑convolution has a low maximum value.

The objective that is actually minimised is

    E(a) = 2·n·max(a*a) / (sum(a))**2

where n = len(a) and a*a is the ordinary (zero‑padded) discrete convolution.
The global optimum is E≈1 (the average of the convolution is S²/(2n‑1)),
while the trivial uniform sequence gives E=2.  The routine below performs a
simple stochastic “flatten‑the‑peak” search:

* start from a uniform sequence (or from the previously known construction);
* repeatedly locate the index k where the convolution is maximal;
* pick a pair (i, k‑i) that contributes heavily to that peak;
* move a small amount of mass from i to a random location with a small value;
* accept the move only if the score improves (otherwise revert);
* keep the best sequence seen.

All operations are carried out with NumPy, the score is computed directly
(bypassing the provided evaluate_sequence for speed – the result is
identical for valid inputs).  The whole procedure runs for the allotted
time budget and returns the best sequence found.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: fast evaluation – identical to evaluate_sequence for valid data
def _score(seq: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative float array."""
    # security checks performed by the original function are reproduced here
    # (negative → 0, >1000 → 1000)
    seq = np.clip(seq, 0.0, 1000.0)
    total = seq.sum()
    if total < 0.01:
        return np.inf
    conv = np.convolve(seq, seq)          # length 2·n‑1
    max_conv = conv.max()
    return 2.0 * len(seq) * max_conv / (total * total)


# ------------------------------------------------------------
def _flatten_peak(a: np.ndarray, rng: np.random.Generator) -> bool:
    """
    Perform one stochastic move that tries to reduce the current convolution peak.
    Returns True if the move was accepted (i.e. the score improved), False otherwise.
    """
    n = a.shape[0]

    # current convolution and its maximal position
    conv = np.convolve(a, a)
    kmax = int(np.argmax(conv))

    # indices i that actually participate in the kmax bin
    i_min = max(0, kmax - (n - 1))
    i_max = min(n - 1, kmax)
    idx = np.arange(i_min, i_max + 1, dtype=int)
    jdx = kmax - idx                     # the partner indices (kmax - i)

    # contribution weights = a[i] * a[j]; use them to pick a heavy contributor
    weights = a[idx] * a[jdx]
    wsum = weights.sum()
    if wsum == 0.0:
        # fallback – pick any contributing index uniformly
        i = rng.integers(i_min, i_max + 1)
    else:
        probs = weights / wsum
        i = rng.choice(idx, p=probs)

    # partner of i for the peak (might be the same when i == jdx)
    partner = kmax - i

    # choose a destination index (prefer a position with a small value)
    # simple heuristic: uniform random draw; if the draw lands on a large entry
    # we simply reject the move later.
    dest = rng.integers(0, n)

    # amount to move – at most half of what is present at the source,
    # and never exceeding the free room at the destination (1000 limit).
    max_move = min(a[i], 0.5 * a[i])
    max_room = 1000.0 - a[dest]
    max_move = min(max_move, max_room)
    if max_move <= 1e-12:
        return False

    delta = rng.random() * max_move        # 0 ≤ delta ≤ max_move
    if delta <= 1e-12:
        return False

    # ----------------------------------------------------------------
    # Apply the tentative move
    a[i] -= delta
    a[dest] += delta

    new_score = _score(a)

    # ----------------------------------------------------------------
    # Accept only if the score improves; otherwise revert.
    if new_score < _score._best_score:          # type: ignore[attr-defined]
        # accepted – the global best score is updated outside this function
        _score._best_score = new_score          # type: ignore[attr-defined]
        return True
    # revert
    a[i] += delta
    a[dest] -= delta
    return False
