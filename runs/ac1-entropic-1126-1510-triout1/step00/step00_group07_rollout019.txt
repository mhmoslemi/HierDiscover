    # returns w such that w >=0, sum w = 1 and closest to v in Euclidean norm.
    # Implementation by sorting.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    """Project vector v onto the probability simplex {x >=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a sequence of non‑negative coefficients that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the full discrete convolution.
The method combines three cheap heuristics that work well within the
1000 s time budget:

1. **Random initialisation** – a Dirichlet draw on a randomly chosen length.
2. **Sub‑gradient descent on the ℓ∞‑convolution term** – the gradient of the
   current maximal convolution entry is known analytically and a small
   projected step reduces the peak.
3. **Linear‑programming refinement** – after a few descent steps we solve the
   LP  

        maximize    Σ g_i
        subject to  (f * g)_k ≤ M   for every k
                    g_i ≥ 0

   where ``f`` is the current sequence and ``M = max(f * f)``.
   Blending the LP solution back into the current iterate (with a line‑search
   over a few mixing factors) often lowers the objective further.

All updates keep the ℓ₁‑norm fixed to ``√(2·n)`` (the normalisation used in
the reference implementation) by a simple rescaling – this does not change the
objective because it is scale‑invariant.  The best sequence found over all
restarts is returned.

The routine respects the 1000 s budget, prints occasional progress messages
and never performs any I/O besides ``print``.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  The implementation follows the classic O(n)
    algorithm by Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_gradient(p: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (p * p)[k]   w.r.t. p.
    For each index k where the maximum is attained the gradient is
        g_i = 2 * p_{k-i}   (if 0 ≤ k-i < n, else 0).
    The returned gradient is the average over all maximal indices.
    """
    n = p.size
    grad = np.zeros_like(p)
    for k in max_idxs:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        grad[i] += 2.0 * p[k - i]
    if max_idxs.size > 0:
        grad /= max_idxs.size
    return grad


def _solve_convolution_lp(f: np.ndarray, bound: float) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise    Σ g_i
        subject to  (f * g)_k ≤ bound   for all k = 0 … 2n‑2
                    g_i ≥ 0
    Returns the optimal g as a dense 1‑D array, or ``None`` if the solver fails.
    """
    n = f.size
    rows = 2 * n - 1

    # Build the constraint matrix A where each row k implements
    #   Σ_i f_i * g_{k-i} ≤ bound
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i                     # j are the column indices of g
        A[k, j] = f[i]

    b = np.full(rows, bound, dtype=float)

    # Use the modern HiGHS interior‑point solver – fast and reliable.
    res = linprog(
        c=-np.ones(n),                 # maximise Σ g_i  ⇔ minimise −Σ g_i
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None

            else:
                # Occasionally inject a tiny random perturbation to escape plateaus.
                if np.random.rand() < 0.02:
                    noise = (np.random.rand(n) - 0.5) * 0.02 * target_sum
                    p_noisy = p + noise
                    p_noisy = np.maximum(p_noisy, 0.0)
                    p_noisy = p_noisy * (target_sum / p_noisy.sum())
                    p = p_noisy
                    cur_score = evaluate_sequence(p.tolist())

        # ----- Optional LP refinement on the best sequence so far -----
        if best_seq is not None:
            # Run a few LP refinement steps on a copy of the current best.
            seq = np.array(best_seq, dtype=float)
            n_seq = seq.size
            target_sum_seq = np.sqrt(2.0 * n_seq)
            seq = seq * (target_sum_seq / seq.sum())

            for lp_iter in range(5):
                if time.time() >= deadline:
                    break
                cur_conv = np.convolve(seq, seq)
                M_seq = cur_conv.max()
                g = _solve_convolution_lp(seq, M_seq)
                if g is None:
                    break

                # Try several mixing factors and keep the best blended candidate.
                t_vals = (0.1, 0.2, 0.5, 1.0)
                best_local_seq = seq
                best_local_score = evaluate_sequence(seq.tolist())
                for t in t_vals:
                    cand = (1.0 - t) * seq + t * g
                    cand = np.maximum(cand, 0.0)
                    cand = cand * (target_sum_seq / cand.sum())
                    cand_score = evaluate_sequence(cand.tolist())
                    if cand_score < best_local_score:
                        best_local_score = cand_score
                        best_local_seq = cand
                seq = best_local_seq
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_seq = seq.tolist()
                    print(f"LP refinement improved global best to {best_score:.6f}")

    # Safety fallback – return a uniform sequence if nothing was found.
    if best_seq is None:
        n = 1000
        best_seq = (np.full(n, np.sqrt(2.0 * n) / n)).tolist()

    return best_seq
