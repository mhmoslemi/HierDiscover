def propose_candidate(seed=123, budget_s=1000):
    """
    Generate a step function sequence using a bounded-sum set construction.
    The algorithm builds a set of positions such that each unordered sum appears
    at most u times (a generalized Sidon set). A small constant epsilon is added
    to every entry to increase the ℓ1 norm without significantly increasing the
    convolution peak. The best sequence found before the time limit is returned.
    """
    import numpy as np, time, random
    # Determine length
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2153
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float('inf')
    attempts = 0

    # baseline epsilon for all positions
    eps = 0.001

    rng = np.random.RandomState(seed)

    # try several values of u (max unordered collisions per sum)
    for u in range(2, 6):   # try 2,3,4,5
        # approximated number of attempts for each u
        while time.time() < deadline:
            attempts += 1
            order = np.arange(n)
            rng.shuffle(order)
            cur_set = []
            sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

            # greedy construction
            for i in order:
                # quick feasibility test
                if sum_counts[2 * i] >= u:
                    continue
                feasible = True
                for j in cur_set:
                    if sum_counts[i + j] >= u:
                        feasible = False
                        break
                if not feasible:
                    continue
                # add i to set
                for j in cur_set:
                    sum_counts[i + j] += 1
                sum_counts[2 * i] += 1
                cur_set.append(i)

            # build sequence with epsilon baseline
            seq = np.full(n, eps, dtype=float)
            if cur_set:
                seq[cur_set] = 1.0 + eps   # spikes of height 1+eps

            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.tolist()
                print(f'New best score {best_score:.6f} with u={u}, set size {len(cur_set)} after {attempts} attempts')
                # early stop if excellent
                if best_score < 1.4:
                    # good enough to exit early
                    break
            # allow early exit if time near.
            if time.time() >= deadline:
                break
        if best_score < 1.4:
            break
    # fallback: If we never improved, return uniform or height_sequence_1
    if best_seq is None:
        best_seq = height_sequence_1.copy()
    return best_seq

"""
Propose a candidate sequence.

The algorithm builds a “generalised Sidon” set of indices: each unordered sum
i+j (0 ≤ i,j < n) is allowed to appear at most *u* times (u = 2,3,…).  With a
larger *u* we can place many more non‑zero entries while keeping the
self‑convolution peak bounded (the peak is at most 2·u for binary values).
After a greedy construction we add a tiny baseline ε to every entry – this
increases the ℓ₁‑norm without appreciably raising the convolution peak.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

def _greedy_b2_set(n: int, u: int, rng: np.random.RandomState):
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that each unordered sum
    i+j (i ≤ j) occurs at most u times.
    Returns the list of selected indices.
    """
    order = np.arange(n)
    rng.shuffle(order)

    cur_set = []                     # selected indices
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)   # unordered‑pair counters

    for i in order:
        # quickly reject if the self‑sum would overflow
        if sum_counts[2 * i] >= u:
            continue

        feasible = True
        for j in cur_set:
            if sum_counts[i + j] >= u:
                feasible = False
                break
        if not feasible:
            continue

        # add i to the set and update counters
        for j in cur_set:
            sum_counts[i + j] += 1
        sum_counts[2 * i] += 1
        cur_set.append(i)

    # try to add any remaining positions (single pass)
    present = np.zeros(n, dtype=bool)
    present[cur_set] = True
    for i in range(n):
        if present[i]:
            continue
        if sum_counts[2 * i] >= u:
            continue
        feasible = True
        for j in cur_set:
            if sum_counts[i + j] >= u:
                feasible = False
                break
        if not feasible:
            continue
        for j in cur_set:
            sum_counts[i + j] += 1
        sum_counts[2 * i] += 1
        cur_set.append(i)
        present[i] = True

    return cur_set
