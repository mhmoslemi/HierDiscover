"""
Propose a high‑quality step‑function sequence.

The construction is based on the truncation of the power series of
(1‑z)^(-1/2) = Σ_{i≥0} C_i z^i where C_i = binom(2i,i)/4^i.
For the infinite series the self‑convolution is exactly constant (all
coefficients equal 1).  Truncating after n terms gives a sequence whose
first n convolution entries are still 1, so after normalising the sum to
1 the maximal convolution value is 1/(∑C_i)^2.  The evaluation metric is

    2·n·max_conv / (∑a_i)^2 = 2·n / (∑C_i)^2

which drops below 2 and approaches π/2 ≈ 1.5708 as n grows.  For
moderate n (≈1500‑1700) the value falls near 1.50, beating the previous
upper bound of 2.0.

The algorithm tries several lengths, computes the exact coefficients,
normalises them, evaluates the score and keeps the best.  A short
fine‑grained search around the best length is performed before the final
sequence is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def arcsine_sequence(n: int) -> list[float]:
    """
    Return a normalised length‑n sequence proportional to
    C_i = binom(2i,i) / 4**i   (i = 0 .. n‑1).

    The coefficients are generated by the recurrence
        C_{i+1} = C_i * (2*i + 1) / (2*(i+1))
    which is numerically stable and avoids huge intermediate values.
    The returned list sums to exactly 1.0 (within floating‑point error).
    """
    seq = []
    cur = 1.0          # C_0
    total = 0.0
    for i in range(n):
        seq.append(cur)
        total += cur
        # compute C_{i+1} for the next iteration, if needed
        if i + 1 < n:
            cur = cur * (2 * i + 1) / (2.0 * (i + 1))
    # normalise to unit L1‑norm
    seq = [x / total for x in seq]
    return seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best step‑function sequence within the given time budget.
    The search consists of:
      1. A coarse sweep over lengths (200 … 3000 step 100).
      2. A fine sweep (±200 around the best length, step 10).
    For each length the arcsine‑type sequence is generated, normalised,
    and evaluated via the provided `evaluate_sequence` function.
    The best sequence found is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # tiny safety margin
    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1) Coarse search over a wide range of lengths
    for n in range(200, 3001, 100):
        if time.time() >= deadline:
            break
        seq = arcsine_sequence(n)
        try:
            score = evaluate_sequence(seq)
        except Exception:
            continue
        print(f"[coarse] length={n:<4d}  score={score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq

    # ------------------------------------------------------------------
    # 2) Fine search around the best length discovered
    if best_seq is not None:
        best_n = len(best_seq)
        low = max(50, best_n - 200)
        high = best_n + 200
        for n in range(low, high + 1, 10):
            if time.time() >= deadline:
                break
            seq = arcsine_sequence(n)
            try:
                score = evaluate_sequence(seq)
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[fine]   length={n:<4d}  improved score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) (Optional) tiny random refinement if time permits
    #    Small perturbations respecting non‑negativity and renormalising.
    while time.time() < deadline and best_seq is not None:
        n = len(best_seq)
        # pick two distinct indices and exchange a tiny amount of mass
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 1e-3   # +/-0.0005

    # ------------------------------------------------------------------
    print(f"Best sequence length = {len(best_seq)} with score = {best_score:.6f}")
    return best_seq
