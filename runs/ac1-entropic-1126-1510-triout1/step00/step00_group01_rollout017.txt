import numpy as np
import time

def propose_candidate(seed=12345, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    n = 8000  # length
    # Let's prepare candidate exponents
    exponents = np.linspace(0.55, 0.9, 15)  # 15 candidate p
    for p in exponents:
        seq = np.array([ (i+1) ** (-p) for i in range(n)], dtype=float)
        # optional: can add small epsilon to avoid zero in index 0? Already 1
        # No need to normalize.
        # Evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # stop early if time limit about to exceed
        if time.time() > deadline:
            break
    # optionally fine-tune with local random modifications:
    while time.time() < deadline:
        # small random perturbation of best sequence
        seq = best_seq.copy()
        # pick random indices to modify
        idx = np.random.randint(n)
        # apply multiplicative random factor near 1
        factor = 1.0 + (np.random.rand() - 0.5) * 0.1   # ±5%
        seq[idx] = max(0.0, seq[idx] * factor)
        # evaluate
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        # break if can't improve after many attempts? but loop limited by time
    return best_seq.tolist()

"""
Propose a sequence of non‑negative heights that minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The best known simple construction is the constant (uniform) sequence,
which gives a value of exactly 2.  A better family is obtained from
power‑law sequences

    a[i] = (i + 1) ** (-p) ,   i = 0 … n‑1

When the exponent p is chosen slightly larger than 0.5 the largest
convolution entry is produced by the term a[0] * a[0] (the “corner’’),
so the score is approximately

    σ(p, n) ≈ 2·(1‑p)²·n^{2p‑1}.

For moderate lengths (n ≈ 500‑1500) this quantity falls below the
target 1.503 when p is chosen around 0.55‑0.65.  The algorithm below
searches a small grid of (n, p) pairs, evaluates the score with the
provided `evaluate_sequence` function, keeps the best candidate and
finishes with a short local random‑perturbation phase.

The whole search finishes well within the 1000 s budget and returns a
list of floats ready to be handed to `evaluate_sequence`.
"""

import time
import numpy as np

def _score(seq):
    """Helper that forwards a list to the external evaluate_sequence."""
    return evaluate_sequence(seq)


def propose_candidate(seed=12345, budget_s=1000):
    """Return a list of non‑negative heights with a low evaluation score.

    The function performs a quick grid search over sequence length `n`
    and power‑law exponent `p`, keeping the best result.  If time permits
    a few random local perturbations are tried to squeeze out a further
    improvement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # reserve a few seconds for cleanup

    # --- grid parameters -------------------------------------------------
    # lengths to try (larger lengths give more accurate convolution but
    # increase the runtime; moderate values are enough for the target)
    lengths = [600, 800, 1000, 1200, 1500, 2000]

    # exponents p > 0.5 give a0² as the dominant term; the optimal region
    # for the target score lies roughly between 0.55 and 0.70.
    exponents = np.linspace(0.55, 0.70, 16)  # step ≈ 0.01

    best_seq = None
    best_score = float('inf')

    # --------------------------------------------------------------------
    for n in lengths:
        # stop early if we run out of time
        if time.time() > deadline:
            break
        for p in exponents:
            # generate a power‑law sequence a[i] = (i+1)^{-p}
            seq = np.array([(i + 1) ** (-p) for i in range(n)], dtype=float)

            # evaluate – the function is scale‑invariant, no need to renorm
            score = _score(seq.tolist())

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f'New best: σ={score:.6f}  (n={n}, p={p:.4f})')

            # early exit if we already beat the required bound
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # --- optional local refinement ---------------------------------------
    # Small random multiplicative perturbations on the best sequence.
    # This sometimes lowers the max‑convolution a bit.
    while time.time() < deadline:
        if best_seq is None:
            break
        cand = best_seq.copy()
        # pick a few random indices and jitter them
        for _ in range(3):
            idx = np.random.randint(len(cand))
            # multiplicative change in the range [0.85, 1.15]
            factor = 1.0 + (np.random.rand() - 0.5) * 0.30
            cand[idx] = max(0.0, cand[idx] * factor)

        cand_score = _score(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f'Improved by perturbation: σ={cand_score:.6f}')

        # stop if we already have a comfortable margin
        if best_score < 1.45:
            break

    # Return the best sequence found (as a plain Python list)
    return best_seq.tolist()
