```
def improve_sequence(a, steps=5000, step_size=0.01):
    for iter in range(steps):
        conv = np.convolve(a, a)
        M = conv.max()
        k = conv.argmax()
        # gradient: for each i s.t. 0 <= i < n and 0 <= k-i < n, gradient_i = 2*a[k-i]
        grad = np.zeros_like(a)
        for i in range(len(a)):
            j = k - i
            if 0 <= j < len(a):
                grad[i] = 2 * a[j]
        # update
        a_new = a - step_size * grad
        # ensure nonnegative
        a_new = np.maximum(a_new, 0)
        # renorm to keep sum constant if needed
        s = a_new.sum()
        if s < 1e-6:
            a_new = a / np.linalg.norm(a)  # fallback
        else:
            a_new = a_new / s * a.sum()
        a = a_new
        # track best
```

```
def smooth_max_objective(a, beta=10):
    conv = np.convolve(a,a)
    max_approx = np.log(np.sum(np.exp(beta*conv))) / beta
    # gradient weights
    exp_beta = np.exp(beta*conv - beta*max_approx)  # but we need numerator = exp(beta*conv)
    w = exp_beta / np.sum(exp_beta)  # sum to 1
    # gradient = 2 * sum_k w_k * conv_gradient_k
    # conv_gradient_k_i = a_{k-i} if valid else 0
    # So gradient_i = 2 * sum_{k} w_k * a_{k-i} (valid)
    # = 2 * (correlation of a with w) at lag i
    corr = np.correlate(a, w, mode='full')
    # slice 
    n = len(a); m = len(w) = 2*n-1
    grad = 2 * corr[m-1 : m-1 + n]  # as earlier
    return max_approx, grad
```

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """Searches for the optimal exponent p in a power-law sequence a_i = (i+1)^{-p}
    to minimize the evaluation metric. Uses a coarse grid then a fine local search.
    """
    import numpy as np, time, math
    start = time.time()
    n = 5000  # length of sequence, large enough to approximate continuous limit
    indices = np.arange(1, n+1, dtype=np.float64)

    best_score = float('inf')
    best_seq = None
    best_p = None

    # Coarse grid search for exponent p
    for p in np.linspace(0.45, 0.75, 61):
        seq = indices ** (-p)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_p = p
        if time.time() - start > budget_s - 5:
            break

    # Fine grid around current best p if time permits
    if best_p is not None and time.time() - start < budget_s - 5:
        # explore +/- 0.02 around best_p with step 0.001
        lo = max(0.3, best_p - 0.03)
        hi = min(0.9, best_p + 0.03)
        for p in np.arange(lo, hi, 0.001):
            seq = indices ** (-p)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p = p
            if time.time() - start > budget_s - 5:
                break

    # Optional: normalize to ensure sum not too small (not required because scaling invariance)
    # but we keep original values.
    print(f"Best exponent p={best_p:.6f}, score={best_score:.6f}, length={n}")

    # Return as pure Python list
    return best_seq.tolist()

import numpy as np
import time
