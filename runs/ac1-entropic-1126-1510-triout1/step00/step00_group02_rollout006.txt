def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = 200  # Choose length
    # Initialize with Dirichlet distribution
    a = np.random.dirichlet(alpha=np.ones(n))
    # convert to list
    curr = a.tolist()
    best = curr.copy()
    best_score = evaluate_sequence(best)
    # Precompute initial temperature schedule
    init_temp = 0.1  # maybe relative to score magnitude

    while time.time() < deadline:
        # propose move
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # compute amount to move: delta fraction of a[i]
        max_delta = min(curr[i], 0.5*curr[i])  # ensure non-negative
        delta = (np.random.rand() - 0.5) * max_delta  # could be negative, but only move from i to j
        # Actually we want to move from i to j: take delta positive from i to j
        delta = np.random.rand() * curr[i] * 0.5  # move up to half of i's mass to j
        new_i = curr[i] - delta
        new_j = curr[j] + delta
        if new_i < 0:
            continue
        new_seq = curr.copy()
        new_seq[i] = new_i
        new_seq[j] = new_j
        # maybe also perform small random noise on others
        # evaluate
        new_score = evaluate_sequence(new_seq)
        if new_score < best_score:
            best_score = new_score
            best = new_seq.copy()
            print(f"New best {best_score}")
        # acceptance

"""
propose_candidate: simulated‑annealing / gradient descent search for a
non‑negative step function that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works on a moderate sized vector (default n = 500) and
keeps the total mass equal to 1 (scaling does not affect the score).
At each iteration it:

1. Computes the full convolution using an FFT (O(n log n)).
2. Finds the index k* where the convolution attains its maximum.
3. Builds the sub‑gradient of the quantity max(conv) w.r.t. the
   coefficients (for the active index k*).
4. Moves a small amount of mass from the coefficient with the largest
   gradient (largest contribution to the peak) to the one with the
   smallest gradient.
5. Accepts the move if it improves the score, otherwise accepts it
   with a temperature‑dependent probability (simulated annealing).

Occasionally a random jitter is added to avoid local minima.
The best sequence seen during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def max_convolution(seq: np.ndarray) -> float:
    """
    Return max(conv(seq, seq)) using an FFT based linear convolution.
    seq is a 1‑D ndarray of non‑negative floats.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    # Real FFT is enough because inputs are real.
    fft_seq = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)[: 2 * n - 1]
    return float(conv.max())


def gradient_for_peak(seq: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the peak value max(conv(seq, seq)) at the active
    convolution index `peak_idx`.  For an ordered pair (i,j) with i+j=k,
    the derivative w.r.t. a_i receives a contribution a_j, and the same
    contribution appears again when the roles are swapped.  Consequently:

        g_i = 2 * a_{k-i}   if 0 ≤ k-i < n and i ≠ k-i
        g_i = a_i           if i == k-i (diagonal term)
        g_i = 0             otherwise
    """
    n = seq.shape[0]
    i = np.arange(n)
    j = peak_idx - i
    mask = (j >= 0) & (j < n)

    grad = np.zeros_like(seq)

    # diagonal case (i == j)
    diag = mask & (i == j)
    grad[diag] = seq[diag]                     # derivative of a_i^2

    # off‑diagonal case (i != j)
    off = mask & (i != j)
    # For those positions we need a_{k-i}
    grad[off] = 2.0 * seq[j[off]]

    return grad


def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence for speed."""
    # The external function works on Python lists; converting once is cheap.
    return evaluate_sequence(seq.tolist())


# ----------------------------------------------------------------------
# Main candidate proposal
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    The routine runs for roughly `budget_s` seconds, then returns the
    best sequence it discovered.
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Parameters (feel free to tweak)
    # ------------------------------------------------------------------
    n = 500                     # length of the sequence we optimise
    total_mass = 1.0            # we keep sum(a) = 1 (scale invariant)
    max_iter = 10_000_000       # safety cap; loop breaks on time out
    init_step = 0.15            # initial fraction of source mass to move
    step_decay = 0.997          # geometric decay per successful move
    jitter_prob = 0.02          # occasional random perturbation probability
    # ------------------------------------------------------------------

    deadline = time.time() + budget_s - 5.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # Initial sequence: Dirichlet(1) gives a random probability vector.
    # Use the globally supplied sequence as a possible fallback.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        # try the existing construction as a starting point (scaled to sum=1)
        init_seq = np.array(height_sequence_1, dtype=float)
        if init_seq.sum() < 1e-12:          # degenerate, fall back to random
            init_seq = np.random.dirichlet(np.ones(n))
        else:
            init_seq = init_seq / init_seq.sum()
        # If the length differs from our chosen n we truncate / pad with zeros.
        if init_seq.size > n:
            init_seq = init_seq[:n]
        elif init_seq.size < n:
            pad = np.zeros(n - init_seq.size)
            init_seq = np.concatenate([init_seq, pad])
    else:
        init_seq = np.random.dirichlet(np.ones(n))

    # Ensure the sum is exactly total_mass (scale‑invariant objective)
    init_seq = init_seq * (total_mass / init_seq.sum())
    current = init_seq.copy()
    best_seq = current.copy()
    best_score = evaluate(best_seq)

    # Main optimisation loop -------------------------------------------------
    step = init_step
    iteration = 0

    while time.time() < deadline and iteration < max_iter:
        # ----- compute convolution and locate the current peak -------------
        peak_val = max_convolution(current)
        # (evaluate() would recompute the same quantity; we reuse it)
        # compute the raw score for the current sequence (needed for acceptance)
        curr_score = 2.0 * n * peak_val / (current.sum() ** 2)

        # ----- gradient of the active peak ---------------------------------
        # find the index where the convolution attains the maximum.
        # We recompute the convolution with FFT again only to locate its argmax.
        size = 1 << ((2 * n - 1).bit_length())
        fft_curr = np.fft.rfft(current, n=size)
        conv_full = np.fft.irfft(fft_curr * fft_curr, n=size)[: 2 * n - 1]
        peak_idx = int(conv_full.argmax())

        grad = gradient_for_peak(current, peak_idx)

        # ----- choose source (large gradient) and target (small gradient) --
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            # degenerate case – fall back to random indices
            src, tgt = np.random.choice(n, size=2, replace=False)

        # ----- propose a move: shift a fraction of mass from src to tgt -----
        delta = step * current[src]          # move up to `step` of src's mass
        if delta <= 0.0:
            # nothing to move; break to avoid endless loop
            break
        proposal = current.copy()
        proposal[src] -= delta
        proposal[tgt] += delta

        # Numerical safety (non‑negativity)
        if proposal[src] < 0.0:
            proposal[tgt] += proposal[src]   # give back the negative part
            proposal[src] = 0.0

        # ----- occasionally apply a random jitter -------------------------
        if np.random.rand() < jitter_prob:
            idx = np.random.randint(n)
            eps = (np.random.rand() - 0.5) * step * current[idx]
            proposal[idx] = max(0.0, proposal[idx] + eps)

        # ----- evaluate the proposal ---------------------------------------
        try:
            prop_score = evaluate(proposal)
        except Exception:
            # In the very unlikely event evaluate fails, reject the proposal
            prop_score = float("inf")

        # ----- acceptance rule (simulated annealing) -----------------------
        if prop_score < curr_score:
            accept = True
        else:
            # temperature decays with the step size; smaller step → lower T
            temperature = max(step, 1e-6)
            prob = np.exp(-(prop_score - curr_score) / temperature)
            accept = np.random.rand() < prob

        if accept:
            current = proposal
            # keep shrinking the step size when an improvement occurs;
            # this acts like a simple learning‑rate schedule.
            step *= step_decay
            if prop_score < best_score:
                best_score = prop_score
                best_seq = proposal.copy()
                print(
                    f"[{iteration:06d}] New best score: {best_score:.6f} "
                    f"(step={step:.5f})"
                )
        else:
            # occasional random restart if we are stuck for a long time
            if iteration % 5000 == 0 and iteration > 0:
                # re‑initialise from a fresh random Dirichlet vector
                current = np.random.dirichlet(np.ones(n))
                current = current * (total_mass / current.sum())

        iteration += 1

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    return best_seq.tolist()
