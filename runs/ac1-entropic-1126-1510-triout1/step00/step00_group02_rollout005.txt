import numpy as np
import time

def next_pow2(x):
    return 1 << (x - 1).bit_length()

def compute_convolution_fft(a):
    n = a.shape[0]
    m = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv[:2 * n - 1]

def objective(a):
    n = a.shape[0]
    conv = compute_convolution_fft(a)
    M = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * M / (s * s))

def max_and_subgradient(a):
    n = a.shape[0]
    conv = compute_convolution_fft(a)
    M = conv.max()
    idxs = np.where(conv == M)[0]
    grad = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        grad[i_range] += 2.0 * a[j_range]
    grad /= len(idxs)
    return M, grad

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Subgradient descent optimizer for the step‑function upper‑bound.
    Starts from an exponential sequence (parameter t≈1) or from the
    provided height_sequence_1 if available, then iteratively moves
    away from the current maximal convolution index while keeping the
    ℓ1‑norm fixed (≈√(2n)).  The best sequence observed during the
    1000 s budget is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # leave margin

    # Determine length
    try:
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=float)
        print(f"Using provided height_sequence_1 of length {n}")
    except NameError:
        n = 3000
        base_seq = None
        print(f"No global height_sequence_1, using length {n}")

    target_sum = np.sqrt(2.0 * n)

    # ---- initialise sequence -------------------------------------------------
    if base_seq is not None and np.random.rand() < 0.6:
        seq = base_seq.copy()
        # normalise sum
        if seq.sum() == 0:
            seq = np.random.rand(n)
        seq = seq * (target_sum / max(seq.sum(), 1e-12))
        print("Start from height_sequence_1")
    else:
        # exponential shape a_i = r**i
        t_param = 1.0   # gives r = 1 - t/n
        r = 1.0 - t_param / n
        seq = np.array([r ** i for i in range(n)], dtype=float)
        seq = seq * (target_sum / seq.sum())
        print("Start from exponential shape")

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x must be > 0)."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the full discrete convolution a * a using an FFT (real‐valued).
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    m = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Fast evaluation of the same quantity as ``evaluate_sequence`` but
    without the security clamping (the algorithm keeps the vector inside
    the allowed domain).  Returns a float; ``np.inf`` if the sum is too
    small.
    """
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.shape[0]
    conv = _conv_fft(a)
    M = conv.max()
    return float(2.0 * n * M / (s * s))


def _max_and_subgrad(a: np.ndarray):
    """
    Compute the current maximal convolution value and a subgradient of the
    (non‑smooth) max‑function at the current point.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    M = conv.max()
    # treat values within a tiny tolerance as maximisers
    max_idx = np.where(np.isclose(conv, M, rtol=0, atol=1e-12))[0]

    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        j_range = k - i_range
        grad[i_range] += 2.0 * a[j_range]
    grad /= float(len(max_idx))
    return M, grad

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin
