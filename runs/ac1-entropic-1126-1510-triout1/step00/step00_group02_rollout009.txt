"""
Propose a step‑function that minimises the evaluation metric.

The algorithm works in three stages:

1. **Initialisation** – start from the provided best construction
   (``height_sequence_1``) if it exists, otherwise from a uniform vector.

2. **Parametric shape search** – generate many candidate sequences from
   simple monotone families (exponential, linear, power‑law).  Each candidate
   is scaled so that its ℓ¹‑norm equals √(2·n) (the same normalisation used
   in the reference implementation).  The best candidate found so far is
   stored.

3. **LP‑guided refinement** – repeatedly solve the linear programme that
   finds a direction ``g`` with a larger ℓ¹‑sum while keeping the
   convolution ``f*g`` bounded by the current maximum ``‖f*f‖∞``.
   For the current best vector ``f`` we compute the optimal mixing weight
   ``t∈[0,1]`` (by a tiny line‑search) that minimises the objective after
   the update ``h = (1‑t)·f + t·g``.  The update is applied and the best
   vector is updated if the objective improves.

The search runs for the allotted time (default 1000 s) and finally returns
the best sequence found.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: fast evaluation of the objective (identical to evaluate_sequence
# but without the extra safety checks – all our candidates satisfy the
# required constraints).
def fast_evaluation(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    sum_a = np.sum(seq)
    if sum_a < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Linear programme used by the SOTA update step.
def solve_convolution_lp(f_seq, rhs):
    """
    Solve:
        maximise   Σ b_j
        subject to conv(f_seq, b) ≤ rhs   (for every shift)
                b ≥ 0
    Returns the optimal array b (as a 1‑D numpy array) or ``None`` if
    the optimiser fails.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise Σ b_j  → minimise -Σ b_j
    a_ub = []
    b_ub = []

    # convolution constraints: for each possible shift k
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  –b ≤ 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # linprog’s ``highs`` method is robust; we set a modest time limit.
    opts = {"disp": False, "time_limit": 8.0}
    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method="highs", options=opts)

    if res.success:
        # clip tiny negatives that may appear due to numerical noise
        return np.maximum(res.x, 0.0)
    return None


# ----------------------------------------------------------------------
def lp_refine(seq, deadline):
    """
    Perform one LP‑guided refinement step on ``seq``.
    Returns a new sequence (list) if an improvement is possible,
    otherwise ``None``.
    """
    n = len(seq)
    if n > 2000:          # LP grows quadratically – keep it tractable
        return None

    seq = np.asarray(seq, dtype=float)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return None

    # Normalise to the standard ℓ¹‑norm √(2n)
    target_norm = np.sqrt(2.0 * n)
    scale = target_norm / sum_seq
    f = seq * scale

    # rhs = ‖f*f‖∞
    rhs = np.max(np.convolve(f, f))

    # Solve the LP for a promising direction g₀
    g0 = solve_convolution_lp(f.tolist(), rhs)
    if g0 is None:
        return None

    # Rescale g₀ so that it also has norm √(2n)
    sum_g0 = np.sum(g0)
    if sum_g0 < 1e-12:
        return None
    g = g0 * (target_norm / sum_g0)

    # Pre‑compute the three convolutions needed for any mixing weight t
    conv_f = np.convolve(f, f)
    conv_g = np.convolve(g, g)
    conv_fg = np.convolve(f, g)

    # Because both f and g have the same ℓ¹‑norm, the denominator of the
    # objective stays constant ( = target_norm**2 ), so we only have to
    # minimise the peak of the mixed convolution.
    best_t = 0.0
    best_val = float("inf")

    # Coarse grid search
    for t in np.linspace(0.0, 1.0, 21):
        if time.time() > deadline:
            break
        mixed = ((1.0 - t) ** 2) * conv_f + 2.0 * t * (1.0 - t) * conv_fg + (t ** 2) * conv_g
        max_b = np.max(mixed)
        val = 2.0 * n * max_b / (target_norm ** 2)
        if val < best_val:
            best_val = val
            best_t = t

    # Refine locally around the best coarse point
    lo = max(0.0, best_t - 0.05)
    hi = min(1.0, best_t + 0.05)
    for t in np.linspace(lo, hi, 41):
        if time.time() > deadline:
            break
        mixed = ((1.0 - t) ** 2) * conv_f + 2.0 * t * (1.0 - t) * conv_fg + (t ** 2) * conv_g
        max_b = np.max(mixed)
        val = 2.0 * n * max_b / (target_norm ** 2)
        if val < best_val:
            best_val = val
            best_t = t

    # Build the new candidate in the original (un‑scaled) space
    g_original_scale = g * (sum_seq / target_norm)     # back to original ℓ¹‑norm
    new_seq = (1.0 - best_t) * seq + best_t * g_original_scale
    return new_seq.tolist()


# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the supplied ``evaluate_sequence`` function.  The routine runs for at
    most ``budget_s`` seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Initialise with the known construction if it exists, otherwise uniform.
    try:
        best_seq = list(height_sequence_1)          # provided globally
        best_score = evaluate_sequence(best_seq)
        print(f"Starting from supplied height_sequence_1, score={best_score:.6f}")
    except Exception:
        n0 = 1200
        uniform_val = np.sqrt(2.0 * n0) / n0
        best_seq = [float(uniform_val)] * n0
        best_score = evaluate_sequence(best_seq)
        print(f"No supplied sequence; start uniform (n={n0}), score={best_score:.6f}")

    best_seq = np.asarray(best_seq, dtype=float)

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate quickly and update the global best.
    def try_candidate(seq):
        nonlocal best_seq, best_score
        seq = np.asarray(seq, dtype=float)
        if seq.ndim != 1 or np.any(seq < 0):
            return
        score = fast_evaluation(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  New best: {best_score:.6f} (len={len(seq)})")
        return

    # ------------------------------------------------------------------
    # Stage 1 – parametric shape exploration
    shape_kinds = ("exponential", "linear", "power")
    # a few lengths that keep the LP cheap but still give good potential
    lengths = [500, 800, 1200, 2000]

    while time.time() < deadline:
        # pick a random shape family and a length
        n = int(np.random.choice(lengths))
        kind = np.random.choice(shape_kinds)

        if kind == "exponential":
            # r in (0.5, 0.999) – the smaller r, the faster the decay
            r = np.random.uniform(0.5, 0.9995)
            seq = np.array([r ** i for i in range(n)], dtype=float)

        elif kind == "linear":
            # decreasing line from a0 to 0 (or close)
            a0 = np.random.uniform(0.5, 2.0)          # height before scaling
            slope = np.random.uniform(0.0, a0 / (n - 1))
            seq = np.maximum(0.0, a0 - slope * np.arange(n, dtype=float))

        else:   # power‑law
            p = np.random.uniform(0.2, 3.0)           # exponent
            seq = np.array([ (i + 1) ** (-p) for i in range(n) ], dtype=float)

        # Scale to the reference ℓ¹‑norm √(2·n)
        s = np.sum(seq)
        if s < 1e-12:
            continue
        target = np.sqrt(2.0 * n)
        seq = seq * (target / s)

        try_candidate(seq)

        # Occasionally perform a tiny random perturbation of the current best
        if np.random.rand() < 0.01:
            pert = best_seq.copy()
            idx = np.random.randint(pert.size)
            pert[idx] = max(0.0, pert[idx] + np.random.randn() * 0.05)
            try_candidate(pert)

        # Every now and then launch a LP‑guided refinement on the best so far
        if np.random.rand() < 0.02:
            refined = lp_refine(best_seq.tolist(), deadline)
            if refined is not None:
                try_candidate(refined)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    print(f"Finished search – best score {best_score:.6f}, length {len(best_seq)}")
    return best_seq.tolist()
