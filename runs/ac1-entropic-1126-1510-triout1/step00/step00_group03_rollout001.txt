"""
Step‑function optimizer.

The algorithm works in three stages:

1) **Parametric initialization** – generate candidates from the
   family a_i = 1 / ((i+0.5)*(n-0.5-i))**p   (p≈0.5 gives the
   classic arcsine shape).  Small values of `p` flatten the curve,
   larger values concentrate mass at the ends.  For several lengths
   (300, 500, 800) and a grid of exponents the best initial sequence
   (lowest `evaluate_sequence` score) is kept.

2) **Local stochastic refinement** – starting from the best candidate,
   repeatedly perturb a random subset of entries by a modest multiplicative
   factor (±10 %).  After each perturbation the whole vector is renormalised
   to keep the total sum unchanged (the score is scale‑invariant).
   Whenever a perturbation yields a lower score, it is accepted.

3) **Return** – after the allotted time (default 1000 s) the best
   sequence found is returned as a plain list of floats.

All numbers are non‑negative, well below the clipping thresholds of
`evaluate_sequence`, and the routine respects the time budget.
"""

import time
import numpy as np

# `evaluate_sequence` is provided in the execution environment.


def _eval(seq_arr: np.ndarray) -> float:
    """Convenient wrapper – `evaluate_sequence` expects a Python list."""
    return evaluate_sequence(seq_arr.tolist())


def _generate_family(n: int, p: float) -> np.ndarray:
    """
    Produce the sequence a_i = 1 / ((i+0.5)*(n-0.5-i))**p, i = 0..n‑1.
    The function returns a NumPy array (float64).
    """
    idx = np.arange(n, dtype=np.float64)
    denom = (idx + 0.5) * (n - 0.5 - idx)
    # `denom` is always > 0 because of the +0.5 shift.
    seq = 1.0 / np.power(denom, p)
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function.
    The function must finish before `budget_s` seconds and return a list of
    non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # keep a small safety margin

    # ------------------------------------------------------------------
    # 1)  Parametric sweep (different lengths and exponents)
    # ------------------------------------------------------------------
    lengths = [300, 500, 800]               # moderate sizes – quick conv.
    exponents = np.linspace(0.20, 0.80, 7)   # 0.20, 0.30, … , 0.80

    best_seq = None
    best_score = float("inf")
    best_len = None
    best_exp = None

    for n in lengths:
        for p in exponents:
            seq = _generate_family(n, p)
            # Small numerical safety – ensure positivity.
            seq = np.maximum(seq, 0.0)
            # Optional renormalisation (scale does not affect the score).
            # We keep the raw values to preserve the shape.
            score = _eval(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_len = n
                best_exp = p
                print(
                    f"[init] n={n}, p={p:.2f} → score={score:.6f}"
                )
        # give a tiny chance to stop early if time runs out
        if time.time() >= deadline:
            print("[init] time exhausted")
            return best_seq.tolist()

    print(
        f"Best after grid search: n={best_len}, p={best_exp:.2f}, "
        f"score={best_score:.6f}"
    )

    # ------------------------------------------------------------------
    # 2)  Stochastic hill‑climbing refinement
    # ------------------------------------------------------------------
    # The score is invariant under a global scaling, therefore we keep the
    # total sum constant during the search (helps numerical stability).
    target_sum = best_seq.sum()
    seq_len = best_seq.shape[0]

    # Pre‑compute a few parameters for the perturbation loop.
    max_modifications = max(1, seq_len // 100)   # modify up to ~1 % each step
    factor_range = 0.10                         # ±10 % multiplicative change

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --- create a neighbour ----------------------------------------
        cand = best_seq.copy()

        # how many positions to touch this iteration?
        k = rng.integers(1, max_modifications + 1)
        indices = rng.choice(seq_len, size=k, replace=False)

        # multiplicative perturbation in (1‑δ, 1+δ)
        factors = 1.0 + (rng.random(k) - 0.5) * 2.0 * factor_range
        cand[indices] *= factors

        # non‑negativity (clip tiny negatives that may appear due to rounding)
        np.maximum(cand, 0.0, out=cand)

        # renormalise to original total sum (scale‑invariance)
        cand_sum = cand.sum()
        if cand_sum < 1e-8:          # avoid division by zero – reject this neighbour
            continue
        cand *= target_sum / cand_sum

        # --- evaluate --------------------------------------------------
        score = _eval(cand)

        # improvement ?
        if score < best_score:
            best_score = score
            best_seq = cand
            print(
                f"[iter {iteration:05d}] new best = {best_score:.6f} "
                f" (elapsed {time.time() - start:.1f}s)"
            )

        # occasional small chance to re‑seed with a fresh random shape
        if iteration % 2000 == 0 and rng.random() < 0.05:
            # fresh random start (same length) – may escape plateaus
            fresh = _generate_family(seq_len, rng.uniform(0.2, 0.8))
            fresh = np.maximum(fresh, 0.0)
            fresh *= target_sum / fresh.sum()
            fresh_score = _eval(fresh)
            if fresh_score < best_score:
                best_score = fresh_score
                best_seq = fresh
                print(
                    f"[reset] refreshed candidate -> score {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3)  Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"Finished. Best score = {best_score:.6f} (length {best_seq.shape[0]})")
    # Convert to a plain Python list as required.
    return best_seq.tolist()
