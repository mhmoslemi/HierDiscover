        try:
            score = evaluate_sequence(seq)
            if score < best_score - 1e-12:
                best_score = score
                best_seq = seq.copy()
                print(f"New best score: {best_score:.6f} len {len(seq)}")

            # optional normalize sum to 1 (not needed)

        n = len(seq)
        sum_seq = np.sum(seq)
        if sum_seq < 1e-12:
            return None
        normalized = seq * np.sqrt(2 * n) / sum_seq
        rhs = np.max(np.convolve(normalized, normalized))
        g = solve_convolution_lp(normalized, rhs)

        g_norm = g * np.sqrt(2 * n) / sum_g
        new_seq = [(1 - t) * x + t * y for x, y in zip(seq, g_norm)]
        return new_seq

    def gradient_step(seq, step=0.01):
        n = len(seq)
        conv = np.convolve(seq, seq)
        k = int(np.argmax(conv))

        grad = np.zeros(n)

        # renormalize to original sum
        total = new_seq.sum()

        new_seq = new_seq * (np.sum(seq) / total)
        return new_seq.tolist()

    current_max = max_convolution(seq)

    best_seq = seq.copy()
    best_score = current_score

    step_factor = 0.05
    temp = 0.01  # maybe adjust
    iteration = 0
    while time.time() < deadline:
        # pick random i,j
        i,j = np.random.randint(n, size=2)
        if i == j:
            continue
        # compute delta, ensure nonnegative
        delta = step_factor * seq[i]
        if delta <= 0:
            continue

        # no need renormalize sum preserved

        new_max = max_convolution(new_seq)

        if new_score < current_score:

            # maybe accept with probability
            prob = np.exp( -(new_score - current_score) / max(temp, 1e-12) )

        if accept:
            seq = new_seq
            current_score = new_score

"""
Propose a candidate step‐function sequence.

The algorithm starts from a smooth exponential profile (the known near‑optimal
shape) and then runs a fast simulated‑annealing / greedy local search that
moves a tiny amount of mass from indices that contribute most to the current
maximum of the auto‑convolution to indices that contribute least.  The score
to minimise is the exact value returned by ``evaluate_sequence``:
    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

Because the objective is scale‑invariant we normalise every candidate to
sum(a)=1 – this lets us compute the score cheaply as ``2*n*max_conv``.
The search works entirely with NumPy FFTs (O(n log n) per iteration) and
fits comfortably into the 1000 s budget.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions – must be defined at the top level (no closures)
# --------------------------------------------------------------

def max_convolution(seq: np.ndarray) -> float:
    """
    Return max(a * a) where * is the linear convolution.
    Uses real FFT for speed: O(n log n).
    """
    n = seq.shape[0]
    size = 2 * n - 1                     # length of the linear convolution
    fft_seq = np.fft.rfft(seq, n=size)   # zero‑pad to the required length
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    return float(np.max(conv))


def score_of(seq: np.ndarray) -> float:
    """
    The exact score used by ``evaluate_sequence`` when the sequence
    already has sum(seq)=1 (the function is scale‑invariant).
    """
    n = seq.shape[0]
    return 2.0 * n * max_convolution(seq)


def greedy_move(current: np.ndarray, step_frac: float) -> np.ndarray:
    """
    Perform a deterministic move that reduces the current maximal
    convolution entry:
        – locate the shift k where the convolution attains its maximum;
        – find the index i that contributes the largest product a_i*a_{k-i};
        – move a small fraction of mass from i to the index with the
          smallest current value.
    The total sum stays unchanged and all entries stay non‑negative.
    """
    n = current.shape[0]
    size = 2 * n - 1
    # convolution (needed only for the index of the maximum)
    fft_seq = np.fft.rfft(current, n=size)
    conv = np.fft.irfft(fft_seq * fft_seq, n=size)
    k = int(np.argmax(conv))

    # contributions to this shift
    contrib = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            contrib[i] = current[i] * current[j]

    i_max = int(np.argmax(contrib))
    # pick a destination with minimal value (to keep the distribution flat)
    min_val = np.min(current)
    candidates = np.where(current == min_val)[0]
    j_min = int(np.random.choice(candidates))

    delta = step_frac * current[i_max]
    if delta <= 0.0:
        return current

    new_seq = current.copy()
    new_seq[i_max] -= delta
    new_seq[j_min] += delta
    return new_seq


def random_swap(current: np.ndarray, step_frac: float) -> np.ndarray:
    """
    Randomly pick two indices i, j and transfer a fraction of the mass
    from i to j (keeping the sum fixed).  Used as the stochastic move in
    the annealing loop.
    """
    n = current.shape[0]
    i, j = np.random.randint(n, size=2)
    if i == j:
        return current
    delta = step_frac * current[i]
    if delta <= 0.0:
        return current
    new_seq = current.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    return new_seq

        # Metropolis acceptance criterion
        if new_score < current_score or \
           np.exp(-(new_score - current_score) / max(temperature, 1e-12)) > np.random.rand():
            current_seq = new_seq
            current_score = new_score

            # Record improvement
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {it:8d} – new best score: {best_score:.6f}")

        # Gradually cool the temperature and shrink the step size
        it += 1
        if it % 2000 == 0:
            temperature *= 0.95
            step_frac *= 0.95

        # Occasionally re‑seed the search to escape possible stagnation
        if it % 20000 == 0:
            rnd = np.random.rand(n)
            rnd /= np.sum(rnd)
            current_seq = rnd
            current_score = score_of(current_seq)

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    # ``evaluate_sequence`` requires non‑negative numbers; ensure that
    # the final sequence respects the additional clipping performed there.
    final = np.clip(best_seq, 0.0, None)

    # In the very unlikely event that the sum became too small after clipping,
    # inject a tiny uniform background to keep the denominator away from zero.
    if final.sum() < 0.01:
        final += (0.01 - final.sum()) / final.size

    return final.tolist()
