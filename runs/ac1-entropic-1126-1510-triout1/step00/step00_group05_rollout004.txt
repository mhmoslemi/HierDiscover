def gradient(a):

    max_b = b.max()

"""
Propose a sequence of non‑negative heights that minimizes the evaluation
function

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm is a simple stochastic mirror‑descent on the probability
simplex.  The objective is approximated by the smooth
log‑sum‑exp of the convolution values, which converges to the
ℓ∞‑norm as the temperature parameter α grows.
Updates are performed with an exponentiated‑gradient step and a
renormalisation onto the simplex.  Several random restarts with different
lengths are run until the time budget (≈ 1000 s) expires; the best sequence
found is returned.

Key ideas
* Use FFT‑based convolution for speed (O(n log n) per iteration).
* Replace the nonsmooth max by a softmax (log‑sum‑exp) with a temperature
  α that is increased during the run.
* Perform mirror‑descent (exponentiated gradient) which naturally keeps
  the coefficients non‑negative and enforces Σa = 1 by a final
  normalisation.
* Keep the best sequence (according to the provided ``evaluate_sequence``
  function) and periodically output progress.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Convolution via FFT (real‐valued input). Returns the full
    discrete convolution of length len(a)+len(b)-1.
    """
    n = len(a) + len(b) - 1
    size = _next_pow_two(n)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:n]


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Softmax (log‑sum‑exp) weights on the convolution vector ``b``.
    The maximum of ``b`` is subtracted for numerical stability.
    """
    maxb = b.max()
    exp_part = np.exp(alpha * (b - maxb))
    total = exp_part.sum()
    if total == 0.0:
        # fallback – uniform weights
        return np.full_like(b, 1.0 / b.size)
    return exp_part / total


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max objective with respect to ``a``.
    It is the (partial) convolution of the weight vector ``w`` (length
    2·n‑1) with the reversed probability vector ``a``.
    """
    a_rev = a[::-1]
    conv = _fft_convolve(w, a_rev)          # length = 3·n‑2
    n = a.size
    # slice that corresponds to indices 0 … n‑1 of the gradient
    return conv[n - 1 : n - 1 + n]


def _mirror_descent_step(a: np.ndarray, eta: float, alpha: float) -> np.ndarray:
    """
    Perform one mirror‑descent (exponentiated‑gradient) step.
    """
    # convolution a * a
    b = _fft_convolve(a, a)
    # soft‑max weights approximating the ℓ∞‑norm of the convolution
    w = _softmax_weights(b, alpha)
    # gradient w.r.t. a
    g = _gradient_from_weights(a, w)
    # exponentiated update (keeps positivity)
    a_new = a * np.exp(-eta * g)
    # renormalise onto the simplex Σa = 1
    s = a_new.sum()
    if s == 0.0 or np.isnan(s):
        # restart from a fresh random point if something went wrong
        a_new = np.random.rand(a.size)
        a_new /= a_new.sum()
    else:
        a_new /= s
    return a_new


def _run_descent(
    n: int,
    time_deadline: float,
    best_global_seq: np.ndarray | None,
    best_global_score: float,
) -> tuple[np.ndarray, float]:
    """
    Run a short mirror‑descent optimisation for a given length ``n``.
    Returns the best local sequence and its score.
    """
    # Initialise a random probability vector; sometimes warm‑start from the
    # current global best (truncated / padded).
    if best_global_seq is not None and random.random() < 0.2:
        if best_global_seq.size >= n:
            a = best_global_seq[:n].copy()
        else:
            pad = np.random.rand(n - best_global_seq.size)
            a = np.concatenate([best_global_seq, pad])
    else:
        a = np.random.rand(n)

    a = a.astype(np.float64)
    a /= a.sum()

    # hyper‑parameters (these are deliberately simple and cheap)
    eta0 = 2.0                # initial step size
    alpha_start = 0.5         # start temperature for softmax
    alpha_end = 50.0          # final temperature (sharper max approximation)

    # Determine a cheap iteration budget that scales with n.
    max_iters = int(20000 / max(1, n)) + 500

    best_local_seq = a.copy()
    try:
        best_local_score = evaluate_sequence(a.tolist())
    except Exception:
        best_local_score = float("inf")

    for it in range(max_iters):
        if time.time() > time_deadline:
            break

        # Step‑size schedule (decreasing) and temperature schedule (increasing)
        eta = eta0 / np.sqrt(it + 1)
        t = it / max_iters
        alpha = alpha_start + (alpha_end - alpha_start) * t

        a = _mirror_descent_step(a, eta, alpha)

        # Evaluate only every few iterations (reduces overhead).
        if it % 20 == 0:
            try:
                score = evaluate_sequence(a.tolist())
            except Exception:
                continue

            if score < best_local_score:
                best_local_score = score
                best_local_seq = a.copy()
                # Occasionally print progress
                print(
                    f"[{time.time():.1f}] n={n} iter={it} "
                    f"local_best={score:.6f} global_best={best_global_score:.6f}"
                )
                # Early exit if we already beat the target strongly
                if score < 1.5:
                    return best_local_seq, best_local_score

    return best_local_seq, best_local_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds,
    performs many random restarts of a mirror‑descent optimiser (see above),
    and finally returns the best list of floats found.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # leave a safety margin so that we can always return a result
    deadline = start_time + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # A collection of candidate lengths – the exact value is not crucial.
    # We also add the length of the previously known sequence if available.
    lengths = [400, 600, 800, 1000, 1200, 1500, 1800, 2000]
    try:
        if height_sequence_1 is not None:
            lengths.append(len(height_sequence_1))
    except NameError:
        pass

    # Main optimisation loop – keep trying new random starts until time runs out.
    while time.time() < deadline:
        n = int(random.choice(lengths))

        # Run a short descent for this length.
        seq, score = _run_descent(
            n=n,
            time_deadline=deadline,
            best_global_seq=best_seq,
            best_global_score=best_score,
        )

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"*** New global best: {best_score:.6f} (length={n}) ***")
            # If we already beat the target we can finish early.
            if best_score < 1.5:
                break

    # Fallback – uniform distribution (should never be needed, but protects
    # against an empty result).
    if best_seq is None:
        best_seq = np.ones(100) / 100.0

    # Return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()
