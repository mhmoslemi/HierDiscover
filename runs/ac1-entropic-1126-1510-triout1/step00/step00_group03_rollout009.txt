def max_feasible_t(f, g, M):
    b_f = np.convolve(f, f)
    b_g = np.convolve(g, g)
    b_fg = np.convolve(f, g)
    # Compute roots for each k
    t_max = 1.0
    for i in range(len(b_f)):
        A = b_f[i] - 2 * b_fg[i] + b_g[i]
        B = -2 * b_f[i] + 2 * b_fg[i]
        C = b_f[i] - M
        # inequality: A t^2 + B t + C <= 0, with C <= 0
        if A == 0:
            if B == 0:
                # no restriction (C <=0)
                continue
            else:
                t_candidate = -C / B
                if B > 0:
                    t_max = min(t_max, t_candidate)
                # if B < 0, inequality holds for all t <= +inf (since linear decreasing)
                # nothing to restrict
        else:
            disc = B * B - 4 * A * C
            if disc < 0:
                # due to numerical errors treat as zero
                disc = 0.0
            sqrt_disc = math.sqrt(disc)
            # Compute the two roots
            t1 = (-B - sqrt_disc) / (2 * A)
            t2 = (-B + sqrt_disc) / (2 * A)
            # Determine interval where inequality holds for t >= 0
            # Since at t=0 we have C <=0, the inequality holds near zero.

            # We take the smallest positive root > 0 as bound.
            # Evaluate which root >0:
            feasible_t = None
            for t_root in (t1, t2):
                if t_root > 0:
                    feasible_t = t_root
                    break
            if feasible_t is not None:

    return max(0.0, min(1.0, t_max * 0.9999)) # safety margin

def solve_lp_max_sum(f_seq, M):
    n = len(f_seq)
    c = -np.ones(n)   # maximize sum -> minimize negative sum
    # constraints: conv(f, g) <= M  for each k
    # Build A_ub rows: each row corresponds to constraints a_i * g_{k-i} sum <= M
    rows = []
    ub = []
    for k in range(2*n -1):
        row = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # For i from i_min to i_max, j = k-i, and coefficient is f_i (since g_j)
        # Actually conv: (f * g)[k] = sum_i f_i * g_{k-i}
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        ub.append(M)
    # nonnegativity:
    rows.extend(-np.eye(n))
    ub.extend([0.0]*n)
    A_ub = np.vstack(rows)
    b_ub = np.array(ub, dtype=float)

    # Solve LP using high-s performance
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 5})
    if res.success:
        return res.x
    else:
        return None

"""
Propose a candidate step function sequence minimizing the evaluation metric.
Algorithm:
1. Start from a random decreasing (geometric) sequence or the existing best.
2. Repeatedly:
   a) Compute current max convolution M.
   b) Solve linear program max sum(g) s.t. conv(f,g) <= M, g >= 0.
   c) Determine maximal mixing coefficient t (binary search) such that
      max conv( (1-t)f + t g ) <= M.
   d) Update f.
   e) If no progress, break.
3. Occasionally perform small random transfers of mass to try to lower M.
4. Keep the best sequence seen.
All operations respect the input validation of evaluate_sequence.
"""

def lp_maximize_sum(f_seq, M, time_limit=5.0):
    n = len(f_seq)
    c = -np.ones(n)
    rows = []
    rhs = []
    for k in range(2*n - 1):
        j = np.arange(n)
        i = k - j
        mask = (i >= 0) & (i < n)
        row = np.zeros(n)
        row[mask] = f_seq[i[mask]]
        rows.append(row)
        rhs.append(M)
    # nonnegativity constraints
    A_ub = np.vstack(rows + [-np.eye(n)])
    b_ub = np.concatenate([rhs, np.zeros(n)])
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': time_limit})
    if res.success:
        return np.clip(res.x, 0, 1000.0)  # clip to obey evaluate bounds
    else:
        return None

def random_perturb(seq, scale=0.1):
    n = len(seq)
    i, j = np.random.randint(n), np.random.randint(n)
    delta = np.random.randn() * scale
    # ensure non-neg
    if delta > 0:
        # Transfer from i to j
        if seq[i] > delta:
            seq[i] -= delta
            seq[j] += delta
    else:
        # negative delta: transfer from j to i
        delta = -delta
        if seq[j] > delta:
            seq[j] -= delta
            seq[i] += delta
    # Clip
    seq = np.clip(seq, 0, 1000)
    return seq

def solve_convolution_lp(f_seq, M, time_limit=2.0):
    n = len(f_seq)
    # objective: maximize sum(g) => minimize negative sum
    c = -np.ones(n)
    # Build constraint matrix
    total_rows = (2 * n - 1) + n
    A = np.zeros((total_rows, n), dtype=np.float64)
    b = np.zeros(total_rows, dtype=np.float64)
    # conv(f,g) constraints
    row = 0
    for k in range(2*n - 1):
        # inner loop fill
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # vectorized fill
        for i in range(i_start, i_end + 1):
            j = k - i
            A[row, j] = f_seq[i]
        b[row] = M
        row += 1
    # nonnegativity: -I * g <= 0 => g >=0
    A[row:, :] = -np.eye(n)
    # b already zeros for those rows
    # Solve LP
    res = linprog(c, A_ub=A, b_ub=b, method='highs', options={'time_limit': time_limit, 'presolve': True})
    if res.success:
        # Clip to keep within limits enforced later
        g = np.clip(res.x, 0, 1000.0)
        return g
    else:
        return None

"""
Propose a step‑function (a list of non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv is the discrete convolution.  The algorithm is a
coordinate‑ascent scheme similar to the published state‑of‑the‑art
but with a larger, adaptive step size.

Main ideas
----------
1.  Start from a non‑uniform sequence (random geometric decay or the
    provided `height_sequence_1`).  Such a sequence is not optimal for
    its current peak M = max(conv(a,a)), so we can increase the total
    sum while keeping the peak unchanged.

2.  For a current sequence f and its peak M, solve the linear program

        max   sum(g)
        s.t.  conv(f, g) <= M    (2 n‑1 linear constraints)
              g >= 0

    This gives a direction g that enlarges the sum without violating the
    convolution bound.

3.  Determine the largest feasible mixing coefficient `t` such that

        h = (1‑t)·f + t·g

    still satisfies `max(conv(h, h)) <= M`.  The bound on `t` is found
    by a cheap binary search that uses the closed‑form expression for the
    convolution of the mixture (quadratic in `t`).

4.  Replace `f` by `h` and repeat steps 1‑3 until the mixing coefficient
    becomes negligible.  Afterwards a few small random mass‑transfer
    perturbations are tried to escape possible stagnation.

5.  The whole process is repeated with different random restarts until the
    allocated budget expires.  The best sequence encountered is returned.

All sequence values are kept inside the interval `[0, 1000]` to respect the
security checks performed by `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# Helper utilities
# ----------------------------------------------------------------------


def _evaluation_score(seq: np.ndarray, conv_cache: np.ndarray | None = None) -> float:
    """
    Compute the same score that evaluate_sequence returns, but without the
    extra security overhead.  The result is mathematically identical.
    """
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    if conv_cache is None:
        max_b = np.max(np.convolve(seq, seq))
    else:
        max_b = np.max(conv_cache)
    return float(2.0 * n * max_b / (s * s))


def _solve_convolution_lp(f_seq: np.ndarray, M: float, time_limit: float = 2.0) -> np.ndarray | None:
    """
    Linear program:
        maximise   sum(g)
        subject to conv(f_seq, g) <= M   (2*n-1 constraints)
                   g >= 0
    Returns the optimal g as a NumPy array, or None if the LP failed.
    """
    n = len(f_seq)

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # Prepare A_ub and b_ub
    total_rows = (2 * n - 1) + n
    A = np.zeros((total_rows, n), dtype=np.float64)
    b = np.zeros(total_rows, dtype=np.float64)

    # convolution constraints: (f * g)[k] = Σ_i f_i * g_{k-i}
    row = 0
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # columns that receive a coefficient are j = k-i
        col_start = k - i_end
        col_end = k - i_start
        # the coefficients are f[i_start … i_end] in reverse order
        A[row, col_start : col_end + 1] = f_seq[i_start : i_end + 1][::-1]
        b[row] = M
        row += 1

    # non‑negativity: -g <= 0
    A[row:, :] = -np.eye(n, dtype=np.float64)
    # b[row:] already zero

    # Solve with the HiGHS dual simplex method (fast for moderate sizes)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            method="highs",
            options={"presolve": True, "time_limit": time_limit},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        # Clip to the limits enforced by evaluate_sequence
        return np.clip(res.x, 0.0, 1000.0)
    return None


def _max_feasible_t(f: np.ndarray, g: np.ndarray, M: float) -> float:
    """
    Binary‑search the largest t∈[0,1] such that
        max(conv( (1‑t)·f + t·g , (1‑t)·f + t·g )) ≤ M
    Uses the quadratic form of the mixed convolution for O(n) checks.
    """
    bf = np.convolve(f, f)
    bg = np.convolve(g, g)
    bfg = np.convolve(f, g)

    lo, hi = 0.0, 1.0
    # 30 iterations give ~1e‑9 precision – more than enough.
    for _ in range(30):
        mid = (lo + hi) * 0.5
        # (1‑t)^2·bf + 2t(1‑t)·bfg + t^2·bg   (element‑wise)
        mixed = (1.0 - mid) ** 2 * bf + 2.0 * mid * (1.0 - mid) * bfg + mid ** 2 * bg
        if np.max(mixed) <= M * (1.0 + 1e-12):
            lo = mid
        else:
            hi = mid
    return lo


def _random_mass_transfer(seq: np.ndarray, scale: float = 0.1) -> np.ndarray:
    """
    Choose two distinct indices i, j and move a small random amount δ from
    i to j (or vice‑versa).  Keeps the sequence non‑negative and respects
    the hard upper bound of 1000.
    """
    n = len(seq)
    i, j = np.random.randint(n), np.random.randint(n)
    while i == j:
        j = np.random.randint(n)

    delta = np.random.randn() * scale
    if delta > 0:
        # move from i → j
        d = min(delta, seq[i])
        seq[i] -= d
        seq[j] += d
    else:
        d = min(-delta, seq[j])
        seq[j] -= d
        seq[i] += d

    np.clip(seq, 0.0, 1000.0, out=seq)
    return seq

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a low‑value step‑function within the given time budget.
    Returns the best sequence found as a list of Python floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # keep a small safety margin

    # ------------------------------------------------------------------
    # Initial fallback (uniform) – guaranteed to be valid.
    # ------------------------------------------------------------------
    uniform_len = 500
    best_seq = [1.0] * uniform_len
    best_val = _evaluation_score(np.array(best_seq, dtype=float))
    print(f"[init] uniform fallback score {best_val:.6f}")

    # ------------------------------------------------------------------
    # If the competition already gave us a reference sequence, try it.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            ref = np.array(height_sequence_1, dtype=float)
            val = _evaluation_score(ref)
            if val < best_val:
                best_val = val
                best_seq = ref.tolist()
            print(f"[init] height_sequence_1 score {val:.6f}")
        except Exception as exc:  # pragma: no cover
            print("[init] failed to use height_sequence_1:", exc)

    # ------------------------------------------------------------------
    # Main optimisation loop – many independent restarts.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Choose a length for this restart (moderate size keeps LP cheap).
        # ------------------------------------------------------------------
        n = np.random.randint(300, 1300)

        # ------------------------------------------------------------------
        # Random geometric start – not optimal for its peak, therefore improvable.
        # ------------------------------------------------------------------
        r = 0.97 + 0.03 * np.random.rand()          # decay factor in [0.97, 1.00)
        scale = np.random.rand() * 5.0 + 0.5        # random overall scaling
        seq = np.array([r ** i for i in range(n)], dtype=float) * scale
        np.clip(seq, 0.0, 1000.0, out=seq)

        cur_score = _evaluation_score(seq)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = seq.tolist()
            print(f"[restart] initial improvement {best_val:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Inner optimisation: enlarge the sum while keeping the peak unchanged.
        # ------------------------------------------------------------------
        for outer in range(20):
            # current convolution peak
            bf = np.convolve(seq, seq)
            M = float(bf.max())

            # solve the LP for a direction that increases the sum
            g = _solve_convolution_lp(seq, M, time_limit=1.5)
            if g is None:
                # occasional random tweak if LP fails
                seq = _random_mass_transfer(seq, scale=0.05)
                continue

            # maximal safe mixing factor
            t = _max_feasible_t(seq, g, M)
            if t < 1e-5:
                break  # cannot move any further in this regime

            # perform the convex combination
            seq = (1.0 - t) * seq + t * g
            np.clip(seq, 0.0, 1000.0, out=seq)

            # evaluate new sequence (use cached convolution for speed)
            bf_new = np.convolve(seq, seq)
            cur_score = _evaluation_score(seq, conv_cache=bf_new)
            if cur_score < best_val:
                best_val = cur_score
                best_seq = seq.tolist()
                print(f"[improve] score {best_val:.6f} iter={outer} t={t:.4f} n={n}")

        # ------------------------------------------------------------------
        # Small random mass‑transfer sweeps to possibly lower the peak.
        # ------------------------------------------------------------------
        for _ in range(5):
            seq = _random_mass_transfer(seq, scale=0.1)
            cur_score = _evaluation_score(seq)
            if cur_score < best_val:
                best_val = cur_score
                best_seq = seq.tolist()
                print(f"[perturb] improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # Stop early if we are out of time.
        # ------------------------------------------------------------------
        if time.time() + 1.0 >= deadline:
            break

    print(f"[final] best score {best_val:.6f}  length {len(best_seq)}")
    return best_seq
