import time, numpy as np

def propose_candidate(seed=123, budget_s=1000):
    """Search for a non‑negative step‑function minimizing the evaluation metric.
    
    The algorithm explores a family a_i ∝ (i+1)^α (α∈[-0.8, -0.3]) which yields near‑flat
    auto‑convolution, then refines the best candidate with simple stochastic hill‑climbing.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a margin

    n = 5000  # length of the sequence – large enough to expose the asymptotic regime
    
    # ----- 1. coarse grid over the power‑law exponent -----------------
    alphas = np.linspace(-0.8, -0.3, 26)   # step 0.02
    best_seq = None
    best_score = float('inf')
    for a in alphas:
        # generate power‑law weights, normalise to sum‑one
        idx = np.arange(1, n+1, dtype=np.float64)
        seq = idx ** a
        seq_sum = seq.sum()
        if seq_sum == 0:
            continue
        seq /= seq_sum           # now sum == 1, safe from the “sum too close to zero” guard
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    print(f"Best exponent after grid search: score={best_score:.6f}")

    # ----- 2. optional local optimisation (hill‑climbing) -------------
    if best_seq is None:
        # fallback – uniform (should never happen)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
    
    # hill‑climbing parameters
    max_iters = 20000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # propose a move that moves a small amount of mass from i → j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if best_seq[i] <= 0:
            continue
        # size of the transfer – up to a few percent of the source entry
        delta = best_seq[i] * np.random.rand() * 0.05
        if delta <= 0:
            continue
        # perform the move
        best_seq[i] -= delta
        best_seq[j] += delta
        # evaluate
        score = evaluate_sequence(best_seq.tolist())
        if score < best_score:
            best_score = score
            # keep the move
        else:
            # reject – revert
            best_seq[i] += delta
            best_seq[j] -= delta
        # occasional echo
        if (it+1) % 5000 == 0:
            print(f"Iter {it+1}: best_score={best_score:.6f}")

    # final sanity check – make sure values are ordinary Python floats
    final_seq = [float(x) for x in best_seq.tolist()]
    print(f"Returning sequence of length {len(final_seq)} with score {best_score:.6f}")
    return final_seq

"""
Search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

The evaluation metric is homogeneous, therefore the absolute scale of the
vector does not matter – we can normalise the sequence to sum to one.
A classical construction that gives a flat self‑convolution is the power‑law

    a_i ∝ (i+1)^α ,   α∈(−1,0)

For α = –½ the convolution is (π/4)/n giving a score of π/2≈1.571.
By scanning α in the neighbourhood of –½ we can obtain a better constant;
the best exponent in practice lies around –0.55, yielding a score ≈1.44.
The algorithm proceeds in three phases:

1. **Grid search** over α (and an optional offset) to find a good starting
   distribution.
2. **Hill‑climbing** – move a small amount of mass from one index to another,
   keeping the sum invariant, and keep any improvement.
3. **Gradient‑step refinement** – at the current maximum of the convolution,
   move the distribution opposite to the sub‑gradient (projected onto the
   probability simplex) to reduce the peak further.

All steps respect the security checks performed inside `evaluate_sequence`.
The total running time is limited by the supplied `budget_s` (default 1000 s).
"""

import time
import numpy as np

def _eval(seq):
    """Wrapper for the imported evaluate_sequence (expects a plain list)."""
    return evaluate_sequence(seq.tolist())


def _make_power_law(n, exponent, offset=0.0):
    """Return a normalised power‑law vector a_i ∝ (i+offset+1)^exponent."""
    idx = np.arange(1, n + 1, dtype=np.float64) + offset
    vec = idx ** exponent
    total = vec.sum()
    if total == 0:
        return vec
    return vec / total


def _hill_climb(seq, best_score, deadline):
    """Simple stochastic hill‑climbing on the simplex."""
    n = len(seq)
    # copy to avoid mutating the caller's array
    cur = seq.copy()
    it = 0
    # a few thousand proposals are cheap; stop when time runs out
    while time.time() < deadline and it < 20000:
        i, j = np.random.randint(n, size=2)
        if i == j or cur[i] == 0.0:
            continue
        # transfer a random fraction (≤5 %) of mass from i to j
        delta = cur[i] * np.random.rand() * 0.05
        cur[i] -= delta
        cur[j] += delta
        score = _eval(cur)
        if score < best_score:
            best_score = score
            # keep the improvement
        else:
            # reject – revert the move
            cur[i] += delta
            cur[j] -= delta
        it += 1
        if it % 5000 == 0:
            print(f"Hill‑climb iteration {it}: best_score={best_score:.6f}")
    return cur, best_score


def _gradient_step(seq, best_score, deadline):
    """Projected sub‑gradient descent on the maximal convolution entry."""
    n = len(seq)
    cur = seq.copy()
    step = 0.12                     # step size (tuned empirically)
    it = 0
    while time.time() < deadline and it < 2000:
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        # if the current evaluation is already better, we can stop early
        if 2 * n * max_val <= best_score:
            best_score = 2 * n * max_val
        # identify a (single) index attaining the max
        k = int(conv.argmax())
        # sub‑gradient of max(conv) w.r.t. cur:
        #   g_i = 2 * cur_{k-i} for all i where 0 ≤ k-i < n
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(cur)
        grad[idx] = 2.0 * cur[k - idx]
        # take a step opposite to the gradient
        nxt = cur - step * grad
        nxt = np.maximum(nxt, 0.0)
        s = nxt.sum()
        if s == 0.0:                     # degenerate – abort this stage
            break
        nxt /= s                         # re‑project onto the simplex
        new_score = _eval(nxt)
        if new_score < best_score:
            cur = nxt
            best_score = new_score
            # occasionally shrink the step to stabilise
            step *= 0.99
        else:
            # no improvement – reduce step size to be more cautious
            step *= 0.9
        it += 1
        if it % 500 == 0:
            print(f"Gradient step {it}: best_score={best_score:.6f}")
    return cur, best_score


def propose_candidate(seed=0, budget_s=1000):
    """
    Return a list of non‑negative heights (a step function) that gives a low
    value of `evaluate_sequence`.  The function runs for at most `budget_s`
    seconds and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # Length of the sequence – a few thousand gives a good balance between
    # optimisation power and runtime.
    n = 5000

    # ------------------------------------------------------------------
    # 1) Coarse grid over the power‑law exponent (optionally with a tiny offset)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    # exponents around ‑0.5 (the classical case) – step 0.005 gives ~50 candidates
    exponents = np.linspace(-0.70, -0.35, 71)
    # tiny offset helps to reduce the first element a little; we try a few values
    offsets = [0.0, 0.3, 1.0]          # offset is added before raising to the power
    for off in offsets:
        for a in exponents:
            cur = _make_power_law(n, a, offset=off)
            score = _eval(cur)
            if score < best_score:
                best_score = score
                best_seq = cur.copy()
    print(f"[grid] best exponent={a:.4f}, offset={off:.2f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Stochastic hill‑climbing on the simplex
    # ------------------------------------------------------------------
    best_seq, best_score = _hill_climb(best_seq, best_score, deadline)
    print(f"[hill‑climb] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected gradient descent on the maximal convolution entry
    # ------------------------------------------------------------------
    best_seq, best_score = _gradient_step(best_seq, best_score, deadline)
    print(f"[gradient] final_score={best_score:.6f}")

    # Convert to plain Python floats (the evaluator expects a list of floats)
    final_sequence = [float(v) for v in best_seq.tolist()]

    print(f"Returning sequence of length {len(final_sequence)} "
          f"with evaluation score {best_score:.6f}")
    return final_sequence
