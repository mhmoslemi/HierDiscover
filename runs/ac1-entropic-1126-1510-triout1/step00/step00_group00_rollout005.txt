def project_simplex(v, s=1.0):
    """Project onto the probability simplex: w >=0, sum w = s."""
    v = np.array(v, dtype=np.float64)
    if np.sum(v) == s and np.all(v >= 0):
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def subgradient_step(a, step, kmax):
    n = a.shape[0]
    # compute gradient g: 2*a[kmax - i] if valid else 0
    g = np.zeros_like(a)
    # compute reversed indices

    mask = (idx >= 0) & (idx < n)
    g[mask] = 2 * a[idx[mask]]

"""
Propose a sequence of non‑negative heights that minimizes the
`evaluate_sequence` score.

Algorithm
---------
1.  The continuous‐time solution of the problem
    “minimise the L∞‑norm of the self‑convolution under a fixed L¹‑mass’’
    is the arcsine density
          f(x) = 1/(π·√(x(1‑x))) ,  x∈(0,1) .
    Its discrete analogue gives a sequence whose autocorrelation is almost
    flat, i.e. the maximal value of the convolution is close to its average.
    The resulting score is close to the theoretical lower bound 1.0.

2.  Build a discrete arcsine distribution on a chosen length `n`:
        a_i ∝ 1/√((i+½)(n‑i‑½)) ,   i = 0 … n‑1
    and normalise it to have unit sum (the evaluation is scale‑invariant).

3.  Refine the shape with a short projected sub‑gradient descent.
    At each iteration we:
        * compute the discrete convolution `b = a * a`;
        * locate the index `k` where `b` attains its maximum;
        * the sub‑gradient of `b_k` w.r.t. `a` is `g_i = 2·a_{k‑i}` for
          those indices that lie inside the support;
        * take a step `a ← a – η·g` and project back onto the probability
          simplex (non‑negative entries summing to 1).
      The step size is decreasing (`η = η₀ / √t`) to guarantee stability.

4.  Keep the best sequence seen during the run (including the initial
    arcsine one) and return it as a plain Python list.

The whole routine respects the 1000 s budget, uses only NumPy, and never
touches the filesystem or network.  It works even if the global variable
`height_sequence_1` is unavailable – in that case a default length of
2000 is used.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (non‑negative, sum = s)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast special case
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: generate the discrete arcsine distribution of length n
def _arcsine_distribution(n: int) -> np.ndarray:
    """Return a length‑n probability vector approximating the arcsine law."""
    i = np.arange(n, dtype=np.float64)
    # 1/√((i+½)*(n‑i‑½)) – the ½ avoids the exact singularities at the ends
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    # Normalise to unit ℓ₁‑mass
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Core optimisation for a single length
def _optimise_one_length(
    n: int,
    time_deadline: float,
    init: str = "arcsine",
    max_iters: int = 4000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Perform a short projected sub‑gradient descent starting from `init`.
    Returns the best sequence found together with its score.
    """
    # ----- initial point -------------------------------------------------
    if init == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init == "random":
        a = np.random.rand(n)
        a /= a.sum()
    else:  # arcsine
        a = _arcsine_distribution(n)

    # ----- bookkeeping ---------------------------------------------------
    best_seq = a.copy()
    # evaluate: score = 2·n·max(conv)  (sum(a)=1 ⇒ denominator disappears)
    b = np.convolve(a, a)
    best_score = float(2.0 * n * b.max())

    # ----- main loop -----------------------------------------------------
    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break

        # convolution and its maximiser
        b = np.convolve(a, a)
        k_max = int(np.argmax(b))
        max_b = b[k_max]
        cur_score = 2.0 * n * max_b

        # track best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # occasional progress report
            print(
                f"[{time.time():.1f}] length={n} iter={it} "
                f"score={best_score:.6f}"
            )

        # sub‑gradient of the active constraint (the maximal convolution entry)
        idx = k_max - np.arange(n)                     # potential partner indices
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection onto the simplex
        a = a - eta * g
        a = _project_onto_simplex(a, s=1.0)

        # small random jitter every few hundred iterations to avoid stagnation
        if it % 250 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score
