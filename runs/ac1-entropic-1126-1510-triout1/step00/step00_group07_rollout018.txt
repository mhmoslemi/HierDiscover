def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)
    # build A_ub matrix
    A_ub = []
    b_ub = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # for i from max(0, k-(n-1)) to min(n-1, k)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # then j = k - i
        # row[j] = f_seq[i]
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f_seq[i]
        A_ub.append(row)
        b_ub.append(rhs)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def reduce_max_convolution(seq):
    c = np.convolve(seq, seq)
    k = int(np.argmax(c))
    # compute contributions
    # indices i where 0 <= i < n and 0 <= k-i < n
    # product = seq[i] * seq[k-i]
    # find i_max maximizing product
    i_vals = np.arange(max(0, k - (n-1)), min(n-1, k) + 1)
    prods = seq[i_vals] * seq[k - i_vals]
    idx = np.argmax(prods)
    i_max = i_vals[idx]
    j_max = k - i_max
    # amount to move
    delta = min(seq[i_max], seq[j_max]) * 0.1  # reduce both by 10%?
    # maybe reduce both equally (or more from larger)
    # create new seq copy
    new_seq = seq.copy()
    # reduce both
    new_seq[i_max] -= delta
    new_seq[j_max] -= delta
    # add mass to smallest index
    low_idx = np.argmin(new_seq)
    new_seq[low_idx] += 2 * delta
    # ensure nonnegative
    new_seq = np.maximum(new_seq, 0)
    return new_seq

"""
Propose a candidate sequence that minimizes the evaluation function.

The algorithm explores several families of sequences (uniform, geometric,
piecewise geometric with a constant tail, linear, etc.) and then performs a
simple local search that reduces the maximal convolution value.
It uses random sampling and a targeted “mass‑transfer” step that moves weight
away from the entries that contribute most to the current maximal convolution.
The method works for any sequence length (uses the existing height_sequence_1
if available) and stays well within the 1000 s budget.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
# Helper: evaluate_sequence is assumed to be available in the global scope.
# ----------------------------------------------------------------------


def _eval(seq):
    """Wrapper for the supplied evaluate_sequence (handles list conversion)."""
    return evaluate_sequence(seq)


def _convolution(seq):
    """Return the convolution of a sequence with itself (numpy array)."""
    return np.convolve(seq, seq)


def _init_candidates(n):
    """
    Build a small set of deterministic candidate sequences.
    Returns a list of numpy arrays.
    """
    cand = []

    # Uniform
    cand.append(np.ones(n, dtype=float))

    # Linear decreasing (a_i = n-i)
    cand.append(np.arange(n, 0, -1, dtype=float))

    # Quadratic decreasing (a_i = (n-i)^2)
    cand.append((np.arange(n, 0, -1, dtype=float)) ** 2)

    # Pure geometric families (r = 1 - c/n)
    for c in np.linspace(0.5, 4.0, 12):
        r = 1.0 - c / n
        if r <= 0:
            continue
        cand.append(r ** np.arange(n, dtype=float))

    # Piecewise: block of ones + geometric tail
    for p_frac in [0.2, 0.4, 0.6]:
        p = int(p_frac * n)
        for c in [1.0, 2.0, 3.0]:
            r = 1.0 - c / n
            if r <= 0:
                continue
            tail_len = n - p
            tail = r ** np.arange(tail_len, dtype=float)
            seq = np.concatenate([np.ones(p, dtype=float), tail])
            cand.append(seq)

    return cand


def _geo_tail_candidate(n):
    """
    Produce a random “geometric + constant tail” candidate.
    The first part is geometric a_i = r**i for i < m,
    the tail (indices m … n‑1) is a constant value t.
    """
    # choose geometric part size
    m_frac = random.uniform(0.1, 0.8)
    m = max(1, int(m_frac * n))

    # choose decay: r = 1 - c/n with c in a reasonable range
    c = random.uniform(0.5, 4.0)
    r = 1.0 - c / n
    if r <= 0.0:
        r = 0.5  # fallback, should be rare

    # geometric part
    geo = r ** np.arange(m, dtype=float)

    # tail length
    tail_len = n - m
    if tail_len == 0:
        return np.concatenate([geo])

    # estimate a safe upper bound for the tail height:
    # M_geo = max(conv(geo,geo)) – we compute it explicitly
    full_geo = np.concatenate([geo, np.zeros(tail_len, dtype=float)])
    M_geo = np.max(_convolution(full_geo))

    # the tail constant t must not create a larger convolution value than M_geo.
    # A very safe bound is t <= sqrt(M_geo / tail_len).
    t_max = np.sqrt(M_geo / max(1, tail_len))
    # choose a random fraction of this bound (0 … 1)
    t = random.uniform(0.0, 0.9) * t_max

    tail = np.full(tail_len, t, dtype=float)
    return np.concatenate([geo, tail])


def _mass_transfer_step(seq):
    """
    One deterministic local improvement step.
    Reduce the two entries that contribute most to the current maximal
    convolution and move that mass to a currently smallest entry.
    """
    n = len(seq)
    conv = _convolution(seq)
    k_max = int(np.argmax(conv))

    # indices that participate in this convolution entry
    i_min = max(0, k_max - (n - 1))
    i_max = min(n - 1, k_max)
    i_vals = np.arange(i_min, i_max + 1, dtype=int)
    # product contributions a_i * a_{k-i}
    prods = seq[i_vals] * seq[k_max - i_vals]
    idx = int(np.argmax(prods))
    i1 = int(i_vals[idx])
    i2 = int(k_max - i1)

    # move a small fraction of the smaller of the two entries
    delta = min(seq[i1], seq[i2]) * 0.07  # 7 % of the smaller entry
    if delta <= 0.0:
        return seq  # nothing to do

    new_seq = seq.copy()
    new_seq[i1] -= delta
    new_seq[i2] -= delta

    # add the moved mass to the currently smallest entry
    i_min_val = int(np.argmin(new_seq))
    new_seq[i_min_val] += 2.0 * delta
    # clamp to non‑negative (should already hold)
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq


def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative numbers that yields a low value
    of the provided evaluation function.  The routine runs for at most
    `budget_s` seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence we will optimise.
    # Prefer the length of the known construction if it exists.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore
    except Exception:
        n = 2000                     # fallback length

    # ------------------------------------------------------------------
    # Initialise with a diverse set of deterministic candidates.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for cand in _init_candidates(n):
        score = _eval(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[init] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main stochastic search loop.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 1.0   # small safety margin
    while time.time() < deadline:
        # ----------------------------------------------------------------
        # Occasionally try a fresh random geometric+tail candidate.
        # ----------------------------------------------------------------
        if random.random() < 0.35:
            cand = _geo_tail_candidate(n)
        else:
            # small random perturbation of the current best
            noise = np.random.randn(n) * 0.02
            cand = best_seq + noise
            cand = np.maximum(cand, 0.0)

        # Evaluate the candidate
        score = _eval(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[rand] new best score = {best_score:.6f}")

        # ----------------------------------------------------------------
        # Perform a few deterministic local improvement steps starting from the
        # current best sequence.
        # ----------------------------------------------------------------
        cur = best_seq.copy()
        cur_score = best_score
        for _ in range(3):
            cur = _mass_transfer_step(cur)
            cur_score = _eval(cur.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[local] new best score = {best_score:.6f}")

        # ----------------------------------------------------------------
        # Early exit if target reached.
        # ----------------------------------------------------------------
        if best_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # final sanity check (ensure non‑negative values)
    best_seq = np.maximum(best_seq, 0.0).tolist()
    print(f"Finished search – best score = {best_score:.6f}")
    return best_seq
