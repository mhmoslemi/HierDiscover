            s = a_candidate.sum()
        a_candidate /= s                            # re‑normalise to unit sum

        # ----- 5.4  Evaluate the candidate -------------------------------
        cand_val = current_score(a_candidate)

        # ----- 5.5  Acceptance test --------------------------------------
        if cand_val < cur_val:                      # improvement
            a = a_candidate
            cur_val = cand_val
            step = min(step_max, step * 1.02)       # gently increase step size
            stagnation = 0
            if cand_val < best_val:
                best_val = cand_val
                best_seq = a_candidate.copy()
                print(f"[iter {it}] new best = {best_val:.6f}")
        else:                                        # no improvement
            step = max(step_min, step * 0.5)        # shrink step size
            stagnation += 1

        # ----- 5.6  Escape plateaus --------------------------------------
        if stagnation > 2000:
            # inject a very small random jitter and re‑normalise
            noise = np.random.randn(n) * 0.001
            a = np.maximum(a + noise, 0.0)
            a /= a.sum()
            stagnation = 0
            print(f"[iter {it}] perturbation injected")

        # ----- 5.7  Occasional full random restart -----------------------
        if it % 5000 == 0:
            a = np.random.rand(n)
            a /= a.sum()
            step = 0.5
            stagnation = 0
            print(f"[iter {it}] random restart")

    # ----------------------------------------------------------------------
    # 6.  Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    return best_seq.tolist()
