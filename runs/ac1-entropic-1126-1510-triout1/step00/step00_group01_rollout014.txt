def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Determine sequence length randomly between 200 and 800
    n = np.random.randint(200, 800)
    # start with random positive values
    seq = np.random.rand(n)
    # normalize sum to sqrt(2*n)
    seq = seq / np.sum(seq) * np.sqrt(2.0 * n)
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq)
    # main loop
    while time.time() < deadline:
        # compute current M via convolution
        conv = np.convolve(seq, seq)
        M = conv.max()
        # solve LP for g
        g = solve_convolution_lp(seq, M)
        if g is not None:
            # rescale g to match sum sqrt(2n)
            g = g / np.sum(g) * np.sqrt(2.0 * n)
            # try several mixing rates
            improved = False
            for t in (0.05, 0.1, 0.2, 0.4, 0.6):
                cand = (1 - t) * seq + t * g
                score = evaluate_sequence(cand)
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    seq = cand.copy()
                    improved = True
                    break
            if not improved:
                # small random perturbation
                i, j = np.random.randint(n), np.random.randint(n)
                delta = np.random.randn() * 0.01
                cand = seq.copy()
                cand[i] = max(0, cand[i] + delta)
                # adjust to keep sum constant (simple rebalancing)
                cand[j] = max(0, cand[j] - delta)
                cand = np.clip(cand, 0, None)
                # renormalize sum
                cand = cand / np.sum(cand) * np.sqrt(2.0 * n)
                score = evaluate_sequence(cand)
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    seq = cand.copy()
        else:
            # random perturbation as fallback
            i, j = np.random.randint(n), np.random.randint(n)
            delta = np.random.randn() * 0.05
            cand = seq.copy()
            cand[i] = max(0, cand[i] + delta)
            cand[j] = max(0, cand[j] - delta)
            cand = np.clip(cand, 0, None)
            cand = cand / np.sum(cand) * np.sqrt(2.0 * n)
            score = evaluate_sequence(cand)
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                seq = cand.copy()
        # optional: occasional restart if no improvement for many steps
        # we can check time and restart with random seq
    return best_seq.tolist()

def reduce_max_convolution(seq, target_sum):
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    n = len(seq)
    # compute contributions
    contributions = []
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            contributions.append((seq[i] * seq[j], i, j))
    # sort contributions descending
    contributions.sort(reverse=True, key=lambda x: x[0])
    # choose top pair (i,j)
    if contributions:
        _, i_max, j_max = contributions[0]
        # move small epsilon from i_max and j_max to positions far away (e.g., opposite ends)
        eps = min(seq[i_max], seq[j_max]) * 0.05
        if eps > 0:
            seq_new = seq.copy()
            seq_new[i_max] -= eps
            seq_new[j_max] -= eps
            # add to random other positions
            idx1 = np.random.randint(n)
            idx2 = np.random.randint(n)
            seq_new[idx1] += eps/2
            seq_new[idx2] += eps/2
            seq_new = np.clip(seq_new, 0, None)
            seq_new = seq_new / seq_new.sum() * target_sum

"""
Propose a candidate step‑function sequence that (hopefully) improves the
evaluation metric

    score = 2 * n * max(conv(seq,seq)) / (sum(seq)**2)

(the smaller the score, the better).  The algorithm is a hybrid
stochastic / linear‑programming search:

*   Initialise a random sequence of moderate length (200‑600) and scale it
    to the standard normalisation Σa = sqrt(2·n) (the scoring function is
    scale‑invariant, this just keeps the numbers well‑behaved).

*   Repeatedly try two kinds of improvement steps while time permits:

    1. **LP direction** –  Solve a linear programme that finds a non‑negative
       vector `g` with the largest possible ℓ₁‑norm under the constraint
       `conv(f,g) ≤ M`, where `M` is the current maximum convolution of the
       sequence `f`.  Mixing `f` with `g` (using several mixing factors
       `t`) often reduces the maximal convolution after the mixture is
       renormalised.

    2. **Local smoothing** –  Look at the shift `k` where the convolution
       attains its maximum, identify the pair `(i,j)` that contributes most
       (`a_i*a_j`) to this peak and move a small amount of mass away from
       the offending indices.  The move is accepted only if it improves the
       score.

*   Occasionally apply a tiny random perturbation to escape plateaus and,
    if no improvement has been seen for a while, restart from a fresh
    random sequence (possibly of a slightly different length).

The search runs for the allotted time (default 1000 s) and returns the best
sequence found.  The implementation uses only `numpy`, `scipy.optimize.linprog`
and the already‑provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import linprog


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve the LP:
        maximise   sum(b)
        subject to conv(f_seq, b) <= rhs   (for every shift)
                   b >= 0
    Returns the optimal b or None if the LP fails.
    """
    n = len(f_seq)
    # objective: minimise -sum(b)  (i.e. maximise sum(b))
    c = -np.ones(n)

    # Build A_ub * b <= b_ub constraints for each convolution entry
    a_ub = []
    b_ub = []
    # each row corresponds to a shift k = 0 .. 2n-2
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                # conv entry is sum_i f_i * b_{k-i}
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints: -b_i <= 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.concatenate(b_ub)

    # a modest time limit keeps the overall search fast
    try:
        res = linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": 5.0},
        )
    except Exception:  # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


def mix_with_direction(seq: np.ndarray, direction: np.ndarray, target_sum: float):
    """
    Try several mixing factors between `seq` and `direction`,
    renormalise to `target_sum`, and return the best candidate
    (sequence, score).  If no improvement is found, (None, None) is returned.
    """
    best_cand = None
    best_score = None
    for t in (0.05, 0.1, 0.2, 0.4, 0.6):
        cand = (1.0 - t) * seq + t * direction
        cand = np.clip(cand, 0.0, None)
        # renormalise to keep the total sum constant (scale‑invariant)
        cand = cand / np.sum(cand) * target_sum
        score = evaluate_sequence(cand.tolist())
        if best_score is None or score < best_score:
            best_score = score
            best_cand = cand.copy()
    return best_cand, best_score


def reduce_max_convolution(seq: np.ndarray, target_sum: float):
    """
    Perform a tiny deterministic move that tries to lower the
    current maximal convolution value.
    """
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    n = len(seq)

    # find the pair (i, j) with the largest product a_i*a_j contributing to conv[k]
    max_prod = -1.0
    i_star = j_star = None
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            prod = seq[i] * seq[j]
            if prod > max_prod:
                max_prod = prod
                i_star, j_star = i, j

    if i_star is None:
        return seq  # should not happen

    # move a small epsilon away from the offending indices
    eps = min(seq[i_star], seq[j_star]) * 0.05
    if eps <= 0.0:
        return seq

    cand = seq.copy()
    cand[i_star] -= eps
    cand[j_star] -= eps
    # donate the mass to two random positions (different from the donors)
    idx1 = np.random.randint(0, n)
    idx2 = np.random.randint(0, n)
    cand[idx1] += eps / 2.0
    cand[idx2] += eps / 2.0

    cand = np.clip(cand, 0.0, None)
    cand = cand / np.sum(cand) * target_sum
    return cand
