def project_simplex(v):
    # v is a numpy array, returns w s.t. w>=0, sum(w)=1, minimal Euclidean distance to v.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    a = np.full(n, 1/n)  # uniform
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    step0 = 0.5
    it = 0
    while time.time() < tmax:
        b = np.convolve(a, a)
        k_max = int(np.argmax(b))
        grad = np.zeros(n)
        i_min = max(0, k_max - (n-1))
        i_max = min(k_max, n-1)
        # for i in range(i_min, i_max+1):
        #    grad[i] = a[k_max - i]
        idxs = np.arange(i_min, i_max+1)
        grad[idxs] = a[k_max - idxs]
        step = step0 / np.sqrt(it + 1)
        a_new = a - step * grad
        # Project onto simplex
        a = project_simplex(a_new)
        # Evaluate
        score = 2.0 * n * np.max(np.convolve(a, a)) # Since sum=1
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional random kick

    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # choose length n maybe 200 or 500.
    n = 200
    a = np.random.dirichlet(np.ones(n))
    # optionally start from height_sequence_1 truncated/resized
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    cur_seq = a.copy()
    cur_score = best_score
    
    T0 = 0.1
    T = T0
    iteration = 0
    while time.time() < deadline:
        # propose move
        i, j = np.random.choice(n, 2, replace=False)
        # epsilon drawn from uniform [-delta, delta] but ensure a[i]-epsilon >=0
        max_eps = min(cur_seq[i], 0.1) # limit max move
        eps = (np.random.rand() - 0.5) * 2 * max_eps  # random in [-max_eps, max_eps]
        if eps == 0: continue
        new_seq = cur_seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps
        # ensure non-neg
        if new_seq[i] < 0:
            continue
        # evaluate
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < cur_score or np.random.rand() < np.exp((cur_score - new_score)/max(T, 1e-12)):
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iteration}, new best {best_score}")
        # cooling
        iteration += 1
        if iteration % 1000 == 0:
            T *= 0.99  # exponential cooling
            # optional random restart
            if T < 1e-6:
                # reinitialize random?
                T = T0
    # return best_seq as list of floats
    return best_seq.tolist()

"""
Simple simulated‑annealing search for a step‑function that reduces the
evaluate_sequence objective.

The algorithm works on a fixed length `n` (chosen between 200 and 500) and
keeps the vector on the probability simplex (non‑negative, sum = 1).
The convolution `b = a * a` is maintained incrementally; a move
transfers a small amount ε from a “donor” index i to a “receiver” index j.
The change of the convolution caused by the move can be updated in O(n)
with vectorised NumPy operations, so millions of proposals can be tried
within the 1000‑second budget.

A temperature schedule controls the step size and the probability of
accepting uphill moves (standard simulated‑annealing).  The best sequence
found (lowest score) is returned.  If it does not improve over the
previous best `height_sequence_1`, the old construction is kept.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: incremental update of the ordered convolution when a single
# entry a[idx] changes by `delta` (delta may be positive or negative).
def _apply_delta(a: np.ndarray, b: np.ndarray, idx: int, delta: float) -> None:
    """
    Apply a change `delta` to a[idx] and update the full convolution `b`.
    Both a and b are modified in place.
    """
    n = a.shape[0]
    # Ordered convolution contribution change:
    #   b[k] += 2 * delta * a[k-idx]  for k = idx .. idx+n-1
    #   plus the quadratic term delta^2 at position 2*idx.
    b[idx:idx + n] += 2.0 * delta * a
    b[2 * idx] += delta * delta
    a[idx] += delta


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimizes the supplied
    `evaluate_sequence` function.

    The method uses a simulated‑annealing style stochastic search on the
    probability simplex, maintaining the convolution incrementally for
    speed.  It returns a Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  Too large would make each move expensive,
    # too small would limit achievable improvement.
    n = np.random.randint(200, 500)          # length ~ 200–500

    # Initial sequence: either a random Dirichlet draw or the previously
    # known construction (if available).  Normalise to sum = 1.
    if 'height_sequence_1' in globals() and np.random.rand() < 0.3:
        a = np.array(height_sequence_1, dtype=float)
        a = a[:n]                     # truncate if it is longer
        if a.size < n:                # pad with tiny mass if needed
            a = np.concatenate([a, np.full(n - a.size, 1e-12)])
        a = a / np.sum(a)
    else:
        a = np.random.dirichlet(np.ones(n))   # sum = 1, all positive

    # Full ordered convolution a*a (length 2n-1)
    b = np.convolve(a, a)

    # Current score (exactly the value produced by evaluate_sequence
    # for sum == 1, because evaluate_sequence = 2*n*max(b) / sum^2).
    cur_score = 2.0 * n * np.max(b)

    best_seq = a.copy()
    best_score = cur_score

    # Simple simulated‑annealing parameters
    temperature = 0.2          # initial temperature (controls step size)
    iteration = 0

    # Main SA loop
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Propose a move: transfer some mass from donor i to receiver j.
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1                     # ensure i != j

        if a[i] <= 1e-12:
            continue                   # nothing to give

        # Step size is a fraction of the donor mass, scaled by temperature.
        max_move = min(a[i], temperature * a[i])
        eps = np.random.rand() * max_move   # always non‑negative

        # Apply the move (donor i loses eps, receiver j gains eps)
        _apply_delta(a, b, i, -eps)
        _apply_delta(a, b, j, eps)

        new_max = b.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # Acceptance test
        accept = False
        if delta_score < 0:
            accept = True
        else:
            prob = np.exp(-delta_score / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"Iter {iteration:>7d} — new best score {best_score:.6f}")
        else:
            # Revert the move
            _apply_delta(a, b, i, eps)   # give back to i
            _apply_delta(a, b, j, -eps)  # take back from j

        # Cooling schedule
        iteration += 1
        if iteration % 1000 == 0:
            temperature *= 0.95
            # Occasionally reset temperature to avoid freezing too early
            if temperature < 1e-4:
                temperature = 0.2
                # Random restart to escape possible local traps
                a = np.random.dirichlet(np.ones(n))
                b = np.convolve(a, a)
                cur_score = 2.0 * n * np.max(b)

    # ------------------------------------------------------------------
    # Convert the best sequence to a plain Python list.
    best_list = best_seq.tolist()

    # Verify that the returned sequence is at least as good as the
    # previously known construction (if any).  If not, fall back.
    try:
        prev_score = evaluate_sequence(list(height_sequence_1))
    except Exception:
        prev_score = float('inf')

    final_score = evaluate_sequence(best_list)
    if final_score >= prev_score:
        # fallback to the older construction – it is guaranteed to be valid
        return list(height_sequence_1)

    return best_list
