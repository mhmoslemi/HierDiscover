```
def simplex_projection(v, s):
    # Project v onto simplex {x >= 0, sum x = s}
    # Implementation based on sorting.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    # comparable size at random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and np.random.rand() < 0.5:
        base = np.array(height_sequence_1, dtype=float)
        n = base.shape[0]
        start = base.copy()
    else:
        n = np.random.randint(2500, 3500)          # a size close to the known one
        start = np.random.exponential(scale=1.0, size=n)

    # ---------------------------------------------------------------
    # Normalise the initial vector so that Σa_i = √(2n) (required by the
    # evaluation function).  All entries are forced to be non‑negative.
    # ---------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    start = np.maximum(start, 0.0)
    if start.sum() < 1e-6:
        start = np.full(n, target_sum / n)
    else:
        start *= target_sum / start.sum()

    cur = start.copy()
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a tiny safety margin

    it = 0
    no_improve = 0
    eps_factor = 0.12   # initial fraction of mass moved per iteration

    while time.time() < deadline:
        it += 1

        # -----------------------------------------------------------
        # Compute the full convolution and locate the current largest
        # coefficient.  This value is exactly the objective when the L1‑norm
        # equals √(2n).
        # -----------------------------------------------------------
        conv = np.convolve(cur, cur)
        max_val = conv.max()
        max_idx = int(conv.argmax())

        # Theoretical lower bound is 1.0; stop early if we are extremely close.
        if max_val <= 1.01:
            break

        # -----------------------------------------------------------
        # Propose a deterministic move that reduces the dominant term.
        # -----------------------------------------------------------
        nxt = _reduce_max_pair(cur, max_idx, eps_factor=eps_factor)

        if nxt is None:
            # a fallback: a tiny random jitter to avoid numerical dead‑ends
            cur = np.maximum(cur + np.random.normal(scale=1e-4 * target_sum, size=n), 0.0)
            cur = _project_to_simplex(cur, target_sum)
            continue

        # Project back onto the simplex to keep the exact L1‑norm.
        nxt = _project_to_simplex(nxt, target_sum)

        # -----------------------------------------------------------
        # Evaluate the candidate.
        # -----------------------------------------------------------
        score = evaluate_sequence(nxt.tolist())
        if score < best_score:
            best_score = score
            best_seq = nxt.copy()
            cur = nxt
            no_improve = 0
            eps_factor = max(0.02, eps_factor * 0.99)   # slowly tighten steps
            print(f"Iter {it:6d} – new best {best_score:.6f}")
        else:
            no_improve += 1
            # Occasionally accept a non‑improving move (simulated‑annealing style)
            if np.random.rand() < 1.0 / (10 + no_improve):
                cur = nxt
                eps_factor = min(0.30, eps_factor * 1.05)

        # -----------------------------------------------------------------
        # Global jitter every few thousand iterations to keep the search
        # from getting trapped in a shallow basin.
        # -----------------------------------------------------------------
        if it % 2000 == 0:
            cur = np.maximum(cur + np.random.normal(scale=5e-5 * target_sum, size=n), 0.0)
            cur = _project_to_simplex(cur, target_sum)

    return best_seq.tolist()
