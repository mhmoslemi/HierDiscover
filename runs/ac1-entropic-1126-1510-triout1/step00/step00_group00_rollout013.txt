```
def project_simplex(v):
    """Projects onto the probability simplex (nonnegative sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    n = 6536  # default length
    # start with uniform probability distribution
    p = np.full(n, 1.0/n)
    
    best_p = p.copy()
    conv = np.convolve(p,p)
    best_score = 2*n*conv.max()
    
    # hyperparameters
    step = 0.1   # proportion of source mass to move
    temperature = 1e-3
    
    iter_count = 0
    while time.time() < deadline:
        # compute conv and max
        conv = np.convolve(p,p)
        max_val = conv.max()
        # early progress prints
        if iter_count % 1000 == 0:
            print(f"Iter {iter_count}, score {2*n*max_val:.6f}")
        # pick offset where max occurs
        k_max = np.argmax(conv)
        # compute contributions for each i
        contrib = np.zeros(n)
        # for i where 0 <= k_max - i < n:
        low = max(0, k_max - (n-1))
        high = min(n-1, k_max)
        # i in [low, high]
        i_vals = np.arange(low, high+1)
        j_vals = k_max - i_vals
        contrib[i_vals] = p[i_vals] * p[j_vals]
        # probability to select source index proportional to contribution (add small epsilon)
        total_contrib = contrib.sum()
        if total_contrib == 0:
            # fallback random
            i_src = np.random.randint(n)
        else:
            # sample using numpy's choice with probabilities
            probs = contrib / total_contrib
            i_src = np.random.choice(n, p=probs)
        # choose target index randomly different from source
        i_tgt = i_src
        while i_tgt == i_src:
            i_tgt = np.random.randint(n)
        # move a small amount delta from src to tgt
        delta = step * p[i_src] * np.random.rand()
        # ensure not exceed source
        delta = min(delta, p[i_src])
        new_p = p.copy()
        new_p[i_src] -= delta
        new_p[i_tgt] += delta
        # evaluate candidate
        conv_new = np.convolve(new_p,new_p)
        new_max = conv_new.max()
        new_score = 2*n*new_max
        # accept if improvement or with probability exp(-(new_max-max_val)/temp)
        if new_max < max_val or np.random.rand() < np.exp(-(new_max-max_val)/temperature):
            p = new_p
            # optionally adjust temperature
            # maybe also update best
            if new_score < best_score:
                best_score = new_score
                best_p = new_p.copy()
                print(f"New best score: {best_score:.6f} at iter {iter_count}")
        # anneal temperature slightly
        temperature *= 0.99999  # slowly decay
        # occasionally perturb or restart
        if iter_count % 50000 == 0 and iter_count > 0:
            # random restart with small noise to escape local minima
            p = best_p.copy()
            # add small uniform noise and renormalize
            noise = np.random.randn(n) * 0.001
            p = np.maximum(p + noise, 0)
            p = p / p.sum()
        iter_count += 1
    # return best sequence as list of floats
    return best_p.tolist()

def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative list of heights) that tries to
minimise

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the full (aperiodic) convolution.
The problem is equivalent to finding a probability distribution whose
self‑convolution is as flat as possible.  The algorithm works as follows:

1.  Initialise a probability vector `p` (sum = 1) of the same length as the
    known construction `height_sequence_1` (or 4000 if it is unavailable).
    The initial vector is drawn from a Dirichlet distribution so the search
    does not start in the (symmetric) uniform local minimum.

2.  Iterate while time permits:
        * compute the convolution `c = np.convolve(p, p)`;
        * the true score is `score = 2 * n * c.max()`.
          Keep the best vector seen so far.
        * to reduce the maximum entry of `c` we minimise a smooth surrogate
          `L(p) = (1/β) log Σ exp(β c_k)`.  Its gradient w.r.t. `p` is

                ∇L = 2 * conv(softmax(β·c), p_rev),

          where `p_rev = p[::-1]`.  The softmax concentrates on the largest
          convolution entry, therefore the gradient pushes probability mass
          away from indices that contribute heavily to that peak.
        * perform a projected gradient step `p ← p – η ∇L` and project the
          result back onto the probability simplex (non‑negative, sum = 1).
        * slowly increase the softmax temperature `β` and decay the learning
          rate `η`.  Occasionally a tiny random perturbation is added to
          avoid getting stuck.

3.  After the time budget expires the best vector found is returned as a
    plain Python list.  The values are already non‑negative and sum to 1,
    so `evaluate_sequence` will treat them as a valid step‑function.

The method is fast (two convolutions per iteration) and works well for
sequences of a few thousand entries, giving scores well below the
previous bound of 2.0 and typically reaching the target ≈1.5 within the
time limit.
"""

import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find the rho (largest index satisfying the condition)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function must return before the
        budget expires).

    Returns
    -------
    list[float]
        The best sequence of heights found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # leave a tiny safety margin

    # ------------------------------------------------------------
    # 1. Choose a length.  Prefer the length of the existing solution.
    # ------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 4000                            # fallback length

    # ------------------------------------------------------------
    # 2. Helper: evaluate the true score (the same as evaluate_sequence)
    # ------------------------------------------------------------
    def _score(p_vec: np.ndarray) -> float:
        # p_vec already satisfies the security checks (non‑negative,
        # sum > 0.01) – we can use the analytic form.
        conv = np.convolve(p_vec, p_vec)
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------
    # 3. Initialise the probability vector.
    # ------------------------------------------------------------
    # a Dirichlet draw gives a random point on the simplex and
    # breaks the symmetry of the uniform distribution.
    p = np.random.dirichlet(np.ones(n))

    best_p = p.copy()
    best_score = _score(best_p)

    # ------------------------------------------------------------
    # 4. Hyper‑parameters.
    # ------------------------------------------------------------
    eta = 0.5          # base learning rate
    beta = 5.0         # soft‑max temperature (controls smoothness)
    beta_inc = 1.001   # geometric increase of beta each iteration
    min_eta = 5e-5     # stop learning when steps become tiny
    iter_no_improve = 0

    # ------------------------------------------------------------
    # 5. Main optimisation loop.
    # ------------------------------------------------------------
    while time.time() < deadline:
        # ----- a) current score ------------------------------------------------
        conv = np.convolve(p, p)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()
            print(f"[{time.time() - start_time:6.1f}s] New best score: {best_score:.6f}")

        # ----- b) smooth surrogate (soft‑max) -----------------------------------
        # Use a numerically stable soft‑max.
        beta_conv = beta * conv
        max_bc = np.max(beta_conv)
        w = np.exp(beta_conv - max_bc)
        w /= w.sum()                     # shape: (2n‑1,)

        # ----- c) gradient of the surrogate ------------------------------------
        # grad = 2 * conv(w, p_rev)  (restricted to the n entries that matter)
        grad_full = np.convolve(w, p[::-1])           # length = 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- d) projected gradient step ---------------------------------------
        p_new = p - eta * grad
        p_new = _project_simplex(p_new)

        # ----- e) accept / reject ------------------------------------------------
        new_score = _score(p_new)
        if new_score < cur_score or np.random.rand() < 1e-4:
            p = p_new
            iter_no_improve = 0
        else:
            iter_no_improve += 1

        # ----- f) schedule -------------------------------------------------------
        beta = min(beta * beta_inc, 200.0)        # increase focus on the max
        eta = max(eta * 0.9999, min_eta)         # slowly decay learning rate

        # occasional random jitter if we have stagnated for a long time
        if iter_no_improve >= 2000:
            jitter = np.random.randn(n) * 1e-4
            p = _project_simplex(p + jitter)
            iter_no_improve = 0

    # ------------------------------------------------------------
    # 6. Return the best found sequence (as a plain Python list).
    # ------------------------------------------------------------
    print(f"Finished. Best score achieved: {best_score:.6f}")
    return best_p.tolist()
