    # random lengths
    lengths = np.random.multisample? Use np.random.rand(k) and scale.
    lengths = np.floor(np.random.rand(k) * n).astype(int)
    # ensure sum exactly n: adjust last.
    # Actually generate cut points: sample k-1 random integers between 1 and n-1, sort, define intervals lengths.
    cuts = sorted(np.random.choice(range(1,n), size=k-1, replace=False))
    starts = [0] + cuts
    ends = cuts + [n]
    lengths = [e - s for s, e in zip(starts, ends)]
    # generate values for each block
    values = np.random.rand(k) * 5  # random values up to 5
    # build sequence
    seq = np.empty(n, dtype=float)
    idx = 0
    for L, val in zip(lengths, values):
        seq[idx:idx+L] = val
        idx += L
    # Normalize sum to n: i.e., scale so sum = n (average=1)
    seq = seq * (n / np.sum(seq))
    return seq.tolist()

"""
propose_candidate
-----------------
Search for a step‑function (list of non‑negative heights) that gives a low
`evaluate_sequence` score.

The key observation is that for a sequence whose non‑zero entries all have the
same value `w`, the evaluation reduces to

    score = 2 * n * max_count / m**2

where
    n         – length of the list,
    m         – number of non‑zero entries,
    max_count – maximal number of unordered pairs (i,j) with i+j = k
                (i.e. maximal collision of pair‑sums).

If we can place many indices so that each possible sum occurs only a few
times, `max_count` stays small while `m` grows, dramatically lowering the
score.  This is the classic “B₂[g]” (Sidon‑type) construction.  We build such
sets greedily: we keep an array `cnt` of how many pairs already produce each
sum and add a new index only if no sum would exceed a chosen bound `c_max`
(2 or 3 in practice).  After a set of size `m` is obtained we assign the same
weight `w = n / m` to every selected position – the scaling does not change
the score because both numerator and denominator scale quadratically.

The algorithm repeatedly tries random `n` (1500–2500) and random
`c_max` (2 or 3) within the time budget, keeps the best sequence found and
finally returns it.
"""

import time
import numpy as np


def _greedy_b2_set(n: int, c_max: int) -> tuple[list[int], int]:
    """
    Build a set of indices {0,…,n‑1} such that each sum i+j appears at most
    `c_max` times (including i=j).  The construction is randomised – indices
    are examined in a random order and kept if the bound is respected.

    Returns
    -------
    positions : list[int]
        Accepted indices.
    max_cnt : int
        The maximum collision count after the construction (should be ≤ c_max).
    """
    # count of pairs giving each possible sum, length = 2*n-1 (0 … 2n‑2)
    cnt = np.zeros(2 * n - 1, dtype=np.int16)

    # random permutation of all candidate indices
    cand = np.arange(n, dtype=np.int32)
    np.random.shuffle(cand)

    positions = []
    for i in cand:
        # self‑pair i+i
        if cnt[i + i] >= c_max:
            continue

        # check all cross‑pairs i+j for already chosen j
        collision = False
        for j in positions:
            if cnt[i + j] >= c_max:
                collision = True
                break
        if collision:
            continue

        # accept i – update counters
        for j in positions:
            cnt[i + j] += 1
        cnt[i + i] += 1
        positions.append(i)

    max_cnt = int(cnt.max())
    return positions, max_cnt


def _build_sequence(n: int, positions: list[int]) -> list[float]:
    """
    From a list of chosen positions create the actual step‑function.
    All non‑zero entries get the same weight so that the total sum equals `n`
    (average height = 1).  This normalisation does not affect the score.
    """
    m = len(positions)
    if m == 0:          # should never happen, safeguard
        return [1.0] * n
    w = n / m           # weight for every selected position
    seq = np.zeros(n, dtype=float)
    seq[positions] = w
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of heights (non‑negative floats) that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s` seconds;
    during that time a greedy B₂[g] construction is repeatedly performed with
    different parameters and the best sequence found is kept.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2  # small safety margin

    best_seq = None
    best_score = float("inf")

    # fallback – uniform sequence (score = 2) in case nothing better is found
    fallback_n = 1600
    fallback_seq = [1.0] * fallback_n
    try:
        fallback_score = evaluate_sequence(fallback_seq)
    except Exception:
        fallback_score = float("inf")
    best_seq = fallback_seq
    best_score = fallback_score

    # main search loop
    while time.time() < deadline:
        # random length and allowed collision bound
        n = np.random.randint(1500, 2501)           # 1500 … 2500
        c_max = np.random.choice([2, 3])           # allow 2 or 3 collisions

        # try a few independent greedy constructions and keep the best one
        for _ in range(3):
            positions, max_cnt = _greedy_b2_set(n, c_max)
            if not positions:
                continue

            seq = _build_sequence(n, positions)
            # evaluate – this also applies the protection logic inside the
            # provided `evaluate_sequence` function.
            try:
                score = evaluate_sequence(seq)
            except Exception:
                continue

            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"[{time.time():.1f}] New best: score={score:.6f}, "
                      f"n={n}, m={len(positions)}, max_cnt={max_cnt}")

            # Early stop if we already beat the target of 1.503
            if best_score < 1.503:
                # Still keep running a little while to possibly improve further,
                # but we have already reached the required region.
                break

        # Occasionally try a simple perturbation of the current best sequence
        # (randomly shift a small amount of mass).  This may escape local minima.
        if best_seq is not None and np.random.rand() < 0.2:
            seq = np.array(best_seq, copy=True)
            idx = np.random.randint(len(seq))
            delta = np.random.randn() * 0.5
            seq[idx] = max(0.0, seq[idx] + delta)
            # re‑normalise to keep the same total sum (scaling does not change
            # the objective, but keeps the numbers in a reasonable range)
            s = seq.sum()
            if s > 0.0:
                seq = seq * (len(seq) / s)   # force sum = n
            try:
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f"[{time.time():.1f}] Perturb improvement: score={score:.6f}")
            except Exception:
                pass

    # final safety: ensure the sequence is a plain Python list of floats
    if best_seq is None:
        return fallback_seq
    return [float(x) for x in best_seq]
