"""
Propose a candidate step‑function height sequence.

Algorithm
---------
* Randomly pick a length `n` (200‑1500) and generate a sparse, non‑negative
  vector whose sum is the normalisation √(2 n).  The sparsity favours low
  convolution peaks.
* Perform an iterative “flatten‑convolution’’ optimisation:
    - Let `h = a * a` be the linear convolution.
    - The average of `h` is `μ = (∑a)² / (2 n‑1)`.  We minimise the squared
      deviation `∑(h‑μ)²`.  Its gradient w.r.t. `a` is
          ∇ = 4 * (a  ⊛  (h‑μ)),
      where `⊛` denotes convolution.
    - After each gradient step we project back onto the simplex
      `{a ≥ 0,  ∑a = √(2 n)}`.
    - The current best sequence (smallest `evaluate_sequence`) is recorded.
* Occasionally improve the current iterate with the linear‑programming
  direction used by the state‑of‑the‑art code (`get_good_direction_to_move_into`);
  a tiny line‑search on the convex combination of the two sequences is performed.
* The whole search runs until the supplied time budget expires (≈ 1000 s).
  The best sequence seen is returned.
"""

import time
import math
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x ≥ 0, sum(x)=s}
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of sum s."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]          # the last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# LP direction used by the reference implementation
def solve_convolution_lp(f_sequence, rhs):
    """Linear programme:
       maximise   Σ b_j
       subject to   (f * b)_k ≤ rhs   for all k
                    b_j ≥ 0
    Returns the optimal b (numpy array) or None.
    """
    n = len(f_sequence)
    c = -np.ones(n)                     # maximise Σ b_j  <=> minimise -Σ b_j
    a_ub = []
    b_ub = []
    # convolution constraints
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints  -b ≤ 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)
    a_ub = np.vstack([a_ub, a_ub_nonneg])
    b_ub = np.hstack([b_ub, b_ub_nonneg])
    # solve – we use a small time limit to keep the outer loop fast
    res = optimize.linprog(c,
                           A_ub=a_ub,
                           b_ub=b_ub,
                           method='highs',
                           options={'time_limit': 5.0})
    return res.x if res.success else None


def get_good_direction_to_move_into(sequence):
    """Return a new sequence obtained by the LP direction described in the
       reference code.  The returned sequence is already normalised to the
       required sum √(2 n)."""
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-9:
        return None
    # normalise to √(2 n)
    norm_seq = np.array(sequence) * math.sqrt(2.0 * n) / sum_seq
    rhs = np.max(np.convolve(norm_seq, norm_seq))
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None
    # renormalise g to the same total sum √(2 n)
    sum_g = np.sum(g)
    norm_g = g * math.sqrt(2.0 * n) / sum_g
    # a small convex combination (the reference used t=0.01);
    # we keep the pure direction – the outer routine will decide the step.
    return norm_g.tolist()

        """Run the gradient‑based flatten‑convolution optimisation for a single
           length n.  Returns (seq, score) where seq is the best sequence seen."""
        S = math.sqrt(2.0 * n)                # required sum of the coefficients
        # ----- sparse random start -------------------------------------------------
        # we deliberately generate a heavy‑tailed start (exponential) and then
        # enforce a modest level of sparsity.
        raw = np.random.exponential(scale=1.0, size=n)
        # keep roughly 30‑70 % of the entries
        keep_mask = np.random.rand(n) < np.random.uniform(0.3, 0.7)
        raw[~keep_mask] = 0.0
        if raw.sum() == 0.0:
            raw = np.random.rand(n)  # fallback
        a = raw / raw.sum() * S

        # ----- optimisation loop ---------------------------------------------------
        avg = S * S / (2 * n - 1)      # average value of the convolution
        lr0 = 1.0                      # initial learning rate
        max_iters = 800                # enough for convergence but cheap
        best_local_seq = a.copy()
        best_local_score = evaluate_sequence(a.tolist())

        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # current convolution
            conv = np.convolve(a, a)          # length 2n‑1
            cur_max = conv.max()
            if cur_max < best_local_score:
                best_local_score = cur_max
                best_local_seq = a.copy()

            # gradient of Σ (h‑avg)²  (where h = a*a)
            residual = conv - avg
            full = np.convolve(a, residual)   # length 3n‑2
            start = len(residual) - 1         # = 2n‑2
            grad = 4.0 * full[start:start + n]   # shape (n,)

            # simple decaying step size
            lr = lr0 / (1.0 + it * 0.02)
            a = a - lr * grad
            # enforce feasibility
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, S)

        # final evaluation of the best local sequence
        final_score = evaluate_sequence(best_local_seq.tolist())
        return best_local_seq, final_score

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Choose a length for this trial; vary it to explore different
        # dimensionalities (larger n gives more freedom but costs more).
        n = np.random.randint(250, 1500)

        # Run the optimiser for this length (it will stop early if time runs out)
        seq, score = optimise_for_length(n, deadline - time.time())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[{time.strftime('%H:%M:%S')}] New best: {best_score:.6f} "
                  f"with n={n} (iteration {iteration})")

        # ------------------------------------------------------------------
        # Occasionally try the LP direction on the current best sequence.
        # This cheap extra step can give a noticeable boost.
        if best_seq is not None and iteration % 3 == 0:
            direction = get_good_direction_to_move_into(best_seq.tolist())
            if direction is not None:
                # tiny line‑search on the convex combination (ternary search)
                low, high = 0.0, 1.0
                for _ in range(12):
                    t1 = low + (high - low) / 3.0
                    t2 = high - (high - low) / 3.0
                    cand1 = (1.0 - t1) * best_seq + t1 * np.array(direction)
                    cand2 = (1.0 - t2) * best_seq + t2 * np.array(direction)
                    s1 = evaluate_sequence(cand1.tolist())
                    s2 = evaluate_sequence(cand2.tolist())
                    if s1 < s2:
                        high = t2
                    else:
                        low = t1
                t_opt = (low + high) / 2.0
                new_seq = (1.0 - t_opt) * best_seq + t_opt * np.array(direction)
                # re‑project onto the simplex (maintains the required sum)
                S = math.sqrt(2.0 * len(new_seq))
                new_seq = np.maximum(new_seq, 0.0)
                new_seq = _project_to_simplex(new_seq, S)
                new_score = evaluate_sequence(new_seq.tolist())
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    print(f"[{time.strftime('%H:%M:%S')}] LP boost: score {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should never happen) return a trivial sequence
    if best_seq is None:
        return [1.0]

    return best_seq.tolist()
