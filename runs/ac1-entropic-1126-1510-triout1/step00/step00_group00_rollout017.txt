def propose_candidate(...):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    n = len(height_sequence_1)
    target_sum = np.sqrt(2 * n)

    # start with uniform
    base = np.full(n, target_sum / n)
    # optionally add small anti-symmetric noise
    current = base.copy()
    best_seq = current.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score}")

    # function to compute max conv quickly
    def compute_score(seq):
        # conv via FFT
        size = 2*n -1
        nfft = 1 << (size-1).bit_length()
        f = np.fft.rfft(seq, n=nfft)
        conv = np.fft.irfft(f*f, n=nfft)[:size]
        return float(np.max(conv.real))

    # main loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # decide move type
        if np.random.rand() < 0.7:
            # antisymmetric move
            i = np.random.randint(n//2)
            j = n - 1 - i
            # choose direction random
            if np.random.rand() < 0.5:
                src, dst = j, i
            else:
                src, dst = i, j
            # amount
            eps = np.random.rand() * 0.02 * target_sum / n   # small fraction of average
            if eps > current[src]:
                eps = current[src] * 0.9
            # Apply move
            new_seq = current.copy()
            new_seq[src] -= eps
            new_seq[dst] += eps
            # Ensure non-negative
            if new_seq.min() < 0:
                continue
            new_score = compute_score(new_seq)
            # accept if better
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iteration}: improved to {best_score:.6f}")
                current = new_seq
            else:
                # maybe accept if improvement relative to current? For simulated annealing we can accept if new_score < compute_score(current) or with prob.
                # simple hill climbing: only accept if better than current
                current_score = compute_score(current)
                if new_score < current_score:
                    current = new_seq
        else:
            # random perturbation of two random indices
            i, j = np.random.choice(n, size=2, replace=False)
            eps = np.random.randn() * 0.01 * target_sum / n
            new_seq = current.copy()
            new_seq[i] = max(0.0, new_seq[i] + eps)
            new_seq[j] = max(0.0, new_seq[j] - eps)  # keep sum approx constant; we will renorm later
            # renormalize
            new_seq = new_seq * (target_sum / np.sum(new_seq))
            new_score = compute_score(new_seq)
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iteration}: perturbed best to {best_score:.6f}")
                current = new_seq
            else:
                # maybe accept if better than current
                cur_score = compute_score(current)
                if new_score < cur_score:
                    current = new_seq

        # periodic LP step every, say, 200 iterations
        if iteration % 200 == 0:
            # use current normalized for LP
            cur_sum = np.sum(current)
            if cur_sum == 0:
                continue
            f_norm = current * (np.sqrt(2*n) / cur_sum)
            M = compute_score(f_norm)
            g = solve_convolution_lp(f_norm.tolist(), M)
            if g is not None:
                g = np.array(g)
                # scale g to same sum
                g = g * (np.sqrt(2*n) / np.sum(g))
                # compute conv arrays via FFT
                size = 2*n -1
                nfft = 1 << (size-1).bit_length()
                fa = np.fft.rfft(f_norm, n=nfft)
                fg = np.fft.rfft(g, n=nfft)
                aa = np.fft.irfft(fa*fa, n=nfft)[:size]
                ag = np.fft.irfft(fa*fg, n=nfft)[:size]
                gg = np.fft.irfft(fg*fg, n=nfft)[:size]
                # binary search for largest t with max <= M
                lo, hi = 0.0, 1.0
                for _ in range(30):
                    mid = (lo+hi)/2
                    conv_mix = (1-mid)**2 * aa + 2*mid*(1-mid) * ag + (mid**2) * gg
                    if conv_mix.max() <= M:
                        lo = mid
                    else:
                        hi = mid
                t_opt = lo

                    new_seq = (1-t_opt) * f_norm + t_opt * g

                    current = new_seq
    # final convert to list of floats

"""
Propose a candidate sequence for the self‑convolution objective.

Algorithm
---------
1. **Initialisation** –  Build several start vectors:
   * uniform distribution,
   * a sparse “Sidon‑like” set where the number of unordered
     pairs that share the same sum is bounded (greedy construction),
   * a random Dirichlet sample.
   The one with the lowest maximum convolution (computed by FFT) is kept.

2. **Iterative sub‑gradient optimisation** –  With the current sequence `a`
   (always kept normalised so that sum(a)=sqrt(2·n)):
   * Compute the full convolution `c = a*a` (FFT, O(n log n)).
   * Let `k` be an index where `c` attains its maximum.
   * The derivative of `c_k` w.r.t. `a_i` is `2·a_{k‑i}` (zero when the index
     is out of range).  Form the vector  
        `d_i = 2·a_{k‑i}`  (sub‑gradient of the active constraint).
   * Subtract the mean `μ = 2·sum(a)/n` so that the direction has zero
     sum: `g = d – μ·1`.  A step `a ← a – η·g` keeps the total sum unchanged.
   * Clip negatives to zero and re‑normalise to the required sum.
   * Accept the step only if the maximum convolution decreases; otherwise
     shrink the step size `η`.  Random tiny perturbations are injected
     occasionally to escape plateaus.

3. **Time budget** –  The loop runs until the supplied time budget expires.
   The best sequence seen so far is returned.

The method never uses the expensive `np.convolve` inside the optimisation
loop – only FFT based convolutions are used, making it fast enough to
explore many thousands of iterations within the 1000 s limit.
"""

import numpy as np
import time
from scipy.optimize import linprog   # kept for possible fallback (not used)

# --------------------------------------------------------------------
# Helper: fast full convolution using FFT (real input)
def fft_convolve_self(x, nfft, size):
    """Return the full convolution of x with itself using a pre‑computed FFT size."""
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:size]
    return conv


# --------------------------------------------------------------------
def build_sparse_initial(n, target_sum, max_collisions=2, attempts=300):
    """
    Greedy construction of a set S ⊂ {0,…,n‑1} such that each unordered sum
    appears at most `max_collisions` times.  All indices in S receive the same
    weight w = target_sum / |S|.
    """
    best_set = []
    best_len = 0

    for _ in range(attempts):
        counts = np.zeros(2 * n - 1, dtype=int)      # unordered‑pair counters
        S = []
        order = np.random.permutation(n)
        for idx in order:
            # check whether inserting idx would violate the bound
            ok = True
            for j in S:
                if counts[idx + j] + 1 > max_collisions:
                    ok = False
                    break
            if not ok:
                continue
            if counts[2 * idx] + 1 > max_collisions:
                continue
            # accepted – update counters
            for j in S:
                counts[idx + j] += 1
            counts[2 * idx] += 1
            S.append(idx)

        if len(S) > best_len:
            best_len = len(S)
            best_set = S[:]

    seq = np.zeros(n)
    if best_len > 0:
        w = target_sum / best_len
        seq[best_set] = w
    return seq


# --------------------------------------------------------------------
def random_dirichlet_initial(n, target_sum):
    """Draw a random point on the simplex and scale to the required sum."""
    rnd = np.random.gamma(1.0, 1.0, size=n)
    seq = rnd / rnd.sum() * target_sum
    return seq


# --------------------------------------------------------------------
def compute_max_conv(seq, nfft, size):
    """Maximum value of the full self‑convolution of `seq`."""
    conv = fft_convolve_self(seq, nfft, size)
    return float(conv.max()), conv

    """
    Return a list of non‑negative heights that (hopefully) beat the
    current bound 2.0.  The function obeys the supplied time budget.
    """
    np.random.seed(seed)

    # ----------------------------------------------------------------

    try:

    nfft = 1 << (conv_size - 1).bit_length()

    # Build a few starting points and pick the best one
    starts = []

    # 1) uniform
    uniform = np.full(n, target_sum / n)
    starts.append(uniform)

    sparse = build_sparse_initial(n, target_sum, max_collisions=2, attempts=400)
    if sparse.sum() > 0:
        starts.append(sparse)

    # 3) random Dirichlet
    starts.append(random_dirichlet_initial(n, target_sum))

    # evaluate the three starters (fast FFT based evaluation)
    best_seq = None
    best_score = np.inf
    for s in starts:
        score, _ = compute_max_conv(s, nfft, conv_size)
        if score < best_score:
            best_score = score
            best_seq = s.copy()
    print(f"[init] best starting score = {best_score:.6f}")

    # ----------------------------------------------------------------
    # Sub‑gradient optimisation loop
    current = best_seq.copy()
    current_score = best_score
    eta = 0.6                         # initial step size
    no_improve = 0
    iteration = 0

    # constant that appears in the sub‑gradient mean term
    avg_term = 2.0 * target_sum / n   # = 2·sum(a)/n  (sum(a) = target_sum)

    deadline = time.time() + budget_s - 2.0   # keep a safety margin

    while time.time() < deadline:
        iteration += 1

        # compute full convolution of the current iterate
        max_conv, conv_vec = compute_max_conv(current, nfft, conv_size)
        # locate an index attaining the maximum (any one is fine)
        k_max = int(np.argmax(conv_vec))

        # ---- sub‑gradient of the active quadratic form ----
        # d_i = 2 * a_{k_max-i}  for admissible i, else 0
        d = np.zeros_like(current)
        i_start = max(0, k_max - (n - 1))
        i_end = min(k_max, n - 1)
        idx = np.arange(i_start, i_end + 1)
        d[idx] = 2.0 * current[k_max - idx]

        # centred direction (zero‑sum) – ensures the total sum stays unchanged
        direction = d - avg_term

        # propose a new point
        cand = current - eta * direction
        np.maximum(cand, 0.0, out=cand)          # enforce non‑negativity
        s = cand.sum()
        if s == 0.0:                     # degenerate – skip this iteration
            continue
        cand *= (target_sum / s)         # re‑normalise to the required sum

        # evaluate the candidate
        cand_score, _ = compute_max_conv(cand, nfft, conv_size)

        if cand_score < max_conv - 1e-12:
            # successful descent – keep the move
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{iteration:6d}] new best score = {best_score:.6f}")
            no_improve = 0
            # slowly increase the step size after a success (but cap it)
            eta = min(1.0, eta * 1.02)
        else:
            # no improvement – shrink the step size
            no_improve += 1
            eta *= 0.70
            if eta < 1e-5:
                eta = 1e-5

        # occasional random jitter to help escape plateaus
        if iteration % 250 == 0:
            jitter = np.random.randn(n) * 0.005 * avg_term
            cand2 = current + jitter
            np.maximum(cand2, 0.0, out=cand2)
            cand2 *= (target_sum / cand2.sum())
            cand2_score, _ = compute_max_conv(cand2, nfft, conv_size)
            if cand2_score < best_score:
                best_score = cand2_score
                best_seq = cand2.copy()
                current = cand2
                print(f"[{iteration:6d}] jitter improved best to {best_score:.6f}")

    # ----------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    print(f"[final] best score achieved = {best_score:.6f}")
    return best_seq.tolist()
