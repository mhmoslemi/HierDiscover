"""
Propose a step‑function sequence using the discrete arcsine distribution.

The continuous arcsine density f(x)=1/(π√{x(1−x)}) on (0,1) has the property
that f∗f is exactly the uniform density on (0,2).  After discretising
the density on n points (x_i=(i+½)/n) and normalising the total sum to
√(2n) – the same normalisation used by the reference algorithm – the
self‑convolution becomes almost constant.  Consequently the evaluation
metric 2 n·max(conv)/sum(a)² (which equals max(conv) after the
normalisation) approaches its theoretical lower bound 1.

The routine builds this arcsine‑shaped sequence, evaluates its score,
and then spends any remaining time on a tiny hill‑climbing loop that
preserves the total sum while trying small random transfers of mass
between two positions.  The loop is deliberately lightweight so that
the whole search finishes well within the 1000 s budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for the optional local search phase.
    budget_s : float, optional
        Time budget in seconds.  The function stops a few seconds before
        the limit to avoid a hard timeout.
    kwargs : dict
        Ignored – present for compatibility with the evaluation harness.

    Returns
    -------
    list of float
        The candidate height sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # 1) Build the discrete arcsine distribution.
    # ------------------------------------------------------------------
    # Choose a length that balances quality and speed.
    n = 8000                     # can be increased if ample time remains
    target_sum = np.sqrt(2.0 * n)   # the normalisation used by the SOTA code

    # x_i = (i + 0.5) / n lies strictly inside (0,1), avoiding division by zero.
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # Unnormalised arcsine density (continuous version would be divided by π;
    # the factor π is absorbed in the later normalisation to target_sum.)
    raw = 1.0 / np.sqrt(x * (1.0 - x))

    # Scale so that the sum equals sqrt(2n); this makes the evaluation
    # metric directly equal to the maximum convolution entry.
    seq = raw * (target_sum / raw.sum())

    # Ensure non‑negativity (the formula is strictly positive, but we mask
    # any possible tiny negative round‑off just in case).
    seq = np.maximum(seq, 0.0)

    # ------------------------------------------------------------------
    # 2) Initial evaluation.
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Light‑weight local search (optional, runs only while time permits).
    # ------------------------------------------------------------------
    # We keep the total sum fixed by re‑normalising after each tweak.
    deadline = start_time + budget_s - 5.0   # keep a small safety margin
    iteration = 0
    max_iters = 5000                         # hard cap on loop count

    while time.time() < deadline and iteration < max_iters:
        # Choose two distinct indices and transfer a small amount of mass.
        i_idx = np.random.randint(n)
        j_idx = np.random.randint(n - 1)
        if j_idx >= i_idx:
            j_idx += 1   # ensure i_idx != j_idx

        # Propose a transfer proportional to the source value.
        delta = best_seq[i_idx] * 0.02 * np.random.randn()  # ~2 % perturbation
        if delta == 0.0:
            continue

        # Apply the transfer, respecting non‑negativity.
        src = best_seq[i_idx] + delta
        dst = best_seq[j_idx] - delta

        if src < 0.0:
            # Too much taken away – clamp.
            delta = -best_seq[i_idx]  # send all mass from i_idx to j_idx
            src = 0.0
            dst = best_seq[j_idx] - delta
        if dst < 0.0:
            # Too much taken from j_idx – clamp.
            delta = best_seq[j_idx]  # move everything from j_idx to i_idx
            src = best_seq[i_idx] + delta
            dst = 0.0

        new_seq = best_seq.copy()
        new_seq[i_idx] = src
        new_seq[j_idx] = dst

        # Renormalise to keep the prescribed total sum.
        s = new_seq.sum()
        if s <= 0.0:
            continue
        new_seq *= (target_sum / s)

        # Evaluate the candidate.
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:
            # Defensive: any unexpected error just abort this move.
            continue

        # Accept if we improve (lower score is better).
        if score < best_score:
            best_score = score
            best_seq = new_seq
            print(f"[iter {iteration}] improved score={best_score:.6f}")

        iteration += 1

    print(f"[final] iterations={iteration}, best score={best_score:.6f}")
    return best_seq.tolist()
