def convolve(a):
    size = len(a)
    m = 1 << ((2*size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    B = A * A
    conv = np.fft.irfft(B, n=m)[:2*size-1]
    return conv

    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 80  # moderate size
    # start with uniform distribution
    a = np.random.rand(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    cur_score = best_score

    iteration = 0
    # temperature schedule (initial high for exploration)
    T0 = 0.02
    while time.time() < deadline:
        # compute convolution and max index
        conv = np.convolve(a, a)
        M = conv.max()
        kmax = np.argmax(conv)
        # decide indices
        # active i indices that affect kmax
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        active_i = np.arange(i_start, i_end + 1)
        # pick i from active
        i = np.random.choice(active_i)
        # pick j from whole range (maybe not active)
        j = np.random.randint(n)
        # small transfer
        epsilon = min(a[i], 0.1 * a[i])  # proportion
        if epsilon <= 0:
            continue
        # propose new distribution
        a_new = a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon
        # ensure nonnegative
        if a_new[i] < 0:
            continue
        # evaluate
        new_score = evaluate_sequence(a_new.tolist())
        # accept if better or with probability
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (T0 * max(1e-9, (deadline - time.time())))):
            a = a_new
            cur_score = new_score
            # maybe update best
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"Iter {iteration}: new best {best_score:.6f}")
        # occasionally random perturb all
        if iteration % 1000 == 0 and iteration > 0:
            # small random jitter
            a = a + 0.01 * np.random.randn(n) * a
            a = np.maximum(a, 0)
            a /= a.sum()
            cur_score = evaluate_sequence(a.tolist())
        iteration += 1
    # final best sequence
    return best_a.tolist()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    # try different lengths
    lengths = [100, 150, 200, 250]
    best_score = float('inf')
    best_seq = None

    for n in lengths:
        # initialize with uniform
        a = np.full(n, 1.0 / n)

        # parameters
        eta = 0.5
        decay = 0.9995
        iter_no_improve = 0
        best_local_score = float('inf')
        best_local_a = a.copy()
        iteration = 0

        while time.time() < deadline:
            # compute convolution
            b = np.convolve(a, a)
            avg = 1.0 / (2 * n - 1)   # since sum(a)=1 -> sum(b)=1
            diff = b - avg

            # gradient of variance = 4 * correlation(diff, a)
            corr = np.correlate(diff, a, mode='full')

            # gradient step
            a_new = a - eta * g
            # ensure nonnegative
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s == 0:

                a_new = a_new / a_new.sum()
            else:
                a_new = a_new / s

"""
Propose a candidate step‑function by minimizing the variance of its
self‑convolution (i.e. making the autocorrelation as flat as possible).
The algorithm works for any length n:
    1. Initialise a non‑negative vector a with sum = 1.
    2. Repeatedly:
        * Compute b = a * a (full convolution).
        * The average of b is 1/(2n‑1) when sum(a)=1.
        * Let diff = b – avg.  The gradient of the sum of squared
          deviations Σ(diff_k²) w.r.t. a is 4·corr(diff, a).
        * Update a multiplicatively: a ← a·exp(‑η·grad) and renormalise.
        * Keep the best sequence seen according to the provided
          evaluate_sequence() function.
    3. After the gradient phase, perform a short hill‑climbing
       (move a small amount of mass between two coordinates) to
       polish the solution.
The whole search respects the 1000 s budget and stays well within the
value limits (all entries stay < 1, far below the 1000 cap).
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence().  The routine uses a variance‑reduction gradient
    descent on the self‑convolution, followed by a tiny hill‑climbing
    refinement.  It returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # small safety margin

    best_score = float("inf")
    best_seq = None

    # try a few different lengths (larger n gives a tighter theoretical bound)
    for n in (120, 160, 200, 240):
        # stop early if we run out of time
        if time.time() > deadline:
            break

        # start from the uniform distribution (sum = 1)
        a = np.full(n, 1.0 / n, dtype=np.float64)

        # gradient‑descent parameters
        eta = 0.45          # step size for the multiplicative update
        decay = 0.9995      # slowly decay the step size
        it_no_improve = 0   # counter for stagnation detection

        # local best for this length
        local_best_score = evaluate_sequence(a.tolist())
        local_best_a = a.copy()
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = a.copy()
            print(f"[init] n={n}, score={best_score:.6f}")

        iteration = 0
        while time.time() < deadline:
            # 1) convolution and deviation from the flat target
            b = np.convolve(a, a)                     # length 2n‑1
            avg = 1.0 / (2 * n - 1)                   # because sum(a)=1 ⇒ sum(b)=1
            diff = b - avg

            # 2) gradient of the variance of b w.r.t. a
            #    grad = 4 * correlation(diff, a) ; correlation via np.correlate
            corr = np.correlate(diff, a, mode="full")
            #   corr[l] corresponds to lag l-(len(a)-1); we need lags 0..n‑1 → indices n‑1‑i
            grad = 4.0 * corr[n - 1 :: -1][:n]

            # 3) multiplicative update (keeps a >= 0 automatically)
            a_new = a * np.exp(-eta * grad)

            # renormalise to keep sum = 1 (scaling does not affect the objective)
            sum_a = a_new.sum()
            if sum_a == 0.0:
                # numerically collapsed – re‑initialise randomly
                a_new = np.random.rand(n)
                a_new /= a_new.sum()
            else:
                a_new /= sum_a

            # evaluate the candidate
            score = evaluate_sequence(a_new.tolist())

            # keep the globally best sequence
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                print(f"[global] n={n}, iter={iteration}, new best {best_score:.6f}")

            # keep the best for this length (used for the final polishing)
            if score < local_best_score:
                local_best_score = score
                local_best_a = a_new.copy()
                it_no_improve = 0
            else:
                it_no_improve += 1

            a = a_new
            iteration += 1
            eta *= decay

            # if no improvement for a while, restart from a fresh random point
            if it_no_improve > 2000:
                a = np.random.rand(n)
                a /= a.sum()
                it_no_improve = 0
                eta = 0.45

            # avoid overrunning the global deadline
            if time.time() > deadline - 0.5:
                break

        # -----------------------------------------------------------------
        # Small hill‑climbing phase to polish the current local optimum.
        # -----------------------------------------------------------------
        a = local_best_a.copy()
        cur_score = local_best_score
        for _ in range(5000):
            if time.time() > deadline - 0.2:
                break

            # pick two different indices and move a tiny amount of mass
            i, j = np.random.choice(n, 2, replace=False)
            # transfer up to 10 % of the source weight, but not more than the weight itself
            eps = min(a[i], 0.1 * a[i])
            if eps <= 0.0:
                continue
            a_candidate = a.copy()
            a_candidate[i] -= eps
            a_candidate[j] += eps

            # re‑normalise (sum is unchanged, but keep for safety against round‑off)
            a_candidate /= a_candidate.sum()

            cand_score = evaluate_sequence(a_candidate.tolist())
            if cand_score < cur_score:
                a = a_candidate
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = a.copy()
                    print(f"[polish] n={n}, iter={iteration}, improved to {best_score:.6f}")

    # fallback – should never happen, but guarantees a return value
    if best_seq is None:
        best_seq = np.full(10, 1.0 / 10)

    # convert to plain Python list of floats (the evaluator expects a list)
    return best_seq.tolist()
