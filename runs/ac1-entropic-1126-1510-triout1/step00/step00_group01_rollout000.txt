def compute_score_and_conv(a):
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 1e-12:
        return np.inf, conv
    score = 2*n*max_conv/(sum_a**2)
    return score, conv

def grad_softmax(a, gamma):
    n = len(a)
    conv = np.convolve(a, a)
    # compute w = softmax(gamma * conv)
    # stable compute:
    max_val = np.max(gamma*conv)
    exp = np.exp(gamma*conv - max_val)
    w = exp/np.sum(exp)
    # compute gradient
    a_rev = a[::-1]
    grad_full = np.convolve(w, a_rev, mode='full')
    grad = 2 * grad_full[n-1:n-1+n]
    return grad

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """...
    """

    import numpy as np, time, math, sys

    # set random seeds
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5.0  # safety margin

    # Choose length
    n = 5000  # near the given construction length
    # we could use original length if available
    try:
        # Use same length as existing sequence if it's longer
        global height_sequence_1
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            n_alt = len(height_sequence_1)
            if n_alt >= 2000:
                n = n_alt
    except NameError:
        pass

    # Initialization
    a = np.ones(n, dtype=np.float64) / n  # uniform positive

    # If existing sequence present, optionally use it as start (normalized)
    try:
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            seq = np.array(height_sequence_1, dtype=np.float64)
            if seq.shape[0] == n:
                # Use that as start
                a = np.maximum(seq, 0.0)
                s = a.sum()
                if s > 0:
                    a /= s
            elif seq.shape[0] > 0:
                # Interpolate to n length
                x_old = np.linspace(0, 1, len(seq))
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, seq)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 0:
                    a /= s
    except NameError:
        pass

    # Helper functions
    def compute_score(vec):
        """Return the evaluate_sequence score for a given vector (numpy array)."""
        n_local = len(vec)
        s = np.sum(vec)
        if s < 1e-12:
            return np.inf
        conv = np.convolve(vec, vec)
        max_conv = np.max(conv)
        return 2.0 * n_local * max_conv / (s * s)

    # Initialize best
    best_seq = a.copy()
    best_score = compute_score(best_seq)

    # Optimization parameters
    eta = 0.5          # initial step size
    eta_min = 1e-4
    gamma = 1.0        # initial softmax temperature
    gamma_max = 200.0
    iter_count = 0
    no_improve = 0
    # For timing logs
    next_log = start + 10.0

    while time.time() < deadline:
        iter_count += 1

        # forward conv
        conv = np.convolve(a, a)
        max_conv = np.max(conv)
        sum_a = np.sum(a)
        # score for current a (could be computed more efficiently since we have them)
        score = 2.0 * n * max_conv / (sum_a * sum_a)

        # compute softmax weights
        # use scaling to avoid overflow
        scaled = gamma * conv
        max_scaled = np.max(scaled)
        exp_vals = np.exp(scaled - max_scaled)
        w = exp_vals / np.sum(exp_vals)

        # gradient of the softmax approximation of max(conv)
        # grad_i = 2 * sum_k w_k * a_{k-i}
        # compute via convolution of w with reversed a
        a_rev = a[::-1]
        grad_full = np.convolve(w, a_rev, mode='full')
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # gradient descent step
        a_new = a - eta * grad
        # enforce non-negativity
        a_new = np.maximum(a_new, 0.0)

        # re-normalize to keep scale (any scale works, we keep sum=1)
        sum_new = np.sum(a_new)
        if sum_new > 0:
            a_new /= sum_new
        else:
            # if everything went to zero, reinitialize uniformly
            a_new = np.ones_like(a) / n

        # compute new score
        conv_new = np.convolve(a_new, a_new)
        max_conv_new = np.max(conv_new)
        sum_new = np.sum(a_new)
        new_score = 2.0 * n * max_conv_new / (sum_new * sum_new)

        # accept if improvement
        if new_score < score:
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # no improvement; maybe still accept with small probability (simulated annealing)
            # probability decays with iteration count
            T = max(0.1, 1.0 - iter_count / 200000.0)  # temperature schedule
            if np.random.rand() < math.exp((score - new_score) / T):
                a = a_new
                no_improve = 0
            else:
                no_improve += 1

        # occasional random perturbation to escape stagnation
        if no_improve >= 2000:
            # add small Gaussian noise and renorm
            noise = np.random.randn(n) * 0.02
            a = np.maximum(a + noise, 0.0)
            a_sum = np.sum(a)
            if a_sum > 0:
                a /= a_sum
            no_improve = 0

        # schedule update
        eta = max(eta * 0.9995, eta_min)
        gamma = min(gamma * 1.001, gamma_max)

        # logging
        now = time.time()
        if now >= next_log:
            print(f"Iter {iter_count:6d} | Score {score:.6f} | Best {best_score:.6f} | eta {eta:.6e} | gamma {gamma:.3f}")
            next_log = now + 10.0

    # final best sequence as list
    print(f"Optimization finished after {iter_count} iterations. Best score: {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a candidate step‑function (non‑negative heights) that drives the
evaluation score 2·n·max(conv(a,a))/sum(a)^2 as low as possible.

The algorithm is a simple stochastic projected gradient descent on a smooth
approximation of the ℓ∞‑norm of the self‑convolution:

    f(a) = max_k (a*a)[k]      (scale‑invariant)

The max is approximated by a soft‑max with a temperature γ that is slowly
increased.  At each iteration we

1. compute the convolution a*a,
2. build soft‑max weights w_k,
3. obtain the gradient of  Σ_k w_k·(a*a)_k  (which equals
   2·(w * a_rev)[n‑1+i] for component i),
4. take a projected step a←max(a‑η·grad,0) and renormalise the sum to 1.

Random perturbations and an annealed acceptance rule help escape
stationary points (the uniform distribution is a trivial fixed point).
The routine runs until the allotted budget (default 1000 s) expires and
returns the best sequence found.
"""

import time
import math
import numpy as np


def _score_of(vec: np.ndarray) -> float:
    """
    Compute the score defined in evaluate_sequence for a 1‑D numpy array.
    The function assumes non‑negative entries.
    """
    n = vec.shape[0]
    s = np.sum(vec)
    if s < 1e-12:
        return float("inf")
    conv = np.convolve(vec, vec)
    return 2.0 * n * np.max(conv) / (s * s)


def _softmax_gradient(vec: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv(vec,vec)).
    For a given temperature γ the weights are
        w_k = exp(γ·conv_k) / Σ_j exp(γ·conv_j)

    The gradient w.r.t. component i is
        ∂/∂a_i Σ_k w_k·conv_k = 2·Σ_k w_k·a_{k‑i}
    which can be computed as a convolution of w with the reversed vector.
    """
    n = vec.shape[0]

    # self‑convolution
    conv = np.convolve(vec, vec)                     # length 2n‑1

    # soft‑max weights – numerically stable
    scaled = gamma * conv
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    w = exp_vals / np.sum(exp_vals)                  # shape (2n‑1,)

    # gradient = 2 * (w * reverse(vec))[n‑1 : n‑1+n]
    rev = vec[::-1]
    grad_full = np.convolve(w, rev, mode='full')    # length 3n‑2
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑scoring sequence using stochastic projected gradient
    descent with an annealing schedule.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (a few seconds are kept as a safety margin).

    Returns
    -------
    list[float]
        The best sequence (list of non‑negative floats) found within the
        time budget.
    """
    np.random.seed(seed)

    start_time = time.time()
    # keep a small safety margin to avoid being killed exactly at the limit
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose a length.  We target the length of the known construction
    # (≈5 000) because the theoretical lower bound improves with n.
    # If a previous construction `height_sequence_1` is available we keep
    # its length, otherwise we fall back to 5000.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 5000

    # ------------------------------------------------------------------
    # Initialise a non‑uniform positive vector (uniform is a fixed point of
    # the gradient and would never move).  Normalise to unit sum – the
    # score is scale‑invariant, so any normalisation works.
    # ------------------------------------------------------------------
    a = np.random.rand(n).astype(np.float64) + 0.1   # avoid exact zeros
    a /= np.sum(a)

    # If a previous construction exists, use a (normalised) interpolated
    # version as a second starting point with some probability.
    try:
        old = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if old.ndim == 1 and old.size > 0:
            # Interpolate to the current length to keep the overall shape.
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            else:
                a = np.random.rand(n) + 0.1
                a /= a.sum()
    except Exception:
        # No previous sequence – keep the random start.
        pass

    # Keep track of the best sequence seen so far.
    best_vec = a.copy()
    best_score = _score_of(best_vec)

    # ------------------------ optimisation hyper‑parameters -------------
    eta = 0.8               # initial step size
    eta_min = 5e-5          # minimal step size (prevents stagnation)
    gamma = 0.5             # start with a low temperature (almost average)
    gamma_max = 250.0       # final temperature – the soft‑max becomes a hard max
    iteration = 0
    no_improve_cnt = 0      # counts iterations without improvement
    log_interval = 10.0
    next_log = start_time + log_interval

    # ------------------------------------------------------------------
    # Main optimisation loop – runs until the time budget is exhausted.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # Current score (we already have a, but recompute for safety)
        cur_score = _score_of(a)

        # Gradient of the soft‑max approximation of the maximal convolution entry.
        grad = _softmax_gradient(a, gamma)

        # Projected gradient step.
        a_next = a - eta * grad
        a_next = np.maximum(a_next, 0.0)          # enforce non‑negativity
        s_next = np.sum(a_next)
        if s_next > 0.0:
            a_next /= s_next                       # renormalise to unit sum
        else:
            # Rare case where everything vanished – re‑initialise uniformly.
            a_next = np.full_like(a_next, 1.0 / n)

        # Evaluate the candidate.
        cand_score = _score_of(a_next)

        # Acceptance rule – accept improving moves; otherwise accept with a
        # tiny probability that decays with the iteration count (simulated
        # annealing flavour).  This helps to jump out of flat regions.
        if cand_score < cur_score:
            a = a_next
            no_improve_cnt = 0
            if cand_score < best_score:
                best_score = cand_score
                best_vec = a_next.copy()
        else:
            # Temperature for the occasional uphill step.
            T = max(0.05, 1.0 - iteration / 50000.0)
            if np.random.rand() < math.exp((cur_score - cand_score) / T):
                a = a_next
                no_improve_cnt = 0
            else:
                no_improve_cnt += 1

        # If we have been stuck for a while, inject a small random perturbation.
        if no_improve_cnt >= 3000:
            perturb = np.random.randn(n) * 0.02
            a = np.maximum(a + perturb, 0.0)
            a_sum = np.sum(a)
            if a_sum > 0:
                a /= a_sum
            no_improve_cnt = 0

        # ------------------------------------------------------------------
        # Annealing schedules for step size and temperature.
        # ------------------------------------------------------------------
        eta = max(eta * 0.9995, eta_min)          # slowly decay step size
        gamma = min(gamma * 1.001, gamma_max)    # slowly raise temperature

        # Periodic logging – useful for debugging and progress monitoring.
        now = time.time()
        if now >= next_log:
            print(
                f"Iter {iteration:6d} | Score {cur_score:.6f} | Best {best_score:.6f} "
                f"| η {eta:.2e} | γ {gamma:.2f}"
            )
            next_log = now + log_interval

    # ----------------------------------------------------------------------
    # Final report
    # ----------------------------------------------------------------------
    final_time = time.time() - start_time
    print(f"Search finished after {iteration} iterations (elapsed {final_time:.1f}s).")
    print(f"Best score obtained: {best_score:.6f}")

    # The evaluation system expects a plain Python list.
    return best_vec.tolist()
