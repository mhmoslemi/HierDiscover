def project_simplex(v, z):
    # v is a numpy array
    # return array x that is closest (L2) to v, with x_i >=0, sum x = z.
    # Use algorithm from Duchi et al.
    if z < 0:
        raise ValueError
    # if sum(v) == z and all >=0, just return v
    # else:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    # find rho = max i s.t. u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""...

import numpy as np
import time
import math

def project_simplex(v, z=1.0):
    """Project vector v onto simplex {x >= 0, sum(x)=z}."""
    # Implementation based on Duchi et al. (2008).
    n = v.shape[0]
    # Ensure v is float
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0)
    return w

def convolution(a):
    """Return linear convolution of a with itself."""
    return np.convolve(a, a)

def eval_val(a, conv=None):
    """Compute the evaluation metric for sequence a."""
    n = a.shape[0]
    if conv is None:
        conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

def gradient_step(a, sum_target, step=0.2):
    """Take a single gradient step minimizing a smooth approximation of max(conv)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    cmax = conv.max()
    # choose epsilon relative to magnitude to avoid overflow
    eps = max(cmax * 0.5, 1e-4)
    # stable soft-max weights
    w = np.exp((conv - cmax) / eps)  # shape 2n-1
    sum_w = w.sum()
    # reverse w for convolution
    w_rev = w[::-1]
    # convolution of a and reversed w
    full = np.convolve(a, w_rev)  # length 3n-2
    # slice of length n, reversed back
    grad = 2.0 * full[n-1:2*n-1][::-1] / sum_w
    # take step and project onto simplex (sum constraint)
    a_new = a - step * grad
    a_new = project_simplex(a_new, sum_target)
    return a_new

def random_transfer(a, sum_target, max_frac=0.5):
    """Randomly move mass from one index to another while preserving the sum."""
    n = a.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:
        return a.copy()
    # ensure donor has enough mass
    if a[i] < 1e-12:
        donor, recipient = j, i
    else:
        donor, recipient = i, j
    # choose delta up to a fraction of donor mass
    delta = np.random.rand() * max_frac * a[donor]
    a_new = a.copy()
    a_new[donor] -= delta
    a_new[recipient] += delta
    # projection not needed because sum unchanged and nonnegativity preserved
    return a_new

    sum_target = 1.0
    a = a * (sum_target / a.sum())

    # ----------------------------------------------------------------------
    it = 0

"""
Propose a sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv denotes the linear (self)‑convolution.
The algorithm mixes a smooth gradient descent on a log‑sum‑exp
approximation of the max‑convolution with occasional random mass‑transfer
moves and a simple simulated‑annealing acceptance rule.
All iterates are projected onto the probability simplex (fixed total sum,
non‑negative entries) so the objective value stays well‑defined.
The method works for any length; if a previous construction
`height_sequence_1` exists its length is reused (longer sequences give a
better theoretical lower bound).  The routine runs for the allotted time
budget and returns the best sequence found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: projection onto the simplex {x >= 0, sum(x) = z}
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Evaluation of the objective (fast, no extra security checks)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, conv: np.ndarray | None = None) -> float:
    """Return 2 * n * max(conv(a,a)) / (sum(a)**2)."""
    n = a.shape[0]
    if conv is None:
        conv = np.convolve(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# One gradient step on a smooth surrogate of max(conv)
# ----------------------------------------------------------------------
def _gradient_step(a: np.ndarray, total: float, step: float = 0.2) -> np.ndarray:
    """
    Perform a projected gradient step that reduces a smooth approximation
    of max(conv(a,a)).  The surrogate is the log‑sum‑exp of the convolution.
    """
    n = a.shape[0]

    # convolution of the current iterate
    conv = np.convolve(a, a)
    cmax = conv.max()

    # choose ε proportionally to the magnitude to keep exp stable
    eps = max(cmax * 0.5, 1e-6)

    # soft‑max weights (shifted by the maximum to avoid overflow)
    w = np.exp((conv - cmax) / eps)          # shape 2n‑1, all non‑negative
    sum_w = w.sum()

    # gradient of the log‑sum‑exp w.r.t. a:
    #   grad_i = 2 * Σ_k w_k * a_{k-i} / Σ_k w_k
    w_rev = w[::-1]                          # reversed for convolution
    full = np.convolve(a, w_rev)             # length 3n‑2
    # slice the part that corresponds to the needed sums, then reverse
    grad = 2.0 * full[n - 1 : 2 * n - 1][::-1] / sum_w

    # gradient descent step and projection back onto the simplex
    a_next = a - step * grad
    a_next = project_simplex(a_next, total)
    return a_next


# ----------------------------------------------------------------------
# Random move that keeps the total mass unchanged
# ----------------------------------------------------------------------
def _random_transfer(a: np.ndarray, total: float, max_frac: float = 0.5) -> np.ndarray:
    """
    Pick two positions i (donor) and j (receiver) and move a random
    fraction of the donor's mass to the receiver.  The total sum remains
    unchanged and non‑negativity is preserved.
    """
    n = a.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:
        return a.copy()

    # decide which index will donate mass (must have enough to give)
    if a[i] < 1e-12:
        donor, recv = j, i
    else:
        donor, recv = i, j

    delta = np.random.rand() * max_frac * a[donor]   # amount to move
    a_new = a.copy()
    a_new[donor] -= delta
    a_new[recv] += delta
    # projection is unnecessary: sum is preserved and both entries stay≥0
    return a_new


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation value.  The method combines smooth gradient descent,
    random mass‑transfer moves and a simple simulated‑annealing acceptance
    rule, always projecting onto the simplex to keep the total sum fixed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose a length.  Re‑use the length of a previously known sequence if
    # it exists; otherwise pick a moderate size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = np.random.randint(800, 2000)

    # initialise a uniform point on the simplex (total mass = 1.0)
    total_mass = 1.0
    a = np.full(n, total_mass / n)

    # initial objective
    conv = np.convolve(a, a)
    best_val = _objective(a, conv)
    best_seq = a.copy()
    print(f"[init] length = {n}, objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the stochastic search
    # ------------------------------------------------------------------
    it = 0
    step_size = 0.4               # gradient step size (will slowly shrink)
    temperature = 1.0             # for simulated‑annealing acceptance
    temp_decay = 0.9995
    no_improve = 0
    max_no_improve = 20000        # trigger a restart if stuck too long

    while time.time() < deadline:
        it += 1

        # 80 % gradient step, 20 % random transfer
        if np.random.rand() < 0.8:
            a_new = _gradient_step(a, total_mass, step=step_size)
        else:
            a_new = _random_transfer(a, total_mass, max_frac=0.3)

        # evaluate the candidate (fast, using numpy)
        conv_new = np.convolve(a_new, a_new)
        val_new = _objective(a_new, conv_new)

        # simulated‑annealing acceptance
        accept = False
        if val_new < best_val:
            accept = True
        else:
            # probability to accept a worse move
            prob = math.exp(-(val_new - best_val) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            conv = conv_new
            if val_new < best_val:
                best_val = val_new
                best_seq = a_new.copy()
                no_improve = 0
                print(f"[iter {it:06d}] new best = {best_val:.6f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        # decay scheduling
        step_size *= 0.9999
        temperature *= temp_decay

        # occasional restart if progress stalls
        if no_improve > max_no_improve:
            a = np.random.rand(n)
            a = a * (total_mass / a.sum())
            conv = np.convolve(a, a)
            no_improve = 0
            print(f"[iter {it:06d}] restart triggered")

    print(f"[done] iterations = {it}, best value = {best_val:.6f}")
    # return plain Python list as required by the evaluator
    return best_seq.tolist()
