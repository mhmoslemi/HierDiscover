import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Construct a linearly increasing step function.

    Theory
    -------
    For a sequence a_i = i+1 (i = 0 … n‑1) the convolution at the
    centre (k = n‑1) is

        (a * a)[n‑1] = n (n+1) (n+2) / 6 .

    The total sum is S = n (n+1) / 2.  After the mandatory scaling
    performed inside ``evaluate_sequence`` (which is homogeneous
    of degree 0) the evaluation value reduces to

        score = 2·n·max(b) / S² = 4·(n+2) / (3·(n+1)) → 4/3 ≈ 1.333

    for large n.  This is well below the required 1.503, and it also
    respects all security checks (non‑negative, ≤1000).

    Implementation
    ---------------
    * Use the length of the currently best construction (global
      ``height_sequence_1``) if available, otherwise fall back to a
      reasonable default (5000).
    * Scale the sequence so that the largest entry is exactly 1000,
      avoiding the automatic truncation performed by ``evaluate_sequence``.
    * Return the resulting list of floats.
    """

    # Determine the length of the sequence we will build.
    # If the global variable ``height_sequence_1`` exists we reuse its
    # length (the competition already stores a good‑size example there).
    # Otherwise we pick a convenient size that is large enough for a
