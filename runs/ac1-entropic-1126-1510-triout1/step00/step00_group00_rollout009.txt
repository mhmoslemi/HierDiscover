def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Choose length
    n = np.random.randint(3000, 8000)  # variable length
    # Initialize shape (exponential)
    r = 0.9995  # start param
    p = np.array([(1 - r) * (r ** i) for i in range(n)])
    p = p / p.sum()
    best_seq = p.copy()
    best_score = evaluate_sequence(p.tolist())
    # parameters
    t0 = 0.5
    t_min = 0.001
    decay = 0.999
    lambda_val = 5.0
    lambda_inc = 1.02  # per iteration
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(p, p)
        max_conv = conv.max()
        # compute smooth weights
        # shift for numeric stability
        shift = max_conv
        exp_term = np.exp(lambda_val * (conv - shift))
        w = exp_term / exp_term.sum()
        # gradient
        rev_p = p[::-1]
        grad_full = np.convolve(rev_p, w)
        # slice out gradient of length n
        grad = 2 * grad_full[n-1: n-1 + n]
        # step size schedule
        step = t0 * (best_score / (best_score + max_conv))  # may not needed
        # use decaying step
        step = max(t_min, t0 * (deadline - time.time()) / budget_s)  # decreasing linearly
        # update
        p_new = p - step * grad
        # projection onto simplex
        p_new = np.maximum(p_new, 0)
        sum_p = p_new.sum()
        if sum_p > 0:
            p_new = p_new / sum_p
        else:
            p_new = np.full_like(p_new, 1.0/n)
        # evaluate
        cur_score = evaluate_sequence(p_new.tolist())
        # keep best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p_new.copy()
        # accept new point (maybe hill climbing)
        p = p_new
        # update lambda
        lambda_val = min(100, lambda_val * lambda_inc)
    return best_seq.tolist()

import time

def _project_simplex(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        return v / s
    else:
        return np.full_like(v, 1.0 / len(v))

def _smooth_max_gradient(p, lam):
    # compute convolution
    conv = np.convolve(p, p)
    max_c = conv.max()
    # compute softmax weights
    # shifting by max for stability
    exp_terms = np.exp(lam * (conv - max_c))
    w = exp_terms / exp_terms.sum()
    # gradient: g_i = 2 * sum_j p_j * w_{i+j}
    rev_p = p[::-1]
    grad_full = np.convolve(rev_p, w)  # length 3n-2
    n = len(p)
    grad = 2.0 * grad_full[n-1:n-1+n]
    return grad, max_c

    deadline = time.time() + budget_s - 2.0   # a safety margin

The evaluation function
```python
2 * n * max(conv(a,a)) / (sum(a) ** 2)
```
is *scale invariant*: multiplying the whole vector by a constant does not change the score.
Therefore we can restrict the search to **probability vectors**
`p_i ≥ 0 , Σ p_i = 1`.  
For such a vector the score simplifies to  

```
score(p) = 2 * n * max_k (p * p)[k]          (1)
```

The problem is to make the auto‑convolution as flat as possible.
A direct optimisation of (1) is hard because the “max’’ makes the objective non‑smooth.
We use a **smooth‑max** approximation

```
Fλ(p) = (1/λ) * log Σ_k exp( λ * (p*p)[k] )
```

For large λ the value of `Fλ` approaches `max_k (p*p)[k]`.  
`Fλ` is differentiable and its gradient can be written in closed form:

```
g_i = 2 Σ_j p_j * w_{i+j}                (2)

    with   w_k = exp( λ * (p*p)[k] ) / Σ_t exp( λ * (p*p)[t] )
```

Thus we can apply a projected (sub)‑gradient descent:
* compute the convolution,
* build the soft‑max weights `w`,
* obtain the gradient by a **single convolution**,
* take a small step and project back onto the simplex (non‑negative, sum = 1).

The algorithm consists of three stages:

1. **Very cheap parametric sweep** – exponential, power‑law and simple polynomial shapes – to obtain a decent starting point.
2. **Smooth‑max gradient descent** – the core optimiser described above.
3. **Periodic random jitter** – helps to escape plateaus.

All work is performed inside the allowed 1000 s wall‑clock limit and the best
sequence found is returned.

import numpy as np
import time

# ------------------------------------------------------------
# Helper: projection onto the probability simplex
# ------------------------------------------------------------
def _project_simplex(v):
    """Clamp to non‑negative and renormalise so that sum(v)=1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        return v / s
    # extremely unlikely, but keep a uniform distribution as fall‑back
    return np.full_like(v, 1.0 / len(v))

# ------------------------------------------------------------
# Helper: gradient of the smooth‑max approximation
# ------------------------------------------------------------
def _smooth_max_gradient(p, lam):
    """
    Returns the gradient of   Fλ(p) = (1/λ) log Σ exp(λ (p*p))[k]
    and the current maximum convolution entry.
    """
    conv = np.convolve(p, p)                 # full convolution, length 2n‑1
    max_c = conv.max()
    # soft‑max weights, shifted by max_c for numerical stability
    exp_terms = np.exp(lam * (conv - max_c))
    w = exp_terms / exp_terms.sum()          # length 2n‑1, Σ w = 1

    # gradient (2) can be obtained by a single convolution
    rev_p = p[::-1]
    grad_full = np.convolve(rev_p, w)       # length 3n‑2
    n = len(p)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n
    return grad, max_c

        _try(a)

        a /= a.sum()
        _try(a)

        a /= a.sum()
        _try(a)

    # 3. Gradient‑descent refinement (smooth max)
    # --------------------------------------------------------
    p = best_seq.copy()
    cur_score = best_score

    lam = 5.0               # temperature for the smooth‑max
    lam_inc = 1.001         # slowly increase → true max
    step = 0.2              # initial learning rate
    step_decay = 0.9995     # geometric decay
    no_improve = 0
    improve_limit = 1200    # after this many non‑improving iterations we jitter

    it = 0
    while time.time() < deadline:
        it += 1

        # gradient of the smooth‑max objective
        grad, max_c = _smooth_max_gradient(p, lam)

        # simple back‑tracking line‑search (max 3 attempts)
        for _ in range(3):
            cand = _project_simplex(p - step * grad)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score <= cur_score or _ == 2:
                break
            step *= 0.5          # shrink step if it does not improve

        # accept candidate (even if slightly worse – we keep exploring)
        p = cand
        cur_score = cand_score

        # keep global best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.copy()
            no_improve = 0
            print(f"[iter {it}] new global best = {best_score:.6f}")
        else:
            no_improve += 1

        # decay learning‑rate and raise temperature
        step *= step_decay
        lam = min(200.0, lam * lam_inc)

        # occasional random jitter to escape flat regions
        if no_improve >= improve_limit:
            jitter = np.random.randn(n) * 0.01
            p = _project_simplex(p + jitter)
            no_improve = 0
            print(f"[iter {it}] random jitter applied")

    # --------------------------------------------------------
    # 4. Return the best sequence found
    # --------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()
