"""
propose_candidate
-----------------

Search for a non‑negative sequence of heights that lowers the evaluation metric

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

(When the sequence is normalized to sum(a)=sqrt(2 n) this score equals the
maximum value of the auto‑convolution `max(conv(a, a))`.)

The algorithm works in three stages:

1. **Initial shape** – a strongly decreasing geometric sequence.  This makes
   the peak of the convolution appear at a small shift (k≈0) and leaves a lot
   of slack for the “full‑overlap’’ shift (k=n‑1).

2. **Linear‑programming improvement** – given the current sequence `f`
   we solve a LP that maximises the total sum of a candidate `b` while
   forcing the mixed convolution `conv(f, b)` to stay below the current
   bound `M = max(conv(f, f))`.  Because `M` is not attained at the full‑overlap
   shift for our non‑uniform start, the LP can increase the raw sum of `b`.
   After scaling `b` back to the target sum `sqrt(2 n)` the self‑convolution
   of the scaled vector is strictly smaller than `M`.  Replacing the current
   vector by this scaled LP solution therefore reduces the score.

3. **Iterated refinement** – we repeat the LP step many times (mixing the
   old and the new vector when the direct replacement does not improve)
   and occasionally apply tiny random perturbations to escape stagnation.
   The whole search runs until the time budget is exhausted.

The whole routine is deterministic (seeded) and respects the 1000‑second
budget.  The best sequence found during the run is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _max_conv(seq: np.ndarray) -> float:
    """Return max value of the linear convolution of seq with itself."""
    return np.max(np.convolve(seq, seq))


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve:
        maximise   sum(b)
        subject to conv(f, b) <= M   (for all shifts)
                   b >= 0

    Returns the raw solution b (not yet rescaled) or None on failure.
    """
    n = f.shape[0]
    # Build the matrix A_ub where each row k corresponds to the inequality
    #   sum_j f_{k-j} * b_j <= M   (indices out of range are ignored)
    rows = []
    for k in range(2 * n - 1):
        # row_j = f_{k-j} if 0 <= k-j < n else 0
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        # vector of length n filled with zeros
        row = np.zeros(n, dtype=float)
        # fill the admissible part
        j_indices = np.arange(start, end + 1)
        i_indices = k - j_indices
        row[j_indices] = f[i_indices]
        rows.append(row)

    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], M, dtype=float)

    # bounds: b_j >= 0
    bounds = [(0, None) for _ in range(n)]

    # linear objective: minimise -sum(b)  <=> maximise sum(b)
    c = -np.ones(n, dtype=float)

    # Use the modern HiGHS solver – fast enough for the moderate sizes we use.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5.0})
    if res.success:
        # Clip tiny negatives that can appear due to numerical tolerance.
        b = np.where(res.x < 0, 0.0, res.x)
        return b
    return None


def propose_candidate(seed: int = 123, budget_s: int = 900, **kwargs) -> list[float]:
    """
    Return a list of non‑negative floats (the step heights) that tries to
    minimise the evaluation function.  The routine runs for roughly
    ``budget_s`` seconds and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # keep a little head‑room

    # -----------------  configuration  -----------------
    # we try a few different lengths; shorter lengths are cheaper to optimise.
    candidate_lengths = [150, 200, 250]  # feel free to adjust if time permits
    target_ratio = 1.5  # stop early if we beat the target bound

    best_overall_seq = None
    best_overall_score = float("inf")

    for n in candidate_lengths:
        # target sum = sqrt(2 * n)  ->  score = max_conv after normalisation
        sum_target = np.sqrt(2.0 * n)

        # ----- initialise with a decreasing geometric sequence -----
        r = 0.93  # decay factor (tuned empirically)
        a = np.array([r ** i for i in range(n)], dtype=float)
        a = a / a.sum() * sum_target

        # keep the best for this length
        best_seq = a.copy()
        best_score = _max_conv(best_seq)
        print(f"[init] n={n}, initial score={best_score:.6f}")

        # ----- main optimisation loop -----
        it = 0
        while time.time() < deadline:
            it += 1
            # current bound
            M = _max_conv(a)

            # try to improve using the LP step
            b_raw = _solve_lp(a, M)
            if b_raw is None:
                # fall back to a tiny random walk
                idx = np.random.randint(n)
                delta = np.random.randn() * 0.02 * sum_target / n
                a[idx] = max(0.0, a[idx] + delta)
                # renormalise to the fixed sum
                if a.sum() > 0:
                    a = a / a.sum() * sum_target
                else:
                    a = np.full(n, sum_target / n)
                continue

            # raw sum of the LP solution
            raw_sum = b_raw.sum()
            if raw_sum <= 1e-12:
                # something went wrong – re‑initialise slightly
                a = np.full(n, sum_target / n)
                continue

            # scale the LP solution back to the required sum.
            scale = sum_target / raw_sum
            g = b_raw * scale

            # compute the new bound
            M_g = _max_conv(g)

            # direct replacement if it improves the bound
            if M_g + 1e-12 < M:
                a = g
                M = M_g
            else:
                # try a convex combination (mix) – small set of candidate alphas
                mixed = False
                for alpha in (0.25, 0.5, 0.75):
                    cand = (1.0 - alpha) * a + alpha * g
                    M_cand = _max_conv(cand)
                    if M_cand + 1e-12 < M:
                        a = cand
                        M = M_cand
                        mixed = True
                        break
                if not mixed:
                    # tiny random perturbation to escape a possible plateau
                    i1, i2 = np.random.choice(n, size=2, replace=False)
                    eps = np.random.randn() * 0.01 * sum_target / n
                    a[i1] = max(0.0, a[i1] - eps)
                    a[i2] = max(0.0, a[i2] + eps)
                    # re‑normalise
                    if a.sum() > 0:
                        a = a / a.sum() * sum_target
                    else:
                        a = np.full(n, sum_target / n)
                    continue

            # keep the best ever seen for this length
            cur_score = M  # because sum is fixed, score == max convolution
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[improve] n={n}, it={it}, score={best_score:.6f}")

            # early exit if we have already beaten the target
            if best_score <= target_ratio:
                print(f"[early stop] n={n} reached target score {best_score:.6f}")
                break

        # compare with global best
        if best_score < best_overall_score:
            best_overall_score = best_score
            best_overall_seq = best_seq.copy()
            print(f"[global] new best n={n}, score={best_overall_score:.6f}")

    # Fallback: if nothing was found (should never happen) use a uniform vector.
    if best_overall_seq is None:
        n = 200
        best_overall_seq = np.full(n, np.sqrt(2 * n) / n)

    print(f"[final] best score achieved = {best_overall_score:.6f}")
    # convert to plain Python list as required by the external evaluator
    return best_overall_seq.tolist()
