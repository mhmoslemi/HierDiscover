def project_simplex(v, s):
    """Project vector v onto the probability simplex scaled to s.
    Returns w such that w >= 0, sum w = s, and w is closest to v in Euclidean norm."""
    # from https://github.com/justheuristics/ColabProjects...
    n = len(v)
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last True
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v, s):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho)==0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑score step‑function using a simple gradient‑descent on the
maximum of the auto‑convolution.

The evaluation function (after the mandatory sum rescaling) is
    score = max_k (a * a)[k] .
Hence we try to minimise the maximum entry of the linear convolution of a
with itself while keeping the vector non‑negative and with L1‑norm
√(2·n).  The algorithm is:

1. Choose a random feasible start (non‑negative, correct L1‑norm).
2. Repeatedly
   * compute the full convolution,
   * locate the index k where the convolution is maximal,
   * form the gradient of that entry:  ∇_i g_k = 2·a_{k‑i}   (zero outside the
     support of the sum),
   * subtract the mean of the gradient so that the update has zero total
     mass (the L1‑norm stays unchanged to first order),
   * take a small step opposite to that direction,
   * project back onto the simplex {x ≥ 0 , Σx = √(2·n)},
   * keep the move if the score improves, otherwise reject (with a tiny
     probability to escape plateaus).
3. Perform many independent random restarts within the allotted time and
   keep the best sequence seen.

The projection onto the simplex is the O(n log n) algorithm of
Duchi et al. (2008).  All operations are vectorised with NumPy; the inner
loop is fast enough to run many thousands of iterations inside the
1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the simplex {x >= 0, sum(x)=s}
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum = s."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]        # last True index (1‑based)
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Local score – identical to evaluate_sequence but cheaper
def _fast_score(a: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative vector a."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_conv = np.max(conv)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    given `evaluate_sequence` function.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5          # a small safety margin

    # --------------------------------------------------------------
    # Determine length n.  If the previous best construction is available
    # we keep the same length; otherwise we use a reasonable default.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)                 # 3130 in the current leaderboard
    else:
        n = 3000

    target_sum = np.sqrt(2.0 * n)                 # the required L1‑norm

    # --------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # parameters for the inner optimisation
    init_lr = target_sum / n                      # typical magnitude of an entry
    min_lr = 1e-6
    max_iter_per_restart = 5000                  # safe upper bound per restart
    no_improve_limit = 300                       # restart if stagnation

    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # ------- random initial feasible vector --------------------
        a = np.random.rand(n)
        a = a / np.sum(a) * target_sum

        cur_score = _fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] init restart {restart} score {best_score:.6f}")

        lr = init_lr
        it_without_improve = 0

        for it in range(max_iter_per_restart):
            if time.time() >= deadline:
                break

            # ---- compute convolution and its worst index ------------
            conv = np.convolve(a, a)
            k_max = int(np.argmax(conv))

            # ---- gradient of the active convolution entry ----------
            # grad_i = 2 * a_{k_max - i}   (zero outside support)
            i_low = max(0, k_max - (n - 1))
            i_high = min(k_max, n - 1)
            idx = np.arange(i_low, i_high + 1, dtype=int)
            grad = np.zeros_like(a)
            grad[idx] = 2.0 * a[k_max - idx]

            # make the step sum‑preserving (subtract the mean)
            mean_g = grad.mean()
            direction = grad - mean_g

            # ---- tentative update ---------------------------------
            a_new = a - lr * direction
            a_new = np.maximum(a_new, 0.0)          # keep non‑negative
            a_new = _simplex_projection(a_new, target_sum)   # exact L1‑norm

            new_score = _fast_score(a_new)

            # ---- accept / reject ----------------------------------
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                it_without_improve = 0
                # enlarge step a little after a success
                lr = min(lr * 1.05, init_lr * 10)
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] restart {restart} iter {it} NEW BEST {best_score:.6f}")
            else:
                # occasional uphill move to escape flat regions
                if np.random.rand() < 0.001:
                    a = a_new
                    cur_score = new_score
                else:
                    # shrink step size on failure
                    lr = max(lr * 0.5, min_lr)
                it_without_improve += 1

            # ---- occasional random perturbation --------------------
            if (it + 1) % 50 == 0:
                # tiny random shuffle of mass between two random positions
                i1, i2 = np.random.choice(n, 2, replace=False)
                delta = min(a[i1] * 0.02, a[i2] * 0.02, target_sum * 1e-3)
                a[i1] -= delta
                a[i2] += delta
                a = _simplex_projection(a, target_sum)
                cur_score = _fast_score(a)

            # ---- stagnation handling --------------------------------
            if it_without_improve >= no_improve_limit:
                break   # give up this restart early

        restart += 1

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to the uniform construction (should never happen)
        best_seq = np.full(n, target_sum / n)

    # convert to plain Python list for the external evaluator
    return best_seq.tolist()
