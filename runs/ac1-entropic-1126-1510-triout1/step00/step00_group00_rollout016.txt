def fast_score(seq):
    arr = np.asarray(seq, dtype=np.float64)
    arr = np.clip(arr, 0, 1000)
    n = len(arr)
    conv = np.convolve(arr, arr)

    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a*sum_a)

def improve_one_step(seq):
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    # compute contributions
    n = len(seq)
    contributions = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            contributions[i] = seq[i] * seq[j]
    i_max = np.argmax(contributions)
    # choose target index with lowest contribution
    zero_idxs = np.where(contributions == 0)[0]
    if len(zero_idxs) > 0:
        i_target = rng.choice(zero_idxs)
    else:
        # choose index with minimum contributions but not i_max
        i_target = np.argmin(contributions + (np.arange(n) == i_max) * np.max(contributions))
    # amount to transfer: fraction of seq[i_max]
    epsilon = seq[i_max] * rng.uniform(0.01, 0.1)
    # ensure we don't go negative
    epsilon = min(epsilon, seq[i_max])
    new_seq = seq.copy()
    new_seq[i_max] -= epsilon
    new_seq[i_target] += epsilon
    return new_seq
```

"""
Propose a candidate sequence of non‑negative heights that (hopefully) lowers the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a simple simulated‑annealing / hill‑climbing loop that
operates directly on the sequence.  At each iteration we
* compute the full convolution,
* locate the index k where the convolution attains its maximum,
* identify the pair (i, j = k‑i) that contributes most to this peak,
* move a small fraction of mass from the offending index `i` to an index
  that hardly influences the peak (preferably where the product a[i]*a[j] is
  zero),
* accept the move if it reduces the score or with a probability that
  depends on a slowly cooling temperature.

The move is deliberately “targeted’’ at the current bottleneck, which is
much more effective than blind random perturbations.  A small fraction of
steps are pure random transfers to explore the space, and occasional tiny
Gaussian noise helps to escape flat regions.

The routine works for any length (the length of the initial sequence is
chosen randomly between 500 and 2500 or taken from the supplied
`height_sequence_1`).  All numbers are kept non‑negative and the sum is
renormalised to 1 (the objective is scale‑invariant).  The function returns
the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation without the heavy input‑validation overhead.
def fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for the given sequence."""
    arr = np.clip(seq, 0.0, 1000.0)        # mimic evaluate_sequence clamping
    n = arr.shape[0]
    conv = np.convolve(arr, arr)          # full convolution
    max_b = np.max(conv)
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height sequence that minimises the provided
    evaluation function.  The routine runs for up to ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5  # a few seconds spare for cleanup

    # --------------------------------------------------------------
    # 1) Initialise a starting sequence.
    # --------------------------------------------------------------
    use_existing = (
        "height_sequence_1" in globals()
        and isinstance(globals()["height_sequence_1"], (list, np.ndarray))
        and rng.random() < 0.5
    )
    if use_existing:
        seq = np.asarray(height_sequence_1, dtype=np.float64).copy()
        if seq.ndim != 1 or seq.size == 0:
            # fallback to random if the provided object is malformed
            use_existing = False
    if not use_existing:
        # random length between 500 and 2500 (fast enough for many iterations)
        n = rng.integers(500, 2500)
        # generate a power‑law shape a[i] ∝ (i+1)^p with p∈[0.5, 2.0]
        p = rng.uniform(0.5, 2.0)
        idx = np.arange(n, dtype=np.float64)
        seq = (idx + 1) ** p
    # Normalise to sum = 1 (scale‑invariant, keeps numbers well‑behaved)
    seq = seq / np.sum(seq)

    # --------------------------------------------------------------
    # 2) Initial scoring.
    # --------------------------------------------------------------
    curr_score = fast_score(seq)
    best_seq = seq.copy()
    best_score = curr_score

    # --------------------------------------------------------------
    # 3) Simulated‑annealing loop.
    # --------------------------------------------------------------
    iteration = 0
    T = 0.1 * best_score if best_score > 0 else 0.01   # initial temperature
    n = seq.shape[0]

    while time.time() < deadline:
        iteration += 1

        # ------------------------------------------------------------------
        # Choose a move: 70 % targeted, 30 % pure random transfer.
        # ------------------------------------------------------------------
        if rng.random() < 0.7:
            # ----------- Targeted move based on the current maximal convolution ---
            conv = np.convolve(seq, seq)
            k = int(np.argmax(conv))                     # index of the peak

            # contributions[i] = seq[i] * seq[k-i] for all valid i
            i_arr = np.arange(n, dtype=np.int64)
            j_arr = k - i_arr
            mask = (j_arr >= 0) & (j_arr < n)
            contrib = np.zeros_like(seq)
            contrib[i_arr[mask]] = seq[i_arr[mask]] * seq[j_arr[mask]]

            i_max = int(np.argmax(contrib))               # index that hurts the peak most

            # Prefer a target where the contribution to this peak is zero.
            zero_idxs = np.flatnonzero(contrib == 0.0)
            if zero_idxs.size > 0:
                i_target = int(rng.choice(zero_idxs))
            else:
                # otherwise pick the smallest contribution (avoid i_max itself)
                tmp = contrib.copy()
                tmp[i_max] = np.max(contrib) + 1.0
                i_target = int(np.argmin(tmp))

            # Transfer a modest fraction of the mass from i_max to i_target.
            eps_frac = rng.uniform(0.05, 0.2)              # 5‑20 % of the source amount
            epsilon = min(seq[i_max] * eps_frac, seq[i_max])
            if epsilon <= 0.0:
                continue

            new_seq = seq.copy()
            new_seq[i_max] -= epsilon
            new_seq[i_target] += epsilon
        else:
            # ------------------- Pure random transfer --------------------
            i, j = rng.choice(n, size=2, replace=False)
            eps_frac = rng.uniform(0.05, 0.2)
            epsilon = min(seq[i] * eps_frac, seq[i])
            if epsilon <= 0.0:
                continue
            new_seq = seq.copy()
            new_seq[i] -= epsilon
            new_seq[j] += epsilon

        # ------------------------------------------------------------------
        # Keep feasibility: non‑negative, optional tiny Gaussian noise.
        # ------------------------------------------------------------------
        new_seq = np.clip(new_seq, 0.0, None)

        if rng.random() < 0.01:                           # occasional jitter
            noise = rng.normal(scale=0.001, size=n)
            new_seq = np.clip(new_seq + noise, 0.0, None)

        # Minor renormalisation to avoid drift (objective is scale‑invariant)
        new_seq /= np.sum(new_seq)

        # ------------------------------------------------------------------
        # Evaluate and decide on acceptance.
        # ------------------------------------------------------------------
        new_score = fast_score(new_seq)

        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - curr_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            seq = new_seq
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"Iter {iteration:6d} – new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Temperature schedule (slow exponential cooling, refreshed every 100 iters).
        # ------------------------------------------------------------------
        if iteration % 100 == 0:
            T = max(T * 0.99, 1e-6)

    # --------------------------------------------------------------
    # 4) Return the best sequence found (as a plain Python list).
    # --------------------------------------------------------------
    best_seq = best_seq.tolist()
    print(f"Search finished after {iteration} iterations. "
          f"Best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq
