        p = expb / expb.sum()
        # compute gradient of LSE (softmax weighted conv derivative)
        # gradient = sum_k p_k * d b_k / d a_i
        # where d b_k / d a_i = a_{k-i} + (i == k-i) * a_i
        # compute first term: grad1_i = sum_k p_k * a_{k-i} where valid
        # This is convolution of a and p (as earlier). We'll compute using dot slices.
        # vectorized using np.dot for each i but perhaps more efficient using FFT cross-correlation.
        # We'll compute using convolution of a and p (full) and then adjust.
        # compute conv = np.convolve(a, p)
        # conv[t] = sum_j a_j * p[t-j]
        # we need for i: sum_j a_j * p[i+j] = conv[i + n - 1]? Let's test.
        # Check: conv[t] includes p[t-j]; set t = i + (some). Suppose we set t = i + (number?) such that t - j = i + j => t = i + 2j (depends).
        # So not simple.

        #   start = i
        #   grad1[i] = np.dot(a, p[i:i+n])  # because p length 2n-1.

        grad1 = np.array([np.dot(a, p[i:i+n]) for i in range(n)], dtype=np.float64)

        # Actually LSE objective f = (1/tau) log sum exp(tau * b). Its gradient w.r.t a = sum_k p_k * d b_k/da_i = grad (as above).

def project_simplex(v):
    """Project onto the probability simplex (nonnegative, sum=1)."""
    # Standard algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑value step‑function using projected gradient descent on a
smooth approximation of the L∞ norm of the self‑convolution.
The algorithm works on the probability simplex (non‑negative entries that
sum to 1) so that the evaluation function is scale‑invariant.
A soft‑max (temperature τ) is used to turn the max over the convolution
into a differentiable objective; the gradient is obtained analytically
via a convolution, then the iterate is projected back onto the simplex.
Multiple random restarts and a small grid of hyper‑parameters are tried
within the given time budget.  The best sequence found is returned.
"""

import time
import numpy as np

# ------------------------------------------------------------
# helper: projection onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x ≥ 0, sum(x)=1} (the probability simplex)."""
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        return np.ones_like(v) / len(v)
    u = np.sort(v)[::-1]                     # sorted (decreasing)
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / len(v)
    else:
        rho = ind[cond][-1]                  # last index where condition holds
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
