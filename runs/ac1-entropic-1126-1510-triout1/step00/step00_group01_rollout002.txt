def arcsine_seq(n):
    i = np.arange(n)
    # Avoid zero division: use i+1 and n-i
    return 1.0 / (np.sqrt(i + 1) * np.sqrt(n - i))

def fast_conv(seq):
    n = len(seq)
    m = 1 << ((2*n - 1).bit_length())
    fft = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[:2*n-1]
    return conv

def fast_score(seq):
    n = len(seq)
    conv = fast_conv(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float('inf'), None, None
    score = 2.0 * n * max_b / (sum_a * sum_a)
    return score, conv, max_b

def arcsine_sequence(n: int) -> np.ndarray:
    """
    Return a length‑n array whose entries follow the discrete arcsine law:
        a[i] = 1 / (sqrt(i+1) * sqrt(n-i))
    This distribution has a near‑uniform self‑convolution.
    """
    i = np.arange(n, dtype=np.float64)
    # avoid division by zero – the formula is well‑defined for i = 0 … n‑1
    a = 1.0 / (np.sqrt(i + 1.0) * np.sqrt(n - i))
    return a


def conv_self(seq: np.ndarray) -> np.ndarray:
    """
    Compute the full convolution of `seq` with itself using FFT.
    Returns an array of length 2*len(seq)-1.
    """
    n = seq.shape[0]
    # length for convolution – next power of two >= 2*n-1
    m = 1 << ((2 * n - 1).bit_length())
    # real FFT is a bit faster
    f = np.fft.rfft(seq, n=m)
    conv = np.fft.irfft(f * f, n=m)[: 2 * n - 1]
    return conv

    Return (score, convolution_array, max_value) for a given sequence.
