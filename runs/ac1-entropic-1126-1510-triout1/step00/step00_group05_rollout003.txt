def simplex_projection(v):
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    import time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1

    best_seq = None
    best_score = float('inf')

    def init_random():

        return simplex_projection(x)

    def evaluate(a):
        return evaluate_sequence(a.tolist()) # external function

    # compute L_conv for conv and L_corr for correlation
    L_conv = 1 << ((2*n -1).bit_length())
    L_corr = 1 << ((3*n -2).bit_length())

    iteration = 0
    while time.time() < deadline:
        iteration+=1
        # random starting point each restart
        a = init_random()
        # randomize alpha and step schedule maybe
        alpha = 5.0  # initial smoothing factor
        for i in range(5000):
            if time.time() >= deadline:
                break
            # compute convolution by FFT
            fa = np.fft.rfft(a, L_conv)
            conv = np.fft.irfft(fa * fa, L_conv)[:2*n-1]
            max_conv = conv.max()
            # evaluation
            score = 2 * n * max_conv
            # track best
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Iter {iteration}-{i}: new best {best_score:.6f}")
                # early exit if near target
                if best_score <= 1.503 + 1e-4:
                    return best_seq.tolist()

            # stable softmax:
            exp_vals = np.exp(alpha * (conv - max_conv))
            weights = exp_vals / exp_vals.sum()

            fw = np.fft.rfft(weights, L_corr)
            # correlation (a ⋆ weights) => irfft(fa * conj(fw))
            corr = np.fft.irfft(fa * np.conj(fw), L_corr)

            # step size schedule

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The algorithm uses a projected (or multiplicative) gradient descent on a
smooth approximation of  max(conv(a,a)).
The smooth approximation is the log‑sum‑exp (soft‑max) of the
auto‑convolution values; its gradient can be written as a correlation
between the current sequence and the soft‑max weights.
Both the convolution and the correlation are evaluated with FFTs,
so each iteration is O(N log N) and very fast.

Multiple random restarts are performed until the time budget is exhausted.
The best sequence found (in terms of `evaluate_sequence`) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex (sum = 1, a_i >= 0)
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Core optimisation step: one gradient descent iteration
def gd_step(a: np.ndarray, alpha: float, eta: float, L: int):
    """
    Perform one iteration of gradient descent on the soft‑max of the
    auto‑convolution.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1, non‑negative).
    alpha : float
        Soft‑max temperature (larger => closer to true max).
    eta : float
        Step size.
    L : int
        Length of the FFT (power of two) used for both convolution
        and correlation (must be >= 3*len(a)‑2).

    Returns
    -------
    a_new : np.ndarray
        Updated probability vector (still on the simplex).
    max_conv : float
        Current maximum convolution value (before the step).
    """
    n = a.shape[0]

    # ---- convolution a*a (linear, not circular) via FFT ----------
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

    # maximum entry of the convolution
    max_conv = np.max(conv_full)

    # ---- soft‑max weights for the convolution entries ---------------
    # using max‑subtraction for numerical stability
    exp_vals = np.exp(alpha * (conv_full - max_conv))
    weights = exp_vals / np.sum(exp_vals)                # shape (2n‑1,)

    # ---- gradient = 2 * (a ⋆ weights) (cross‑correlation) -------
    fw = np.fft.rfft(weights, n=L)
    corr_full = np.fft.irfft(fa * np.conj(fw), n=L)      # length L
    grad = 2.0 * corr_full[:n]                           # first n entries

    # ---- update (additive) and projection -------------------------
    a_new = a - eta * grad
    a_new = simplex_projection(a_new)

    return a_new, max_conv

    """

    deadline = time.time() + budget_s - 5.0      # small safety margin

    # --------------------------------------------------------------

    N = 800

    L = 1 << ((3 * N - 2).bit_length())   # >= 3N‑2

    def eval_score(a_vec: np.ndarray) -> float:
        # `evaluate_sequence` works with a Python list and enforces its own
        # clipping/normalisation.  To avoid the overhead of converting each time
        # we keep the sum exactly 1, so the score simplifies to 2*N*max(conv).
        # Still we call the given function to be safe.

    best_seq = None
