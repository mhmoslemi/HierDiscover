Algorithm (Duchi et al.):

```
def simplex_projection(v, z=1):
    # projecting onto {x >= 0, sum x = z}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v, s):
    # similar as earlier
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works as follows.

1.  The length n is taken from the provided `height_sequence_1` if present,
    otherwise a default of 2000 is used.
2.  The L‑1 norm of the sequence is fixed to  √(2 n)  (the same normalisation
    used in the reference code).  All updates therefore preserve this sum by
    projecting onto the probability simplex scaled by that constant.
3.  Starting from a random point on the simplex we minimise a smooth
    approximation of the 𝓁∞‑norm of the convolution:  ∑ b_k^p  with p=6.
    The gradient of this objective can be written in closed form
    (using the fact that ∂(a*a)_k / ∂a_i = 2 a_{k‑i} when the index is
    in range).  Both the convolution and the gradient are computed with a
    real FFT, which makes each iteration O(n log n) and fast enough for
    thousands of iterations.
4.  After each gradient step the vector is projected back onto the
    non‑negative simplex (preserving the required L‑1 norm).  A simple
    back‑tracking line‑search guarantees that the smooth objective never
    increases.
5.  The true objective (the value returned by ``evaluate_sequence``) is
    monitored; the best vector found during the allotted time is remembered.
6.  To avoid being trapped in a local minimum the algorithm occasionally
    injects a small random jitter and also restarts from a fresh random
    point when no improvement has been seen for a while.

The whole optimisation runs for at most ``budget_s`` seconds (default
1000 s).  The returned list is the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full convolution (length len(x)+len(y)-1) using FFT."""
    n = len(x) + len(y) - 1
    size = _next_pow2(n)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy)[:n]
    return conv


def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project vector v onto the simplex {x >= 0, sum(x) = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # Work on a copy to avoid side effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero after projection
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_lp_norm(a: np.ndarray, b: np.ndarray, p: int) -> np.ndarray:
    """
    Gradient of   F(a) = sum_k b_k**p   where b = a * a (full convolution).
    Returns ∇F (same shape as a).
    """
    # b has length 2*n-1, we need weights w_k = p * b_k**(p-1)
    w = p * np.power(b, p - 1)
    # compute grad_i = 2 * sum_k w_k * a_{k-i}
    # which is 2 * (a cross‑correlated with w)
    w_rev = w[::-1]
    conv = _full_convolution(a, w_rev)  # length 3*n-2
    offset = len(w) - 1               # = 2*n-2
    # indices offset, offset-1, ..., offset-(n-1)
    grad = 2.0 * conv[offset - np.arange(len(a))]
    return grad

        n = len(height_sequence_1)  # type: ignore[name-defined]

    p = 6                     # exponent for smooth Lp‑approximation of max

    restart_interval = 3000   # re‑initialise after this many iterations
    no_improve_limit = 5000   # give up and restart if no improvement

    best_seq = None

    deadline = start_time + budget_s - 5.0  # small safety margin

        return evaluate_sequence(seq.tolist())

    # ------------------------------------------------------------------
