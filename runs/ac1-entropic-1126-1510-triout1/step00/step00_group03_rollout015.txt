def simplex_projection(v):
    v = np.array(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v):
    # v is numpy array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if not np.any(cond):
        # all negative, project to uniform
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def convolution_max_and_indices(a):
    n = len(a)
    size = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    c = np.fft.irfft(A * A, n=size)
    c = c[:2*n-1]
    max_val = np.max(c)
    indices = np.where(np.abs(c - max_val) < 1e-12)[0]  # tolerance
    return max_val, indices

    n=len(a)
    size = 1 << ((2*n-1).bit_length())
    Fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Fa * Fa, n=size)
    conv = conv[:2*n-1]
    max_val = np.max(conv)
    indices = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return max_val, indices

"""
Fast gradient based search for a low‑value step‑function.
The goal is to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

which, after normalising the sequence to sum(a)=1, is simply
    score = 2 * n * max(conv(a, a))

The algorithm:
1. Randomly pick a length n and an initial shape (geometric,
   power‑law or a Dirichlet draw).  The vector is normalised to sum=1.
2. Repeatedly reduce the maximum of the self‑convolution by a
   projected sub‑gradient step.
   * The convolution maximum and its locations are obtained with an FFT.
   * The sub‑gradient for the max‑convolution is 2·a shifted by each
     active index.
   * After a step we project back onto the probability simplex
     (non‑negative entries that sum to 1).
   * Only steps that strictly decrease the score are kept; otherwise a
     small random perturbation is tried.
3. Keep the best sequence seen across many restarts until the time
   budget (≈ 1000 s) is exhausted.
The returned list is ready for the external `evaluate_sequence` call.
"""

import numpy as np
import time


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all entries are zero or negative
        theta = cssv[-1] / n
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max_fft(a: np.ndarray):
    """Return (max value, array of indices achieving the max) of a*a."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = np.max(conv)
    # indices of the maximum (tolerance to avoid floating noise)
    idx = np.where(np.abs(conv - max_val) < 1e-12)[0]
    return float(max_val), idx.astype(int)


def _gradient_for_max(a: np.ndarray, max_idx):
    """Sub‑gradient of the max‑convolution w.r.t. a."""
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]          # derivative of Σ a[i] a[j] w.r.t. a[i]
    return grad


def _improve_sequence(a0: np.ndarray, n, deadline):
    """Run a projected sub‑gradient descent, returning the best (a, score)."""
    a = a0.copy()
    best_a = a.copy()
    max_c, max_idx = _conv_max_fft(a)
    best_score = 2.0 * n * max_c

    it = 0
    while time.time() < deadline and it < 3000:
        it += 1
        max_c, max_idx = _conv_max_fft(a)
        cur_score = 2.0 * n * max_c

        # compute sub‑gradient for the active indices
        grad = _gradient_for_max(a, max_idx)

        # simple backtracking line‑search
        eta0 = 0.2 / np.sqrt(it)          # decreasing step size
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            eta = eta0 * factor
            a_candidate = _project_to_simplex(a - eta * grad)
            max_c2, _ = _conv_max_fft(a_candidate)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break

        # if no improvement, try a tiny random perturbation
        if not improved:
            a_pert = a.copy()
            idx = np.random.randint(n, size=5)
            a_pert[idx] += np.random.randn(5) * 0.001
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / np.sum(a_pert)
            max_c2, _ = _conv_max_fft(a_pert)
            cand_score = 2.0 * n * max_c2
            if cand_score < cur_score:
                a = a_pert
                cur_score = cand_score

        # keep the best ever seen for this run
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 200 == 0:
                print(f"    iter {it}, new local best {best_score:.6f}")

        # abort if we have already reached a very good bound
        if best_score < 1.51:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Entry point used by the evaluator.

    Returns a list of non‑negative heights giving a low evaluation score.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a little margin

    best_score = float("inf")
    best_seq = None

    while time.time() < deadline:
        # ---- choose problem size and a random starting shape ----
        n = np.random.randint(1500, 3000)           # length of the step function
        shape = np.random.choice(["geom", "power", "mixed", "random"])

        if shape == "geom":
            # geometric decay: a_i = r^i
            t = np.random.uniform(0.0003, 0.0015)   # t = -log(r)
            r = np.exp(-t)
            a = r ** np.arange(n, dtype=float)

        elif shape == "power":
            # power‑law decay: a_i = (i+1)^{-p}
            p = np.random.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** -p)

        elif shape == "mixed":
            # product of geometric and power‑law
            t = np.random.uniform(0.0003, 0.001)
            r = np.exp(-t)
            p = np.random.uniform(0.5, 1.5)
            a = (r ** np.arange(n, dtype=float)) * (np.arange(1, n + 1, dtype=float) ** -p)

        else:  # "random"
            # Dirichlet(1) – random positive vector, then normalised
            a = np.random.gamma(1.0, 1.0, size=n)

        # random permutation (does not change the multiset of values)
        if np.random.rand() < 0.5:
            np.random.shuffle(a)

        # normalise to sum = 1 (scale does not affect the evaluation)
        a = a / np.sum(a)

        # ---- local optimisation (projected sub‑gradient) ----
        sub_deadline = min(deadline, time.time() + 15.0)   # at most 15 s per start
        a_opt, score_opt = _improve_sequence(a, n, sub_deadline)

        # keep the globally best sequence seen so far
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            print(
                f"Global improvement: n={n}, shape={shape}, score={best_score:.6f}"
            )

        # early exit if we have already hit the target
        if best_score < 1.51:
            break

    # fallback – uniform sequence (never should happen)
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n)

    # return a plain Python list (the evaluator will clip values automatically)
    return best_seq.tolist()
