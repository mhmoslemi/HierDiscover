def flatten_sequence(seq, steps=5000, lr=0.01):
   n = len(seq)
   S = np.sum(seq)   # keep sum constant
   for _ in range(steps):
       conv = np.convolve(seq, seq)
       max_val = np.max(conv)
       max_idx = np.argmax(conv)
       # compute gradient = 2 * seq[max_idx - i] for i where index in range
       grad = np.zeros(n)
       for i in range(n):
           j = max_idx - i
           if 0 <= j < n:
               grad[i] = 2 * seq[j]
       # update
       seq = seq - lr * grad
       # ensure nonnegative
       seq = np.maximum(seq, 0)
       # renormalize sum to original S
       sum_seq = np.sum(seq)
       if sum_seq > 0:
           seq = seq * (S / sum_seq)
       else:
           # restart
           seq = np.random.rand(n)
           seq *= S / np.sum(seq)
   return seq

"""
Propose candidate step‑function heights.

Algorithm
---------
1.  Normalize the target L¹‑norm to  sqrt(2·n)  (this does not change the
    evaluation value but keeps the numbers in a convenient range).
2.  Initialise a pool of starting sequences:
       * the supplied `height_sequence_1` (renormalised),
       * a flat (uniform) sequence,
       * a random Dirichlet‑distributed sequence.
   Keep the best of these as the current elite.
3.  Repeatedly try two different improvement strategies until the
    time‑budget expires.
   * **Gradient flattening** –  we minimise the L∞‑norm of the
     auto‑convolution by a sub‑gradient step.
       * Find the index `k*` where `conv = f * f` attains its maximum.
       * The sub‑gradient of `conv[k*]` w.r.t. `f_i` is `2·f_{k*-i}`
         (if the index is inside the array, otherwise 0).
       * Take a small step opposite to the gradient, project to
         non‑negative values and renormalise the vector to the target sum.
   * **LP move** –  once in a while call the existing LP routine
     `get_good_direction_to_move_into` (the same tool used in the
     reference solution).  This gives a direction that usually reduces
     the convolution bound while preserving the L¹‑norm.
4.  After every modification evaluate the sequence with the provided
    `evaluate_sequence`.  If the new value is smaller (remember we are
    minimising) store it as the current best.
5.  Return the best sequence found before the deadline.

The core of the improvement is the simple sub‑gradient descent,
which actively reduces the contribution of the most constraining
convolution entry.  Random restarts and occasional LP moves help the
search escape shallow plateaus.  All operations are vectorised and the
convolution is performed with an FFT (O(n log n)) so that thousands of
iterations easily fit into the 1000 s budget.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: fast convolution using FFT (real input)
def _conv_fft(x: np.ndarray) -> np.ndarray:
    n = x.size
    size = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << (size - 1).bit_length()
    fx = np.fft.rfft(x, fft_len)
    conv = np.fft.irfft(fx * fx, fft_len)[:size]
    return conv


# ----------------------------------------------------------------------
# LP helper (taken from the reference implementation)
def solve_convolution_lp(f_sequence, rhs):
    """Solve  max Σ b_j   s.t.   (f * b)_k ≤ rhs  and  b ≥ 0."""
    n = len(f_sequence)
    c = -np.ones(n)                     # maximise Σ b_j
    a_ub = []
    b_ub = []

    # constraints for each possible sum index
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity b ≥ 0  →  -b ≤ 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    result = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 8.0, "presolve": True},
    )
    if result.success:
        return result.x
    return None


def get_good_direction_to_move_into(sequence):
    """Return a convex combination (1‑t)·seq + t·g that respects the
    convolution bound, where g comes from the LP."""
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return None

    # renormalise to satisfy the LP’s  sqrt(2 n)‑norm convention
    normalized = sequence * np.sqrt(2 * n) / sum_seq
    rhs = np.max(_conv_fft(normalized))

    g = solve_convolution_lp(normalized, rhs)
    if g is None:
        return None

    # scale g back to the original L¹‑norm
    g = g * np.sqrt(2 * n) / np.sum(g)

    t = 0.01                     # a tiny convex step
    new_seq = (1 - t) * sequence + t * g
    # protect against accidental tiny negatives caused by numerical noise
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq.tolist()


# ----------------------------------------------------------------------
def _gradient_flatten(seq: np.ndarray, target_sum: float,
                     steps: int, lr: float) -> np.ndarray:
    """One run of sub‑gradient descent that tries to flatten f*f."""
    n = seq.size
    for step in range(steps):
        conv = _conv_fft(seq)
        k_star = int(np.argmax(conv))

        # build the sub‑gradient of conv[k_star] w.r.t. seq
        idx = np.arange(n)
        j = k_star - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(seq)
        grad[mask] = 2.0 * seq[j[mask]]

        # decreasing learning rate (simple linear schedule)
        eta = lr * (1.0 - step / steps)

        seq = seq - eta * grad
        seq = np.maximum(seq, 0.0)

        # renormalise to keep Σ seq_i = target_sum
        cur_sum = np.sum(seq)
        if cur_sum > 0:
            seq *= target_sum / cur_sum
        else:
            # degenerated case – re‑initialise randomly
            seq = np.random.rand(n)
            seq *= target_sum / np.sum(seq)

    return seq
