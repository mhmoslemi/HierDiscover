The algorithm works in three stages:

1. **Parametric sweep** –  a one‑parameter family 
   a_i(p) ∝ (i+1)^{-p} (i = 0 … n‑1) is examined.
   The family contains the uniform vector (p = 0) and many
   decreasing shapes (p > 0).  For each p the evaluation
   score is computed and the best exponent is kept.
   A finer sweep around the best exponent refines the result.

   most to the current peak (k_max) is identified.  A small amount
   of mass is transferred from a high‑contributing index to an

   accepted if it improves the score (or with a tiny

All moves respect the non‑negativity constraint; the

                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity taken care of by explicit bounds below
    a_ub = np.vstack(a_ub)
    b_ub = np.array(b_ub)
    bounds = [(0, None) for _ in range(n)]
    # small time limit to avoid hangs
    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub,
                          bounds=bounds,
                          method='highs',
                          options={'time_limit': 5.0, 'disp': False})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def parametric_sequence(p, n):
    """
    Return a probability vector of length ``n`` of the form
    a_i ∝ (i+1)^{-p}  (i = 0 … n‑1)  and normalised to sum = 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)   # i+1
    seq = idx ** (-p)
    total = seq.sum()
    if total == 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / total
    return seq.tolist()


# ----------------------------------------------------------------------
def flatten_step(curr_seq, conv, k_max):
    """
    Perform a small mass transfer that tends to reduce the current peak
    at index ``k_max``.
    Returns a new sequence (list) and the proposed delta for debugging.
    """
    n = len(curr_seq)
    # contribution of each position to the peak
    contrib = np.zeros(n, dtype=np.float64)
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            contrib[i] = curr_seq[i] * curr_seq[j]

    # indices with largest and (ideally) zero contribution
    if np.any(contrib > 0):
        i_high = int(np.argmax(contrib))
    else:
        i_high = np.random.randint(n)

    zero_idxs = np.where(contrib == 0)[0]
    if zero_idxs.size > 0:
        i_low = int(np.random.choice(zero_idxs))
    else:
        i_low = int(np.argmin(contrib))

    # move a fraction of the mass from i_high to i_low
    delta = min(curr_seq[i_high] * 0.08, 0.5)   # at most 8 % of the source
    if delta <= 0:
        return None, None

    new_seq = np.array(curr_seq, copy=True)
    new_seq[i_high] -= delta
    new_seq[i_low]  += delta
    # ensure we stay non‑negative (tiny numerical safety)
    new_seq[i_high] = max(new_seq[i_high], 0.0)
    return new_seq.tolist(), delta

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function height vector that minimises
    evaluate_sequence.  The function respects the 1000 s time budget
    and returns the best vector seen.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a little safety margin

    # --------------------------------------------------------------
    # Choose a length.  If ``height_sequence_1`` exists we reuse its size,
    # otherwise we fall back to a moderate length.
    n = globals().get("height_sequence_1", None)
    if isinstance(n, (list, tuple)):
        n = len(n)
    if not isinstance(n, int) or n <= 0:
        n = 2000                     # fallback, still reasonably fast

    # --------------------------------------------------------------
    # 1) Coarse grid search over exponent p
    best_score = float('inf')
    best_seq = None
    best_p = 0.0
    p_grid = np.linspace(0.0, 2.0, 21)    # 0,0.1,…,2.0  → 21 points

    print("=== coarse grid search ===")
    for p in p_grid:
        seq = parametric_sequence(p, n)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_p = p
        # very cheap progress print
        if (p - p_grid[0]) % 5 == 0:
            print(f"p={p:.2f}, score={score:.6f}")

    # --------------------------------------------------------------
    # 2) Fine grid around the best exponent found
    fine_low = max(0.0, best_p - 0.5)
    fine_high = best_p + 0.5
    fine_grid = np.linspace(fine_low, fine_high, 31)   # 31 points

    print("\n=== fine grid search ===")
    for p in fine_grid:
        seq = parametric_sequence(p, n)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_p = p
        if (p - fine_low) % 0.2 < 1e-9:
            print(f"p={p:.3f}, score={score:.6f}")

    print(f"\nBest exponent after grids: p={best_p:.4f}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Local refinement – flattening of the convolution peak
    curr_seq = best_seq.copy()
    curr_score = best_score
    temperature = 0.01 * best_score   # initial temperature for occasional uphill moves

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # recompute convolution and the location of the maximal entry
        conv = np.convolve(curr_seq, curr_seq)
        k_max = int(np.argmax(conv))

        # attempt a flattening step
        new_seq, delta = flatten_step(curr_seq, conv, k_max)
        if new_seq is not None:
            new_score = evaluate_sequence(new_seq)
            # classic simulated‑annealing acceptance rule
            accept = False
            if new_score < curr_score:
                accept = True
            else:
                prob = np.exp((curr_score - new_score) / max(temperature, 1e-12))
                if rng.random() < prob:
                    accept = True
            if accept:
                curr_seq = new_seq
                curr_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] New best via flatten: score={best_score:.6f}")

        # occasional random perturbation to diversify search
        if rng.random() < 0.001:
            idx = rng.integers(n)
            pert = max(0.0, curr_seq[idx] + rng.normal(scale=0.1))
            temp_seq = curr_seq.copy()
            temp_seq[idx] = pert
            temp_score = evaluate_sequence(temp_seq)
            if temp_score < curr_score:
                curr_seq = temp_seq
                curr_score = temp_score
                if temp_score < best_score:
                    best_score = temp_score
                    best_seq = temp_seq.copy()
                    elapsed = time.time() - start_time
                    print(f"[{elapsed:6.1f}s] New best via random tweak: score={best_score:.6f}")

        # an occasional LP step (like the SOTA approach) can increase the sum
        # while keeping the current bound – we invoke it very sparsely.
        if iteration % 5000 == 0:
            # normalise current vector to “unit” L1 to stay comparable with the LP routine
            total = sum(curr_seq)
            if total > 0:
                norm_curr = [x * np.sqrt(2 * n) / total for x in curr_seq]
                rhs = max(np.convolve(norm_curr, norm_curr))
                g = solve_convolution_lp(norm_curr, rhs)
                if g is not None:
                    # scale g to original L1
                    sum_g = sum(g)
                    if sum_g > 0:
                        g_scaled = [x * np.sqrt(2 * n) / sum_g for x in g]
                        t = 0.02   # slightly larger step than the baseline code
                        mixed = [(1 - t) * a + t * b for a, b in zip(curr_seq, g_scaled)]
                        mixed_score = evaluate_sequence(mixed)
                        if mixed_score < curr_score:
                            curr_seq = mixed
                            curr_score = mixed_score
                            if mixed_score < best_score:
                                best_score = mixed_score
                                best_seq = mixed.copy()
                                elapsed = time.time() - start_time
                                print(f"[{elapsed:6.1f}s] New best via LP mix: score={best_score:.6f}")

        # geometric cooling of temperature
        temperature *= 0.99995

    # --------------------------------------------------------------
    print(f"\nSearch finished. Best score = {best_score:.6f}")
    # Ensure the returned type matches the required list[float]
    return [float(v) for v in best_seq]
