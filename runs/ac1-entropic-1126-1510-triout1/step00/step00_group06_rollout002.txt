"""
Propose a step‑function (list of non‑negative heights) with a low
evaluate_sequence() value.

Algorithm
---------
* The evaluation is  2·n·max(conv(seq,seq)) / (sum(seq)²).  Because the score
  is invariant under scaling of the whole sequence we fix the total sum to
  `sqrt(2·n)` (the same normalisation used by the reference method).
* We search for a shape that “flattens’’ the self‑convolution.
  The search is performed by projected gradient descent on the peak of the
  convolution combined with occasional random perturbations (a simple
  simulated‑annealing style scheme).
* For the chosen peak index `k*` the sub‑gradient of `max(conv)` w.r.t.
  the coefficients is
      ∂/∂a_i  (a*a)_{k*} = 2·a_{k*‑i}   (if the partner index exists, else 0)
  and we move opposite to this gradient while keeping the coefficients
  non‑negative and re‑normalising the sum.
* Several independent restarts (different lengths, different initial
  shapes – either the known SOTA sequence or an “arcsine’’ distribution)
  are tried until the time budget expires.  The best sequence seen
  overall is returned.

The method is cheap (only O(n log n) FFT convolutions per iteration) and
fits comfortably into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper utilities ------------------------------------------------------
# ----------------------------------------------------------------------


def _score_from_seq(seq: np.ndarray) -> float:
    """Return the exact objective value for a given sequence."""
    n = seq.size
    if seq.sum() < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    maxb = conv.max()
    return 2.0 * n * maxb / (seq.sum() ** 2)


def _arcsine_shape(n: int) -> np.ndarray:
    """Return a positive “arcsine’’ shape: a_i ∝ 1/√((i+½)(n‑i‑½))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    # add a tiny amount of noise to avoid exact symmetry which can stall GD
    a += 0.01 * a.mean() * np.random.rand(n)
    return a


def _project_and_renorm(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative and scale to the required total sum."""
    x = np.clip(x, 0.0, None)
    s = x.sum()
    # the sum never becomes zero because we start from a feasible point
    if s == 0.0:
        # fall back to a uniform distribution
        x = np.full_like(x, target_sum / x.size)
        return x
    return x * (target_sum / s)


def _gradient_step(seq: np.ndarray, peak_idx: int) -> np.ndarray:
    """
    Compute one projected gradient step that tries to reduce the peak
    (sub‑gradient of the peak convolution value).
    """
    n = seq.size
    # indices that actually participate in the peak convolution
    lo = max(0, peak_idx - (n - 1))
    hi = min(n - 1, peak_idx)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    partner = peak_idx - idx                     # partner index = k*‑i
    grad = np.zeros_like(seq)
    grad[idx] = 2.0 * seq[partner]               # sub‑gradient 2·a_{k*‑i}
    return grad


def _run_one_restart(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    best_global_score: float,
    best_global_seq: np.ndarray,
) -> tuple[float, np.ndarray]:
    """
    Perform a short projected‑gradient descent run on a sequence of length `n`.
    Returns the best score/sequence found in this run.
    """
    # Normalise the initial sequence to the standard sum √(2·n)
    target_sum = np.sqrt(2.0 * n)
    seq = _project_and_renorm(init_seq.astype(np.float64), target_sum)

    # current score (the objective is invariant to a global scale, we keep the
    # sum fixed, therefore we can recompute only the max‑convolution)
    conv = np.convolve(seq, seq)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / (target_sum ** 2)

    best_score = cur_score
    best_seq = seq.copy()

    # hyper‑parameters (tuned heuristically)
    step_factor = 0.15          # base step size (relative to the mean value)
    patience = 0                # counts iterations without improvement

    max_iters = 2000            # hard limit per restart (will break earlier if out of time)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # locate the convolution peak
        conv = np.convolve(seq, seq)
        peak_idx = int(conv.argmax())
        peak_val = float(conv.max())

        # compute sub‑gradient and take a step opposite to it
        grad = _gradient_step(seq, peak_idx)

        # adaptive step length – we use a fraction of the mean coefficient
        mean_a = seq.mean()
        alpha = step_factor * mean_a / (np.linalg.norm(grad) + 1e-12)

        new_seq = seq - alpha * grad
        new_seq = _project_and_renorm(new_seq, target_sum)

        # evaluate new objective
        new_conv = np.convolve(new_seq, new_seq)
        new_score = 2.0 * n * new_conv.max() / (target_sum ** 2)

        # keep the globally best sequence seen so far
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(
                f"[restart n={n}] iter={it:04d}  new best = {best_score:.6f}"
            )

        # acceptance rule (simple greedy + occasional random jitter)
        if new_score < cur_score:
            seq = new_seq
            cur_score = new_score
            patience = 0
            # slightly increase step size after a successful move
            step_factor = min(step_factor * 1.02, 0.5)
        else:
            patience += 1
            # shrink step size when we are not improving
            step_factor = max(step_factor * 0.98, 0.02)

            # after a while without improvement inject a random perturbation
            if patience >= 80:

                idx = np.random.randint(n)
                bump = (np.random.rand() - 0.5) * 0.2 * mean_a
                seq[idx] = max(0.0, seq[idx] + bump)
                seq = _project_and_renorm(seq, target_sum)
                # recompute current score after the perturbation
                conv = np.convolve(seq, seq)
                cur_score = 2.0 * n * conv.max() / (target_sum ** 2)
                patience = 0

    # return the best discovered in this restart
    return best_score, best_seq
