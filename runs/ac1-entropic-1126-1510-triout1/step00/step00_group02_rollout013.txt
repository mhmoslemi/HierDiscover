```python
import numpy as np
import time

def generate_beta_sequence(n, alpha):
    # generate symmetric beta-dist shape
    # x in (0,1)
    # use midpoints of intervals to avoid exact 0 or1.
    idx = np.arange(n, dtype=np.float64)
    x = (idx + 0.5) / n
    # ensure values >0 and <1
    # compute raw using log to avoid overflow
    log_raw = (alpha - 1) * (np.log(x) + np.log(1.0 - x))
    raw = np.exp(log_raw)
    # Normalize to sum 1
    total = raw.sum()
    seq = raw / total
    return seq.tolist()
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave 5 seconds buffer

    best_seq = None
    best_score = np.inf

    # candidate lengths
    lengths = [2000, 5000, 8000]  # adjust based on time
    alphas = np.linspace(0.1, 0.9, 9)  # for beta distribution
    geos = np.linspace(0.5, 3.0, 6)    # for geometric parameter alpha in exponent scale

    # Evaluate beta families
    for n in lengths:
        for a in alphas:
            seq = generate_beta_sequence(n, a)
            try:
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    print(f"New best (beta): n={n}, alpha={a:.3f}, score={score:.6f}")
            except Exception as e:
                print(f"Error evaluating beta n={n} a={a}: {e}")

    # Evaluate geometric families
    for n in lengths:
        for g in geos:
            seq = generate_geometric_sequence(n, g)
            try:
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    print(f"New best (geo): n={n}, gamma={g:.3f}, score={score:.6f}")
            except Exception as e:
                print(f"Error evaluating geo n={n} g={g}: {e}")

    # Combine both families? face.

    # Additional local search from best_seq if found
    if best_seq is not None:
        seq = np.array(best_seq, dtype=np.float64)
        # local random hill climbing
        while time.time() < deadline:
            # random pair of indices
            i, j = np.random.randint(len(seq), size=2)
            # propose small transfer epsilon
            eps = np.random.randn() * 0.01
            # ensure we keep sum unchanged
            # We'll modify: seq[i] = max(0, seq[i] + eps), adjust seq[j] to keep sum constant
            new_i = max(0.0, seq[i] + eps)
            delta = new_i - seq[i]
            new_j = max(0.0, seq[j] - delta)   # subtract delta to keep sum constant
            # compute new sum
            new_seq = seq.copy()
            new_seq[i] = new_i
            new_seq[j] = new_j
            # renormalize to avoid drift from rounding
            total = new_seq.sum()
            if total < 1e-12:
                continue
            new_seq = new_seq / total  # keep sum=1
            try:
                score = evaluate_sequence(new_seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = new_seq.tolist()
                    seq = new_seq  # keep as current
                    print(f"Improved via hill climb: score={score:.6f}")
            except Exception:
                pass

    # If we never found a sequence (unlikely), fallback to constant
    if best_seq is None:
        best_seq = [1.0] * 100
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback constant sequence with score {best_score}")

    print(f"Final best score: {best_score}")
    return best_seq

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = np.inf

    lengths = [2000, 5000, 8000, 10000]
    alphas = np.linspace(0.2, 0.9, 8)
    p_vals = np.linspace(0.2, 1.0, 9)
    decay_vals = np.linspace(0.5, 5.0, 10)

    for n in lengths:
        # constant baseline
        seq = [1.0/n] * n
        # evaluate
        try:
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
        except Exception:
            pass

        for a in alphas:
            seq = generate_beta_sequence(n, a)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"beta n={n} alpha={a:.3f} score={score:.6f}")

        for p in p_vals:
            seq = generate_powerlaw_sequence(n, p)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"pow n={n} p={p:.3f} score={score:.6f}")

        for d in decay_vals:
            seq = generate_geometric_sequence(n, d)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"geo n={n} decay={d:.3f} score={score:.6f}")

        seq = generate_inverse_sqrt_sequence(n)
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score

"""
propose_candidate
=================

Search for a step‐function (non‑negative list of heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the discrete self‑convolution.  The algorithm combines
three ideas:

1. **Parametric seeds** – simple families (geometric, power‑law,
   random) are sampled at several lengths and parameter values.
2. **Projected gradient descent** – starting from the best seed the
   maximal convolution entry is identified, its sub‑gradient
   (2*a_{k‑i}) is used to move the vector downhill, the result is
   projected onto the probability simplex (non‑negative, unit sum) and
   re‑evaluated.  The step size is gradually reduced.
3. **Hill climbing & random perturbations** – when the gradient step
   does not improve the score a small random jitter is occasionally
   applied to escape plateaus.

All candidates are evaluated with the provided ``evaluate_sequence``
function; the best sequence found before the time budget expires is
returned.  The implementation uses only NumPy, SciPy (for optional
LPs) and standard library modules, respects the 1000 s budget and
conforms to the required function signature.
"""

import time
import numpy as np
from scipy import optimize

# --------------------------------------------------------------
# Helper sequence generators
# --------------------------------------------------------------

def generate_geometric_sequence(n: int, gamma: float) -> np.ndarray:
    """Monotonically decreasing exponential shape.
    a_i ∝ exp(-gamma * i / (n-1)).
    """
    if n == 1:
        return np.ones(1)
    i = np.arange(n, dtype=np.float64)
    raw = np.exp(-gamma * i / (n - 1))
    raw_sum = raw.sum()
    return raw / raw_sum


def generate_powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """Power‑law decreasing shape a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    raw = i ** (-p)
    raw_sum = raw.sum()
    return raw / raw_sum


def generate_random_sequence(n: int) -> np.ndarray:
    """Uniform random positive vector, normalised to sum = 1."""
    raw = np.random.rand(n)
    raw_sum = raw.sum()
    if raw_sum == 0:
        raw = np.ones(n)
        raw_sum = n
    return raw / raw_sum


# --------------------------------------------------------------
# Gradient‑descent refinement
# --------------------------------------------------------------

def refine_by_gradient(seq: np.ndarray, deadline: float, best_score: float,
                       best_seq: list) -> tuple[float, list]:
    """
    Perform projected gradient descent on ``seq`` (which already sums to 1).
    Returns the best score and the best sequence encountered during the
    refinement.
    """
    arr = seq.copy()
    n = arr.size

    while time.time() < deadline:
        # convolution and index of the current worst (largest) entry
        conv = np.convolve(arr, arr)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # sub‑gradient of the maximal convolution entry:
        # d/d a_i  (conv_kmax)  = 2 * a_{k_max-i}  (when the index is valid)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(arr)
        grad[mask] = 2.0 * arr[partner[mask]]

        # step size schedule – starts at ~0.1 and shrinks to ~0.001
        remaining = deadline - time.time()
        progress = (deadline - remaining) / deadline
        eta = 0.1 * (1.0 - progress) + 0.001

        new_arr = arr - eta * grad
        new_arr = np.maximum(new_arr, 0.0)
        s = new_arr.sum()
        if s == 0.0:
            break
        new_arr /= s  # projection onto the simplex (sum = 1)

        # evaluate the new candidate
        try:
            cur_score = evaluate_sequence(new_arr.tolist())
        except Exception:
            break

        # keep the improvement
        if cur_score < best_score:
            best_score = cur_score
            best_seq = new_arr.tolist()
            arr = new_arr
            print(f"  grad‑improve: n={n} score={cur_score:.6f}")
            continue

        # occasional random jitter to escape stagnation
        if np.random.rand() < 0.02:
            jitter = 0.01 * np.random.randn(n)
            new_arr = arr + jitter
            new_arr = np.maximum(new_arr, 0.0)
            new_arr /= new_arr.sum()
            arr = new_arr

    return best_score, best_seq


# --------------------------------------------------------------
# Main entry point required by the problem
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) achieves a
    low value of ``evaluate_sequence`` before the time budget runs out.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # keep a tiny safety margin

    best_score = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # 1.  Try a few parametric families as seeds
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 8000, 10000]          # various problem sizes
    geom_gammas = [0.5, 1.0, 1.2, 1.5, 2.0]      # exponential decay controls
    pow_exps = [0.4, 0.6, 0.8, 1.0, 1.5]        # power‑law exponents

    for n in lengths:
        # constant (reference)
        const = np.full(n, 1.0 / n, dtype=np.float64)
        try:
            sc = evaluate_sequence(const.tolist())
            if sc < best_score:
                best_score, best_seq = sc, const.tolist()
                print(f"init constant n={n} score={sc:.6f}")
        except Exception:
            pass

        # geometric families
        for gamma in geom_gammas:
            seq = generate_geometric_sequence(n, gamma)
            try:
                sc = evaluate_sequence(seq.tolist())
                if sc < best_score:
                    best_score, best_seq = sc, seq.tolist()
                    print(f"init geom n={n} γ={gamma:.2f} score={sc:.6f}")
            except Exception:
                pass

        # power‑law families
        for p in pow_exps:
            seq = generate_powerlaw_sequence(n, p)
            try:
                sc = evaluate_sequence(seq.tolist())
                if sc < best_score:
                    best_score, best_seq = sc, seq.tolist()
                    print(f"init pow  n={n} p={p:.2f} score={sc:.6f}")
            except Exception:
                pass

        # a few random seeds (useful when the deterministic families miss a
        # good region of the search space)
        for _ in range(3):
            seq = generate_random_sequence(n)
            try:
                sc = evaluate_sequence(seq.tolist())
                if sc < best_score:
                    best_score, best_seq = sc, seq.tolist()
                    print(f"init rnd  n={n} score={sc:.6f}")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # 2.  Refine the current best by projected gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None:
        arr = np.array(best_seq, dtype=np.float64)
        best_score, best_seq = refine_by_gradient(arr, deadline,
                                                  best_score, best_seq)

    # ------------------------------------------------------------------
    # 3.  If nothing succeeded (unlikely), fall back to a simple constant
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 5000
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print("fallback constant used")

    print(f"final best score: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
