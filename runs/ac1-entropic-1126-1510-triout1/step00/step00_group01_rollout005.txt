def subgradient_max(a, conv, max_val, eps=1e-12):
    n = len(a)
    active = np.where(conv >= max_val - eps)[0]
    g = np.zeros(n)

        end = min(k, n-1)

        segment = a[start: end+1]
        rev = segment[::-1]

"""
Propose a candidate step‑function height sequence that improves the
upper‑bound 2.0 produced by the uniform sequence.

Algorithm
---------

* Initialise a feasible sequence `a` (non‑negative, sum = √(2·n)).
  By default we start from the globally available `height_sequence_1`
  if it exists, otherwise from the uniform sequence of length *n*.

* The objective after normalisation is simply the maximum entry of
  the self‑convolution (`M = max(a * a)`).  Hence we try to reduce `M`.

* The main optimisation loop runs until the time budget expires.
  Three complementary moves are used:

  1. **LP‑move** – solve the linear programme
        maximise   Σ b_j
        subject to  (a * b)_k ≤ M  (∀k)          (linear in b)
                     b_j ≥ 0 .
     The solution `b` has the same L¹‑norm as `a`.  By mixing
     `a` and `b` (convex combination) we strictly decrease the
     maximal convolution value.  A small line‑search over several
     step‑sizes `t ∈ {0.05,…,0.8}` is performed and the best mixture
     is kept.

  2. **Gradient‑descent step** – a sub‑gradient of the `max`‐operator
     is computed from the active convolution indices
     (those attaining the current maximum).  A short step in the
     negative direction together with a projection onto the simplex
     (non‑negative entries and fixed L¹‑norm) yields a cheap improvement.

  3. **Random perturbation** – occasional random Gaussian noise on a
     few coordinates helps to escape shallow local minima.

* After every move the current sequence is projected onto the feasible
  set (clipping to `[0,1000]` and renormalising the sum) and its score
  (`evaluate_sequence`) is recorded.  The best sequence found during the
  whole run is returned.

The implementation uses FFT‑based convolution (`O(n log n)`) and the
high‑performance “highs” interior‑point solver from SciPy for the LP.
All hyper‑parameters (step sizes, frequencies of the three moves, etc.)
are tuned to give many cheap iterations while still performing the
expensive LP moves often enough to drive the bound below 2.0.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper – projection onto the feasible simplex (non‑negative, fixed sum)
def _project_to_target(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to ≥0, renormalise to `target_sum`, and enforce the 0‑1000 bound."""
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12:                      # fallback to uniform if degenerate
        a = np.full_like(a, target_sum / a.size)
    else:
        a = a * (target_sum / total)
    a = np.minimum(a, 1000.0)
    return a


# ----------------------------------------------------------------------
# Fast convolution via FFT (real inputs)
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.size
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return b


# ----------------------------------------------------------------------
# Linear programme that maximises sum(b) under (a * b) ≤ M and b ≥ 0.
def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """
    Solve
        max Σ b_j      s.t.   (f * b)[k] ≤ rhs   ∀k,   b ≥ 0.
    Returns the optimal `b` as a NumPy array, or None if infeasible.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  → minimise -1·b
    c = -np.ones(n)

    # build the convolution constraints
    rows = []
    rhs_vec = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # (f * b)[k] = Σ_j f_j * b_{k-j}
        # fill the appropriate entries of `row`
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # indices j = k - i where i runs over valid range
        # equivalently, for each j in [i_min, i_max] we have f_{k-j}
        # easier: loop over i and set row[j] = f_i where j = k - i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)
        rhs_vec.append(rhs)

    # non‑negativity constraints: -I · b ≤ 0
    rows.append(-np.eye(n))
    rhs_vec.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.hstack(rhs_vec)

    # solve with the HiGHS interior‑point method (fast and reliable)
    options = {"time_limit": time_limit, "presolve": True}
    result = optimize.linprog(
        c, A_ub=A_ub, b_ub=b_ub, method="highs", options=options
    )
    if result.success:
        return result.x
    return None
