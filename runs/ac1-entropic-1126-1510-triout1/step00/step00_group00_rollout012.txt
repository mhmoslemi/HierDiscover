        eps = curr_seq[i_max] * step_frac
        # we don't want to move everything, cap epsilon
        eps = min(eps, curr_seq[i_max] * 0.9)

        # apply move
        new_seq = curr_seq.copy()
        new_seq[i_max] -= eps
        new_seq[i_min] += eps

        # enforce nonnegativity (numerical safety)
        new_seq = np.clip(new_seq, 0, None)

        if s_new == 0:

        new_seq = new_seq / s_new

        elapsed = time.time() - start

        # acceptance

        if new_score < curr_sc:
            accept = True
        else:

            if np.random.rand() < prob:

            if new_score < best_sc:

    print(f'Finished. Best score (official) = {final_score:.6f}')

"""
Search for a step‑function (non‑negative sequence) that minimises
    score = 2·n·max(conv(a,a)) / (sum(a))²

The algorithm is a simple stochastic sub‑gradient descent with simulated
annealing:

* Start from a smooth exponential shape (close to the analytically‑derived
  family a_i ∝ r^i) which already gives a score ≈ 1.8 for the current
  length (≈6500).
* Repeatedly locate the index k where the self‑convolution attains its
  maximum.  The gradient of this maximal entry w.r.t. a is simply
  grad_i = a_{k‑i} (when the term exists).  We move a small amount of
  mass from the index with the largest contribution (grad_i·a_i) to the
  index with the smallest contribution, thereby reducing the peak.
* The move is accepted if it improves the score; otherwise it may be
  accepted with a temperature‑dependent probability (simulated annealing).
* Periodic random perturbations and occasional restarts provide exploration.
* Fast convolution is performed with an FFT (O(n log n)) and everything
  runs well within the 1000 s budget.

The returned list is a plain Python list of floats ready for the
`evaluate_sequence` checker.
"""

import math
import time
import numpy as np

# ------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ------------------------------------------------------------

def _fft_convolution_self(x: np.ndarray) -> np.ndarray:
    """
    Return the linear self‑convolution of a real vector x using FFT.
    """
    n = x.size
    size = 2 * n - 1
    # next power of two for efficient FFT
    nfft = 1 << (size - 1).bit_length()
    X = np.fft.rfft(x, n=nfft)
    conv_full = np.fft.irfft(X * X, n=nfft)[:size]
    return conv_full


def _max_conv(x: np.ndarray) -> float:
    """Maximum value of the self‑convolution of x."""
    return _fft_convolution_self(x).max()


def _score(x: np.ndarray) -> float:
    """Score defined in the challenge (uses the exact formula)."""
    s = x.sum()
    if s < 0.01:                     # protect against division by zero
        return float("inf")
    return 2.0 * x.size * _max_conv(x) / (s * s)


# ------------------------------------------------------------
# Main candidate generator
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (the function will stop a few seconds early).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5          # safety margin

    # --------------------------------------------------------
    # Problem size – use the length of the known construction
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 6500                                 # fallback

    # --------------------------------------------------------
    # Initial sequence – exponential shape (close to optimal)
    # --------------------------------------------------------
    # Parameter (c) roughly 1.2 gives a good starting point.
    c0 = 1.2
    r = math.exp(-c0 / n)
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()                                 # normalise to sum = 1

    # Small random jitter to avoid exact symmetry
    a += np.random.randn(n) * 1e-4
    a = np.clip(a, 0, None)
    a /= a.sum()

    best_seq = a.copy()
    best_score = _score(best_seq)

    current = a.copy()
    current_score = best_score

    # --------------------------------------------------------
    # Simulated‑annealing parameters
    # --------------------------------------------------------
    T_start = 0.2
    T_end = 1e-4
    it = 0

    # Main optimisation loop
    while time.time() < deadline:
        # ---------- compute convolution and the maximal index ----------
        conv_full = _fft_convolution_self(current)
        max_val = conv_full.max()
        k_max = np.flatnonzero(conv_full == max_val)      # could be several

        # ---------- sub‑gradient of the maximal convolution entry ----------
        grad = np.zeros_like(current)
        for k in k_max:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad[i] += current[k - i]                     # ∂/∂a_i conv(k)

        # The contribution of each coordinate to the peak is grad_i * a_i
        contrib = grad * current
        i_src = int(np.argmax(contrib))                    # move mass away from here
        i_dst = int(np.argmin(contrib))                    # move mass to here
        if i_src == i_dst:                                 # safety
            i_dst = (i_dst + 1) % n

        # ---------- propose a small transfer ----------
        # Fraction of the source mass (10%–60%) – randomised for exploration
        frac = np.random.rand() * 0.5 + 0.1                 # 0.1 … 0.6
        eps = current[i_src] * frac
        eps = min(eps, current[i_src] * 0.9)               # never empty the source

        new_seq = current.copy()
        new_seq[i_src] -= eps
        new_seq[i_dst] += eps
        new_seq = np.clip(new_seq, 0, None)                # guard against negatives
        new_seq /= new_seq.sum()                           # renormalise (sum stays 1)

        # ---------- evaluate the new candidate ----------
        new_score = _score(new_seq)

        # ---------- temperature for simulated annealing ----------
        elapsed = time.time() - start_time
        progress = (elapsed) / (deadline - start_time)
        T = T_start * (1.0 - progress) + T_end * progress

        # ---------- acceptance rule ----------
        accept = False
        if new_score < current_score:
            accept = True
        else:
            prob = math.exp(-(new_score - current_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            current = new_seq
            current_score = new_score
            # keep the best ever found
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[{it:6d}] New best score: {best_score:.6f} "
                      f"(time {elapsed:.1f}s)")

        # ---------- occasional diversification ----------
        if it % 5000 == 0 and it > 0:
            # perturb the current best a little and continue from there
            pert = np.random.randn(n) * 0.005
            tmp = best_seq + pert
            tmp = np.clip(tmp, 0, None)
            tmp /= tmp.sum()
            current = tmp
            current_score = _score(current)

        it += 1

    # Final verification with the official evaluator (avoids tiny FFT rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Search finished. Official score = {final_score:.6f}")
    return best_seq.tolist()
