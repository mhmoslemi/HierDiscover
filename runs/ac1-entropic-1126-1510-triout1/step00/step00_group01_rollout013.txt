def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    # function to normalize seq to sum=1
    def normalize(v):
        s = np.sum(v)
        if s == 0:
            v = np.random.rand(len(v))
            s = np.sum(v)
        return v / s

    def evaluate(v):
        return evaluate_sequence(v.tolist())
    
    # try a set of lengths
    lengths = [3,4,5,8,16,32,64,128,256,512,1024]
    # but ensure not too large for time
    # We'll start with moderate lengths: up to 512

    for n in lengths:
        if time.time() > deadline:
            break
        # initial random solution
        a = np.random.rand(n)
        a = normalize(a)
        # maybe also try linear decreasing shape
        a_lin = np.arange(n, 0, -1, dtype=float)
        a_lin = normalize(a_lin)
        # add to candidate pool
        candidates = [a, a_lin]

        for init in candidates:
            cur = init.copy()
            cur_score = evaluate(cur)
            if cur_score < best_score:
                best_seq = cur.copy()
                best_score = cur_score
            # annealing schedule
            T0 = 0.5
            T_end = 1e-4
            steps = 300
            for step in range(steps):
                if time.time() > deadline:
                    break
                # linear cooling
                T = T0 * ((T_end/T0) ** (step/steps))
                # propose move: randomly select i, j
                i, j = random.sample(range(n), 2)
                # move fraction of i to j
                delta = cur[i] * (random.random() * 0.2)  # up to 20%
                new = cur.copy()
                new[i] -= delta
                new[j] += delta
                # ensure nonnegative - already fine
                new = normalize(new)
                new_score = evaluate(new)
                if new_score < cur_score or random.random() < np.exp(-(new_score - cur_score)/T):
                    cur = new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_seq = cur.copy()
                        best_score = cur_score
                    # Optionally reduce temperature based on improvement
                # occasional perturbation large
                if step % 50 == 0:
                    # jitter
                    jitter = np.random.randn(n) * 0.005
                    cur = np.maximum(0, cur + jitter)
                    cur = normalize(cur)
                    cur_score = evaluate(cur)
                    if cur_score < best_score:
                        best_seq = cur.copy()
                        best_score = cur_score

    # after search, return best_seq as list
    if best_seq is None:
        # fallback to existing height_sequence_1 if available
        if 'height_sequence_1' in globals():
            return height_sequence_1
        else:
            return [1.0]  # trivial
    else:
        return best_seq.tolist()

def solve_convolution_lp(f, M):
    n = len(f)
    c = -np.ones(n)  # maximize sum(g) -> minimize -sum(g)
    # Build constraint matrix A_ub: for each k (0..2n-2), sum_i f_i * g_{k-i} <= M
    # where g_{k-i}=0 if index out of bounds.
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        # i from max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # For each i, j = k - i
        # row[j] += f[i]
        # So row entries for j in [k-i_end, k-i_start] etc
        for i in range(i_start, i_end+1):
            j = k - i
            row[j] = f[i]   # set coefficient
        rows.append(row)
    A_ub = np.vstack(rows + [-np.eye(n)])
    b_ub = np.hstack([np.full(2*n-1, M), np.zeros(n)])
    # bounds g_i >= 0
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success:
       return result.x
    else: return None
```

"""
Propose a better coefficient sequence for the given evaluation function.

The algorithm is a lightweight iterative scheme that repeatedly
* normalises a candidate sequence to have ℓ¹‑norm 1 (the ratio is
  homogeneous, any scaling is irrelevant),
* solves a linear programme that finds another non‑negative sequence `g`
  with **as large a total mass as possible** while keeping the convolution
  `f * g` bounded by the current worst convolution value `M = max(f*f)`,
* mixes the current sequence `f` with the LP‑solution `g` using a short
  line‑search over mixing parameters `t∈[0,1]`,
* keeps the best sequence found.

The LP is `max sum(g)  s.t.  (f * g)[k] ≤ M  ∀k, g ≥ 0`.
Because the mixing coefficient `t` can be chosen larger than the tiny value
used in the reference implementation, the reduction of the worst
convolution value `M` is typically faster.

Multiple restarts with random initialisations and a few cheap random
perturbations are performed until the time budget (≈ 1000 s) is exhausted.
The best sequence encountered is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _solve_convolution_lp(f_norm: np.ndarray, M: float):
    """
    Linear programme:
        maximise   sum(g)
        subject to  (f_norm * g)[k] <= M  for all k
                    g_i >= 0
    Returns the optimal g (numpy array) or None on failure.
    """
    n = len(f_norm)
    # objective: maximise sum(g) -> minimise -sum(g)
    c = -np.ones(n)

    # Build convolution constraints: for each k=0..2n-2
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices where 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # fill coefficients: (f[i]) contributes to g[k-i]
        #   i ∈ [i_start, i_end]  →  j = k-i
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f_norm[i]
        rows.append(row)

    # non‑negativity constraints g_i >= 0 (converted to -g_i <= 0)
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, M), np.zeros(n)])

    # Solve with HiGHS (fast and robust)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs")
    if res.success:
        return res.x
    return None


def _mix_and_search(f: np.ndarray, g: np.ndarray, n: int, deadline: float):
    """
    Perform a tiny grid search over mixing coefficient t and return
    the best mixed sequence (normalised) together with its score.
    """
    best_seq = f
    best_score = evaluate_sequence(f.tolist())
    # grid of t values – more points give a finer search but cost a bit more
    for t in np.linspace(0.0, 1.0, 21)[1:]:        # skip t=0 (no change)
        if time.time() >= deadline:
            break
        mixed = (1.0 - t) * f + t * g
        # re‑normalise – scaling does not affect the score
        mixed /= mixed.sum()
        score = evaluate_sequence(mixed.tolist())
        if score < best_score:
            best_score = score
            best_seq = mixed.copy()
    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a coefficient sequence that minimises `evaluate_sequence`.
    The function respects the time budget (default 1000 s) and returns the
    best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    best_overall_seq = None
    best_overall_score = float("inf")

    # ------------------------------------------------------------------
    # Helper to normalise to ℓ¹‑norm 1 (the score is scale‑invariant)
    def _normalise(v):
        s = v.sum()
        if s <= 0.0:
            v = np.abs(v) + 1e-12
            s = v.sum()
        return v / s

    # ------------------------------------------------------------------
    attempt = 0
    while time.time() < deadline:
        attempt += 1
        # ------------------------------------------------------------------
        # pick a problem size – moderate sizes keep the LP cheap
        n = np.random.randint(30, 250)          # 30 … 250 elements
        # initialise either uniformly or at random
        if np.random.rand() < 0.4:
            f = np.full(n, 1.0 / n)            # uniform start
        else:
            f = np.random.rand(n)
            f = _normalise(f)

        # keep a copy for the current restart
        cur_seq = f.copy()
        cur_score = evaluate_sequence(cur_seq.tolist())
        if cur_score < best_overall_score:
            best_overall_score = cur_score
            best_overall_seq = cur_seq.copy()

        # ------------------------------------------------------------------
        # iterative improvement – each iteration solves an LP and mixes
        # the current sequence with the LP‑solution.
        no_improve_iters = 0
        while time.time() < deadline and no_improve_iters < 8:
            # normalise current sequence (scale does not matter)
            f_norm = _normalise(cur_seq)

            # maximum convolution value for the current sequence
            conv_f = np.convolve(f_norm, f_norm)
            M = conv_f.max()

            # solve LP to obtain a promising direction g
            g = _solve_convolution_lp(f_norm, M)
            if g is None:
                # fallback: random jitter to escape possible stagnation
                jitter = np.random.randn(n) * 0.004
                cur_seq = np.maximum(0.0, cur_seq + jitter)
                cur_seq = _normalise(cur_seq)
                cur_score = evaluate_sequence(cur_seq.tolist())
                continue

            # normalise g – the LP does not enforce any particular ℓ¹ norm
            g = _normalise(g)

            # try a few mixing coefficients and keep the best improvement
            new_seq, new_score = _mix_and_search(f_norm, g, n, deadline)

            if new_score + 1e-12 < cur_score:
                cur_seq = new_seq
                cur_score = new_score
                no_improve_iters = 0
                if cur_score < best_overall_score:
                    best_overall_score = cur_score
                    best_overall_seq = cur_seq.copy()
                # occasional debug output
                print(f"[{time.time():.1f}] attempt {attempt}, n={n}, "
                      f"new best score {best_overall_score:.6f}")
            else:
                # no improvement – try a small random perturbation
                jitter = np.random.randn(n) * 0.002
                cur_seq = np.maximum(0.0, cur_seq + jitter)
                cur_seq = _normalise(cur_seq)
                cur_score = evaluate_sequence(cur_seq.tolist())
                no_improve_iters += 1

        # ------------------------------------------------------------------
        # occasional extra exploration using the provided SOTA sequence
        if 'height_sequence_1' in globals() and time.time() < deadline:
            seq = np.array(height_sequence_1, dtype=float)
            seq = _normalise(seq + np.random.rand(*seq.shape) * 0.001)  # tiny noise
            sc = evaluate_sequence(seq.tolist())
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall_seq = seq.copy()
                print(f"[{time.time():.1f}] candidate from height_sequence_1 improved score to "
                      f"{best_overall_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best sequence found (as plain Python list of floats)
    if best_overall_seq is None:
        # as a safe fallback return a simple uniform sequence of length 2
        return [0.5, 0.5]
    return best_overall_seq.tolist()
