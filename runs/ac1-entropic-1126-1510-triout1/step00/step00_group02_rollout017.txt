def solve_min_max_convolution(f):
    n = len(f)
    # Variables: x (n) + t (1) => total n+1
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize t
    # Build A_ub matrix rows for each shift k
    rows = []
    for k in range(2*n - 1):
        row = np.zeros(n+1)
        # For each j (0..n-1)
        # coefficient = f[k - j] if 0 <= k-j < n else 0
        # Use slicing to fill faster:
        # j range max(0, k-(n-1)) to min(n-1, k)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        # For j in [low, high], coeff = f[k - j]
        # j goes forward, compute indices:
        # Use row[low:high+1] = f[k - np.arange(low, high+1)]
        row[low:high+1] = f[k - np.arange(low, high+1)]
        row[-1] = -1.0
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.zeros(2*n - 1)

    # Equality constraint sum x = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]  # t >=0

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'time_limit': 5})
    if res.success:
        x = res.x[:n]
        t = res.x[-1]
        return x, t
    else:
        return None, None

"""
Propose a candidate step‐function sequence.

The core idea:
*  Use the well known “arcsine” shape a_i = 1/√(i+1) which yields a
   convolution that is almost constant and gives a much lower
   evaluation value (~1.57) than the trivial uniform sequence (value = 2).
*  Normalise the vector to have unit sum – the evaluation function is
   scale invariant, so the absolute magnitude is irrelevant.
*  Apply a cheap stochastic local‑search (pairwise mass transfers) for a
   short time to try to improve the arcsine shape further.
*  All operations stay within the allowed 1000 s budget; the algorithm
   works for any length – a moderate length (2000) gives a good trade‑off
   between quality and speed.
"""

import numpy as np
import time
import random

# ------------------------------------------------------------
# Helper 1 – generate the arcsine‑type sequence
# ------------------------------------------------------------
def arcsine_sequence(length: int) -> np.ndarray:
    """
    Return a normalised vector a of given length with components
    a_i = 1 / sqrt(i+1).  The vector is scaled to have sum == 1.
    """
    idx = np.arange(1, length + 1, dtype=float)          # 1 … length
    a = 1.0 / np.sqrt(idx)                               # arcsine shape
    a /= a.sum()                                         # normalise to sum‑1
    return a


# ------------------------------------------------------------
# Helper 2 – simple pairwise stochastic improvement
# ------------------------------------------------------------
def stochastic_refine(seq: np.ndarray,
                     deadline: float,
                     max_iter: int = 2000,
                     step_scale: float = 0.02) -> np.ndarray:
    """
    Perform a lightweight stochastic local search that moves a small
    amount of mass between two random positions while preserving the
    total sum.  Only moves that improve the evaluation score are kept.
    The routine stops when the time budget expires or the iteration
    limit is reached.
    """
    best = seq.copy()
    best_score = evaluate_sequence(best.tolist())
    n = len(seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        # choose two distinct indices
        i, j = random.sample(range(n), 2)

        # propose moving a small amount δ from i to j
        delta = (np.random.rand() - 0.5) * step_scale * best[i]
        # ensure we do not create negative entries
        delta = np.clip(delta, -best[i] + 1e-12, best[j])

        candidate = best.copy()
        candidate[i] += delta
        candidate[j] -= delta

        # keep positivity (numpy clip is redundant here but safe)
        candidate = np.maximum(candidate, 0.0)
        # renormalise – total sum must stay 1 (scale invariance)
        candidate /= candidate.sum()

        score = evaluate_sequence(candidate.tolist())
        if score < best_score:                 # improvement
            best = candidate
            best_score = score
        it += 1

    return best


# ------------------------------------------------------------
# Main entry point required by the framework
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Return a sequence of non‑negative heights that (hopefully) gives a
    low value of ``evaluate_sequence``.

    Strategy
    ----------
    1.  Build an arcsine‑type vector of moderate size (default 2000).
    2.  Refine it with a cheap stochastic local search for a short
        period (leaving most of the budget for evaluation).
    3.  Return the best sequence encountered.
    """
    np.random.seed(seed)
    random.seed(seed)

    # Leave a few seconds buffer for the Python interpreter at the end
    deadline = time.time() + budget_s - 5.0

    # --------------------------------------------------------
    # 1) Start from the analytically motivated arcsine shape
    # --------------------------------------------------------
    length = 2000                         # moderate size – fast conv.
    seq = arcsine_sequence(length)

    # --------------------------------------------------------
    # 2) Stochastic refinement (very cheap – a few milliseconds per step)
    # --------------------------------------------------------
    # give the refinement at most 30 % of the total budget
    refine_deadline = time.time() + 0.3 * budget_s
    seq = stochastic_refine(seq, refine_deadline)

    # --------------------------------------------------------
    # 3) Final sanity check and return
    # --------------------------------------------------------
    score = evaluate_sequence(seq.tolist())
    print(f"[propose_candidate] final length={len(seq)}  score={score:.6f}")

    # Ensure the output satisfies the required format (list of floats)
    return seq.tolist()
