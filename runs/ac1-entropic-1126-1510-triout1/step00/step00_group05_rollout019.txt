def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)   # maximize sum
    # build convolution matrix: each k row has coefficients f_seq[i] where j = k-i
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        i = np.arange(i_min, i_max+1)
        j = k - i
        A[k, j] = f_seq[i]
    b = np.full(2*n-1, rhs)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'time_limit': 5})
    if res.success:
        return res.x
    else:
        return None

"""
Search for a step function with low evaluate_sequence score.
The algorithm uses multiple random restarts combined with a simple
projected gradient descent on a smooth approximation of the L∞
(convolution) norm.  For each restart a random probability vector
(p_i ≥ 0, Σp_i = 1) of length n is generated.  At every iteration we
compute the linear convolution of p with itself, turn the maximum
into a soft‑max weighting and take a gradient step that reduces the
large convolution entries.  After the step we project back onto the
simplex (non‑negative, unit sum).  The best vector found during the
budget is returned.  A few deterministic geometric candidates are also
tested for safety.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog

def _project_simplex(v):
    """Project a non‑negative vector onto the probability simplex."""
    # Bring negative entries to zero
    v = np.maximum(v, 0)
    if v.sum() == 0:
        # return uniform vector
        return np.full_like(v, 1.0 / v.size)
    return v / v.sum()

def _smooth_max(conv, alpha):
    """
    Soft‑max approximation of max(conv) with temperature 1/alpha.
    Returns the weights (soft‑max) and the approximate max value.
    """
    # shift for numerical stability
    shift = conv.max()
    ex = np.exp(alpha * (conv - shift))
    w = ex / ex.sum()
    # approximate max = shift + log(sum(exp(alpha*(conv-shift))))/alpha
    approx_max = shift + np.log(ex.sum()) / alpha
    return w, approx_max

def _gradient_step(p, alpha, lr):
    """
    Perform one gradient step on p to reduce the smooth max of its convolution.
    Returns the new probability vector (still sums to 1).
    """
    conv = np.convolve(p, p)
    w, _ = _smooth_max(conv, alpha)
    # gradient of smooth max w.r.t. p is 2 * (w * p) convolution
    grad = 2.0 * np.convolve(w, p[::-1], mode='valid')
    # gradient descent step
    p_new = p - lr * grad
    # project back onto simplex
    p_new = _project_simplex(p_new)
    return p_new

def _evaluate(p):
    """Wrapper that forwards a list to evaluate_sequence."""
    # evaluate_sequence expects a list, but works with list of floats.
    return evaluate_sequence(p.tolist())

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np
    import time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # leave a second margin

    # keep the best sequence we have seen
    best_seq = None
    best_score = float('inf')

    # a small utility to try a concrete probability vector
    def try_candidate(p):
        nonlocal best_seq, best_score
        # keep p as raw values (not scaled) – evaluate_sequence performs its own clipping
        score = _evaluate(p)
        if math.isfinite(score) and score < best_score:
            best_score = score
            best_seq = p.tolist()
            print(f"New best score {best_score:.6f} (len={len(p)})")
        return score

    # -----------------------------------------------------------------
    # 1) deterministic geometric families (fast cheap candidates)
    # -----------------------------------------------------------------
    for n in (500, 800, 1200, 1600):
        # geometric decay p_i ∝ α^i, α close to 1
        for delta in (0.001, 0.0007, 0.0005):
            alpha = 1.0 - delta
            p = np.power(alpha, np.arange(n), dtype=float)
            p = p / p.sum()
            try_candidate(p)

    # -----------------------------------------------------------------
    # 2) stochastic gradient descent with several restarts
    # -----------------------------------------------------------------
    # parameters for the SGD loop
    restart_limits = [
        (800, 2000),   # (length, max_iterations per restart)
        (1000, 1500),
        (1200, 1200),
    ]

    for n, max_iters in restart_limits:
        # how many restarts we can afford with the remaining time?
        while time.time() < deadline:
            # random Dirichlet start – concentration < 1 to get spiky vectors
            concentration = np.random.uniform(0.2, 0.8)
            p = np.random.gamma(concentration, 1.0, size=n)
            p = p / p.sum()               # make it a probability vector

            # evaluate the initial random point
            try_candidate(p)

            # schedule for the temperature (α) and learning rate
            alpha = 5.0                     # start rather smooth
            lr = 0.3                        # initial step size

            for it in range(max_iters):
                if time.time() > deadline:
                    break
                # every few iterations recompute the exact score – cheap enough
                if it % 10 == 0:
                    cur_score = _evaluate(p)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = p.tolist()
                        print(f"[{n}] iteration {it}   score {cur_score:.6f}")

                # take a gradient step
                p = _gradient_step(p, alpha, lr)

                # slowly increase the temperature (make the soft‑max sharper)
                alpha = min(50.0, alpha * 1.02)
                # slowly decay the learning rate
                lr = max(0.02, lr * 0.995)

            # after the inner loop we already recorded the best candidate
            # break the restart loop if we are close to the target
            if best_score < 1.51:
                break
        if best_score < 1.51:
            break

    # -----------------------------------------------------------------
    # 3) fallback – uniform vector if nothing better was found
    # -----------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = _evaluate(np.array(best_seq))

    # final sanity check – make sure the returned list is plain Python floats
    return [float(x) for x in best_seq]

Optimisation of a non‑negative step function for the provided evaluation
criterion.

The algorithm works in three stages:

   probability simplex (non‑negative, unit sum).  The best vector seen so

   length works).

only of non‑negative floats.  The implementation uses only `numpy` and

    Soft‑max approximation of the maximum of `conv`.

    Returns a pair ``(weights, approx_max)`` where ``weights`` sum to one
    and concentrate on the large entries of ``conv`` as ``alpha`` grows.
    """
    shift = conv.max()
    # all entries are ≤ shift, therefore the exponent is ≤ 0 → stable
    exp_vals = np.exp(alpha * (conv - shift))
    weight_sum = exp_vals.sum()
    w = exp_vals / weight_sum
    approx = shift + np.log(weight_sum) / alpha
    return w, approx


def _gradient_step(p: np.ndarray, alpha: float, lr: float) -> np.ndarray:
    """
    One projected gradient step that reduces a smooth approximation of
    ``max(conv(p,p))``.
    """
    conv = np.convolve(p, p)
    w, _ = _smooth_max(conv, alpha)
    # gradient of the smooth‑max w.r.t. p is 2·conv(w, p[::-1])
    grad = 2.0 * np.convolve(w, p[::-1], mode='valid')
    p_new = p - lr * grad
    return _project_simplex(p_new)


def _evaluate(p: np.ndarray) -> float:
    """Thin wrapper around the supplied ``evaluate_sequence``."""
    # ``evaluate_sequence`` works with a plain ``list[float]``.
    return evaluate_sequence(p.tolist())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    ``evaluate_sequence`` score.  The function respects the time budget,
    tries a few deterministic families, then runs multiple stochastic
    gradient‑descent restarts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # a short safety margin

    best_seq = None
    best_score = float('inf')

    def try_update(p_vec: np.ndarray):
        """Evaluate ``p_vec`` and remember it if it beats the current best."""
        nonlocal best_seq, best_score
        score = _evaluate(p_vec)
        if math.isfinite(score) and score < best_score:
            best_score = score
            best_seq = p_vec.tolist()
            print(f"New best: {best_score:.6f}  (len={len(p_vec)})")
        return score

    # ------------------------------------------------------------------
    # 1) deterministic geometric families (very cheap)
    # ------------------------------------------------------------------
    for n in (500, 800, 1200, 1600):
        for delta in (0.001, 0.0007, 0.0005):
            alpha = 1.0 - delta
            p = np.power(alpha, np.arange(n, dtype=float))
            p = p / p.sum()
            try_update(p)

    # ------------------------------------------------------------------
    # 2) stochastic restarts with projected gradient descent
    # ------------------------------------------------------------------
    # (length, allowed iterations) pairs – larger n = fewer iters
    restart_plan = [
        (800, 2500),
        (1000, 2000),
        (1200, 1500),
    ]

    for n, max_iters in restart_plan:
        # keep trying restarts until time runs out or we hit the target
        while time.time() < deadline:
            # ----------------------------------------------------------------
            # generate a random start – Dirichlet with a random concentration
            # ----------------------------------------------------------------
            concentration = np.random.uniform(0.15, 0.8)   # spikier than uniform
            raw = np.random.gamma(concentration, 1.0, size=n)
            p = raw / raw.sum()          # a probability vector

            # evaluate the raw start (useful for logging)
            try_update(p)

            # hyper‑parameters for the inner optimisation loop
            alpha = 5.0                  # start with a smooth soft‑max
            lr = 0.30                    # fairly aggressive step size

            for it in range(max_iters):
                if time.time() > deadline:
                    break

                # every few iterations we check the exact score (cheap)
                if (it & 7) == 0:        # ≈ every 8th iteration
                    cur_score = _evaluate(p)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = p.tolist()
                        print(f"[{n}] iter {it:4d}  score {cur_score:.6f}")

                # ---- gradient descent step --------------------------------
                p = _gradient_step(p, alpha, lr)

                # schedule: make the soft‑max sharper and the step smaller
                alpha = min(60.0, alpha * 1.015)   # anneal temperature up
                lr = max(0.02, lr * 0.997)        # cool learning rate

                # early exit if we have already hit the target bound
                if best_score < 1.503:
                    break

            # stop early if we already are below the desired threshold
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # 3) fallback – plain uniform (should never be used for a winning run)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = _evaluate(np.array(best_seq))
        print(f"Fallback uniform used, score={best_score:.6f}")

    # return a plain list of Python floats (the conversion is cheap)
    return [float(v) for v in best_seq]
