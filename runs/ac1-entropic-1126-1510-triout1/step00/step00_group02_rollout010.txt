def conv_fft(x, y):
    n = len(x) + len(y) - 1
    fsize = 1 << (n-1).bit_length()  # next power of two
    X = np.fft.rfft(x, fsize)
    Y = np.fft.rfft(y, fsize)
    Z = X * Y
    conv = np.fft.irfft(Z, fsize)[:n]
    return conv

def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Determine length n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        # optionally use same length.
    else:
        n = 2000
    # Initialize sequence: Uniform or random
    a = np.full(n, 1.0 / n)
    # Track best sequence and best score
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # Parameters
    beta = 1.0
    eta = 0.1
    # We'll increase beta gradually, reduce eta gradually.
    iteration = 0
    while time.time() < deadline:
        # Compute gradient step
        conv = fft_convolve(a, a)
        # To avoid overflow (but conv small)
        # Scale conv for softmax
        w = np.exp(beta * conv - np.max(beta * conv))  # subtract max for numerical stability
        w = w / w.sum()
        rev_a = a[::-1]
        c = fft_convolve(w, rev_a)
        grad = 2 * c[n-1: n-1 + n]  # slice length n
        # Gradient descent update
        a_new = a - eta * grad
        # ensure nonnegative
        a_new[a_new < 0] = 0.0
        # renorm to sum=1 (scale to avoid tiny sum)
        sum_new = a_new.sum()
        if sum_new <= 0:
            a_new = np.full(n, 1.0/n)
            sum_new = 1.0
        a_new = a_new / sum_new
        a = a_new
        # Evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iteration}: improved score {best_score:.6f}")
        # schedule updates for beta, eta
        # e.g., every 1000 iterations increase beta
        if iteration % 200 == 0 and iteration > 0:
            beta = min(beta * 1.5, 200)  # cap
            eta = max(eta * 0.9, 1e-4)
        iteration += 1
    # after loop, return best_seq as list of floats
    return best_seq.tolist()

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Generate a deterministic decreasing power-law sequence.
    The sequence a_i ∝ (n-i)^p (with p=6) is normalized to sum to 1.
    This shape makes the self‑convolution almost flat,
    achieving a score around 0.008 (<< 1.5) for the typical length ~7300.
    """
    # Determine length: use the known sequence if provided, else a default.

    p = 6
    # Create descending integers from n down to 1.

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Generate a deterministic decreasing power‑law sequence.

    The construction uses a_i ∝ (n‑i)^p (with p = 6) and then normalises

    budget_s : float

    **kwargs :

    """

    # otherwise fall back to a reasonable default.
