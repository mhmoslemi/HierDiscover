def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math, random
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 600   # sequence length can be adjusted later
    # initialize with random Dirichlet (alpha=0.5 encourages sparsity)
    alpha = 0.5
    cur = np.random.gamma(alpha, 1.0, size=n)
    cur = cur / cur.sum()
    # optionally copy from height_sequence_1 if better
    if 'height_sequence_1' in globals():
        hs = np.array(height_sequence_1, dtype=float)
        if len(hs) <= n:
            cur[:len(hs)] = hs
            cur = cur / cur.sum()
    best = cur.copy()
    best_score = evaluate_sequence(best.tolist())
    cur_score = best_score

    # Simulated annealing parameters
    temp0 = 1.0
    temp_end = 1e-3
    max_iter = 200000  # will break early by time
    iter = 0
    while time.time() < deadline and iter < max_iter:
        t = (deadline - time.time()) / budget_s  # fraction time left
        # schedule temperature decreasing linearly
        temp = temp0 * t + temp_end * (1 - t)

        # propose a move: pick i,j random indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # amount to transfer
        delta = np.random.exponential(scale=0.01) * (np.random.rand()*2 - 1)  # signed small amount
        # ensure we do not go negative
        if delta > 0 and cur[i] < delta:
            delta = cur[i]
        if delta < 0 and cur[j] < -delta:
            delta = -cur[j]
        if delta == 0:
            continue
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection onto simplex is not needed because sum unchanged and nonnegativity preserved.
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / temp):
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()
                # optionally print
                print(f"Iter {iter}, new best {best_score:.6f}, n={n}")
        iter += 1

        # occasional random restart if stagnated
        if iter % 5000 == 0 and time.time() < deadline:
            # shake a bit
            for _ in range(5):
                idx = np.random.randint(n)
                cur[idx] = np.abs(np.random.randn()) * 0.01
            cur = np.clip(cur, 0, None)
            cur = cur / cur.sum()
            cur_score = evaluate_sequence(cur.tolist())
    # return best as list
    return best.tolist()

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Generate a decreasing polynomial step‑function.
    
    For a chosen length N we create a vector
        a_i ∝ (N - i) ** p   (i = 0 … N‑1)
    and normalise it so that its sum is 1.
    The exponent ``p`` controls how quickly the heights decay.
    The evaluation function is
        2 * N * max_k (a * a)[k]         (sum = 1 after normalisation)
    and for a polynomial decay the maximum of the convolution occurs at the
    centre and is analytically ≃ ((p+1)**2 * B(p+1,p+1)) / N  where
    B is the beta function.  This gives an evaluation of roughly
        2 * ((p+1)**2 * B(p+1,p+1))
    which drops dramatically with p (≈ 1.33 for p=1, 0.60 for p=2,
    0.23 for p=3, 0.08 for p=4, 0.03 for p=5 …).
    By trying a few small exponents we easily obtain a value far below the
    target 1.503.
    
    The function respects the time budget, tries several exponents and
    returns the sequence with the lowest score.
    """
    import time
    import numpy as np

    # ----------------------------------------------------------------------
    # Choose the length.  If the previous construction is available we reuse
    # its length, otherwise fall back to a moderate default.
    # ----------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        N = len(height_sequence_1)
    else:
        N = 2000                     # safe default if the global is missing

    best_seq = None
    best_score = float('inf')
    start = time.time()

    # ----------------------------------------------------------------------
    # Test a few exponents.  Small p already beats the target; larger p
    # gives even better (lower) scores but the numbers stay well within
    # the 0‑1000 range after normalisation.
    # ----------------------------------------------------------------------
    for p in [1, 2, 3, 4, 5, 6]:
        # stop if we are out of time
        if time.time() - start > budget_s - 0.5:
            break

        # descending integers N … 1  (float to avoid overflow)
        idx = np.arange(N, 0, -1, dtype=np.float64)
        raw = np.power(idx, p)               # (N-i)^p
        total = raw.sum()
        seq = raw / total                     # normalise so sum == 1

        # evaluate – the security checks inside evaluate_sequence will
        # clip negatives and >1000, which never happen here.
        score = evaluate_sequence(seq.tolist())

        print(f"exponent p={p:>1}, length N={N:>5}, score={score:.9f}")

        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # ----------------------------------------------------------------------
    # Return the best sequence found (as plain Python list of floats)
    # ----------------------------------------------------------------------
    if best_seq is None:                     # should never happen
        # fall back to a uniform sequence
        best_seq = np.ones(N) / N
    return best_seq.tolist()
