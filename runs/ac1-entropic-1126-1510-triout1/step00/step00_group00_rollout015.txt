"""
propose_candidate
-----------------

Search for a step‑function (non‑negative sequence) that yields a low value of the
evaluation function

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

The approach is a simple *LP‑based improvement*:

1. Start from a random, non‑uniform sequence (many zero / small entries).  
   The non‑uniformity allows the linear program to increase the total mass
   while keeping the current maximum convolution `M` unchanged.

2. Solve the linear program

       maximise   Σ b_i
       subject to  conv(f,b) ≤ M          (elementwise)
                   b_i ≥ 0

   For a fixed `M` this maximises the L¹‑norm of a new sequence `b`
   that can be added without raising the current peak.

3. Rescale `b` to the standard normalisation
   `sum(b) = √(2·n)`.  Replacing the current sequence `f` by this rescaled
   `b` never increases `M`; in practice it strictly reduces it unless `f`
   is already uniform.

4. Iterate the LP step as long as an improvement is observed.
   When no progress is made a new random start is generated.
   The whole process runs until the time budget expires and the best
   sequence found is returned.

The method is cheap (the LP size is (2n‑1)×n) and works well for
moderate lengths (n≈200‑800).  Even a single LP step starting from a
non‑uniform vector typically beats the trivial uniform bound (E=2)
and can approach the target value ~1.5 for sufficiently many iterations.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# the evaluator is provided in the execution environment
# from ... import evaluate_sequence   # not needed, it is already in globals()


def _build_conv_constraints(f):
    """
    Build the matrix A and vector b for the linear constraints
        A @ b <= M
    where each row corresponds to one entry of conv(f, b).
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    # For each output index k compute the coefficients of b_j.
    #   conv_k = sum_i f_i * b_{k-i}
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        A[k, j] = f[i]
    return A


def _lp_improve(f, M):
    """
    Solve the LP that maximises sum(b) subject to conv(f,b) ≤ M and b ≥ 0.
    Returns a new sequence (already normalised to sqrt(2n)) or None on failure.
    """
    n = len(f)
    A = _build_conv_constraints(f)
    b_vec = np.full(2 * n - 1, M, dtype=float)

    # maximise Σ b_i  ⇒ minimise -Σ b_i
    c = -np.ones(n, dtype=float)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_vec,
        bounds=(0, None),
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None

    g = res.x
    total = g.sum()
    if total <= 0.0:
        return None

    # normalise to the standard L¹‑norm √(2n)
    target = math.sqrt(2.0 * n)
    g = g * (target / total)
    return g


def propose_candidate(seed: int = 12345, budget_s: int = 990, **kwargs):
    """
    Returns a list of non‑negative heights.  The routine runs for up to
    ``budget_s`` seconds, repeatedly trying random starts and improving them
    with the LP scheme described above.  The best sequence seen is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 1.0)  # leave a tiny margin

    # keep the best ever found
    best_seq = None
    best_score = float("inf")

    # a few different problem sizes keep the LP cheap
    candidate_sizes = [200, 400, 600, 800]

    while time.time() < deadline:
        # choose a length (cycle through the list)
        n = np.random.choice(candidate_sizes)

        # ---- random non‑uniform start ---------------------------------
        # exponential distribution gives many tiny entries and a few large ones
        f = np.random.exponential(scale=1.0, size=n)
        # enforce a reasonable amount of zeros to create asymmetry
        zero_mask = np.random.rand(n) < 0.2
        f[zero_mask] = 0.0
        # normalise to the standard L¹‑norm √(2n)
        target_sum = math.sqrt(2.0 * n)
        if f.sum() == 0.0:
            f[:] = 1.0
        f = f * (target_sum / f.sum())

        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[init] n={n} score={cur_score:.6f}")

        # ---- iterative LP improvement ---------------------------------
        # stop inner loop after a modest number of successful steps
        successful_steps = 0
        while time.time() < deadline and successful_steps < 30:
            conv = np.convolve(f, f)
            M = conv.max()

            g = _lp_improve(f, M)
            if g is None:
                break

            new_score = evaluate_sequence(g.tolist())

            # keep the overall best
            if new_score < best_score:
                best_score = new_score
                best_seq = g.copy()
                print(
                    f"[improve] n={n} step={successful_steps+1} "
                    f"score={new_score:.6f}"
                )

            # accept the move only if it improves the current sequence
            if new_score < cur_score - 1e-12:
                f = g
                cur_score = new_score
                successful_steps += 1
            else:
                # no further progress from this start – break to a new random start
                break

        # early exit if we have reached a very good bound
        if best_score < 1.6:   # close to the advertised target of 1.503
            break

    # fallback – uniform sequence (guaranteed to be feasible)
    if best_seq is None:
        n = 500
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
