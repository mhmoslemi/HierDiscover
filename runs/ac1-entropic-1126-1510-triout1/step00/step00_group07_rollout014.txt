def project_simplex(v):
    """Project onto simplex (nonnegative, sum=1)."""
    # based on Duchi et al. 2008.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
