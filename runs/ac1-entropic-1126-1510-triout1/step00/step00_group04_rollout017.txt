"""
Propose a low‑evaluation step‑function using a 1/√(i) shape and greedy
mass‑re‑distribution.  The initial sequence a_i = 1/√(i+1) gives a
convolution close to a constant (≈π) and yields an evaluation near
π/2 ≈ 1.57.  A simple hill‑climbing step repeatedly moves a small
fraction of mass from the indices that contribute the most to the
current maximal convolution entry toward the smallest entries.
Occasional random perturbations and a tiny simulated‑annealing
acceptance rule help escape local plateaus.  The algorithm runs for the
available time budget and returns the best sequence found.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: full linear convolution using a power‑of‑2 FFT (fast for many calls)
def _full_convolution(seq: np.ndarray) -> np.ndarray:
    n = len(seq)
    # length needed for linear convolution
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    fv = np.fft.rfft(seq, L)
    conv = np.fft.irfft(fv * fv, L)
    return conv[: 2 * n - 1]               # discard trailing zeros


# Helper: contribution of each coefficient to a given shift k
def _prod_for_shift(seq: np.ndarray, k: int) -> np.ndarray:
    n = len(seq)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    prod = np.zeros(n, dtype=seq.dtype)
    prod[mask] = seq[i[mask]] * seq[j[mask]]
    return prod


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimizes `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget in seconds (default 1000).
    **kwargs : dict
        Ignored – kept for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence discovered within the time budget.
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # 1.  Choose a moderate length – long enough for a good bound,
    #     short enough for many iterations.
    n = 3000                      # can be tuned; 3k works well within 1 000 s

    # 2.  Initialise with the classic 1/√(i) shape (i starts at 1)
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = 1.0 / np.sqrt(idx)           # non‑negative, already ≤ 1

    # 3.  Keep track of the best sequence we have seen.
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    curr_seq = seq.copy()
    curr_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Main optimisation loop – simple gradient‑free hill climbing.
    deadline = time.time() + budget_s - 5   # leave a couple of seconds margin
    total_iters = 0

    while time.time() < deadline:
        total_iters += 1

        # ----- compute convolution and locate the current worst shift -----
        conv = _full_convolution(curr_seq)
        k_max = int(np.argmax(conv))

        # ----- contributions of each coefficient to that worst shift -----
        prod = _prod_for_shift(curr_seq, k_max)

        # indices to decrease / increase mass
        i_decr = int(np.argmax(prod))          # large product => hurts the max
        i_incr = int(np.argmin(curr_seq))      # smallest coefficient

        if i_decr == i_incr:
            i_incr = (i_incr + 1) % n

        # ----- move a modest fraction of mass from i_decr to i_incr ----------
        delta = 0.05 * curr_seq[i_decr]        # 5 % of the larger entry
        if delta <= 0.0:
            continue

        new_seq = curr_seq.copy()
        new_seq[i_decr] -= delta
        new_seq[i_incr] += delta
        if new_seq[i_decr] < 0.0:   # safety (should never happen)
            continue

        # ----- evaluate the new candidate ----------------------------------
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < curr_score:            # greedy acceptance
            curr_seq = new_seq
            curr_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f"[{total_iters:6d}] improved best: {best_score:.6f}")

        else:
            # occasional uphill move (tiny simulated annealing)
            time_left = deadline - time.time()
            fraction = time_left / (budget_s - 5)
            temperature = max(best_score * 0.05 * fraction, 1e-12)
            if temperature > 0 and np.random.rand() < math.exp((curr_score - new_score) / temperature):
                curr_seq = new_seq
                curr_score = new_score

        # ----- random perturbation every now‑and‑then to diversify ----------
        if np.random.rand() < 0.01:
            ri = np.random.randint(n)
            # small random tweak of one entry
            tweak = (np.random.rand() * 2.0 - 1.0) * 0.02 * curr_seq[ri]
            if curr_seq[ri] + tweak < 0:
                tweak = -curr_seq[ri] * 0.5
            rj = (ri + np.random.randint(1, n)) % n
            pert_seq = curr_seq.copy()
            pert_seq[ri] += tweak
            pert_seq[rj] -= tweak
            if pert_seq[rj] < 0.0:
                continue
            pert_score = evaluate_sequence(pert_seq.tolist())
            if pert_score < best_score:
                best_seq = pert_seq.copy()
                best_score = pert_score
                print(f"[{total_iters:6d}] random tweak improved best: {best_score:.6f}")
            if pert_score < curr_score:
                curr_seq = pert_seq
                curr_score = pert_score

    print(f"[done] iterations={total_iters}, best score={best_score:.6f}")
    # The evaluator is scale‑invariant, but keep the numbers modest.
    return best_seq.tolist()
