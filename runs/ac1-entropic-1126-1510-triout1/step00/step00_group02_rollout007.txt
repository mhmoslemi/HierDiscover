"""
Propose a step‐function (list of non‑negative heights) that minimizes the
evaluation metric

    2·n·max(conv(a,a)) / (sum(a))²

where `conv` is the full (linear) convolution.  The theoretical lower bound of the
ratio is 1 (the average of the convolution values) and the uniform sequence
gives the value 2.  Randomly sprinkling “1”s on the grid makes the pairwise
sums i+j almost uniformly distributed, lowering the maximum convolution value
close to its average and therefore driving the ratio towards 1.

The algorithm therefore performs a fast stochastic search:
* repeatedly generate a random binary sequence with a density chosen uniformly
  in a wide interval,
* evaluate its score using a lightweight implementation (convolution via
  NumPy),
* keep the best sequence seen,
* after a good candidate is found, perform a short hill‑climbing phase that
  swaps/adds/removes a single “1” to try to improve the score further.

The whole search respects the given time budget (default 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np


def _score(seq: np.ndarray, n: int) -> float:
    """
    Compute the evaluation score for a given sequence.
    The implementation mirrors ``evaluate_sequence`` but is much faster
    because the input is already known to be a non‑negative float array
    with values in [0,1000].

    Parameters
    ----------
    seq : np.ndarray
        1‑D array of non‑negative heights.
    n : int
        Length of the sequence (len(seq)) – passed to avoid recomputation.

    Returns
    -------
    float
        The objective value (higher is better, lower is what we want).
    """
    # Full convolution (ordered pairs, i.e. a_i*a_j and a_j*a_i both appear)
    conv = np.convolve(seq, seq)          # shape 2*n-1
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:                      # protect against too small sums
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)


def _random_binary(n: int, density: float) -> np.ndarray:
    """
    Return a binary (0/1) float array of length ``n`` where each entry is 1
    with probability ``density``.
    """
    return (np.random.rand(n) < density).astype(np.float64)


def _hill_climb(seq: np.ndarray,
                cur_score: float,
                n: int,
                stop_time: float,
                max_iter: int = 300) -> tuple[np.ndarray, float]:
    """
    Very simple local optimisation: repeatedly propose a neighbour obtained
    by swapping a 1 with a 0, or by adding / removing a single 1.
    The move is kept only if it strictly improves the score.
    The routine stops after ``max_iter`` successful attempts or when the
    time budget expires.

    Parameters
    ----------
    seq : np.ndarray
        Current binary sequence.
    cur_score : float
        Its score (as returned by ``_score``).
    n : int
        Length of the sequence.
    stop_time : float
        Absolute time (seconds since epoch) at which the search must stop.
    max_iter : int, optional
        Upper bound on the number of neighbour evaluations.

    Returns
    -------
    tuple[np.ndarray, float]
        The best sequence found in the local search and its score.
    """
    best_seq = seq.copy()
    best_score = cur_score

    it = 0
    while it < max_iter and time.time() < stop_time:
        it += 1
        cand = best_seq.copy()

        r = np.random.rand()
        if r < 0.6:                                # swap a 1 with a 0
            ones = np.flatnonzero(cand > 0)
            zeros = np.flatnonzero(cand == 0)
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            cand[i] = 0.0
            cand[j] = 1.0
        elif r < 0.85:                             # add a 1
            zeros = np.flatnonzero(cand == 0)
            if len(zeros) == 0:
                continue
            j = np.random.choice(zeros)
            cand[j] = 1.0
        else:                                      # remove a 1
            ones = np.flatnonzero(cand > 0)
            if len(ones) == 0:
                continue
            i = np.random.choice(ones)
            cand[i] = 0.0

        cand_score = _score(cand, n)
        if cand_score < best_score:
            best_seq = cand
            best_score = cand_score
            # early exit if we already beat a reasonable target
            if best_score < 1.55:
                break
    return best_seq, best_score


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Stochastic search for a low‑evaluation step function.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found (list of non‑negative heights).
    """
    np.random.seed(seed)

    # Length of the problem – use the length of the reference construction
    # if it exists, otherwise fall back to a modest size.
    ref = globals().get("height_sequence_1")
    n = len(ref) if ref is not None else 2000

    deadline = time.time() + budget_s - 2   # keep a small safety margin

    best_seq = np.zeros(n, dtype=np.float64)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main random‑sampling loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Random density between 0.2 and 0.9 – this interval covers the
        # region where the ratio is usually well below the target 1.503.
        density = np.random.uniform(0.2, 0.9)
        candidate = _random_binary(n, density)

        # Ensure the sequence is not empty (the evaluator would return inf)
        if candidate.sum() < 1:
            candidate[0] = 1.0

        cand_score = _score(candidate, n)

        if cand_score < best_score:
            best_seq = candidate.copy()
            best_score = cand_score
            print(f"[{time.time():.1f}] New global best = {best_score:.6f} "
                  f"(density {density:.3f})")

            # -----------------------------------------------------------------
            # Optional short hill‑climbing phase to squeeze a bit more performance
            # -----------------------------------------------------------------
            # We give the local optimiser a small fraction of the remaining time.
            local_deadline = min(deadline, time.time() + 0.4 * (deadline - time.time()))
            improved_seq, improved_score = _hill_climb(best_seq,
                                                       best_score,
                                                       n,
                                                       local_deadline,
                                                       max_iter=250)
            if improved_score < best_score:
                best_seq = improved_seq
                best_score = improved_score
                print(f"[{time.time():.1f}]  ↳ Local improvement → {best_score:.6f}")

        # Early exit if we have already crossed a comfortable margin
        # (e.g. well below the target 1.503).  This saves time for the
        # remaining budget.
        if best_score < 1.45:
            # still give the caller a tiny window to finish cleanly
            break

    print(f"Finished search – best score = {best_score:.6f}")
    return best_seq.tolist()
