def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    # choose list of lengths to try
    lengths = [256, 512, 1024, 2048, 4096]
    # we may also include the length of height_sequence_1 if given.
    for n in lengths:
        # initialize a random Dirichlet distribution
        a = np.random.rand(n)
        a = a / a.sum()
        # compute b
        b = np.convolve(a, a)
        M = b.max()
        score = 2 * n * M
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # SA parameters
        T0 = 0.01 * score   # initial temperature relative to score
        T = T0
        it = 0
        # annealing schedule: geometric decay
        while time.time() < deadline:
            it += 1
            # gradually lower temperature
            if it % 2000 == 0:
                T *= 0.9
                # optional random restart if stuck
            # pick i and j
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            # amount to move: fraction of a[i]
            if a[i] == 0:
                continue
            frac = np.random.rand() * 0.5  # up to half of a[i]
            delta = a[i] * frac
            # ensure a[j] not exceed 1000 after addition? but not needed
            # compute increments
            Δ_i = -delta
            Δ_j = delta
            # compute delta array
            delta_b = np.zeros(2*n-1)
            # i contribution: add 2*Δ_i*a for shift range
            start_i = i
            delta_b[start_i:start_i+n] += 2 * Δ_i * a
            delta_b[2*i] += Δ_i * Δ_i  # extra diagonal term
            # j contribution
            start_j = j
            delta_b[start_j:start_j+n] += 2 * Δ_j * a
            delta_b[2*j] += Δ_j * Δ_j
            # cross term
            delta_b[i + j] += 2 * Δ_i * Δ_j
            # new b
            b_new = b + delta_b
            M_new = b_new.max()
            # acceptance
            if M_new <= M or np.random.rand() < np.exp(-(M_new - M) / (T + 1e-12)):
                # accept
                a[i] = a[i] + Δ_i
                a[j] = a[j] + Δ_j
                b = b_new
                M = M_new
                score = 2 * n * M
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
            # maybe occasional projection to keep sum exactly 1 (should be preserved)
            # we preserve sum automatically because we transfer mass.
            # if time near deadline break
            if time.time() > deadline:
                break
        # after SA for this n, print best
        print(f"Length {n}, best score {best_score}")
    # final convert best_seq to list of floats
    if best_seq is None:
        # fallback to uniform as safe
        n = 1000
        best_seq = np.full(n, 1.0/n)
    return best_seq.tolist()

def solve_lp_direction(a, M):
    n = len(a)
    c = -np.ones(n)  # maximize sum(g)
    A_ub = []
    b_ub = []
    for k in range(2*n-1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = a[i]
        A_ub.append(row)
        b_ub.append(M)
    # nonnegativity constraints
    A_ub = np.vstack([A_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if result.success:
        return result.x
    return None
```

"""
Propose a candidate step‑function (list of non‑negative heights) that beats the
trivial uniform construction.  

The algorithm works in three stages:

1. **Parameterised search** –  Different simple families (geometric decay,
   power‑law decay and a two‑region piecewise constant shape) are generated.
   Each family is mixed with the uniform distribution and the best
   combination (lowest value of the evaluation function) is remembered.
   The evaluation of a candidate is cheap because the convolution maximum is
   obtained with an FFT based convolution.

2. **Local refinement** –  Starting from the best candidate, a short projected
   sub‑gradient descent run is performed.  The sub‑gradient of the current
   maximal convolution entry is used to move the probability mass away from
   the indices that contribute most to that entry.  After each step the
   vector is projected onto the probability simplex (non‑negative, sum = 1).

3. **Return** –  The best sequence seen during the whole budget is returned.
   All numbers are non‑negative; the evaluator rescales them internally, so
   the absolute scale is irrelevant.

The whole procedure respects the supplied time budget (default 1000 s) and
uses only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def max_convolution_fft(a: np.ndarray) -> float:
    """
    Return max_k (a * a)[k] using an FFT based convolution.
    a must be one‑dimensional, non‑negative, and sum(a) > 0.
    """
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:size]
    return float(conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x | x_i >= 0, sum x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def geometric_sequence(n: int, r: float) -> np.ndarray:
    """Return a decreasing geometric sequence of length n, normalised to sum = 1."""
    if r <= 0.0:
        # fall back to uniform if r is out of range
        return np.full(n, 1.0 / n)
    a = r ** np.arange(n, dtype=np.float64)
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """Return a decreasing power‑law sequence (i+1)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def piecewise_two_region(n: int, m: int, prop: float) -> np.ndarray:
    """
    Two‑region piecewise constant distribution.
    First region: first m entries receive total mass `prop`.
    Remaining (n‑m) entries receive the rest.
    Both regions are uniform inside.
    """
    if m <= 0 or m >= n:
        # degenerate – fall back to uniform
        return np.full(n, 1.0 / n)
    a = np.empty(n, dtype=np.float64)
    a[:m] = prop / m
    a[m:] = (1.0 - prop) / (n - m)
    return a


def evaluate_candidate(a: np.ndarray) -> float:
    """
    Compute the exact value returned by `evaluate_sequence` for a given
    coefficient list.  This is 2·n·max(conv(a,a))/(sum(a)**2);  the scaling
    of `a` cancels, so we keep sum(a)=1.
    """
    n = a.shape[0]
    max_b = max_convolution_fft(a)
    return 2.0 * n * max_b


def subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One projected sub‑gradient step aimed at reducing the current maximal
    convolution entry.
    """
    n = a.shape[0]
    # compute convolution and locate the maximal index
    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))

    # gradient of b_{k_max} w.r.t. a_i is 2*a_{k_max-i} (if in range)
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    i_range = np.arange(lo, hi + 1, dtype=int)
    j_range = k_max - i_range
    grad[i_range] = 2.0 * a[j_range]

    # take a step opposite to the gradient and project back onto simplex
    a_new = a - step * grad
    a_proj = simplex_projection(a_new)
    return a_proj


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Return a list of non‑negative heights (the step function) that gives a
    low value of the provided `evaluate_sequence` function.
    The routine spends the allocated time searching over simple parametric
    families and then performs a short local optimisation.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – larger n gives more flexibility but costs
    # more per evaluation.  2000 is a good compromise for the 1000 s budget.
    # ------------------------------------------------------------------
    n = 2000

    # Uniform distribution is the baseline (score = 2.0)
    uniform = np.full(n, 1.0 / n)
    best_a = uniform.copy()
    best_score = evaluate_candidate(best_a)

    # ------------------------------------------------------------------
    # 1) PARAMETRISED SEARCH
    # ------------------------------------------------------------------
    # geometric families
    for r in np.linspace(0.85, 0.9999, 30):
        geom = geometric_sequence(n, r)
        for alpha in np.linspace(0.0, 1.0, 11):
            cand = alpha * uniform + (1.0 - alpha) * geom
            # both components already sum to 1 → cand also sums to 1
            score = evaluate_candidate(cand)
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        if time.time() >= deadline:
            break

    # power‑law families
    if time.time() < deadline:
        for p in np.linspace(0.1, 3.0, 30):
            power = powerlaw_sequence(n, p)
            for alpha in np.linspace(0.0, 1.0, 11):
                cand = alpha * uniform + (1.0 - alpha) * power
                score = evaluate_candidate(cand)
                if score < best_score:
                    best_score = score
                    best_a = cand.copy()
            if time.time() >= deadline:
                break

    # two‑region piecewise families
    if time.time() < deadline:
        for m_frac in np.linspace(0.05, 0.5, 10):
            m = max(1, int(m_frac * n))
            for prop in np.linspace(0.1, 0.9, 9):
                piece = piecewise_two_region(n, m, prop)
                score = evaluate_candidate(piece)
                if score < best_score:
                    best_score = score
                    best_a = piece.copy()
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # 2) LOCAL REFINEMENT – projected sub‑gradient descent
    # ------------------------------------------------------------------
    # start from the best candidate found above
    a = best_a.copy()
    step = 0.1                 # initial step size
    it = 0
    while time.time() < deadline and step > 1e-5 and it < 5000:
        it += 1
        # perform a sub‑gradient step
        a_new = subgradient_step(a, step)
        score_new = evaluate_candidate(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            # when we improve we can try a slightly larger step
            step = min(0.2, step * 1.05)
        else:
            # otherwise shrink the step size
            step *= 0.7
        a = a_new
        # occasional random kick to escape very flat regions
        if it % 500 == 0 and np.random.rand() < 0.2:
            kick = np.random.rand(n)
            kick = kick / kick.sum()
            a = 0.9 * a + 0.1 * kick
            a = simplex_projection(a)

    # ------------------------------------------------------------------
    # 3) Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    # Convert to Python list; values are already non‑negative.
    return best_a.tolist()
