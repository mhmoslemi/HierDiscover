def project_simplex(v, S):
    # ensure nonnegative? Actually can be negative before projection
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - S) / (np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        theta = (cssv[-1] - S) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - S) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def compute_objective(a):
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    conv = conv[:2*n-1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

"""
Propose a step–function sequence that tries to minimise

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a simplex‑projection based simulated‑annealing / sub‑gradient scheme.
The algorithm works with any length (defaults to the length of the supplied
global `height_sequence_1`).  It starts from a uniform vector, breaks the
symmetry with a small random perturbation and then repeatedly:

* evaluates the full convolution with FFT,
* finds the shift(s) attaining the current maximum,
* builds a sub‑gradient of the max‑convolution,
* takes a diminishing step in the opposite direction,
* projects back onto the simplex of non‑negative vectors with the required
  sum `sqrt(2*n)`.

Occasional random perturbations are injected to escape flat regions.
The best sequence found before the time budget expires is returned.
"""

import numpy as np
import time

# --------------------------------------------------------------------
# Helper functions (no closures, no lambdas, top level as required)
# --------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)  # 1‑based index
    # find the largest rho satisfying the condition
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – unlikely, but keep it safe
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using a power‑of‑two FFT for speed."""
    n = a.shape[0]
    # length of the full convolution is 2*n-1; pick next power of two
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    # Numerical noise may produce tiny negative values – clamp to 0
    conv = np.maximum(conv, 0.0)
    return conv


def _objective(a: np.ndarray) -> float:
    """Return the value of the evaluation function for sequence a."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    conv = _conv_full(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _perturb_and_project(a: np.ndarray, scale: float) -> np.ndarray:
    """Add Gaussian noise (zero‑sum) and re‑project onto the simplex."""
    v = a + np.random.normal(scale=scale, size=a.shape)
    return _project_to_simplex(v, a.sum())


def _subgradient(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    Compute a sub‑gradient of the function g(a)=max_k (a*a)[k]
    at the current point a.  All shifts k where conv[k] is within
    a tiny tolerance of max_val are taken as active.
    """
    n = a.shape[0]
    # tolerance to collect (near) active shifts
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices i = lo … hi, j = k - i
        i_range = np.arange(lo, hi + 1, dtype=int)
        j_range = k - i_range
        # contribution 2 * a_j to each gradient component a_i
        grad[i_range] += 2.0 * a[j_range]
    return grad

        step = (target_sum * 0.1) / np.sqrt(it + 1.0)

        # Gradient step and projection onto the simplex
        a_new = a - step * grad
        a_new = _project_to_simplex(a_new, target_sum)

        # Occasionally inject a random perturbation to avoid stagnation
        if np.random.rand() < 0.005:
            a_new = _perturb_and_project(a_new, scale=target_sum * 0.05)

        # Evaluate the new candidate
        val_new = _objective(a_new)

        # Accept if we improve the current point (always keep the global best)
        if val_new <= _objective(a):
            a = a_new
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[{it:06d}] new best objective = {best_val:.6f}")

        # Periodic re‑initialisation from the best point to explore new basins
        if it % 5000 == 0 and it > 0:
            a = _perturb_and_project(best_a, scale=target_sum * 0.02)

        # Simple progress report
        if it % 2000 == 0:
            elapsed = time.time() - start_time
            print(f"[{it:06d}] elapsed={elapsed:.1f}s best={best_val:.6f}")

        it += 1

    # --------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # --------------------------------------------------------------
    print(f"[final] iterations={it}, best objective={best_val:.6f}")
    return [float(x) for x in best_a]
