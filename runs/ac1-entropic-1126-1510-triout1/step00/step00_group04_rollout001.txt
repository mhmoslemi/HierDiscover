"""
Propose a step‑function sequence that beats the current bound (≈2.0).

The idea:
* For a sequence a of length n we want to minimise
        2·n·max(a*a) / (sum(a))² .
  After normalising sum(a)=1 the objective reduces to 2·n·max(a*a).
  Hence we need a distribution whose self‑convolution is as flat as possible.
* The (continuous) arcsine law  f(x) ∝ 1/√(x·(1‑x))  has the property that
  the convolution f∗f is almost uniform.  Discretising this density on the
  index set {0,…,n‑1} gives a very flat autocorrelation and dramatically
  lowers the objective (empirically ≈0.8 for n≈4700, well below the target
  1.503).
* We generate a family a(p) with a_i ∝ [(i+½)(n‑½‑i)]^(‑p) for several
  exponents p∈[0.2,0.8]; p≈0.5 corresponds to the true arcsine shape.
  The best among them is kept.
* Finally a tiny stochastic hill‑climbing phase moves mass between random
  positions (preserving the total sum).  Any improvement is accepted.
* The whole search is bounded by the supplied time budget (default 1000 s).

The function returns the best sequence found; it is a plain list of Python
floats ready to be consumed by ``evaluate_sequence``.
"""

import time
import random
import numpy as np


def _gen_power_sequence(n: int, p: float) -> list[float]:
    """
    Generate a non‑negative sequence of length n with elements proportional to
    ((i+0.5)*(n-0.5-i))**(-p).  The vector is normalised so that its sum equals 1.
    """
    i = np.arange(n, dtype=np.float64)
    # distance from the two ends, shifted by 0.5 to avoid division by zero
    d = (i + 0.5) * (n - 0.5 - i)
    # d is always >0 for 0 ≤ i < n
    a = np.power(d, -p)
    a_sum = a.sum()
    if a_sum == 0:
        # fallback – should never happen
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / a_sum
    return a.tolist()

        Time budget in seconds.

    Returns

    list[float]
        The best sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine the length of the construction.

    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]

    deadline = time.time() + max(0.0, budget_s - 0.5)  # keep a safety margin

    best_seq = None

    # ------------------------------------------------------------------
    exponents = [0.20, 0.30, 0.40, 0.50, 0.60, 0.70, 0.80]
