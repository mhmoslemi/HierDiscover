"""
Propose a low‑score step function.

The algorithm consists of three stages:
1.  Generate a few diverse initial candidates (uniform, exponential,
    antisymmetric perturbations and the previous best if available) and keep
    the one with the lowest max‑self‑convolution.
2.  Run a long Adam‑style projected optimisation that directly minimizes a
    smooth approximation of  maxₖ (a*a)[k]  (log‑sum‑exp).  The central
    convolution entry (index n‑1) is given extra weight because it is often
    the dominant term.
3.  Finish with cheap random block constructions and a tiny pair‑swap hill‑climb.
All steps respect a 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_conv_fft(x: np.ndarray, y: np.ndarray, n2: int) -> np.ndarray:
    """Convolution of two real vectors using FFT, returning the full length."""
    fx = np.fft.rfft(x, n2)
    fy = np.fft.rfft(y, n2)
    conv = np.fft.irfft(fx * fy, n2)
    # keep only the true convolution length
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
def _max_conv(vec: np.ndarray, n2: int) -> float:
    """Maximum entry of the self‑convolution of vec."""
    conv = _full_conv_fft(vec, vec, n2)
    return float(conv.max())


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 30) -> np.ndarray:
    """Sub‑gradient of max conv based on the ``top_k`` largest entries."""
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad


# ----------------------------------------------------------------------

    """
    Search for a non‑negative step function minimizing the evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate wall‑clock budget (seconds).

    Returns
    -------
    list[float]
        The best sequence found (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # ----- problem size -------------------------------------------------
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # we enforce Σa = √(2 n)

    # ----- FFT buffer size (must cover length up to 3 n) ---------------
    n2 = 1 << ((3 * n - 2).bit_length())    # next power of two ≥ 3n‑2

    # ----- helper to evaluate a candidate (fast, no safety checks) -----
    def _evaluate(a: np.ndarray) -> float:
        return _max_conv(a, n2)

    # ----- stage 0: generate a few diversified starts -----------------
    candidates = []

    # uniform start
    uniform = np.full(n, target_sum / n, dtype=float)
    candidates.append(uniform)

    # exponential decay families
    for beta in (0.0005, 0.001, 0.002, 0.005, 0.01, 0.02):
        exp_vec = np.exp(-beta * np.arange(n, dtype=float))
        exp_vec *= target_sum / exp_vec.sum()
        candidates.append(_project_to_simplex(exp_vec, target_sum))

    # antisymmetric random perturbations
    for _ in range(12):
        v = np.random.randn(n)
        # enforce true antisymmetry: v_i = -v_{n-1-i}
        half = n // 2
        v[:half] = np.random.randn(half)
        if n % 2 == 1:
            v[half] = 0.0
        v[-half:] = -v[:half][::-1]
        a = target_sum / n + 0.3 * v  # base + perturb
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        candidates.append(a)

    # previous best if it exists
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = a_ref * (target_sum / max(a_ref.sum(), 1e-12))
        a_ref = _project_to_simplex(a_ref, target_sum)
        candidates.append(a_ref)

    # pick the best among the seeds
    best_a = None
    best_score = np.inf
    for a_cand in candidates:
        sc = _evaluate(a_cand)
        if sc < best_score:
            best_score = sc
            best_a = a_cand.copy()
    a = best_a.copy()
    print(f"[init] best seed score = {best_score:.6f}")

    # ----- stage 1: long Adam‑style projected optimisation -------------
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)              # first moment
    v_vec = np.zeros_like(a)          # second moment
    eta0 = target_sum * 0.5 / n        # base step size
    decay_eta = 5e-6
    mu = best_score                   # initial temperature for smooth max
    mu_decay = 0.9999
    mu_min = 1e-10
    central_weight = 2.0              # extra emphasis on the middle entry
    max_restarts = 5
    # split the remaining time: most for the gradient phase
    restart_interval = (deadline - time.time()) / (max_restarts + 1)
    next_restart = time.time() + restart_interval
    restarts_done = 0
    it = 0

    while time.time() < deadline - 30.0:   # keep ~30 s for final phases
        # self‑convolution (full) and current max
        conv = _full_conv_fft(a, a, n2)[: 2 * n - 1]
        cur_max = conv.max()

        # keep global best
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{it:08d}] new best = {best_score:.6f}")

        # compute gradient
        if mu > 0.02 * cur_max:
            diff = conv - cur_max                 # ≤ 0
            w = np.exp(diff / mu)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            # weighted convolution with reversed a
            conv_wr = _full_conv_fft(w, a[::-1], n2)
            # extract the n‑length slice that forms the gradient
            sum_term = conv_wr[n - 1: n - 1 + n]
            grad = 2.0 * sum_term
        else:
            # hard‑max sub‑gradient on a few active constraints
            grad = _gradient_topk(a, conv, top_k=30)

        # add extra central‑entry pressure (derivative of (a*a)[n‑1])
        grad += central_weight * (2.0 * a[::-1])

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad ** 2)
        it1 = it + 1
        m_hat = m / (1.0 - beta1 ** it1)
        v_hat = v_vec / (1.0 - beta2 ** it1)
        eta = eta0 / (1.0 + decay_eta * it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional tiny random shake to escape shallow minima
        if it % 4000 == 0 and it > 0:
            a_new += np.random.randn(n) * target_sum * 5e-4
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        mu = max(mu * mu_decay, mu_min)

        # restart handling
        if time.time() > next_restart and restarts_done < max_restarts:
            restarts_done += 1
            print(f"[{it:08d}] restart #{restarts_done}")
            # new random start (still respecting simplex)
            a = np.random.rand(n)
            a = _project_to_simplex(a, target_sum)
            m.fill(0.0)
            v_vec.fill(0.0)
            mu = _evaluate(a)
            next_restart = time.time() + restart_interval

        it += 1

    # ----- stage 2: extensive random block constructions ---------------
    block_trials = 8000
    for _ in range(block_trials):
        nb = np.random.randint(2, 21)                     # 2‑20 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _evaluate(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved block score = {best_score:.6f}")

    # ----- stage 3: fine pair‑swap hill‑climb -------------------------
    while time.time() < deadline:
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size == 0:
            break
        i = np.random.choice(nz)
        # with 30 % probability move mass towards the mirror position
        if np.random.rand() < 0.3:
            j = n - 1 - i
        else:
            j = np.random.randint(0, n)
        if i == j:
            continue
        delta = best_a[i] * np.random.uniform(0.1, 0.6)
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        sc = _evaluate(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[swap] new best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
