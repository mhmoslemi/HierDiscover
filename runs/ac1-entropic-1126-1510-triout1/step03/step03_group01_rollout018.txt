"""
Propose a low‑score step‑function.

The algorithm is a two‑phase optimisation on the probability simplex:

1.  Adam‑based minimisation of a smooth approximation of the peak of the
    self‑convolution (soft‑max surrogate).  The temperature α is slowly
    increased so that the optimisation first follows a gentle landscape and
    finally focuses on the true maximum.

2.  A short refinement using the exact ℓ∞‑objective (sub‑gradient descent
    on the true maximum) to polish the result.

Both phases use FFT‑based convolutions (O(n log n)) and project the iterate
back onto the simplex after each update.  Several initialisations (previous
solution, uniform, random, arcsine) and a few lengths around the historic
construction are tried within the allotted time (default 1000 s).  The best
sequence found (according to the black‑box `evaluate_sequence`) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # gradient = 2·rev(a) * w   (restricted to the n entries of a)
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max objective.
    Returns the best sequence encountered and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        lr = lr0 / np.sqrt(t)

        # update & project
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # tiny random jitter every few hundred iterations
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray) -> np.ndarray:
    """
    One sub‑gradient step for the true ℓ∞ objective.
    Uses a uniform weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # active set (allow a tiny absolute tolerance)
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short refinement using the exact max (sub‑gradient descent).
    Stops when time runs out.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a sequence of non‑negative heights with a low score.
    Returns a plain Python list (the sequence) that will be fed to
    `evaluate_sequence`.  The routine tries several lengths and initial
    strategies, using Adam on a smooth surrogate followed by a short exact
    max refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a second margin

    # ------------------------------------------------------------------
    # base sequence (if available) and candidate lengths
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 8000

    # explore a handful of lengths around the historic size
    lengths = [
        max(500, int(base_n * f))
        for f in (0.6, 0.8, 1.0, 1.2, 1.4)
        if int(base_n * f) >= 500
    ]
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # time budget per length (roughly equal split)
    time_per_length = max(2.0, (deadline - start_time) / len(lengths))

    for n in lengths:
        length_deadline = time.time() + time_per_length
        if time.time() >= deadline:
            break

        # initialise strategies
        init_strategies = ["prev", "arcsine", "uniform", "random"]
        for kind in init_strategies:
            if time.time() >= length_deadline:
                break

            # ------------------------------------------------------------------
            # initialise a on the simplex
            if kind == "prev" and base_seq is not None:
                a0 = _interpolate_to_length(base_seq, n)
            elif kind == "arcsine":
                a0 = _arcsine_distribution(n)
            elif kind == "uniform":
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            elif kind == "random":
                a0 = np.random.rand(n)
                a0 /= a0.sum()
            else:               # fallback
                a0 = np.full(n, 1.0 / n, dtype=np.float64)

            a0 = _project_onto_simplex(a0, s=1.0)

            # ------------------------------------------------------------------
            # phase‑1 : Adam on smooth max
            phase1_deadline = min(length_deadline, time.time() + (length_deadline - time.time()) * 0.6)
            a_opt, peak = _adam_optimize(a0, phase1_deadline)

            # ------------------------------------------------------------------
            # phase‑2 : short exact‑max refinement
            phase2_deadline = min(length_deadline, time.time() + (length_deadline - time.time()) * 0.4)
            a_refined = _refine_max(a_opt, phase2_deadline)

            # ------------------------------------------------------------------
            # evaluate candidate (fast approximate score first)
            fast_peak = _fft_self_convolve(a_refined).max()
            fast_score = 2.0 * n * fast_peak
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = a_refined.copy()
                print(f"[{time.time():.1f}] len={n} init={kind} "
                      f"fast_score={fast_score:.6f}")

            # final true evaluation (the black‑box is cheap relative to time budget)
            try:
                true_score = evaluate_sequence(overall_best_seq.tolist())
                if true_score < overall_best_score:
                    overall_best_score = true_score
                    overall_best_seq = np.asarray(overall_best_seq, dtype=np.float64)
                    print(f"*** New true best: len={n} init={kind} score={true_score:.6f}")
            except Exception as exc:
                # fallback – keep the fast estimate
                print("evaluate_sequence raised:", exc)

            # early exit if we already beat the target
            if overall_best_score <= 1.503:
                break

        if overall_best_score <= 1.503:
            break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # fallback in the extremely unlikely case nothing was produced
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback uniform used.")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # convert to plain Python list as required by the harness
    return overall_best_seq.tolist()
