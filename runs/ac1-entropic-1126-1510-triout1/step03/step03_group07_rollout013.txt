"""
Propose a step‑function sequence that minimizes the evaluator’s
score.  The algorithm works in three stages:

1. **Initialization** – if a previous best sequence (`height_sequence_1`)
   is available it is scaled to the required ℓ¹‑norm `S = √(2·N)`;
   otherwise a uniform random start is used.

2. **Fast gradient descent** – a few hundred iterations of the
   soft‑max approximation (the same idea as the reference code) bring
   the solution into a good basin.  Only FFT‑based convolutions are
   used, so the cost remains modest.

3. **Simulated‑annealing polishing** – pairwise “mass‑move’’ updates are
   applied directly to the convolution using an O(N) formula
   (no FFT needed).  The move conserves the ℓ¹‑norm and respects the
   non‑negativity constraints.  An exponential temperature schedule
   allows occasional uphill steps, helping to escape local minima.
   The best sequence seen throughout the run is kept.

The whole routine respects the time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the non‑negative simplex with prescribed sum.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (used only in the gradient phase).
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based valid‑mode correlation (used only in the gradient phase).
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    a_rev = a[::-1]
    L = w.shape[0]                     # = 2n‑1
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length n (valid correlation)


# ----------------------------------------------------------------------
# O(N) update of a*a when a mass δ is moved from src→dst.
# ----------------------------------------------------------------------
def _move_mass(a: np.ndarray, conv: np.ndarray, src: int, dst: int, delta: float) -> None:
    """
    Transfer `delta` mass from a[src] to a[dst].
    The function updates `a` and the full autocorrelation `conv`
    in‑place using the exact algebraic formula:
        conv ← conv + 2·δ·(shift(a, dst) – shift(a, src))
               + δ²·(e_{2dst}+e_{2src}−2e_{src+dst})
    where `shift(a, k)` is a copy of `a` placed starting at index `k`.
    """
    N = a.shape[0]

    # slice updates (use the *old* a – a is unchanged up to this point)
    conv[dst:dst + N] += 2.0 * delta * a
    conv[src:src + N] -= 2.0 * delta * a

    # the δ² impulse terms
    d2 = delta * delta
    conv[2 * dst] += d2
    conv[2 * src] += d2
    conv[src + dst] -= 2.0 * d2

    # finally modify the vector a
    a[src] -= delta
    a[dst] += delta


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a non‑negative sequence (list of floats) that tries to minimise
    `evaluate_sequence`.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # 1️⃣ Initialise a feasible sequence.
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = None

    if base is not None and base.size > 0:
        N = base.shape[0]
        S = math.sqrt(2.0 * N)
        sum_base = base.sum()
        if sum_base <= 0:
            a = np.full(N, S / N, dtype=float)
        else:
            a = base * (S / sum_base)      # rescale to the correct ℓ¹‑norm
        a = _project_onto_simplex(a, S)    # guarantee exact feasibility
    else:
        # fallback: uniform start of a moderate size
        N = 12000
        S = math.sqrt(2.0 * N)
        a = np.full(N, S / N, dtype=float)
        a += 0.001 * S * np.random.rand(N) / N
        a = _project_onto_simplex(a, S)

    # initial convolution & score (with the chosen normalisation the score == max conv)
    conv = _fft_convolve(a)
    cur_score = float(conv.max())
    best_score = cur_score
    best_seq = a.copy()
    print(f"[Init] N={N}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Gradient‑descent warm‑up (few hundred iterations only).
    # --------------------------------------------------------------
    max_grad_iters = 1200                     # limited – cheap compared to SA
    beta_start, beta_end = 0.5, 40.0
    lr_start, lr_end = 0.8, 0.02

    for it in range(1, max_grad_iters + 1):
        if time.time() >= deadline:
            break
        # full convolution
        conv = _fft_convolve(a)
        max_conv = conv.max()

        # schedule
        t_frac = it / max_grad_iters
        beta = beta_start + (beta_end - beta_start) * t_frac
        lr = lr_start + (lr_end - lr_start) * t_frac

        # soft‑max weights (stable formulation)
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth max
        grad = 2.0 * _fft_correlate(w, a)

        # projected gradient step on the simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, S)

        # evaluate new point
        conv_new = _fft_convolve(a_new)
        score_new = float(conv_new.max())
        if score_new < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[Grad {it:05d}] new best {best_score:.6f}")

        a = a_new

    # after the gradient phase clamp variables
    conv = _fft_convolve(a)
    cur_score = float(conv.max())
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()
        print(f"[Post‑Grad] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Simulated‑annealing polishing using O(N) mass moves.
    # --------------------------------------------------------------
    # Temperature & move‑size schedules
    T = max(0.02 * cur_score, 1e-4)      # initial temperature (relative to score)
    T_min = 1e-8
    sigma_frac = 0.5                     # start with up‑to‑50 % of current source mass
    sigma_end = 0.001                    # final move‑size fraction
    cooling_rate = 0.9995                # exponential cooling factor

    # main SA loop – runs until time runs out
    it = 0
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # draw a random move (source → destination)
        # ----------------------------------------------------------
        src = np.random.randint(N)
        dst = np.random.randint(N - 1)
        if dst >= src:
            dst += 1          # ensure src ≠ dst

        if a[src] <= 0.0:
            continue          # nothing to move

        # maximum transferable mass this iteration
        max_delta = sigma_frac * S
        delta = min(a[src], max_delta) * np.random.rand()
        if delta <= 0.0:
            continue

        # ----------------------------------------------------------
        # apply move and compute new score
        # ----------------------------------------------------------
        _move_mass(a, conv, src, dst, delta)
        new_score = float(conv.max())

        # ----------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            # uphill move – accept with probability exp(-(Δ)/T)
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[SA {it:07d}] new best {best_score:.6f}")
        else:
            # revert the move (swap src ↔ dst)
            _move_mass(a, conv, dst, src, delta)

        # ----------------------------------------------------------
        # cooling
        # ----------------------------------------------------------
        T = max(T * cooling_rate, T_min)
        sigma_frac = max(sigma_frac * cooling_rate, sigma_end)

        # early stop if we have already reached the target bound
        if best_score <= 1.503:
            print(f"[Early stop] target reached at iteration {it}")
            break

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # safeguard against tiny negatives
    return best_seq.tolist()
