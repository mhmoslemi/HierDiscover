"""
Hybrid linear‑programming + local refinement for the step‑function bound.

The algorithm proceeds in three stages:

1. **LP‑driven descent**  
   Starting from a feasible non‑negative vector `a` with
   Σa = √(2 n), we repeatedly
   - compute M = max(a∗a),
   - solve the LP   maximise Σb_j  subject to (a∗b)_k ≤ M  (k=0…2n‑2), b≥0,
   - rescale the LP solution to keep the ℓ¹‑norm equal to √(2 n),
   - replace `a` by the convex combination of `a` and the rescaled `b`
     that yields the smallest new max‑convolution (a few trial mixing ratios).

   The LP is built once per iteration using a sparse Toeplitz matrix;
   for n≈2000 this is fast and memory‑friendly.

2. **Block‑search**  
   Random piece‑wise‑constant patterns (2‑12 blocks) are projected onto
   the simplex and kept if they improve the maximal auto‑convolution.

3. **Tiny hill‑climb**  
   Repeatedly move a random fraction of mass from one index to another,
   project back onto the simplex and accept improvements.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found.  All entries are non‑negative, bounded
by 1000, and Σa = √(2 n) – consequently the score returned by
`evaluate_sequence` equals the maximal self‑convolution of the returned
vector.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv(vec: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution of vec."""
    n = vec.shape[0]
    size = 1 << (2 * n - 1).bit_length()          # next power of two
    f = np.fft.rfft(vec, size)
    conv = np.fft.irfft(f * f, size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _build_toeplitz_constraint(indices_i: np.ndarray,
                               indices_j: np.ndarray,
                               a: np.ndarray,
                               n: int):
    """
    Fast construction of the sparse matrix A where
    A[k, j] = a[i] with k = i + j   (i = indices_i, j = indices_j).
    """
    # Data for the non‑zero entries
    data = a[indices_i]
    rows = indices_i + indices_j
    from scipy.sparse import coo_matrix
    return coo_matrix((data, (rows, indices_j)), shape=(2 * n - 1, n))

    Search for a non‑negative step function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Wall‑clock budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """

    try:

        from scipy.sparse import coo_matrix
    except Exception as e:

            best_a = cand.copy()
            print(f"[block] improved M={best_M:.6f}")

    while time.time() < deadline:

            best_M = M_cand
            best_a = cand
            print(f"[swap] improved M={best_M:.6f}")

    print(f"[final] best max‑conv = {best_M:.6f}  (score ≈ {best_M:.6f})")
    return best_a.tolist()
