"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises

    evaluate_sequence(a) = max_k (a * a)[k]

under the normalisation Σ a_i = √(2 n) (the function itself returns the
maximum of the self‑convolution after this normalisation).

The algorithm consists of three stages:

1. **Broad parametric sweep** – many simple families (inverse‑sqrt,
   double‑ended power‑law, exponential, cosine, beta‑distribution,
   random block) are sampled.  Every candidate is projected onto the
   simplex {a ≥ 0, Σa = √(2 n)} and its maximal self‑convolution is
   evaluated quickly via FFT.  The best candidate is kept.

2. **Smooth‑max Adam optimisation** – starting from the best shape we
   minimise a smooth approximation of the max:

        L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

   Its gradient is
        ∇L_α = 2·corr(softmax_α(a*a), a)

   where `corr` is the linear cross‑correlation.
   Adam updates are performed on this gradient, each step being projected
   back onto the simplex.  The smoothing parameter α is increased slowly
   (5 → 20) so the optimiser first sees a softened landscape and later
   focuses on the true maximum.  Random perturbations are injected when
   progress stalls.

3. **Final random‑block refinement** – a handful of fresh block‑structured
   candidates are evaluated; any improvement replaces the current best.

The whole search respects a 1000 s budget, prints intermediate scores,
and finally returns the best sequence as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy import fft

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v via FFT."""
    fu = fft.rfft(u, n=size_fft)
    fv = fft.rfft(v, n=size_fft)
    conv = fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _cross_corr_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Linear cross‑correlation (u ⋆ v)[t] = Σ_i u[i]·v[i+t],
    returned for t = 0 … len(u)+len(v)-2.
    """
    fu = fft.rfft(u, n=size_fft)
    fv = fft.rfft(v, n=size_fft)
    corr = fft.irfft(fu * np.conj(fv), n=size_fft)
    return corr[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _score(a: np.ndarray, size_fft: int) -> float:
    """Return max entry of the self‑convolution (the evaluate_sequence score)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray,
                         conv: np.ndarray,
                         size_fft: int,
                         alpha: float) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·conv) w.r.t. a.
    Using the identity:
        ∂L/∂a = 2·corr(softmax_α(conv), a) .
    """
    # softmax over the convolution entries (avoid overflow)
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    w = exp_term / exp_term.sum()                     # shape (2n‑1,)
    # cross‑correlation w ⋆ a  (valid for shifts 0 … n‑1)
    corr = _cross_corr_fft(w, a, size_fft)
    grad = 2.0 * corr[: a.shape[0]]
    return grad

    deadline = time.time() + budget_s - 0.5   # small safety margin
