import numpy as np, math, time

def _project_to_simplex(v, z):
    ...

def _conv_fft(u, v, size_fft):
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[:len(u) + len(v) - 1]

def _self_convolution(a, size_fft):
    return _conv_fft(a, a, size_fft)

def _weighted_gradient(a, w, size_fft):
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)
    start = len(w) - len(a)
    return 2.0 * conv[start:start+len(a)]

def _softmax_weights(conv, T):
    shifted = conv - conv.max()
    exp_arg = shifted / max(T, 1e-12)
    exp_arg = np.clip(exp_arg, -50, 0)
    w = np.exp(exp_arg)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / w.size
    else:
        w /= s
    return w

def _hard_weights(conv, eps=1e-12):
    maxv = conv.max()
    mask = np.abs(conv - maxv) <= eps
    count = mask.sum()
    if count == 0:
        # fallback to uniform
        w = np.full_like(conv, 1.0/conv.size)
    else:
        w = mask.astype(float) / count
    return w

def _optimize(a0, deadline):
    n = a0.shape[0]
    target_sum = np.sqrt(2.0*n)
    size_fft = 1 << ((3*n - 2).bit_length())
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0*n*_self_convolution(a,size_fft).max()/(target_sum**2) # but simpler use conv max
    # better compute M directly:
    conv = _self_convolution(a,size_fft)
    best_M = conv.max()
    best_score = 2.0*n*best_M/(target_sum**2)

    # Adam states
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    # Temperature schedule
    # initial T as half of range of conv
    conv = _self_convolution(a,size_fft)
    T_init = max(1e-4, (conv.max() - conv.min()) * 0.5)
    T = T_init
    # Use parameters
    eta_base = target_sum * 0.5 / n
    # thresholds and schedules
    warmup_iters = 200_000  # number of iters using softmax
    hard_start = warmup_iters
    # loop
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = _self_convolution(a,size_fft)
        M = conv.max()
        # keep best
        if M < best_M:
            best_M = M
            best_a = a.copy()
            # print progress
            if it % 5000 == 0:
                print(f"[opt] it {it:,} new best M={best_M:.6f}")
        # compute weights
        if it <= warmup_iters:
            w = _softmax_weights(conv, T)
            # cool temperature
            T *= 0.9999
        else:
            w = _hard_weights(conv, eps=1e-12)

        # compute gradient
        grad = _weighted_gradient(a, w, size_fft)

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = eta_base * m_hat / (np.sqrt(v_hat) + eps)  # moving along gradient

        # descent: a = a - step (since grad points direction of increase)
        a_new = a - step
        # projection to simplex
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick
        if it % 10000 == 0:
            noise = np.random.randn(n) * target_sum * 1e-3
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new

    # Return best found
    return best_a, best_M

"""
Propose a step‑function sequence with a very low self‑convolution peak.

Algorithm summary
-----------------
1.  Determine problem size (n) and the normalisation target  Σa = √(2 n).
2.  Generate a modest set of random starters (Dirichlet draws) plus the
    previously discovered best sequence (if available).  Keep the one
    with the smallest initial peak of a*a.
3.  Run a long projected‐gradient descent on the chosen starter.
    * A soft‑max (temperature) approximation of the max is used for the
      first part of the run – this provides a smooth gradient.
    * After a warm‑up phase the temperature is driven to zero and a hard
      indicator on the maximal convolution index is used (true sub‑gradient).
    * An Adam‑style adaptive step is employed and after each step the
      vector is projected back onto the simplex {a≥0, Σa = √(2 n)}.
    * Occasional random kicks help escape plateaus.
4.  The best vector encountered before the 1000 s budget expires is returned.

The optimisation works directly on the objective
    score = max_k (a*a)_k   (because Σa = √(2 n) is fixed)
so a lower peak of the self‑convolution equals a lower evaluation score.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution of u and v using a power‑of‑two FFT."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a * a (full length)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. a   (weights w sum to 1).
    The exact expression is   2·(w ↔ a_rev)   restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)   # length = 3·n‑2
    start = len(w) - len(a)                # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, T: float) -> np.ndarray:
    """Soft‑max on the convolution vector (T = temperature)."""
    shifted = conv - conv.max()                 # ≤ 0
    exp_arg = shifted / max(T, 1e-12)
    exp_arg = np.clip(exp_arg, -50, 0)          # prevent overflow
    w = np.exp(exp_arg)
    s = w.sum()
    if s == 0:
        w[:] = 1.0 / w.size
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
def _hard_weights(conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Indicator weights on the (approximately) maximal convolution entries.
    Used once the temperature has been driven to zero.
    """
    m = conv.max()
    mask = np.abs(conv - m) <= eps
    cnt = mask.sum()
    if cnt == 0:
        # fallback – use uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    w = mask.astype(float) / cnt
    return w


# ----------------------------------------------------------------------
def _optimize(a0: np.ndarray,
              deadline: float,
              max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Long‑run projected Adam optimisation of max(a*a) under the simplex
    constraint Σa = √(2 n).  Returns the best vector seen and its peak.
    """
    n = a0.shape[0]
    target_sum = math.sqrt(2.0 * n)
    size_fft = 1 << ((3 * n - 2).bit_length())   # power of two ≥ 3n‑2

    a = a0.copy()
    best_a = a.copy()
    best_peak = _self_convolution(a, size_fft).max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # learning‑rate scaling (roughly η ≈ 0.5·target_sum / n)
    eta0 = target_sum * 0.5 / n

    # temperature schedule (soft‑max → hard)
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    warmup_iters = 200_000       # soft‑max phase
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # --- current convolution & peak ------------------------------------------------
        conv = _self_convolution(a, size_fft)
        peak = conv.max()

        if peak < best_peak:
            best_peak = peak
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[opt] it {it:,} new best peak = {best_peak:.6f}")

        # --- weights (smooth → hard) ----------------------------------------------------
        if it <= warmup_iters:
            w = _softmax_weights(conv, T)
            T *= 0.9999            # geometric cooling
        else:
            w = _hard_weights(conv)

        # --- gradient ------------------------------------------------------------------
        grad = _weighted_gradient(a, w, size_fft)

        # --- Adam update ---------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)   # move opposite to grad
        a_new = a - step

        # --- simplex projection ---------------------------------------------------------
        a_new = _project_to_simplex(a_new, target_sum)

        # --- occasional random kick ----------------------------------------------------
        if it % 10_000 == 0:
            noise = np.random.randn(n) * target_sum * 1e-3
            a_new += noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new

    return best_a, best_peak
