def adam_optimize(a, size_fft, target_sum, deadline, max_iters=500000, lr0=0.3, beta1=0.9, beta2=0.999):
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_local_a = a.copy()
    best_local_score = evaluate(a)
    it = 0
    no_improve = 0
    while it < max_iters and time.time() < deadline:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # active indices
        eps = max(1e-12, cur_score * 1e-9)
        max_idxs = np.where(conv >= cur_score - eps)[0]

        grad = _gradient_at_max(a, max_idxs)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / (1.0 + 1e-6 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # random perturbation if stuck
        if (it % 5000 == 0 and it > 0) or no_improve > 2000:
            noise = np.random.randn(n) * target_sum * 2e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0

        a = a_new
        it += 1

    return best_local_a, best_local_score
```

def local_annealing(a_best, size_fft, target_sum, deadline):
    a = a_best.copy()
    best_score = evaluate(a)
    best_a = a.copy()
    n = len(a)
    T0 = 1.0
    T_decay = 0.9999
    temperature = T0
    while time.time() < deadline:
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # choose epsilon relative to a[i]
        max_move = a[i] * 0.5
        if max_move <= 0:
            continue
        epsilon = np.random.rand() * max_move
        a_new = a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon
        # Ensure non-neg, sum unchanged
        cur_score = _max_convolution(a_new, size_fft)
        if cur_score < best_score or np.random.rand() < np.exp(-(cur_score - best_score) / max(temperature, 1e-12)):
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
        # decay temperature
        temperature *= T_decay
        # occasionally reset temperature to avoid stagnation
        if temperature < 1e-4:
            temperature = T0
    return best_a, best_score

   Σa = √(2 n) (this makes evaluate_sequence() equal to the maximal
   entry of the self‑convolution).

2. *Parametric search* – generate thousands of candidates from simple
   families (single‑ended inverse‑sqrt, power‑law, exponential,
   polynomial decay, logistic, random block and Dirichlet vectors).
   Keep the best few.

3. *ADAM refinement* – starting from each of the top‑K candidates run a
   projected ADAM optimiser that minimises the true max‑convolution.
   The sub‑gradient at the active indices is computed analytically.
   A small amount of random noise is injected when progress stalls.

4. *Local simulated annealing* – from the best solution perform cheap
   swaps of a random amount of mass between two positions.  Accept
   improvements always and occasional uphill moves with a decaying
   temperature.

The whole search respects a 1000 s budget and prints intermediate scores.
The returned list of floats is the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Maximum entry of the linear self‑convolution."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _gradient_at_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (average) max‑convolution w.r.t. a.
    Implements  grad_i = (2/|max|) Σ_{k∈max} a_{k−i}
    (only terms with valid indices are summed).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    if max_idxs.size == 0:
        return grad
    weight = 1.0 / max_idxs.size
    for k in max_idxs:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        i_range = np.arange(i0, i1 + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    grad *= weight
    return grad


# ----------------------------------------------------------------------
def _evaluate(a: np.ndarray, size_fft: int) -> float:
    """Convenient wrapper returning the max‑convolution value."""
    return _max_convolution(a, size_fft)


# ----------------------------------------------------------------------
def _candidate_from_array(arr: np.ndarray, target_sum: float) -> np.ndarray:
    """Project raw array onto the simplex with the required sum."""
    arr = np.maximum(arr, 0.0)
    return _project_to_simplex(arr, target_sum)


# ----------------------------------------------------------------------
def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Generate a random point on the simplex (Dirichlet(alpha))."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _generate_random_block(n: int,
                           min_blocks: int = 2,
                           max_blocks: int = 6) -> np.ndarray:
    """Create a random piecewise‑constant step function."""
    nb = np.random.randint(min_blocks, max_blocks + 1)
    cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = np.random.rand(nb) + 0.1      # avoid exact zeros
    seq = np.empty(n, dtype=float)
    for blk, v in zip(blocks, vals):
        seq[blk] = v
    return seq


# ----------------------------------------------------------------------
def _adam_refine(start_a: np.ndarray,
                 size_fft: int,
                 target_sum: float,
                 deadline: float,
                 max_iters: int = 400_000,
                 lr0: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected ADAM optimiser that minimises the true maximal self‑convolution.
    Returns the best vector encountered and its score.
    """
    a = start_a.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _evaluate(a, size_fft)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    it = 0
    no_improve = 0

    while it < max_iters and time.time() < deadline:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # active set (numerical tolerance to avoid jitter)
        eps = max(1e-12, cur_score * 1e-9)
        max_idxs = np.where(conv >= cur_score - eps)[0]

        grad = _gradient_at_max(a, max_idxs)

        # ADAM moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # slowly decaying learning rate
        lr = lr0 / (1.0 + 1e-6 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick
        if (it % 5000 == 0 and it > 0) or no_improve > 3000:
            noise = np.random.randn(n) * target_sum * 3e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         size_fft: int,
                         target_sum: float,
                         deadline: float,
                         max_moves: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Simple annealing that swaps a random amount of mass between two positions.
    Accepts worse moves with a temperature that decays exponentially.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _evaluate(a, size_fft)

    n = a.shape[0]
    temperature = 1.0
    decay = 0.9999

    move = 0
    while move < max_moves and time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue
        # move at most half of a[i] to keep non‑negativity
        eps = np.random.rand() * (0.5 * a[i])
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        cur_score = _evaluate(a_new, size_fft)

        # Metropolis acceptance
        if cur_score < best_score or np.random.rand() < math.exp(-(cur_score - best_score) / max(temperature, 1e-12)):
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()

        temperature *= decay
        if temperature < 1e-4:
            temperature = 1.0  # occasional reheating
        move += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score non‑negative step function."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 5000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)
    size_fft = 1 << ((2 * n - 1).bit_length())   # >= 2n-1

    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    def _maybe_update(candidate: np.ndarray, name: str = ""):
        nonlocal best_a, best_score
        sc = _evaluate(candidate, size_fft)
        if sc < best_score - 1e-12:
            best_a, best_score = candidate.copy(), sc
            print(f"[{name}] new best = {sc:.6f}")
        return sc

    # --------------------------------------------------------------
    # 1. Baselines
    uniform = np.full(n, target_sum / n)
    _maybe_update(uniform, "uniform")

    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)
        ref = _candidate_from_array(ref, target_sum)
        _maybe_update(ref, "reference")

    # --------------------------------------------------------------
    # 2. Random parametric families
    N_SAMPLES = 800   # per family – fast enough

    # single‑ended inverse‑sqrt
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 20.0
        raw = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        cand = _candidate_from_array(raw, target_sum)
        _maybe_update(cand, "inv_sqrt")

    # single‑ended power‑law
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 20.0
        p = np.random.rand() * 3.5 + 0.05   # exponent in (0.05, 3.55)
        raw = 1.0 / (np.arange(n, dtype=float) + beta + 1.0) ** p
        cand = _candidate_from_array(raw, target_sum)
        _maybe_update(cand, "pow_law")

    # single‑ended exponential decay
    for _ in range(N_SAMPLES):
        decay = np.random.rand() * 0.025 + 0.0003   # [3e‑4, 2.8e‑2]
        raw = np.exp(-decay * np.arange(n, dtype=float))
        cand = _candidate_from_array(raw, target_sum)
        _maybe_update(cand, "exp")

    # polynomial‑decay ( (1 - (i/(n-1))^p)^q )
    for _ in range(N_SAMPLES):
        p = np.random.rand() * 4.0 + 0.2
        q = np.random.rand() * 5.0 + 0.2
        t = (np.arange(n, dtype=float) / (n - 1)) ** p
        raw = np.maximum(0.0, (1.0 - t) ** q)
        cand = _candidate_from_array(raw, target_sum)
        _maybe_update(cand, "poly")

    # logistic‑type decreasing step
    for _ in range(N_SAMPLES):
        centre = np.random.rand() * n * 0.3          # centre near left side
        scale = np.random.rand() * n * 0.05 + 1.0
        raw = 1.0 / (1.0 + np.exp((np.arange(n, dtype=float) - centre) / scale))
        cand = _candidate_from_array(raw, target_sum)
        _maybe_update(cand, "logistic")

    # random Dirichlet vectors (highly stochastic)
    for _ in range(N_SAMPLES):
        raw = _random_dirichlet(n, alpha=np.random.rand() * 5.0 + 0.1)
        cand = _candidate_from_array(raw, target_sum)
        _maybe_update(cand, "dirichlet")

    # random block structures (asymmetric)
    for _ in range(N_SAMPLES):
        raw = _generate_random_block(n, min_blocks=2, max_blocks=6)
        cand = _candidate_from_array(raw, target_sum)
        _maybe_update(cand, "block")

    print(f"[shape] best score after families = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. ADAM refinement – run on the top few distinct seeds
    # Collect a few promising seeds (different shapes) for refinement.
    # Here we simply reuse the best one plus a handful of random
    # candidates generated anew.
    refinement_seeds = [best_a.copy()]  # best so far

    # add a few random block candidates as extra starting points
    for _ in range(4):
        raw = _generate_random_block(n, min_blocks=2, max_blocks=6)
        refinement_seeds.append(_candidate_from_array(raw, target_sum))

    # limit total ADAM time to roughly half of the remaining budget
    remaining_time = deadline - time.time()
    adam_time_budget = remaining_time * 0.55
    per_refine_budget = adam_time_budget / len(refinement_seeds)

    for idx, seed_vec in enumerate(refinement_seeds):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_refine_budget
        refined, refined_score = _adam_refine(seed_vec,
                                              size_fft,
                                              target_sum,
                                              sub_deadline,
                                              max_iters=300_000,
                                              lr0=0.22)
        if refined_score < best_score - 1e-12:
            best_a, best_score = refined, refined_score
            print(f"[adam-{idx}] improved to {best_score:.6f}")

    print(f"[adam] best score after refinement = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final local simulated annealing
    if time.time() < deadline:
        final_a, final_score = _simulated_annealing(best_a,
                                                    size_fft,
                                                    target_sum,
                                                    deadline,
                                                    max_moves=25_000)
        if final_score < best_score - 1e-12:
            best_a, best_score = final_a, final_score
            print(f"[anneal] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")

    # Convert to plain Python list before returning
    return best_a.tolist()
