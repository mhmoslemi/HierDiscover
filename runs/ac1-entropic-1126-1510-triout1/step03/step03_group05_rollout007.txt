def _lp_refine(a, target_sum, size_fft, max_iters=5, init_top_k=5, tol=1e-8):
    n = len(a)
    active = set()
    # initial active set: top_k indices
    conv = _self_convolution(a, size_fft)
    cur_max = conv.max()
    initial_idxs = np.argpartition(conv, -init_top_k)[-init_top_k:]
    active.update(int(i) for i in initial_idxs)
    for iteration in range(max_iters):
        # build constraints for all in active
        rows = []
        rhs = []
        for k in active:
            w = np.zeros(n)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_range = np.arange(i_min, i_max+1, dtype=int)
            w[i_range] = a[k - i_range]
            rows.append(np.concatenate([w, [-0.5]]))
            rhs.append(conv[k] / 2.0)
        A_ub = np.vstack(rows) if rows else None
        b_ub = np.array(rhs)
        A_eq = np.concatenate([np.ones(n), [0.0]]).reshape(1, n+1)
        b_eq = np.array([target_sum])
        bounds = [(0.0, None)] * n + [(0.0, None)]
        c = np.concatenate([np.zeros(n), [1.0]])  # minimize t
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                    bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {iteration} failed: {res.message}")
            break
        a_new = res.x[:n]
        t_new = res.x[-1]
        # recompute convolution
        conv_new = _self_convolution(a_new, size_fft)
        max_new = conv_new.max()
        # evaluate improvement relative to current max
        if max_new + tol < cur_max:  # improvement
            a = a_new
            cur_max = max_new
            conv = conv_new
            # update active set based on new max
            new_active = set(np.where(conv_new >= max_new - tol * max_new)[0])
            # combine with existing, limit size
            combined = active.union(new_active)
            if len(combined) > 30:
                # keep only top 30 by conv value
                top = np.argpartition(conv_new, -30)[-30:]
                combined = set(int(i) for i in top)
            active = combined
        else:
            # no significant improvement, break
            break
    return a

Propose a non‑negative step‑function (list of heights) minimising
``evaluate_sequence`` (i.e. the maximal entry of the self‑convolution
under the L1‑norm Σa = √(2 n)).  

The algorithm consists of three stages:

1. **Extensive parametric sampling** – many families (inverse‑sqrt,
   double‑ended power‑law, exponential, cosine, random blocks, sparse
   spikes) are sampled, projected onto the simplex and evaluated.
   The best candidate is kept.

2. **Projected Adam optimisation** – starting from the best shape we
   perform Adam updates on the actual max‑convolution (sub‑gradient
   at the active indices) with Euclidean projection onto the simplex.
   Random perturbations are injected when progress stalls.

3. **Active‑set linear‑program refinement** – using the current
   active (maximal) convolution indices we build a linearisation of the
   quadratic constraints and solve a tiny LP (via ``scipy.optimize.linprog``)
   that directly lowers the bound on the maximum.  The LP is solved a
   few times, each time enlarging the active set if the linearisation
   was not tight.

Finally a short random‑block sweep is performed and the best sequence
found is returned.
"""

import time
import math
import numpy as np
from typing import List

# optional import – the LP refinement will be skipped if unavailable
try:
    from scipy.optimize import linprog
except Exception:   # pragma: no cover
    linprog = None


def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0.0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of two vectors using the FFT."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


def _gradient_at_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entries.
    Implements  grad_i = (2/|max|) Σ_{k∈max} a_{k−i}  (ignoring out‑of‑range).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    if max_idxs.size == 0:
        return grad
    weight = 1.0 / max_idxs.size
    for k in max_idxs:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    grad *= weight
    return grad


def _score(a: np.ndarray, size_fft: int) -> float:
    """Return the maximal entry of the self‑convolution (the core score)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


def _lp_refine(a: np.ndarray,
               target_sum: float,
               size_fft: int,
               max_iters: int = 5,
               init_top_k: int = 5,
               tol: float = 1e-8) -> np.ndarray:
    """
    Active‑set linear‑program refinement.
    Starting from ``a``, repeatedly solves a linearisation of the
    max‑convolution constraints and updates ``a`` if an improvement is
    found.
    """
    if linprog is None:                     # LP not available – skip
        return a

    n = a.shape[0]
    conv = _self_convolution(a, size_fft)
    cur_max = conv.max()

    # initialise active set with the indices attaining (near) the maximum
    active = set(int(i) for i in np.argpartition(conv, -init_top_k)[-init_top_k:])

    for it in range(max_iters):
        # ----- build LP matrices -----
        rows = []
        rhs = []
        for k in active:
            # w_i = a_{k-i} (zero outside the range)
            w = np.zeros(n, dtype=float)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            i_range = np.arange(i_min, i_max + 1, dtype=int)
            w[i_range] = a[k - i_range]
            rows.append(np.concatenate([w, [-0.5]]))      # w·a – 0.5·t ≤ c_k/2
            rhs.append(conv[k] / 2.0)                    # c_k/2

        A_ub = np.vstack(rows) if rows else None
        b_ub = np.array(rhs)
        A_eq = np.concatenate([np.ones(n), [0.0]]).reshape(1, n + 1)
        b_eq = np.array([target_sum])

        bounds = [(0.0, None)] * n + [(0.0, None)]          # a_i ≥0 , t ≥0
        c = np.concatenate([np.zeros(n), [1.0]])          # minimise t

        # ----- solve the LP -----
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            # if LP fails we stop refining
            break

        a_new = res.x[:n]
        t_new = res.x[-1]

        # evaluate the true convolution of the candidate
        conv_new = _self_convolution(a_new, size_fft)
        max_new = conv_new.max()

        # improvement test
        if max_new + tol < cur_max:
            a = a_new
            conv = conv_new
            cur_max = max_new

            # enlarge active set with any newly tight constraints
            new_active = set(int(i) for i in np.where(conv >= max_new - tol)[0])
            active.update(new_active)
            # keep the active set reasonably small (top 30)
            if len(active) > 30:
                top = np.argpartition(conv, -30)[-30:]
                active = set(int(i) for i in top)
        else:
            # no further progress – stop
            break
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """Search for a low‑score non‑negative step function."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)    # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3130                      # fallback length
        have_ref = False

    target_sum = math.sqrt(2.0 * n)               # Σa = √(2 n)
    size_fft = 1 << ((2 * n - 2).bit_length())   # power‑of‑2 ≥ 2 n‑1

    # ------------------------------------------------------------------
    # best candidate tracker
    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # helper for fast evaluation (max‑conv only)
    def _eval(arr: np.ndarray) -> float:
        return _score(arr, size_fft)

    # ------------------------------------------------------------------
    # 1. EXTENSIVE PARAMETRIC SHAPE SEARCH
    # --------------------------------------------------------------
    # uniform baseline
    a = np.ones(n, dtype=float)
    a = _project_to_simplex(a, target_sum)
    sc = _eval(a)
    best_a, best_score = a.copy(), sc
    print(f"[shape] uniform score = {sc:.6f}")

    # reference if available
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)
        ref = np.maximum(ref, 0.0)
        ref = _project_to_simplex(ref, target_sum)
        sc = _eval(ref)
        if sc < best_score:
            best_a, best_score = ref.copy(), sc
        print(f"[shape] reference score = {sc:.6f}")

    # families – increase samples relative to the previous script
    SAMPLES_PER_FAMILY = 200

    # ----- inverse‑sqrt (single‑ended) -----
    for _ in range(SAMPLES_PER_FAMILY):
        beta = np.random.rand() * 15.0
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- double‑ended inverse‑sqrt -----
    for _ in range(SAMPLES_PER_FAMILY):
        beta = np.random.rand() * 15.0
        i = np.arange(n, dtype=float)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- power‑law (double‑ended) -----
    for _ in range(SAMPLES_PER_FAMILY):
        beta = np.random.rand() * 20.0
        p = np.random.rand() * 5.0 + 0.05          # p ∈ (0.05, 5.05)
        i = np.arange(n, dtype=float)
        a = (1.0 / (i + beta + 1.0) ** p +
             1.0 / ((n - 1 - i) + beta + 1.0) ** p)
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- exponential decay (double‑ended) -----
    for _ in range(SAMPLES_PER_FAMILY):
        decay = np.random.rand() * 0.018 + 0.0005   # [5e‑4, 1.85e‑2]
        i = np.arange(n, dtype=float)
        a = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- cosine‑based (large at ends) -----
    for _ in range(SAMPLES_PER_FAMILY):
        p = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- random block candidates (few large blocks) -----
    for _ in range(500):
        nb = np.random.randint(2, 7)                # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _eval(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    # ----- sparse random spikes -----
    for _ in range(300):
        K = np.random.randint(5, 60)                # number of spikes
        pos = np.random.choice(n, K, replace=False)
        vals = np.random.rand(K) + 0.1
        cand = np.zeros(n, dtype=float)
        cand[pos] = vals
        cand = _project_to_simplex(cand, target_sum)
        sc = _eval(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    print(f"[shape] best score after families = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. ADAM‑BASED PROJECTED SUB‑GRADIENT DESCENT
    # ------------------------------------------------------------------
    a = best_a.copy()
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.18
    it = 0
    no_improve = 0

    while time.time() < deadline and it < 800_000:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())

        if cur_score < best_score - 1e-12:
            best_score, best_a = cur_score, a.copy()
            no_improve = 0
            print(f"[{it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # active indices (all within a tiny relative tolerance)
        max_val = conv.max()
        eps = max(1e-12, max_val * 1e-9)
        max_idxs = np.where(conv >= max_val - eps)[0]

        grad = _gradient_at_max(a, max_idxs)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # simple decay schedule
        lr = lr0 / (1.0 + 1e-5 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick
        if (it % 4000 == 0 and it > 0) or no_improve > 4000:
            noise = np.random.randn(n) * target_sum * 4e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[{it:07d}] random perturbation")

        a = a_new
        it += 1

    print(f"[adam] finished after {it} iterations, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. ACTIVE‑SET LP REFINEMENT
    # ------------------------------------------------------------------
    a_refined = _lp_refine(best_a, target_sum, size_fft,
                           max_iters=8, init_top_k=7, tol=1e-9)
    refined_score = _eval(a_refined)
    if refined_score < best_score:
        best_score, best_a = refined_score, a_refined.copy()
        print(f"[lp] improvement after refinement = {best_score:.6f}")
    else:
        print(f"[lp] no improvement (score {refined_score:.6f})")

    # ------------------------------------------------------------------
    # 4. FINAL RANDOM BLOCK REFINEMENT
    # ------------------------------------------------------------------
    for _ in range(300):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _eval(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
