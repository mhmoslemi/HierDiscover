"""
Propose a non‑negative step function (list of heights) that minimises
evaluate_sequence().

Algorithm
---------
1. **Parametric shape search** – many simple families (inverse‑sqrt,
   double‑ended power‑law, exponential, cosine, mixtures, random blocks)
   are sampled, projected onto the simplex Σa = √(2 n) and evaluated.
   The best candidate is kept.

2. **Smooth‑max Adam optimisation** – starting from the best shape we
   minimise the *soft* maximum of the self‑convolution:
       SM_α(a) = Σ_k p_k·(a*a)_k ,   p_k ∝ exp(α·(a*a)_k) .
   For moderate α this is a smooth approximation of the true max.
   Its gradient can be computed by a single FFT‑based convolution:
       dSM_i = 2·Σ_j a_j·p_{i+j} .
   Using Adam with Euclidean projection onto the simplex we run many
   iterations, slowly raising α (annealing) so the solution approaches
   the true max.  Random perturbations are injected when progress stalls.

3. **Random‑block refinement** – a handful of fresh block‑structured
   candidates are evaluated and, if they improve the best score, replace
   it.

The whole procedure respects the 1000 s time budget, prints progress
and finally returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of vector v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]                               # sorted descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, n: int, size_fft: int) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·n−1) using FFT.
    `size_fft` must be ≥ 2·n−1 and a power of two.
    """
    fa = np.fft.rfft(a, size_fft)
    conv_full = np.fft.irfft(fa * fa, size_fft)
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> (np.ndarray, float):
    """
    Return softmax weights p_k ∝ exp(alpha·conv_k) and the smoothed
    maximum SM = Σ p_k·conv_k (stable implementation).
    """
    # shift for numerical stability
    shift = np.max(alpha * conv)
    exp_shifted = np.exp(alpha * conv - shift)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    sm = np.dot(weights, conv)                     # weighted average = soft‑max
    return weights, sm


# ----------------------------------------------------------------------
def _gradient_smooth_max(a: np.ndarray, weights: np.ndarray,
                        n: int, size_fft: int,
                        fa: np.ndarray) -> np.ndarray:
    """
    Gradient of SM = Σ p_k·(a*a)_k w.r.t. a.
    Implements   dSM_i = 2·Σ_j a_j·p_{i+j}.
    Computed via a single FFT convolution.
    """
    # reverse the weight vector for correlation
    w_rev = weights[::-1]
    fw_rev = np.fft.rfft(w_rev, size_fft)
    # convolution a * w_rev  (linear)
    conv_rev_full = np.fft.irfft(fa * fw_rev, size_fft)
    # keep the needed prefix (length 3·n−2)
    conv_rev = conv_rev_full[: 3 * n - 2]
    # dSM_i = 2 * conv_rev[2·n−2−i]   for i = 0…n−1
    idx = (2 * n - 2) - np.arange(n)
    dsm = 2.0 * conv_rev[idx]
    return dsm


# ----------------------------------------------------------------------
def _adam_optimize(start_a: np.ndarray,
                   target_sum: float,
                   n: int,
                   size_fft: int,
                   deadline: float) -> (np.ndarray, float):
    """
    Adam optimisation of the smooth‑max objective.
    Returns the best vector seen and its (true) max‑convolution value.
    """
    # constants (with our normalisation Σa = √(2n) we have S² = 2n → factor = 1)
    S = target_sum
    const_term = 2.0 * S / S   # equals 2, but we need 2*SM/S later; keep explicit.

    a = start_a.copy()
    best_a = a.copy()
    # compute first score
    conv = _self_convolution_fft(a, n, size_fft)
    best_score = float(np.max(conv))

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule for the soft‑max temperature
    alpha = 0.5          # start soft
    alpha_max = 2000.0
    alpha_inc = 1.001   # multiplicative increase each iteration

    # step‑size schedule (slow decay)
    lr0 = 0.15

    it = 0
    no_improve = 0
    max_no_improve = 5000
    noise_interval = 4000
    eps = 1e-12

    while time.time() < deadline and it < 300_000:
        # ---- convolution & soft‑max ----
        fa = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
        cur_score = float(conv.max())

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[adam {it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # soft‑max weights and SM
        weights, SM = _softmax_weights(conv, alpha)

        # gradient of SM
        dSM = _gradient_smooth_max(a, weights, n, size_fft, fa)

        # complete gradient of the *real* objective (since factor = 1)
        # gradient_i = dSM_i - 2*SM/S
        grad = dSM - (2.0 * SM / S)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / (1.0 + 1e-5 * it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick to escape plateaus
        if (it % noise_interval == 0 and it > 0) or no_improve > max_no_improve:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[adam {it:07d}] random perturbation")

        a = a_new

        # increase temperature (make soft‑max sharper)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)

        it += 1

    print(f"[adam] finished after {it} iterations, best score = {best_score:.6f}")
    return best_a, best_score

def _generate_random_block(n: int, target_sum: float) -> np.ndarray:
    """Create a random block (few constant pieces) candidate."""
    nb = np.random.randint(2, 7)                     # 2‑6 blocks
    cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = np.random.rand(nb) + 0.1
    cand = np.zeros(n, dtype=float)
    for blk, v in zip(blocks, vals):
        cand[blk] = v
    cand = _project_to_simplex(cand, target_sum)
    return cand
