def _optimise_adam_exp(n, a0, deadline, max_iters=5000, lr0=0.5, lam0=0.1, lam_max=40.0):
    a = a0.copy()
    a = np.clip(a, 0, None)
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)
    # adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # lambda schedule linear
    lam = lam0
    lam_inc = (lam_max - lam0) / max_iters
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # conv
        conv = _conv_full_fft(a)
        cur_score = 2.0 * n * conv.max()
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # compute softmax weights
        maxb = conv.max()
        exp_arg = lam * (conv - maxb)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w /= w_sum
        # gradient via correlation
        m_fft = 1 << ((3 * n - 2).bit_length())
        g = _gradient_via_correlation(w, a, m_fft)
        # Adam update for multiplicative exponents
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        # multiplicative update a = a * exp(-step)
        a = a * np.exp(-step)
        # enforce symmetry
        a = (a + a[::-1]) / 2.0
        # project onto simplex
        a = a / a.sum() if a.sum() > 0 else np.full(n, 1.0/n)
        # increase lambda
        lam = min(lam_max, lam + lam_inc)
    return best_seq, best_score

"""
Propose a step‑function sequence with a low `evaluate_sequence` score.

Algorithm Overview
------------------
1.  Load a historic sequence (if the variable `height_sequence_1` exists)
    and keep it as a baseline.
2.  Explore a few problem sizes (≈ 8k–16k).  For each size:
    * generate a diverse pool of initial probability vectors
      (arcsine, several symmetric Beta(α,α), Dirichlet draws, sparse spikes);
    * refine each candidate with an **Adam‑accelerated exponentiated‑gradient**
      method that optimises a soft‑max approximation of the max‑convolution;
    * optionally improve the best candidate with a projected **sub‑gradient**
      descent on the exact max‑convolution peak;
    * finally run a cheap stochastic **hill‑climb** that moves tiny amounts of
      mass between random positions (symmetrised after each accepted move).
3.  Keep the globally best sequence seen across all lengths.
4.  A final hill‑climb is performed on that global best before returning it
    as a plain Python list.

All optimisation steps work with vectors normalised to sum = 1;
the objective `2·n·max(conv(a,a))` is invariant under scaling, so this
normalisation does not affect the score.  Symmetry (`a[i]=a[n‑1‑i]`) is
enforced after every update – it reduces the effective dimension and
helps flatten the auto‑convolution.

The whole search respects the time budget (default 1000 s) by checking the
deadline after each iteration.  Only NumPy is required.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector of length 2·n‑1,
    a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns vector of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum 1."""
    n = a.shape[0]
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid exact 0/1 – they would blow up for α<1
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a small, diverse set of probability vectors for a given length.
    All vectors are normalised to sum 1.
    """
    cand = []

    # arcsine (α = ½)
    cand.append(_arcsine_distribution(n))

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a grid of symmetric betas (including more aggressive α<½)
    for alpha in (0.15, 0.2, 0.25, 0.3, 0.35, 0.45, 0.55, 0.7):
        cand.append(_beta_distribution(n, alpha))

    # a few random β draws
    for _ in range(3):
        alpha = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, alpha))

    # sparse “spike” vector (few non‑zero entries)
    k = max(2, n // 500)               # roughly 0.2 % spikes
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1   # avoid exact zeros
    cand.append(a / a.sum())

    # Dirichlet with low concentration (many tiny entries)
    a = np.random.dirichlet(np.full(n, 0.2))
    cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _optimise_adam_exp(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 6000,
    lr0: float = 0.6,
    lam0: float = 0.2,
    lam_max: float = 35.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑accelerated exponentiated‑gradient on a soft‑max surrogate of
    max(conv(a,a)).  Returns the best normalised vector and its exact score.
    """
    a = np.maximum(init_seq.astype(np.float64), 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # FFT length for correlation (needs at least len(w)+len(a)-1)
    fft_len = 1 << ((3 * n - 2).bit_length())

    # linear schedule for λ
    lam = lam0
    lam_inc = (lam_max - lam0) / max_iters

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and score
        conv = _conv_full_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} ADAM it={it:5d} score={best_score:.6f}")

        # soft‑max weights (temperature λ)
        maxb = conv.max()
        exp_arg = lam * (conv - maxb)               # ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient via correlation
        g = _gradient_via_correlation(w, a, fft_len)

        # Adam update for the *exponents* (multiplicative update)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a * np.exp(-step)                      # multiplicative move
        # enforce symmetry (helps flatten convolution)
        a = (a + a[::-1]) * 0.5
        # renormalise onto simplex (scale invariance would work, but we keep sum=1)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)

        # increase temperature
        lam = min(lam_max, lam + lam_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Symmetry is enforced after each projection.
    """
    a = np.maximum(init_seq.astype(np.float64), 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a)
        k_star = int(np.argmax(conv))

        # construct sub‑gradient: g_i = 2·a_{k*−i} when index in range
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = 2.0 * a[k_star - idx]

        eta = eta0 / math.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        # enforce symmetry
        a = (a + a[::-1]) * 0.5

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} SUB it={it:5d} score={best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 400 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            a = (a + a[::-1]) * 0.5

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search: transfer a tiny amount of mass from one index
    to another, accept only if the exact objective improves.
    Symmetry is restored after each successful move.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        # donor
        i = np.random.randint(n)
        if best_seq[i] < 1e-14:
            continue
        # receiver
        j = np.random.randint(n)
        if i == j:
            continue

        # amount to shift: up to 4 % of donor, capped at 0.01
        delta = np.random.rand() * min(best_seq[i] * 0.04, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        # enforce symmetry
        trial = (trial + trial[::-1]) * 0.5

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score
