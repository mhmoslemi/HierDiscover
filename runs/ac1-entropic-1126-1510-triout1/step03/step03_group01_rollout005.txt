"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that
minimises

    score = 2·n·max(a*a) / (sum(a)**2)                (evaluate_sequence)

The denominator is invariant to scaling (as long as no entry is clipped
to 1000), so we work on the probability simplex Σ a_i = 1.
The algorithm combines two phases:

1. **Soft‑max (smoothed max) gradient descent** – a smooth surrogate of
   max(a*a) is optimised with a decreasing step size and a momentum term.
2. **Exact max sub‑gradient descent** – after the surrogate has sharpened
   (large temperature) the true max is used; the sub‑gradient is obtained
   analytically from the active convolution entries and a second momentum
   update is performed.

Both phases keep the iterate on the simplex by Euclidean projection.
Multiple lengths, several initialisations (uniform, random, arcsine,
previous best) and jitter are explored within the allotted time budget.
The best sequence found (according to the black‑box `evaluate_sequence`)
is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)) of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _initial_vector(init_kind: str, n: int) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if init_kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init_kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif init_kind == "arcsine":
        a = _arcsine_distribution(n)
    elif init_kind == "prev":
        # try to reuse the previously found construction
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                # linearly interpolate to the required length
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    # guarantee exact feasibility
    a = _project_onto_simplex(a, s=1.0)
    return a

            max(500, int(base_n * f))
            for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)
            if int(base_n * f) >= 500
        )
    )
    if not lengths:
        lengths = [8000]

    overall_best_seq = None          # type: np.ndarray | None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        # Allocate roughly a constant fraction of the remaining budget to this length.
        remaining = max(0.0, deadline - time.time())
        slice_deadline = time.time() + min(remaining * 0.30, 200.0)  # at most 200 s per length

        for init_kind in ("prev", "arcsine", "uniform", "random"):
            # ------------------------------------------------------------------
            # Initialise a feasible vector.
            a = _initial_vector(init_kind, n)

            # Store the best exact (max‑based) vector seen for this run.
            best_local_seq = a.copy()
            b = _fft_self_convolve(a)
            best_local_max = b.max()
            best_local_score = 2.0 * n * best_local_max

            # Momentum buffers for the two phases.
            velocity = np.zeros_like(a)
            beta = 0.9

            # Phase‑1 (soft‑max) parameters.
            eta0 = 0.5          # initial step‑size factor
            alpha = 5.0         # temperature (smoothness)
            alpha_factor = 1.003
            alpha_max = 2000.0

            iteration = 0
            # ------------------------------------------------------------------
            while time.time() < slice_deadline:
                iteration += 1

                # ---- current statistics -------------------------------------------------
                b = _fft_self_convolve(a)
                cur_max = b.max()
                cur_score = 2.0 * n * cur_max

                # keep the best exact sequence seen so far
                if cur_max < best_local_max - 1e-14:
                    best_local_max = cur_max
                    best_local_seq = a.copy()
                    best_local_score = cur_score
                    print(
                        f"[{time.time():.1f}] n={n} init={init_kind} "
                        f"iter={iteration} maxB={best_local_max:.8e} score≈{best_local_score:.6f}"
                    )

                # ---- decide which gradient to use ---------------------------------------
                if alpha < 500.0:
                    # ----- smoothed max (soft‑max) ------------------------------------
                    exp_term = np.exp(alpha * (b - cur_max))
                    w = exp_term / exp_term.sum()
                    conv_rev = _fft_convolve(a[::-1], w)            # length 3n‑2
                    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]          # size n
                    # anneal temperature
                    if alpha < alpha_max:
                        alpha = min(alpha_max, alpha * alpha_factor)
                else:
                    # ----- exact max sub‑gradient ------------------------------------
                    eps = 1e-12 * (abs(cur_max) + 1.0)                # tolerance
                    active = np.flatnonzero(b >= cur_max - eps)
                    w = np.zeros_like(b)
                    w[active] = 1.0
                    conv_rev = _fft_convolve(a[::-1], w)
                    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
                    # diagonal correction for the terms where i=j (k=2i)
                    even_mask = (active % 2 == 0)
                    diag_idx = (active[even_mask] // 2)
                    if diag_idx.size > 0:
                        grad[diag_idx] -= a[diag_idx]

                # ---- momentum + step ----------------------------------------------------
                velocity = beta * velocity + (1.0 - beta) * grad
                eta = eta0 / np.sqrt(iteration)          # diminishing step size
                a = a - eta * velocity
                a = _project_onto_simplex(a, s=1.0)

                # ---- occasional jitter to escape plateaux -------------------------------
                if iteration % 2000 == 0 and np.random.rand() < 0.07:
                    a += np.random.randn(n) * 1e-6
                    a = _project_onto_simplex(a, s=1.0)

                # Optional early‑stop if we already reach a very good bound.
                if best_local_score <= 1.502:
                    break
            # ------------------------------------------------------------------
            # Evaluate the best candidate from this (n, init) run.
            try:
                final_score = evaluate_sequence(best_local_seq.tolist())
            except Exception:
                final_score = best_local_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_seq.copy()
                print(
                    f"*** New global best → n={n} init={init_kind} "
                    f"score={overall_best_score:.6f}"
                )

            # stop early if we have already beaten the target
            if overall_best_score <= 1.503:
                print("Target reached – exiting early.")
                return overall_best_seq.tolist()

            # If the slice time is exhausted, move to the next length.
            if time.time() >= slice_deadline:
                break

        if time.time() >= deadline:
            break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but we return a uniform vector just in case.
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
