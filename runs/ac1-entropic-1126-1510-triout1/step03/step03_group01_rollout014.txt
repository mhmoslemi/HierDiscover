import cvxpy as cp

def solve_lp(a, M):
    n = len(a)
    g = cp.Variable(n, nonneg=True)
    # Convolution a * g <= M
    conv_expr = cp.conv(a, g)   # shape 2*n-1
    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-7, reltol=1e-7)
    if g.value is None:
        # fallback: zero
        return np.zeros(n)
    return np.array(g.value).astype(np.float64)

import numpy as np

def _project_onto_simplex(v, s=1.0):
    """Project v onto the probability simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _arcsine_distribution(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _fft_self_convolve(a):
    """Return a*a (full) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    """Full convolution of real vectors a and b using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _softmax_weights(b, alpha):
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        # fallback to uniform
        w = np.full_like(b, 1.0 / b.size)
    else:
        w = exp_term / w_sum
    return w

def _max_weights(b):
    cur_max = b.max()
    mask = (b == cur_max)
    count = mask.sum()
    if count == 0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w = mask.astype(np.float64) / count
    return w

def _gradient_from_weights(a, w):
    """Given weight vector w (length 2n-1) for convolution entries, compute gradient of weighted sum."""
    n = a.shape[0]
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad

    """Return max self‑convolution of a (assumed sum=1)."""
    return _fft_self_convolve(a).max()

    Returns numpy array g (length n) or None on failure."""

"""
Hybrid optimiser for the “step‑function” challenge.

* Stage 1 – Adam optimisation of a smooth surrogate of the peak
  (soft‑max of the self‑convolution).  The learning‑rate is kept
  moderate, the iterate is projected onto the simplex after each step
  and an annealing schedule slowly turns the surrogate into the true
  max (by raising the soft‑max temperature).

* Stage 2 – optional linear‑programming refinement.
  For the current vector `a` we solve
        maximise   Σ g_j
        s.t.       g ≥ 0 ,  a ∗ g ≤ M   (M = max a∗a)
  using cvxpy (ECOS or SCS).  The resulting direction is mixed with the
  current vector via a tiny line‑search and the result is projected back
  onto the simplex.

Multiple lengths and initialisations are tried; the best sequence found
within the time budget (default 1000 s) is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional import of the convex‑programming library.
try:
    import cvxpy as cp
    _CVXPY_AVAILABLE = True
except Exception:          # pragma: no cover
    _CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution of length *n*."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a via real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(alpha·(b_k‑max(b))) that sum to 1."""
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        return np.full_like(b, 1.0 / b.size)
    return exp_term / w_sum


# ----------------------------------------------------------------------
def _max_weights(b: np.ndarray) -> np.ndarray:
    """Uniform weights on the (possibly several) maximal entries of *b*."""
    cur_max = b.max()
    mask = (b == cur_max)
    count = mask.sum()
    if count == 0:
        return np.full_like(b, 1.0 / b.size)
    return mask.astype(np.float64) / count


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k w.r.t. a.
    Returned vector has the same length as *a*.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)                # length = 3n‑2
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]          # slice the central n entries
    return grad


# ----------------------------------------------------------------------
def _evaluate_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of *a* (sum(a) is assumed = 1)."""
    return _fft_self_convolve(a).max()


# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float):
    """
    Solve the LP  max Σ g_j   s.t.   g≥0 , a∗g ≤ M.
    Returns a NumPy array *g* (or ``None`` on failure).
    """
    if not _CVXPY_AVAILABLE:
        return None
    n = a.shape[0]
    g = cp.Variable(n, nonneg=True)
    conv_ag = cp.conv(a, g)               # length 2n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [conv_ag <= M])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   feastol=1e-7, abstol=1e-7, reltol=1e-7)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, warm_start=True, max_iters=5000)
        except Exception:
            return None
    if g.value is None:
        return None
    return np.array(g.value, dtype=np.float64)

            break
    return best_tau, best_val
