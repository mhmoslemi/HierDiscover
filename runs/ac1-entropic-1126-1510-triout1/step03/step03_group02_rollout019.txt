"""
Light‑weight multi‑restart Adam‑based optimisation of the self‑convolution peak.

The algorithm:
   1. Choose a moderate length n (uses the reference length if available,
      otherwise 2500).  The sum constraint is the classical √(2 n) used in
      the literature.
   2. Generate a diverse pool of seed vectors (uniform, powers, exponentials,
      sin/cos, inverse‑sqrt, random Dirichlet, random block, etc.).
   3. For the ten best seeds run an Adam optimiser on the smooth‑max
      approximation of maxₖ (a∗a)ₖ, keeping the simplex constraint via Euclidean
      projection after each step.  A decreasing temperature (soft‑max) and
      occasional random kick help to escape plateaux.
   4. Keep the overall best vector and return it as a Python list.
   5. All heavy work uses FFT‑based convolutions (O(n log n)), so the whole
      search comfortably fits the 1000 s budget.

The objective that is minimised is exactly the one used by `evaluate_sequence`:
    2·n·max(a∗a) / (sum(a)²)
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution using FFT (real‐valued inputs)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    # keep only the non‑zero part (length = len(u)+len(v)-1)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """a * a (full convolution)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. a.
    The factor 2 appears because the quadratic term is symmetric.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)        # length = 3·n‑2
    start = len(w) - len(a)                     # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int) -> float:
    """Exact objective (no security checks)."""
    conv = _self_convolution(a, _size_fft_global)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _adam_refine(a0: np.ndarray,
                 deadline: float,
                 n: int,
                 size_fft: int,
                 target_sum: float,
                 max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max version of the max‑convolution objective.
    Returns the best vector encountered during the run and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a, n)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # base learning rate – scaled to the typical magnitude of a
    eta0 = target_sum * 0.3 / n
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature for the soft‑max (start relatively high)
    conv = _self_convolution(a, size_fft)
    T = max(1e-3, (conv.max() - conv.min()) * 0.5)

    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iters:
        # ----- current objective -----
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            # occasional feedback
            if it % 10_000 == 0:
                print(f"[adam] iter {it:,} new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ----- soft‑max weights (smooth approximation of the max) -----
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)         # stability
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        it += 1
        eta = eta0 / (1.0 + it * 1e-5)              # mild decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat

        # ----- projection onto the simplex (keeps Σa = target_sum) -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random kick -----
        if (it % 5_000 == 0 and it > 0) or no_improve > 8_000:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[adam] iter {it:,} random perturbation")

        a = a_new
        # cool the temperature slowly
        T *= 0.9995

    return best_a, best_score

# Global placeholder for FFT size – filled in `propose_candidate`.
_size_fft_global = None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation score.  The routine respects a 1000 s budget and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 2500
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # normalisation used by the benchmark
    global _size_fft_global
    _size_fft_global = 1 << ((3 * n - 2).bit_length())   # power of two ≥ 3n‑2

    # Helper to evaluate a candidate quickly (no security checks)
    def quick_score(a: np.ndarray) -> float:
        conv = _self_convolution(a, _size_fft_global)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # --------------------------------------------------------------
    # Keep the K best raw candidates
    K = 12
    best_raw = []    # list of (score, a)

    def keep(score: float, a: np.ndarray) -> None:
        """Maintain a size‑K list of the best raw candidates."""
        if len(best_raw) < K:
            best_raw.append((score, a.copy()))
            best_raw.sort(key=lambda x: x[0])
        elif score < best_raw[-1][0]:
            best_raw[-1] = (score, a.copy())
            best_raw.sort(key=lambda x: x[0])

    idx = np.arange(n)

    # ---------- 1. generate diverse seeds ---------------------------------
    # uniform
    a = np.ones(n)
    a = _project_to_simplex(a, target_sum)
    keep(quick_score(a), a)
    print(f"[seed] uniform score = {quick_score(a):.6f}")

    # exponential decay (both sides)
    for _ in range(70):
        decay = np.random.rand() * 0.02 + 0.001
        a = np.exp(-decay * idx)
        a = _project_to_simplex(a, target_sum)
        keep(quick_score(a), a)

    # double‑ended exponential
    for _ in range(70):
        decay = np.random.rand() * 0.02 + 0.001
        a = np.exp(-decay * idx) + np.exp(-decay * (n - 1 - idx))
        a = _project_to_simplex(a, target_sum)
        keep(quick_score(a), a)

    # inverse‑sqrt (single‑ended)
    for _ in range(70):
        beta = np.random.rand() * 12.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        keep(quick_score(a), a)

    # double‑ended inverse‑sqrt
    for _ in range(70):
        beta = np.random.rand() * 12.0
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        keep(quick_score(a), a)

    # power‑law
    for _ in range(70):
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 2.5 + 0.1   # exponent ∈ [0.1, 2.6]
        a = 1.0 / (idx + beta + 1.0) ** pexp
        a = _project_to_simplex(a, target_sum)
        keep(quick_score(a), a)

    # sin / cos shapes
    for _ in range(60):
        p = np.random.rand() * 6.0 + 0.2
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        keep(quick_score(a), a)

    for _ in range(60):
        p = np.random.rand() * 6.0 + 0.2
        a = np.sin(np.pi * idx / (n - 1)) ** p
        a = _project_to_simplex(a, target_sum)
        keep(quick_score(a), a)

    # random Dirichlet (concentrations < 1 produce sparse spikes)
    for _ in range(80):
        alpha = np.random.rand() * 0.9 + 0.05
        a = np.random.gamma(shape=alpha, scale=1.0, size=n)
        a = _project_to_simplex(a, target_sum)
        keep(quick_score(a), a)

    # random block candidates (few large blocks)
    for _ in range(150):
        nb = np.random.randint(2, 7)                     # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        keep(quick_score(cand), cand)

    # keep the reference solution if we have one
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        ref = _project_to_simplex(ref, target_sum)
        keep(quick_score(ref), ref)

    best_raw.sort(key=lambda x: x[0])
    print(f"[seed phase] best raw score = {best_raw[0][0]:.6f}")

    # --------------------------------------------------------------
    # ---------- 2. Adam based refinement -------------------------------
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    for rank, (raw_score, a0) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        print(f"[refine] start candidate #{rank+1} (raw score {raw_score:.6f})")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 1300))   # rough ops per second estimate
        refined_a, refined_score = _adam_refine(
            a0, deadline, n, _size_fft_global, target_sum, max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best = {global_best_score:.6f}")

    # --------------------------------------------------------------
    # ---------- 3. Quick random block sweep (no gradient) --------------
    while time.time() < deadline:
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[sweep] improved to {global_best_score:.6f}")
        # stop if very little time left
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    # Return as plain Python list (evaluate_sequence will re‑apply its own checks)
    return global_best_a.tolist()
