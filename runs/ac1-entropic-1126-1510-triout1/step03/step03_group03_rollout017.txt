"""
Propose a low‑score non‑negative step‑function.

The algorithm works in three stages:
1. **Smooth‑max continuation** – the heights a_i are represented as
   a soft‑max of free parameters y (ensuring Σa = √(2·n)).  For a given
   temperature μ the smooth surrogate
        φ_μ(y) = max_k (a∗a)[k] + μ·log Σ_k exp((a∗a)[k]/μ)
   is minimised with L‑BFGS‑B.  μ is cooled geometrically (μ←μ·γ) down to
   a tiny value.  FFTs are used for all convolutions, giving O(n log n)
   per gradient evaluation and allowing large n (≈8000) without a
   noticeable slowdown.
2. **Multi‑start & random restarts** – several diverse initial y‑vectors
   (uniform, V‑shaped, random Gaussian, inverted‑V) are optimised;
   the best solution over all runs is kept.
3. **Fine‑grained stochastic refinement** – the current best a is
   repeatedly perturbed by Gaussian noise, projected back onto the
   ℓ¹‑simplex (Σa = √(2·n)) and accepted if the true evaluation score
   improves.  A decreasing step‑size schedule yields a cheap local‑search
   that frequently discovers slight improvements after the continuation
   phase.

All work stops before the given time budget expires; the best sequence
found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _linear_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a∗a using FFT (real‑valued, length 2·n‑1)."""
    n = a.shape[0]
    L = _next_pow2(2 * n)
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[: 2 * n - 1]  # keep linear part only


# ----------------------------------------------------------------------
def _weighted_convolution_grad_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g_a = 2·(w ∗ a_rev)[n‑1 : n‑1+n] using FFT.
    w has length 2·n‑1, a_rev is a reversed.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = _next_pow2(len(w) + len(a_rev) - 1)
    W = np.fft.rfft(w, n=L)
    A_rev = np.fft.rfft(a_rev, n=L)
    conv_wr = np.fft.irfft(W * A_rev, n=L)
    # slice the central part (indices n‑1 … n‑1+n‑1)
    start = n - 1
    g_a = 2.0 * conv_wr[start : start + n]
    return g_a


# ----------------------------------------------------------------------
def _compute_f_and_grad_fft(y: np.ndarray, mu: float, n: int,
                            S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (temperature mu) and its gradient w.r.t. y.
    Uses FFT for all convolutions.
    """
    p = _softmax(y)          # probabilities, Σp = 1
    a = S * p                # heights, Σa = S

    # convolution a*a
    conv = _linear_convolution_fft(a)       # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # protect exp
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – equal weight on all maximisers
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a
    g_a = _weighted_convolution_grad_fft(w, a)   # length n

    # chain rule to obtain gradient w.r.t. y
    inner = np.dot(g_a, p)                # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjectiveFFT:
    """Callable for scipy.optimize – returns (value, gradient)."""

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad_fft(y, self.mu, self.n, self.S)
        return f, g

    # ------------------------------------------------------------------

    init_y = []
