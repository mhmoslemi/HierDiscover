def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a):
    """Full convolution a*a using real FFT (returns length 2n‑1 array)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]

def _gradient_via_softmax(a, b, alpha, size_fft):
    """Gradient of (1/α)·log Σ exp(α·b_k) w.r.t a."""
    n = a.shape[0]
    if alpha > 2000.0:
        # treat max entry as one-hot
        k = int(np.argmax(b))
        # contribution from ordered pairs i+j=k
        idx = k - np.arange(n)          # possible i such that j=k-i
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g
    # softmax weights
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g

def _optimize_adam(init_a, n, deadline):
    """
    Adam‑based projected gradient descent that minimises  max(conv(a,a)).
    The sum of a is kept equal to 1 (simplex) throughout.
    Returns best sequence found and its internal score 2*n*max_b.
    """
    a = init_a.copy()
    best_a = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.5   # base learning rate
    # soft‑max temperature schedule
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 20000.0
    size_fft = 1 << ((2 * n - 1).bit_length())
    it = 0
    while time.time() < deadline:
        it += 1
        # forward pass
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")
        # gradient of smoothed max
        grad = _gradient_via_softmax(a, b, alpha, size_fft)
        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 / (np.sqrt(v_hat) + eps) * m_hat
        a = a - step
        a = _project_onto_simplex(a, s=1.0)
        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)
        # occasional jitter to shake out of plateaus
        if it % 500 == 0 and rng.rand() < 0.1:
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = _project_onto_simplex(a, s=1.0)
    return best_a, best_score

def _local_hill_climb(a, n, deadline):
    """Simple hill‑climb that moves tiny mass between two positions."""
    best_a = a.copy()
    b = _fft_self_convolve(best_a)
    best_score = 2.0 * n * b.max()
    rng = np.random.default_rng()

        i, j = rng.choice(n, size=2, replace=False)

        delta = min(best_a[i], 0.01 * best_a[i])

            best_a = cand
            # print improvement occasionally
            print(f"[{time.time():.2f}] n={n} hill_climb score={best_score:.6f}")
    return best_a, best_score

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


def _initial_sequence(n: int, rng: np.random.Generator, kind: str = "random") -> np.ndarray:
    """Create an initial probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = _initial_sequence(n, rng, kind="arcsine")
    elif kind == "edge_heavy":
        # a few large entries near the borders, rest uniform tiny
        a = np.zeros(n, dtype=np.float64)
        edge_mass = 0.4
        a[0] = a[-1] = edge_mass / 2.0
        if n > 2:
            a[1:-1] = (1.0 - edge_mass) / (n - 2)
        a = _project_onto_simplex(a, s=1.0)
    elif kind == "dirichlet_small":
        # Dirichlet with concentration <1 → spiky vectors
        alpha_vec = np.full(n, 0.2, dtype=np.float64)
        a = rng.dirichlet(alpha_vec)
    else:                                 # random
        a = rng.rand(n)
        a = a / a.sum()
    a = _project_onto_simplex(a, s=1.0)
    return a


def _optimize_adam(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of  max(conv(a,a)).
    Returns the best sequence found (sum fixed to 1) and its internal
    2·n·max_b score (exact value when sum=1).
    """
    rng = np.random.default_rng()
    a = init_a.copy()
    best_a = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5

    # soft‑max temperature schedule
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 20000.0

    size_fft = 1 << ((2 * n - 1).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and objective
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient w.r.t. smoothed max
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.1:
            a += rng.normal(scale=1e-4, size=n)
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_score


def _local_hill_climb(a: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap local search: move a small amount of mass from one index to another.
    Accept only improving moves.
    """
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_a).max()

    while time.time() < deadline:
        i, j = rng.choice(n, size=2, replace=False)
        if best_a[i] < 1e-12:
            continue
        delta = min(best_a[i], 0.01 * best_a[i])
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            print(f"[{time.time():.2f}] n={n} hill_climb score={best_score:.6f}")
    return best_a, best_score


def propose_candidate(seed: int = 47,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds,
    trying several lengths, several random initialisations and a projected
    Adam optimiser (with a soft‑max smoothing of the max‑convolution).  The
    best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the final print
    deadline = time.time() + budget_s - 2.0

    # base length – try to reuse the length of a previously found sequence
    base_n = 8000
    try:
        base_n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        pass

    # candidate lengths (capped at 20000 for speed)
    length_factors = (0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # main optimisation loop over lengths
    for n in lengths:
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate ≈20 % of the remaining budget to this length (minimum 5 s)
        slice_budget = max(5.0, remaining * 0.2)
        slice_deadline = time.time() + slice_budget

        init_kinds = (
            "prev", "arcsine", "uniform", "random",
            "edge_heavy", "dirichlet_small"
        )
        for kind in init_kinds:
            if time.time() >= slice_deadline:
                break
            a0 = _initial_sequence(n, rng, kind)
            # run the Adam optimiser
            seq_opt, int_score = _optimize_adam(a0, n, slice_deadline)

            # cheap local refinement if a little time is left
            leftover = slice_deadline - time.time()
            if leftover > 2.0:
                lc_deadline = time.time() + min(2.0, leftover * 0.5)
                seq_opt, int_score = _local_hill_climb(seq_opt, n, lc_deadline)

            # evaluate with the official black‑box (more reliable)
            try:
                final_score = evaluate_sequence(seq_opt.tolist())
            except Exception:
                final_score = int_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq_opt.copy()
                print(
                    f"*** Global new best → len={n} init={kind} "
                    f"score={overall_best_score:.6f}"
                )

            # early exit if we are comfortably below the target
            if overall_best_score <= 1.45:
                break
        if overall_best_score <= 1.45:
            break

    # safety net – should never be triggered
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} "
          f"(length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
