    m = 1 << ((3*n - 2).bit_length()) # as before

    v_momen2 = np.zeros(n)

        maxb = b.max()
        exp_arg = lam * (b - maxb)
        w = np.exp(exp_arg)

        v_hat = v_momen2 / (1 - beta2**t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)  # Note sign: we want to move opposite to gradient for min; g is gradient of maxb? """
Propose a low‑score step‑function sequence.

Algorithm
---------
1. Choose a target length (≈10 000) and normalise all vectors to sum 1
   (the objective is invariant under scaling).
2. Generate several diverse initial distributions:
   * discrete arcsine law,
   * symmetric Beta(α,α) with α drawn uniformly from [0.05, 0.9],
   * Dirichlet draws with low concentration,
   * uniform.
3. For each start run a **multiplicative Adam** optimiser that minimises a
   soft‑max approximation of the convolution peak.  The temperature λ is
   increased linearly up to 200, while Adam provides per‑coordinate adaptive
   step‑sizes.  The best vector seen is kept.
4. Refine the best candidate with **projected sub‑gradient descent** on the
   exact max‑convolution (using the classic sub‑gradient
   g_i = 2·a_{k*−i} where k* is a maximiser of the convolution).
5. Perform a longer **stochastic hill‑climb** that repeatedly moves a tiny
   amount of mass from a “high‑gradient” index to a “low‑gradient” index;
   only improving moves are accepted.
6. Keep the globally best sequence among all restarts and return it as a
   plain Python list.  The whole search respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α = ½), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on grid x_i = (i+½)/n, normalised."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) via FFT.
    w – weight vector of length 2·n‑1,
    a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns array of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector a with Σa = 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _optimise_exponential_adam(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 15000,
    lr0: float = 0.12,
    lambda0: float = 0.3,
    lambda_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative Adam optimiser on a soft‑max approximation of the peak
    of the self‑convolution.
    Returns the best vector (Σ = 1) seen and its true score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT length for correlation (needs to accommodate w + a)
    m_corr = 1 << ((3 * n - 2).bit_length())

    # Adam moments
    m1 = np.zeros(n, dtype=np.float64)
    m2 = np.zeros(n, dtype=np.float64)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    lam = lambda0
    lam_inc = (lambda_max - lambda0) / max_iters

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and true score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} Adam it={it:5d} score={best_score:.6f}")

        # soft‑max weights
        maxb = b.max()
        exp_arg = lam * (b - maxb)          # ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # gradient (correlation of w with a)
        g = _gradient_via_correlation(w, a, m_corr)

        # Adam update (multiplicative)
        m1 = beta1 * m1 + (1.0 - beta1) * g
        m2 = beta2 * m2 + (1.0 - beta2) * (g * g)

        # bias correction
        m1_hat = m1 / (1.0 - beta1 ** it)
        m2_hat = m2 / (1.0 - beta2 ** it)

        step = lr0 * m1_hat / (np.sqrt(m2_hat) + eps)   # descent direction
        # clip to avoid overflow in exp
        step = np.clip(step, -5.0, 5.0)

        a = a * np.exp(-step)          # multiplicative move (keeps positivity)

        # project onto simplex (just renormalise)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a[:] = 1.0 / n

        # increase temperature
        lam = min(lambda_max, lam + lam_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses the classic sub‑gradient g_i = 2·a_{k*−i} where k* is a maximiser.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # build sub‑gradient for the active index
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} sub it={it:5d} score={best_score:.6f}")

        # occasional jitter to shake out plateaus
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 40000,
) -> tuple[np.ndarray, float]:
    """
    Fast stochastic local search.
    Move a tiny amount of mass from one index to another; keep the change
    if the unsmoothed objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        # pick donor with non‑negligible mass
        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # transfer a small fraction of donor's mass
        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score
