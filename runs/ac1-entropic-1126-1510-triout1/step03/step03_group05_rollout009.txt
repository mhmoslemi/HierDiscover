def gradient_softmax(a, conv, alpha):
    # conv length = 2n-1
    maxc = conv.max()
    # shift for stability
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    w_norm = w / w_sum
    a_rev = a[::-1]
    # using convolution conv of w_norm and a_rev
    size_fft = next_pow2( len(w_norm) + len(a_rev) - 1 )
    grad_conv = _conv_fft(w_norm, a_rev, size_fft)
    # slice
    grad = 2.0 * grad_conv[len(a_rev)-1 : len(a_rev)-1 + len(a)]   # careful index
    # Wait: We want slice at indices (n-1) to (n-1 + n-1) = [n-1, 2n-2].
    start = len(w_norm) - 1  # we need conv at shift p = n-1
    # Actually size = (2n-1)+(n)-1 = 3n-2. So w_norm length = 2n-1, a_rev length = n.
    # We compute conv length 3n-2. The index p = n-1 (zero-indexed) corresponds to start = (p) => n-1.
    # So grad = 2 * conv[p : p + n]
    start = len(a) - 1  # Since a_rev len = n, w_norm len = 2n-1. In convolution, the index where we align center is (len(a_rev)-1) = n-1.
    # Danger: Let's use formula: conv = convolve(w_norm, a_rev) = sum_{i} w_norm[i] * a_rev[p - i]; for p = n-1 gives sum_{i} w_norm[i] * a_rev[n-1 - i] = sum_{i} w_norm[i] * a[i] (since a_rev[t] = a[n-1 - t] => a_rev[n-1 - i] = a[i]).
    # So conv at p = n-1 yields sum_i w[i] * a[i].
    # For general p = n-1 + i: conv[p] = sum_{k} w[k] * a_rev[p - k] = sum_{k} w[k] * a[n-1 - (p - k)] = sum_k w[k] * a[k - i] (since n-1 - (p - k) = n-1 - ((n-1)+i - k) = k - i). Yes matches gradient formula.
    # So we need conv slice from p = n-1 to p = n-1 + (n-1) = 2n-2 inclusive (n values)
    start = len(a)-1  # indeed n-1

def adam_opt(a, target_sum, size_fft, deadline, best_score, best_a):
    beta1=0.9; beta2=0.999; eps=1e-8
    m=np.zeros_like(a); v=np.zeros_like(a)
    lr0=0.2
    it=0
    no_improve=0
    # stage schedule: list of (alpha, lr_factor, max_iters)
    schedule = [(2.0, 1.0, 2000), (5.0, 0.8, 2000), (10.0, 0.6, 2000), (20.0, 0.4, 2000), (40.0, 0.3, 2000)]
    for alpha, lr_factor, max_stage_iter in schedule:
        lr = lr0 * lr_factor
        for s_it in range(max_stage_iter):
            if time.time()>deadline: break
            # compute conv
            conv = _self_convolution(a, size_fft)
            cur_score = float(conv.max())
            if cur_score < best_score:
                best_score = cur_score; best_a = a.copy()
                no_improve = 0
                print(f"[adam alpha={alpha} it={it}] new best {best_score:.6f}")
            else:
                no_improve += 1

            grad = _gradient_softmax(a, conv, alpha)

            m = beta1 * m + (1.0 - beta1) * grad

            m_hat = m / (1.0 - beta1**(it+1))

            a = a - step

                a = _project_to_simplex(a + noise, target_sum)
                no_improve = 0
                print(f"[adam] random perturbation at it {it}")

            it += 1

    return best_a, best_score

"""
Propose a non‑negative step function (list of floats) that minimises the
evaluate_sequence() score, i.e. the maximal entry of the self‑convolution
under the L1‑normalisation Σa = √(2 n).

Strategy
--------
1.  Initialise a collection of candidate shapes (inverse‑sqrt, power‑law,
    exponential, cosine, sine‑squared, arcsine‑like, beta‑like and random
    block patterns).  All candidates are projected onto the simplex
    {a≥0, Σa = √(2 n)} and the best one is kept.

2.  Refine the best candidate by projected Adam optimisation.
    Two gradient flavours are used:
       • a smooth “soft‑max’’ gradient (α‑softmax of the convolution) – this
         yields a dense sub‑gradient that guides the search away from plateaus.
       • the exact active‑set gradient (only indices attaining the current
         maximum) – used in a final polishing stage.

    The Adam moments are kept across the two stages; after each update the
    iterate is projected back onto the simplex.  Periodic random perturbations
    help escape stagnation.

3.  A cheap local hill‑climbing phase swaps small amounts of mass between
    random positions; any improvement is accepted.

The routine respects the 1000 s budget, prints intermediate scores and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _score(a: np.ndarray, size_fft: int) -> float:
    """Return max entry of the self‑convolution (the evaluate_sequence score)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _gradient_at_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution (active‑set version).
    Implements  grad_i = (2/|max|) Σ_{k∈max} a_{k−i}  (if 0≤k−i<n).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    if max_idxs.size == 0:
        return grad
    weight = 1.0 / max_idxs.size
    for k in max_idxs:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        i_range = np.arange(i0, i1 + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    grad *= weight
    return grad


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, conv: np.ndarray, alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Smooth gradient of the α‑soft‑max of the convolution.
    grad_i = 2 * Σ_k w_k a_{k−i} where w_k ∝ exp(α·conv_k).
    """
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    w_norm = w / w_sum                     # shape (2n‑1,)

    a_rev = a[::-1]                        # reversed a, shape (n,)
    grad_full = _conv_fft(w_norm, a_rev, size_fft)   # length 3n‑2
    start = a.shape[0] - 1                 # index n‑1
    grad = 2.0 * grad_full[start:start + a.shape[0]]
    return grad
