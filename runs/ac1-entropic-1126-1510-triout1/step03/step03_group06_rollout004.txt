def _adam_optimize(a_init, deadline, max_iters, n):

   # Convert a_init to logit z
   epsilon = 1e-12
   z = np.log(np.clip(a_init, epsilon, None))

   m = np.zeros_like(z)
   v = np.zeros_like(z)
   beta1, beta2 = 0.9, 0.999
   lr = 0.05
   eps = 1e-8

   best_seq = None
   best_score = float('inf')
   lambda_val = 1.0
   lambda_inc = (200.0 - 1.0) / max_iters

   m_len = 1 << ((3*n-2).bit_length())
   for t in range(1, max_iters+1):
        if time.time() >= deadline: break

        # compute a
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)  # avoid overflow
        a = exp_z / np.sum(exp_z)

        # conv
        conv = _conv_full_fft(a)
        max_conv = conv.max()
        # compute weight softmax
        exp_arg = lambda_val * (conv - max_conv)   # negative or zero
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # gradient of smoothed objective wrt a:
        # g_a = sum_k w_k * grad(conv_k) = 2 * correlation(w, a)
        g_a = _gradient_via_correlation(w, a, m_len)   # returns 2* corr

        # gradient wrt z:
        dot = np.dot(a, g_a)
        g_z = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * (g_z**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate true objective occasionally, perhaps every 10 steps
        if t % 10 == 0:
            cur_score = 2.0 * n * max_conv   # actual
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} iter={t} score={best_score:.6f}")

        # increase lambda
        lambda_val = min(200.0, lambda_val + lambda_inc)

   # after loop compute final best if None
   if best_seq is None:
       best_seq = a # current a
       best_score = 2.0 * n * _conv_full_fft(best_seq).max()

   return best_seq, best_score

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  For several problem sizes `n` generate a small, diverse pool of
    probability vectors (arcsine, uniform, Beta‑distributions,
    Dirichlet, sparse spikes).
2.  Optimise each candidate with **Adam** on a *soft‑max* approximation of
    the ℓ∞‑norm of the self‑convolution.
    * Variables are the logits `z` of a soft‑max: a_i = exp(z_i)/∑exp(z).
    * Loss = (1/λ)·log∑_k exp(λ·conv_k).  λ is increased from 1 to 200.
    * Gradient w.r.t. `z` is obtained via an FFT‑based convolution
      (correlation) and the soft‑max weights.
3.  Keep the best vector (lowest true hard‑max objective) across all
    lengths and initialisations.
4.  Finish with a cheap stochastic hill‑climb that shifts a tiny amount
    of mass between two indices if it improves the true objective.
5.  Return the best sequence as a plain Python list.

The whole search respects the 1000 s budget, uses only NumPy (FFT
operations are O(n log n)) and runs on a single core (up to 2 CPUs are
available but not required).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n, normalised."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) via a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector (length 2·n‑1), a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Diverse set of starting probability vectors for size n."""
    cand = []

    # arcsine law (often a very good baseline)
    cand.append(_arcsine_distribution(n))

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric betas on a modest grid
    for alpha in (0.18, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.9, 1.2):
        cand.append(_beta_distribution(n, alpha))

    # a few random beta draws
    for _ in range(3):
        cand.append(_beta_distribution(n, np.random.uniform(0.1, 2.0)))

    # sparse “spike” vectors (few non‑zero entries)
    k = max(2, n // 500)                 # ≈0.2 % of positions
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    cand.append(a / a.sum())

    # Dirichlet with low concentration (many tiny entries)
    cand.append(np.random.dirichlet(np.full(n, 0.3)))

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 4000,
    lambda_max: float = 200.0,
    lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a soft‑max approximation of max(conv(a,a)).
    Returns the best (hard‑max evaluated) sequence and its score.
    """
    n = len(init_seq)

    # Initialise logits z from the initial probability vector
    eps = 1e-12
    z = np.log(np.clip(init_seq, eps, None))

    # Adam state
    m_hat = np.zeros_like(z)
    v_hat = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # FFT length sufficient for both convolution and correlation
    m_fft = 1 << ((3 * n - 2).bit_length())

    best_seq = None
    best_score = float("inf")
    lam = 1.0
    lam_inc = (lambda_max - lam) / max_iters

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- a = softmax(z) -----
        z_shift = z - np.max(z)          # avoid overflow
        exp_z = np.exp(z_shift)
        a = exp_z / exp_z.sum()

        # ----- convolution -----
        conv = _conv_full_fft(a)
        max_conv = conv.max()

        # ----- soft‑max weights -----
        w = np.exp(lam * (conv - max_conv))   # max element gets weight 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (smooth objective) -----
        g_a = _gradient_via_correlation(w, a, m_fft)   # = 2·corr(w,a)

        # ----- transform gradient to logits -----
        dot = np.dot(a, g_a)
        g_z = a * (g_a - dot)

        # ----- Adam update -----
        m_hat = beta1 * m_hat + (1 - beta1) * g_z
        v_hat = beta2 * v_hat + (1 - beta2) * (g_z ** 2)
        m_corr = m_hat / (1 - beta1 ** t)
        v_corr = v_hat / (1 - beta2 ** t)
        z -= lr * m_corr / (np.sqrt(v_corr) + eps_adam)

        # ----- occasional random jitter to escape plateaus -----
        if t % 500 == 0 and np.random.rand() < 0.15:
            z += np.random.randn(n) * 1e-3

        # ----- evaluate true objective every few steps -----
        if t % 10 == 0:
            cur_score = 2.0 * n * max_conv      # exact hard‑max value
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n:5d} it={t:5d} score={best_score:.6f}")

        # ----- increase λ (sharper soft‑max) -----
        lam = min(lambda_max, lam + lam_inc)

    # If we never updated best_seq (unlikely) fall back to current a
    if best_seq is None:
        best_seq = a
        best_score = 2.0 * n * _conv_full_fft(best_seq).max()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search: move a tiny amount of mass from one
    index to another and keep the move if it improves the exact objective.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = 2.0 * n * _conv_full_fft(best_seq).max()

    for trial in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial_seq = best_seq.copy()
        trial_seq[i] -= delta
        trial_seq[j] += delta

        cur_score = 2.0 * n * _conv_full_fft(trial_seq).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = trial_seq
            print(f"[{time.time():.1f}] hill‑climb trial={trial} score={best_score:.6f}")

    return best_seq, best_score

    except Exception:

    # ------------------------------------------------------------------
    # Problem sizes to explore – modest to stay well inside the time budget

        # generate a pool of diverse initial vectors
        init_pool = _initial_candidates(n)

        for init in init_pool:

            if sc < best_n_score:

                low = max(0, k_star - (n_local - 1))
