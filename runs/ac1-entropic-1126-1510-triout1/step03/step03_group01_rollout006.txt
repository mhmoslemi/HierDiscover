"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Parametric sweep** –  for a given length `n` we evaluate a family of
   symmetric power‑law densities   a_i ∝ ((i+½)(n‑i‑½))^p   (the arcsine
   distribution corresponds to p = –0.5).  The sweep over `p` (and a small
   family of truncated arcsine shapes) is performed with fast FFT based
   self‑convolution, keeping the best candidate.

2. **Local refinement** –  starting from the best sweep result we run a
   momentum gradient descent on the smooth surrogate of the peak
   (`log‑sum‑exp` of the self‑convolution).  The gradient can be written as
   a convolution, so each iteration still uses only FFTs.  The iterates are
   projected back onto the probability simplex.

3. **Return** –  the best sequence found (as a plain Python list) is
   returned.  All entries are non‑negative, sum to 1, and the length stays
   modest (≈20‑30 k) so that the final call to `evaluate_sequence` (which
   uses a naïve `np.convolve`) finishes quickly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of 2
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

        if frac > 0.0:
            cut = int(frac * n)
            a[cut:n - cut] = 0.0
            s = a.sum()
            if s == 0:
                continue
            a /= s
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] trunc frac={frac:.3f} => score≈{cur_score:.6f}")

    # ------------------------------------------------------------------
    # If we somehow failed to obtain any candidate, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()

    # ------------------------------------------------------------------
    # 3) Local refinement with smoothed‑max gradient descent.
    #    (Only performed if there is still time left.)
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    alpha = 5.0
    alpha_factor = 1.005
    alpha_max = 2000.0
    it = 0

    # keep track of the best exact (max‑based) sequence seen during refinement
    best_local_seq = a.copy()
    best_local_score = best_score

    while time.time() < deadline:
        it += 1

        # self‑convolution and its peak
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_max < best_local_score / (2.0 * n):
            best_local_score = cur_score
            best_local_seq = a.copy()
            print(f"[{time.time():.1f}] iter={it} refined score≈{cur_score:.6f}")

        # soft‑max weights over convolution entries
        exp_term = np.exp(alpha * (conv - cur_max))
        w_sum = exp_term.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_term / w_sum

        # gradient of the smoothed objective
        a_rev = a[::-1]
        grad_full = _fft_convolve(a_rev, w)            # length = 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # momentum and step
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter
        if it % 500 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

        # anneal the soft‑max temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    # ------------------------------------------------------------------
    # Choose the best of the sweep and the refinement.
    if best_local_score < best_score:
        best_seq = best_local_seq
        best_score = best_local_score

    print(f"Final best score (approx.) = {best_score:.6f}  (length={len(best_seq)})")
    # Convert to plain Python list for the harness.
    return best_seq.tolist()
