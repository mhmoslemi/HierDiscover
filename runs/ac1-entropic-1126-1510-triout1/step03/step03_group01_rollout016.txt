"""
Propose a candidate step function (non‑negative vector) that minimizes
evaluate_sequence().  The search proceeds in three stages:

1.  Parameterised families.
    •  Symmetric power‑law shapes   a_i ∝ ((i+shift)*(n‑i‑1+shift))**p
       with p∈(-1,0).  The classic arcsine corresponds to p=-0.5, shift=0.5.
    •  A dense grid over (p,shift) is scanned for several lengths.
    •  The best vector (lowest 2·n·max(a*a)) from this coarse search is kept.

2.  Local refinement.
    •  Starting from the best coarse candidate a simple sub‑gradient descent
       on the smoothed max (soft‑max with a growing temperature) is applied.
    •  Each gradient step is projected onto the probability simplex.
    •  Momentum (Adam‑style) and occasional jitter help to escape plateaus.

3.  Final verification.
    •  The best vector found is evaluated with the provided evaluate_sequence()
       and returned as a plain Python list.

The whole routine respects the allotted time budget and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using an FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _make_powerlaw_vector(n: int, p: float, shift: float) -> np.ndarray:
    """
    Symmetric power‑law vector:
        a_i ∝ ((i+shift)*(n-i-1+shift))**p   (p<0, shift>0)
    Returned array is normalised to sum=1.
    """
    i = np.arange(n, dtype=np.float64)
    raw = ((i + shift) * (n - i - 1 + shift)) ** p
    # Numerical safety – replace possible infinities
    raw = np.where(np.isfinite(raw), raw, 0.0)
    total = raw.sum()
    if total <= 0.0:
        # fall back to uniform – caller will discard this candidate
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / total


# ----------------------------------------------------------------------
def _score_from_vector(a: np.ndarray) -> float:
    """
    Fast surrogate for evaluate_sequence when sum(a)=1:
        score = 2 * n * max(a*a)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())
