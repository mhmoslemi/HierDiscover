def simulated_annealing(a_start, target_sum, n, size_fft_self, deadline):

    a = a_start.copy()

    best_a = a.copy()

    best_score = score(a)

    T = 0.01   # initial

    freeze_factor = 0.9999

    while time.time() < deadline:

        # propose perturbation: choose random indices i,j, transfer delta

        i, j = np.random.choice(n, 2, replace=False)

        delta = (np.random.rand() - 0.5) * target_sum * 0.001

        # ensure not negative after change

        a_new = a.copy()

        a_new[i] = max(0, a[i] + delta)

        a_new[j] = max(0, a[j] - delta)

        # re-project onto simplex (ensuring sum), use _project_to_simplex

        a_new = _project_to_simplex(a_new, target_sum)

        cur_score = score(a_new)

        delta_score = cur_score - best_score

        accept = False

        if cur_score < best_score:

            accept = True

        else:

            prob = np.exp(-delta_score / max(T, 1e-8))

            accept = np.random.rand() < prob

        if accept:

            a = a_new

            if cur_score < best_score:

                best_score = cur_score

                best_a = a_new.copy()

        T *= freeze_factor  # cooling

        # break if T too low or time

        if T < 1e-8:

            break

    return best_a, best_score

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a using real FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. a  (weights w sum to 1).
    The exact expression is   2·(w * a_rev)  restricted to n entries.
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size_fft)
    fa_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(fw * fa_rev, n=size_fft)
    start = len(w) - len(a)          # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int) -> float:
    """Exact evaluation score (same formula as `evaluate_sequence`)."""
    conv = _self_convolution(a, _size_fft_self_global)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   n: int,
                   size_fft_self: int,
                   size_fft_grad: int,
                   target_sum: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth‑max of the self‑convolution.
    Returns the best (a, score) seen during the run.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a, n)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate schedule (exponential decay)
    lr_max = target_sum * 0.5 / n          # a generous start
    lr_min = target_sum * 0.005 / n        # final learning rate

    # temperature for the soft‑max (initialised on first iteration)
    T = None

    it = 0
    while time.time() < deadline and it < max_iters:
        # --------------------------------------------------------------
        # evaluate current score (for bookkeeping)
        conv = _self_convolution(a, size_fft_self)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # optionally log progress
            # print(f"[adam] it {it:,} new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # initialise / decay temperature
        if T is None:
            spread = conv.max() - conv.min()
            T = max(1e-5, spread * 0.5)
        else:
            T *= 0.9995               # fairly fast cooling

        # --------------------------------------------------------------
        # soft‑max weights that approximate the max of the convolution
        shifted = conv - conv.max()               # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)        # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # gradient of the smoothed objective
        grad = _weighted_gradient(a, w, size_fft_grad)

        # --------------------------------------------------------------
        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decays exponentially from lr_max to lr_min
        lr = lr_max * (lr_min / lr_max) ** (it / max_iters)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce feasibility: non‑negative and correct L1‑norm
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random jolt to break plateaus
        if it % 5000 == 0:
            noise = np.random.randn(n) * target_sum * 1e-3
            a_new = _project_to_simplex(a_new + noise, target_sum)

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         deadline: float,
                         n: int,
                         size_fft_self: int,
                         target_sum: float) -> tuple[np.ndarray, float]:
    """
    A light‑weight simulated‑annealing walk that performs tiny mass swaps.
    Returns the best (a, score) encountered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a, n)

    # start temperature (relative to current best score)
    T = best_score * 0.05
    decay = 0.9998

    while time.time() < deadline:
        # propose a swap of a small amount of mass between two random positions
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * target_sum * 0.001  # ~0.1 % of total mass
        a_new = a.copy()
        a_new[i] = max(0.0, a[i] + delta)
        a_new[j] = max(0.0, a[j] - delta)

        # re‑project onto the simplex (keeps Σa intact)
        a_new = _project_to_simplex(a_new, target_sum)

        cur_score = _score(a_new, n)

        # Metropolis acceptance
        if cur_score < best_score:
            accept = True
        else:
            prob = math.exp(-(cur_score - best_score) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a = a_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_new.copy()
                # print(f"[anneal] new best = {best_score:.6f}")

        # cool
        T *= decay
        if T < 1e-9:
            break

    return best_a, best_score

        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Power‑law
    for _ in range(80):
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 1.7 + 0.3
        a = 1.0 / (idx + beta + 1.0) ** pexp
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Exponential decay
    for _ in range(80):
        decay = np.random.rand() * 0.015 + 0.0005
        a = np.exp(-decay * idx)
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Cosine‑based (large at ends)
    for _ in range(80):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Sine‑based (zero at ends)
    for _ in range(80):
        p = np.random.rand() * 6.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** p
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Linear ramp (ascending)
    for _ in range(50):
        p = np.random.rand() * 4.0 + 0.5
        a = (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Quadratic / polynomial
    for _ in range(50):
        p = np.random.rand() * 2.0 + 1.0
        a = (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # Random block candidates (few large blocks)
    for _ in range(300):
        nb = np.random.randint(2, 9)                     # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        consider(quick_score(cand), cand)

    # Mixture of previously discovered candidates
    pool = [arr for _, arr in best_candidates]
    for _ in range(40):
        f1 = pool[np.random.randint(len(pool))]
        f2 = pool[np.random.randint(len(pool))]
        alpha = np.random.rand()
        cand = alpha * f1 + (1 - alpha) * f2
        cand = _project_to_simplex(cand, target_sum)
        consider(quick_score(cand), cand)

    # Include the reference sequence if supplied
    if have_ref and base_seq is not None:
        cand = _project_to_simplex(base_seq, target_sum)
        consider(quick_score(cand), cand)

    # ------------------------------------------------------------------
    # Report best raw (pre‑refinement) candidate
    best_candidates.sort(key=lambda x: x[0])
    print(f"[explore] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. Refinement with Adam (smooth‑max descent)
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    for rank, (raw_score, a0) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank+1} (raw {raw_score:.6f})")
        remaining = deadline - time.time()
        max_iters = int(min(300_000, remaining * 1500))  # approx. 1500 it/s
        refined_a, refined_score = _adam_optimize(
            a0, deadline, n,
            _size_fft_self_global,
            _size_fft_grad_global,
            target_sum,
            max_iters=max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Final simulated‑annealing fine‑tuning
    if time.time() < deadline:
        print("[anneal] starting final simulated‑annealing pass")
        annealed_a, annealed_score = _simulated_annealing(
            global_best_a, deadline, n,
            _size_fft_self_global,
            target_sum
        )
        if annealed_score < global_best_score:
            global_best_score = annealed_score
            global_best_a = annealed_a.copy()
            print(f"[global] new best after annealing = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
