"""
Propose a step‑function sequence that (approximately) minimizes the maximum
self‑convolution under a fixed L1‑norm.  The algorithm consists of three
stages:

1. **Broad parametric search** – generate many families (inverse‑sqrt,
   double‑ended, power‑law, exponential, cosine, beta‑distribution,
   logistic, random block, uniform) and keep the few best candidates
   according to an exact convolution score.

2. **Refinement** – for each top candidate run a projected Adam optimiser.
   The objective is the soft‑max approximation of the convolution peak;
   the temperature is slowly annealed so that the optimiser eventually
   focuses on the true max.  After each Adam step the iterate is projected
   onto the simplex {a ≥ 0, Σa = √(2 n)}.

3. **Final random‑block sweep** – a quick pass of random block candidates
   exploiting any remaining plateaus.

All operations respect the security checks performed later by
`evaluate_sequence`.  The function respects a 1000 s time budget and
returns the best sequence discovered.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a (weights sum to 1).
    The exact expression is 2·(w ↔ a_rev) restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact evaluation score (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using a projected Adam optimiser.
    The objective is a smooth approximation of max_k (a*a)_k.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # step‑size base (scaled to problem size)
    eta0 = target_sum * 0.5 / n

    # temperature for soft‑max weights
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.2)
    T_decay = 0.9995

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- evaluate current point -----
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[adam] iter {it:,} new best = {best_score:.6f}")

        # ----- soft‑max weights (smooth max) -----
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)         # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        eta = eta0 / (1.0 + 1e-4 * it)  # slow decay

        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex (keeps Σa = target_sum) -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random perturbation to escape plateaus -----
        if (it % 10000 == 0 and it > 0):
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= T_decay
        it += 1

    return best_a, best_score

    # --- Beta‑distribution family ----------------------------------
    for _ in range(200):
        alpha = np.random.rand() * 4.0 + 0.2   # (0.2, 4.2)
        beta_param = np.random.rand() * 4.0 + 0.2
        x = (idx + 0.5) / n
        # PDF of Beta(α,β) up to a multiplicative constant
        a = (x ** (alpha - 1)) * ((1 - x) ** (beta_param - 1))
        a = np.where(np.isfinite(a), a, 0.0)
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # --- Logistic (sigmoidal) --------------------------------------
    for _ in range(120):
        slope = np.random.rand() * 0.04 + 0.005
        shift = np.random.rand() * n
        a = 1.0 / (1.0 + np.exp(slope * (idx - shift)))
        a = _project_to_simplex(a, target_sum)
        consider(quick_score(a), a)

    # --- Random block candidates (few large blocks) ---------------
    for _ in range(300):
        nb = np.random.randint(2, 8)                     # 2‑7 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        consider(quick_score(cand), cand)

    # --- Include reference sequence if present ---------------------
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        consider(quick_score(a_ref), a_ref)
        print("[shape] reference sequence added to candidate pool")

    # ------------------------------------------------------------------
    # Report best from the exploratory phase
    best_candidates.sort(key=lambda x: x[0])
    print(f"[shape] best raw score after parametric sweep = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # ---------- 2. REFINEMENT (Adam optimiser) --------------------
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    # Process candidates ordered by their raw scores (most promising first)
    for rank, (raw_score, a0) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine] start candidate #{rank+1} (raw score {raw_score:.6f})")
        remaining_time = deadline - time.time()
        # Allocate a reasonable number of iterations based on remaining time
        max_iters = int(min(300_000, remaining_time * 1300))
        refined_a, refined_score = _refine_adam(
            a0, deadline, n, size_fft, target_sum, max_iters
        )
        print(f"[refine] candidate #{rank+1} refined score = {refined_score:.6f}")
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---------- 3. FINAL RANDOM BLOCK SWEEP -----------------------
    while time.time() < deadline:
        nb = np.random.randint(2, 8)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")
        # fast‑exit guard
        if time.time() + 0.02 > deadline:
            break

    # ------------------------------------------------------------------
    # Final report
    print(f"[final] best score achieved = {global_best_score:.6f}")

    # Convert to plain Python list as required by the external evaluator
    return global_best_a.tolist()
