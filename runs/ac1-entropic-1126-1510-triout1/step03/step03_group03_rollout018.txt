"""
Propose a low‑score step‑function sequence.

The algorithm is a multi‑stage optimiser:
  1.  Choose a problem size n (use the reference length if available,
      otherwise a moderately large size ~5000).
  2.  Parameterise the heights a_i as a soft‑max of free variables y_i,
      guaranteeing non‑negativity and a fixed ℓ¹‑norm S = √(2·n).
  3.  For several diverse initial y‑vectors (uniform, decreasing,
      increasing, many random draws and the reference if present) run
      a continuation scheme:
          * initialise the temperature μ as a multiple of the current
            maximum convolution value;
          * minimise the smooth‑max surrogate
                f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)
            using L‑BFGS‑B (Jacobian supplied);
          * after each L‑BFGS run evaluate the true score
                2·n·max(conv)/(∑a)²   (via evaluate_sequence) and keep
            the best candidate;
          * cool the temperature geometrically (μ ← 0.7·μ) until μ becomes
            tiny (≈1e‑5).
  4.  A cheap stochastic block‑search explores piece‑wise‑constant
      candidates with 2‑14 random blocks.
  5.  A final projected sub‑gradient descent on the hard‑max objective
      (μ = 0) refines the best solution for a few thousand iterations.
  6.  The best feasible sequence found before the deadline is returned.

All steps respect the 1000 s time budget, abort early when the deadline
is reached and print progress information.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective and gradient w.r.t. the softmax parameters y.

    Returns
    -------
    f_val : float
        Smooth surrogate value.
    grad_y : np.ndarray
        Gradient ∂f/∂y (shape (n,)).
    """
    # probabilities on the simplex and actual heights
    p = _softmax(y)                # Σp = 1, p_i ≥ 0
    a = S * p                      # Σa = S

    # full convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)
    max_c = conv.max()

    # soft‑max weights for the smooth surrogate
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max (μ = 0) – uniform distribution on all maximisers
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (g_a[i] = 2 Σ_k w_k a_{k-i})
    conv_wr = np.convolve(w, a[::-1])            # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]         # slice length n

    # chain rule to obtain gradient w.r.t. y
    inner = np.dot(g_a, p)                       # scalar = Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize (returns (value, gradient))."""

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score non‑negative step‑function within the time budget."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size ----------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
        print(f"[info] using reference length n = {n}")
    except Exception:
        n = 5000
        have_ref = False
        print(f"[info] no reference found – using default n = {n}")

    S = math.sqrt(2.0 * n)               # ℓ¹‑norm that makes the score = max(conv)

    # ------------------------------------------------------------------
    # bookkeeping ------------------------------------------------------
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")
    best_y = None                        # store the y‑vector of the best point

    # ------------------------------------------------------------------
    # generate a collection of diverse initial y‑vectors ----------------
    init_y = []

    # 0. uniform (all heights equal)
    init_y.append(np.zeros(n, dtype=np.float64))

    # 1. decreasing V‑shape (large at the left, tiny at the right)
    vdec = np.arange(n, dtype=np.float64) + 1.0
    init_y.append(np.log(vdec + 1e-6))

    # 2. increasing V‑shape (large at the right)
    vinc = np.arange(n, 0, -1, dtype=np.float64) + 1.0
    init_y.append(np.log(vinc + 1e-6))

    # 3‑7. several random Gaussian initialisations
    for _ in range(5):
        init_y.append(np.random.randn(n) * 0.5)

    # 8. reference if available (projected onto the simplex of size S)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        eps = 1e-12
        init_y.append(np.log(a_ref + eps))

    # ------------------------------------------------------------------
    # continuation / L‑BFGS parameters ---------------------------------
    mu_factor = 0.7            # slower cooling than the previous code
    mu_min = 1e-5
    maxiter_per_mu = 200       # L‑BFGS‑B iterations per temperature

    # ------------------------------------------------------------------
    # optimisation loop over all initial points -------------------------
    for start_id, y0 in enumerate(init_y):
        if time.time() > deadline:
            break

        print(f"[init {start_id}] starting continuation from this seed")
        y = y0.copy()

        # initialise temperature from the current convolution maximum
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        if mu <= 0.0:
            mu = 1.0

        # geometric cooling loop
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x

            # evaluate the *exact* score
            a = S * _softmax(y)
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_y = y.copy()
                print(f"[improve] new best score = {best_score:.6f}")

            # cool the temperature
            mu *= mu_factor

    # ------------------------------------------------------------------
    # if nothing succeeded, fall back to a uniform candidate
    if best_score == float("inf"):
        best_a = S * np.ones(n) / n
        best_score = evaluate_sequence(best_a.tolist())
        best_y = np.zeros(n, dtype=float)
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # cheap random block search (more exhaustive than the reference code)
    remaining = deadline - time.time()
    if remaining > 2.0:
        block_trials = int(min(8000, remaining * 300))
        print(f"[block] performing {block_trials} random block trials")
        for _ in range(block_trials):
            nb = np.random.randint(2, 15)                     # 2‑14 blocks
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                best_y = np.log(_softmax(np.log(cand + 1e-12)) + 1e-12)  # store a compatible y
                print(f"[block improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final projected‑gradient polishing on the hard max (μ = 0)
    # (sub‑gradient descent with a decreasing step size)
    a = best_a.copy()
    eta0 = S * 0.6 / n          # initial step roughly proportional to the mean height
    decay = 5e-5
    it = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        conv_wr = np.convolve(w, a[::-1])
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        # step size schedule (poly‑decay)
        eta = eta0 / (1.0 + decay * it)

        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[grad refine] new best score = {best_score:.6f}")

        a = a_new
        it += 1
        if it > 20000:          # safety cap
            break

    # ------------------------------------------------------------------
    print(f"[final] best score achieved = {best_score:.6f}")
    return best_a.tolist()
