def smooth_opt(a, n, time_limit):
    start = time.time()
    # hyperparameters
    velocity = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5  # step base
    alpha = 5.0
    alpha_factor = 1.005
    alpha_max = 5000.0
    it = 0
    best_a = a.copy()
    best_max = _fft_self_convolve(a).max()
    while time.time() < time_limit:
        it += 1
        # compute convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
        # softmax weights
        exp_term = np.exp(alpha * (b - cur_max))
        # avoid overflow; clip exponent maybe:
        # stable compute by subtract max ensures highest term = exp(0)=1; others <=1
        w = exp_term / exp_term.sum()
        # gradient
        a_rev = a[::-1]
        conv_rev = _fft_convolve(a_rev, w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
        # momentum
        velocity = beta * velocity + (1-beta) * grad
        # step size
        eta = eta0 / np.sqrt(it)
        a = a - eta * velocity
        # projection to simplex
        a = _project_onto_simplex(a, s=1.0)
        # annealing
        if alpha < alpha_max:
            alpha *= alpha_factor
        # jitter occasional
        if it % 500 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)
    # after loop, return best_a
    return best_a

def generate_sym_beta(n, p):
    i = np.arange(n, dtype=np.float64)
    vals = ( (i + 0.5) * (n - i - 0.5) ) ** p
    # ensure non-negative
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0:
        # fallback uniform
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals

"""
Propose a low‑score step‑function.

The score is 2·n·max(a∗a)/(sum(a))².  Because the objective is
scale‑invariant we restrict ourselves to probability vectors
(sum a = 1, aᵢ ≥ 0).  The convolution max is minimised by a
distribution whose self‑convolution is as flat as possible.
A symmetric Beta‑type family
        aᵢ ∝ ((i+½)(n‑i‑½))^p
covers a wide range of shapes (uniform → p=0, arcsine → p=‑0.5,
edge‑heavy → p<-0.5).  We scan this one‑parameter family (and a few
mixtures with the uniform distribution), keep the best few candidates
according to a fast FFT‑based estimate of max(a∗a) and refine them by
projected gradient descent on a smooth approximation of the max.
The refinement uses the same soft‑max surrogate as the reference code
but with tuned hyper‑parameters and a brief momentum scheme.
All heavy work (FFT convolutions) is O(n log n) and inexpensive, so we
can explore several lengths and many shapes within the 1000 s budget.
The best candidate found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a via real FFT, returning the first 2·n‑1 values."""
    n = a.shape[0]
    # next power of two ≥ 2·n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a∗b via real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_sym_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric Beta‑type distribution:
        a_i ∝ ((i+½)*(n‑i‑½))^p .
    The exponent p may be negative (edge heavy) or zero (uniform).
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    # Guard against overflow / underflow
    with np.errstate(divide='ignore'):
        vals = np.power(prod, p, dtype=np.float64)
    # Replace possible NaNs (e.g. 0**negative) by a large number
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        # fall back to uniform if the shape collapsed
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """
    Projected gradient descent on the soft‑max surrogate of max(a∗a).
    Uses a momentum term and an annealing schedule for the soft‑max
    temperature α.  The function returns the best (lowest max‑convolution)
    vector observed before the deadline.
    """
    a = a0.copy()
    velocity = np.zeros_like(a)
    beta = 0.8               # momentum coefficient
    eta0 = 0.6               # base step size
    alpha = 10.0             # initial soft‑max temperature
    alpha_factor = 1.005
    alpha_max = 4000.0
    it = 0
    best_a = a.copy()
    best_max = _fft_self_convolve(a).max()

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()

        # soft‑max weights (stable because we subtract cur_max)
        exp_term = np.exp(alpha * (b - cur_max))
        w = exp_term / exp_term.sum()

        # gradient of the smooth surrogate:
        # grad_i = 2 * Σ_j a_j w_{i+j}
        a_rev = a[::-1]
        conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection
        a -= eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # anneal the temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape flat regions
        if it % 800 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_a
