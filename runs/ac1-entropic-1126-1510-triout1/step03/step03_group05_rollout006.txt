"""
Optimisation using smoothed max (log‑sum‑exp) and Adam.
Multiple random starts, block‑structured candidates and
projected‑simplex updates.
"""

import math
import time
import numpy as np

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    # Follow the algorithm of Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, put all mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using real FFT, result length len(u)+len(v)-1."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]

def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)

def _smooth_max_and_grad(a: np.ndarray, size_fft: int, temperature: float):
    """
    Returns a tuple (smooth_max, conv, grad) where ``smooth_max`` is
    T·log‑sum‑exp of the full self‑convolution, ``conv`` is the raw
    convolution, and ``grad`` is the (sub‑)gradient of the smooth
    max w.r.t. a.
    """
    conv = _self_convolution(a, size_fft)                     # length 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    shifted = (conv - maxc) / temperature
    exp_shift = np.exp(shifted)
    sum_exp = exp_shift.sum()
    smooth = maxc + temperature * np.log(sum_exp)

    # probability vector – the soft‑max over the convolution entries
    p = exp_shift / sum_exp                                   # sums to 1

    # grad_i = 2·∑_k p_k·a_{k‑i}
    rev_a = a[::-1]
    convp = _conv_fft(p, rev_a, size_fft)                     # length 3n‑2
    grad = 2.0 * convp[len(a) - 1: len(a) - 1 + len(a)]       # slice n entries
    return smooth, conv, grad

def _descent_from(init_a: np.ndarray,
                  size_fft: int,
                  target_sum: float,
                  deadline: float):
    """
    Adam optimisation of the smoothed max, projected onto the simplex.
    Returns the best vector seen and its exact evaluate_sequence score.
    """
    a = init_a.copy()
    n = a.shape[0]

    # hyper‑parameters (they are quite robust)
    temperature = max(_self_convolution(a, size_fft).max() * 0.5, 1e-6)
    temp_min = 1e-6
    temp_decay = 0.9995

    lr0 = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_local_a = a.copy()
    best_local_score = float('inf')
    it = 0

    while time.time() < deadline and it < 50000:
        # one Adam step on the smooth objective
        smooth, conv, grad = _smooth_max_and_grad(a, size_fft, temperature)

        # exact score (what the outer harness will see)
        cur_score = float(2.0 * n * conv.max() / (a.sum() ** 2))
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_a = a.copy()

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        it += 1
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step (learning‑rate schedule decays together with temperature)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_to_simplex(a, target_sum)

        # temperature annealing
        temperature = max(temperature * temp_decay, temp_min)

        # occasional random kick to escape flat regions
        if it % 4000 == 0:
            a = a + np.random.randn(n) * target_sum * 5e-4
            a = _project_to_simplex(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)

    return best_local_a, best_local_score

    """
    Search for a low‑score non‑negative step function.
    Uses several randomised starts, Adam optimisation on a
    smoothed max of the self‑convolution and a final random‑block sweep.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ----------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
        have_ref = True
    except Exception:

"""
Optimisation using a smoothed max (log‑sum‑exp) and Adam.
Multiple random starts (uniform, reference, analytic families,
random block structures) are refined by projected‑simplex Adam.
A cheap final random‑block sweep is performed before returning the
best sequence found within the allotted time budget.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using a real FFT; result length = len(u)+len(v)-1."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a∗a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray,
                         size_fft: int,
                         temperature: float):
    """
    Returns (smooth_max, conv, grad) where

    * smooth_max is  T·log‑sum‑exp of the full self‑convolution,
    * conv       is the raw convolution (length 2·n‑1),
    * grad       = 2·∑_k p_k·a_{k‑i}  (i = 0,…,n‑1) with
                  p_k = exp(conv_k/T)/∑exp(conv/T).

    The gradient is obtained analytically and computed via FFT.
    """
    conv = _self_convolution(a, size_fft)            # length 2n‑1
    maxc = conv.max()
    # stable log‑sum‑exp
    shifted = (conv - maxc) / temperature
    exp_shift = np.exp(shifted)
    sum_exp = exp_shift.sum()
    smooth = maxc + temperature * np.log(sum_exp)

    # soft‑max weights (probability vector)
    p = exp_shift / sum_exp

    # grad_i = 2·∑_k p_k·a_{k‑i}
    rev_a = a[::-1]
    convp = _conv_fft(p, rev_a, size_fft)      # length 3n‑2
    grad = 2.0 * convp[len(a) - 1: len(a) - 1 + len(a)]
    return smooth, conv, grad


# ----------------------------------------------------------------------
def _descent_from(init_a: np.ndarray,
                  size_fft: int,
                  target_sum: float,
                  deadline: float):
    """
    Adam optimisation of the smoothed max, projected onto the simplex.
    Returns the best vector seen (exact evaluate_sequence score) together
    with that score.
    """
    a = init_a.copy()
    n = a.shape[0]

    # hyper‑parameters (robust across runs)
    temperature = max(_self_convolution(a, size_fft).max() * 0.5, 1e-6)
    temp_min = 1e-6
    temp_decay = 0.9995

    lr0 = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = float('inf')
    it = 0

    while time.time() < deadline and it < 50000:
        smooth, conv, grad = _smooth_max_and_grad(a, size_fft, temperature)

        # exact objective (what the outer harness uses)
        cur_score = float(2.0 * n * conv.max() / (a.sum() ** 2))
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        it += 1
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_to_simplex(a, target_sum)

        # anneal temperature
        temperature = max(temperature * temp_decay, temp_min)

        # occasional random kick to escape plateaus
        if it % 4000 == 0:
            a = a + np.random.randn(n) * target_sum * 5e-4
            a = _project_to_simplex(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)

    return best_a, best_score
