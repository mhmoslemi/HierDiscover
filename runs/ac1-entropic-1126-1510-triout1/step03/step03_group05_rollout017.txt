def _smooth_gradient(a, conv, eps, size_fft):
    # conv: a*a array (length 2n-1)
    max_val = conv.max()
    # compute weights
    # diff = conv - max_val (<=0)
    w = np.exp((conv - max_val) / eps)  # safe
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        # fallback to uniform weighting
        w = np.ones_like(conv) / conv.shape[0]

Propose a non‑negative step‑function (list of heights) with a

Algorithm
---------
1.  Choose a problem size n (use the reference sequence if it exists,

    Instead of a pure sub‑gradient on the active set, a soft‑max (log‑sum‑exp)

    projected back onto the simplex.  Random perturbations are injected
    when progress stalls.

4.  **Final random‑block refinement** – a handful of fresh block‑structured
    candidates are sampled to see if any beat the current best.

The whole search respects a 1000 s budget, prints intermediate scores and
returns the best list of heights found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution via FFT; output length = len(u)+len(v)-1."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _gradient_at_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """Sub‑gradient when the objective is max_i (a*a)_i."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    if max_idxs.size == 0:
        return grad
    inv = 1.0 / max_idxs.size
    for k in max_idxs:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        rng = np.arange(i0, i1 + 1, dtype=int)
        grad[rng] += 2.0 * a[k - rng]
    grad *= inv
    return grad


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray,
                     eps: float, size_fft: int) -> np.ndarray:
    """
    Gradient of ε·log Σ exp(conv/ε).  Implemented via one FFT.
    If eps is too small, falls back to the exact sub‑gradient.
    """
    max_val = conv.max()
    if eps < 1e-8:
        # exact sub‑gradient on the active set
        idx = np.where(conv >= max_val - max_val * 1e-12)[0]
        return _gradient_at_max(a, idx)

    # soft‑max weights (stable, because conv-max ≤ 0)
    w = np.exp((conv - max_val) / eps)
    sum_w = w.sum()
    if sum_w == 0.0 or np.isnan(sum_w):
        idx = np.where(conv >= max_val - max_val * 1e-12)[0]
        return _gradient_at_max(a, idx)
    w /= sum_w

    # gradient = 2 * (w ⋆ a) , where ⋆ is correlation.
    # Correlation can be obtained by convolving a with reversed w.
    w_rev = w[::-1]
    grad_full = _conv_fft(a, w_rev, size_fft)                # length 3·n‑2
    L = conv.shape[0]                                      # = 2·n‑1
    # grad_i = 2 * grad_full[L-1-i],  i = 0 … n‑1
    idx_start = L - 1
    grad = 2.0 * grad_full[idx_start - np.arange(a.shape[0])]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score non‑negative sequence."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)     # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3130                        # fallback length
        have_ref = False

    target_sum = math.sqrt(2.0 * n)    # normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())   # ≥ 3 n‑2 for gradient steps

    # ------------------------------------------------------------------
    # keep the best candidate seen so far
    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # helper to evaluate the objective (max‑convolution)
    def _eval(arr: np.ndarray) -> float:
        return float(_self_convolution(arr, size_fft).max())

    # ------------------------------------------------------------------
    # 1. PARAMETRIC SHAPE SEARCH
    # ------------------------------------------------------------------
    # uniform baseline
    a = np.ones(n, dtype=float)
    a = _project_to_simplex(a, target_sum)
    sc = _eval(a)
    best_a, best_score = a.copy(), sc
    print(f"[shape] uniform score = {sc:.6f}")

    # reference if available
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)
        ref = np.maximum(ref, 0.0)
        ref = _project_to_simplex(ref, target_sum)
        sc = _eval(ref)
        if sc < best_score:
            best_a, best_score = ref.copy(), sc
        print(f"[shape] reference score = {sc:.6f}")

    N_SAMPLES = 200   # a bit more samples per family

    # ----- families -----
    # 1) inverse‑sqrt (single‑ended)
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # 2) double‑ended inverse‑sqrt
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        i = np.arange(n, dtype=float)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # 3) double‑ended power‑law
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        p = np.random.rand() * 2.5 + 0.1
        i = np.arange(n, dtype=float)
        a = (1.0 / (i + beta + 1.0) ** p +
             1.0 / ((n - 1 - i) + beta + 1.0) ** p)
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # 4) double‑ended exponential decay
    for _ in range(N_SAMPLES):
        decay = np.random.rand() * 0.012 + 0.0005
        i = np.arange(n, dtype=float)
        a = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # 5) cosine‑based (large at ends)
    for _ in range(N_SAMPLES):
        p = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # 6) triangular (linear decreasing from ends to centre)
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 5.0
        mid = (n - 1) / 2.0
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = (mid - dist + beta)            # decreasing towards centre
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # 7) logistic (sigmoid‑shaped double‑ended)
    for _ in range(N_SAMPLES):
        scale = np.random.rand() * 0.5 + 0.1
        shift = np.random.rand() * 5.0
        i = np.arange(n, dtype=float)
        d = np.minimum(i, n - 1 - i)
        a = 1.0 / (1.0 + np.exp((d - shift) / scale))
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # 8) two‑Gaussian peaks at the ends
    for _ in range(N_SAMPLES):
        sigma = np.random.rand() * 300.0 + 100.0
        i = np.arange(n, dtype=float)
        a = np.exp(- (i ** 2) / (2 * sigma ** 2)) + np.exp(- ((i - (n - 1)) ** 2) / (2 * sigma ** 2))
        a = _project_to_simplex(a, target_sum)
        sc = _eval(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # 9) random block candidates (few large blocks)
    for _ in range(300):
        nb = np.random.randint(2, 7)            # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _eval(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    print(f"[shape] best score after families = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. SMOOTHED ADAM‑BASED DESCENT
    # ------------------------------------------------------------------
    a = best_a.copy()
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.12                # aggressive initial step size
    it = 0
    no_improve = 0

    # Smoothing schedule
    conv = _self_convolution(a, size_fft)
    cur_max = conv.max()
    eps = cur_max * 0.5                # start with half the current peak
    eps_min = cur_max * 1e-5
    eps_decay = 0.9995                 # exponential cooling

    while time.time() < deadline and it < 1_200_000:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            no_improve = 0
            print(f"[{it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # update smoothing parameter
        eps = max(eps * eps_decay, eps_min)

        # gradient (smoothed or exact sub‑gradient)
        grad = _smooth_gradient(a, conv, eps, size_fft)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / (1.0 + 1e-5 * it)      # slow decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick to escape plateaus
        if (it % 5000 == 0 and it > 0) or no_improve > 4000:
            noise = np.random.randn(n) * target_sum * 4e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[{it:07d}] random perturbation")

        a = a_new
        it += 1

    print(f"[adam] finished after {it} iterations, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM‑BLOCK REFINEMENT
    # ------------------------------------------------------------------
    for _ in range(250):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _eval(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
