def refine_candidate_adam(a0, deadline, n, size_fft, target_sum, max_iters):

    a = a0.copy()

    best_a = a.copy(); best_score = quick_score(a)

    m = np.zeros_like(a); v = np.zeros_like(a)

    beta1 = 0.9; beta2 = 0.999; epsilon = 1e-8

    alpha = target_sum * 0.2 / n   # base step size

    # initial temperature T = max(1e-4, (conv.max() - conv.min())*0.4)

    iteration = 1

    while time.time() < deadline and iteration <= max_iters:

        conv = _self_convolution(a, size_fft)  # shape len 2n-1

        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:

            best_score = cur_score; best_a = a.copy()

        # compute smoothed weights

        shifted = conv - conv.max()

        # compute weights via softmax: w_i = exp(shifted_i / T)

        # avoid overflow: shift to <=0; divide by T; clip; compute exp; normalize

        exp_arg = shifted / max(T, 1e-8)

        exp_arg = np.clip(exp_arg, -50, 0)

        w = np.exp(exp_arg); w_sum = w.sum(); w /= w_sum if w_sum else np.full_like(w, 1.0/w.size)

        # compute gradient

        grad = _weighted_gradient(a, w, size_fft)

        # Adam updates

        m = beta1 * m + (1 - beta1) * grad

        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** iteration)

        v_hat = v / (1 - beta2 ** iteration)

        step = alpha / (np.sqrt(v_hat) + epsilon)

        a_new = a - step * m_hat

        # project onto simplex

        a_new = _project_to_simplex(a_new, target_sum)

        a = a_new

        # random perturbations (optional)

        if iteration % 5000 == 0:

            noise = np.random.randn(n) * target_sum * 0.001

            a = _project_to_simplex(a + noise, target_sum)

        # temperature decay

        T = max(1e-4, T * 0.9995)

        iteration += 1

    return best_a, best_score

"""
Propose a non‑negative step function with a low self‑convolution peak.

Algorithm
---------
1. Choose a problem size `n` (>= 3000). If a reference sequence
   `height_sequence_1` is present we up‑sample it to the chosen length;
   otherwise we use `n = 5000`.

2. Generate a large pool of parametric candidates (inverse‑sqrt, power‑law,
   exponential, Gaussian, beta‑distribution, sinusoidal, random block …)
   and keep the `K` best according to the exact score
   `2·n·max(conv)/sum(a)^2`.

3. Refine each of the top candidates with a smooth‑max descent.
   The smooth max is approximated by a soft‑max (temperature `T`);
   the gradient of the smoothed objective is obtained analytically via
   FFTs.  An Adam optimiser (adaptive step size) together with Euclidean
   projection onto the simplex `{a≥0, Σa = √(2n)}` is used.
   Temperature is slowly cooled and occasional random kicks help escaping
   plateaus.

4. A final fast random‑block sweep tries to improve the current best
   solution.

The whole routine respects the 1000 s budget and returns the best
sequence found as a plain Python list of floats.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution of real vectors u and v using FFT."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Compute a * a (full convolution)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   with respect to a.
    (w must sum to 1).  Exact formula: 2·(w ↔ a_rev) restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)      # length = 3·n‑2
    start = len(w) - len(a)                   # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _score_exact(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact evaluation score (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max descent with Adam optimisation.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_exact(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base step size (order of the average entry)
    alpha0 = 0.5 * target_sum / n

    # initialise temperature from the spread of the initial convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 1
    while time.time() < deadline:
        # evaluate current point
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[adam] it {it:,} new best = {best_score:.6f}")

        # soft‑max weights (smooth approximation of the max)
        shifted = conv - conv.max()          # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)   # stability
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        grad = _weighted_gradient(a, w, size_fft)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha0 / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_to_simplex(a_new, target_sum)

        # random perturbation to help escape plateaus
        if it % 5000 == 0:
            noise = np.random.randn(n) * target_sum * 1e-3
            a_new = _project_to_simplex(a_new + noise, target_sum)

        a = a_new

        # cooldown
        T = max(1e-4, T * 0.9995)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects a 1000 s budget and returns
    the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Determine problem size ------------------------------------------------
    try:
        # if the reference sequence is present, up‑sample it
        ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n_ref = ref.shape[0]
        n = max(5000, n_ref)  # work with a larger size for better potential
        # linear interpolation to length n
        x_old = np.linspace(0, n_ref - 1, n_ref)
        x_new = np.linspace(0, n_ref - 1, n)
        a0_ref = np.interp(x_new, x_old, ref)
        a0_ref = _project_to_simplex(a0_ref, math.sqrt(2.0 * n))
        have_ref = True
    except Exception:
        n = 5000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)

    # FFT size >= 3·n‑2 (convolution of length‑(2n‑1) with length‑n)
    global _size_fft_global
    _size_fft_global = 1 << ((3 * n - 2).bit_length())

    idx = np.arange(n)

    # ------------------------------------------------------------------
    # Helper for quick exact scoring (no security checks)
    def quick_score(a: np.ndarray) -> float:
        conv = _self_convolution(a, _size_fft_global)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # Keep the K best raw candidates
    K = 15
    best_candidates: list[tuple[float, np.ndarray]] = []

    def consider(score: float, a: np.ndarray) -> None:
        """Maintain a sorted top‑K list (ascending scores)."""
        if len(best_candidates) < K:
            best_candidates.append((score, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if score < best_candidates[-1][0]:
                best_candidates[-1] = (score, a.copy())
                best_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 1. PARAMETRIC SHAPE SEARCH ----------------------------------------
    # Uniform baseline
    a = np.ones(n)
    a = _project_to_simplex(a, target_sum)
    consider(quick_score(a), a)
    print(f"[init] uniform score = {best_candidates[-1][0]:.6f}")

    # Helper to generate a random candidate from a callable creator
    def sweep(creator, trials: int = 80):
        for _ in range(trials):
            a = creator()
            a = _project_to_simplex(a, target_sum)
            consider(quick_score(a), a)

    # Inverse‑sqrt (single‑ended)
    sweep(lambda: 1.0 / np.sqrt(idx + np.random.rand() * 10.0 + 1.0), 120)

    # Double‑ended inverse‑sqrt
    sweep(
        lambda: (
            1.0 / np.sqrt(idx + np.random.rand() * 10.0 + 1.0)
            + 1.0
            / np.sqrt((n - 1 - idx) + np.random.rand() * 10.0 + 1.0)
        ),
        120,
    )

    # Power‑law
    sweep(
        lambda: 1.0
        / (idx + np.random.rand() * 10.0 + 1.0)
        ** (np.random.rand() * 1.7 + 0.3),
        120,
    )

    # Exponential decay
    sweep(
        lambda: np.exp(-np.random.rand() * 0.01 - 0.0005 * idx),
        120,
    )

    # Gaussian bump
    sweep(
        lambda: np.exp(
            -0.5
            * ((idx - np.random.rand() * n) / (np.random.rand() * n * 0.2 + 1e-3)) ** 2
        ),
        80,
    )

    # Beta‑like shape (a·(i+1)^α·(n‑i)^β)
    sweep(
        lambda: (idx + 1) ** (np.random.rand() * 4.0 - 2.0)
        * (n - idx) ** (np.random.rand() * 4.0 - 2.0),
        80,
    )

    # Cosine (large at ends)
    sweep(
        lambda: np.abs(np.cos(np.pi * idx / (n - 1))) ** (np.random.rand() * 5.0 + 0.5),
        80,
    )

    # Sine (zero at ends, peak middle)
    sweep(
        lambda: np.sin(np.pi * idx / (n - 1)) ** (np.random.rand() * 5.0 + 0.5),
        80,
    )

    # Linear ramp (ascending)

    # Quadratic / polynomial

    # Random block candidates (few large blocks)

        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)

        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        consider(quick_score(cand), cand)

    # Include the up‑sampled reference if available

    # Report best from the exploratory phase

    print(f"[shape search] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. REFINEMENT (Adam‑based smooth‑max descent) --------------------
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    # Iterate over the top few candidates; allocate time fairly.

        refined_a, refined_score = _refine_adam(

            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM‑BLOCK SWEEP ---------------------------------------
    while time.time() < deadline - 0.1:
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)

        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
