def _smooth_max_and_grad(a, beta, fft_len):
    # a: np.ndarray shape (n,)
    # Returns approx_max, gradient np.ndarray shape (n,)
    # Compute FFT of a
    Fa = np.fft.rfft(a, n=fft_len)
    # self-convolution
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:2*n-1]
    # logsumexp
    max_conv = conv.max()
    # Using stable logsumexp with beta factor:
    x = beta * conv
    max_x = x.max()
    lse = max_x + np.log(np.sum(np.exp(x - max_x)))
    approx_max = lse / beta
    # weights w_k = exp(beta * conv_k) / sum(exp(beta * conv_k)) = exp(x - max_x) * exp(max_x) ??? Actually exp(x)/exp(lse*beta?) Wait compute w = np.exp(beta * conv - lse) ; lse = log-sum-exp * beta? Actually lse = log sum exp(beta conv). So w = exp(beta * conv - lse)

    # correlation gradient: grad = 2 * correlation(w, a)
    # compute FFT of w
    Fw = np.fft.rfft(w, n=fft_len)

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]

def _smooth_max_and_gradient(a: np.ndarray, beta: float, fft_len: int,
                             lam: float) -> tuple[float, np.ndarray, float]:
    """
    Returns
       approx_max  – smooth approximation of max_k (a*a)_k,
       grad        – gradient of the same quantity + λ·variance term,
       real_max    – actual (unsmoothed) max value of the convolution.
    """
    n = a.shape[0]
    Fa = np.fft.rfft(a, n=fft_len)

    # self‑convolution
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]

    # ----- smooth max (log‑sum‑exp) -----
    x = beta * conv
    max_x = np.max(x)
    lse = max_x + np.log(np.sum(np.exp(x - max_x)))
    approx_max = lse / beta
    # soft‑max weights (sum to 1)
    w = np.exp(beta * conv - lse)   # shape 2n‑1, non‑negative, sums to 1

    # ----- variance regulariser -----
    # average of the convolution (exactly sum(a)^2 / (2n‑1), but we use the
    # computed value to avoid numerical drift)
    avg = np.mean(conv)
    diff = conv - avg                # can be positive or negative

    # combine the two weight vectors; the variance term contributes
    # a factor 2·λ·diff to the effective weight used in the correlation
    w_total = w + 2.0 * lam * diff   # still length 2n‑1 (may contain negatives)

    # ----- gradient (2·correlation(weight,total) ) -----
    Fw = np.fft.rfft(w_total, n=fft_len)
    # correlation via conj(Fw) * Fa
    grad = 2.0 * np.fft.irfft(np.conj(Fw) * Fa, n=fft_len)[:n]

    real_max = float(conv.max())
    return approx_max, grad, real_max

    # --- reference sequence, if available --------------------------------
    if have_ref:
        ref = np.maximum(reference, 0.0)
        ref = _project_to_simplex(ref, target_sum)
        sc = _score(ref)
        if sc < best_score:
            best_a, best_score = ref.copy(), sc
        print(f"[init] reference score = {sc:.6f}")

    N_SAMPLES = 80

    # ----- inverse‑sqrt (single‑ended) -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- double‑ended inverse‑sqrt -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        i = np.arange(n, dtype=float)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- power‑law (double‑ended) -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        p = np.random.rand() * 2.5 + 0.1
        i = np.arange(n, dtype=float)
        a = (1.0 / (i + beta + 1.0) ** p +
             1.0 / ((n - 1 - i) + beta + 1.0) ** p)
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- exponential decay (double‑ended) -----
    for _ in range(N_SAMPLES):
        decay = np.random.rand() * 0.012 + 0.0005
        i = np.arange(n, dtype=float)
        a = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- cosine‑based (large at ends) -----
    for _ in range(N_SAMPLES):
        p = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- mixture of uniform & inverse‑sqrt -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        lam_mix = np.random.rand()
        a = lam_mix * np.ones(n, dtype=float) + (1.0 - lam_mix) * (
            1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- Beta‑distribution family (new) -----
    for _ in range(N_SAMPLES):
        alpha = np.random.rand() * 3.0 + 0.1   # (0.1, 3.1)
        beta = np.random.rand() * 3.0 + 0.1
        i = np.arange(n, dtype=float) + 0.5
        a = (i ** (alpha - 1.0)) * ((n - i + 0.5) ** (beta - 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- random block candidates (few large blocks) -----
    for _ in range(200):
        nb = np.random.randint(2, 7)            # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    print(f"[shape] best after families = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. ADAM‑BASED SMOOTHED‑MAX MINIMISATION WITH VARIANCE TERM
    # ------------------------------------------------------------------
    a = best_a.copy()

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature (β) schedule for the log‑sum‑exp
    beta_temp = 0.5
    beta_max = 80.0
    beta_mult = 1.006  # multiplicative increase per iteration

    # variance regulariser weight
    lam = 0.6

    # base learning rate (will decay slowly)
    lr0 = 0.15

    it = 0
    no_improve = 0
    max_iters = 250_000

    while time.time() < deadline and it < max_iters:
        # smooth‑max value, gradient, and exact max of the convolution
        approx_max, grad, cur_max = _smooth_max_and_gradient(
            a, beta_temp, fft_len, lam
        )

        # keep the best exact score seen so far
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:06d}] new best = {best_score:.6f}  β={beta_temp:.2f}")
        else:
            no_improve += 1

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # learning‑rate schedule (very mild decay)
        lr = lr0 / (1.0 + 1e-5 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # random perturbation when stuck
        if (it % 4000 == 0 and it > 0) or no_improve > 5000:
            noise = np.random.randn(n) * target_sum * 6e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[{it:06d}] random perturbation")

        a = a_new
        beta_temp = min(beta_max, beta_temp * beta_mult)
        it += 1

    print(f"[adam] finished after {it} iterations, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM‑BLOCK REFINEMENT
    # ------------------------------------------------------------------
    for _ in range(300):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
