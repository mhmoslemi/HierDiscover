"""
Multi‑phase optimizer for the non‑negative step‑function problem.

The algorithm consists of:
  1. Softmax‑parameterised L‑BFGS‑B continuation (smooth‑max surrogate)
     from several diverse initial points.
  2. Projected sub‑gradient descent on the true hard‑max objective.
  3. An alternating linear‑programming refinement:
        – maximise Σ g_j subject to (a * g)_k ≤ M  (M = max(a*a))
        – rescale g to the target ℓ¹‑norm and mix it with a.
  4. A cheap random‑block search for the remaining time budget.

All intermediate sequences are projected onto the simplex
   {x ≥ 0 , Σ x = √(2·n)}  so that the evaluation score equals the maximal
convolution entry.  The best sequence found before the deadline is returned.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max surrogate of max(a*a) and its gradient w.r.t. softmax parameters y.
    a = S * softmax(y)   satisfies Σ a = S.
    """
    p = _softmax(y)                 # simplex probabilities
    a = S * p                       # actual heights

    conv = np.convolve(a, a)        # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (g_a) : 2 Σ_k w_k a_{k-i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule to obtain ∂f/∂y
    inner = np.dot(g_a, p)                     # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """
    Callable for scipy.optimize.minimize:
    returns (smooth‑max value, gradient) for a given temperature μ.
    """

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, S: float, deadline: float, best_score: float,
               best_a: np.ndarray):
    """
    Perform one LP‑based refinement step.
    Returns (updated_a, updated_best_score, updated_best_a, improved_flag).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()

    # Build convolution matrix A where (A @ g)[k] = Σ_i a_i g_{k-i}
    offsets = np.arange(n, dtype=int)                     # 0 … n‑1
    diag_data = [np.full(n, a[d], dtype=float) for d in range(n)]
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csc')

    c = -np.ones(n)                                       # maximise Σ g_j
    b_ub = np.full(2 * n - 1, M, dtype=float)
    bounds = [(0, None)] * n

    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as exc:
        print("[LP] exception:", exc)
        return a, best_score, best_a, False

    if res.status != 0:
        # non‑optimal solution – abort LP refinement
        return a, best_score, best_a, False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, best_score, best_a, False

    # Scale g to have the same ℓ¹‑norm as a (which is S)
    g_scaled = g * (S / sum_g)

    # Mix the current a with the LP solution
    t = 0.5
    a_new = (1.0 - t) * a + t * g_scaled
    # the mixing preserves non‑negativity and the sum exactly S

    # Evaluate
    sc = evaluate_sequence(a_new.tolist())
    improved = False
    if sc < best_score:
        best_score = sc
        best_a = a_new.copy()
        improved = True
        print(f"[LP] new best = {best_score:.6f}")

    # Stop if time ran out
    if time.time() >= deadline:
        return a_new, best_score, best_a, improved

    return a_new, best_score, best_a, improved
