"""
Propose a low‑score non‑negative step‑function.

The algorithm consists of three stages:

1. **Multi‑restart smooth‑max continuation** –  Each restart starts from a
   different random point on the probability simplex (softmax
   parametrisation).  For a temperature μ the smooth surrogate of the
   max‑convolution is minimised with L‑BFGS‑B; μ is geometrically cooled
   (μ ← μ·μ_factor) until a very small value is reached.  After every
   L‑BFGS call the true hard‑max score is evaluated and the best feasible
   sequence is remembered.

2. **Heavy random‑block search** –  After the continuations a cheap
   “block‑wise” random search (piecewise‑constant candidates) is performed.
   Each block candidate is projected onto the simplex and tested; the
   best result is kept.

3. **Simulated‑annealing + projected sub‑gradient polishing** –  The
   current best sequence is further refined by a stochastic walk that
   adds Gaussian noise, projects onto the simplex and accepts improvements
   (or occasional uphill moves) according to a linear‑in‑time temperature
   schedule.  Finally a simple projected sub‑gradient descent on the hard
   max is executed for a few hundred iterations.

All operations respect the non‑negativity and sum‑normalisation
constraints; the final sequence is returned as a plain Python list.
"""

import time
import math
import random
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities -------------------------------------------------------
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}.
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1          # Python index → count
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _smooth_max_and_grad(y: np.ndarray, mu: float, n: int,
                         S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate of the convolution peak and its gradient w.r.t. y.
    """
    p = _softmax(y)                 # Σ p = 1
    a = S * p                       # Σ a = S

    conv = np.convolve(a, a)        # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        # smooth log‑sum‑exp trick
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight over all maximisers
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. heights a (see derivation in the description)
    conv_wr = np.convolve(w, a[::-1])        # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]     # slice length n

    # chain rule: ∂f/∂y = S * p * (g_a - ⟨g_a,p⟩)
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObj:
    """Callable for scipy.optimize; returns (value, gradient)."""

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _smooth_max_and_grad(y, self.mu, self.n, self.S)
        return f, g


def _evaluate(a: np.ndarray) -> float:
    """Wrapper around the supplied evaluate_sequence."""
    return evaluate_sequence(a.tolist())


# ----------------------------------------------------------------------
# Continuation optimisation ---------------------------------------------
# ----------------------------------------------------------------------
def _continuation_opt(y0: np.ndarray, n: int, S: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Runs the smooth‑max continuation from a starting point y0.
    Returns the best (feasible) a‑vector found and its hard‑max score.
    """
    best_a = None
    best_score = float("inf")
    y = y0.copy()

    # initialise temperature from the current (hard) max‑convolution
    a_cur = S * _softmax(y)
    mu = float(np.convolve(a_cur, a_cur).max())
    if mu <= 0.0:
        mu = 1.0

    mu_factor = 0.6            # slower cooling → more accurate descent
    mu_min = 1e-7
    maxiter_per_mu = 200       # L‑BFGS‑B iterations per temperature

    while mu > mu_min and time.time() < deadline:
        obj = _SmoothMaxObj(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu,
                         "disp": False,
                         "gtol": 1e-12}
            )
        except Exception as exc:
            print(f"[error] L‑BFGS failed: {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = _evaluate(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[cont] improved score = {best_score:.6f}")

        mu *= mu_factor

    # If nothing was ever accepted, fall back to the final y
    if best_a is None:
        best_a = S * _softmax(y)
        best_score = _evaluate(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random block generator -------------------------------------------------
# ----------------------------------------------------------------------
def _random_block_candidate(n: int, S: float) -> np.ndarray:
    """
    Generates a piecewise‑constant random candidate and projects it onto
    the simplex with total sum S.
    """
    nb = np.random.randint(2, 7)          # 2–6 blocks
    cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = np.random.rand(nb)
    cand = np.zeros(n, dtype=float)
    for blk, v in zip(blocks, vals):
        cand[blk] = v
    cand = _project_to_simplex(cand, S)
    return cand


# ----------------------------------------------------------------------
# Simulated annealing refinement -----------------------------------------
# ----------------------------------------------------------------------
def _annealing_refine(a0: np.ndarray, n: int, S: float,
                     current_best_score: float,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic walk with a simple linear temperature schedule.
    Returns the best sequence found during the walk.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = current_best_score

    total_time = deadline - time.time()
    start_time = time.time()
    # linear decay schedule
    sigma0 = 0.5 * S / n          # initial perturbation magnitude

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        # temperature linearly from 1 → 0
        T = max(0.0, 1.0 - frac)
        sigma = max(1e-8, sigma0 * (1.0 - frac))

        # propose a new point by adding Gaussian noise
        cand = a + sigma * np.random.randn(n)
        cand = np.maximum(cand, 0.0)          # keep non‑negative before projection
        cand = _project_to_simplex(cand, S)

        sc = _evaluate(cand)

        # Metropolis acceptance
        accept = False
        if sc < best_score:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            if T > 0.0:
                prob = math.exp(-(sc - best_score) / T)
                if random.random() < prob:
                    accept = True

        if accept:
            a = cand
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[anneal] new best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Hard‑max sub‑gradient polishing -----------------------------------------
# ----------------------------------------------------------------------
def _gradient_refine(a0: np.ndarray, n: int, S: float,
                    current_best_score: float,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = current_best_score

    # step‑size schedule parameters
    eta0 = S * 0.5 / n        # roughly the average entry magnitude
    decay = 1e-4
    it = 0

    while time.time() < deadline and it < 5000:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        conv_wr = np.convolve(w, a[::-1])
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = _evaluate(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[grad] improved to {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score
