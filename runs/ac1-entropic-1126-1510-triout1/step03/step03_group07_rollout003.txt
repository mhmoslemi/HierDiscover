"""
Propose Candidate – Improved Non‑Negative Step Function Optimiser
----------------------------------------------------------------
We minimise the evaluation score 2·n·max(conv)/sum² by fixing the ℓ₁‑norm
to S = √(2·n) (the score then equals max(conv)).  The optimisation
uses three stages:

1. **Initialisation** – several deterministic families of windows
   (uniform, triangular, Hann, Hamming, Blackman) and a random Dirichlet
   sample are projected onto the simplex of sum S.

2. **Projected Adam optimisation** – a smooth approximation of the
   max‑operator is built with a soft‑max weighting w ∝ exp(β·(conv−max)).
   The gradient of the surrogate objective is
        ∇ = 2·corr(w, a)
   (obtained with FFT‑based correlation).  Adam updates are performed on
   the raw coefficients, followed by a Euclidean projection onto the
   non‑negative simplex (preserving the ℓ₁‑norm).  β is annealed from
   0.5 to 200, while the learning‑rate decays from 0.6 to 0.01.

3. **Polishing (simulated annealing)** – random pairwise mass swaps
   (preserving the sum) are accepted if they improve the max‑convolution,
   otherwise they are accepted with a tiny probability that decays with a
   linear temperature schedule.  This helps escape shallow local minima.

The whole routine respects the time budget (default 1000 s) and keeps the
best sequence ever seen across all restarts.  The returned list is a plain
Python list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x ≥ 0, Σx = target}.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the non‑negative simplex with given L1 sum."""
    if target_sum <= 0:
        return np.zeros_like(v)
    # Duchi et al. (2008) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (length 2n‑1).
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    # size for convolution (next power of two)
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based correlation w ⋆ a (valid mode, length n).
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]  # = 2n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]  # length n (valid)


# ----------------------------------------------------------------------
# Random pairwise mass‑swap that preserves the L1 sum.
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two random indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    # Δ uniformly from (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    # clamp to keep entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Initialisation families (all return a vector of length N with ℓ1‑norm S).
# ----------------------------------------------------------------------
def _init_uniform(N: int, S: float) -> np.ndarray:
    a = np.full(N, S / N, dtype=float)
    a += 0.001 * S * np.random.rand(N) / N  # tiny noise
    return _project_onto_simplex(a, S)


def _init_triangular(N: int, S: float) -> np.ndarray:
    # symmetric triangle peaking in the centre
    idx = np.arange(N)
    centre = (N - 1) / 2.0
    a = 1.0 - np.abs(idx - centre) / centre
    a = np.maximum(a, 0.0)
    a = a.astype(float)
    a *= S / a.sum()
    return _project_onto_simplex(a, S)


def _init_hann(N: int, S: float) -> np.ndarray:
    idx = np.arange(N)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * idx / (N - 1)))
    a = a.astype(float)
    a *= S / a.sum()
    return _project_onto_simplex(a, S)


def _init_hamming(N: int, S: float) -> np.ndarray:
    idx = np.arange(N)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (N - 1))
    a = a.astype(float)
    a *= S / a.sum()
    return _project_onto_simplex(a, S)


def _init_blackman(N: int, S: float) -> np.ndarray:
    idx = np.arange(N)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * idx / (N - 1))
        + 0.08 * np.cos(4.0 * np.pi * idx / (N - 1))
    )
    a = a.astype(float)
    a *= S / a.sum()
    return _project_onto_simplex(a, S)


def _init_random_dirichlet(N: int, S: float) -> np.ndarray:
    # Gamma(1,1) = exponential → Dirichlet(1,…,1) = uniform on simplex
    g = np.random.gamma(shape=1.0, scale=1.0, size=N)
    a = g.astype(float)
    a = a * (S / a.sum())
    return _project_onto_simplex(a, S)

# Core optimisation for a single initialisation.
# ----------------------------------------------------------------------
def _optimize_one(
    a0: np.ndarray,
    S: float,
    deadline: float,
    max_grad_iters: int = 6000,
    max_polish_iters: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam‑style projected optimisation followed by simulated‑annealing polishing.
    Returns the best sequence found and its max‑convolution value.
    """
    N = a0.shape[0]
    a = a0.copy()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Schedules
    lr_start, lr_end = 0.6, 0.01
    beta_start, beta_end = 0.5, 200.0

    best_a = a.copy()
    best_score = _fft_convolve(a).max()

    it = 0
    while time.time() < deadline and it < max_grad_iters:
        it += 1

        # 1) convolution and soft‑max weights
        conv = _fft_convolve(a)
        max_conv = conv.max()

        t_frac = it / max_grad_iters
        beta = beta_start + (beta_end - beta_start) * t_frac
        lr = lr_start + (lr_end - lr_start) * t_frac

        # stable soft‑max (subtract max_conv)
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # 2) gradient of the soft‑max objective
        grad = 2.0 * _fft_correlate(w, a)  # shape (N,)

        # 3) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, S)

        a = a_new

        # occasional monitoring
        if it % 10 == 0:
            score = _fft_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Grad] iter {it:05d} – new best score {best_score:.6f}")

    # --------------------------------------------------------------
    # Polishing phase – simulated annealing with random pair swaps.
    # --------------------------------------------------------------
    curr_a = best_a.copy()
    curr_score = best_score

    # Temperature schedule (linear decay)
    T_start = 0.05 * curr_score
    T_end = 0.0
    polish_it = 0
    while time.time() < deadline and polish_it < max_polish_iters:
        polish_it += 1
        frac = polish_it / max_polish_iters
        T = T_start * (1.0 - frac) + T_end * frac
        sigma = 0.4 * (1.0 - frac) + 0.02  # amplitude of the swap

        cand = _random_pair_move(curr_a, sigma)
        cand_score = _fft_convolve(cand).max()
        delta = cand_score - curr_score

        accept = False
        if delta < 0:
            accept = True
        elif T > 0 and np.random.rand() < math.exp(-delta / T):
            accept = True

        if accept:
            curr_a = cand
            curr_score = cand_score
            if curr_score < best_score:
                best_score = curr_score
                best_a = curr_a.copy()
                print(f"[Polish] iter {polish_it:05d} – new best score {best_score:.6f}")

    return best_a, best_score
