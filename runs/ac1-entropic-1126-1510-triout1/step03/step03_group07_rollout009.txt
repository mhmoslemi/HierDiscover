"""
Propose a candidate step‑function sequence that minimizes the evaluation
score (maximal value of the self‑convolution).  The algorithm consists of
three stages:

1.  Initialise a long non‑negative vector on the ℓ₁‑simplex (sum = √(2·N)).
    If a previous construction is available in the global variable
    ``height_sequence_1`` it is used as a warm start.

2.  Perform a projected Adam optimisation on a smooth soft‑max
    approximation of the max‑convolution.  At iteration *t* we compute
    the convolution, build soft‑max weights wₖ∝exp(βₜ·(cₖ−c_max)),
    obtain the gradient ∇=2·corr(w,a) and take an Adam step.
    After each step we project back onto the non‑negative simplex.

3.  Polishing by many random pairwise mass swaps (the ℓ₁‑norm stays
    exactly preserved).  Every successful move that lowers the peak
    convolution is kept.

All heavy operations use FFT‑based convolution/correlation for speed.
The routine respects the 1000 s budget and returns the best sequence
found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (length 2N‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based valid‑mode correlation w * a  → length N
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                     # = 2N‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, n=nfft)
    A_rev = np.fft.rfft(a_rev, n=nfft)
    full = np.fft.irfft(W * A_rev, n=nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length N (valid)


# ----------------------------------------------------------------------
# Random pairwise mass‑swap (keeps ℓ₁‑norm exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two distinct indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])          # take from j, give to i
        new_seq = seq.copy()
        new_seq[i] += delta
        new_seq[j] -= delta
    else:
        delta = -delta                       # take from i, give to j
        delta = min(delta, seq[i])
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
    return new_seq

    # Polishing – random pairwise mass swaps.
    # --------------------------------------------------------------
    polish_iters = 12000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        # linear cooling of the swap magnitude
        sigma = 0.5 * (1.0 - it / polish_iters) + 0.01
        cand = _random_pair_move(best_seq, sigma)

        # compute its peak convolution
        conv = _fft_convolve(cand)
        score = conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[Polish {it:05d}] New best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Return result.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # safeguard against tiny negatives
    return best_seq.tolist()
