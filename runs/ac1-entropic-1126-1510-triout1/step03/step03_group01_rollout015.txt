    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / ((i + 0.5) * (n - i - 0.5))**gamma
    # ensure non-negative.
    vals[vals < 0] = 0
    vals_sum = vals.sum()
    if vals_sum == 0:
        vals = np.full(n, 1.0/n)
        vals_sum = n*(1.0/n)
    vals /= vals_sum
    return vals

def _distribution_beta(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid zero: use x**(p-1) for p<1; x=0 and x=1 cause infinite, but using epsilon.
    eps = 1e-12
    x = np.clip(x, eps, 1-eps)
    vals = x**(p-1) * (1 - x)**(p-1)
    vals_sum = vals.sum()
    if vals_sum == 0:
        vals = np.full(n, 1.0/n)
        vals_sum = n*(1.0/n)
    vals /= vals_sum
    return vals

"""
Propose a low‑score step‑function.

Algorithm overview
------------------
1. Choose a moderate length (≈2000–3000).  This keeps the linear‑program
   (LP) tractable while still allowing a good peak reduction.
2. Initialise with a symmetric “power‑law’’ distribution
   a_i ∝ ((i+½)(n‑i‑½))^{‑γ} (γ=0.5 gives the classic arcsine shape).
3. Iteratively tighten the peak of the self‑convolution:
   * Compute M = max(a*a).
   * Solve the LP   maximise 1ᵀg   subject to   a * g ≤ M   and   g ≥ 0.
     The constraints are linear in the unknown g because a is fixed.
   * Rescale g to keep the total mass unchanged, then mix
     a ← (1‑t)·a + t·g (t≈0.5) and re‑project onto the probability simplex.
   * Repeat a few times – each round strictly lowers the peak.
4. Refine the resulting vector with a few thousand Adam‑type gradient
   steps on a smooth surrogate of the max‑convolution (soft‑max with a
   slowly increasing temperature α).  After every step the vector is
   projected back onto the simplex.
5. Keep the best exact candidate seen, evaluate it with the provided
   ``evaluate_sequence`` and retain the globally best result over all
   tried lengths.
6. If time runs out a uniform distribution is returned as a safe fallback.

The whole search respects the 1000 s budget, prints progress messages
and finally returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# Optional SciPy – used for the linear programme.
try:
    from scipy.optimize import linprog
    from scipy import sparse
    _has_scipy = True
except Exception:  # pragma: no cover

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _distribution_power(n: int, gamma: float) -> np.ndarray:
    """
    Symmetric power‑law distribution: a_i ∝ ((i+0.5)(n-i-0.5))^{-γ}.
    Normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / ((i + 0.5) * (n - i - 0.5)) ** gamma
    # remove possible infinities (very small n)
    vals[np.isinf(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise 1ᵀg   subject to   a * g ≤ M   and   g ≥ 0.
    Returns the optimal g (or zeros on failure).  Only called when SciPy
    is available and the problem size is modest.
    """
    if not _has_scipy:
        return np.zeros_like(a)

    n = a.shape[0]
    # Build the convolution matrix A (size (2n‑1) × n) as a sum of
    # constant‑value diagonals.
    diagonals = [np.full(n, a[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = sparse.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme: minimise -1ᵀg  ⇔  maximise 1ᵀg
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})

    if not res.success or res.x is None:
        return np.zeros_like(a)

    g = np.asarray(res.x, dtype=np.float64)
    g[g < 0] = 0.0
    return g

    """Fast approximation of the evaluation score (sum(a) is assumed ≈1)."""
    n = a.shape[0]
    max_b = _fft_self_convolve(a).max()
    return 2.0 * n * max_b

                sum_g = g.sum()
                sum_a = a.sum()
                # If the LP cannot increase the total mass, stop early.
                if sum_g <= sum_a * (1.0 + 1e-12):
                    break
                # Rescale g to keep the total mass equal to sum_a.
                g = g * (sum_a / sum_g)
                # Mix the new direction into a.
                t = 0.5
                a = (1.0 - t) * a + t * g
                a = _project_onto_simplex(a)

                # Re‑evaluate the peak.
                M_new = _fft_self_convolve(a).max()
                score_new = 2.0 * n * M_new
                if score_new < best_score_local:
                    best_score_local = score_new
                    best_seq_local = a.copy()
                    best_max_local = M_new
                    M = M_new
                    print(f"[{time.time():.1f}] len={n} LP iter={lp_iter} "
                          f"score≈{best_score_local:.6f}")
                else:
                    # No improvement – stop the LP loop.
                    break

        # ---------- 3) Adam‑type gradient refinement on the smooth max ----------
        # Hyper‑parameters
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        eta = 0.5            # base step size
        alpha = 5.0          # start temperature for soft‑max
        alpha_factor = 1.005
        alpha_max = 2000.0
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        it = 0
        while time.time() < slice_deadline:
            it += 1
            # Self‑convolution and its current maximum.
            b = _fft_self_convolve(a)
            cur_max = b.max()
            cur_score = 2.0 * n * cur_max

            if cur_score < best_score_local:
                best_score_local = cur_score
                best_seq_local = a.copy()
                best_max_local = cur_max
                print(f"[{time.time():.1f}] len={n} GD iter={it} "
                      f"score≈{best_score_local:.6f}")

            # Soft‑max over the convolution entries.
            exp_term = np.exp(alpha * (b - cur_max))
            w = exp_term / exp_term.sum()

            # Gradient of the smoothed objective.
            a_rev = a[::-1]
            conv_rev = _fft_convolve(a_rev, w)
            grad = 2.0 * conv_rev[n - 1: n - 1 + n]

            # Adam update (bias‑corrected).
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = eta * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = _project_onto_simplex(a)

            # Anneal temperature.
            if alpha < alpha_max:
                alpha = min(alpha_max, alpha * alpha_factor)

            # Tiny random jitter every few thousand steps.
            if it % 2000 == 0 and np.random.rand() < 0.1:
                a += np.random.randn(n) * 1e-5
                a = _project_onto_simplex(a)

        # ---------- 4) Exact evaluation of the best local candidate ----------
        try:
            final_score = evaluate_sequence(best_seq_local.tolist())
        except Exception:                     # pragma: no cover
            final_score = best_score_local

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = best_seq_local.copy()
            print(f"*** New global best (len={n}) score={overall_best_score:.6f}")

        # Early exit if we have already reached the target.
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but guarantees a return value.
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback to uniform distribution.")

    print(f"Final best score: {overall_best_score:.6f} "
          f"(length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
