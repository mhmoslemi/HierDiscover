def _run_gradient(a0, target_sum, overall_deadline, time_budget):
    a = a0.copy()
    N = a.shape[0]
    # Adam parameters
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    conv = _fft_convolve(a)
    best_score = conv.max()

    start = time.time()
    deadline = start + time_budget

    # schedule parameters
    max_iters = 30000  # upper bound
    beta_start = 0.5
    beta_end = 120.0
    lr_start = 1.0
    lr_end = 0.02

    it = 0
    while time.time() < deadline and it < max_iters and time.time() < overall_deadline:
        it += 1
        conv = _fft_convolve(a)
        max_conv = conv.max()
        # compute softmax weights
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)
        lr = lr_start + (lr_end - lr_start) * (it / max_iters)

        # safe exponentiation: conv - max_conv <= 0
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = 2.0 * _fft_correlate(w, a)

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        step = lr * m / (np.sqrt(v) + eps)
        a_new = a - step
        # projection onto simplex
        a = _project_onto_simplex(a_new, target_sum)

        # evaluate score occasionally, say every 5 it
        if it % 5 == 0:
            conv = _fft_convolve(a)
            cur_score = conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Grad it {it:05d}] New local best: {best_score:.6f}")

    # final eval
    conv = _fft_convolve(a)
    cur_score = conv.max()
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()
        print(f"[Grad final] New local best: {best_score:.6f}")

    return best_seq, best_score

def _polish_sequence(seq, overall_deadline):
    a = seq.copy()
    N = a.shape[0]
    cur_conv = _fft_convolve(a)
    cur_score = cur_conv.max()
    best_seq = a.copy()
    best_score = cur_score

    # parameters
    total_time = overall_deadline - time.time()
    if total_time <= 0:
        return best_seq, best_score
    polish_start = time.time()
    polish_deadline = overall_deadline
    T0 = 0.05
    Tend = 0.001
    sigma0 = 0.6
    sigma_end = 0.02

    it = 0
    rng = np.random.RandomState(12345)   # maybe seed independent

    while time.time() < polish_deadline:
        it += 1
        elapsed = time.time() - polish_start
        frac = elapsed / (polish_deadline - polish_start)
        T = T0 * (1 - frac) + Tend * frac
        sigma = sigma0 * (1 - frac) + sigma_end * frac

        # propose move: random pair mass transfer
        i, j = rng.choice(N, size=2, replace=False)
        delta = (rng.rand() - 0.5) * sigma
        # adjust to keep non-negative
        if delta > 0:
            # want to move mass from j to i, take at most a[j]
            delta = min(delta, a[j])
        else:
            # move mass from i to j, bounded by a[i]
            delta = -min(-delta, a[i])
        new_a = a.copy()
        new_a[i] += delta
        new_a[j] -= delta

        # No need to project: maintains sum and non-negativity by delta correction
        # Compute score
        conv = _fft_convolve(new_a)
        new_score = conv.max()
        # acceptance
        if new_score < cur_score or rng.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_a.copy()
                print(f"[Polish it {it:06d}] New best: {best_score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the simplex with prescribed ℓ1‑norm."""
    if target_sum <= 0:
        return np.zeros_like(v)
    # Duchi et al. (2008) O(n log n) algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (length 2N‑1).
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# FFT‑based correlation (valid mode)  corr = w ⊛ a, length N.
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]                  # length of a
    L = w.shape[0]                  # length of w (2N‑1)
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]   # valid part, length N

# ----------------------------------------------------------------------
# Initialise a feasible sequence of length N and ℓ1‑norm = target_sum.
# ----------------------------------------------------------------------
def _initialize_sequence(N: int, target_sum: float, seed: int,
                         init_type: str, base_seq=None) -> np.ndarray:
    rng = np.random.RandomState(seed)
    if init_type == 'base' and base_seq is not None:
        base_arr = np.asarray(base_seq, dtype=float)
        if base_arr.size > N:
            seq = base_arr[:N].copy()
        else:
            seq = np.concatenate([base_arr, np.zeros(N - base_arr.size)])
        seq = _project_onto_simplex(seq, target_sum)
        return seq

    if init_type == 'uniform':
        seq = np.full(N, target_sum / N, dtype=float)
        # small random perturbation to break symmetry
        seq += 0.001 * target_sum * rng.rand(N) / N
        seq = _project_onto_simplex(seq, target_sum)
        return seq

    if init_type == 'sparse':
        seq = np.zeros(N, dtype=float)
        K = max(2, N // 50)               # ~2 % non‑zeros
        idx = rng.choice(N, size=K, replace=False)
        vals = rng.rand(K)
        vals = vals / vals.sum() * target_sum
        seq[idx] = vals
        seq = _project_onto_simplex(seq, target_sum)
        return seq

    # fallback to uniform
    seq = np.full(N, target_sum / N, dtype=float)
    seq = _project_onto_simplex(seq, target_sum)
    return seq

# ----------------------------------------------------------------------
# Run a projected Adam optimisation for a bounded amount of time.
# ----------------------------------------------------------------------
def _run_gradient(a0: np.ndarray, target_sum: float,
                  overall_deadline: float, time_budget: float):
    a = a0.copy()
    N = a.shape[0]

    # Adam hyper‑parameters
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # keep the best vector seen in this run
    conv = _fft_convolve(a)
    best_score = conv.max()
    best_seq = a.copy()

    start = time.time()
    deadline = start + time_budget

    # schedules (linear annealing)
    max_iters = 30000
    beta_start, beta_end = 0.5, 120.0
    lr_start, lr_end = 1.0, 0.02

    it = 0
    while time.time() < deadline and it < max_iters and time.time() < overall_deadline:
        it += 1

        conv = _fft_convolve(a)
        max_conv = conv.max()

        # annealed parameters
        t_frac = it / max_iters
        beta = beta_start + (beta_end - beta_start) * t_frac
        lr = lr_start + (lr_end - lr_start) * t_frac

        # soft‑max weights (stable because max_conv is subtracted)
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth surrogate
        grad = 2.0 * _fft_correlate(w, a)

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        step = lr * m / (np.sqrt(v) + eps)

        a_new = a - step
        a = _project_onto_simplex(a_new, target_sum)

        # occasional full evaluation
        if it % 5 == 0:
            cur_score = _fft_convolve(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Grad it {it:05d}] new local best = {best_score:.6f}")

    # final check
    cur_score = _fft_convolve(a).max()
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()
        print(f"[Grad final] new local best = {best_score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Simulated‑annealing polishing on the current best sequence.
# ----------------------------------------------------------------------
def _polish_sequence(seq: np.ndarray, overall_deadline: float):
    a = seq.copy()
    N = a.shape[0]

    cur_score = _fft_convolve(a).max()
    best_seq = a.copy()
    best_score = cur_score

    # temperature / perturbation schedules (linear)
    total_time = max(0.0, overall_deadline - time.time())
    if total_time <= 0.0:
        return best_seq, best_score

    polish_start = time.time()
    polish_deadline = overall_deadline
    T0, T_end = 0.05, 0.001
    sigma0, sigma_end = 0.6, 0.02

    rng = np.random.RandomState(12345)
    it = 0
    while time.time() < polish_deadline:
        it += 1
        elapsed = time.time() - polish_start
        frac = elapsed / (polish_deadline - polish_start + 1e-12)

        T = T0 * (1 - frac) + T_end * frac
        sigma = sigma0 * (1 - frac) + sigma_end * frac

        # propose a random pairwise mass transfer
        i, j = rng.choice(N, size=2, replace=False)
        delta = (rng.rand() - 0.5) * sigma
        if delta > 0:
            delta = min(delta, a[j])          # move from j → i
        else:
            delta = -min(-delta, a[i])        # move from i → j
        new_a = a.copy()
        new_a[i] += delta
        new_a[j] -= delta

        new_score = _fft_convolve(new_a).max()
        if new_score < cur_score or rng.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_a.copy()
                print(f"[Polish it {it:06d}] new global best = {best_score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Optimises a non‑negative step‑function to minimise the maximal value of its
    self‑convolution.  The algorithm combines several projected‑Adam runs with
    a simulated‑annealing polishing phase while respecting the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 1.0)

    # ------------------------------------------------------------------
    # Problem size (N) and normalisation (S = √(2·N))
    # ------------------------------------------------------------------
    base_seq = globals().get('height_sequence_1', None)
    if base_seq is not None:
        n0 = len(base_seq)
    else:
        n0 = None

    if n0 is None:
        N = 16000
    else:
        N = min(20000, max(8000, n0 * 2))

    S = math.sqrt(2.0 * N)

    # ------------------------------------------------------------------
    # Gradient‑descent stage (multiple restarts)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    restarts = 4
    # allocate roughly 70 % of the remaining time to this phase
    grad_total_budget = (deadline - time.time()) * 0.70
    time_per_restart = grad_total_budget / max(1, restarts)

    for r in range(restarts):
        if time.time() >= deadline:
            break

        if r == 0 and base_seq is not None:
            init_type = 'base'
        elif r == 1:
            init_type = 'uniform'
        elif r == 2:
            init_type = 'sparse'
        else:
            init_type = 'uniform'

        a0 = _initialize_sequence(N, S, seed + r, init_type, base_seq)
        seq_local, score_local = _run_gradient(
            a0, S, deadline, time_per_restart
        )
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[Restart {r}] global best updated = {best_score:.6f}")

    # fallback if all restarts failed (unlikely)
    if best_seq is None:
        best_seq = np.full(N, S / N, dtype=float)
        best_seq = _project_onto_simplex(best_seq, S)
        best_score = _fft_convolve(best_seq).max()

    # ------------------------------------------------------------------
    # Polishing stage (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_seq, best_score = _polish_sequence(best_seq, deadline)

    # final safety clipping (the evaluator also clamps, but we keep it tidy)
    best_seq = np.maximum(best_seq, 0.0)
    best_seq = np.minimum(best_seq, 1000.0)

    return best_seq.tolist()
