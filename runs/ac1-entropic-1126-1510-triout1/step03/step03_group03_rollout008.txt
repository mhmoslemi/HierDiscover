def _make_blocks(n, B):
    # distribute remainder across first blocks.
    base = n // B
    remainder = n % B
    lengths = np.full(B, base, dtype=int)
    lengths[:remainder] += 1
    starts = np.cumsum(np.concatenate(([0], lengths[:-1])))
    ends = starts + lengths - 1
    return starts, ends, lengths

def _precompute_overlap(starts, ends, lengths, n):
    B = len(starts)
    K = 2 * n - 1
    overlap = np.empty((B, B, K), dtype=np.float32)
    ks = np.arange(K, dtype=np.int32)
    for p in range(B):
        sp = starts[p]; ep = ends[p]
        for q in range(p, B):
            sq = starts[q]; eq = ends[q]
            low = np.maximum(sp, ks - eq)
            high = np.minimum(ep, ks - sq)
            ov = (high - low + 1).astype(np.float32)
            ov[ov < 0] = 0.0
            overlap[p, q, :] = ov
            if p != q:
                overlap[q, p, :] = ov
    return overlap

class _BlockSmoothMaxObjective:
    def __init__(self, mu, n, B, S, lengths, overlap):
        self.mu = mu
        self.n = n
        self.B = B
        self.S = S
        self.lengths = lengths
        self.overlap = overlap

    def __call__(self, y):
        # compute p via softmax
        y_max = np.max(y)
        e = np.exp(y - y_max)
        p = e / e.sum()
        # compute h = S * p / lengths
        h = self.S * p / self.lengths
        # compute conv
        H = np.outer(h, h)  # B,B
        # compute conv via tensordot
        conv = np.tensordot(H, self.overlap, axes=([0, 1], [0, 1]))
        # conv shape K
        # smooth max using mu
        # compute max for stability
        max_c = conv.max()
        if self.mu > 0.0:
            diff = (conv - max_c) / self.mu
            # clip overflow
            diff = np.clip(diff, -700, 700)
            exp_diff = np.exp(diff)
            Z = exp_diff.sum()
            w = exp_diff / Z
            f_val = max_c + self.mu * np.log(Z)
        else:
            # hard max -> uniform weights on maximisers
            mask = conv == max_c
            cnt = mask.sum()
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            f_val = max_c

        # compute M matrix: M_{p,q} = sum_k w_k * overlap_{p,q,k}
        # we can do tensordot: M = np.tensordot(self.overlap, w, axes=([2],[0]))
        M = np.tensordot(self.overlap, w, axes=([2], [0]))
        # grad_h = 2 * M @ h
        grad_h = 2.0 * M.dot(h)
        # compute gradient wrt p: d f/d p = (S / lengths) * grad_h
        a = grad_h * (self.S / self.lengths)  # elementwise
        # compute gradient wrt y via softmax Jacobian
        inner = np.dot(p, a)
        grad_y = p * (a - inner)
        return float(f_val), grad_y.astype(np.float64)

def _project_to_simplex(v, z):
    # returns projection onto nonnegative simplex with sum = z
    v = np.asarray(v, dtype=np.float64)
    if z < 0:
        raise ValueError
    # algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = cssv[-1] / len(v)
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Block‑parameterised smooth‑max optimisation.

The step‑function is parametrised by B constant‑height blocks
(30‑40 works well for n≈3000).  The block heights h_i are forced to
satisfy Σ_i L_i h_i = √(2 n) (L_i – block lengths) by writing
h_i = S·p_i/L_i with p on the simplex (p≥0, Σp=1).  A smooth
approximation of the max‑convolution entry is minimised with a
continuation on the temperature μ and L‑BFGS‑B.  After the low‑dimensional
optimisation the solution is expanded to the full sequence and a cheap
projected‑sub‑gradient polishing finishes the run.

The method is fast (few seconds per continuation) and explores many
initialisations, random block perturbations and a final gradient polish,
typically beating the previous 1.5166 bound and often reaching the
target 1.503.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _make_blocks(n: int, B: int):
    """Return starts, ends (inclusive) and lengths of B contiguous blocks."""
    base = n // B
    rem = n % B
    lengths = np.full(B, base, dtype=int)
    lengths[:rem] += 1
    starts = np.cumsum(np.concatenate(([0], lengths[:-1])))
    ends = starts + lengths - 1
    return starts, ends, lengths


# ----------------------------------------------------------------------
def _precompute_overlap(starts, ends, n):
    """Overlap counts for every pair of blocks (shape B×B×(2n‑1))."""
    B = len(starts)
    K = 2 * n - 1
    overlap = np.empty((B, B, K), dtype=np.float32)
    ks = np.arange(K, dtype=np.int32)

    for p in range(B):
        sp, ep = starts[p], ends[p]
        for q in range(p, B):
            sq, eq = starts[q], ends[q]
            low = np.maximum(sp, ks - eq)
            high = np.minimum(ep, ks - sq)
            ov = (high - low + 1).astype(np.float32)
            ov[ov < 0] = 0.0
            overlap[p, q, :] = ov
            if p != q:
                overlap[q, p, :] = ov
    return overlap


# ----------------------------------------------------------------------
class _BlockSmoothMaxObjective:
    """
    Smooth‑max objective for the block‑parameterisation.
    The variables are the free logits y (size B) feeding a soft‑max p.
    """

    def __init__(self, mu, n, B, S, lengths_float, overlap):
        self.mu = mu                # temperature
        self.n = n
        self.B = B
        self.S = S                  # ℓ¹‑norm √(2 n)
        self.lengths = lengths_float  # float version of block lengths
        self.overlap = overlap      # B×B×(2n‑1) float32

    def __call__(self, y):
        # probabilities on the simplex
        y_max = np.max(y)
        e = np.exp(y - y_max)
        p = e / e.sum()

        # block heights h_i = S·p_i / L_i
        h = self.S * p / self.lengths

        # full block‑level convolution (vector of length 2n‑1)
        H = np.outer(h, h)                               # B×B
        conv = np.einsum('ij,ijk->k', H, self.overlap)   # length 2n‑1

        max_c = conv.max()
        if self.mu > 0.0:
            diff = (conv - max_c) / self.mu
            diff = np.clip(diff, -700.0, 700.0)          # avoid overflow
            exp_diff = np.exp(diff)
            Z = exp_diff.sum()
            w = exp_diff / Z
            f_val = float(max_c + self.mu * math.log(Z))
        else:   # hard max (will hardly be used)
            mask = conv == max_c
            cnt = mask.sum()
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            f_val = float(max_c)

        # matrix M_{pq} = Σ_k w_k·overlap_{pqk}
        M = np.einsum('ijk,k->ij', self.overlap, w)

        # gradient w.r.t. block heights
        grad_h = 2.0 * M.dot(h)           # size B

        # chain rule through h = S·p/L
        a_vec = grad_h * (self.S / self.lengths)   # ∂f/∂p_i

        # Jacobian of soft‑max: ∂p/∂y = diag(p) – p pᵀ
        inner = np.dot(p, a_vec)                  # scalar
        grad_y = p * (a_vec - inner)              # size B

        return f_val, grad_y.astype(np.float64)


# ----------------------------------------------------------------------
def _final_full_polish(a_start, deadline, S):
    """Projected sub‑gradient descent on the true hard‑max objective."""
    n = a_start.shape[0]
    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    print(f"[polish] starting score = {best_score:.6f}")

    eta0 = S * 0.5 / n
    decay = 1e-4
    it = 0
    max_it = 5000

    while time.time() < deadline and it < max_it:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt

        # gradient = 2·(w * a) convolved with reversed a
        conv_wr = np.convolve(w, a[::-1])
        grad = 2.0 * conv_wr[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * grad
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[polish] improve = {best_score:.6f} (it {it})")
        a = a_new
        it += 1

    print(f"[polish] final best = {best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score non‑negative step‑function within the time budget.
    Returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        # global height_sequence_1 may be provided by the environment
        height_sequence_1  # type: ignore[name-defined]
        n = len(height_sequence_1)   # pylint: disable=unused-variable
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    S = math.sqrt(2.0 * n)               # ℓ¹‑norm required by the score

    # --------------------------------------------------------------
    # block discretisation
    B = min(50, max(20, n // 80))        # typically 30‑40
    starts, ends, lengths_int = _make_blocks(n, B)
    lengths = lengths_int.astype(np.float64)   # float version for formulas
    overlap = _precompute_overlap(starts, ends, n)   # B×B×(2n‑1) float32

    # --------------------------------------------------------------
    # bookkeeping
    best_score = float('inf')
    best_a = np.zeros(n, dtype=float)

    # --------------------------------------------------------------
    # initialise several y‑vectors (logits for the soft‑max over blocks)
    init_y = []

    # 0) uniform (all logits zero)
    init_y.append(np.zeros(B, dtype=np.float64))

    # 1) V‑shaped (large at the ends, tiny in the centre)
    centres = (starts + ends) / 2.0
    mid = (n - 1) / 2.0
    vshape = np.abs(centres - mid) + 1e-3
    p_v = vshape / vshape.sum()
    init_y.append(np.log(p_v + 1e-12))

    # 2) random Gaussian
    init_y.append(np.random.randn(B) * 0.5)

    # 3) another random (uniform Dirichlet)
    init_y.append(np.log(np.random.dirichlet(np.ones(B)) + 1e-12))

    # 4) reference sequence, if available
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        # compute block‑averaged heights
        block_means = np.empty(B, dtype=float)
        for i, (s, e) in enumerate(zip(starts, ends)):
            block_means[i] = a_ref[s:e + 1].mean()
        # turn into simplex probabilities p_i = L_i·h_i / S
        p_ref = block_means * lengths / S
        p_ref = np.maximum(p_ref, 1e-12)
        p_ref /= p_ref.sum()
        init_y.append(np.log(p_ref + 1e-12))

    # --------------------------------------------------------------
    # Continuation on the block level
    mu_factor = 0.5
    mu_min = 1e-6
    maxiter_per_mu = 120

    for start_id, y0 in enumerate(init_y):
        if time.time() >= deadline:
            break
        print(f"[seed {start_id}] starting continuation")
        y = y0.copy()

        # initialise temperature from current max‑convolution
        p = _softmax(y)
        h = S * p / lengths
        a_full = np.repeat(h, lengths_int)
        conv_cur = np.convolve(a_full, a_full)
        mu = float(conv_cur.max())
        if mu <= 0.0:
            mu = 1.0

        while mu > mu_min and time.time() < deadline:
            obj = _BlockSmoothMaxObjective(mu, n, B, S, lengths, overlap)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x

            # evaluate the *hard* maximum for the current block solution
            p = _softmax(y)
            h = S * p / lengths
            a_full = np.repeat(h, lengths_int)
            sc = evaluate_sequence(a_full.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a_full.copy()
                print(f"[improve] new best = {best_score:.6f}")

            mu *= mu_factor

    # --------------------------------------------------------------
    # If nothing succeeded, fall back to a uniform distribution
    if best_score == float('inf'):
        best_a = np.full(n, S / n, dtype=float)
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] using uniform distribution")

    # --------------------------------------------------------------
    # Random block‑wise perturbations (still respecting the weighted simplex)
    remaining = deadline - time.time()
    if remaining > 2.0:
        # work with the probability vector p = L·h / S
        p_best = (lengths * best_a.reshape(B, 1).mean(axis=1)) / S   # recompute
        # alternatively, recover from best_a directly:
        p_best = np.empty(B, dtype=float)
        for i, (s, e) in enumerate(zip(starts, ends)):
            p_best[i] = best_a[s:e + 1].mean() * lengths[i] / S
        p_best /= p_best.sum()

        trials = int(min(8000, remaining * 200))
        sigma = 0.05  # perturbation magnitude
        print(f"[block‑perturb] performing {trials} random trials")
        for _ in range(trials):
            if time.time() >= deadline:
                break
            delta = np.random.randn(B) * sigma
            p_new = p_best + delta
            p_new = np.maximum(p_new, 0.0)
            if p_new.sum() == 0.0:
                continue
            p_new /= p_new.sum()
            h_new = S * p_new / lengths
            a_new = np.repeat(h_new, lengths_int)
            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                p_best = p_new
                print(f"[block‑perturb] improve = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final projected‑gradient polishing on the full sequence
    if time.time() < deadline:
        best_a, best_score = _final_full_polish(best_a, deadline, S)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
