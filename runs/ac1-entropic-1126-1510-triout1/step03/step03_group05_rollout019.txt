    conv_self = _self_convolution(a, size_fft) # length 2n-1
    max_conv = conv_self.max()
    # compute softmax weights
    exps = np.exp(alpha * (conv_self - max_conv))

def _softmax_gradient(a, conv_self, alpha, size_fft):
    max_conv = conv_self.max()
    # compute softmax weights
    adj = conv_self - max_conv
    exp_alpha = np.exp(alpha * adj)
    w = exp_alpha / exp_alpha.sum()
    # convolution of w and reversed a

    # gradient slice
    n = a.shape[0]
    grad = 2.0 * conv_w_rev[n-1: n-1 + n]
    return grad

def shape_search(n, target_sum, size_fft, time_limit):
    best_a = None
    best_score = float('inf')
    def try_candidate(a):
        nonlocal best_a, best_score
        # ensure a sums to target and non-neg
        a = _project_to_simplex(a, target_sum)
        # evaluate max conv via FFT
        max_val = _max_convolution(a, size_fft)
        if max_val < best_score:
            best_a = a.copy()
            best_score = max_val
    # baseline uniform
    a = np.ones(n) / n * target_sum
    try_candidate(a)
    # families
    N = 120  # number of samples per family
    # inverse sqrt single-ended
    for _ in range(N):
        beta = np.random.rand()*10
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        try_candidate(a)
    # double-ended inverse sqrt
    for _ in range(N):
        beta = np.random.rand()*10
        i = np.arange(n, dtype=float)
        a = 1.0 / np.sqrt(i + beta + 1.0) + 1.0 / np.sqrt((n-1 - i) + beta + 1.0)
        try_candidate(a)
    # power law
    for _ in range(N):
        beta = np.random.rand()*10
        p = np.random.rand()*2.5 + 0.1
        i = np.arange(n, dtype=float)
        a = 1.0 / (i + beta + 1.0)**p + 1.0 / ((n-1 - i) + beta + 1.0)**p
        try_candidate(a)
    # exponential double-ended
    for _ in range(N):
        decay = np.random.rand()*0.012 + 0.0005
        i = np.arange(n, dtype=float)
        a = np.exp(-decay*i) + np.exp(-decay*(n-1 - i))
        try_candidate(a)
    # cosine
    for _ in range(N):
        p = np.random.rand()*5.0 + 0.5
        a = np.abs(np.cos(np.pi*np.arange(n, dtype=float)/(n-1)))**p
        try_candidate(a)
    # mixture
    for _ in range(N):
        beta = np.random.rand()*10
        lam = np.random.rand()
        a = lam*np.ones(n) + (1-lam)*(1.0/np.sqrt(np.arange(n, dtype=float)+beta+1.0))
        try_candidate(a)
    # random blocks
    for _ in range(400):
        nb = np.random.randint(2,8)
        cuts = np.sort(np.random.choice(n-1, nb-1, replace=False)+1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)+0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        try_candidate(cand)
    # random gaussian
    for _ in range(200):
        cand = np.abs(np.random.randn(n))
        try_candidate(cand)
    if best_a is None:
        # fallback uniform
        best_a = np.ones(n) * (target_sum/n)
        best_score = _max_convolution(best_a, size_fft)
    print(f"[shape] best candidate max conv = {best_score:.6f} for n={n}")
    return best_a, best_score

def refine_adam(a, target_sum, size_fft, deadline):
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.15
    alpha = 1.0
    alpha_max = 200.0
    no_improve = 0
    it = 0
    while time.time() < deadline:
        conv = _self_convolution(a, size_fft)
        cur_score = conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[adam iter {it}] new best max conv = {best_score:.6f}")
        else:
            no_improve += 1

        # compute softmax gradient
        max_conv = conv.max()
        adj = conv - max_conv
        exp_alpha = np.exp(alpha * adj)
        w = exp_alpha / exp_alpha.sum()
        # compute gradient via convolution
        a_rev = a[::-1]
        conv_w_rev = _conv_fft(w, a_rev, size_fft)
        grad = 2.0 * conv_w_rev[n-1: n-1 + n]

        # Adam updates
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        lr = lr0 / (1.0 + 1e-5 * it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional perturbation
        if (it % 5000 == 0 and it > 0) or no_improve > 3000:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[adam iter {it}] noise injection")

        # update alpha gradually
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * 1.001)

        a = a_new
        it += 1
        # optional early break
        if it >= 500000:  # safe limit
            break

    final_score = _max_convolution(best_a, size_fft)
    print(f"[adam] finished after {it} iters, best max conv = {best_score:.6f}")
    return best_a, best_score

"""
Propose a non‑negative step function with a low evaluate_sequence value.

Algorithm
---------

1. **Problem size** – choose a length `n`.  If the reference sequence
   `height_sequence_1` exists we use its length, otherwise we fall back to
   `n = 4000`.  The ℓ₁‑norm is fixed to `sqrt(2·n)` (the normalisation used in
   the reference paper).  All vectors are projected onto the simplex
   `{a≥0, Σa = sqrt(2·n)}` after each update.

2. **Parametric shape search** – sample a large collection of simple families
   (inverse‑sqrt, double‑ended power‑law, exponential, cosine, mixtures,
   random block patterns, Gaussian noise).  Every candidate is projected onto
   the simplex and its maximal self‑convolution entry is evaluated
   (fast FFT based).  The best candidate is kept.

3. **Soft‑max Adam refinement** – starting from the best shape we minimise the
   smooth approximation of the max‑convolution

        Lα(a) = (1/α)·log Σₖ exp(α·(a*a)ₖ)

   The gradient of `Lα` w.r.t. `a` is a weighted sum of the sub‑gradients of
   the individual convolution entries and can be computed by a single FFT
   convolution:

        grad = 2·conv(w, a_rev)[n‑1 : n‑1+n]   ,   wₖ ∝ exp(α·((a*a)ₖ‑max))

   An Adam optimiser (β₁=0.9, β₂=0.999) with bias correction updates `a`,
   followed by Euclidean projection onto the simplex.  The temperature `α`
   is increased gradually (geometric schedule) so the optimisation first
   follows a smooth landscape and later focuses on the true maximum.
   Occasional random perturbations help to escape flat regions.

4. **Final random‑block refinement** – a handful of fresh block‑structured
   candidates are generated and evaluated; any improvement updates the best
   solution.

The whole routine respects a 1000 s wall‑clock budget, prints intermediate
scores and finally returns the best sequence as a plain list of floats.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Return max entry of a*a (fast, assumes sum already normalised)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv_self: np.ndarray,
                      alpha: float, size_fft: int) -> np.ndarray:
    """
    Gradient of the smooth max (log‑sum‑exp) of the self‑convolution.
    Uses the identity  grad = 2·conv(w, a_rev)[n‑1:n‑1+n],
    where wₖ ∝ exp(alpha·(conv_selfₖ – max(conv_self)).
    """
    n = a.shape[0]
    max_c = conv_self.max()
    # stable soft‑max weights
    w = np.exp(alpha * (conv_self - max_c))
    w /= w.sum()
    # convolution of weights with reversed a
    a_rev = a[::-1]
    conv_wr = _conv_fft(w, a_rev, size_fft)          # length 3·n‑2
    grad = 2.0 * conv_wr[n - 1 : n - 1 + n]
    return grad

def _shape_search(n: int, target_sum: float, size_fft: int,
                  deadline: float) -> tuple[np.ndarray, float]:

    best_a = None
    best_score = float("inf")
    rng = np.random.default_rng()

    def try_candidate(cand: np.ndarray) -> None:
        nonlocal best_a, best_score
        cand = _project_to_simplex(cand, target_sum)
        score = _max_convolution(cand, size_fft)
        if score < best_score:
            best_score = score
            best_a = cand.copy()

    # uniform baseline
    uniform = np.full(n, target_sum / n, dtype=float)
    try_candidate(uniform)

    # ----- families (increase samples modestly) -----
    SAMPLES = 120

    # single‑ended inverse‑sqrt
    for _ in range(SAMPLES):

    return best_a, best_score

def _adam_refinement(a: np.ndarray, target_sum: float,
                     size_fft: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max‑convolution.
    Temperature α is increased geometrically up to α_max.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.15

    # temperature schedule
    alpha = 1.0
    alpha_max = 200.0
    alpha_factor = 1.001

    it = 0
    no_improve = 0

    while time.time() < deadline:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[adam {it:07d}] new best max‑conv = {best_score:.6f}")
        else:
            no_improve += 1

        # gradient of smooth max
        grad = _softmax_gradient(a, conv, alpha, size_fft)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / (1.0 + 1e-5 * it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick
        if (it % 5000 == 0 and it > 0) or no_improve > 3000:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[adam {it:07d}] random perturbation")

        a = a_new

        # increase temperature slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        it += 1
        # safety upper bound on iterations (helps in tight time budgets)
        if it >= 500_000:
            break

    print(f"[adam] finished after {it} iterations, best max‑conv = {best_score:.6f}")
    return best_a, best_score

                      size_fft: int, deadline: float) -> tuple[np.ndarray, float]:
    """Random block search for a quick last improvement."""
    n = a.shape[0]
