- Initialize m = zeros(n), v = zeros(n). Keep best.

- Compute FFT length m = next power of two >= 3n for correlation.

- For it in range(max_iters):

    - check time.

    - Compute conv = _conv_full_fft(a). Compute cur_score = 2 * n * conv.max(). Update best.

    - Weighted softmax weights as before: w = np.exp(lambda*(conv - max(conv))) normalized.

"""
Hybrid Adam‑softmax gradient descent + sub‑gradient refinement + hill‑climbing.

* Generates a rich pool of initial probability vectors:
  - arcsine, uniform, symmetric Beta(α,α) (grid + random α)
  - classic signal‑processing windows (Hann, Hamming, Blackman, Kaiser, Gaussian)
  - sparse “spike” vectors, low‑concentration Dirichlet draws.
* Phase 1 – smooth optimisation:
  – Uses a soft‑max (log‑sum‑exp) approximation of the max convolution.
  – Gradient is obtained by correlation via FFT.
  – Adam optimizer on the simplex (additive update + Euclidean projection).
  – λ (soft‑max temperature) is increased linearly up to a large value
    (≈150) so the smooth objective approaches the true ∞‑norm.
* Phase 2 – exact sub‑gradient descent (projected) to polish the best
  candidate of the current length.
* Phase 3 – cheap stochastic hill‑climbing that moves a tiny amount of
  mass between two positions and keeps the change if the (exact) score
  improves.
* Multiple problem sizes (n≈2 k … 20 k) are explored; the time budget
  (default 1000 s) is respected by checking the deadline at every
  iteration.  The best sequence found (as a plain list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using a power‑of‑two rFFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector of length 2·n‑1 (soft‑max weights),
    a – current sequence (length n),
    m – length of the FFT (power of two, ≥ len(w)+len(a)-1).
    Returns gradient g of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the grid x_i = (i+½)/n, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α = ½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _hann_window(n: int) -> np.ndarray:
    """Classic Hann window, normalised to sum 1."""
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    w /= w.sum()
    return w


def _hamming_window(n: int) -> np.ndarray:
    """Hamming window, normalised."""
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    w /= w.sum()
    return w


def _blackman_window(n: int) -> np.ndarray:
    """Blackman window, normalised."""
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    i = np.arange(n)
    w = (a0
         - a1 * np.cos(2.0 * np.pi * i / (n - 1))
         + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


def _kaiser_window(n: int, beta: float = 5.0) -> np.ndarray:
    """Kaiser window with parameter β, normalised."""
    i = np.arange(n)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = np.i0(beta * np.sqrt(1 - t * t)) / np.i0(beta)
    w /= w.sum()
    return w


def _gaussian_window(n: int, sigma_factor: float = 1.0 / 6.0) -> np.ndarray:
    """Truncated discrete Gaussian, σ = sigma_factor·n, normalised."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Diversified pool of starting probability vectors for length n."""
    cand = []

    # 1) arcsine law – a strong baseline
    cand.append(_arcsine_distribution(n))

    # 2) simple uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) symmetric betas with a deterministic grid of α
    for alpha in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.9, 1.2):
        cand.append(_beta_distribution(n, alpha))

    # 4) random betas
    for _ in range(3):
        alpha = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, alpha))

    # 5) classic windows
    cand.append(_hann_window(n))
    cand.append(_hamming_window(n))
    cand.append(_blackman_window(n))
    cand.append(_kaiser_window(n, beta=5.0))
    cand.append(_gaussian_window(n, sigma_factor=0.15))

    # 6) a few‑spike sparse vector
    k = max(2, n // 500)                     # ~0.2 % non‑zeros
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    cand.append(a / a.sum())

    # 7) low‑concentration Dirichlet draw
    cand.append(np.random.dirichlet(np.full(n, 0.3)))

    return cand

def _optimise_adam(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 8000,
    eta0: float = 0.12,
    lambda0: float = 0.2,
    lambda_max: float = 150.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed (log‑sum‑exp) approximation of the max‑convolution.
    Returns the best sequence (Σa = 1) seen and its true objective value.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT length for correlation (needs to accommodate w length 2n‑1 and a length n)
    m_fft = 1 << ((3 * n - 2).bit_length())

    # Adam state
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)

    lambda_inc = (lambda_max - lambda0) / max_iters
    lamb = lambda0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exact score (track best)
        conv = _conv_full_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # ---------- soft‑max weights ----------
        maxb = conv.max()
        # to improve numerical stability we shift by maxb
        exp_arg = lamb * (conv - maxb)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # ---------- gradient ----------
        g = _gradient_via_correlation(w, a, m_fft)

        # ---------- Adam step ----------
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * (g * g)
        m_hat = m_vec / (1.0 - beta1 ** it)
        v_hat = v_vec / (1.0 - beta2 ** it)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

        # additive update on the simplex
        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # occasional Gaussian jitter (helps escape flat plateaus)
        if it % 900 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # increase soft‑max temperature
        lamb = min(lambda_max, lamb + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    eta0: float = 0.28,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Returns the best sequence (Σa = 1) encountered.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a)
        k_star = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k*−i} for positions that actually contribute
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = a[k_star - idx]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        # jitter every few hundred steps
        if it % 600 == 0 and np.random.rand() < 0.18:
            a += np.random.randn(n) * 8e-5
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 18000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny amount of mass from one index
    to another and keep the change if the exact objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-13:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # transfer up to 5 % of donor (or 0.01, whichever is smaller)
        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score
