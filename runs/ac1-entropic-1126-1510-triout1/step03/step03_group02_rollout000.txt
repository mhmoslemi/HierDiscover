"""
Propose a step‑function (list of non‑negative heights) that minimizes
the self‑convolution peak defined in `evaluate_sequence`.

Algorithm
---------
1. **Parametric exploration** – generate many candidates from a rich
   collection of analytic families (inverse‑sqrt, power‑law,
   exponential, cosine, Gaussian, triangular, beta‑shaped, double‑exponential,
   random block).  Each candidate is projected onto the simplex
   {a ≥ 0, Σa = √(2 n)} and evaluated with an exact FFT‑based score.
   The best `K` candidates are retained.

2. **Refinement** – each retained candidate is improved by a smooth
   max‑convolution descent.  The objective is approximated by a soft‑max
   (temperature `T`), whose gradient is computed analytically via FFTs.
   An Adam optimiser with projected simplex updates is used.
   Temperature is slowly cooled; occasional random kicks help escape
   plateaus.  The best point seen during the run is kept.

3. **Final random‑block sweep** – a quick pass of random block candidates
   (few large constant blocks) is performed on the remaining time budget,
   updating the global best whenever a lower score is found.

All operations respect the security checks of `evaluate_sequence`.
The function returns the best sequence discovered before the 1000 s budget
expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    (w must sum to 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact score used during the search (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Smooth max‑convolution minimisation using Adam.
    Returns the best sequence observed and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning rate (scaled to the problem size)
    eta0 = target_sum * 0.5 / n

    # initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights (smooth approximation of max) -----
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)         # prevent overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam step -----
        eta = eta0 / (1.0 + it * 1e-5)               # simple decay
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random kick to escape plateaus -----
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9997                               # slower cooling
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises the evaluation score.  The function respects a 1000 s
    time budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)               # normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # Keep the K best raw candidates from the exploratory phase
    K = 15
    best_candidates: list[tuple[float, np.ndarray]] = []   # (score, array)

    def consider(score: float, a: np.ndarray) -> None:
        """Maintain a sorted list of the top‑K candidates (ascending score)."""
        if len(best_candidates) < K:
            best_candidates.append((score, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if score < best_candidates[-1][0]:
                best_candidates[-1] = (score, a.copy())
                best_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # ---------- 1. PARAMETRIC SHAPE SEARCH -------------------------

    idx = np.arange(n)

    # Helper to create a candidate, project, score and store.
    def add_candidate(arr: np.ndarray) -> None:
        arr = _project_to_simplex(arr, target_sum)
        sc = quick_score(arr, n, size_fft)
        consider(sc, arr)

    # (a) Uniform
    add_candidate(np.ones(n))

    # (b) Inverse‑sqrt (single‑ended)
    for _ in range(100):
        beta = np.random.rand() * 12.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        add_candidate(a)

    # (c) Double‑ended inverse‑sqrt
    for _ in range(100):
        beta = np.random.rand() * 12.0
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        add_candidate(a)

    # (d) Power‑law
    for _ in range(100):
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 1.8 + 0.2      # exponent ∈ [0.2, 2.0]
        a = 1.0 / (idx + beta + 1.0) ** pexp
        add_candidate(a)

    # (e) Exponential decay (single)
    for _ in range(100):
        decay = np.random.rand() * 0.015 + 0.0002    # [2e‑4, 1.7e‑2]
        a = np.exp(-decay * idx)
        add_candidate(a)

    # (f) Exponential decay (double‑ended, possibly asymmetric)
    for _ in range(100):
        decay_l = np.random.rand() * 0.015 + 0.0002
        decay_r = np.random.rand() * 0.015 + 0.0002
        a = np.exp(-decay_l * idx) + np.exp(-decay_r * (n - 1 - idx))
        add_candidate(a)

    # (g) Cosine‑based (large at ends, small in the middle)
    for _ in range(100):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        add_candidate(a)

    # (h) Sine‑based (zero at ends, peak in the middle)
    for _ in range(100):
        p = np.random.rand() * 6.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** p
        add_candidate(a)

    # (i) Gaussian‑like bump (centered)
    for _ in range(100):
        sigma = np.random.rand() * 0.25 + 0.05          # [0.05,0.30] of n
        sigma = sigma * n
        centre = (n - 1) / 2.0 + (np.random.rand() - 0.5) * n * 0.1
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        add_candidate(a)

    # (j) Triangular (symmetric peak at centre)
    for _ in range(80):
        peak = np.random.rand() * 3.0 + 0.5
        a = np.minimum(idx + 1, n - idx) ** peak
        add_candidate(a)

    # (k) Beta‑shaped (a_i ∝ (i+1)^α * (n-i)^β)
    for _ in range(80):
        alpha = np.random.rand() * 3.0 + 0.2
        beta = np.random.rand() * 3.0 + 0.2
        a = (idx + 1) ** alpha * (n - idx) ** beta
        add_candidate(a)

    # (l) Random block candidates (few large constant blocks)
    for _ in range(300):
        nb = np.random.randint(2, 8)                 # 2‑7 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        add_candidate(cand)

    # (m) Reference sequence, if available
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        add_candidate(a_ref)

    # Report exploration status
    best_candidates.sort(key=lambda x: x[0])
    print(f"[exploration] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # ---------- 2. REFINEMENT (Adam‑based smoothed max descent) -----

    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    for rank, (raw_score, a0) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank + 1} (raw score {raw_score:.6f})")
        remaining_time = deadline - time.time()
        # allocate a modest number of iterations proportionally to remaining time
        max_iters = int(min(200_000, remaining_time * 1400))
        refined_a, refined_score = _refine_candidate_adam(
            a0, deadline, n, size_fft, target_sum, max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---------- 3. FINAL RANDOM BLOCK SWEEP -----------------------

    while time.time() < deadline:
        nb = np.random.randint(2, 8)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand, n, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")
        # break early if only a tiny amount of time remains
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
