"""
Propose a step‑function (non‑negative sequence) with a low self‑convolution peak.

Algorithm
---------
1. Determine the problem size `n` (use the provided reference sequence if any,
   otherwise fall back to n≈3000) and the target ℓ₁‑norm  S = √(2 n).

2. **Exploratory sampling** – build a pool of promising raw candidates.
   Families include:
   * Dirichlet draws with various concentration parameters (sparse to dense)
   * Power‑law / exponential shapes with random exponents / decay rates
   * Random block‑wise constant sequences
   * The reference sequence (if available)

   Every candidate is projected onto the simplex {x ≥ 0 , Σ x = S} and its
   exact score `max(conv)` is evaluated (the normalization makes the score
   equal to the peak of the convolution).  The `K` best raw candidates are kept.

3. **Refinement** – each of the `K` candidates is run through a projected
   Adam‑style sub‑gradient descent that minimises the *hard* max‑convolution
   (the weight vector is a Dirac at the current worst index).  The update is
   projected back onto the simplex after every step.  The learning rate
   decays roughly as 1/√t.  The best sequence seen during the run is retained.

4. **Ordering optimisation** – for the current global best, a simple
   hill‑climbing on the permutation of the values is performed:
   repeatedly swap two positions, keep the swap if it lowers the peak.
   A modest number of swaps (≈ 2000) is affordable within the remaining time.

5. Return the best sequence discovered before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return a power‑of‑2 size ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a using FFT (length = 2 n‑1)."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution (no clipping)."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)_k   at the current maximal index k_max.
    This is the exact sub‑gradient of the L∞‑norm of the convolution.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        # overlapping part is a[0..k_max] with reversed a[k_max::-1]
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        # k_max ≥ n
        start = k_max - n + 1                # first i that participates
        length = 2 * n - 1 - k_max           # number of terms
        # a[start:] pairs with a[:length] reversed
        g[start:] = 2.0 * a[:length][::-1]

    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent on the hard max‑convolution.
    Returns the best (lowest peak) sequence and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base step size (order of magnitude of one entry)
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and current worst index
        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            # occasionally report progress
            if it % 5000 == 0:
                print(f"[refine] it {it:,} new best = {best_score:.6f}")

        # subgradient at the worst index
        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate decays like 1/√t
        eta = eta0 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection to keep Σa = target_sum and a ≥ 0
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score
