import time, math, numpy as np

def _project_to_simplex(v, z):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]          # sorted descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(u, v, size_fft):
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[:len(u) + len(v) - 1]

def _self_convolution(a, size_fft):
    return _conv_fft(a, a, size_fft)

Propose a non‑negative step function (list of heights) that minimises the
value returned by evaluate_sequence().  The optimisation proceeds in three
phases:

1. **Seeding** – a handful of simple parametric families (uniform,
   decreasing exponential, decreasing power‑law, linear ramp) are sampled.
   Each candidate is projected onto the simplex Σa = √(2 n) and the best
   one is kept.

2. **Smoothed‑max Adam** – we minimise a smooth approximation of the
   max‑convolution:
        f_T(a) = (1/T)·log Σ_k exp(T·(a*a)_k) .
   For a given temperature T the gradient is the correlation of *a* with the
   soft‑max weights w_k ∝ exp(T·(a*a)_k).  The gradient is computed with two
   FFT‑based convolutions, the update follows Adam with Euclidean projection
   onto the simplex, and the temperature T is increased gradually (annealing)
   so that the objective approaches the true max.  Random perturbations are
   injected when progress stalls.

3. **Random‑block refinement** – a final round of random block‑structured
   candidates is evaluated to possibly improve the best solution found.

All helper functions are top‑level, no closures are used, and the code
stops automatically before the 1000 s budget expires.

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (output length |u|+|v|-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]

def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n−1)."""
    return _conv_fft(a, a, size_fft)

        # gradient = 2·(a ⋆ w)   (cross‑correlation)
        w_rev = w[::-1]
        fw = np.fft.rfft(w_rev, n=size_fft)
        corr_full = np.fft.irfft(fa * fw, n=size_fft)[: 3 * n - 2]

        idx = 2 * n - 2 - np.arange(n)   # mapping to correlation entries
        grad = 2.0 * corr_full[idx]

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        # decaying learning rate
        lr = 0.06 / (1.0 + 1e-5 * it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # exact objective for monitoring
        cur_score = _max_conv(a_new)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            no_improve = 0
            print(f"[{it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        # temperature annealing
        if it % 5000 == 0 and it > 0:
            T = min(T * T_mult, T_max)

        # occasional random kick to escape plateaus
        if (it % 30000 == 0 and it > 0) or no_improve > 5000:
            noise = np.random.randn(n) * target_sum * 3e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            # reset moments after a large perturbation
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{it:07d}] random kick")

        a = a_new
        it += 1

    print(f"[adam] finished after {it} iterations, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. RANDOM BLOCK REFINEMENT
    for _ in range(300):
        nb = np.random.randint(2, 8)                # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.zeros(n, dtype=float)
        for blk, vblk in zip(blocks, vals):
            cand[blk] = vblk
        cand = _project_to_simplex(cand, target_sum)
        sc = _max_conv(cand)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
