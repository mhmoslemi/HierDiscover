def compute_subgradient(a, conv, max_b, tol=1e-12):
    active = np.where(conv >= max_b - tol)[0]
    grad = np.zeros_like(a)
    n = a.shape[0]
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        grad[i] += a[i] + a[j]

"""
Gradient‐Descent with Exact Sub‑gradients for the max self‑convolution problem.

The algorithm works on a non‑negative step function a of length n
(≈3 k when a reference sequence is available, otherwise a larger
n can be used).  The objective to minimise is
    eval(a) = 2·n·max(a*a)/(sum(a)^2) .
For a fixed ℓ¹–norm S = √(2n) the evaluation equals max(a*a), so we
directly minimise the peak of the (full) auto‑convolution while keeping
the vector on the simplex {a≥0, Σa = S}.  The method consists of

1.  Projection onto the simplex after every update.
2.  A fast FFT based convolution to obtain the peak value.
3.  Exact sub‑gradient computation from *all* active shift indices
    (those attaining the maximum); if they are too many a cheap
    top‑k approximation is used.
4.  Momentum (Nesterov‑style) and a slowly decaying step size.
5.  Several random restarts and a short exhaustive block‑search /
    pair‑wise mass‑transfer hill‑climb at the end.

The routine respects a 1000 s wall‑clock budget and returns the best
sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    if np.all(v == 0):
        # trivial case – put all mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv_fft(a: np.ndarray) -> float:
    """Return max of the full self‑convolution a*a using a real FFT."""
    n = a.size
    n2 = 1 << ((2 * n - 1).bit_length())       # next power of two ≥ 2n‑1
    f = np.fft.rfft(a, n2)
    conv = np.fft.irfft(f * f, n2)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
def _full_conv_fft(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2n‑1) via real FFT."""
    n = a.size
    n2 = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(a, n2)
    conv = np.fft.irfft(f * f, n2)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _subgradient_from_active(a: np.ndarray, conv: np.ndarray,
                             cur_max: float, rel_tol: float = 1e-8,
                             max_active: int = 200,
                             top_k: int = 20) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    If the number of active indices exceeds `max_active`,
    a top‑k approximation is used instead.
    """
    n = a.size
    # indices where the convolution is (almost) maximal
    tol = max(rel_tol * cur_max, 1e-12)
    active = np.where(conv >= cur_max - tol)[0]

    if active.size > max_active:
        # fall back to a cheap top‑k approximation
        active = np.argpartition(-conv, top_k - 1)[:top_k]

    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        # grad_i receives a_i + a_j (diag case yields 2·a_i)
        grad[i] += a[i] + a[j]
    return grad
