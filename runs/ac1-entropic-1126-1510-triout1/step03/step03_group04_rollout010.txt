"""
Optimise a non‑negative step‑function for the evaluation function.

The optimiser uses a projected Adam scheme with an exact (or soft‑max)
sub‑gradient of the max‑convolution, several diverse initialisations,
periodic random restarts and a final cheap hill‑climb that swaps mass
between two positions.  It respects the sum constraint Σa = √(2 n) so
that the final score printed by the provided ``evaluate_sequence``
coincides with the maximum of the self‑convolution.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    { x ≥ 0 , Σx = target_sum }.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = target_sum
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Full convolution a * a using a real FFT.
    Returns an array of length 2·n‑1.
    """
    n = a.shape[0]
    size = 1 << (2 * n - 1).bit_length()          # next power of two ≥ 2n‑1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_and_grad(a: np.ndarray, mu: float | None) -> tuple[float, np.ndarray]:
    """
    Compute ``max(conv)`` and a sub‑gradient of the max‑convolution.

    If ``mu`` is given and sufficiently large a soft‑max weighting is used;
    otherwise the exact active set (all indices attaining the maximum) is used.
    """
    n = a.shape[0]
    conv = _conv_fft(a)
    max_val = conv.max()

    # ----- weight vector for the gradient --------------------------------
    if mu is not None and mu > 0.02 * max_val:
        # smooth (soft‑max) weighting
        diff = conv - max_val                # ≤ 0
        w = np.exp(diff / mu)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum
    else:
        # exact active set – equal weights on all maximal entries
        tol = max_val * 1e-12 + 1e-15
        w = (conv >= max_val - tol).astype(float)

    # ----- gradient = 2 * (w * a_rev) convolved ---------------------------
    a_rev = a[::-1]
    m_len = w.shape[0] + a_rev.shape[0] - 1      # = 3·n‑2
    size = 1 << m_len.bit_length()               # FFT size ≥ linear length
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a_rev, size)
    grad_full = np.fft.irfft(fw * fa_rev, size)[:m_len]
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return float(max_val), grad


# ----------------------------------------------------------------------
def _initial_sequences(n: int, target_sum: float) -> list[np.ndarray]:
    """Create a handful of diverse starting points."""
    seqs = []

    # uniform
    seqs.append(np.full(n, target_sum / n))

    # symmetric triangular
    idx = np.arange(n)
    half = n // 2
    tri = np.where(idx <= half, idx + 1, n - idx).astype(float)
    seqs.append(tri / tri.sum() * target_sum)

    # sine‑shaped (half‑wave)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    seqs.append(s / s.sum() * target_sum)

    # exponential decay
    e = np.exp(-np.arange(n) / (n / 10.0))
    seqs.append(e / e.sum() * target_sum)

    # a few random points on the simplex
    rng = np.random.default_rng(12345)
    for _ in range(3):
        rnd = rng.random(n)
        seqs.append(_project_to_simplex(rnd, target_sum))

    return seqs


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a sequence of non‑negative heights that (hopefully) attains a
    low score for ``evaluate_sequence``.  The routine respects the given
    time budget, prints intermediate progress and ends with the best
    sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1          # safety margin

    # --------------------------------------------------------------
    # Determine problem size.
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)                 # Σa = √(2 n) → score = max(conv)

    # --------------------------------------------------------------
    # Build a pool of initial candidates.
    init_pool = []

    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = a_ref * (target_sum / max(a_ref.sum(), 1e-12))
        a_ref = _project_to_simplex(a_ref, target_sum)
        init_pool.append(a_ref)

    init_pool.extend(_initial_sequences(n, target_sum))

    # Pick the best among the cheap initial candidates.
    best_a = None
    best_score = float("inf")
    for cand in init_pool:
        score = _conv_fft(cand).max()
        if score < best_score:
            best_score = score
            best_a = cand.copy()
    print(f"[init] best initial max‑conv = {best_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for the projected Adam optimiser.
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2                               # base learning‑rate
    m = np.zeros_like(best_a)              # first‑moment estimate
    v = np.zeros_like(best_a)              # second‑moment estimate

    # Temperature for the soft‑max gradient (starts large, cools).
    mu = best_score * 0.8
    mu_decay = 0.9995
    mu_min_factor = 1e-6

    # Restart schedule – a few random restarts help escaping plateaux.
    max_restarts = 4
    restarts_done = 0
    restart_interval = (deadline - time.time()) / (max_restarts + 1)
    next_restart = time.time() + restart_interval

    cur = best_a.copy()                    # current iterate
    it = 0

    # --------------------------------------------------------------
    # Main optimisation loop.
    while time.time() < deadline - 20.0:    # keep ~20 s for final refinements
        max_val, grad = _max_and_grad(cur, mu if mu > mu_min_factor * max_val else None)

        # Keep the globally best solution.
        if max_val < best_score:
            best_score = max_val
            best_a = cur.copy()
            print(f"[{it:08d}] new best = {best_score:.6f}")

        # Adam update (minimise, so move opposite to the gradient).
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        cur = cur - step
        cur = _project_to_simplex(cur, target_sum)

        # Occasional tiny random jitter to escape flat regions.
        if it % 4000 == 0:
            jitter = np.random.randn(n) * target_sum * 2e-4
            cur += jitter
            cur = _project_to_simplex(cur, target_sum)

        # Cool the temperature.
        mu = max(mu * mu_decay, max_val * mu_min_factor)

        # Random restart handling.
        if time.time() > next_restart and restarts_done < max_restarts:
            restarts_done += 1
            print(f"[{it}] restart #{restarts_done}")
            cur = np.random.rand(n)
            cur = _project_to_simplex(cur, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            mu = _conv_fft(cur).max() * 0.8
            next_restart = time.time() + restart_interval

    # --------------------------------------------------------------
    # Phase 2 – cheap local hill‑climb (swap a fraction of mass between two spots).
    print("[final] start local refinement")
    while time.time() < deadline:
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size == 0:
            break
        i = int(np.random.choice(nz))
        j = int(np.random.randint(0, n))
        if i == j:
            continue
        delta = best_a[i] * np.random.uniform(0.05, 0.30)
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        cand_score = _conv_fft(cand).max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand
            print(f"[swap] improved = {best_score:.6f}")

    # --------------------------------------------------------------
    # Report final result (the value printed here should match evaluate_sequence).
    print(f"[final] best max‑conv = {best_score:.6f}")
    return best_a.tolist()
