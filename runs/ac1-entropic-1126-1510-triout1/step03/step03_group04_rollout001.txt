def conv_fft(vec):
   n = len(vec)
   n2 = 1 << ((2*n-1).bit_length())
   f = np.fft.rfft(vec, n2)
   conv = np.fft.irfft(f * f, n2)
   return conv[:2*n-1]

def grad_softmax(a, conv, mu):
   # conv length = 2n-1
   max_c = conv.max()
   diff = conv - max_c
   w = np.exp(diff / mu)
   w_sum = w.sum()
   w = w / w_sum
   # convolution of w with reversed a
   a_rev = a[::-1]
   # compute conv_w_rev via FFT
   len_out = len(w) + n - 1
   n2 = 1 << (len_out.bit_length())
   fw = np.fft.rfft(w, n2)

"""
Propose a low‑score step function.

The algorithm improves on the reference by using a *mirror‑descent* (exponentiated‑gradient)
scheme on the simplex, a smooth‑max gradient based on a temperature µ, and a
slow annealing schedule.  The method performs several timed restarts,
followed by a cheap block‑search and a final pairwise‑mass‑swap hill‑climb.
All operations are O(n log n) via FFT, allowing hundreds of thousands of
iterations within the 1000 s budget.

The score of a sequence is exactly the maximal entry of its self‑convolution,
provided the ℓ¹‑norm equals √(2 n).  The optimizer therefore directly
minimises this maximum.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto { x ≥ 0 , Σx = z }."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv_fft(a: np.ndarray, n2: int) -> np.ndarray:
    """Full self‑convolution of a using FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n2)
    conv = np.fft.irfft(fa * fa, n2)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 20) -> np.ndarray:
    """Sub‑gradient of max conv using the largest `top_k` entries."""
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, conv: np.ndarray, mu: float,
                     n2_a: int, n2_w: int) -> np.ndarray:
    """
    Gradient of the smooth‑max (log‑sum‑exp) of the convolution.
    Uses the temperature `mu` and FFTs of length `n2_a` and `n2_w`.
    """
    n = a.size
    max_c = conv.max()
    # weights for the soft‑max
    w = np.exp((conv - max_c) / mu)
    w_sum = w.sum()
    w /= w_sum

    # convolution of w with the reversed a
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n2_w)
    fr = np.fft.rfft(a_rev, n2_w)
    conv_wr = np.fft.irfft(fw * fr, n2_w)

    # slice that corresponds to the needed range
    sum_term = conv_wr[n - 1 : n - 1 + n]
    grad = 2.0 * sum_term

    # diagonal correction (i = j) as in the reference implementation
    diag = 2 * np.arange(n)
    mask = diag < conv.size
    grad[mask] -= w[diag[mask]] * a[mask]
    return grad

            best_a = cand
            print(f"[swap] new best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
