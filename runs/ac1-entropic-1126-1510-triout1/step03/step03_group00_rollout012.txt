        else:

            idx = np.floor(np.arange(n) * prev_a.size / n).astype(int)
            init_a = prev_a[idx] + 1e-12
            init_a = init_a / init_a.sum()

        beta2 = 0.999

        lr = 1.0
        alpha = 1.0
        alpha_mul = 1.015
        it = 0
        while time.time() < slice_deadline:
            it += 1
            # compute a
            max_log = logits.max()
            a = np.exp(logits - max_log)
            a = a / a.sum()
            # convolution a*a
            # compute FFT convolution
            size_fft = 1 << ((2*n-1).bit_length())
            A = np.fft.rfft(a, n=size_fft)
            conv_full = np.fft.irfft(A * A, n=size_fft)[:2*n-1]
            b = conv_full
            cur_max = b.max()
            cur_score = 2.0 * n * cur_max
            if cur_score < overall_best_score:
                overall_best_score = cur_score
                overall_best_seq = a.copy()
                print(f"[{time.time():.2f}] n={n} it={it} score={cur_score:.6f}")
            # compute weight vector w
            if alpha > 2000:
                # one-hot
                w = np.zeros_like(b)
                w[int(np.argmax(b))] = 1.0
            else:
                shift = b - cur_max
                w = np.exp(alpha * shift)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w[int(np.argmax(b))] = 1.0

def propose_candidate(seed:int=0, budget_s:int=1000):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm runs a projected Adam optimisation
    on the simplex using a temperature‑controlled soft‑max approximation of
    the maximal self‑convolution.  Several sequence lengths are tried within
    the time budget and the best sequence found is returned.
    """
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    # possible lengths (increase to explore larger n)
    lengths = [5000, 8000, 12000, 16000, 20000]

    overall_best_seq = None
    overall_best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break
        # allocate slice of time for this length (30% of remaining)
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(10.0, remaining * 0.3)

        # initialise with discrete arcsine distribution (works well)
        init_a = _arcsine_distribution(n)  # returns sum=1
        # add a small random perturbation to avoid exact symmetry
        init_a = init_a + 0.001 * np.random.rand(n)
        init_a = init_a / init_a.sum()
        # logits for softmax parametrisation
        logits = np.log(init_a + 1e-12)

        # Adam state
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        lr = 0.6  # learning rate, calibrated empirically
        # temperature for soft‑max on the convolution
        alpha = 0.5
        alpha_mul = 1.03
        max_alpha = 15000.0

        it = 0
        while time.time() < slice_deadline:
            it += 1

            # ---- compute current a from logits (softmax) ----
            max_log = logits.max()
            a = np.exp(logits - max_log)
            a = a / a.sum()

            # ---- self‑convolution (FFT) ----
            size_fft = 1 << ((2 * n - 1).bit_length())
            A = np.fft.rfft(a, n=size_fft)
            conv_full = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
            b = conv_full
            cur_max = b.max()
            cur_score = 2.0 * n * cur_max

            if cur_score < overall_best_score:
                overall_best_score = cur_score
                overall_best_seq = a.copy()
                print(f"[{time.time():.2f}] n={n} it={it} score={cur_score:.6f}")

            # ---- soft‑max weights for the max entry ----
            if alpha > 2000.0:
                # use a pure sub‑gradient (one‑hot)
                w = np.zeros_like(b)
                w[int(np.argmax(b))] = 1.0
            else:
                shift = b - cur_max          # max(b) = cur_max
                w = np.exp(alpha * shift)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w[int(np.argmax(b))] = 1.0
                else:
                    w = w / w_sum

            # ---- gradient of (soft‑max) max(b) w.r.t a ----
            a_rev = a[::-1]
            size_fft_g = 1 << ((3 * n - 2).bit_length())
            W = np.fft.rfft(w, n=size_fft_g)
            A_rev = np.fft.rfft(a_rev, n=size_fft_g)
            conv_g = np.fft.irfft(W * A_rev, n=size_fft_g)
            # take the part that corresponds to the gradient entries
            g_a = 2.0 * conv_g[n - 1:n - 1 + n]

            # ---- gradient on logits (mirror descent) ----
            dot = np.dot(g_a, a)               # scalar <g_a, a>
            g_x = a * (g_a - dot)

            # optional gradient clipping (prevents exploding updates)
            grad_norm = np.linalg.norm(g_x)
            max_norm = 5.0
            if grad_norm > max_norm:
                g_x = g_x * (max_norm / grad_norm)

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * g_x
            v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---- anneal temperature ----
            if alpha < max_alpha:
                alpha = min(max_alpha, alpha * alpha_mul)

            # ---- occasional random jitter to escape plateaus ----
            if it % 4000 == 0 and np.random.rand() < 0.12:
                logits += np.random.randn(n) * 1e-3

        # after finishing this length we continue with the next one
        # no need to keep state – we start fresh for the new length

        # early exit if target already achieved
        if overall_best_score <= 1.503:
            break

    # make sure we have something
    if overall_best_seq is None:
        # fallback: uniform distribution
        n_fallback = 5000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # final evaluation with the official black‑box
    try:
        final_score = evaluate_sequence(overall_best_seq.tolist())
    except Exception:
        final_score = None
    print(f"Final best score (official): {final_score:.6f} length={len(overall_best_seq)}")
    return overall_best_seq.tolist()

"""
Propose a step‑function (list of non‑negative heights) with a low
`evaluate_sequence` score.

Algorithm
---------
* Parameterise a probability vector a (sum = 1) by logits x and use the soft‑max
  a = softmax(x) – this guarantees the simplex constraints automatically.
* The objective 2 n·max(a∗a) is approximated by the smooth
  (1/α)·log ∑ exp(α·(a∗a)) (soft‑max of the self‑convolution).  For large α
  this becomes the true max, for small α it provides a smooth surrogate.
* Compute the gradient of the surrogate w.r.t. a via a convolution
  (soft‑max weights * reversed a).  Convert it to a gradient w.r.t. the
  logits (mirror‑descent geometry) and apply Adam with adaptive step sizes.
* The temperature α is increased geometrically, so optimisation starts
  smooth and finishes with a true sub‑gradient step.
* Several sequence lengths (up to 20 000) are tried within the time budget.
  The best sequence found over all lengths is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length n (sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will stop a few seconds early).

    Returns
    -------
    list[float]
        The best sequence (list of heights) discovered.
    """
    np.random.seed(seed)
    # Leave a small margin for final printing / evaluation
    deadline = time.time() + budget_s - 2.0

    # Candidate lengths (the problem is homogeneous – longer sequences can
    # achieve smaller constants, but the optimisation gets harder).
    lengths = [5000, 8000, 12000, 16000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        # Time slice for this length (30 % of what is left, but at least 10 s)
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(10.0, remaining * 0.30)

        # ---------- initialise ----------
        init_a = _arcsine_distribution(n)
        # a tiny random jitter helps to break perfect symmetry of the arcsine
        init_a = init_a + 0.001 * np.random.rand(n)
        init_a /= init_a.sum()
        logits = np.log(init_a + 1e-12)          # parametrisation a = softmax(logits)

        # ---------- Adam state ----------
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        lr = 0.6                                 # base learning rate

        # ---------- temperature for soft‑max of the convolution ----------
        alpha = 0.5                               # start smooth
        alpha_mul = 1.03
        max_alpha = 15000.0

        it = 0
        while time.time() < slice_deadline:
            it += 1

            # ----- a from logits (soft‑max) -----
            max_log = logits.max()
            a = np.exp(logits - max_log)
            a /= a.sum()                         # guarantees sum = 1

            # ----- self‑convolution a*a (FFT) -----
            size_fft = 1 << ((2 * n - 1).bit_length())
            A = np.fft.rfft(a, n=size_fft)
            conv_full = np.fft.irfft(A * A, n=size_fft)[: 2 * n - 1]
            b = conv_full                         # b_k = (a*a)_k
            cur_max = b.max()
            cur_score = 2.0 * n * cur_max

            # keep global best
            if cur_score < overall_best_score:
                overall_best_score = cur_score
                overall_best_seq = a.copy()
                print(f"[{time.time():.2f}] n={n} it={it} score={cur_score:.6f}")

            # ----- soft‑max weights w for the max entry -----
            if alpha > 2000.0:
                # sub‑gradient (one‑hot) – avoids overflow
                w = np.zeros_like(b)
                w[int(np.argmax(b))] = 1.0
            else:
                shift = b - cur_max                # for numerical stability
                w = np.exp(alpha * shift)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w[int(np.argmax(b))] = 1.0
                else:
                    w /= w_sum

            # ----- gradient of soft‑max max(b) w.r.t a -----
            a_rev = a[::-1]
            size_fft_g = 1 << ((3 * n - 2).bit_length())
            W = np.fft.rfft(w, n=size_fft_g)
            A_rev = np.fft.rfft(a_rev, n=size_fft_g)
            conv_g = np.fft.irfft(W * A_rev, n=size_fft_g)
            # gradient component for each a_i lives in the central slice
            g_a = 2.0 * conv_g[n - 1:n - 1 + n]

            # ----- gradient w.r.t. logits (mirror‑descent geometry) -----
            dot = np.dot(g_a, a)                 # <g_a, a>
            g_x = a * (g_a - dot)

            # optional gradient clipping (prevents exploding steps)
            grad_norm = np.linalg.norm(g_x)
            max_norm = 5.0
            if grad_norm > max_norm:
                g_x = g_x * (max_norm / grad_norm)

            # ----- Adam update on logits -----
            m = beta1 * m + (1.0 - beta1) * g_x
            v = beta2 * v + (1.0 - beta2) * (g_x ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- increase temperature -----
            if alpha < max_alpha:
                alpha = min(max_alpha, alpha * alpha_mul)

            # ----- occasional random jitter to escape plateaus -----
            if it % 4000 == 0 and np.random.rand() < 0.12:
                logits += np.random.randn(n) * 1e-3

        # early exit if we have already beaten the target
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback (should never trigger)
    if overall_best_seq is None:
        n_fallback = 5000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Final official evaluation (more reliable than the internal estimate)
    try:
        final_score = evaluate_sequence(overall_best_seq.tolist())
    except Exception:                     # if something goes wrong, keep internal score
        final_score = overall_best_score
    print(f"Final best score (official): {final_score:.6f} length={len(overall_best_seq)}")
    return overall_best_seq.tolist()
