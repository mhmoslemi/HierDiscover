def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2*n-1]

  * build soft‑max weights w_k ∝ exp(α·(b_k−max(b))) (α grows to a large
    value, eventually turning w into a hard indicator of the current peak);
  * obtain the gradient  ∇L = 2·rev(a) * w  (again via FFT);
  * perform an Adam update on a, symmetrise it (a←(a+a_rev)/2) and
    project back onto the probability simplex (∑a=1, a_i≥0).

A time budget of ~1000 s is split among several lengths (≈500–20000) and
a few random restarts per length.  The best exact sequence seen is
evaluated with the provided ``evaluate_sequence`` and returned.

The method differs from the previous smoothed‑max gradient descent by
using Adam’s adaptive steps, an explicit hard‑max phase, symmetry
enforcement and a broader exploration of lengths.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a via real FFT (full length 2·len(a)‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (real output)."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function within the supplied time budget.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # reserve a couple of seconds

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Length factors (can be tuned).  Keep ≤ 25000 for speed.
    length_factors = (0.7, 1.0, 1.3, 1.6, 2.0)
    lengths = sorted(
        {
            max(500, min(25000, int(base_n * f)))
            for f in length_factors
        }
    )

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # Main outer loop: keep sampling lengths while time permits.
    while time.time() < deadline:
        for n in lengths:
            if time.time() >= deadline:
                break

            # Allocate a slice of time for this length (≈30 % of remaining).
            slice_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.3)

            # ------------------------------------------------------------------
            # Several independent runs for the same length.
            for run_id in range(3):
                if time.time() >= slice_deadline:
                    break

                # ----------- initialise a ---------------------------------
                init_type = np.random.choice(["arcsine", "uniform", "random"])
                if init_type == "arcsine":
                    a = _arcsine_distribution(n)
                elif init_type == "uniform":
                    a = np.full(n, 1.0 / n, dtype=np.float64)
                else:  # random Dirichlet‑like
                    a = np.random.rand(n)
                    a /= a.sum()

                # Occasionally blend with the current global best (if same size).
                if overall_best_seq is not None and np.random.rand() < 0.3:
                    best_arr = np.asarray(overall_best_seq, dtype=np.float64)
                    if best_arr.size == n:
                        a = (a + best_arr) / 2.0

                a = _project_onto_simplex(a, s=1.0)

                # ----------- local optimisation state --------------------
                best_local_a = a.copy()
                best_local_max = np.max(_fft_self_convolve(a))
                best_local_score = 2.0 * n * best_local_max

                # Adam state
                m = np.zeros_like(a)
                v = np.zeros_like(a)
                beta1 = 0.9
                beta2 = 0.999
                eps = 1e-8
                it = 0

                # Soft‑max temperature schedule
                alpha = 1.0
                alpha_factor = 1.02
                max_alpha = 2000.0

                # ------------------------------------------------------------------
                while time.time() < slice_deadline:
                    it += 1

                    # self‑convolution and current peak
                    b = _fft_self_convolve(a)
                    cur_max = b.max()

                    # record best exact (peak‑based) sequence seen so far
                    if cur_max < best_local_max:
                        best_local_max = cur_max
                        best_local_a = a.copy()
                        best_local_score = 2.0 * n * cur_max
                        print(
                            f"[{time.time():.1f}] n={n} run={run_id} it={it} "
                            f"max={cur_max:.8e} score≈{best_local_score:.6f}"
                        )

                    # ------- compute weights (soft‑max or hard max) ----------
                    if alpha < max_alpha:
                        exp_term = np.exp(alpha * (b - cur_max))
                        w = exp_term / exp_term.sum()
                    else:
                        # hard‑max sub‑gradient (all entries within eps of max)
                        mask = (b >= cur_max - 1e-12)
                        w = mask.astype(float) / mask.sum()

                    # ------- gradient via convolution -------------------------
                    # grad_i = 2 * Σ_j a_j * w_{i+j}
                    a_rev = a[::-1]
                    conv_rev = _fft_convolve(a_rev, w)          # length 3n‑2
                    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]    # slice length n

                    # ------- Adam update ------------------------------------
                    m = beta1 * m + (1 - beta1) * grad
                    v = beta2 * v + (1 - beta2) * (grad ** 2)

                    # bias‑corrected estimates
                    m_hat = m / (1 - beta1 ** it)
                    v_hat = v / (1 - beta2 ** it)

                    # step size that decays gently
                    lr = 0.4 / np.sqrt(it)

                    a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

                    # enforce symmetry (helps flatten the convolution)
                    a = (a + a[::-1]) / 2.0

                    # project back onto the simplex {a_i≥0, Σa_i=1}
                    a = _project_onto_simplex(a, s=1.0)

                    # occasional tiny jitter (helps escape plateaux)
                    if it % 500 == 0 and np.random.rand() < 0.05:
                        a += np.random.randn(n) * 1e-6
                        a = _project_onto_simplex(a, s=1.0)

                    # anneal temperature
                    if alpha < max_alpha:
                        alpha = min(max_alpha, alpha * alpha_factor)

                # ------------------------------------------------------------------
                # Evaluate the best exact sequence found in this run
                try:
                    run_score = evaluate_sequence(best_local_a.tolist())
                except Exception:
                    run_score = best_local_score

                if run_score < overall_best_score:
                    overall_best_score = run_score
                    overall_best_seq = best_local_a.copy()
                    print(
                        f"*** New global best *** n={n} run={run_id} "
                        f"score={overall_best_score:.6f}"
                    )

                # Early exit if we have already reached the target
                if overall_best_score <= 1.503:
                    break
            if overall_best_score <= 1.503:
                break
        # End of loop over lengths
    # ----------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found.
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"Fallback sequence evaluated to {overall_best_score:.6f}")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
