import cvxpy as cp

def lp_step(f, n, target_sum):
    g = cp.Variable(n, nonneg=True)
    # Build convolution constraints: f restricted, we need (f*g)_k <= M.
    # Compute convolution coefficients of f * g as linear function: for each k (0..2n-2), sum_{i=0}^{k} f_i * g_{k-i} (with boundaries).
    # This is linear constraints: sum_i f_i * g_{k-i} <= M.
    # We can precompute for each k a vector of coefficients for g: coefficient for g_j is f_{k-j} if 0 <= k-j < n else 0.
    # Represent as matrix A shape (2n-1, n). Build using for loop with sparse representation.

    M = max(np.convolve(f, f))

    obj = cp.Maximize(cp.sum(g))
    constraints = []
    for k in range(2*n-1):
        coeff = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # g_{j} multiplies f_{k-j}, where j = i
        # Actually g_j appears when f_{k-j} * g_j, for j from i_min..i_max.
        # So constraint: sum_{j=i_min..i_max} f_{k-j} * g_j <= M
        coeff[i_min:i_max+1] = f[k - np.arange(i_min,i_max+1)]  # vectorized
        constraints.append(coeff @ g <= M)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)   # maybe ECOS fastest for LP.

    if g.value is not None:
        g_vec = np.maximum(g.value, 0)  # ensure nonnegative
        sum_g = g_vec.sum()
        if sum_g > 0:
            scale = target_sum / sum_g
            g_vec = g_vec * scale
            return g_vec
    else:
        return None

"""
Propose a low‑score step function using a sparse, simulated‑annealing optimiser.

The search works in two phases.
* Phase 1 – a fast stochastic optimiser on a *sparse* representation
  (only a few hundred non‑zero entries).  The objective is the maximal
  value of the self‑convolution; candidate moves transfer mass between
  entries or relocate a non‑zero to a new index.  A simple temperature
  schedule (simulated annealing) accepts occasional worsening moves,
  allowing the algorithm to escape local minima.
* Phase 2 – a short dense gradient‑descent refinement that starts from the
  best sparse solution found.  The gradient is the sub‑gradient of the
  max‑convolution (active index only) and the iterate is projected onto
  the simplex {a≥0, Σa = √(2 n)} after each step.

Both phases respect the 1000 s wall‑clock budget, keep the globally best
solution seen and finally return a plain Python list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv_sparse(pos: np.ndarray, w: np.ndarray, n: int) -> float:
    """
    Compute max_k (a*a)_k for a sparse vector.
    a[pos[i]] = w[i]; all other entries are zero.
    """
    # pairwise sums of positions
    idx = pos[:, None] + pos[None, :]               # shape (s, s)
    # outer product of weights
    prod = np.outer(w, w)                           # shape (s, s)
    # each unordered pair (i≠j) contributes twice
    contrib = prod * 2.0
    np.fill_diagonal(contrib, np.diag(prod))        # diagonal stays as w_i^2
    # accumulate to convolution vector via bincount
    conv = np.bincount(idx.ravel(),
                       weights=contrib.ravel(),
                       minlength=2 * n - 1)
    return float(conv.max())


# ----------------------------------------------------------------------
def _grad_max_conv_dense(vec: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max‑convolution for a *dense* vector.
    Returns a vector g such that:   max_conv(vec+Δ) ≈ max_conv(vec) + g·Δ
    (valid for sufficiently small Δ).
    """
    conv = np.convolve(vec, vec)
    k = int(conv.argmax())          # index attaining the maximum
    n = vec.shape[0]
    grad = np.zeros_like(vec)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] += 2.0 * vec[j[mask]]
    diag = mask & (i == j)
    grad[diag] -= vec[i[diag]]
    return grad

    # initialise
