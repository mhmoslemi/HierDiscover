    n = v.shape[0]

        w = np.zeros_like(v)

        return w

    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a non‑negative sequence with a low evaluate_sequence score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # problem size -------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = len(base)
        have_ref = True
    except Exception:
        n = 4000
        have_ref = False
        base = None

    S = math.sqrt(2.0 * n)      # we enforce Σa = S, then score = max(conv)

    # ------------------------------------------------------------------
    # bookkeeping ---------------------------------------------------------
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # generate a variety of initial soft‑max parameters ------------------
    init_ys = []

    init_ys.append(np.zeros(n, dtype=np.float64))                 # uniform
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    vshape = np.abs(idx - mid) + 1e-4
    init_ys.append(np.log(vshape))                                 # V‑shaped
    init_ys.append(np.random.randn(n) * 0.5)                       # Gaussian
    init_ys.append(np.log(np.random.dirichlet(np.ones(n) * 0.3) + 1e-12))
    init_ys.append(np.log(np.random.dirichlet(np.ones(n) * 5.0) + 1e-12))
    if have_ref:
        a_ref = _project_to_simplex(base, S)
        init_ys.append(np.log(a_ref + 1e-12))
    # a few more random seeds
    for _ in range(3):
        scale = np.random.uniform(0.2, 1.2)
        init_ys.append(scale * np.random.randn(n))

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max continuation with L‑BFGS‑B --------------------
    mu_factor = 0.7            # slower cooling
    mu_min = 1e-7
    maxiter_per_mu = 300

    for run_id, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[run {run_id}] start continuation")
        y = y0.copy()
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        mu = mu if mu > 0 else 1.0

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
                y = res.x
            except Exception as exc:
                print("[error] L‑BFGS failed:", exc)
                break

            a = S * _softmax(y)
            cur_score = np.convolve(a, a).max()   # with Σa=S the score equals max
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[improve] new best = {best_score:.6f}")

            mu *= mu_factor

    # ------------------------------------------------------------------
    # Phase 2 – linear‑program refinement ----------------------------------
    while time.time() + 6.0 < deadline:
        a = best_a.copy()
        conv = np.convolve(a, a)
        M = conv.max()

        rows = 2 * n - 1
        # Build sparse Toeplitz matrix A such that (A @ b) = conv(a, b)
        A = sp.lil_matrix((rows, n), dtype=float)
        for i in range(n):
            A[i:i + n, i] = a[i]
        A = A.tocsr()

        c = -np.ones(n, dtype=float)                # maximise Σb  ⇔ minimise –Σb
        b_ub = np.full(rows, M, dtype=float)
        bounds = [(0, None)] * n

        try:
            res_lp = opt.linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method="highs",
                options={"presolve": True},
            )
        except Exception as exc:
            print("[lp error]", exc)
            break

        if not res_lp.success:
            print("[lp] optimisation failed")
            break

        b_opt = res_lp.x
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break
        g = S * b_opt / sum_b

        # convex combination with step size t
        t = 0.2
        new_f = (1 - t) * a + t * g
        new_f = _project_to_simplex(new_f, S)

        new_score = np.convolve(new_f, new_f).max()
        if new_score < best_score:
            best_score = new_score
            best_a = new_f.copy()
            print(f"[lp refine] improve = {best_score:.6f}")

        # accept move (it never raises the max by more than t²·(new_score‑M))
        best_a = new_f.copy()
        if abs(M - new_score) < 1e-9:
            break

    # ------------------------------------------------------------------
    # Phase 3 – random block search ---------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        block_trials = int(min(4000, remaining * 150))
        print(f"[block] performing {block_trials} block trials")
        for _ in range(block_trials):
            nb = np.random.randint(2, 9)            # 2‑8 blocks
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            cur_score = np.convolve(cand, cand).max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand.copy()
                print(f"[block improve] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simple hill‑climbing / simulated annealing -----------------
    a = best_a.copy()
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 1e-12:
            continue
        delta = np.random.rand() * 0.1 * a[i]      # move up to 10 % of donor
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, 0.0)
        cand = _project_to_simplex(cand, S)

        new_score = np.convolve(cand, cand).max()
        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()
                print(f"[anneal] new best = {best_score:.6f} (it {it})")
        else:
            # occasional uphill move, temperature falls with remaining time
            temp = max(1e-5, (deadline - time.time()) / budget_s)
            if np.random.rand() < math.exp(-(new_score - cur_score) / temp):
                a = cand
                cur_score = new_score
        it += 1
        if it % 1000 == 0 and time.time() > deadline - 5:
            break

    print(f"[final] best score = {best_score:.6f}")
    # return as plain Python list – evaluate_sequence will clamp as required
    return best_a.tolist()
