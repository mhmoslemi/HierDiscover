"""
Propose a step‑function (list of non‑negative heights) with a very low
`evaluate_sequence` score.

Algorithm
---------
* A family of symmetric beta‑like distributions
      a_i ∝ ((i+½)(n‑i‑½))^{-γ},   γ∈[0.3,0.5],
  is used as a flexible starting point (γ=0.5 recovers the classic arcsine).
* For each (γ, n) pair a projected Adam optimiser minimises a smooth
  approximation of the maximal self‑convolution:
    – the max is replaced by a soft‑max with temperature α.
    – the gradient of the soft‑max is computed by a single FFT‑based
      convolution (O(n log n)).
    – after every step the vector is projected back onto the probability
      simplex (Euclidean projection).
    – α is increased geometrically so the optimisation ends with a true
      sub‑gradient of the max‑norm.
* When the Adam phase stagnates a short stochastic local‑search (random
  mass transfers between two coordinates) is applied.
* All candidates are evaluated with the official ``evaluate_sequence``.
  The best sequence found within the allocated 1000 s is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑_k exp(α·b_k) w.r.t. a.
    For very large α the max entry is taken as a one‑hot weight.
    """
    n = a.shape[0]

    # treat the true max as a one‑hot weight when α is huge
    if alpha > 2000.0:
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g

    # soft‑max weights
    shift = b - b.max()                # numerical stability
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                   # should not happen, fallback
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution w * a_rev (a_rev = a reversed)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _beta_distribution(n: int, gamma: float) -> np.ndarray:
    """
    Symmetric beta‑like distribution:
        a_i ∝ ((i+½)·(n‑i‑½))^{-γ}
    Normalised to sum to one (probability simplex).
    """
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-gamma)
    vals /= vals.sum()
    # exact projection onto the simplex to eliminate tiny numerical drift
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth max‑convolution objective.
    Returns the best sequence found and its internal score
    (2·n·max_conv), i.e. the exact value returned by evaluate_sequence
    when the sum of the coefficients equals 1.
    """
    a = init_a.copy()
    best_a = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # soft‑max temperature schedule
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for w (2n‑1) + a_rev
    t = 0
    while time.time() < deadline:
        t += 1

        # current convolution & score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] iter={t} score={best_score:.6f}")

        # gradient of the smoothed max
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update (bias‑corrected)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # increase the temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if t % 3000 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a small amount of mass
    from one coordinate to another if it reduces the max‑convolution.
    """
    cur_a = a.copy()
    b = _fft_self_convolve(cur_a)
    cur_score = 2.0 * n * b.max()
    best_a = cur_a.copy()
    best_score = cur_score

    attempts = 0
    while time.time() < deadline and attempts < 3000:
        attempts += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j or cur_a[i] <= 0.0:
            continue

        # move at most 5 % of the source mass
        eps = cur_a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # positivity is guaranteed by construction

        b_new = _fft_self_convolve(new_a)
        new_score = 2.0 * n * b_new.max()
        if new_score < cur_score - 1e-12:
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                print(f"[{time.time():.2f}] local search improved to {best_score:.6f}")

    return best_a, best_score

        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # a few different lengths (all ≥ 500, ≤ 20000 for speed)
    lengths = sorted(set(
        max(500, int(base_n * f))
        for f in (0.7, 0.9, 1.0, 1.2, 1.5, 2.0)
        if max(500, int(base_n * f)) <= 20000
    ))

    # gamma values to explore (γ=0.5 reproduces the classical arcsine)
    gammas = [0.45, 0.4, 0.35, 0.3]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for gamma in gammas:
        for n in lengths:
            if time.time() >= deadline:
                break

            # allocate roughly 25 % of the remaining time to this (γ,n) pair
            remaining = deadline - time.time()
            if remaining <= 0:
                break
            slice_deadline = time.time() + max(0.1, remaining * 0.25)

            # ---- initialise -------------------------------------------------
            a0 = _beta_distribution(n, gamma)

            # ---- Adam optimisation -----------------------------------------
            seq_opt, _ = _adam_optimize(a0, n, slice_deadline)

            # ---- brief local refinement ------------------------------------
            seq_refined, _ = _local_search(seq_opt, n, slice_deadline)

            # ---- final evaluation with the official black‑box ---------------
            try:
                final_score = evaluate_sequence(seq_refined.tolist())
            except Exception:                     # fallback to internal estimate
                b = _fft_self_convolve(seq_refined)
                final_score = 2.0 * n * b.max()

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq_refined.copy()
                print(
                    f"*** NEW GLOBAL BEST γ={gamma:.2f} n={n} "
                    f"score={overall_best_score:.6f}"
                )

            # stop early if we already beat the target
            if overall_best_score <= 1.503:
                break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # safety net – should never be triggered
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
