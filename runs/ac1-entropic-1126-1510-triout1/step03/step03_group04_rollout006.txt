"""
Propose a low‐score step function using a fast sub‑gradient descent on the
ℓ∞‑norm of the self‑convolution, followed by a cheap block‑search and a
final tiny swap hill‑climb.

The algorithm works on the invariant that the evaluation score is
scale‑free:
    score = 2·n·max(a*a) / (∑a)² .
Therefore we keep the vector on the simplex
    Σa = √(2 n) , a_i ≥ 0 .
The main loop repeatedly
    1. evaluates the full convolution via an FFT,
    2. builds the exact sub‑gradient of the maximum convolution entry,
    3. takes a diminishing step opposite to the gradient and projects back
       onto the simplex.
Occasional Gaussian “shakes’’ and restarts prevent stagnation.
When the time budget is almost exhausted a lightweight block‑search
generates piecewise‑constant candidates, and finally a tiny mass‑swap
hill‑climb finishes the optimisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv_full(a: np.ndarray, n2: int) -> np.ndarray:
    """
    Full self‑convolution of a (length n) using FFT.
    Returns the first (2·n‑1) entries as a 1‑D array.
    """
    f = np.fft.rfft(a, n2)
    conv = np.fft.irfft(f * f, n2)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _subgradient_at_max(a: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution functional.
    For every index k where conv[k] == max_val we add
        2·a_{k‑i}
    to the i‑th component (if the index `k‑i` is inside the vector).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    # treat numerical noise generously
    max_idxs = np.where(np.isclose(conv, max_val, atol=1e-12))[0]
    for k in max_idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        # each contribution is 2·a_j
        np.add.at(g, i, 2.0 * a[j])
    return g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate wall‑clock budget (seconds). The routine stops a bit
        before the limit to avoid a hard timeout.

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # safety margin

    # --------------------------------------------------------------
    # Problem size.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # Σa = √(2 n)
    n2 = 1 << ((2 * n - 1).bit_length())    # power of two ≥ 2n‑1

    # --------------------------------------------------------------
    # Initialise.
    if have_ref:
        a = np.asarray(height_sequence_1, dtype=float)
        a = a * (target_sum / max(a.sum(), 1e-12))
        a = _project_to_simplex(a, target_sum)
    else:
        a = np.random.rand(n)
        a = _project_to_simplex(a, target_sum)

    conv = _max_conv_full(a, n2)
    max_val = conv.max()
    best_score = 2.0 * n * max_val / (target_sum ** 2)
    best_a = a.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – sub‑gradient descent on the max‑convolution.
    step0 = target_sum * 0.8 / n          # initial step magnitude
    it = 0
