"""
Propose a non‑negative step function with a low self‑convolution peak.

Algorithm
---------
1. **Problem data** – length `n` (taken from the supplied reference
   sequence or default 3130) and target L1‑norm `S = sqrt(2*n)`.
   The evaluation of a candidate is exactly the maximal entry of
   its self‑convolution because the L1‑norm is normalised.

2. **Candidate generation** – a large pool of simple parametric families
   (exponential, power‑law, inverse‑sqrt, cosine, mixtures and random
   block‑structures).  Each candidate is projected onto the simplex
   `{a≥0, Σa=S}` and its score is recorded.  The ten best candidates are
   kept for the next stage.

3. **Multi‑restart Adam sub‑gradient descent** – each of the retained
   candidates is locally refined with an Adam optimiser that uses the
   exact sub‑gradient of the max‑convolution (the gradient at all active
   indices).  The optimiser runs until the time budget is exhausted or
   progress stalls; the best sequence found over all restarts is kept.

4. **Linear‑programming polishing** – using the current best sequence
   `f` we solve the LP  

        minimise   t
        s.t.       (f * g)_k ≤ t   for all k
                   Σ g_i = S
                   g_i ≥ 0

   (the convolution constraints are linear in `g`).  The LP is solved
   with `scipy.optimize.linprog` (high‑s) on a sparse constraint matrix.
   If the returned sequence `g` yields a smaller peak, it replaces the
   current best.

The whole routine is bounded by the supplied time budget (≈ 1000 s) and
always returns the best sequence found as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (the simplex)."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·n‑1)."""
    fu = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fu * fu, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the self‑convolution (the true score)."""
    conv = _self_convolution(a, fft_len)
    return float(conv.max())


# ----------------------------------------------------------------------
def _gradient_at_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max_k (a*a)_k` w.r.t. `a`.
    For each active index k we add 2·a_{k-i} (if it exists) to grad_i.
    The contribution of all active indices is averaged.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    if max_idxs.size == 0:
        return grad
    weight = 1.0 / max_idxs.size
    for k in max_idxs:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    grad *= weight
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(
    a_start: np.ndarray,
    target_sum: float,
    fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam sub‑gradient descent on the max‑convolution objective."""
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.15
    best_a = a.copy()
    best_score = _conv_max(a, fft_len)

    it = 0
    no_improve = 0

    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, fft_len)
        cur_score = float(conv.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # indices where the convolution is (almost) maximal
        eps_tol = max(1e-12, cur_score * 1e-9)
        max_idxs = np.where(conv >= cur_score - eps_tol)[0]

        grad = _gradient_at_max(a, max_idxs)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        lr = lr0 / (1.0 + 1e-5 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_to_simplex(a, target_sum)

        # occasional random kick to escape plateaus
        if (it % 3000 == 0) and (it > 0):
            noise = np.random.randn(a.shape[0]) * target_sum * 5e-4
            a = _project_to_simplex(a + noise, target_sum)
            no_improve = 0

        # early stop if we have been stuck for a long time
        if no_improve > 12_000:
            break

        it += 1

    return best_a, best_score

def _lp_polish(a: np.ndarray, target_sum: float, deadline: float):
    """
    Given a current sequence a, solve the linear program

        minimise   t
        s.t.       (a * g)_k ≤ t   for all k
                   Σ g_i = target_sum
                   g_i ≥ 0

    The LP is solved once (if time permits).  Returns a possibly better
    sequence g and its score; otherwise (None, None).
    """
    n = a.shape[0]
    rows = np.empty(n * n + (2 * n - 1), dtype=np.int32)
    cols = np.empty(n * n + (2 * n - 1), dtype=np.int32)
    data = np.empty(n * n + (2 * n - 1), dtype=np.float64)

    # Fill the Toeplitz‑style convolution constraints
    pos = 0
    for i in range(n):
        # rows k = i .. i+n-1  (inclusive)
        rows[pos : pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos : pos + n] = i
        data[pos : pos + n] = a[i]
        pos += n

    # Add the -t column (coefficient -1)
    rows[pos:] = np.arange(2 * n - 1, dtype=np.int32)
    cols[pos:] = n                       # last column = t
    data[pos:] = -1.0

    A_ub = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n + 1))

    # Equality Σ g_i = target_sum
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0

    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0               # minimise t

    bounds = [(0, None)] * n + [(0, None)]   # t ≥ 0 as well

    # Solve with HiGHS (fast and works with sparse matrices)
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=np.zeros(2 * n - 1),
            A_eq=A_eq,
            b_eq=np.array([target_sum]),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return None, None

    if not res.success:
        return None, None

    g = res.x[:n]
    # g already satisfies Σ g = target_sum
    # Small numerical cleanup
    g = np.maximum(g, 0.0)
    fft_len = 1 << ((2 * n - 1).bit_length())
    score = _conv_max(g, fft_len)
    return g, score

    """Search for a low‑score non‑negative step function."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3130
        have_ref = False

    target_sum = math.sqrt(2.0 * n)       # the normalisation used in the task
    fft_len = 1 << ((2 * n - 1).bit_length())   # power‑of‑two ≥ 2·n‑1

    # --------------------------------------------------------------
    # 1. generate a diverse pool of candidates (keep the best 10)
    top_k = 10
    pool = []   # list of (score, array)

    def _add_to_pool(arr: np.ndarray):
        sc = _conv_max(arr, fft_len)
        if len(pool) < top_k:
            pool.append((sc, arr.copy()))
            pool.sort(key=lambda x: x[0])
        else:
            if sc < pool[-1][0]:
                pool[-1] = (sc, arr.copy())
                pool.sort(key=lambda x: x[0])

    # Uniform baseline
    uniform = np.full(n, target_sum / n, dtype=float)
    _add_to_pool(uniform)

    # Reference, if available
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)
        ref = np.maximum(ref, 0.0)
        ref = _project_to_simplex(ref, target_sum)
        _add_to_pool(ref)

    # Helper to generate symmetric shapes
    def _sym(arr):
        """Force exact symmetry a_i = a_{n-1-i}."""
        half = (n + 1) // 2
        arr[:half] = (arr[:half] + arr[-half:][::-1]) / 2.0
        arr[half:] = arr[:n - half][::-1]
        return arr

    # --------------------------------------------------------------
    # parametric families (more samples than the previous baseline)
    SAMPLES_PER_FAMILY = 200

    # Inverse‑sqrt, double‑ended
    for _ in range(SAMPLES_PER_FAMILY):
        beta = np.random.rand() * 12.0
        idx = np.arange(n, dtype=float)
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        a = _sym(a)
        a = _project_to_simplex(a, target_sum)
        _add_to_pool(a)

    # Power‑law, double‑ended
    for _ in range(SAMPLES_PER_FAMILY):
        beta = np.random.rand() * 12.0
        p = np.random.rand() * 2.8 + 0.1
        idx = np.arange(n, dtype=float)
        a = (1.0 / (idx + beta + 1.0) ** p +
             1.0 / ((n - 1 - idx) + beta + 1.0) ** p)
        a = _sym(a)
        a = _project_to_simplex(a, target_sum)
        _add_to_pool(a)

    # Exponential decay, double‑ended
    for _ in range(SAMPLES_PER_FAMILY):
        decay = np.random.rand() * 0.018 + 0.0003
        idx = np.arange(n, dtype=float)
        a = np.exp(-decay * idx) + np.exp(-decay * (n - 1 - idx))
        a = _sym(a)
        a = _project_to_simplex(a, target_sum)
        _add_to_pool(a)

    # Cosine‑based (large at the ends)
    for _ in range(SAMPLES_PER_FAMILY):
        p = np.random.rand() * 5.0 + 0.3
        a = np.abs(np.cos(math.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        a = _sym(a)
        a = _project_to_simplex(a, target_sum)
        _add_to_pool(a)

    # Mixture of uniform & inverse‑sqrt
    for _ in range(SAMPLES_PER_FAMILY):
        beta = np.random.rand() * 12.0
        lam = np.random.rand()
        a = lam * np.ones(n, dtype=float) + (1.0 - lam) * (
            1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0))
        a = _sym(a)
        a = _project_to_simplex(a, target_sum)
        _add_to_pool(a)

    # Random block structures (few blocks, possibly zero interior)
    for _ in range(500):
        nb = np.random.randint(2, 9)               # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        _add_to_pool(cand)

    print(f"[generation] retained {len(pool)} best candidates, best score = {pool[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2. Multi‑restart Adam optimisation
    best_a = None
    best_score = float("inf")

    for idx, (init_score, init_a) in enumerate(pool):
        if time.time() >= deadline:
            break
        print(f"[adam] start restart {idx+1}/{len(pool)} (init score {init_score:.6f})")
        a_opt, sc_opt = _adam_optimize(
            init_a, target_sum, fft_len, deadline, max_iters=500_000
        )
        print(f"[adam] restart {idx+1} finished – local best {sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[adam] new global best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Optional LP polishing (if time permits)
    if best_a is not None and time.time() + 30.0 < deadline:
        print("[lp] polishing with linear program")
        g, sc_g = _lp_polish(best_a, target_sum, deadline)
        if g is not None and sc_g < best_score:
            best_score, best_a = sc_g, g.copy()
            print(f"[lp] improved to {best_score:.6f}")
        else:
            print("[lp] no improvement")

    # --------------------------------------------------------------
    # final output
    if best_a is None:
        # Fallback – uniform (should never happen)
        best_a = np.full(n, target_sum / n, dtype=float)
        best_score = _conv_max(best_a, fft_len)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
