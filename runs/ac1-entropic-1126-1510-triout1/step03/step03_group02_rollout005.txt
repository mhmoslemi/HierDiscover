        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        # gradient
        grad = _weighted_gradient(a, w, size_fft)
        # Adam update
        t += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**t)

"""
Propose a step‑function sequence with a low self‑convolution peak.

Algorithm
---------
1. Determine problem size (n) and normalisation target Σa = √(2 n).
2. Generate many parametric families (uniform, power‑law, inverse‑sqrt,
   exponential, cosine, sine, Gaussian, logistic, product‑power, random
   block).  Each candidate is projected onto the simplex and scored with a
   fast FFT‑based convolution.  Keep the top K raw candidates.
3. Perform a lightweight evolutionary sweep: mix two top candidates,
   add small Gaussian noise, enforce symmetry, project onto the simplex,
   and keep any improvement.
4. Refine the best candidates with a projected Adam optimiser that
   minimises a soft‑max approximation of the convolution peak.
   The temperature of the soft‑max is slowly annealed.  After every
   update the vector is symmetrised and projected onto the simplex.
5. Finish with a quick random‑block sweep to capture any remaining
   improvements.
6. Return the best sequence found before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a  (weights w sum to 1).
    Implements 2·(w ↔ a_rev) restricted to the n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Fast evaluation of the objective (same formula as evaluate_sequence)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max approximation of the max
    self‑convolution value.  Symmetry (a_i = a_{n-1-i}) is enforced after
    each update to encourage low peaks.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # step‑size (scaled to problem size)
    lr0 = 0.5 * target_sum / n

    # initialise temperature for soft‑max
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current score
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # occasional progress report
            print(f"[adam] it {it:,} new best = {best_score:.6f}")

        # soft‑max weights (smooth approximation of max)
        shifted = conv - conv.max()               # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)        # guard overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smoothed objective
        grad = _weighted_gradient(a, w, size_fft)

        # Adam update (minimisation -> subtract step)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # enforce non‑negativity & sum constraint
        a_new = _project_to_simplex(a_new, target_sum)

        # enforce symmetry (helps lower the peak)
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick to escape plateaus
        if (it % 5000 == 0 and it > 0):
            a_new += np.random.randn(n) * target_sum * 1e-3
            a_new = _project_to_simplex(a_new, target_sum)
            a_new = (a_new + a_new[::-1]) * 0.5
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9999          # slow cooling

    return best_a, best_score

    # cosine (large at ends)

    for _ in range(80):
