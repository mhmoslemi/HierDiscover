"""
Improved search for low‑score step‑functions.

The objective of evaluate_sequence equals the infinity norm of the
autoconvolution a*a when the ℓ¹‑norm of a is fixed to √(2·n).  We therefore
parameterise a by a soft‑max vector on the simplex (sum = √(2·n)) and
 minimise the smooth‑max surrogate

    f_μ(y) = max_k (a*a)[k] + μ·log Σ_k exp((a*a)[k]/μ),

with a geometric continuation μ→0 (smooth → hard max).  A few
independent initialisations (uniform, exponential, reversed‑exponential,
random Gaussian and a projected reference) are run, each with many
L‑BFGS‑B iterations per temperature.  After the continuation we polish
the best candidate with a sub‑gradient descent on the true hard max,
followed by a cheap large‑scale random‑block search and a final
monotone‑sorting heuristic.  The whole procedure respects the 1000 s
budget and returns the best feasible sequence found.

The code is deliberately written with only top‑level functions (no
closures, no lambdas) and uses only numpy/scipy.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate value and gradient w.r.t. the free parameters y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained parameters (length n).
    mu : float
        Temperature (>0) – the smoother the larger μ.
    n : int
        Length of the step‑function.
    S : float
        Desired ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : np.ndarray
        Gradient ∂f/∂y (shape (n,)).
    """
    # probabilities on the simplex and actual heights
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    # full convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weight vector w_k ∝ exp((conv_k - max)/μ)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # guard against overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                               # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a
    # g_a[i] = 2 * Σ_k w_k * a_{k‑i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable required by scipy.optimize – returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)

        # continuation loop
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)

            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            # cool down
            mu *= mu_factor

    # --------------------------------------------------------------
    # cheap random block search (large number of blocks, many trials)
    remaining = deadline - time.time()
    if remaining > 3.0:
        # allow up to ~5000 block trials (adjusted to the remaining time)
        max_trials = int(min(5000, remaining * 150))
        print(f"[block] performing up to {max_trials} random block trials")
        for _ in range(max_trials):
            # choose a random number of blocks between 2 and 12
            nb = rng.integers(2, 13)
            # random cut positions
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            # random values for each block
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # --------------------------------------------------------------
    # monotone‑sorting heuristic – a descending ordering is often beneficial
    if time.time() < deadline:
        sorted_desc = np.sort(best_a)[::-1]
        sorted_desc = _project_to_simplex(sorted_desc, S)
        _maybe_update(sorted_desc)

        sorted_asc = np.sort(best_a)  # just for curiosity
        sorted_asc = _project_to_simplex(sorted_asc, S)
        _maybe_update(sorted_asc)

    # --------------------------------------------------------------
    # final projected‑gradient descent on the hard max (μ = 0)
    if time.time() < deadline:
        a = best_a.copy()
        it = 0
        # step‑size schedule parameters (tuned empirically)
        eta0 = S * 0.6 / n
        decay = 5e-5
        while time.time() < deadline and it < 5000:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            conv_wr = np.convolve(w, a[::-1])
            g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

            eta = eta0 / (1.0 + decay * it)
            a_new = a - eta * g_a
            a_new = _project_to_simplex(a_new, S)

            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"[grad] improved to {best_score:.6f}")

            a = a_new
            it += 1

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
