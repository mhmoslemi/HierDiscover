    # init
    a = init_seq / init_seq.sum()
    # ensure symmetry if want
    a = (a + a[::-1]) / 2
    a = a / a.sum()
    # log parameters
    x = np.log(a + 1e-16)    # avoid log0
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_seq = a.copy()
    best_score = _score(a)
    # Precompute FFT size and maybe allocate w array
    m_fft = 1 << ((3 * n - 2).bit_length())
    lambda_inc = (lambda_max - lambda0) / max_iters
    lamb = lambda0
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute a = softmax(x)
        maxx = np.max(x)
        expx = np.exp(x - maxx)   # stable
        a = expx / expx.sum()
        # enforce symmetry: average with reversed
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()
        # compute conv
        b = _conv_full_fft(a)
        max_b = b.max()
        # Score
        cur_score = 2 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] iter={t:,} λ={lamb:.2f} score={best_score:.6f}")
        # compute w
        # softmax weighting of conv b:
        exp_arg = lamb * (b - max_b)   # negative
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(b, 1.0 / b.size)
        else:

        # gradient wrt a: g_i = 2 * Σ_k w_k * a_{k-i}
        # correlation: Use _gradient_via_correlation()
        g = _gradient_via_correlation(w, a, m_fft)
        # compute grad_x = a * (g - <a,g>)
        dot = np.dot(a, g)
        grad_x = a * (g - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        # bias correction
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = alpha / (np.sqrt(v_hat) + eps)
        x = x - step * m_hat
        # increase λ (linearly)
        lamb = min(lambda_max, lamb + lambda_inc)
    return best_seq, best_score

"""
Propose a low‑score step‑function sequence.

This version searches with a long Adam‑style optimisation on the log‑
parameters of a probability vector.  The algorithm works with a symmetric
distribution (a[i] = a[-i‑1]) and uses a soft‑max approximation of the
maximum of the auto‑convolution.  The temperature λ is increased linearly
to a large value (≈200) while Adam adapts per‑coordinate step sizes.
After the main search a cheap hill‑climbing phase moves tiny amounts of
mass between positions.  The whole routine respects the global 1000 s
budget and returns the best sequence (as a plain list) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT (length 2·n‑1)."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – length 2·n‑1 weight vector,
    a – current sequence (length n),
    m – FFT length (power of two, ≥ len(w)+len(a)-1).
    Returns a vector of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (sum 1)."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric Beta(α,β) on a uniform grid, then normalised."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _spike_distribution(n: int, k: int = None) -> np.ndarray:
    """Few‑spike distribution: put random mass on k positions (≈0.2 % of n)."""
    if k is None:
        k = max(2, n // 500)
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    return a / a.sum()


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 50000,
    alpha: float = 0.06,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    lambda0: float = 1.0,
    lambda_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on log‑probabilities with a soft‑max approximation
    of the maximal self‑convolution.  The sequence is kept symmetric.
    Returns the best vector (sum 1) seen and its exact score.
    """
    # initialise
    a = init_seq.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    a = (a + a[::-1]) / 2.0                # enforce symmetry
    a /= a.sum()

    # log‑parameters (softmax)
    x = np.log(a + 1e-30)

    m = np.zeros_like(x)                   # first moment
    v = np.zeros_like(x)                   # second moment

    best_seq = a.copy()
    best_score = _score(a)

    # FFT size for correlation (w length 2n‑1, a length n)
    m_fft = 1 << ((3 * n - 2).bit_length())
    lambda_inc = (lambda_max - lambda0) / max_iters
    lamb = lambda0

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- softmax to obtain a probability vector -----
        maxx = np.max(x)
        expx = np.exp(x - maxx)
        a = expx / expx.sum()
        # enforce symmetry after each softmax step
        a = (a + a[::-1]) / 2.0
        a /= a.sum()

        # ----- convolution and soft‑max weighting -----
        b = _conv_full_fft(a)
        maxb = b.max()
        cur_score = 2.0 * n * maxb
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] it={t:,} λ={lamb:.1f} score={best_score:.6f}")

        # soft‑max weights over the 2·n‑1 convolution entries
        w = np.exp(lamb * (b - maxb))        # max entry becomes 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (softmax‑approximated loss) -----
        g_a = _gradient_via_correlation(w, a, m_fft)

        # ----- transform to gradient w.r.t. log‑variables -----
        dot = np.dot(a, g_a)
        grad_x = a * (g_a - dot)             # Jᵀ·g where J is softmax Jacobian

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha / (np.sqrt(v_hat) + eps)
        x = x - step * m_hat

        # occasional jitter to escape plateaus
        if t % 1200 == 0 and np.random.rand() < 0.3:
            x += np.random.randn(n) * 1e-4

        # increase λ slowly towards λ_max
        lamb = min(lambda_max, lamb + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic local search: move a tiny mass from one index to another.
    Keeps the change if the exact objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low `evaluate_sequence` score.
    The routine:
      1. builds several diverse initial probability vectors,
      2. runs a long Adam optimiser on the log‑space,
      3. refines the best result with a cheap hill‑climbing phase,
      4. returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Use historic sequence if it exists (optional)
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: len={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Choose a single problem size – large enough to approximate the
    #     continuous limit but still fast.  8192 (=2¹³) is a good compromise.
    n = 8192
    print(f"--- Optimising length n={n} ---")

    # ------------------------------------------------------------------
    # 2️⃣  Build a pool of initial candidates
    init_pool = []

    # arcsine (the classic good one)
    init_pool.append(_arcsine_distribution(n))

    # symmetric β‑distributions with a sweep over α (β = α)
    for alpha in (0.12, 0.18, 0.25, 0.33, 0.45, 0.60, 0.80, 1.2):
        init_pool.append(_beta_distribution(n, alpha, alpha))

    # a few random Beta draws
    for _ in range(3):
        a = np.random.uniform(0.1, 0.6)
        b = np.random.uniform(0.1, 0.6)
        init_pool.append(_beta_distribution(n, a, b))

    # sparse spike vector
    init_pool.append(_spike_distribution(n))

    # Dirichlet with low concentration (many tiny entries)
    init_pool.append(np.random.dirichlet(np.full(n, 0.3)))

    # uniform – just a sanity check
    init_pool.append(np.full(n, 1.0 / n, dtype=np.float64))

    # ------------------------------------------------------------------
    # 3️⃣  Optimise each candidate (time‑budget aware)
    for idx, init in enumerate(init_pool):
        if time.time() >= deadline:
            break
        print(f"--- Starting Adam on candidate {idx+1}/{len(init_pool)} ---")
        seq, sc = _adam_optimize(
            n=n,
            init_seq=init,
            deadline=deadline,
            max_iters=50000,
            alpha=0.06,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            lambda0=1.0,
            lambda_max=200.0,
        )
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** Global best updated (candidate {idx+1}) score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final hill‑climbing refinement (if time permits)
    if overall_best_seq is not None and time.time() < deadline:
        hill_deadline = deadline
        seq, sc = _hill_climb(overall_best_seq, hill_deadline, max_trials=20000)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** After hill‑climb: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Fallback (should never be needed)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"Final best: length={len(overall_best_seq)} score={overall_best_score:.6f}")

    # Return a plain Python list for the evaluation harness
    return overall_best_seq.tolist()
