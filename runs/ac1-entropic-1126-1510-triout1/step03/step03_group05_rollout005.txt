   random block).  Each candidate is projected onto the simplex
   {a≥0, Σa = √(2n)} and its max‑self‑convolution is evaluated.  The
   best of these seeds becomes the starting point.

2. **Directed simulated‑annealing** – keep the current vector `a`,
   its full self‑convolution `c` and the current maximum `M = max(c)`.
   At each iteration the gradient of the active constraints
   (indices where `c` attains `M`) is computed exactly:
        grad_i = 2·∑_{k∈active} a_{k‑i}   (if the index is in range)
   The algorithm moves a small amount of mass from the index with the
   largest gradient (most responsible for the peak) to the index with
   the smallest gradient.  The convolution is updated in O(n) by a
   slice‑add operation, and the move is accepted if it improves `M` or
   with a Metropolis probability that slowly cools.

   The step size and temperature decay exponentially; occasional
   full recomputation of the convolution (via FFT) eliminates accumulated
   floating‑point drift.

3. **Final symmetrisation & projection** – the best vector discovered
   is optionally symmetrised (averaged with its reverse) and re‑projected
   onto the simplex once more before returning.

The whole routine respects a 1000 s time budget, prints intermediate
best scores and returns the best sequence as a plain Python list.
"""

import math
import time
import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution of a using an FFT (length 2·n‑1)."""
    n = a.shape[0]
    size_fft = 1 << ((2 * n - 2).bit_length())   # next power of two ≥ 2n‑1
    fu = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fu * fu, n=size_fft)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient for the max‑convolution constraints.
    Returns grad_i = 2·∑_{k∈active} a_{k‑i} (if 0≤k‑i<n, else 0).
    """
    n = a.shape[0]
    grad = np.zeros(n, dtype=float)
    for k in active:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        # indices i0 … i1 correspond to a[k‑i] for i in that range
        idx = np.arange(i0, i1 + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function with low evaluate_sequence() value."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.4   # small safety margin

    # --------------------------------------------------------------
    # Problem size (use the reference length if available)
    try:
        n = len(height_sequence_1)    # type: ignore[name-defined]
        have_ref = True
    except Exception:                 # fallback length (fast enough)
        n = 2500
        have_ref = False

    target_sum = math.sqrt(2.0 * n)   # Σa = √(2n) makes the score = max(conv)

    # --------------------------------------------------------------
    # Helper to compute max self‑convolution quickly (FFT for fresh eval)
    def _max_conv(arr: np.ndarray) -> float:
        return float(_self_convolution_fft(arr).max())

    # --------------------------------------------------------------
    # 1. PARAMETRIC SHAPE SEARCH
    best_a = None
    best_score = float("inf")

    # uniform baseline
    a = np.full(n, target_sum / n, dtype=float)
    sc = _max_conv(a)
    best_a, best_score = a.copy(), sc
    print(f"[init] uniform score = {sc:.6f}")

    # reference if present
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)
        ref = np.maximum(ref, 0.0)
        ref = _project_to_simplex(ref, target_sum)
        sc = _max_conv(ref)
        if sc < best_score:
            best_a, best_score = ref.copy(), sc
        print(f"[init] reference score = {sc:.6f}")

    # families – a moderate number of samples (fast)
    N_SAMPLES = 80

    # ----- inverse‑sqrt (single‑ended) -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        sc = _max_conv(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- double‑ended inverse‑sqrt -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        i = np.arange(n, dtype=float)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _max_conv(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- double‑ended power‑law -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        p = np.random.rand() * 2.5 + 0.1
        i = np.arange(n, dtype=float)
        a = (1.0 / (i + beta + 1.0) ** p +
             1.0 / ((n - 1 - i) + beta + 1.0) ** p)
        a = _project_to_simplex(a, target_sum)
        sc = _max_conv(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- exponential decay (double‑ended) -----
    for _ in range(N_SAMPLES):
        decay = np.random.rand() * 0.012 + 0.0005
        i = np.arange(n, dtype=float)
        a = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
        a = _project_to_simplex(a, target_sum)
        sc = _max_conv(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- cosine‑based (large at ends) -----
    for _ in range(N_SAMPLES):
        p = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        sc = _max_conv(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- mixture uniform & inverse‑sqrt -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        lam = np.random.rand()
        a = lam * np.ones(n, dtype=float) + (1.0 - lam) * (
            1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _max_conv(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- random block candidates (few large blocks) -----
    for _ in range(300):
        nb = np.random.randint(2, 7)            # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _max_conv(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    print(f"[shape] best score after families = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. DIRECTED SIMULATED‑ANNEALING REFINEMENT
    a = best_a.copy()
    conv = _self_convolution_fft(a)                     # length 2n‑1
    cur_max = float(conv.max())
    best_score = cur_max
    best_a = a.copy()

    # SA hyper‑parameters
    step_factor = 0.5                # controls size of transferred mass
    step_decay = 2e-6                # exponential decay per iteration
    temp = 1e-3                      # temperature for Metropolis moves
    temp_decay = 5e-6                # temperature decay
    it = 0
    accept_cnt = 0

    # main loop
    while time.time() < deadline and it < 5_000_000:
        # identify active constraints (indices where conv ≈ cur_max)
        eps = max(1e-12, cur_max * 1e-9)
        active = np.where(conv >= cur_max - eps)[0]

        # gradient of active constraints
        grad = _gradient_from_active(a, active)

        # source = index with largest gradient, destination = smallest
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        # if the worst indices happen to be equal or source has no mass, fall back
        if i_src == i_dst or a[i_src] <= 1e-14:
            # random fallback
            i_src = np.random.randint(n)
            i_dst = np.random.randint(n - 1)
            if i_dst >= i_src:
                i_dst += 1
            if a[i_src] <= 1e-14:
                it += 1
                continue

        # step size (transfer a fraction of source mass)
        delta = a[i_src] * (step_factor * np.random.rand())
        if delta <= 0.0:
            it += 1
            continue

        # ----- incremental convolution update (O(n)) -----
        new_conv = conv.copy()
        # add/subtract shifted copies of a
        new_conv[i_dst:i_dst + n] += 2.0 * delta * a
        new_conv[i_src:i_src + n] -= 2.0 * delta * a
        # delta² point contributions
        d2 = delta * delta
        new_conv[2 * i_src] += d2
        new_conv[2 * i_dst] += d2
        new_conv[i_src + i_dst] -= 2.0 * d2

        new_max = float(new_conv.max())

        # acceptance rule
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max) / max(temp, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            a[i_src] -= delta
            a[i_dst] += delta
            conv = new_conv
            cur_max = new_max
            accept_cnt += 1
            if cur_max < best_score - 1e-12:
                best_score = cur_max
                best_a = a.copy()
                print(f"[{it}] new best = {best_score:.6f}")

        # schedule decay
        step_factor = max(step_factor * (1.0 - step_decay), 1e-4)
        temp = max(temp * (1.0 - temp_decay), 1e-6)
        it += 1

        # occasional full recomputation to avoid drift
        if it % 2000 == 0:
            conv = _self_convolution_fft(a)
            cur_max = float(conv.max())

    print(f"[anneal] finished after {it} iterations, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. OPTIONAL SYMMETRIZATION (often lowers the peak)
    sym = (best_a + best_a[::-1]) * 0.5
    sym = _project_to_simplex(sym, target_sum)
    sym_score = _max_conv(sym)
    if sym_score < best_score:
        best_a = sym
        best_score = sym_score
        print(f"[sym] symmetrised score = {best_score:.6f}")

    # final safety projection (ensures exact feasibility)
    best_a = _project_to_simplex(best_a, target_sum)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
