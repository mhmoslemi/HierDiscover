def max_conv(a):
    m = len(a)
    L = 1 << ((2*m - 1).bit_length())
    # compute FFT for real numbers
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa*fa, n=L)[:2*m-1]
    # due to small negative values due to numeric errors, clip to zero
    conv = np.maximum(conv, 0.0)
    return conv.max(), conv

def gradient_max_conv(a, conv):
    # conv is the full convolution array (length 2n-1)
    max_c = np.max(conv)
    mask = (conv == max_c)
    # number of max locations
    cnt = np.count_nonzero(mask)
    w = np.zeros_like(conv)
    w[mask] = 1.0 / cnt
    # compute conv_w_rev = convolution of w and reversed a
    L = 1 << ((2*len(a) - 1).bit_length())
    fw = np.fft.rfft(w, n=L)
    fa_rev = np.fft.rfft(a[::-1], n=L)
    conv_wr = np.fft.irfft(fw * fa_rev, n=L)[:2*len(a)-1]
    g_a = 2.0 * conv_wr[len(a)-1:len(a)-1+len(a)]
    return g_a

   # Precompute function to compute max conv quickly using FFT.

       conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
       conv = np.maximum(conv, 0.0)
       return conv.max()

   a = a_start.copy()

   best_a = a.copy()

"""
Propose a step‑function (non‑negative sequence) that minimizes
the maximum of its self‑convolution under the ℓ¹‑norm constraint
∑a_i = sqrt(2·n).  The algorithm combines three phases:

1.  Simulated‑annealing (SA) on the simplex – cheap random mass moves
    that directly minimise the max‑convolution (the true score, because
    the ℓ¹‑norm is fixed).

2.  A continuation scheme based on a smooth approximation of the
    max‑operator (same ideas as the reference code) using L‑BFGS‑B.
    Starts from the best SA solution, runs a geometric cooling schedule
    and keeps the best feasible point found.

3.  Final projected sub‑gradient polishing (hard max) for the remaining
    time budget.

All helpers are top‑level functions (no closures) and the code works
with any sequence length – it uses the length of the optional global
`height_sequence_1` if present, otherwise defaults to 2000.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (soft‑max surrogate) and its gradient w.r.t. y.
    """
    p = _softmax(y)          # Σp = 1, p_i ≥ 0
    a = S * p                # Σa = S

    conv = np.convolve(a, a)      # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. heights a
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)                     # scalar = Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for all FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    # numerical noise may produce tiny negatives – clamp to zero
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


def _gradient_max_convolution(a: np.ndarray, conv: np.ndarray,
                             size: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a (ordered‑pair version).
    """
    max_c = conv.max()
    # treat values within 1e‑12 of the maximum as active
    mask = np.isclose(conv, max_c, atol=1e-12)
    cnt = mask.sum()
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt

    # convolution w * a_rev
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * len(a) - 1]
    g_a = 2.0 * conv_wr[len(a) - 1 : len(a) - 1 + len(a)]
    return g_a

def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: repeatedly move a random amount of mass
    from one index to another.  The score is the max‑convolution (which
    equals the true evaluation because ∑a = S = √(2·n)).
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    # temperature schedule – start fairly high, cool to near zero
    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-6

    it = 0
    while time.time() - start_time < budget:
        # linear cooling
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        # pick a different index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of a[i] (up to 30 %)
        delta = np.random.rand() * a[i] * 0.3
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = False
        if diff < 0.0:
            accept = True
        else:
            # Metropolis criterion
            if T > 0.0 and np.random.rand() < math.exp(-diff / T):
                accept = True

        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score


def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation on the smooth‑max surrogate (same as the reference
    implementation) but with a longer schedule and more inner
    L‑BFGS iterations.
    """
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    # initialise temperature from current max‑convolution
    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    mu_factor = 0.5
    mu_min = 1e-6
    maxiter_per_mu = 250

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true hard max (no smoothing).
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # step‑size parameters
    eta0 = S * 0.5 / n            # roughly the average entry magnitude
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 5000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # gradient of the hard max
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step‑function within the allotted time.
    Returns the best sequence found (list of non‑negative floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -------------------- problem size --------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 2000          # reasonable default when no reference is available
        have_ref = False

    S = math.sqrt(2.0 * n)            # ℓ¹‑norm that makes the score equal max‑conv

    # -------------------- initial sequence --------------------
    if have_ref:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = _project_to_simplex(a0, S)
    else:
        a0 = np.full(n, S / n, dtype=float)   # uniform start

    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init]   score = {best_score:.6f}")

    # -------------------- 1) Simulated annealing --------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sa_time = remaining * 0.60
        print(f"[SA] running for {sa_time:.1f}s")
        best_a, best_score = _simulated_annealing(best_a, S, sa_time, time.time())
        print(f"[SA]    best = {best_score:.6f}")

    # -------------------- 2) L‑BFGS continuation (smooth max) --------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print(f"[L‑BFGS] refining for {remaining:.1f}s")
        best_a, best_score = _l_bfgs_refine(best_a, S, remaining, time.time())
        print(f"[L‑BFGS] best = {best_score:.6f}")

    # -------------------- 3) Final hard‑max sub‑gradient polish --------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[final] polishing for {remaining:.1f}s")
        best_a, best_score = _final_subgradient(best_a, S, remaining, time.time())
        print(f"[final]  best = {best_score:.6f}")

    # safety fallback (should never be needed)
    if best_score == float("inf"):
        best_a = S * np.ones(n) / n
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
