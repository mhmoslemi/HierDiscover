"""
Step‑function optimizer for the `evaluate_sequence` challenge.

The algorithm mixes several complementary heuristics:

1.  A diverse pool of initial distributions (arcsine, symmetric Beta,
    uniform, Dirichlet, sparse spikes, plus up‑sampled best‑so‑far).

2.  Phase 1 – Exponentiated‑gradient (mirror descent) on a soft‑max
    approximation of the self‑convolution peak.  The temperature λ is
    raised to a large λ_max (≈80) while the step‑size decays as 1/√t.

3.  Phase 2 – Adam optimisation on the same smooth objective (adds
    momentum and adaptive step‑sizes).

4.  Phase 3 – Exact sub‑gradient descent on the true max‑convolution
    (projected onto the simplex).

5.  Phase 4 – Simulated‑annealing hill‑climbing moving tiny amounts of
    mass between indices; moves that improve the unsmoothed score are
    always kept, otherwise accepted with probability exp(−Δ/T) with a
    linear cooling schedule.

All vectors are kept normalised to ℓ₁‑norm 1, which leaves the
`evaluate_sequence` objective invariant.  The routine respects the
global time budget (default 1000 s) and returns the best sequence found
as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via power‑of‑two FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Compute g = 2·corr(w, a) where w has length 2·n‑1 and a length n.
    Returns a vector of length n.
    """
    n = a.shape[0]
    m = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = n - 1
    return 2.0 * corr[start:start + n]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum 1."""
    n = a.shape[0]
    b = _conv_full_fft(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid (i+½)/n, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid exact 0/1 which would give nan for alpha<1
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1.0 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a modest but diverse set of starter probability vectors."""
    cand = []

    cand.append(_arcsine_distribution(n))                     # arcsine
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))        # uniform

    # symmetric betas on a coarse grid of α
    for alpha in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.9, 1.2):
        cand.append(_beta_distribution(n, alpha))

    # a few random betas
    for _ in range(3):
        alpha = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, alpha))

    # sparse “spike” vectors
    k = max(2, n // 500)                     # around 0.2 % non‑zero entries
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    cand.append(a / a.sum())

    # low‑concentration Dirichlet draw
    a = np.random.dirichlet(np.full(n, 0.3))
    cand.append(a)

    return cand

def _optimise_exponential(a0: np.ndarray,
                          max_iters: int,
                          lam_max: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on a soft‑max of the
    convolution.  Returns the best vector seen and its exact score.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_sc = _score(a)

    n = a.shape[0]
    # FFT size for the correlation step
    m = 1 << ((3 * n - 2).bit_length())

    lam = 0.3
    lam_inc = (lam_max - lam) / max(max_iters, 1)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_sc = 2.0 * n * b.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] exp n={n:5d} it={it:5d} score={best_sc:.6f}")

        # soft‑max weights
        maxb = b.max()
        exp_arg = lam * (b - maxb)                     # ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient
        g = _gradient_via_correlation(w, a)

        # exponentiated‑gradient step (mirror descent with KL divergence)
        eta = 0.15 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex (simple ℓ₁ projection = renorm)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # occasional jitter to escape tiny plateaus
        if it % 800 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        lam = min(lam_max, lam + lam_inc)

    return best_seq, best_sc

def _optimise_adam(a0: np.ndarray,
                   max_iters: int,
                   lam_max: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the same soft‑max objective used in the mirror
    descent.  The gradient is the same correlation vector.  Returns the
    best sequence seen and its exact score.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_sc = _score(a)

    n = a.shape[0]
    m = 1 << ((3 * n - 2).bit_length())

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m_t = np.zeros_like(a)
    v_t = np.zeros_like(a)

    lam = 0.3
    lam_inc = (lam_max - lam) / max(max_iters, 1)
    alpha = 0.02    # base learning rate

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_sc = 2.0 * n * b.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] adam n={n:5d} it={it:5d} score={best_sc:.6f}")

        # soft‑max weights
        maxb = b.max()
        w = np.exp(lam * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient
        g = _gradient_via_correlation(w, a)

        # Adam update
        m_t = beta1 * m_t + (1 - beta1) * g
        v_t = beta2 * v_t + (1 - beta2) * (g * g)
        m_hat = m_t / (1 - beta1 ** it)
        v_hat = v_t / (1 - beta2 ** it)

        a = a - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 5e-5
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, s=1.0)

        lam = min(lam_max, lam + lam_inc)

    return best_seq, best_sc

def _subgradient_descent(a0: np.ndarray,
                         max_iters: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Uses the classic sub‑gradient g_i = 2·a_{k*−i}.
    """
    a = a0.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_sc = _score(a)

    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # build sub‑gradient g_i = 2·a_{k*−i}
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = 0.25 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_sc = _score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] sub n={n:5d} it={it:5d} score={best_sc:.6f}")

        # jitter to avoid stagnation
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_sc

def _hill_climb_sa(a0: np.ndarray,
                  max_trials: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climbing with simulated‑annealing acceptance.
    Moves a tiny amount of mass from a donor index to a receiver.
    """
    n = a0.shape[0]
    cur = a0.copy()
    cur_sc = _score(cur)
    best_seq = cur.copy()
    best_sc = cur_sc

    # linear cooling schedule
    T0 = 0.015
    for t in range(max_trials):
        if time.time() >= deadline:
            break
        T = max(1e-6, T0 * (1.0 - t / max_trials))

        # choose donor (biased towards larger entries)
        donor = np.random.choice(n, p=cur / cur.sum())
        if cur[donor] < 1e-12:
            continue
        receiver = np.random.randint(n)
        if donor == receiver:
            continue

        # transfer amount (up to 5 % of donor, but never >0.01)
        delta = np.random.rand() * min(cur[donor] * 0.05, 0.01)

        trial = cur.copy()
        trial[donor] -= delta
        trial[receiver] += delta

        # trial is already feasible (non‑negative, sum unchanged)
        trial_sc = _score(trial)

        accept = False
        if trial_sc < cur_sc:
            accept = True
        else:
            # simulated‑annealing acceptance
            if np.random.rand() < np.exp(-(trial_sc - cur_sc) / T):
                accept = True

        if accept:
            cur = trial
            cur_sc = trial_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_seq = cur.copy()
                print(f"[{time.time():.1f}] SA trial={t} score={best_sc:.6f}")

    return best_seq, best_sc

                overall_score = sc
                overall_best = seq.copy()
                print(f"[{time.time():.1f}] *** Global best updated (n={n}) score={overall_score:.6f}")

            if overall_score <= 1.503:
                print("[early exit] target achieved")
                break

        # ------------------------------------------------------------------
        # b) Final stochastic hill‑climb on the best vector for this length
        if best_n_seq is not None and time.time() < deadline:
            seq, sc = _hill_climb_sa(best_n_seq,
                                     max_trials=20000,
                                     deadline=deadline)
            if sc < overall_score:
                overall_score = sc
                overall_best = seq.copy()
                print(f"[{time.time():.1f}] *** After SA: new global best score={overall_score:.6f}")
                if overall_score <= 1.503:
                    print("[early exit] target achieved after SA")
                    break

    # ------------------------------------------------------------------
    # 2️⃣  Fallback (unlikely to be used)
    if overall_best is None:
        n = 3000
        overall_best = np.full(n, 1.0 / n, dtype=np.float64)
        overall_score = _score(overall_best)
        print(f"[{time.time():.1f}] fallback uniform sequence, score={overall_score:.6f}")

    print(f"[{time.time():.1f}] Final best: length={len(overall_best)} score={overall_score:.6f}")
    # Return a plain list for the evaluation harness
    return overall_best.tolist()
