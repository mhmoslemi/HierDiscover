def evaluate_sequence(sequence: list[float]) -> float:
    # security checks: ensure list, not empty, each element int/float, not NaN or inf, etc.
    sequence = [float(x) for x in sequence]
    # protect against negative numbers: after conversion: sequence = [max(0, x) for x in sequence]
    # protect against numbers that are too large: sequence = [min(1000.0, x) ...]
    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))
```

"""
Propose a non‑negative step function (list of heights) that minimizes
`evaluate_sequence`.  

The algorithm combines:
  * a projected (soft‑max) gradient descent on the probability simplex,
    annealing the soft‑max temperature to approach the true max‑convolution
    sub‑gradient;
  * Nesterov‑style momentum and a √t diminishing step size;
  * occasional random jitter to escape plateaus;
  * a cheap local‑search phase (pairwise mass swaps) after each gradient run;
  * multiple restarts with diverse initial shapes (arcsine, uniform,
    random, previous best, and a two‑spike pattern) across several
    sequence lengths.

All operations are performed with FFT‑based convolutions for O(n log n)
complexity.  The best sequence found before the time budget expires is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                      # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":

        except Exception:

    else:

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·b_k) w.r.t. a.
    For very large α this reduces to a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # ----- compute soft‑max weights (or uniform over all maximal entries) -----
    if alpha > 2000:
        max_b = b.max()
        # treat all entries within a tiny tolerance as active
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()           # for numerical stability
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ----- convolution w * a_rev (a_rev = a reversed) -----
    a_rev = a[::-1]
    # size_fft is guaranteed to be ≥ len(w)+len(a_rev)-1
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g

def _optimize_one_length(init_a: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex to minimise max(self‑convolution).
    Returns the best sequence discovered for this length and the associated
    internal score 2·n·max_b (i.e. the exact value of evaluate_sequence when
    the sum of the coefficients equals 1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # hyper‑parameters
    beta = 0.8                     # momentum factor
    eta0 = 0.8                     # initial step‑size constant
    velocity = np.zeros_like(a)
    alpha = 5.0                    # start temperature for soft‑max
    alpha_mul = 1.06               # geometric increase
    alpha_max = 20000.0
    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for w (2n‑1) + a_rev
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient (soft‑max or sub‑gradient)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Nesterov‑style momentum update
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection onto simplex
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny random jitter to escape plateaux
        if it % 2500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: repeatedly move a random fraction
    of mass from one index to another and keep the move if the max‑convolution
    improves.  Stops when the deadline is reached.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] <= 1e-12:
            continue
        # transfer a random fraction (up to 30%) of the mass at i to j
        frac = np.random.rand() * 0.30
        delta = best_seq[i] * frac
        a_new = best_seq.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate
        b = _fft_self_convolve(a_new)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_seq = a_new
        # tiny jitter every few thousand iterations
        if it % 5000 == 0:
            a_new = best_seq + np.random.randn(n) * 1e-6
            a_new = _project_onto_simplex(a_new, s=1.0)
            b = _fft_self_convolve(a_new)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_seq = a_new

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point called by the grader.  It searches for a step‑function
    (list of non‑negative heights) with a low `evaluate_sequence` score.
    """
    np.random.seed(seed)
    # leave a small safety margin for final printing / cleanup
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # base length – reuse a previously discovered length if available
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # candidate lengths (capped at 30 000 for speed)
    length_factors = (0.7, 0.85, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(300, int(base_n * f))
        for f in length_factors
        if max(300, int(base_n * f)) <= 30000
    })

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate roughly 30 % of remaining time to this length
        slice_deadline = time.time() + max(0.1, remaining * 0.30)

        # initialise with a variety of shapes
        for init_kind in ("prev", "arcsine", "uniform", "random", "spike"):
            if time.time() >= slice_deadline:
                break

            if init_kind == "spike":
                # two‑spike pattern: half the mass near each end
                a0 = np.zeros(n, dtype=np.float64)
                pos1 = np.random.randint(max(1, n // 10))
                pos2 = np.random.randint(n - max(1, n // 10), n)
                a0[pos1] = 0.5
                a0[pos2] = 0.5
                a0 = _project_onto_simplex(a0, s=1.0)
            else:
                a0 = _initial_sequence(n, init_kind)

            # gradient‑based optimisation
            seq, int_score = _optimize_one_length(a0, n, slice_deadline)

            # short local‑search phase (if any time left)
            time_left = slice_deadline - time.time()
            if time_left > 0.5:
                seq, int_score = _local_search(seq, n,
                                               time.time() + min(time_left, 5.0))

            # evaluate with the official black‑box (more reliable)
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = int_score

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq.copy()
                print(f"*** New global best → len={n} init={init_kind} "
                      f"score={global_best_score:.6f}")

            # early exit if target already reached
            if global_best_score <= 1.503:
                break

        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # final refinement on the overall best (if time permits)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _local_search(global_best_seq,
                                       len(global_best_seq),
                                       deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
