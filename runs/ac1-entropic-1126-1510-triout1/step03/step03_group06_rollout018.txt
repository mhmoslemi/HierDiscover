    b = conv(a, a)
    maxb = b.max()
    # Weighted softmax with shift for numerical stability:
    b_shift = b - maxb
    w = np.exp(lam * b_shift)
    w_sum = w.sum()
    # Softmax approx of max: f = maxb + (1/lam) * log(w_sum)

def _adam_optimize(...):
    a = init_seq.astype(np.float64).copy()
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_seq = a.copy()
    best_score = _score(a)
    ft = 1 << ((3*n - 2).bit_length()) # FFT length, as earlier

    lam = lambda0
    lam_inc = (lambda_max - lambda0) / max_iters

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break

        # compute conv and current true score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} score={best_score:.6f}")

        # compute softmax weights
        maxb = b.max()
        exp_arg = lam * (b - maxb)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # compute gradient g
        g = _gradient_via_correlation(w, a, ft)

        # Adam update on log-space: use multiplicative update directly
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)

        # bias correction
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

"""
Propose a very low‑score step‑function.

Algorithm
---------
1. For a few problem sizes (4000‑8000) generate a small set of diverse
   initial probability vectors (arcsine, beta, uniform, Dirichlet, random spikes).
2. For each start run a long mirror‑descent with an Adam‑style
   multiplicative update on a soft‑max approximation of the peak of the
   self‑convolution.  The “temperature’’ λ is annealed linearly from a small
   value up to a large one (400) so that the approximation becomes tight.
3. Refine the best candidate of a given length with an Adam‑style *exact*
   sub‑gradient descent (the weight vector is a Kronecker delta at the
   current maximal convolution index).
4. Keep the globally best vector seen over all lengths.
5. Finally run a cheap stochastic hill‑climb that moves a tiny amount of
   mass between two positions and accepts the change if the (exact) score
   improves.
6. Return the best sequence as a plain Python list.

All vectors are kept non‑negative and normalised to sum 1 – the
evaluation function is invariant under this scaling.
The routine respects the global 1000 s budget.
"""

from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised to sum 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> List[np.ndarray]:
    """Generate a modest, diverse pool of starting probability vectors."""
    cand = []

    # 1) arcsine (known good performer)
    cand.append(_arcsine_distribution(n))

    # 2) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) a few symmetric betas
    for alpha in (0.25, 0.4, 0.55, 0.7, 0.9, 1.2):
        cand.append(_beta_distribution(n, alpha))

    # 4) random betas
    for _ in range(2):
        alpha = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, alpha))

    # 5) sparse “spike’’ vector
    k = max(2, n // 500)
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    cand.append(a / a.sum())

    # 6) Dirichlet with low concentration
    a = np.random.dirichlet(np.full(n, 0.3))
    cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 15000,
    eta0: float = 0.12,
    lambda0: float = 0.1,
    lambda_max: float = 400.0,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent with Adam‑style multiplicative updates on a
    soft‑max approximation of max(conv(a,a)).
    Returns the best vector (sum 1) and its true (unsmoothed) score.
    """
    a = init_seq.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    # FFT length for correlation (>= 3n‑2)
    m = 1 << ((3 * n - 2).bit_length())

    # Adam moments
    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta1_pow = 1.0
    beta2_pow = 1.0

    # linear lambda schedule
    lam = lambda0
    lam_inc = (lambda_max - lambda0) / max_iters

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current exact score
        conv_full = _conv_full_fft(a)
        cur_score = 2.0 * n * conv_full.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} ADAM it={it:5d} score={best_score:.6f}")

        # soft‑max weights
        maxb = conv_full.max()
        exp_arg = lam * (conv_full - maxb)          # ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv_full, 1.0 / conv_full.shape[0])
        else:
            w /= w_sum

        # gradient w.r.t. a
        g = _gradient_via_correlation(w, a, m)

        # Adam updates (multiplicative)
        m1 = beta1 * m1 + (1 - beta1) * g
        m2 = beta2 * m2 + (1 - beta2) * (g * g)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m1_hat = m1 / (1 - beta1_pow)
        m2_hat = m2 / (1 - beta2_pow)

        eta = eta0 / np.sqrt(it)   # slowly decaying base step size
        factor = np.exp(-eta * m1_hat / (np.sqrt(m2_hat) + eps))
        a = a * factor

        # renormalise to the simplex
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total > 0.0:
            a = a / total
        else:
            a = np.full(n, 1.0 / n)

        # increase temperature
        lam = min(lambda_max, lam + lam_inc)

        # occasional jitter to avoid getting stuck
        if it % 2000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score

def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 50000,
) -> Tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a small amount of mass from
    one index to another and keep the change if the (unsmoothed) objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    overall_best_seq: np.ndarray = None
    overall_best_score: float = float("inf")

    # ------------------------------------------------------------------
    # 0) historic sequence if available
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) problem sizes to explore (moderate lengths keep FFT cheap)
    candidate_lengths = [4000, 5000, 6000, 8000]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        print(f"\n--- Exploring length n={n} ---")
        init_pool = _initial_candidates(n)

        best_n_seq = None
        best_n_score = float("inf")

        # ------------------------------------------------------------------
        # a) Adam mirror‑descent on soft‑max objective
        for init in init_pool:
            if time.time() >= deadline:
                break
            seq, sc = _adam_optimize(
                n=n,
                init_seq=init,
                deadline=deadline,
                max_iters=15000,
                eta0=0.12,
                lambda0=0.1,
                lambda_max=400.0,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"   >>> new best for n={n}: score={best_n_score:.6f}")

        # ------------------------------------------------------------------
        # b) Adam sub‑gradient refinement (if time permits)
        if best_n_seq is not None and time.time() < deadline:
            seq, sc = _adam_subgradient(
                n=n,
                init_seq=best_n_seq,
                deadline=deadline,
                max_iters=8000,
                eta0=0.08,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"   *** sub‑grad refined n={n}: score={best_n_score:.6f}")

        # ------------------------------------------------------------------
        # c) update global best
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # early exit if target already met
        if overall_best_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # 2) final local hill‑climb refinement
    if overall_best_seq is not None and time.time() < deadline:
        seq, sc = _hill_climb(overall_best_seq, deadline, max_trials=50000)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** After hill‑climb: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) fallback (should never be needed)
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"Final best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    return overall_best_seq.tolist()
