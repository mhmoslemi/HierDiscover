analysisFirst, we need to understand the problem: The evaluate_sequence function takes a sequence and computes:

- n = len(sequence)
- b = np.convolve(sequence, sequence), length 2*n-1.
- max_b = max(b)
- sum_a = sum(sequence)
- score = 2 * n * max_b / (sum_a**2).

So to minimize the score, we want to keep max_b as low as possible relative to sum_a, or maximize sum_a while keeping max_b low: but sum_a cannot be too small, as sum_a < 0.01 is penalized. Also there is a cap on each coefficient: after conversion, they are clipped to at most 1000, but that won't be an issue.

We can use normalization: many algorithms assume we restrict the ℓ¹ norm to certain value, e.g., sqrt(2n) to set target sum. Indeed, if we fix sum_a = s, then score = 2n * max_b / s^2. Since s appears in denominator squared, raising s while fixing shape (like scaling a) reduces score quadratically. """
Propose a low‑score step function.

Algorithm
---------
1. Initialise the sequence either from a known reference (height_sequence_1)
   or from a random point on the ℓ¹‑simplex {a ≥ 0, Σa = √(2 n)}.
2. A short projected‑gradient phase reduces the peak of the self‑convolution.
3. Repeated linear‑programming “mix‑steps’’:
   * For the current a compute M = max(a*a).
   * Solve   maximise Σ g_j  subject to   a⋆g ≤ M , g ≥ 0 .
   * Scale g to the target ℓ¹‑norm and convex‑combine a with g.
   * Accept the combination if it improves the score.
   This LP‑step is the core of the reference method but is implemented
   explicitly (SciPy/HiGHS) and used several times.
4. A cheap block‑search explores piecewise‑constant patterns.
5. Finally, tiny random mass‑swaps finish the optimisation.

The whole routine respects a 1000 s wall‑clock budget and prints progress.
"""

import time
import math
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog


def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(a: np.ndarray) -> float:
    """Maximum entry of the full self‑convolution a∗a."""
    # Use direct convolution; for n≈3000 it is fast enough.
    conv = np.convolve(a, a)
    return float(conv.max())


def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int) -> np.ndarray:
    """Sub‑gradient of max conv based on the `top_k` largest convolution entries."""
    n = a.shape[0]
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i = np.arange(n)
    for k in idxs:
        j = k - i
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        diag = mask & (i == j)
        grad[diag] -= a[i[diag]]
    return grad


def _prepare_lp_structure(n: int):
    """Pre‑compute the row/column index arrays for the Toeplitz convolution matrix."""
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    shape = (2 * n - 1, n)
    return rows, cols, shape


def _lp_max_sum(a: np.ndarray, M: float, rows: np.ndarray,
                cols: np.ndarray, shape: tuple, target_sum: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to a⋆g ≤ M  (component‑wise)
                   g ≥ 0
    Returns the optimal g (may have sum > target_sum).
    """
    n = a.shape[0]
    # Data for the sparse Toeplitz matrix: each column repeats the whole vector a.
    data = np.tile(a, n)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=shape)
    b_ub = np.full(shape[0], M, dtype=float)

    # maximise Σ g  → minimise −Σ g
    c = -np.ones(n, dtype=float)

    # simple finite upper bound prevents unboundedness when a_i == 0
    bounds = [(0.0, target_sum) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x

    while time.time() < deadline:
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size == 0:
            break
        i = np.random.choice(nz)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = best_a[i] * np.random.uniform(0.1, 0.5)
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        cand_max = _max_conv(cand)
        cand_score = 2.0 * n * cand_max / (target_sum ** 2)
        if cand_score < best_score:
            best_score = cand_score
            best_max = cand_max
            best_a = cand
            print(f"[swap] new best score={best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
