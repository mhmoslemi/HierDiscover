The implementation is pure NumPy, runs under the 1000 s budget and returns

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast O(n log n) projection (see Duchi et al. 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(½,½)) distribution of length n, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _beta_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric Beta(p,p) distribution on a grid of n points.
    For p=0.5 this coincides with the arcsine distribution.

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (no padding artefacts)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Returns the soft‑max weights w_j = exp(alpha·b_j) / Σ exp(alpha·b_j).
    The shift by max(b) guarantees numerical stability.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    return w

            except Exception:
                final_score = internal_score

            if final_score < overall_best_score:
                overall_best_seq = seq.copy()
                overall_best_score = final_score
                print(f"*** New global best → len={n} init={init_kind} "
                      f"score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break

        if overall_best_score <= target_score:
            break

        # ------------------------------------------------------------------
        # a lightweight grid over symmetric Beta(p,p) shapes
        beta_ps = (0.3, 0.4, 0.5, 0.6, 0.8, 1.0)
        for p in beta_ps:
            if time.time() >= slice_deadline:
                break
            a0 = _beta_distribution(n, p)
            seq, internal_score = _optimize_one_length(a0, n, slice_deadline)

            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = internal_score

            if final_score < overall_best_score:
                overall_best_seq = seq.copy()
                overall_best_score = final_score
                print(f"*** New global best → len={n} beta={p} "
                      f"score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break

        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Final refinement of the currently best candidate (if time permits)
    if overall_best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            n = len(overall_best_seq)
            seq, _ = _optimize_one_length(overall_best_seq, n, deadline)
            try:
                final_score = evaluate_sequence(seq.tolist())
                if final_score < overall_best_score:
                    overall_best_seq = seq.copy()
                    overall_best_score = final_score
                    print(f"*** Refined global best → len={n} score={overall_best_score:.6f}")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # safety net – should never be triggered
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
