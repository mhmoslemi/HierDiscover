def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law on {0,…,n‑1} (probability vector)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a via a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, conv: np.ndarray,
                 alpha: float) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·conv) w.r.t. a.
    Implements the convolution w * a_rev, where w are the soft‑max
    weights of the convolution entries.
    """
    n = a.shape[0]
    # soft‑max weights
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # convolution w * a_rev
    a_rev = a[::-1]
    size_fft = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    tmp = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    grad = 2.0 * tmp[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _adam_step(a: np.ndarray, grad: np.ndarray,
               m: np.ndarray, v: np.ndarray,
               it: int, lr0: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Perform one Adam update (with Euclidean projection onto the simplex).
    Returns the new a, and the updated first‑ and second‑moment vectors.
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)

    m_hat = m / (1.0 - beta1 ** it)
    v_hat = v / (1.0 - beta2 ** it)

    lr = lr0 / (np.sqrt(it) + 1.0)          # a mild decay
    a_new = a - lr * (m_hat / (np.sqrt(v_hat) + eps))
    a_new = _project_onto_simplex(a_new, s=1.0)
    return a_new, m, v


# ----------------------------------------------------------------------
def _optimise_length(n: int, init_a: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Run the two‑phase optimisation for a fixed length n.
    Returns the best sequence found (as a numpy array) and its exact score.
    """
    # ----- Phase 1 : smooth Adam optimisation -------------------------
    a = init_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 1.0               # temperature for the soft‑max
    alpha_mul = 1.025
    alpha_max = 15000.0
    lr0 = 0.25                 # base learning rate for Adam

    best_a = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} Adam it={it} score={best_score:.6f}")

        # gradient of the smooth approximation
        grad = _smooth_grad(a, conv, alpha)

        # Adam update + simplex projection
        a, m, v = _adam_step(a, grad, m, v, it, lr0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    # ----- Phase 2 : local hill‑climbing -----------------------------
    # Use the remainder of the allocated time for a simple stochastic
    # refinement that works directly on the exact objective.
    a_cur = best_a.copy()
    conv_cur = _fft_self_convolve(a_cur)
    cur_score = 2.0 * n * conv_cur.max()
    best_seq = a_cur.copy()
    best_seq_score = cur_score

    # hill‑climbing parameters
    eps_move = 1e-3          # relative amount of mass moved per trial
    attempts = 0

    while time.time() < deadline:
        attempts += 1
        # pick donor i with non‑trivial mass
        i = np.random.randint(n)
        if a_cur[i] < 1e-14:
            continue
        # pick receiver j ≠ i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a small amount of mass
        delta = eps_move * np.random.rand() * a_cur[i]
        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:
            continue

        # evaluate exact objective
        conv_new = _fft_self_convolve(a_new)
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_seq_score - 1e-12:
            best_seq_score = new_score
            best_seq = a_new.copy()
            a_cur = a_new
            conv_cur = conv_new
            print(f"[{time.time():.2f}] n={n} hill‑climb iter={attempts} score={best_seq_score:.6f}")
        else:
            # occasional random walk to escape tiny plateaus
            if np.random.rand() < 0.001:
                a_cur = a_new
                conv_cur = conv_new

    return best_seq, best_seq_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence (a plain Python list) discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for printing

    # ------------------------------------------------------------------
    # Choose a set of candidate lengths.
    try:
        base_n = len(height_sequence_1)   # may be defined externally
    except Exception:
        base_n = 8000

    # lengths are between 500 and 20000, centre around base_n
    length_factors = (0.7, 0.85, 1.0, 1.15, 1.3, 1.6)
    lengths = sorted(set(
        max(500, int(base_n * f)) for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    ))
    # ensure we also probe a relatively large length
    if 15000 not in lengths and 15000 <= 20000:
        lengths.append(15000)
    lengths = sorted(lengths)

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly 30 % of the remaining budget to this length
        slice_deadline = time.time() + max(0.1, remaining * 0.30)

        # initialise with the discrete arcsine law (good starting point)
        a0 = _arcsine_distribution(n)
        a0 = _project_onto_simplex(a0, s=1.0)

        # run the optimiser for this length
        seq, internal_score = _optimise_length(n, a0, slice_deadline)

        # final evaluation with the provided black‑box (may differ slightly
        # due to clipping; fallback to internal score if it fails)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = internal_score

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq.copy()
            print(
                f"*** Global best → len={n} score={overall_best_score:.6f}"
            )

        # stop early if we already beat the target
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case we found nothing
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print(f"Fallback score: {overall_best_score:.6f}")

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    # Convert to plain Python list as required by the environment
    return overall_best_seq.tolist()
