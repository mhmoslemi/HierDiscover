    a = np.copy(a0)
    best_a = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr0 = target_sum * 0.5 / n   # base LR perhaps
    # initialize T
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute weights
        shifted = conv - conv.max()
        exp_arg = shifted / max(T, 1e-12)
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad = _weighted_gradient(a, w, size_fft)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = lr0 / (1 + 1e-6 * it)  # simple decay maybe

def _conv_fft(u, v, size_fft):
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[:len(u)+len(v)-1]

def _self_convolution(a, size_fft):
    return _conv_fft(a, a, size_fft)

def _weighted_gradient(a, w, size_fft):

        blocks = np.split(idx, cuts)

        # break if time too near end
        if time.time() + 0.02 > deadline:

    print(f"[final] best score = {best_score:.6f}")

"""
search for a non‑negative step‑function (list of heights) that minimises
the self‑convolution peak as measured by ``evaluate_sequence``.

Algorithm
---------

1.  Initialise the candidate.
    * If the global variable ``height_sequence_1`` (the best sequence found
      so far) exists, use it.
    * Otherwise fall back to a simple decreasing power‑law and project it
      onto the simplex {a≥0, Σa = √(2 n)}.

2.  Long optimisation phase – projected Adam descent on a smooth
    approximation of the max‑convolution.
    * The objective is  ϕ(a)= Σ_k w_k·(a∗a)_k  with
      w_k ∝ exp(( (a∗a)_k – max(a∗a) ) / T)   (soft‑max on the
      convolution).  As ``T→0`` the weighted sum approaches the true max.
    * Gradients are obtained analytically by one additional FFT
      (convolution of the weight vector with the reversed signal).
    * After each Adam step the iterate is projected onto the simplex
      (preserving non‑negativity and the normalisation Σa=√(2 n)).
    * Temperature ``T`` is slowly cooled and a tiny Gaussian kick is added
      every few thousand iterations to escape plateaus.

3.  Final quick sweep – random block patterns are generated and kept if
    they improve the exact score.

The whole routine respects the 1000 s budget; the best sequence found
before the deadline is returned as a plain ``list[float]``.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v (full length) using FFT."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a∗a)_k   w.r.t. a .
    The exact formula is   2·(w * a_rev)[n‑1 : n‑1+n] .
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact evaluation without the security wrapper."""
    conv = _self_convolution(a, size_fft)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _refine_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Long Adam‑type optimisation of the soft‑max approximation.
    Returns the best vector encountered and its exact score.
    """
    a = _project_to_simplex(a0.astype(float), target_sum)

    # Adam hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # base learning rate (similar magnitude to the previous Nesterov step)
    lr0 = target_sum * 0.5 / n

    # initialise temperature from the current convolution spread
    conv = _self_convolution(a, size_fft)
    T = max(1e-6, (conv.max() - conv.min()) * 0.5)

    best_a = a.copy()
    best_score = _quick_score(a, n, size_fft)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ----- current convolution and score -----
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] it {it:,} new best = {best_score:.6f}")

        # ----- soft‑max weights (smooth max) -----
        shifted = conv - conv.max()                       # ≤ 0
        exp_arg = shifted / max(T, 1e-12)
        exp_arg = np.clip(exp_arg, -50, 0)                # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # gentle learning‑rate decay
        lr = lr0 / (1.0 + 1e-5 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        a = a_new

        # ----- temperature schedule -----
        T *= 0.99995                     # slow cooling

        # ----- occasional random kick -----
        if it % 5000 == 0:
            noise = np.random.randn(n) * target_sum * 5e-4
            a = _project_to_simplex(a + noise, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point for the optimisation.  Returns a list of floats
    representing the step‑function heights.
    """
    np.random.seed(seed)
    # keep a small safety margin for the final print statements
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size and initial candidate
    # --------------------------------------------------------------
    global_seq = globals().get("height_sequence_1")
    if global_seq is not None:
        n = len(global_seq)
        a0 = np.asarray(global_seq, dtype=float)
        print("[init] using provided height_sequence_1")
    else:
        # fallback – a simple decreasing power‑law
        n = 3000
        idx = np.arange(n)
        a0 = 1.0 / np.sqrt(idx + 1.0)
        print("[init] no reference sequence – using synthetic start")

    target_sum = math.sqrt(2.0 * n)                     # normalisation √(2 n)
    size_fft = 1 << ((3 * n - 2).bit_length())          # ≥ 3 n‑2, power of two

    a0 = _project_to_simplex(a0, target_sum)
    init_score = _quick_score(a0, n, size_fft)
    print(f"[init] initial score = {init_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long Adam optimisation
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate almost all remaining time to the Adam run
    refined_a, refined_score = _refine_adam(
        a0, deadline, n, size_fft, target_sum, max_iters=int(1e7)
    )
    best_a = refined_a
    best_score = refined_score
    print(f"[phase2] best after Adam = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final fast random‑block sweep
    # --------------------------------------------------------------
    idx = np.arange(n)
    while time.time() < deadline:
        nb = np.random.randint(2, 7)                                 # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1                               # positive heights
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _quick_score(cand, n, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[random block] improved to {best_score:.6f}")

        # avoid overshooting the deadline
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
