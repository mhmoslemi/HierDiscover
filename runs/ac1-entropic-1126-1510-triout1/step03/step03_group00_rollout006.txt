"""
Propose a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.  The algorithm works in three stages:

1.  **Initialisation** – generate a probability vector (sum = 1) using
    several heuristics (uniform, arcsine, Dirichlet, previous best,
    random).

2.  **Adam optimisation on the simplex** – minimise the maximal
    self‑convolution `max(a*a)` by projected Adam updates.
    The gradient is obtained from a soft‑max smoothing of the
    convolution values; the temperature is increased gradually so that
    the optimisation starts smooth and ends with an (almost) true
    sub‑gradient of the max‑operator.

3.  **Local greedy refinement** – after the Adam run a cheap
    hill‑climbing phase repeatedly moves a tiny amount of mass from a
    large entry to a small entry and keeps the move only if the
    objective improves.

The whole search stays inside the 1000 s budget, tries several
sequence lengths (≈ 5 k–15 k) and keeps the best sequence found.
All updates respect the simplex constraint via Euclidean projection.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2*len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max‑convolution.
    When alpha is very large the sub‑gradient of the true max is returned.
    """
    n = a.shape[0]

    # Sub‑gradient (one‑hot) for very large alpha
    if alpha > 2000.0:
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g

    # Soft‑max weights
    shift = b - b.max()                 # for numerical stability
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # Convolution w * a_rev  (a_rev = a reversed)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Return a probability vector of length n according to ``kind``."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)

    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()

    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()

    elif kind == "dirichlet":
        # concentration between 0.1 and 1.0 gives a varied sparse shape
        conc = np.random.uniform(0.1, 1.0)
        a = np.random.dirichlet(np.full(n, conc))
        a /= a.sum()

    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # noqa: F821
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = _initial_sequence(n, "arcsine")
    else:
        a = _initial_sequence(n, "arcsine")

    # Ensure we are exactly on the simplex
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the max‑self‑convolution objective.
    Returns the best sequence found (sum = 1) and its internal score
    ``2·n·max(a*a)``.
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha = 1.0          # soft‑max temperature
    alpha_mul = 1.02
    alpha_max = 20000.0
    size_fft = 1 << ((3 * n - 2).bit_length())   # sufficient for w + a_rev

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution & score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient (soft‑max or sub‑gradient)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size – simple schedule (decrease like 1/sqrt(it))
        eta = 0.8 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter to leave flat plateaus
        if it % 2500 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 1e-4
            a = a + jitter
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refine(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: repeatedly move a tiny amount of mass from a
    large entry to a small entry and keep the move if the objective
    improves.  Stops when the time budget expires.
    """
    best_a = a.copy()
    b = _fft_self_convolve(best_a)
    best_score = 2.0 * n * b.max()

    # Pre‑compute a list of indices sorted by magnitude (updated lazily)
    while time.time() < deadline:
        # pick a source index among the largest 10 % of entries
        top_k = max(1, int(0.10 * n))
        src = np.argpartition(-best_a, top_k - 1)[np.random.randint(top_k)]

        # pick a destination index among the smallest 10 % of entries
        bot_k = max(1, int(0.10 * n))
        dst = np.argpartition(best_a, bot_k - 1)[np.random.randint(bot_k)]

        if src == dst or best_a[src] < 1e-12:
            continue

        # transfer a random fraction (up to 40 %) of the source mass
        eps = best_a[src] * np.random.rand() * 0.4
        a_new = best_a.copy()
        a_new[src] -= eps
        a_new[dst] += eps

        # short‑circuit: if any entry becomes negative (should not happen) skip
        if a_new[src] < 0.0:
            continue

        # evaluate
        b_new = _fft_self_convolve(a_new)
        new_score = 2.0 * n * b_new.max()
        if new_score + 1e-12 < best_score:
            best_a, best_score = a_new, new_score

    return best_a, best_score
