"""
Propose a non‑negative step function with a low evaluate_sequence score.

Algorithm
---------
1. **Setup** – Determine the problem size `n` (from the supplied reference,
   otherwise 3130) and the target L¹‑norm `S = sqrt(2*n)`.  Initialise a
   projection onto the simplex {x≥0, Σx=S}.

2. **Exploratory families** – Sample a large collection of parametric shapes
   (uniform, inverse‑sqrt, power‑law, exponential, cosine, beta‑pdf,
   mixtures and random block structures).  Each candidate is projected onto
   the simplex and its maximal self‑convolution is evaluated (via FFT).  The
   best candidate becomes the starting point for refinement.

3. **Smoothed‑max Adam optimisation** – minimise the *soft‑max* of the
   self‑convolution:
        f_T(a) = T * log Σ_k exp( (a*a)_k / T )
   using Adam with Euclidean projection onto the simplex after every step.
   The temperature `T` starts relatively large (≈ current max) and is
   exponentially cooled, which yields a smoothly varying gradient that
   quickly drives the maximum down.  Gradient computation is performed with
   FFTs (O(n log n)).

   cheap random “mass‑transfer” perturbations that keep the sum fixed.
   Accept a move only if it reduces the true maximal convolution value.

5. **Final block search** – Evaluate a handful of fresh random block‑wise

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    out_len = u.shape[0] + v.shape[0] - 1
    return conv[:out_len]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _score(a: np.ndarray, size_fft: int) -> float:
    """Return the maximal entry of the self‑convolution (the raw score)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())

def _gradient_softmax(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of the soft‑max objective f_T(a) = T·log Σ exp(conv/T)
    where w_k = exp(conv_k/T) / Σ exp(conv/T).
    The result is 2 * (w ⋆ a_rev)[n‑1 : n‑1+n].

    a_rev = a[::-1]
    conv_rev = _conv_fft(w, a_rev, size_fft)          # length 3n‑2
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad
