"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Load a historic sequence (if the global `height_sequence_1` exists) and
   treat it as an initial solution.
2. For a handful of lengths `n` (≈5 k … 16 k) generate several diverse
   probability vectors:
   * discrete arcsine law,
   * a few symmetric Beta(α,α) densities,
   * uniform,
   * Dirichlet draws with low concentration,
   * a “few‑spike’’ vector.
3. Optimise each candidate with **exponentiated‑gradient mirror descent**
   on a soft‑max surrogate of the peak of the self‑convolution.
   * The gradient of  Σ w_k·(a∗a)_k equals 2·corr(w,a) and is obtained by
     three FFTs.
   * A momentum term (β≈0.9) is accumulated to smooth the noisy updates.
   * The temperature λ grows linearly up to a large value (default 200)
     so that the surrogate converges to the true `max`.
   * After each step the vector is re‑projected onto the simplex
     (renormalised to sum 1).
4. Refine the best vector for each length with a **projected sub‑gradient
   descent** on the exact max‑convolution.
5. Keep the globally best sequence, then run a **longer momentum descent**
   from that sequence using any time that remains.
6. Finish with a cheap **stochastic hill‑climb** that moves a tiny amount
   of mass between two coordinates whenever it improves the true score.

All convolutions use FFTs (O(n log n)) and the routine checks the wall‑clock
after every iteration, guaranteeing that the total running time stays
below the 1000 s budget.  The final result is returned as a plain Python
list, ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k‑i} (correlation) using FFT.
    w – weight vector (length 2·n‑1), a – current sequence (length n),
    m – FFT length (≥ len(w)+len(a)‑1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid (i+½)/n, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Create a small, diverse pool of initial probability vectors for size n."""
    candidates = []

    # arcsine – often a strong baseline
    candidates.append(_arcsine_distribution(n))

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a handful of symmetric beta laws
    for alpha in (0.2, 0.35, 0.5, 0.75, 1.0, 1.5, 2.0):
        candidates.append(_beta_distribution(n, alpha))

    # low‑concentration Dirichlet (many tiny entries)
    candidates.append(np.random.dirichlet(np.full(n, 0.25)))

    # a sparse “spike’’ vector – a few random positions carry most of the mass
    k = max(2, n // 600)                     # ~0.15 % of coordinates
    idx = np.random.choice(n, size=k, replace=False)
    spike = np.zeros(n, dtype=np.float64)
    spike[idx] = np.random.rand(k) + 0.1     # keep them strictly positive
    candidates.append(spike / spike.sum())

    return candidates


# ----------------------------------------------------------------------
def _optimise_exponential_momentum(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 12000,
    eta0: float = 0.12,
    lambda_max: float = 200.0,
    beta_mom: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a soft‑max surrogate of the peak self‑convolution.
    Uses an exponential‑gradient update with momentum.
    Returns the best vector (∑=1) seen and its exact score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT length for correlation
    m = 1 << ((3 * n - 2).bit_length())

    # linear schedule for λ
    lambda0 = 0.3
    lambda_inc = (lambda_max - lambda0) / max_iters
    lam = lambda0

    # momentum accumulator
    mom = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and current score
        conv = _conv_full_fft(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # soft‑max weights (stable)
        maxc = conv.max()
        exp_arg = lam * (conv - maxc)            # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)

        # momentum update
        mom = beta_mom * mom + (1.0 - beta_mom) * g

        # step size (decays like 1/√t)
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * mom)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        # occasional jitter to avoid stagnation
        if it % 1000 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        lam = min(lambda_max, lam + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2500,
    eta0: float = 0.22,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Returns the best vector seen and its true score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a)
        k_star = int(conv.argmax())          # an argmax index (0‑based)

        # sub‑gradient: g_i = 2·a_{k*‑i} for all i where the index is valid
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_vals = np.arange(low, high + 1, dtype=int)
        g[i_vals] = a[k_star - i_vals]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        if it % 500 == 0 and np.random.rand() < 0.18:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 18000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local improvement: move a tiny amount of mass from
    one coordinate to another whenever it reduces the exact score.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 5 % of the donor (capped at 0.01 absolute)
        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score
