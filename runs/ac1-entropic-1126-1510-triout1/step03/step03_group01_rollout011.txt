"""
Propose a candidate step‑function (non‑negative heights) that minimizes the
score returned by ``evaluate_sequence``.

The algorithm works in three phases (all within the supplied time budget):

1. **Parameterized sweep** – for several lengths we generate simple
   families of distributions a_i ∝ ((i+½)(n‑i‑½))**p** with a list of
   exponents *p* (including the classic arcsine p = –0.5).  Each candidate
   is normalised to sum = 1 and its score is evaluated using a fast FFT
   based convolution.  The best candidate from this phase is kept.

2. **Projected soft‑max gradient descent** – starting from the best
   candidate we run the smoothed‑max gradient descent that was used in the
   reference implementation, but with a few tuned hyper‑parameters and a
   tighter time allocation.  The iterate is always projected onto the
   probability simplex.

3. **Simulated‑annealing refinement** – a short SA loop performs tiny
   mass transfers between two random positions and accepts moves with a
   Metropolis criterion.  This helps to escape shallow local minima and
   often reduces the peak of the self‑convolution a little further.

All three phases keep track of the globally best sequence (lowest
score).  The final sequence is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a via real FFT (full length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (real output)."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _score_from_vector(a: np.ndarray) -> float:
    """Fast surrogate of evaluate_sequence (sum(a) == 1)."""
    n = a.shape[0]
    max_b = _fft_self_convolve(a).max()
    return 2.0 * n * max_b   # because sum(a)=1 → denominator = 1


# ----------------------------------------------------------------------
def _distribution_exponent(n: int, p: float) -> np.ndarray:
    """Generate a_i ∝ ((i+0.5)*(n‑i‑0.5))**p, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** p
    total = vals.sum()
    if total == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


# ----------------------------------------------------------------------
def _gradient_descent(start_a: np.ndarray,
                      time_limit: float,
                      n: int) -> tuple[np.ndarray, float]:
    """
    Smoothed‑max gradient descent (same spirit as the reference code).
    Returns the best vector encountered and its score.
    """
    a = start_a.copy()
    best_seq = a.copy()
    best_score = _score_from_vector(a)

    velocity = np.zeros_like(a)
    beta = 0.9                # momentum
    eta0 = 0.4                # base step size
    alpha = 5.0               # temperature for soft‑max
    alpha_factor = 1.001
    alpha_max = 2000.0

    it = 0
    # pre‑compute FFT size for this n (used inside the loop)
    size_self = 1 << ((2 * n - 1) - 1).bit_length()
    size_conv = 1 << ((3 * n - 2) - 1).bit_length()

    deadline = time.time() + time_limit
    while time.time() < deadline:
        it += 1

        # 1) self‑convolution and score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] GD iter={it} n={n} score≈{cur_score:.6f}")

        # 2) soft‑max weights
        exp_term = np.exp(alpha * (b - cur_max))
        w_sum = exp_term.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = exp_term / w_sum

        # 3) gradient of surrogate objective
        a_rev = a[::-1]
        conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # 4) momentum update
        velocity = beta * velocity + (1.0 - beta) * grad

        # 5) diminishing step size
        eta = eta0 / math.sqrt(it)

        # 6) gradient step + projection onto simplex
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # 7) tiny jitter every few thousand steps
        if it % 2000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

        # 8) increase temperature slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         time_limit: float,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly transfers a tiny amount of mass between two
    positions and accepts the move according to a decreasing temperature.
    Returns the best sequence seen and its score.
    """
    a = start_a.copy()
    cur_score = _score_from_vector(a)
    cur_max = cur_score / (2.0 * n)  # because score = 2 n max_b
    best_seq = a.copy()
    best_score = cur_score

    # initial temperature: a fraction of the current excess over the average
    avg = 1.0 / (2 * n - 1)           # average b value when sum(a)=1
    excess = cur_max - avg
    T0 = max(excess, 1e-6) * 0.8      # start a bit below the excess
    start_time = time.time()
    deadline = start_time + time_limit

    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling schedule
        elapsed = time.time() - start_time
        T = T0 * max(0.0, 1.0 - elapsed / time_limit)

        # propose a transfer
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1  # ensure j != i
        if a[i] == 0.0:
            continue
        delta = a[i] * np.random.rand() * 0.05   # move up to 5 % of a[i]

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        new_max_b = _fft_self_convolve(a_new).max()
        new_score = 2.0 * n * new_max_b

        # acceptance test
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            cur_max = new_max_b

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} n={n} score≈{cur_score:.6f}")

    return best_seq, best_score
