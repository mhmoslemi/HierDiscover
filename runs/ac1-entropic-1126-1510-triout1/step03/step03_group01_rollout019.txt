"""
Propose a low‑score step function.

The algorithm explores several problem sizes (down to 500 entries) and a
hand‑crafted family of “inverse‑power” shapes

    a_i ∝ ((i+γ)·(n‑i‑γ))^{-p}            (γ>0, p>0)

which contain the classical arcsine distribution as the special case p=½.
For each size a quick score estimate (2·n·max(conv(a,a))) is computed.
The best few candidates are then refined with an Adam optimiser on the
smooth‑max surrogate used in the reference code.  All updates are projected
onto the probability simplex so that the sum constraint stays satisfied.
The whole run respects the 1000 s time budget and returns the sequence
with the smallest evaluation from `evaluate_sequence`.
"""
import time
import numpy as np

# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# --------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# --------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (p=½, γ=½) – already normalised to sum‑1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# --------------------------------------------------------------
def _power_distribution(n: int, p: float, gamma: float = 0.5) -> np.ndarray:
    """a_i ∝ ((i+γ)*(n‑i‑γ))^{-p}, normalised to sum‑1."""
    idx = np.arange(n, dtype=np.float64)
    # avoid zero denominator at the ends
    den = (idx + gamma) * (n - idx - gamma)
    vals = den ** (-p)
    vals /= vals.sum()
    return vals


# --------------------------------------------------------------
def _raw_score(seq: np.ndarray) -> float:
    """2·n·max(self‑convolution) – the denominator (sum) is 1."""
    n = len(seq)
    b = np.convolve(seq, seq)                # plain convolution is fast enough for modest n
    return 2.0 * n * b.max()


# --------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   max_iter: int,
                   time_deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed‑max objective.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature (smooth‑max surrogate)
    alpha = 5.0
    alpha_factor = 1.005
    alpha_max = 2000.0

    best_seq = a.copy()
    best_score = _raw_score(a)

    for it in range(1, max_iter + 1):
        if time.time() >= time_deadline:
            break

        # 1) current convolution and peak
        b = _fft_self_convolve(a)
        cur_max = b.max()

        # keep exact best
        if cur_max * 2 * n < best_score:
            best_score = cur_max * 2 * n
            best_seq = a.copy()

        # 2) soft‑max weights over the convolution entries
        exp_term = np.exp(alpha * (b - cur_max))
        w = exp_term / exp_term.sum()                     # shape: 2·n‑1

        # 3) gradient of the smooth‑max surrogate
        a_rev = a[::-1]
        conv_rev = _fft_convolve(a_rev, w)               # length = 3·n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]          # slice back to length n

        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # diminishing learning‑rate (≈1/√it)
        eta = 0.5 / np.sqrt(it)

        a -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # 5) simplex projection (keeps sum=1 and non‑negative)
        a = _project_onto_simplex(a, s=1.0)

        # 6) occasional tiny jitter to escape plateaus
        if it % 1000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

        # 7) anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    final_raw = _raw_score(best_seq)
    return best_seq, final_raw


# --------------------------------------------------------------
def propose_candidate(budget_s: int = 1000, seed: int = 42) -> list[float]:
    """
    Search for a step‑function with the smallest possible evaluation
    (according to the supplied `evaluate_sequence`).  The routine respects
    the time budget, tries several problem sizes down to 500, explores a
    small parametric family of shapes, and refines the most promising
    candidates with an Adam optimiser.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # reserve a couple of seconds

    # ----------------------------------------------------------------------
    # Problem sizes we will explore – we deliberately go down to the minimum
    # allowed size (500) because the score 2·n·max(b) becomes easier to beat.
    candidate_lengths = [500, 800, 1200, 1600, 2000, 4000, 8000]

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    for n in candidate_lengths:
        # allocate a slice of the remaining budget (≈30 % of what is left)
        slice_deadline = time.time() + max(0.1,
                                           (deadline - time.time()) * 0.3)

        # --------------------------------------------------------------
        # build a small pool of initial vectors for this length
        init_pool: list[np.ndarray] = []

        # uniform – baseline
        init_pool.append(np.full(n, 1.0 / n, dtype=np.float64))

        # arcsine (p=0.5)
        init_pool.append(_arcsine_distribution(n))

        # a few power‑law shapes (p from 0.3 to 1.0)
        for p in np.linspace(0.3, 1.0, 8):
            init_pool.append(_power_distribution(n, p=p, gamma=0.5))

        # a few random starts
        for _ in range(3):
            r = np.random.rand(n)
            r /= r.sum()
            init_pool.append(r)

        # keep only the most promising 3 initialisers according to the cheap score
        scored_inits = [(seq, _raw_score(seq)) for seq in init_pool]
        scored_inits.sort(key=lambda x: x[1])
        top_inits = [seq for seq, _ in scored_inits[:3]]

        # --------------------------------------------------------------
        # local optimisation of each promising start
        for a0 in top_inits:
            # quick sanity – ensure we start on the simplex
            a0 = _project_onto_simplex(a0, s=1.0)

            # run Adam optimiser; give it a modest iteration budget
            max_iters = 20000
            best_seq, best_raw = _adam_optimize(a0, max_iters, slice_deadline)

            # final evaluation through the black‑box routine
            try:
                final_score = evaluate_sequence(best_seq.tolist())
            except Exception:
                final_score = best_raw  # fallback to our own estimate

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_seq.copy()
                print(f"*** New global best: n={n} score={overall_best_score:.6f}")

            # stop early if we already beat the target
            if overall_best_score <= 1.503:
                break
        if time.time() >= deadline or overall_best_score <= 1.503:
            break

    # ----------------------------------------------------------------------
    # safety net – should never happen, but we return a uniform vector if needed
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
