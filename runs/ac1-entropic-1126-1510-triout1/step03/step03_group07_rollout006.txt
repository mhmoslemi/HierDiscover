"""
Propose a candidate step‑function by stochastic projected gradient descent
with a soft‑max approximation of the max‑convolution objective, combined
with a simulated‑annealing polishing phase.

The evaluation metric is scale‑invariant; we therefore enforce the ℓ₁‑norm
∑aᵢ = √(2·n) (denoted S) so that the objective reduces to minimising
max(conv(a,a)).  The algorithm works as follows:

1. Choose a problem size N (≈2·|height_sequence_1|, capped at 30 000).
2. Initialise a on the simplex (Dirichlet/Exponential draw).
3. Run several restarts of projected gradient descent.
   * Objective approximation: w ∝ exp(β·(conv–max(conv))) (soft‑max).
   * Gradient of the soft‑max objective: ∇ = 2·corr(w,a).
   * Update with momentum, then Euclidean projection onto the
     non‑negative simplex with prescribed sum S.
   * β is increased from 1 to 200, step‑size is annealed from 2 to 0.01.
4. Keep the best sequence observed over all restarts.
5. Polish the best candidate with a simulated‑annealing random‑pair
   mass‑swap procedure (still respecting the simplex constraint).

The whole procedure respects the 1000 s time budget, prints occasional
progress messages and finally returns the best sequence as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the non‑negative simplex with given sum.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    # Duchi et al. (2008) algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution (a*a) – length 2n‑1.
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based valid‑mode correlation (len = n) needed for the gradient.
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]               # length of a
    L = w.shape[0]               # = 2n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1                # position of zero lag
    return full[start : start + n]   # length n (valid)


# ----------------------------------------------------------------------
# Small random mass‑swap that preserves the ℓ₁‑norm exactly.
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two distinct indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # draw delta uniformly from (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    # keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  Returns the best sequence (list of floats) discovered before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for the harness (a couple of seconds)
    deadline = start_time + max(0.0, budget_s - 2.0)

    # ------------------------------------------------------------------
    # Problem size.
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n0 = base.size
    except Exception:
        n0 = 15000

    # Choose a length that is large enough to grant many degrees of freedom
    # but still fits comfortably into memory and FFT work‑size limits.
    N = min(30000, max(12000, n0 * 2))
    S = math.sqrt(2.0 * N)           # ℓ₁‑norm we enforce (score == max(conv))

    print(f"[Setup] N = {N}, target ℓ₁‑norm S = {S:.6f}")

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the (soft‑max) gradient descent.
    # ------------------------------------------------------------------
    beta_start = 1.0
    beta_end = 200.0
    lr_start = 2.0
    lr_end = 0.01
    momentum = 0.9

    # ------------------------------------------------------------------
    # Main optimisation loop: several stochastic restarts.
    # ------------------------------------------------------------------
    restart_id = 0
    while time.time() < deadline - 20.0:          # reserve ~20 s for polishing
        restart_id += 1

        # Initialise a random point on the simplex (Dirichlet via exponentials).
        a = np.random.exponential(1.0, size=N)
        a = a / a.sum() * S
        v = np.zeros_like(a)                      # momentum buffer

        remaining = deadline - time.time()
        max_iters_restart = min(8000, int(remaining * 30))   # very generous estimate
        if max_iters_restart < 500:

        # --------------------------------------------------------------

        # --------------------------------------------------------------
        for it in range(1, max_iters_restart + 1):
            # Time‑check – abort early if we are about to exceed the budget.
            if time.time() >= deadline - 20.0:

            conv = _fft_convolve(a)
            cur_max = conv.max()
            # Record best solution seen so far.
            if cur_max < best_score:

            # stable soft‑max
            exp_vals = np.exp(beta * (conv - cur_max))
            w = exp_vals / exp_vals.sum()

            # Gradient of the smooth approximation.
            grad = 2.0 * _fft_correlate(w, a)

            # Momentum step and projection onto the simplex.
            v = momentum * v - lr * grad
            a = a + v
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, S)

        # Small random perturbation before the next restart (helps escaping)
        if time.time() < deadline - 20.0:

    # ------------------------------------------------------------------
    # If nothing was found (should never happen), fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(N, S / N, dtype=float)

    best_seq = np.maximum(best_seq, 0.0)
    best_score = _fft_convolve(best_seq).max()
    print(f"[Before polishing] best max(conv) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Polishing phase – simulated‑annealing random pair swaps.
    # ------------------------------------------------------------------
    polish_iters = 20000
    sigma_start = 0.5
    sigma_end = 0.02
    temp_start = 0.1
    temp_end = 1e-5

    for it in range(1, polish_iters + 1):
        if time.time() >= deadline:
            break
        frac = it / polish_iters
        sigma = sigma_start * (1.0 - frac) + sigma_end * frac
        temp = temp_start * (1.0 - frac) + temp_end * frac

        cand = _random_pair_move(best_seq, sigma)
        cand_score = _fft_convolve(cand).max()
        delta = cand_score - best_score
        if delta < 0 or np.exp(-delta / max(temp, 1e-12)) > np.random.rand():
            best_seq = cand
            best_score = cand_score
            print(f"[Polish {it:05d}] New best max(conv) = {best_score:.6f}")

    # Final sanity clamp.
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
