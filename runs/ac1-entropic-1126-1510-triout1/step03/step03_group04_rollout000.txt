def propose_candidate(seed=42, budget_s=1000):

        fa = np.fft.rfft(a, n=nfft)
        conv = np.fft.irfft(fa * fa, n=nfft)

        return float(conv[:2*n-1].max())

            cand = _project_to_simplex(cand, target_sum)

    best_idx = int(scores.argmin())

    CR = 0.9

"""
Propose a low‑score step function.

Algorithm
---------
1. **Problem size** – Use the length of the previously known best
   construction (`height_sequence_1`) if it exists, otherwise fall back
   to `n = 3000`.  The ℓ¹‑norm is fixed to √(2 n) so that the evaluation
   score equals the maximum of the self‑convolution.

2. **Differential Evolution** – A population of 30 simplex‑feasible
   candidates is evolved.  Each offspring is created by the classic
   “DE/rand/1/bin” mutation and crossover, then projected back onto the
   simplex.  The best individual is tracked.  Stagnation is prevented by
   occasional random‑restart of the worst half of the population.

3. **Projected sub‑gradient descent** – Starting from the best DE
   individual, a simple sub‑gradient method (using the `top_k` largest
   convolution entries) refines the solution.  Momentum and a decaying
   step size are employed, together with Euclidean projection onto the
   simplex.

4. **Final random‐swap hill‑climb** – With the remaining time a cheap
   pairwise mass‑transfer move is tried; after each move the vector is
   projected onto the simplex and accepted only if it improves the
   maximum convolution.

All heavy work (convolution, projection, gradient) is done with NumPy
and FFTs; the total running time respects the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = z}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero except the first
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution(vec: np.ndarray) -> np.ndarray:
    """
    Compute the full (linear) convolution of `vec` with itself using FFT.
    Returns a real array of length 2*len(vec)-1.
    """
    n = vec.shape[0]
    # size for FFT: next power of two ≥ 2n‑1
    nfft = 1 << (2 * n - 1).bit_length()
    f = np.fft.rfft(vec, n=nfft)
    conv = np.fft.irfft(f * f, n=nfft)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution(vec: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of `vec`."""
    return float(_full_convolution(vec).max())


# ----------------------------------------------------------------------
def _gradient_topk(a: np.ndarray, conv: np.ndarray, top_k: int = 20) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution using the `top_k` largest
    convolution entries.  For each active index k we add the contribution
    2·a_{k-i} to the gradient w.r.t. a_i.
    """
    n = a.shape[0]
    # indices of the k largest values (could contain ties)
    idxs = np.argpartition(-conv, top_k - 1)[:top_k]
    grad = np.zeros_like(a)
    i_arr = np.arange(n)
    for k in idxs:
        j = k - i_arr
        mask = (j >= 0) & (j < n)
        grad[mask] += 2.0 * a[j[mask]]
        # diagonal entries (i=j) are already counted correctly by the
        # expression above (they contribute 2·a_i), which matches the
        # true derivative of a_i².
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # --------------------------------------------------------------
    # Problem size and target sum.
    try:
        # Use the previous best construction if available.
        n = len(height_sequence_1)          # type: ignore[name-defined]
        init_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        init_seq = None
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # enforced ℓ¹‑norm

    # --------------------------------------------------------------
    # Helper to evaluate the objective (max of self‑convolution).
    def _evaluate(a: np.ndarray) -> float:
        """Return the max convolution (equals evaluate_sequence score)."""
        return _max_convolution(a)

    # --------------------------------------------------------------
    # Initialise a Differential Evolution population.
    popsize = 30
    pop = np.empty((popsize, n), dtype=float)

    for i in range(popsize):
        if have_ref and i == 0:
            cand = init_seq.copy()
            # Rescale to exact target ℓ¹‑norm and project.
            cand = cand * (target_sum / max(cand.sum(), 1e-12))
            cand = _project_to_simplex(cand, target_sum)
        else:
            cand = np.random.rand(n)
            cand = _project_to_simplex(cand, target_sum)
        pop[i] = cand

    scores = np.empty(popsize, dtype=float)
    for i in range(popsize):
        scores[i] = _evaluate(pop[i])

    best_idx = int(scores.argmin())
    best_a = pop[best_idx].copy()
    best_score = float(scores[best_idx])
    print(f"[init] population best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Differential Evolution (DE/rand/1/bin).
    F = 0.6          # mutation factor
    CR = 0.9         # crossover probability
    generation = 0

    # Leave ~30 % of the budget for the gradient phase.
    while time.time() < deadline - 200:
        for i in range(popsize):
            # Choose three distinct individuals different from i.
            idxs = list(range(popsize))
            idxs.remove(i)
            a_idx, b_idx, c_idx = np.random.choice(idxs, 3, replace=False)
            a_vec = pop[a_idx]
            b_vec = pop[b_idx]
            c_vec = pop[c_idx]

            # Mutation
            mutant = a_vec + F * (b_vec - c_vec)

            # Crossover (binomial)
            trial = pop[i].copy()
            j_rand = np.random.randint(n)
            cross_mask = np.random.rand(n) < CR
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]

            # Projection onto simplex (ensures feasibility)
            trial = _project_to_simplex(trial, target_sum)

            # Selection
            trial_score = _evaluate(trial)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
                    print(f"[DE gen {generation}] new best = {best_score:.6f}")

        generation += 1

        # Simple restart heuristic every 50 generations.
        if generation % 50 == 0:
            worst_idxs = np.argsort(scores)[popsize // 2 :]
            for wi in worst_idxs:
                newcand = np.random.rand(n)
                newcand = _project_to_simplex(newcand, target_sum)
                pop[wi] = newcand
                scores[wi] = _evaluate(newcand)

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (fine‑tuning).
    a = best_a.copy()
    momentum = np.zeros_like(a)
    it = 0
    # Hyper‑parameters for the descent.
    eta0 = target_sum * 0.5 / n          # initial step magnitude
    decay_eta = 1e-5
    beta = 0.9                           # momentum coefficient
    top_k = 30                           # size of active set

    while time.time() < deadline - 5:
        conv = _full_convolution(a)     # full self‑convolution
        cur_max = conv.max()
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best_a = a.copy()
            print(f"[GD {it}] new best = {best_score:.6f}")

        # Sub‑gradient from the top‑k largest convolution entries.
        grad = _gradient_topk(a, conv, top_k=top_k)

        # Momentum update.
        momentum = beta * momentum + (1.0 - beta) * grad

        # Decaying step size.
        eta = eta0 / (1.0 + decay_eta * it)

        # Gradient step + projection.
        a_new = a - eta * momentum
        a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    # --------------------------------------------------------------
    # Phase 3 – final tiny hill‑climb (pairwise mass swaps).
    while time.time() < deadline:
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size == 0:
            break
        i = int(np.random.choice(nz))
        j = int(np.random.randint(0, n))
        if i == j:
            continue
        # Move a random fraction of the mass at i to position j.
        delta = best_a[i] * np.random.uniform(0.05, 0.5)
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        sc = _evaluate(cand)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = cand
            print(f"[swap] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
