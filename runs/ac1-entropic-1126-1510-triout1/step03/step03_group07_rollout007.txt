"""
Propose a high‑quality step‑function sequence.

Algorithm
---------
1.  Determine a problem size N (≈10k–20k) and the ℓ1‑norm target
    S = √(2·N).  Initialise the sequence either by resampling the
    previously supplied `height_sequence_1` or by a small random
    perturbation of the uniform simplex point.

2.  **Sub‑gradient phase** – minimise the true objective
    `max(conv)` (the evaluation is scale‑invariant when Σa = S).
    In each iteration
        * compute the full convolution `conv = a*a`,
        * build a probability vector `w` that puts equal weight on
          every index where `conv` attains its maximum,
        * the sub‑gradient of `max(conv)` is
              ∇ = 2 * corr(w, a)          (valid correlation)
        * take a diminishing projected‑sub‑gradient step
          (`step = 0.01 / sqrt(it+1)`) and project back onto the
          non‑negative simplex {x≥0, Σx = S}.

    The best sequence seen during this phase is stored.

3.  **Simulated‑annealing polishing** – with the best sequence as a
    start, perform many pairwise mass‑transfer moves while accepting
    uphill moves with probability `exp(-(Δ)/T)`.  Temperature `T`
    decays linearly to zero, and the move magnitude `σ` also cools
    down.  Whenever a new best value is found it is recorded.

4.  The routine stops as soon as the allotted time (≈1000 s) is
    exhausted and returns the best sequence as a plain Python list.

All heavy work (convolution, correlation, simplex projection) is
implemented with NumPy/FFT and runs in O(N log N) per evaluation,
allowing several hundred thousand objective evaluations within the
budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the non‑negative simplex with prescribed sum.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    # Duchi et al. (2008) – O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Cheap random pairwise mass‑swap (preserves sum exactly target_sum).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two random indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # draw delta uniformly from (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    # keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# FFT‑based convolution (full) – returns a*a of length 2n‑1.
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based correlation: corr = np.correlate(w, a, mode='valid')
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlate w (length 2n‑1) with a (length n) in “valid” mode.
    Returns an array of length n.
    """
    n = a.shape[0]
    L = w.shape[0]                     # = 2n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length n (valid correlation)


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Find a non‑negative step‑function sequence with a low value of
    `evaluate_sequence`.  The method combines a projected sub‑gradient
    descent on the exact max‑convolution objective with a simulated‑annealing
    polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 1.5)

    # --------------------------------------------------------------
    # Choose problem size.
    # --------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        base_n = base.size
    except Exception:
        base = None
        base_n = 0

    # Aim for a moderately large N (up to 20 000) while staying fast.
    if base_n > 0:
        N = min(20000, max(10000, int(base_n * 1.2)))
    else:
        N = 15000
    S = math.sqrt(2.0 * N)          # target ℓ1‑norm (scale‑invariant)

    # --------------------------------------------------------------
    # Initialise a feasible sequence (resampled previous best or uniform).
    # --------------------------------------------------------------
    if base is not None:
        # Normalise base to the new ℓ1‑norm.
        base_sum = base.sum()
        if base_sum == 0:
            base_norm = np.full_like(base, 1.0)
        else:
            base_norm = base * (S / base_sum)

        if base_n == N:

        else:

            x_old = np.linspace(0.0, 1.0, base_n)
            x_new = np.linspace(0.0, 1.0, N)

    else:

    a += 0.001 * S * np.random.rand(N) / N
    a = np.maximum(a, 0.0)
    a = _project_onto_simplex(a, S)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    conv = _fft_convolve(a)
    max_conv = conv.max()
    cur_score = float(2.0 * N * max_conv / (np.sum(a) ** 2))
    best_seq = a.copy()
    best_score = cur_score

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing polishing.
    # --------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # remaining time determines how many SA steps we can afford
    remaining = deadline - time.time()

        max_conv_c = conv_c.max()
        cand_score = float(2.0 * N * max_conv_c / (np.sum(cand) ** 2))

            if T > 0 and np.random.rand() < math.exp(-delta / T):

            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[SA {i:06d}] New best score: {best_score:.6f}")

            cur_score = float(
                2.0 * N * _fft_convolve(cur_seq).max() / (np.sum(cur_seq) ** 2)
            )

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # safety
    return best_seq.tolist()
