"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.

Algorithm
---------
* Work on the probability simplex (∑a_i = 1) – the evaluation is
  invariant under scaling, so we can safely enforce the sum.
* Several families of initial shapes are tried:
    – discrete arcsine,
    – power‑law shapes  a_i ∝ ((i+0.5)*(n‑i‑0.5)+ε)^{-q}  for q∈{0.3,…,0.7},
    – random points on the simplex.
* For a fixed length `n` we run a **projected Adam optimiser** that
  minimises a smooth approximation of `max(a*a)`.
    – The smooth max is realised by a soft‑max on the convolution values
      with a temperature `α`.  `α` starts modest and grows geometrically;
      this provides a good trade‑off between smooth gradients and a true
      sub‑gradient for the max.
    – Gradient of the soft‑max is obtained via FFT‑based convolutions,
      which makes each iteration O(n log n).
    – After each Adam step we project back onto the simplex and
      enforce symmetry (a_i = a_{n‑1‑i}), because the problem is
      invariant under reversal.  Symmetrising speeds up convergence.
    – Small random jitter is added occasionally to escape plateaus.
* The optimiser is run repeatedly for several initialisations and for a
  handful of lengths (from a few thousand up to 20 000).  The total
  running time is limited to the supplied budget (≈1000 s).  The best
  sequence found (according to the official `evaluate_sequence`) is
  returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                      # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _powerlaw_distribution(n: int, q: float, eps: float = 1e-12) -> np.ndarray:
    """Family a_i ∝ ((i+0.5)*(n‑i‑0.5)+eps)^(-q).  Normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5) + eps
    vals = denom ** (-q)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()     # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of the smooth max (soft‑max) of the self‑convolution.
    Returns ∂/∂a ( (1/α)·log Σ_k exp(α·b_k) ).
    """
    n = a.shape[0]

    # soft‑max weights (stable)
    shift = b - b.max()
    # protect against overflow
    w = np.exp(np.clip(alpha * shift, -700, 700))
    w_sum = w.sum()
    if w_sum == 0.0:                     # extreme case – fall back to hard max
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution w * a_rev  (a_rev = a reversed)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _optimize_one_length_adam(init_a: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation for a fixed length `n`.
    Returns the best sequence (as a NumPy array) discovered before `deadline`
    together with its *internal* score 2·n·max(a*a) (i.e. the value that
    `evaluate_sequence` would return when the vector is normalised to sum = 1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.12                # learning‑rate – works well for the simplex
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 10.0
    alpha_mul = 1.02
    alpha_max = 5000.0

    # FFT size large enough for convolution w (2n‑1) with a_rev (n)
    size_fft = 1 << ((3 * n - 2) - 1).bit_length()

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution & score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n:5d} it={it:6d} score={best_score:.6f}")

        # gradient of smooth max
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex and enforce symmetry
        a = _project_onto_simplex(a, s=1.0)

        # increase soft‑max temperature
