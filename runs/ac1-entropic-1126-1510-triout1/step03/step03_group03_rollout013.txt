"""
Hill‑climbing optimizer for the non‑negative step‑function problem.

The algorithm starts from the best known construction (global variable
`height_sequence_1` if present, otherwise a uniform distribution), projects it
onto the ℓ¹‑simplex with sum √(2·n) and then performs a large number of
gradient‑guided moves.  A move transfers a small amount of mass from a
high‑gradient index to a low‑gradient index while keeping the total sum
exactly constant.  The convolution is updated incrementally (O(n)) so that
many iterations fit easily into the 1000 s time budget.  An annealing schedule
allows occasional uphill moves early on, helping to escape plateaus.
The best feasible sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the hard‑max convolution objective.
    g[i] = 2 * average_{k in active} a[k-i] (zero when out of range).
    """
    n = a.shape[0]
    m = active.size
    weight = 1.0 / m
    g = np.zeros(n, dtype=float)

    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # indices i where the term a_{k-i} is defined
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k - i_idx                 # = k-i
        g[i_start:i_end + 1] += 2.0 * a[j_idx] * weight
    return g

    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    S = math.sqrt(2.0 * n)            # target ℓ¹‑norm (sqrt(2·n))

    # --------------------------------------------------------------
    # initialise a (heights) on the simplex Σa = S
    if have_ref:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = np.clip(a0, 0.0, 1000.0)
        a0 = a0 / a0.sum() * S
        a = _project_to_simplex(a0, S)
    else:
        a = np.full(n, S / n, dtype=float)

    # --------------------------------------------------------------
    # bookkeeping
    conv = np.convolve(a, a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_score = cur_max
    print(f"[init] n={n}, initial max={cur_max:.6f}")

    # --------------------------------------------------------------
    # hill‑climbing parameters
    step_factor = 0.5                 # maximal fraction of source mass to move
    min_step = 1e-12
    it = 0
    start_time = time.time()
    # annealing: initial temperature as a small fraction of current max
    T0 = 0.02 * cur_max

    while time.time() < deadline:
        # recompute active indices (might have changed)
        active = np.where(conv == cur_max)[0]
        # sub‑gradient w.r.t. a
        g = _subgradient_max_conv(a, active)

        # choose source (largest gradient) and destination (smallest gradient)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if dst == src:
            dst = (dst + 1) % n

        # pick a feasible transfer amount
        max_transfer = a[src]
        if max_transfer < min_step:
            # source is essentially empty – pick another source
            possible = np.where(a > min_step)[0]
            if possible.size == 0:
                break
            src = int(np.random.choice(possible))
            max_transfer = a[src]

        delta = max_transfer * step_factor * np.random.rand()
        if delta < min_step:
            continue

        # incremental update of the convolution (O(n))
        conv_new = conv.copy()
        # linear part: -2*delta*a shifted by src, +2*delta*a shifted by dst
        conv_new[src:src + n] -= 2.0 * delta * a
        conv_new[dst:dst + n] += 2.0 * delta * a
        # quadratic correction (exact)
        conv_new[2 * src] += delta * delta
        conv_new[2 * dst] += delta * delta
        conv_new[src + dst] -= 2.0 * delta * delta

        new_max = float(conv_new.max())

        # annealing temperature (linear cooling)
        elapsed = time.time() - start_time
        T = max(0.0, T0 * (1.0 - elapsed / budget_s))

        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        elif T > 0.0:
            # accept uphill moves with probability exp(-Δ/T)
            prob = math.exp(-(new_max - cur_max) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            # apply the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            cur_max = new_max

            if new_max < best_score - 1e-12:
                best_score = new_max
                best_a = a.copy()
                print(f"[improve] iter={it} new best max={best_score:.6f}")

            # slowly increase the step size after success
            step_factor = min(step_factor * 1.03, 0.9)
        else:
            # shrink step size after failure
            step_factor = max(step_factor * 0.97, 0.01)

        # occasional full recomputation to avoid drift
        if it % 2500 == 0 and it > 0:
            conv = np.convolve(a, a)
            cur_max = float(conv.max())

        it += 1

        # safety break if we have exhausted the budget without progress
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # final check with the official evaluator (safety)
    final_score = evaluate_sequence(best_a.tolist())
    if final_score == float("inf"):
        # fallback to the current best (should not happen)
        final_score = best_score
    print(f"[final] best max={best_score:.6f}, evaluator score={final_score:.6f}")

    return best_a.tolist()
