"""
Propose a step‑function (non‑negative sequence) that minimizes
the evaluation score defined by `evaluate_sequence`.

Algorithm summary
-----------------
1. **Problem size** – If the previous best `height_sequence_1` is
   defined we keep its length, otherwise we use a larger default
   (≈ 4000).  The target L1‑norm is set to √(2 n) which makes the
   objective equal to the peak of the self‑convolution.

2. **Exploratory families** – Generate many parametric families
   (inverse‑sqrt, power‑law, exponential, Gaussian, beta‑like,
   arcsine, sinusoidal, double‑ended, random block, smoothed random).
   Each candidate is projected onto the simplex (non‑negative,
   Σ = √(2 n)) and the exact peak of its self‑convolution is scored.
   The K≈15 best raw candidates are kept.

3. **Refinement** – For every kept candidate we run a
   projected (sub‑)gradient descent on the *max‑convolution* objective.
   The gradient is obtained exactly via FFTs:
        ∇ max_k (a*a)_k = 2·conv(w, a_rev)
   where w is a probability vector supported on the indices attaining the
   current maximum (sub‑gradient).  Early on a soft‑max temperature is used
   to provide smoothness, later we switch to the pure sub‑gradient.
   Momentum, a decaying step‑size and occasional Gaussian kicks help
   escaping plateaus.  The best sequence seen during the run is stored.

4. **Post‑refinement random walk** – Starting from the current global
   best we repeatedly apply tiny random mass transfers followed by a
   short gradient descent; any improvement updates the global best.

5. **Return** – The best sequence found before the 1000 s budget expires
   is returned as a plain ``list[float]``.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# GLOBAL placeholders
_size_fft_global = None   # will be set inside ``propose_candidate``

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v (full length) using FFT."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    The exact expression is   2·(w ↔ a_rev)   restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = len(w)+n-1
    start = len(w) - len(a)                       # = n-1  when len(w)=2n-1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, target_sum: float) -> float:
    """Exact evaluation score (no security checks)."""
    conv = _self_convolution(a, _size_fft_global)
    # because Σa = target_sum = √(2n) => denominator = 2n, so objective = max(conv)
    return float(2 * n * conv.max() / (target_sum ** 2))


# ----------------------------------------------------------------------
def _refine_candidate(a0: np.ndarray,
                      deadline: float,
                      n: int,
                      size_fft: int,
                      target_sum: float,
                      max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected (sub‑)gradient descent on the max‑convolution peak.
    Uses a soft‑max temperature at the beginning and switches to a
    pure sub‑gradient once the temperature is very low.
    """
    a = _project_to_simplex(a0.copy(), target_sum)
    best_a = a.copy()
    best_score = _objective(a, n, target_sum)

    # momentum variables
    vel = np.zeros_like(a)
    momentum = 0.95
    eta0 = target_sum * 0.4 / n            # base step size
    decay = 1e-5

    # initialise temperature from the spread of the initial convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.6)

    no_improve = 0
    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- evaluate current point ----
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (target_sum ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            # occasional progress report
            if it % 2000 == 0:
                print(f"[refine] it {it:,} new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ---- weight vector for gradient ----
        if T > 1e-4:                     # smooth phase
            shifted = conv - conv.max()
            exp_arg = shifted / max(T, 1e-9)
            exp_arg = np.clip(exp_arg, -50, 0)   # avoid under/overflow
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full_like(w, 1.0 / w.size)
        else:                           # pure sub‑gradient on the active set
            max_val = conv.max()
            eps = 1e-12 * max_val
            active = np.where(conv >= max_val - eps)[0]
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size

        # ---- gradient ----
        grad = _weighted_gradient(a, w, size_fft)

        # ---- Nesterov‑type momentum update ----
        eta = eta0 / (1.0 + decay * it)
        vel = momentum * vel - eta * grad
        a_new = a + vel

        # ---- projection onto simplex ----
        a_new = _project_to_simplex(a_new, target_sum)

        # ---- random kick to escape plateaus ----
        if (it % 5000 == 0 and it > 0) or no_improve > 4000:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            if it % 5000 == 0:
                print(f"[refine] it {it:,} random perturbation")

        a = a_new
        T *= 0.9997                     # slow cooling
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises the given score.

    The function respects a 1000 s time budget (default) and returns the best
    sequence discovered.  It does **not** call ``evaluate_sequence`` itself;
    the returned list can be fed to that function later.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    prev = globals().get('height_sequence_1')
    if isinstance(prev, (list, tuple, np.ndarray)):
        n = len(prev)
        have_ref = True
    else:
        n = 4000                # a comfortable size that is still fast
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # normalisation used in the paper
    global _size_fft_global
    _size_fft_global = 1 << ((3 * n - 2).bit_length())   # power‑of‑2 ≥ 3n‑2

    # ------------------------------------------------------------------
    # Helper for fast scoring (exact, but no security checks)
    def quick_score(a: np.ndarray) -> float:
        conv = _self_convolution(a, _size_fft_global)
        return float(2 * n * conv.max() / (target_sum ** 2))

    # ------------------------------------------------------------------
    # Keep the K best raw candidates
    K = 15
    best_candidates = []          # list of (score, ndarray)

    def consider(sc: float, a: np.ndarray) -> None:
        """Maintain a sorted list of the K best candidates."""
        if len(best_candidates) < K:
            best_candidates.append((sc, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if sc < best_candidates[-1][0]:
                best_candidates[-1] = (sc, a.copy())
                best_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # --------------- 1. PARAMETRIC SHAPE SEARCH --------------------
    idx = np.arange(n, dtype=float)

    # Uniform (baseline)
    a = np.ones(n)
    a = _project_to_simplex(a, target_sum)
    consider(quick_score(a), a)

    # Helper to sample and evaluate a raw family
    def sample_family(gen_func, repeats: int):
        for _ in range(repeats):
            a = gen_func()
            a = _project_to_simplex(a, target_sum)
            consider(quick_score(a), a)

    # ---- Inverse‑sqrt (single‑ended) ----
    def gen_inv_sqrt():
        beta = np.random.rand() * 10.0
        return 1.0 / np.sqrt(idx + beta + 1.0)
    sample_family(gen_inv_sqrt, 80)

    # ---- Double‑ended inverse‑sqrt ----
    def gen_double_inv_sqrt():
        beta = np.random.rand() * 10.0
        left = 1.0 / np.sqrt(idx + beta + 1.0)
        right = 1.0 / np.sqrt((n - 1 - idx) + beta + 1.0)
        return left + right
    sample_family(gen_double_inv_sqrt, 80)

    # ---- Power‑law ----
    def gen_power():
        beta = np.random.rand() * 10.0
        pexp = np.random.rand() * 1.7 + 0.3          # exponent ∈ [0.3, 2.0]
        return 1.0 / (idx + beta + 1.0) ** pexp
    sample_family(gen_power, 80)

    # ---- Exponential decay ----
    def gen_expo():
        decay = np.random.rand() * 0.01 + 0.0005    # ∈ [5e‑4, 1.05e‑2]
        return np.exp(-decay * idx)
    sample_family(gen_expo, 80)

    # ---- Gaussian bump (centered) ----
    mid = (n - 1) / 2.0
    def gen_gauss():
        sigma = np.random.rand() * (n / 6.0) + n / 30.0   # reasonable width
        return np.exp(-0.5 * ((idx - mid) / sigma) ** 2)
    sample_family(gen_gauss, 60)

    # ---- Beta‑like shapes (a·(1‑a) type) ----
    def gen_beta():
        a_param = np.random.rand() * 5.0 + 0.2
        b_param = np.random.rand() * 5.0 + 0.2
        x = (idx + 1) / n
        return (x ** (a_param - 1)) * ((1 - x) ** (b_param - 1))
    sample_family(gen_beta, 60)

    # ---- Arcsine distribution ----
    def gen_arcsine():
        # add epsilon to avoid division by zero at the ends
        eps = 1e-6
        return 1.0 / np.sqrt((idx + eps) * (n - idx + eps))
    sample_family(gen_arcsine, 40)

    # ---- Sinusoidal (zero at ends, peak middle) ----
    def gen_sin():
        p = np.random.rand() * 5.0 + 0.5
        return (np.sin(np.pi * idx / (n - 1)) ** p)
    sample_family(gen_sin, 80)

    # ---- Cosine‑based (large at ends, small in centre) ----
    def gen_cos():
        p = np.random.rand() * 5.0 + 0.5
        return (np.abs(np.cos(np.pi * idx / (n - 1))) ** p)
    sample_family(gen_cos, 80)

    # ---- Random block structure (up to 15 blocks) ----
    def gen_random_blocks():
        nb = np.random.randint(2, 16)                     # 2‑15 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05                  # avoid exact zeros
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        return cand
    sample_family(gen_random_blocks, 200)

    # ---- Smoothed random noise (low‑pass filtered) ----
    def gen_smooth_noise():
        noise = np.random.randn(n)
        # simple moving average filter of width ≈ n/30
        wlen = max(3, n // 30)
        kernel = np.ones(wlen) / wlen
        smooth = np.convolve(noise, kernel, mode='same')
        smooth -= smooth.min()                # shift to non‑negative
        return smooth + 1e-3
    sample_family(gen_smooth_noise, 80)

    # ------------------------------------------------------------------
    # Report best raw candidate

    print(f"[shape search] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # --------------- 2. REFINEMENT (gradient descent) -------------

        print(f"[refine start] candidate #{rank+1} (raw score {raw_score:.6f})")
        remaining = deadline - time.time()
        # allocate iterations proportionally to remaining time

        refined_a, refined_score = _refine_candidate(
            a0, deadline, n, _size_fft_global, target_sum, max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # --------------- 3. POST‑REFINEMENT RANDOM WALK ---------------
    # small random mass‑transfer steps followed by a short descent
    while time.time() < deadline:
        # create a perturbed copy of the current global best
        a = global_best_a.copy()
        # random transfer: pick two indices and move a tiny amount
        i, j = np.random.randint(0, n, size=2)
        delta = (np.random.rand() * 0.01) * target_sum   # ≤1 % of total mass
        if a[i] >= delta:
            a[i] -= delta
            a[j] += delta
        else:
            # if not enough mass, just shift whatever is there
            delta = a[i]
            a[i] = 0.0
            a[j] += delta
        a = _project_to_simplex(a, target_sum)

        # a quick short descent (few hundred iterations)
        refined_a, refined_score = _refine_candidate(
            a, deadline, n, _size_fft_global, target_sum, max_iters=800
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[walk] improved to {global_best_score:.6f}")

        # break if only a tiny sliver of time remains
        if time.time() + 0.05 > deadline:
            break

    # ------------------------------------------------------------------
    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
