def propose_candidate(seed=0, budget_s=1000):
    import time, math, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000
    target_sum = math.sqrt(2.0 * n)
    n2 = 1 << (2 * n - 1).bit_length()
    def max_conv(vec):
        f = np.fft.rfft(vec, n2)
        conv = np.fft.irfft(f * f, n2)
        return float(conv[:2*n-1].max())
    def score_from_vec(vec):
        s = vec.sum()
        mb = max_conv(vec)
        return 2.0 * n * mb / (s * s)
    # keep best
    best_vec = None
    best_score = float('inf')
    iters = 0
    # Phase 1: random family sampling
    while time.time() < deadline - 5.0:
        # choose family uniformly at random
        family = np.random.randint(5)
        if family == 0:  # random dirichlet
            cand = np.random.exponential(1.0, size=n)
        elif family == 1:  # power law
            alpha = np.random.uniform(0.1, 2.5) # exponent
            idx = np.arange(n)
            cand = (idx + 1) ** (-alpha)
        elif family == 2:  # exponential

        elif family == 3:  # sinusoidal

        else:  # block pattern

def max_conv_fft(a, n, n2):
    # Using real FFT optimisation.
    f = np.fft.rfft(a, n2)
    conv = np.fft.irfft(f * f, n2)
    # conv is length n2, we need first 2*n-1 entries.

        return x
    return x * (target / s)

    f = np.fft.rfft(a, n2)
    conv = np.fft.irfft(f * f, n2)
    # only first 2n-1 entries are relevant
    return float(conv[:2*n-1].max())

# Window generation functions
def gen_hann(n):
    return np.hanning(n)

def gen_hamming(n):
    return np.hamming(n)

def gen_blackman(n):
    return np.blackman(n)

def gen_gaussian(n, sigma_frac):
    i = np.arange(n)
    mu = (n-1)/2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)

def gen_tukey(n, alpha):
    # Tukey window
    if alpha <= 0:
        return np.ones(n)
    if alpha >= 1:
        return np.hanning(n)
    t = np.arange(n) / (n-1)
    w = np.ones_like(t)
    # first region
    first = t < alpha/2
    w[first] = 0.5 * (1 + np.cos(np.pi * (2*t[first]/alpha - 1)))
    # last region
    last = t > (1 - alpha/2)
    w[last] = 0.5 * (1 + np.cos(np.pi * (2*t[last]/alpha - 2/alpha + 1)))
    return w

def gen_kaiser(n, beta):
    if i0 is None:
        # fallback simple: Hamming if scipy not available
        return np.hamming(n)
    i = np.arange(n) - (n-1)/2.0
    denom = (n-1)/2.0
    denom = max(denom, 1e-12)
    w = np.i0(beta * np.sqrt(1 - (i/denom)**2)) / np.i0(beta)
    return w

def gen_exponential(n, lam):
    # lam > 0
    i = np.arange(n)
    return np.exp(-lam * i)

def gen_power_law(n, alpha):
    # alpha >= 0
    i = np.arange(1, n+1)
    return i.astype(float) ** (-alpha)

"""
Propose a low‑score non‑negative step function.

The algorithm explores a wide variety of parametric window families
(Hann, Hamming, Blackman, Gaussian, Tukey, Kaiser, exponential,
power‑law, sinusoidal‑power) and also random Dirichlet / block patterns.
For each candidate the sum is normalised to √(2 n) (the normalisation
does not affect the final score) and the maximum of the self‑convolution
is computed by an FFT‑based O(n log n) routine.  The best candidate found
is further refined by a cheap hill‑climb that repeatedly moves a random
fraction of mass from one index to another; any improvement is kept.
The whole search stops just before the allotted wall‑clock budget.
"""

import time
import math
import numpy as np

try:
    from scipy.special import i0  # optional, used for Kaiser windows
except Exception:  # pragma: no cover
    i0 = None


def _scale_to_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector so that its ℓ¹ norm equals *target*."""
    s = x.sum()
    if s == 0.0:
        return x
    return x * (target / s)


def _max_conv_fft(a: np.ndarray, n: int, n2: int) -> float:
    """Return max entry of the full self‑convolution of *a* using FFT."""
    # real‑FFT for speed
    f = np.fft.rfft(a, n2)
    conv = np.fft.irfft(f * f, n2)
    # conv length = n2, only first 2n‑1 entries are relevant
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Window / pattern generators (all return non‑negative arrays of length n)
def _gen_hann(n: int) -> np.ndarray:
    return np.hanning(n)


def _gen_hamming(n: int) -> np.ndarray:
    return np.hamming(n)


def _gen_blackman(n: int) -> np.ndarray:
    return np.blackman(n)


def _gen_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((i - mu) / sigma) ** 2)


def _gen_tukey(n: int, alpha: float) -> np.ndarray:
    """Tukey (tapered cosine) window; alpha∈[0,1]."""
    if alpha <= 0.0:
        return np.ones(n, dtype=float)
    if alpha >= 1.0:
        return np.hanning(n)
    t = np.arange(n, dtype=float) / (n - 1)
    w = np.ones_like(t)
    first = t < alpha / 2
    w[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1 - alpha / 2
    w[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return w


def _gen_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window; falls back to Hamming if scipy.special.i0 is unavailable."""
    if i0 is None:
        return np.hamming(n)
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    denom = (n - 1) / 2.0
    arg = beta * np.sqrt(1 - (i / denom) ** 2)
    return np.i0(arg) / np.i0(beta)


def _gen_exponential(n: int, lam: float) -> np.ndarray:
    return np.exp(-lam * np.arange(n, dtype=float))


def _gen_power_law(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    return i ** (-alpha)


def _gen_sin_pow(n: int, gamma: float) -> np.ndarray:
    return np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n) ** gamma


def _gen_random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex using exponential variables."""
    return np.random.exponential(1.0, size=n)


def _gen_block_random(n: int) -> np.ndarray:
    """Piecewise‑constant pattern with a random number of blocks."""
    nb = np.random.randint(2, 9)          # 2‑8 blocks
    cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)

    out = np.empty(n, dtype=float)

    return out
