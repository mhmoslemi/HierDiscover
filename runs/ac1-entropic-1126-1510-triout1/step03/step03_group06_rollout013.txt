def _optimise_adam(n, init_seq, deadline, max_iters=8000, eta0=0.1, lambda0=0.1, lambda_max=80.0):
    a = init_seq.copy()
    a /= a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_score = _score(a)
    m_fft = 1 << ((3 * n - 2).bit_length())
    lambda_inc = (lambda_max - lambda0) / max_iters
    lamb = lambda0

    for t in range(1, max_iters+1):
        if time.time()>= deadline:
            break
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"adam n={n} it={t} score={best_score:.6f}")

        w = _softmax_weights(b, lamb)
        g = _gradient_via_correlation(w, a, m_fft)
        # update Adam
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # increase lambda
        lamb = min(lambda_max, lamb + lambda_inc)
        # occasional jitter
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
    return best_seq, best_score
```

"""
Propose a low‑score step‑function.

The algorithm builds on the previous approach but adds a few
enhancements:

*   A larger search space (up to 20 000 points).  As the discretisation
    becomes finer the achievable bound approaches the theoretical limit.
*   A new optimiser – Adam – applied to a soft‑max approximation of the
    peak of the self‑convolution.  Adam’s adaptive steps often reach
    lower minima than plain exponentiated‑gradient descent.
*   More diverse initial distributions (arcsine, beta with a wide range
    of α, triangular, sinusoidal, sparse spikes and low‑concentration
    Dirichlet draws).
*   A simple time‑budget manager that keeps the total runtime ≤ budget s.
*   A final stochastic hill‑climb that nudges a tiny amount of mass
    between positions only if the true (unsmoothed) objective improves.

All vectors are kept non‑negative and normalised to sum = 1, which makes
the evaluator’s scaling invariant.  The routine returns a plain Python
list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangular (tent) distribution."""
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _sinusoidal_distribution(n: int) -> np.ndarray:
    """Half‑period sine distribution, normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals[vals < 0] = 0.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised to sum = 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights for a given temperature λ."""
    maxb = b.max()
    exp_arg = lam * (b - maxb)          # ≤ 0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(b, 1.0 / b.shape[0])
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """A modest, diverse pool of starting probability vectors for size n."""
    cand = []

    cand.append(_arcsine_distribution(n))                     # arcsine
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))        # uniform

    # symmetric betas on a grid of α values
    for alpha in (0.2, 0.35, 0.5, 0.8, 1.2, 1.5):
        cand.append(_beta_distribution(n, alpha))

    # a few random beta draws
    for _ in range(2):
        alpha = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, alpha))

    cand.append(_triangular_distribution(n))                 # tent
    cand.append(_sinusoidal_distribution(n))                 # sine

    # sparse spikes
    k = max(2, n // 500)
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    cand.append(a / a.sum())

    # low‑concentration Dirichlet
    cand.append(np.random.dirichlet(np.full(n, 0.3)))

    return cand


# ----------------------------------------------------------------------
def _optimise_exponential(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 8000,
    eta0: float = 0.15,
    lambda0: float = 0.3,
    lambda_max: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the soft‑max approximation
    of max(conv(a,a)).
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    m_fft = 1 << ((3 * n - 2).bit_length())
    lam = lambda0
    lam_inc = (lambda_max - lambda0) / max_iters

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] exp n={n:5d} it={it:5d} score={best_score:.6f}")

        w = _softmax_weights(b, lam)
        g = _gradient_via_correlation(w, a, m_fft)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # projection onto the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)

        # occasional jitter to escape tiny plateaus
        if it % 800 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        lam = min(lambda_max, lam + lam_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _optimise_adam(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 12000,
    eta0: float = 0.05,
    lambda0: float = 20.0,
    lambda_max: float = 80.0,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser applied to the same soft‑max surrogate.
    Uses projection onto the simplex after each update.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    m_fft = 1 << ((3 * n - 2).bit_length())
    lam = lambda0
    lam_inc = (lambda_max - lambda0) / max_iters

    pow1 = 1.0
    pow2 = 1.0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] adam n={n:5d} it={it:5d} score={best_score:.6f}")

        w = _softmax_weights(b, lam)
        g = _gradient_via_correlation(w, a, m_fft)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        pow1 *= beta1
        pow2 *= beta2
        m_hat = m / (1.0 - pow1)
        v_hat = v / (1.0 - pow2)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project back onto the probability simplex
        a = _project_onto_simplex(a, s=1.0)

        # jitter occasionally
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        lam = min(lambda_max, lam + lam_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact peak of the self‑convolution.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # sub‑gradient g_i = 2·a_{k*−i} on the overlap interval
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] sub n={n:5d} it={it:5d} score={best_score:.6f}")

        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny mass from one index
    to another and keep the change if the (unsmoothed) objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  use historic sequence if present
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  problem sizes – we concentrate on a large size but also try a few
    #       smaller ones for diversity.
    candidate_lengths = [4000, 8000, 20000]  # 20k gives the most refined discretisation

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        print(f"\n--- Optimising length n={n} ---")
        init_pool = _initial_candidates(n)

        best_n_seq = None
        best_n_score = float("inf")

        for init in init_pool:
            if time.time() >= deadline:
                break

            # ---- exponentiated‑gradient phase (fast, coarse) ----
            seq_exp, sc_exp = _optimise_exponential(
                n=n,
                init_seq=init,
                deadline=deadline,
                max_iters=4000,
                eta0=0.12,
                lambda0=0.2,
                lambda_max=30.0,
            )

            # ---- Adam refinement (more precise) ----
            seq_adam, sc_adam = _optimise_adam(
                n=n,
                init_seq=seq_exp,
                deadline=deadline,
                max_iters=8000,
                eta0=0.04,
                lambda0=30.0,
                lambda_max=80.0,
            )

            # pick the better of the two local runs
            if sc_adam < sc_exp:
                local_seq, local_sc = seq_adam, sc_adam
            else:
                local_seq, local_sc = seq_exp, sc_exp

            if local_sc < best_n_score:
                best_n_score = local_sc
                best_n_seq = local_seq.copy()
                print(f"  >>> New best for n={n}: score={best_n_score:.6f}")

        # ---- optional sub‑gradient polishing (if time permits) ----
        if best_n_seq is not None and time.time() < deadline:
            sub_deadline = deadline - 0.5
            seq_sub, sc_sub = _subgradient_descent(
                n=n,
                init_seq=best_n_seq,
                deadline=sub_deadline,
                max_iters=2000,
                eta0=0.25,
            )
            if sc_sub < best_n_score:
                best_n_score = sc_sub
                best_n_seq = seq_sub.copy()
                print(f"  *** Sub‑grad refined n={n}: score={best_n_score:.6f}")

        # ---- update the global best ----
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # early exit if target already achieved
        if overall_best_score <= 1.503:
            print("Target bound reached – terminating early.")
            break

    # ------------------------------------------------------------------
    # 3️⃣  final stochastic hill‑climbing refinement
    if overall_best_seq is not None and time.time() < deadline:
        hill_deadline = deadline
        seq_hc, sc_hc = _hill_climb(overall_best_seq, hill_deadline, max_trials=20000)
        if sc_hc < overall_best_score:
            overall_best_score = sc_hc
            overall_best_seq = seq_hc.copy()
            print(f"*** After hill‑climb: final score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  fallback (should never be needed)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"\nFinal best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    return overall_best_seq.tolist()
