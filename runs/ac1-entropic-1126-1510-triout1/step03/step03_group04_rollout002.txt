def golden_section_search(f, a, b, tol=1e-4, max_iter=30):
    phi = (1 + sqrt(5))/2
    resphi = 2 - phi  # 1/phi^2
    c = b - resphi * (b - a)
    d = a + resphi * (b - a)
    fc = f(c)
    fd = f(d)
    for i in range(max_iter):
        if time.time()>deadline: break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - resphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + resphi * (b - a)
            fd = f(d)
        if abs(b-a) < tol:
            break
    # get the best point near mid
    if fc < fd:
        return c, fc
    else:
        return d, fd

"""
Propose a low‑score step function.

The algorithm is deliberately simple but exploits a known near‑optimal
family of shapes:

*  a_i = r**i  (geometric decay)  with r = 1 - c / n .
   The parameter c ≈ 2 gives the smallest possible value of the
   evaluation score 2·n·max(a*a) / (∑a)² for large n.
   The bound is essentially independent of the length n, so we reuse the
   length of the best known construction if it exists, otherwise we fall
   back to a medium‑size n≈3000.

The search proceeds in three phases:

1. **Coarse grid** over c in [0.5, 4.5] → coarse candidate.
2. **Golden‑section refinement** of c around the best grid point.
3. **Tiny stochastic hill‑climb** that moves a fraction of mass between
   two positions to possibly escape discretisation artefacts.

All scores are computed with an O(n log n) FFT based max‑convolution,
so thousands of candidates fit easily into the 1000 s wall‑clock budget.
The final sequence is returned as a plain Python list of non‑negative
floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fast_score(arr: np.ndarray, n: int, n2: int) -> float:
    """Return the evaluation score using the same formula as evaluate_sequence,
    but with an FFT based max‑convolution (much faster)."""
    # FFT of the sequence, zero‑padded to length n2 (next power of two ≥2n‑1)
    f = np.fft.rfft(arr, n2)
    conv = np.fft.irfft(f * f, n2)
    max_conv = float(conv[: 2 * n - 1].max())
    sum_a = float(arr.sum())
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
def _grid_search(n: int, n2: int) -> tuple[float, np.ndarray]:
    """Coarse scan of the parameter c (where r = 1 - c/n). Returns the best
    c and the corresponding array."""
    best_val = float("inf")
    best_c = None
    best_arr = None

    # coarse grid: 81 points uniformly spaced in [0.5, 4.5]
    for c in np.linspace(0.5, 4.5, 81):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        arr = r ** np.arange(n, dtype=np.float64)
        val = _fast_score(arr, n, n2)
        if val < best_val:
            best_val = val
            best_c = c
            best_arr = arr
    return best_c, best_arr


# ----------------------------------------------------------------------
def _golden_section_refine(
    c0: float, n: int, n2: int, deadline: float, arr0: np.ndarray
) -> tuple[float, np.ndarray]:
    """Refine the parameter c around c0 with golden‑section search."""
    # widen the interval a little, but stay inside (0, n)
    delta = max(0.5, 0.1 * c0)
    a = max(1e-6, c0 - delta)
    b = min(n - 1e-6, c0 + delta)

    phi = (1 + 5 ** 0.5) / 2.0
    resphi = 2 - phi  # 1/phi**2

    # initial interior points
    c = b - resphi * (b - a)
    d = a + resphi * (b - a)

    def score_for(cval: float) -> float:
        r = 1.0 - cval / n
        if r <= 0.0:
            return float("inf")
        arr = r ** np.arange(n, dtype=np.float64)
        return _fast_score(arr, n, n2)

    fc = score_for(c)
    fd = score_for(d)

    # iterate until interval is tiny or time runs out
    while time.time() < deadline - 0.5 and (b - a) > 1e-5:
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - resphi * (b - a)
            fc = score_for(c)
        else:
            a, fc = c, fd
            c = d
            d = a + resphi * (b - a)
            fd = score_for(d)

    # pick the best of the two interior points
    if fc < fd:
        best_c = c
        best_val = fc
        r = 1.0 - best_c / n
        best_arr = r ** np.arange(n, dtype=np.float64)
    else:
        best_c = d
        best_val = fd
        r = 1.0 - best_c / n
        best_arr = r ** np.arange(n, dtype=np.float64)

    # also compare with the initial array just in case numerical noise
    init_val = _fast_score(arr0, n, n2)
    if init_val < best_val:
        return c0, arr0
    return best_c, best_arr


# ----------------------------------------------------------------------
def _hill_climb(arr: np.ndarray, n: int, n2: int, deadline: float) -> np.ndarray:
    """Very cheap stochastic hill‑climb that swaps a fraction of mass between
    two coordinates."""
    best_arr = arr.copy()
    best_score = _fast_score(best_arr, n, n2)

    while time.time() < deadline - 0.5:
        # choose a source index with noticeable mass
        nz = np.flatnonzero(best_arr > 1e-12)
        if nz.size == 0:
            break
        i = np.random.choice(nz)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # move a random fraction (10‑50 %) of the source mass
        delta = best_arr[i] * np.random.uniform(0.1, 0.5)
        cand = best_arr.copy()
        cand[i] -= delta
        cand[j] += delta
        # no need for projection – sums stay constant and all entries stay ≥0
        cand = np.clip(cand, 0.0, None)
        cand_score = _fast_score(cand, n, n2)
        if cand_score < best_score:
            best_score = cand_score
            best_arr = cand
    return best_arr


# ----------------------------------------------------------------------

    """

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05  # safety margin

    # --------------------------------------------------------------
    # Determine the problem size.
    try:
        n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        n = 3000                     # a comfortable default

    # Pre‑compute the FFT length (next power of two ≥ 2n‑1)

    # --------------------------------------------------------------

    best_c, best_arr = _grid_search(n, n2)
