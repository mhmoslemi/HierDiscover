def _build_convolution_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # column index i (b_i)
        # For rows k=i..i+n-1, coefficient f_{k-i}=f_{k-i}
        # That is for j from 0 to n-1: k = i + j, coefficient f_j
        cols.extend([i] * n)
        rows.extend(range(i, i + n))
        data.extend(f)
    # rows range up to 2n-2
    T = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return T

```
def _gradient_from_weights(a, w, size_fft):
    n = a.shape[0]
    len_w = w.shape[0]
    # compute correlation w cross a using FFT
    W = np.fft.rfft(w, n=size_fft)
    A = np.fft.rfft(a, n=size_fft)
    corr = np.fft.irfft(W * np.conj(A), n=size_fft)
    start = len_w - 1
    grad = 2.0 * corr[start:start + n]
    return grad
```

Propose a non‑negative step function that minimises
evaluate_sequence().  The method consists of four stages:

1. **Broad parametric search** – many simple families (uniform,
   single‑ended power‑law, double‑ended power‑law, exponential,
   cosine, logistic, Gaussian, random block) are sampled,
   projected onto the simplex Σa = √(2 n) and scored via a fast FFT
   convolution. The best candidate is retained.

   compute the full self‑convolution g = a∗a, build a soft‑max
   weighting wₖ ∝ exp(α·(gₖ−max g)) (α grows during the run) and
   obtain the gradient from the correlation w◦a.
   A back‑tracking line‑search together with Euclidean projection
   onto the simplex guarantees monotone improvement.

3. **Simple hill‑climbing** – random mass transfers between two
   positions are accepted only if they lower the maximal convolution.
   This refines the solution locally after the gradient phase.

4. **Random‑block final sweep** – a few fresh block‑structured
   candidates are evaluated in the remaining time.

All heavy work (convolutions, correlations) uses real‑valued FFTs,
so each iteration is O(n log n) with n≈3000.  The total wall‑clock
budget is 1000 s; the routine stops early if the time limit is

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear self‑convolution a∗a using FFT."""
    A = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(A * A, n=size_fft)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Maximum entry of the self‑convolution."""
    conv = _self_convolution_fft(a, size_fft)
    return float(conv.max())

def _gradient_from_weights(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of  Σₖ wₖ·(a∗a)ₖ  w.r.t. a.
    Implements  grad = 2·(w ⋆ a)_{0…n‑1},
    i.e. correlation of w with a (real FFT version).
    """
    n = a.shape[0]
    # correlation w ⋆ a = irfft( fft(w) * conj(fft(a)) )
    W = np.fft.rfft(w, n=size_fft)
    A = np.fft.rfft(a, n=size_fft)
    corr = np.fft.irfft(W * np.conj(A), n=size_fft)
    start = w.shape[0] - 1            # index of zero lag in full correlation
    grad = 2.0 * corr[start:start + n]
    return grad
