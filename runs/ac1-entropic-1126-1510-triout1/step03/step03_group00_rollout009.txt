"""
Hybrid search for a low‑score step function.

The routine consists of two stages.

1️⃣  **Parameter sweep** – a simple family of symmetric β‑distributions
    a_i ∝ ((i+0.5)*(n‑i‑0.5))^(‑β) is examined for a handful of lengths.
    The internal objective 2·n·max(a∗a) is evaluated (the true score after
    evaluate_sequence is identical when the coefficients sum to one).
    The best candidate of this cheap sweep seeds stage 2.

2️⃣  **Fine optimisation** – starting from the best candidate we run an
    Adam‑based projected optimisation of the smoothed max–convolution.
    The “soft‑max’’ of the convolution values (temperature α) yields a
    differentiable surrogate for the supremum; its gradient is obtained by
    a pair of FFT‑based convolutions.  α is increased geometrically so the
    optimisation finishes with a true sub‑gradient step.  After each
    gradient step the vector is projected onto the probability simplex.
    The best vector found before the time budget expires is returned.

The whole process respects the 1000‑second limit; the last few seconds are
reserved for a final call to ``evaluate_sequence`` and for printing the
result.
"""

import time
from typing import List

import numpy as np


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]          # descending sorted copy
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _beta_distribution(n: int, beta: float) -> np.ndarray:
    """
    Symmetric β‑distribution of length n.
    For β=0            → uniform,
    β=0.5 (arcsine)   → 1/√((i+½)(n‑i‑½)).
    """
    if beta == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        i = np.arange(n, dtype=np.float64)
        vals = ((i + 0.5) * (n - i - 0.5)) ** (-beta)
        a = vals / vals.sum()
    # guarantee exact simplex membership
    return _project_onto_simplex(a)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a via a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return np.max(_fft_self_convolve(a))


def _gradient_via_softmax(
    a: np.ndarray,
    b: np.ndarray,
    alpha: float,
    size_fft: int,
) -> np.ndarray:
    """
    Gradient of  (1/α)·log∑_k exp(α·b_k)  with respect to a.
    For very large α we fall back to the true sub‑gradient (the max entry only).
    """
    n = a.shape[0]

    # very large α → use sub‑gradient at the max index
    if alpha > 2000.0:
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g

    # soft‑max weights
    shift = b - b.max()                       # for numerical stability
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution w * a_rev  (a_rev = a reversed)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


def _adam_optimize(
    init_a: np.ndarray,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max surrogate of the max self‑convolution.
    Returns the best vector encountered and its exact internal score
    (2·n·max(a*a)).
    """
    a = init_a.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _max_self_convolution(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 0.5
    alpha_mul = 1.07
    alpha_max = 15000.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # full convolution needed for both objective and gradient
        b_full = _fft_self_convolve(a)
        cur_max = b_full.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Adam it={it} n={n} score={best_score:.6f}")

        # gradient of the smooth max
        grad = _gradient_via_softmax(a, b_full, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex (sum = 1, non‑negative)
        a = _project_onto_simplex(a, s=1.0)

        # increase the temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if it % 3000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the given time budget and returns the best sequence
    found as a plain Python list.
    """
    import numpy as np

    np.random.seed(seed)

    # keep a small safety margin for final printing/output
    deadline = time.time() + budget_s - 3.0

    # ------------------------------------------------------------------
    # 1️⃣  Base length (use previous best if available)
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # lengths to explore (between 500 and 20000)
    length_factors = (0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted(
        {
            max(500, int(base_n * f))
            for f in length_factors
            if max(500, int(base_n * f)) <= 20000
        }
    )
    # a few extra hand‑picked lengths
    for extra in (4000, 6000, 8000, 10000, 12000):
        if 500 <= extra <= 20000:
            lengths.append(extra)
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # 2️⃣  Very cheap parameter sweep (≈30 % of the time budget)
    sweep_deadline = time.time() + max(0.05, (deadline - time.time()) * 0.3)

    best_seq = None
    best_score = float("inf")
    best_n = None

    beta_grid = np.linspace(0.0, 1.0, 21)   # 0 → uniform, 0.5 → arcsine, etc.

    for n in lengths:
        if time.time() >= sweep_deadline:
            break

        for beta in beta_grid:
            a = _beta_distribution(n, beta)

            # internal objective (exact because sum(a)=1)
            max_conv = _max_self_convolution(a)
            score = 2.0 * n * max_conv

            if score < best_score:
                best_score = score
                best_seq = a.copy()
                best_n = n
                print(
                    f"[{time.time():.2f}] sweep n={n} beta={beta:.3f} → score={score:.6f}"
                )

    # Fallback – should never be needed
    if best_seq is None:
        best_n = max(500, int(base_n * 1.0))
        best_seq = np.full(best_n, 1.0 / best_n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3️⃣  Fine optimisation with Adam (use remaining time)
    remaining = deadline - time.time()
    if remaining > 0.5:
        opt_deadline = time.time() + remaining
        # run Adam starting from the best sweep candidate
        seq_opt, internal_opt = _adam_optimize(best_seq, best_n, opt_deadline)

        # compare against the sweep result using the official black‑box
        try:
            final_score = evaluate_sequence(seq_opt.tolist())
        except Exception:                     # extremely unlikely
            final_score = internal_opt

        if final_score < best_score:
            best_score = final_score
            best_seq = seq_opt.copy()
            print(f"[{time.time():.2f}] Adam improvement → final score={final_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final safety check & return
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final best score: {final_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
