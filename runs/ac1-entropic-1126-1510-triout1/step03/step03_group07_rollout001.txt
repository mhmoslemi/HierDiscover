Algorithm
---------
1.  Determine a problem size N (≈1.5× the length of the previously
    discovered sequence, capped at 30 000).  The ℓ1‑norm is fixed to
    S = √(2·N) so that the evaluation score coincides with max(conv).

    (``height_sequence_1``) to length N and projecting onto the
    non‑negative simplex of sum S (small random noise is added).

3.  Minimise max(conv) with Adam‑accelerated projected gradient descent
    on a smooth soft‑max approximation of the max:

        •  Learning rate decays from 0.6 → 0.02.
        •  Gradient = 2·corr(w, a) where w are the soft‑max weights.
        •  After each step the vector is projected onto the simplex.

    small amount of mass between two random positions (mass is conserved).
    The move is accepted if it improves the score, otherwise with
    probability exp(−Δ/ T) where the temperature T cools from 0.02 to 1e‑5.

5.  The best sequence seen during the whole budget is returned.

The whole routine respects the 1000 s time budget, prints progress

import time
import math

# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random pairwise mass move (conserves total sum)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Shift a small amount of mass from one index to another."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# FFT‑based full convolution (a * a) – length 2·n‑1
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based correlation of w (length 2·n‑1) with a (length n) – result length n
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                    # = 2·n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]   # valid correlation, length n


# ----------------------------------------------------------------------
# Resize an existing sequence (linear interpolation) to a new length.
# ----------------------------------------------------------------------
def _resize_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    x_old = np.arange(old_len, dtype=float)
    x_new = np.linspace(0, old_len - 1, new_len, dtype=float)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low max‑convolution.
    Returns the best sequence (list of floats) found before the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Determine problem size.
    # --------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=float)   # type: ignore
        n0 = base_seq.size
    except Exception:
        base_seq = None
        n0 = 15000

    # Target length – 1.5× the known construction, capped at 30000.
    N = int(min(30000, max(12000, int(1.5 * n0))))
    S = math.sqrt(2.0 * N)                # ℓ1‑norm to enforce (score = max(conv))

    # --------------------------------------------------------------
    # Initialise feasible point.
    # --------------------------------------------------------------
    if base_seq is not None:
        a = _resize_sequence(base_seq, N)
    else:
        a = np.full(N, S / N, dtype=float)

    # small random perturbation to break symmetry
    a += 0.001 * S * np.random.rand(N) / N
    a = _project_onto_simplex(a, S)

    # keep global best
    best_seq = a.copy()
    conv = _fft_convolve(a)
    best_score = conv.max()
    print(f"[Init] N={N}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Adam‑accelerated projected gradient descent on a soft‑max of conv.
    # --------------------------------------------------------------
    max_iters = 50000                     # may be cut early by time
    beta_start = 5.0
    beta_end = 250.0
    lr_start = 0.6
    lr_end = 0.02

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(N, dtype=float)
    v = np.zeros(N, dtype=float)

    for it in range(1, max_iters + 1):
        # Time check – abort if out of budget
        if time.time() >= deadline:
            break

        # Full convolution of current iterate
        conv = _fft_convolve(a)
        max_conv = conv.max()

        # Soft‑max temperature (log‑linear schedule)
        frac = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** frac

        # Compute soft‑max weights (stable formulation)
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # Gradient of the smooth objective (2·corr(w, a))
        grad = 2.0 * _fft_correlate(w, a)

        # Adam update (minimise → move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Learning‑rate schedule (linear decay)
        lr = lr_start + (lr_end - lr_start) * frac
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a + step
        # Project back onto the feasible simplex
        a = _project_onto_simplex(a, S)

        # Periodic evaluation
        if it % 50 == 0 or it == max_iters:
            conv = _fft_convolve(a)
            score = conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[Iter {it:05d}] New best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Polishing phase – simulated‑annealing random‑pair moves.
    # --------------------------------------------------------------
    polish_iters = 25000
    T_start = 0.02
    T_end = 0.00001

    for it in range(1, polish_iters + 1):
        if time.time() >= deadline:
            break

        # Linear cooling of the move amplitude σ
        sigma = 0.4 * (1.0 - it / polish_iters) + 0.02
        cand = _random_pair_move(best_seq, sigma)

        conv_cand = _fft_convolve(cand)
        score_cand = conv_cand.max()
        delta = score_cand - best_score

        # Temperature for Metropolis acceptance (log‑linear schedule)
        t_frac = it / polish_iters
        T = T_start * (T_end / T_start) ** t_frac

        if delta < 0:
            best_seq = cand
            best_score = score_cand
            print(f"[Polish {it:05d}] Improved to {best_score:.6f}")
        else:
            if np.random.rand() < math.exp(-delta / T):
                # Accept a worse state (helps escaping local minima)
                best_seq = cand
                best_score = score_cand

        # occasional logging
        if it % 5000 == 0:
            print(f"[Polish {it:05d}] Current best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # safety clamp
    return best_seq.tolist()
