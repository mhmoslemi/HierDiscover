    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    try:
        n = len(height_sequence_1)
        a0 = np.array(height_sequence_1, dtype=float)
    except NameError:
        n = 3000
        a0 = np.ones(n, dtype=float)

    # project to simplex sum=1
    a = _project_to_simplex(a0, 1.0)
    best_a = a.copy()
    best_score = compute_score(a)

    # Evaluations counter
    it = 0

    # Simulated annealing wrapper
    while time.time() < deadline:
        # decide move type
        move_type = np.random.choice(['swap','smooth','noise'])
        if move_type == 'swap':
            i = np.random.randint(n)
            j = np.random.randint(n-1)
            if j >= i: j += 1  # ensure j != i
            delta = (np.random.rand() * 0.1) * a[i]  # at most 10% of a[i]
            # ensure not exceed a[i]
            delta = min(delta, a[i])
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
        elif move_type == 'smooth':
            L = np.random.randint(3, 50)
            start = np.random.randint(0, n-L+1)
            block = slice(start, start+L)
            avg = a[block].mean()
            cand = a.copy()
            cand[block] = avg
        else: # noise
            sigma = 0.01 * (np.random.rand() + 0.5) # ~0.005-0.015
            cand = a + sigma * np.random.randn(n)
            cand = np.maximum(cand, 0.0)
            # project onto simplex sum=1
            cand = _project_to_simplex(cand, 1.0)

        # compute score quickly
        sc = compute_score(cand)

        # accept if improvement or with small probability
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand  # move to new location
            print(f"[improve] new best = {best_score:.6f} at it {it}")
        else:
            # maybe accept worse to escape, with probability exp(-(sc-best)/temp)
            # temperature schedule decreasing ~ (deadline - time)/deadline
            temp = max(1e-6, (deadline - time.time()) / (budget_s))
            if np.random.rand() < np.exp(-(sc - best_score) / temp):
                a = cand  # move anyway
        it += 1
        if it % 10000 == 0:
            print(f"[status] iter {it}, best_score {best_score:.6f}")

    # optionally final polishing: subgradient descent
    a = best_a.copy()
    eta0 = 0.1 / n
    decay = 1e-5
    it2 = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = (conv == max_c)
        cnt = mask.sum()
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        g = 2 * np.convolve(w, a[::-1])[n-1:n-1+n]  # gradient shape n
        eta = eta0 / (1 + decay * it2)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0)
        a_new = _project_to_simplex(a_new, 1.0)
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[grad improve] new best = {best_score:.6f} it2 {it2}")
        a = a_new
        it2 += 1
        if it2 > 10000:
            break

    print(f"[final] best score = {best_score:.6f}")

"""
Propose a low‑score step‑function using a multi‑phase optimisation:

1. Initialise several diverse starting points on the probability simplex.
2. Continuation with a smooth‑max surrogate (log‑sum‑exp) minimised by L‑BFGS‑B.
3. Large random block perturbations (cheap but global) to escape local minima.
4. Projected sub‑gradient descent on the true max‑convolution objective.
5. Simple stochastic hill‑climbing (mass swaps / Gaussian noise) for final polishing.

All vectors are kept on the simplex (sum = 1); the final answer is scaled to
∑a = sqrt(2·n) which leaves the evaluation score unchanged.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector (sum = 1)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (log‑sum‑exp surrogate) and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Free parameters (real vector of length n).
    mu : float
        Temperature (>0). Smaller mu gives a tighter surrogate.
    n : int
        Length of the sequence.
    S : float
        Desired ℓ¹‑norm; for a probability vector we use S = 1.0.

    Returns
    -------
    f_val : float
        Value of the smooth surrogate.
    grad_y : np.ndarray
        Gradient ∂f/∂y (shape (n,)).
    """
    # probabilities and actual heights
    p = _softmax(y)        # Σp = 1, p_i ≥ 0
    a = S * p              # Σa = S

    # full convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)
    max_c = conv.max()

    # smooth‑max (log‑sum‑exp) weights
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:   # hard max case (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a (see paper for derivation)
    conv_wr = np.convolve(w, a[::-1])         # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]      # slice length n

    # chain rule to obtain gradient w.r.t. y
    inner = np.dot(g_a, p)                    # scalar = Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable wrapper required by scipy.optimize.minimize."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def _score_from_prob(a: np.ndarray) -> float:
    """
    Compute the evaluation score for a probability vector a (Σa = 1).
    The score equals 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score non‑negative step‑function within the time budget."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size and reference sequence (if any)
    try:
        height_sequence_1  # type: ignore[name-defined]
        have_ref = True
        n = len(height_sequence_1)       # type: ignore[name-defined]
    except Exception:
        have_ref = False
        n = 3000

    # ------------------------------------------------------------------
    # target ℓ¹‑norm for the soft‑max parameterisation (probability simplex → sum = 1)
    S = 1.0

    # ------------------------------------------------------------------
    # initialise best candidate on the simplex (sum = 1)
    if have_ref:
        a0 = np.asarray(height_sequence_1, dtype=float)      # type: ignore[name-defined]
        a0 = _project_to_simplex(a0, S)
    else:
        a0 = np.ones(n, dtype=float) / n

    best_a = a0.copy()
    best_score = _score_from_prob(best_a)
    print(f"[init] length={n}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max continuation (L‑BFGS‑B) from several starts
    # ------------------------------------------------------------------
    init_y = []

    # 0. uniform (y = 0)
    init_y.append(np.zeros(n, dtype=np.float64))

    # 1. V‑shaped (large at ends, tiny in the centre)
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    vshape = np.abs(idx - mid) + 1e-3
    init_y.append(np.log(vshape))

    # 2. random Gaussian draws
    for _ in range(3):
        init_y.append(np.random.randn(n) * 0.5)

    # 3. reference if available
    if have_ref:
        eps = 1e-12
        init_y.append(np.log(a0 + eps))

    # continuation parameters
    mu_factor = 0.7            # slower cooling than before
    mu_min = 1e-7
    maxiter_per_mu = 200       # more L‑BFGS steps per temperature

    for start_id, y0 in enumerate(init_y):
        if time.time() > deadline:
            break
        print(f"[cont] start #{start_id}")

        y = y0.copy()
        a_cur = _softmax(y)    # stays on simplex, sum = 1
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        if mu <= 0.0:
            mu = 1.0

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)

            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = _softmax(y)                 # probability vector
            sc = _score_from_prob(a)

            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[improve] new best = {best_score:.6f}")

            mu *= mu_factor

    # ------------------------------------------------------------------
    # Phase 2 – cheap random block search (global exploration)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        block_trials = int(min(20000, remaining * 200))  # up to ~10 000–20 000 trials
        print(f"[block] performing {block_trials} random block trials")
        for _ in range(block_trials):
            # random number of blocks (2‑8)
            nb = np.random.randint(2, 9)
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)  # uniform random heights
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            sc = _score_from_prob(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[block improve] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the hard max (μ = 0)
    # ------------------------------------------------------------------
    a = best_a.copy()
    eta0 = 0.5 / n           # initial step size roughly the average entry
    decay = 1e-5
    it = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        # gradient w.r.t a
        g_a = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_to_simplex(a_new, S)

        sc = _score_from_prob(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[grad improve] it {it}, new best = {best_score:.6f}")

        a = a_new
        it += 1
        if it > 8000:      # safety stop
            break

    # ------------------------------------------------------------------
    # Phase 4 – stochastic hill‑climbing (mass swaps & Gaussian perturbations)
    # ------------------------------------------------------------------
    a = best_a.copy()
    it_hc = 0
    while time.time() < deadline:
        # choose move type
        if np.random.rand() < 0.6:
            # mass transfer between two indices
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            if a[i] == 0:
                continue
            delta = np.random.rand() * 0.02 * a[i]   # up to 2 % of a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # Gaussian noise + projection
            sigma = 0.01 * (0.5 + np.random.rand())
            cand = a + sigma * np.random.randn(n)
            cand = np.maximum(cand, 0.0)
            cand = _project_to_simplex(cand, S)

        sc = _score_from_prob(cand)

        # simple annealing schedule (temperature decays linearly)
        temp = max(1e-6, (deadline - time.time()) / (budget_s))
        accept = sc < best_score or (np.random.rand() < math.exp(-(sc - best_score) / temp))

        if accept:
            a = cand
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[hc improve] it {it_hc}, new best = {best_score:.6f}")

        it_hc += 1
        if it_hc % 5000 == 0:
            print(f"[hc status] iter {it_hc}, best_score {best_score:.6f}")

    # ------------------------------------------------------------------
    # final scaling (optional, restores the conventional ℓ¹‑norm sqrt(2·n))
    # ------------------------------------------------------------------
    scaling = math.sqrt(2.0 * n) / best_a.sum()
    final_a = (best_a * scaling).tolist()
    final_score = evaluate_sequence(final_a)  # sanity check with the provided routine
    print(f"[final] best_score (eval) = {final_score:.6f}")

    return final_a
