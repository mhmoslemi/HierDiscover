"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1.  Generate a diverse pool of initial probability vectors for several
    lengths (4000‑10000).  The pool contains arcsine, uniform, several
    symmetric Beta laws, sparse “spike” vectors and low‑concentration
    Dirichlet draws.

2.  For each length run a long exponentiated‑gradient (mirror) descent
    on a soft‑max approximation of the peak of the self‑convolution.
    The temperature λ is increased slowly up to 80 and the step size
    decays as 1/√t.  After each step the vector is projected back onto
    the simplex.

3.  Refine the best candidate of the length with projected sub‑gradient
    descent on the exact max‑convolution peak.

4.  Perform a simulated‑annealing local search on the global best
    sequence.  Each move transfers a small amount of mass from one index
    to another; the convolution array is updated in O(n) using the exact
    quadratic update formula, and moves are accepted with a temperature
    schedule (exponential cooling).

5.  Optionally finish with a cheap hill‑climb.  The algorithm respects a
    1000 s wall‑clock budget and prints intermediate scores.

The returned sequence is a plain Python list of non‑negative numbers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised to sum 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a small, diverse set of initial probability vectors for size n."""
    cand = []

    # arcsine
    cand.append(_arcsine_distribution(n))

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric betas with a grid of α
    for alpha in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.9, 1.2):
        cand.append(_beta_distribution(n, alpha))

    # random beta draws
    for _ in range(3):
        cand.append(_beta_distribution(n, np.random.uniform(0.1, 2.0)))

    # sparse spikes
    k = max(2, n // 500)
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    cand.append(a / a.sum())

    # low‑concentration Dirichlet
    cand.append(np.random.dirichlet(np.full(n, 0.3)))

    return cand


# ----------------------------------------------------------------------
def _optimise_exponential(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    eta0: float = 0.2,
    lambda0: float = 0.5,
    lambda_max: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent (exponentiated‑gradient) on a soft‑max approximation of
    max(conv(a,a)). Returns the best vector (sum 1) seen and its true score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT length for correlation (needs to accommodate w length 2n‑1)
    m = 1 << ((3 * n - 2).bit_length())

    # Linear schedule for λ
    lambda_inc = (lambda_max - lambda0) / max_iters
    lamb = lambda0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} exp it={it:5d} score={best_score:.6f}")

        maxb = b.max()
        exp_arg = lamb * (b - maxb)            # ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        g = _gradient_via_correlation(w, a, m)

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # Simple simplex projection (renormalise)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = np.full(n, 1.0 / n)

        # occasional jitter to escape tiny plateaus
        if it % 1000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        lamb = min(lambda_max, lamb + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # sub‑gradient g_i = 2·a_{k*−i}
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        if it % 800 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(
    a: np.ndarray,
    deadline: float,
    max_steps: int = 30000,
    init_temp_ratio: float = 0.04,
    final_temp_ratio: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Simple SA on the exact objective.
    Maintains the convolution array and updates it in O(n) per move.
    """
    n = len(a)
    a = a.copy()
    conv = _conv_full_fft(a)
    cur_score = 2.0 * n * conv.max()
    best_seq = a.copy()
    best_score = cur_score

    # temperature schedule (exponential)
    T0 = init_temp_ratio * cur_score
    Tf = final_temp_ratio * cur_score
    if max_steps <= 1:
        decay = 1.0
    else:
        decay = (Tf / T0) ** (1.0 / max_steps)

    T = T0

    for step in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # Pick donor (i) with some mass and receiver (j)
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # random transfer amount
        max_delta = min(a[i] * 0.05, 0.02)   # up to a few percent of donor
        delta = np.random.rand() * max_delta

        # Save old values for revert
        old_i = a[i]
        old_j = a[j]

        # Linear part of convolution update using old a (pre‑move)
        # slice updates: c[i:i+n] -= 2*delta*old_a ; c[j:j+n] += 2*delta*old_a
        conv[i:i + n] -= 2.0 * delta * a
        conv[j:j + n] += 2.0 * delta * a

        # Quadratic adjustments
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        # Apply the move to a
        a[i] = old_i - delta
        a[j] = old_j + delta

        # Compute new score
        new_score = 2.0 * n * conv.max()

        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - cur_score) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA step={step} score={best_score:.6f}")
        else:
            # revert move
            a[i] = old_i
            a[j] = old_j
            conv[i:i + n] += 2.0 * delta * a   # add back the previous subtraction
            conv[j:j + n] -= 2.0 * delta * a
            conv[2 * i] -= delta * delta
            conv[2 * j] -= delta * delta
            conv[i + j] += 2.0 * delta * delta
            # cur_score stays unchanged

        # temperature cooling
        T *= decay

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣  Load historic sequence if present
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2️⃣  Lengths to explore (larger lengths give finer granularity)
    candidate_lengths = [4000, 6000, 8000, 10000]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        print(f"--- Exploring length n={n} ---")
        init_pool = _initial_candidates(n)

        best_n_seq = None
        best_n_score = float("inf")

        # a) exponentiated‑gradient phase (long)
        for init in init_pool:
            if time.time() >= deadline:
                break
            seq, sc = _optimise_exponential(
                n=n,
                init_seq=init,
                deadline=deadline,
                max_iters=20000,
                eta0=0.2,
                lambda0=0.5,
                lambda_max=80.0,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  >>> New best for n={n}: score={best_n_score:.6f}")

        # b) sub‑gradient refinement
        if best_n_seq is not None and time.time() < deadline:
            seq, sc = _subgradient_descent(
                n=n,
                init_seq=best_n_seq,
                deadline=deadline,
                max_iters=5000,
                eta0=0.25,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  *** Sub‑grad refined n={n} score={best_n_score:.6f}")

        # c) update global best
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # Early exit if we already beat a strong target
        if overall_best_score <= 1.51:
            print("Early exit: target region reached.")
            break

    # --------------------------------------------------------------
    # 3️⃣  Simulated annealing refinement on the global best
    if overall_best_seq is not None and time.time() < deadline:
        sa_deadline = deadline
        seq, sc = _simulated_annealing(
            overall_best_seq,
            sa_deadline,
            max_steps=30000,
            init_temp_ratio=0.04,
            final_temp_ratio=0.001,
        )
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** After SA: score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final cheap hill‑climb (optional)
    if overall_best_seq is not None and time.time() < deadline:
        # reuse the hill‑climb from the previous run (simple version)
        n = len(overall_best_seq)
        a = overall_best_seq.copy()
        best_score = _score(a)
        for t in range(8000):
            if time.time() >= deadline:
                break
            i = np.random.randint(n)
            if a[i] < 1e-12:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = np.random.rand() * min(a[i] * 0.05, 0.01)
            trial = a.copy()
            trial[i] -= delta
            trial[j] += delta
            cur = _score(trial)
            if cur < best_score:
                best_score = cur
                a = trial
                print(f"[{time.time():.1f}] final hill‑climb t={t} score={best_score:.6f}")
        overall_best_seq = a
        overall_best_score = best_score

    # --------------------------------------------------------------
    # 5️⃣  Fallback (should not happen)
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"Final best: length={len(overall_best_seq)} score={overall_best_score:.6f}")

    # Return as plain Python list for the evaluation harness
    return overall_best_seq.tolist()
