"""
Propose a low‑score non‑negative step‑function.

Algorithm
---------
1. Load a historic sequence (if present) and keep it as a baseline.
2. For a few problem sizes (8000, 12000, 20000) generate a diverse pool of
   initial candidates:
        – discrete arcsine law,
        – uniform,
        – symmetric Beta(α,α) for a fine grid of α,
        – low‑concentration Dirichlet draws,
        – sparse symmetric “spike’’ vectors.
3. For every candidate run a long exponentiated‑gradient (mirror‑descent)
   optimisation on a soft‑max approximation of the convolution peak.
   The temperature λ is increased up to 150 and the step size η decays as
   1/√t.  After each update the vector is symmetrised (a ← (a+a[::-1])/2)
   and projected onto the probability simplex.
4. Refine the best candidate for each length with projected sub‑gradient
   descent on the exact peak.
5. Upsample the current global best to a finer grid (≈2× longer) and
   perform a short extra exponential run – this reduces discretisation
   error.
6. Finish with a cheap stochastic hill‑climbing phase that moves a tiny
   amount of mass between two indices whenever the true (unsmoothed)
   score improves.
7. Return the best sequence seen as a plain list.  All operations are
   O(n log n) thanks to FFTs, allowing thousands of iterations well within
   the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT (length 2·n‑1)."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _make_symmetric(a: np.ndarray) -> np.ndarray:
    """Force a to be symmetric around the centre (a_i = a_{n-1-i})."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
def _optimise_exponential(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 8000,
    eta0: float = 0.12,
    lambda0: float = 0.2,
    lambda_max: float = 150.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a soft‑max approximation of max(conv(a,a)).
    Uses multiplicative updates (exponentiated gradient) and enforces symmetry.
    Returns the best vector (∑a = 1) seen and its exact unsmoothed score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    a = _make_symmetric(a)               # ensure symmetry from the start
    a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    # FFT length sufficient for correlation of length 2n‑1 with a (n)
    m = 1 << ((3 * n - 2).bit_length())

    lamb = lambda0
    lambda_inc = (lambda_max - lambda0) / max_iters

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and (soft) score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # soft‑max weights
        maxb = b.max()
        exp_arg = lamb * (b - maxb)            # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient wrt a
        g = _gradient_via_correlation(w, a, m)

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # enforce symmetry, then renormalise onto simplex
        a = _make_symmetric(a)
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)

        # occasional jitter to escape plateaus
        if it % 800 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = _make_symmetric(a)
            a /= a.sum()

        # increase temperature
        lamb = min(lambda_max, lamb + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.22,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Enforces symmetry after each projection.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    a = _make_symmetric(a)
    a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # sub‑gradient: g_i = 2·a_{k*−i} for indices that contribute
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        # restore symmetry and re‑normalise
        a = _make_symmetric(a)
        a /= a.sum()

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        # jitter occasionally
        if it % 500 == 0 and np.random.rand() < 0.18:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = _make_symmetric(a)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny mass from one index to
    another and keep the change if the (unsmoothed) objective improves.
    The candidate stays symmetric by moving the same amount to the mirrored
    partner of the receiver (if needed).
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)                     # donor
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)                     # receiver
        if i == j:
            continue

        # amount to transfer (up to 5 % of donor, capped)
        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        # enforce symmetry explicitly (mirror the pair of the moved mass)
        trial = _make_symmetric(trial)
        trial /= trial.sum()      # tiny drift after mirroring, re‑normalise

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a diverse set of initial probability vectors for size n."""
    cand = []

    # 1) arcsine
    cand.append(_arcsine_distribution(n))

    # 2) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) a fine grid of symmetric betas
    for alpha in np.linspace(0.25, 2.0, 12):
        cand.append(_beta_distribution(n, alpha))

    # 4) random betas
    for _ in range(3):
        alpha = np.random.uniform(0.1, 2.5)
        cand.append(_beta_distribution(n, alpha))

    # 5) sparse symmetric spikes (few non‑zero entries)
    for _ in range(3):
        k = max(2, n // 250)  # roughly 0.4 % of positions
        idx = np.random.choice(n // 2, size=k, replace=False)
        a = np.zeros(n, dtype=np.float64)
        # assign random mass to each selected index and its mirror
        vals = np.random.rand(k) + 0.1
        a[idx] = vals
        a[-np.array(idx) - 1] = vals  # mirror
        cand.append(a / a.sum())

    # 6) Dirichlet draws with low concentration (many tiny entries)
    a = np.random.dirichlet(np.full(n, 0.2))
    cand.append(a)

    return cand


# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.clip(up, 0.0, None)
    up /= up.sum()
    up = _make_symmetric(up)
    up /= up.sum()
    return up

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # --------------------------------------------------------------
    # 1️⃣  Historic sequence, if available
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2️⃣  Problem sizes to explore (larger n gives a finer discretisation)
    candidate_lengths = [8000, 12000, 20000]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        print(f"\n--- Exploring length n={n} ---")

        # pool of starting points
        init_pool = _initial_candidates(n)

        best_n_seq = None
        best_n_score = float("inf")

        # --- a) Exponentiated‑gradient (mirror descent) -----------------
        for init in init_pool:
            if time.time() >= deadline:
                break
            seq, sc = _optimise_exponential(
                n=n,
                init_seq=init,
                deadline=deadline,
                max_iters=8000 if n <= 12000 else 5000,
                eta0=0.12,
                lambda0=0.2,
                lambda_max=150.0,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  >>> New best for n={n}: score={best_n_score:.6f}")

        # --- b) Sub‑gradient refinement --------------------------------
        if best_n_seq is not None and time.time() < deadline:
            seq, sc = _subgradient_descent(
                n=n,
                init_seq=best_n_seq,
                deadline=deadline,
                max_iters=3000,
                eta0=0.22,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  *** Sub‑grad refined n={n}: score={best_n_score:.6f}")

        # --- c) Update global best ------------------------------------
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_seq = best_n_seq.copy()
            overall_best_score = best_n_score
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # early exit if we already beat the target
        if overall_best_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # --------------------------------------------------------------
    # 3️⃣  Upsample the best vector to a finer grid and fine‑tune ----------
    if overall_best_seq is not None and time.time() < deadline:
        # aim for roughly double the current resolution (but not beyond 25000)
        target_n = min(25000, max(20000, len(overall_best_seq) * 2))
        if target_n > len(overall_best_seq):
            up_seq = _upsample_sequence(overall_best_seq, target_n)
            print(f"Upsampling to n={target_n} for final refinement.")
            seq, sc = _optimise_exponential(
                n=target_n,
                init_seq=up_seq,
                deadline=deadline,
                max_iters=4000,
                eta0=0.08,
                lambda0=0.2,
                lambda_max=150.0,
            )
            if sc < overall_best_score:
                overall_best_seq = seq.copy()
                overall_best_score = sc
                print(f"Upsample refinement improved to score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Final cheap hill‑climbing ---------------------------------------
    if overall_best_seq is not None and time.time() < deadline:
        seq, sc = _hill_climb(overall_best_seq, deadline, max_trials=20000)
        if sc < overall_best_score:
            overall_best_seq = seq.copy()
            overall_best_score = sc
            print(f"*** After hill‑climb: score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Fallback (should never be needed) -------------------------------
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _arcsine_distribution(n)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform-like sequence used.")

    print(f"\nFinal best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    # Return as plain Python list for the evaluation harness
    return overall_best_seq.tolist()
