"""
Propose a low‑score step function.

The algorithm works on a single (large) length chosen from the historic
construction.  It starts from a smooth “arcsine” distribution, then runs a
two‑stage gradient descent:

1.  A soft‑max surrogate of the peak of the self‑convolution provides a
    smooth gradient.  The temperature (α) is annealed so the surrogate
    approaches the true max.

2.  After the temperature is large we switch to the exact sub‑gradient of the
    max‑entry (the derivative of the convolution at its largest index).

Both stages use simple momentum and a projection onto the probability
simplex (non‑negative, sum = 1).  Random jitter is added occasionally to
escape plateaus, and several restarts are performed in the allocated time.
The best sequence found (according to the black‑box `evaluate_sequence`) is
returned.
"""

import time
import numpy as np


def _simplex_normalize(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback to uniform distribution
        v = np.full_like(v, 1.0 / v.size)
    else:
        v = v / s
    return v


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (β(½,½)) of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors a and b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    Search for a non‑negative step function (list of heights) with a low
    `evaluate_sequence` score.  The routine runs for up to `budget_s` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # a couple of seconds for tidy‑up

    # ------------------------------------------------------------------
    # Choose a length (power of two for fast FFT).  Prefer the length of the
    # historic best construction if it exists; otherwise pick a large default.
    try:

    except Exception:

    # theoretical limit.
    lengths = []

        raw = int(base_n * factor)
        raw = max(500, raw)
        n_pow2 = 1 << ((raw - 1).bit_length())      # next power of two
