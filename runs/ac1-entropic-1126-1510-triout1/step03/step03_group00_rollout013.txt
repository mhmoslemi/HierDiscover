"""
Propose a non‑negative step‑function (list of heights) that minimises
`evaluate_sequence`.  

The algorithm is a multi‑restart projected Adam optimiser on the simplex.
For each candidate length we initialise the vector with several sensible
patterns (uniform, arcsine, random, a previous best and a linear
interpolation of the previous best).  The objective is
    f(a) = 2·n·max_k (a * a)[k]      (the sum of a is fixed to 1)
which is non‑smooth because of the inner maximum.  We smooth the max by a
soft‑max (log‑sum‑exp) with a temperature α that is increased
geometrically; for large α the gradient reduces to the sub‑gradient at the
current maximal convolution entry.  Adam’s adaptive steps together with a
projection onto the probability simplex provide fast, stable progress.
Occasional random jitter helps to escape plateaus.  The best sequence found
over all lengths and restarts is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial probability vector of length n for the standard kinds."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    else:                      # fallback
        a = _arcsine_distribution(n)
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _initial_sequence_custom(n: int, kind: str) -> np.ndarray:
    """
    Initialise a vector of length n.
    Supported kinds:
        'uniform', 'random', 'arcsine', 'prev', 'interp'
    """
    # ------------------------------------------------------------------
    # previous best (global variable `height_sequence_1`) if it exists
    prev = None
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
        except Exception:
            prev = None

    # ------------------------------------------------------------------
    if kind == "prev":
        if prev is not None and prev.size == n:
            a = prev.copy()
        else:
            a = _arcsine_distribution(n)
    elif kind == "interp":
        # linearly interpolate the previous best to the new length
        if prev is not None:
            if prev.size == 1:
                a = np.full(n, prev[0], dtype=np.float64)
            else:
                x_src = np.linspace(0.0, 1.0, prev.size)
                x_dst = np.linspace(0.0, 1.0, n)
                a = np.interp(x_dst, x_src, prev)
        else:
            a = _arcsine_distribution(n)
    else:
        a = _initial_sequence(n, kind)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑_k exp(α·b_k) w.r.t. a.
    For very large α we fall back to the hard max sub‑gradient.
    """
    n = a.shape[0]

    # hard max case – a single dominant index
    if alpha > 2000.0:
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g

    # smooth case – full soft‑max
    shift = b - b.max()               # for numerical stability
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                  # fallback (should not happen)
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _optimize_one_length_adam(init_a: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for a fixed length n.
    Returns the best sequence seen and its internal (2·n·max_b) score.
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = 1.0                      # base step‑size
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # soft‑max temperature schedule
    alpha = 1.0
    alpha_mul = 1.03
    alpha_max = 5e4

    size_fft = 1 << ((3 * n - 2).bit_length())   # for w * a_rev

    while time.time() < deadline:
        t += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={t} score={best_score:.6f}")

        # gradient of smoothed max
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = eta0 / np.sqrt(t)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter to shake out of plateaus
        if t % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score
