"""
Propose a non‑negative step‑function that minimizes
    2·n·max(a*a) / (sum(a)**2)

The algorithm works in three stages:

1. **Exploratory parametric sweep** – generate many simple families
   (uniform, inverse‑sqrt, power‑law, exponential, cosine, sine,
   Gaussian, double‑ended, random block) and keep the K best according
   to the exact score.

2. **Subgradient descent on the exact max‑convolution** – for each of the
   top candidates run a projected sub‑gradient method that uses the true
   max‑convolution (no soft‑max) and Euclidean projection onto the simplex
   {a≥0, Σa = √(2 n)}.  Momentum (Nesterov‑type) and occasional random
   perturbations accelerate escape from plateaus.

3. **Fine‑grained random pairwise swaps** – starting from the best refined
   sequence, perform cheap random mass‑reallocation moves (i↔j) that keep
   the total sum fixed.  Accept only improving moves; this typically
   chisels away the remaining few hundredths of the score.

All heavy work uses FFT‑based convolution (O(n log n)) and runs for at most
the 1000 s budget; the function returns the best sequence found.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a  (weights w sum to 1).
    The exact expression is 2·(w ↔ a_rev) restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _quick_score(a: np.ndarray, size_fft: int) -> float:
    """Exact evaluation score (no security checks)."""
    n = len(a)
    conv = _self_convolution(a, size_fft)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _refine_subgrad_candidate(a0: np.ndarray,
                              deadline: float,
                              n: int,
                              target_sum: float,
                              size_fft: int,
                              max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max(a*a) with momentum.
    Returns best sequence seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _quick_score(a, size_fft)

    # momentum vars
    vel = np.zeros_like(a)
    momentum = 0.9
    # base step; scaled to typical magnitude of a
    eta0 = target_sum * 0.6 / n
    decay = 1e-5

    it = 0
    while time.time() < deadline and it < max_iters:
        # 1) current convolution and max‑set
        conv = _self_convolution(a, size_fft)
        max_val = conv.max()
        mask = (conv == max_val)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / mask.sum()            # uniform over all maximal positions

        # 2) gradient of the max‑convolution (sub‑gradient)
        grad = _weighted_gradient(a, w, size_fft)

        # 3) step‑size schedule & momentum update
        eta = eta0 / (1.0 + decay * it)
        vel = momentum * vel - eta * grad
        a_new = a + vel

        # 4) project onto the simplex (keeps Σa = target_sum, a≥0)
        a_new = _project_to_simplex(a_new, target_sum)

        # 5) evaluate
        conv_new = _self_convolution(a_new, size_fft)
        cur_score = float(2.0 * n * conv_new.max() / (target_sum ** 2))

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # occasional progress report
            print(f"[subgrad] it {it:,} new best = {best_score:.6f}")

        # occasional random kick to avoid stagnation
        if (it % 5000 == 0 and it > 0) or (it % 20000 == 0 and it > 0):
            noise = np.random.randn(n) * target_sum * 5e-4
            a_kick = a_new + noise
            a_kick = _project_to_simplex(a_kick, target_sum)
            a_new = a_kick
            print(f"[subgrad] it {it:,} random perturbation")

        a = a_new
        it += 1

    return best_a, best_score

def _local_pairwise_search(a0: np.ndarray,
                           deadline: float,
                           n: int,
                           target_sum: float,
                           size_fft: int) -> tuple[np.ndarray, float]:
    """
    Cheap random pairwise mass swaps (i↔j) that keep the sum fixed.
    Accept only improving moves. Returns the best sequence found.
    """
    a = a0.copy()
    best_score = _quick_score(a, size_fft)

    it = 0
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)

        # propose a small transfer delta
        delta = (np.random.rand() - 0.5) * target_sum * 0.02
        if a[i] + delta < 0 or a[j] - delta < 0:
            continue

        a_new = a.copy()
        a_new[i] += delta
        a_new[j] -= delta

        # sum unchanged, non‑negativity kept
        cur_score = _quick_score(a_new, size_fft)

        if cur_score < best_score:
            best_score = cur_score
            a = a_new
            print(f"[local] it {it:,} new best = {best_score:.6f}")

        # occasional larger random block reshuffle
        if np.random.rand() < 0.001:
            nb = np.random.randint(2, 7)
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb) + 0.1
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, target_sum)
            cand_score = _quick_score(cand, size_fft)
            if cand_score < best_score:
                best_score = cand_score
                a = cand
                print(f"[local] random block improved to {best_score:.6f}")

        it += 1
        # break early if virtually no time left
        if time.time() + 0.02 > deadline:
            break

    return a, best_score

        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        _push_candidate(_quick_score(cand, size_fft), cand)

    best_candidates.sort(key=lambda x: x[0])
    print(f"[shape search] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # ---------- 2. REFINEMENT (exact max‑convolution sub‑gradient) --
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    for rank, (raw_sc, a_start) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank+1} (raw score {raw_sc:.6f})")
        remaining = deadline - time.time()
        # allocate a generous slice of the remaining budget
        max_iters = int(min(300_000, remaining * 1400))
        refined_a, refined_sc = _refine_subgrad_candidate(
            a_start, deadline, n, target_sum, size_fft, max_iters
        )
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---------- 3. FINAL LOCAL PAIRWISE SWEEP ----------------------
    if time.time() < deadline:
        print("[final] starting local pairwise search")
        local_a, local_sc = _local_pairwise_search(
            global_best_a, deadline, n, target_sum, size_fft
        )
        if local_sc < global_best_score:
            global_best_score = local_sc
            global_best_a = local_a.copy()
            print(f"[global] new best after local search = {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
