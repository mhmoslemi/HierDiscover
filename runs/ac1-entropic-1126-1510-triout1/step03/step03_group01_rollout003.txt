def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1
    # Determine base length
    try:
        base_seq = np.array(height_sequence_1, dtype=np.float64)
        base_n = base_seq.size
    except Exception:
        base_seq = None
        base_n = 6000
    # We'll try lengths: base_n, int(base_n*0.9), int(base_n*1.1) ensuring >=500
    lengths = [base_n]
    for f in (0.8, 1.2):
        n_candidate = int(base_n * f)
        if n_candidate >= 500:
            lengths.append(n_candidate)
    lengths = list(dict.fromkeys(lengths))
    best_seq = None
    best_score = float('inf')
    # For each length:
    for n in lengths:
        # Set initial a
        if base_seq is not None and base_seq.size == n:
            a = base_seq.copy()
        else:
            a = _arcsine_distribution(n)
        a = a / a.sum()
        # optimization hyperparameters
        eta0 = 0.4
        alpha = 5.0
        alpha_factor = 1.004
        max_alpha = 5000.0
        it = 0
        # Adam parameters maybe not needed if using EG
        # We'll just use EG with decaying eta.
        while time.time() < deadline:
            it += 1
            # compute convolution
            b = _fft_self_convolve(a)
            cur_max = b.max()
            # compute weight w
            if alpha < max_alpha:
                # softmax
                exp_term = np.exp(alpha * (b - cur_max))
                w = exp_term / exp_term.sum()
                # increase alpha
                alpha = min(max_alpha, alpha * alpha_factor)
                phase = "softmax"
            else:
                # hard max subgradient
                # Determine indices within tolerance of max
                eps = 1e-12
                max_idx = np.where(np.abs(b - cur_max) <= eps)[0]
                if max_idx.size == 0:
                    max_idx = np.array([np.argmax(b)])
                w = np.zeros_like(b)
                w[max_idx] = 1.0 / max_idx.size
                phase = "hard"
            # gradient via convolution:
            a_rev = a[::-1]
            # compute conv via fft, full length of a_rev (n) + w (2n-1) - 1 = 3n-2, we need slice after n-1
            conv_rev = _fft_convolve(a_rev, w)  # length = n + (2n-1) -1 = 3n-2
            grad = 2.0 * conv_rev[n-1 : n-1 + n]  # shape n
            # scale step size
            eta = eta0 / np.sqrt(it)
            # limit grad magnitude to avoid overflow
            max_abs = np.max(np.abs(grad))
            if max_abs > 20:
                grad = grad * (20 / max_abs)
            # exponentiated gradient update
            a = a * np.exp(-eta * grad)
            # re-normalize
            sum_a = a.sum()
            if sum_a <= 0 or np.isnan(sum_a):
                a = np.full(n, 1.0/n)
                sum_a = 1.0
            else:
                a = a / sum_a
            # store best exact after some iterations
            if it % 2000 == 0 or time.time() > deadline - 10:
                # compute exact score
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] n={n} iter={it} phase={phase} cur_max={cur_max:.8e} cur_score≈{cur_score:.6f}")
                # also evaluate using black-box evaluate_sequence occasionally
                if it % 10000 == 0:
                    eval_score = evaluate_sequence(a.tolist())
                    if eval_score < best_score:
                        best_score = eval_score
                        best_seq = a.copy()
                        print(f"[{time.time():.1f}] Evaluated score {eval_score:.6f} (better) at n={n} iter={it}")
            # occasional jitter
            if it % 5000 == 0 and np.random.rand() < 0.1:
                a += np.random.randn(n) * 1e-6
                a = np.maximum(a, 0)
                a = a / a.sum()
        # end while per length
        # after finishing this length, evaluate best_seq if not yet done
        if best_seq is not None:
            eval_final = evaluate_sequence(best_seq.tolist())
            if eval_final < best_score:
                best_score = eval_final
                best_seq = best_seq.copy()
    # fallback if none.
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0/n)
        best_score = evaluate_sequence(best_seq.tolist())
    print(f"Final best score: {best_score:.6f} length={len(best_seq)}")
    return best_seq.tolist()

the simplex constraint is enforced by a simple normalisation after each
exponentiated‑gradient step.  Several candidate lengths (around the length
of the previously discovered sequence, if any) are explored sequentially,
and the best exact score seen (using the provided ``evaluate_sequence``)

The algorithm is deliberately lightweight (no Python closures, no lambdas)

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project onto the simplex {x≥0, Σx = s} (Euclidean projection)."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution on {0,…,n‑1} (probability vector)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]
