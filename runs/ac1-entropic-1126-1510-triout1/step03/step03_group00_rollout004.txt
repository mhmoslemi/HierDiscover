def _optimize_exp_gradient(a, n, deadline):
    best_seq, best_score = a.copy(), evaluate_score_approx(a) # using 2n*max_b
    α = 0.5
    α_mul = 1.03
    α_max = 20000
    eta = 0.5
    eta_decay = 0.999
    iteration = 0
    size_fft = 1 << ((3 * n - 2).bit_length())
    while time.time() < deadline:
        iteration+=1
        # compute convolution
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={iteration} score={best_score}")
        # compute softmax weights
        shift = b - b.max()
        w = np.exp(α * shift)
        w /= w.sum()
        # gradient
        grad = _gradient_via_softmax(a, b, α, size_fft)  # returns same as earlier.
        # multiplicative update
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0:
            # reinitialize randomly
            a = np.random.rand(n)
        a = a / a_sum
        # schedule
        α = min(α_max, α * α_mul)
        eta *= eta_decay
        if iteration % 5000 == 0:
            # add jitter
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0)
            a = a / a.sum()

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2
    alpha = 0.5
    alpha_mul = 1.03
    alpha_max = 20000
    size_fft = 1 << ((3*n - 2).bit_length())
    best_a = a.copy()
    best_score = 2.0*n * _fft_self_convolve(a).max()
    t = 0
    while time.time() < deadline:
        t += 1
        # compute conv and score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} it={t} score={best_score:.6f}")
        # compute gradient via softmax
        grad = _gradient_via_softmax(a, b, alpha, size_fft)
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # update schedules
        alpha = min(alpha_max, alpha * alpha_mul)
        lr *= 0.9995  # slow decay
        # occasional jitter
        if t % 5000 == 0 and np.random.rand() < 0.3:
            a += np.random.randn(n) * 1e-4
            a = np.maximum(a, 0)
            a = a / a.sum()
    return best_a, best_score

"""
propose_candidate
=================

Search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm works on the probability simplex
(sum = 1) and repeatedly reduces the maximal self‑convolution
`max(a*a)` using a smooth approximation (soft‑max) together with an
Adam optimiser and Euclidean projection onto the simplex.

The optimisation is performed for several sequence lengths (derived
from the previous best) and with a few random restarts.  The best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) w.r.t. a.
    If α is very large the gradient reduces to a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # sub‑gradient when α is huge
    if alpha > 2000.0:
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g

    # soft‑max weights
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution w * a_rev
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _optimize_adam_softmax(a0: np.ndarray,
                           n: int,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution.
    Returns the best sequence (sum = 1) seen before ``deadline`` and its
    internal score 2·n·max(a*a).
    """
    a = a0.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12               # initial learning rate
    # temperature for soft‑max
    alpha = 0.1
    alpha_mul = 1.05
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # ≥ (2n‑1)+(n)‑1
    best_a = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution & score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient of smoothed max
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # schedules
        alpha = min(alpha_max, alpha * alpha_mul)

            a += np.random.randn(n) * 5e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()
