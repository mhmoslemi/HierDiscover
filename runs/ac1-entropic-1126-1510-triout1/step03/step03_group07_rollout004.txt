def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy, math, random
    np.random.seed(seed)
    start = time.time()
    deadline = start + max(0, budget_s - 2)
    # Determine N
    try:
        base = np.array(height_sequence_1, dtype=float)
        n0 = base.size
    except Exception:
        base = None
        n0 = 2000
    # Choose N
    N = min(30000, max(10000, n0*2))
    # Choose initial a
    if base is not None:
        # Upsample or interpolate base to length N
        a = np.interp(np.linspace(0, len(base)-1, N), np.arange(len(base)), base)
    else:
        a = np.full(N, 1.0/N)
    # Enforce sum S
    S = math.sqrt(2.0 * N)
    a = a * (S/ a.sum())
    # Add tiny random noise
    a += 0.001 * S * np.random.rand(N)/N
    a = _project_onto_simplex(a, S)

    best_seq = a.copy()
    conv = _fft_convolve(a)
    best_score = conv.max()
    print("[Init] length", N, "score", best_score)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # gradient phase parameters
    max_iter = 30000
    it=0
    while time.time() < deadline and it < max_iter:
        # compute conv
        conv = _fft_convolve(a)
        max_conv = conv.max()
        # if best found improvement
        if max_conv < best_score:
            best_score = max_conv
            best_seq = a.copy()
            print("[Iter {}] new best {:.6f}".format(it, best_score))
        # compute weight vector as "peak set"
        # choose epsilon = 0.01*max_conv or top K
        eps_thresh = max_conv * 0.01
        peak_idx = np.where(conv >= max_conv - eps_thresh)[0]
        # If too many pick top 10
        if peak_idx.size > 20:
            # pick top 20
            topk = np.argsort(conv)[-20:]
            peak_idx = topk
        w = np.zeros_like(conv)
        w[peak_idx] = 1.0/peak_idx.size
        # gradient
        grad = 2.0 * _fft_correlate(w, a)

        # Adam update
        it += 1
        lr = 0.4 * (1 - it / max_iter) + 0.02   # start 0.42, end 0.02
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # keep nonneg and project
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, S)
        a = a_new

    # Polishing phase: use random pair moves with simulated annealing
    polish_iters = 20000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        # temperature schedule decreasing
        sigma = 0.5 * (1 - it / polish_iters) + 0.01
        cand = _random_pair_move(best_seq, sigma)
        # compute conv and evaluate
        conv = _fft_convolve(cand)
        score = conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print("[Polish {}] new best {:.6f}".format(it, best_score))
        # occasionally try to escape using random restart: if no improvement after many steps, maybe random perturb.
        if it % 2000 == 0 and time.time() < deadline:
            # perturb best_seq slightly by random noise to escape
            prob = 0.1
            if np.random.rand() < prob:
                tmp = best_seq + 0.001 * S * np.random.randn(N) / N
                tmp = np.maximum(tmp, 0)
                tmp = _project_onto_simplex(tmp, S)
                best_seq = tmp

    # Return best_seq as list
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()

"""
Improved search for a non‑negative step function that minimizes the
evaluation score.

Algorithm outline
-----------------
1. Choose a problem size N (≈ 2·previous length, up to 30 000) and fix the
   ℓ1‑norm S = √(2·N).  The evaluation score after this scaling equals
   max(conv) where conv = a * a.

2. Initialise a from a few diverse candidates (uniform, random Dirichlet,
   sine‑squared, Gaussian and the provided `height_sequence_1`).  For each
   candidate we run a fast optimisation phase.

3. Optimisation phase – Adam accelerated projected gradient descent.
   * The gradient of max(conv) is approximated by the correlation of a
     weight vector w with the current a.
   * w is concentrated on the K biggest convolution entries (top‑K
     approximation of the soft‑max).  This yields a clean sub‑gradient.
   * After each Adam step we Euclidean‑project onto the non‑negative simplex
     {x≥0, Σx = S}.

4. Polishing phase – simulated‑annealing style random pair swaps.
   Each move transfers a small amount of mass between two indices while
   preserving the ℓ1‑norm.  If the move reduces max(conv) it is kept.

5. The best sequence seen over all restarts is returned.  Early exit is
   performed as soon as the score drops below the target 1.503.

All heavy work is performed with FFT‑based convolution / correlation, which
makes each iteration O(N log N) and fast enough to explore many thousands of
iterations within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho_arr = u - cssv / ind > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.where(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT‑based (full) convolution a*a, length 2·N‑1
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# FFT‑based correlation w (len 2·N‑1) with a (len N), valid mode → length N
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                     # should be 2·n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]

# ----------------------------------------------------------------------
# Random tiny mass transfer preserving the ℓ1‑norm.
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq

# ----------------------------------------------------------------------
# Initialisers for diverse starting points.
# ----------------------------------------------------------------------
def _init_uniform(N: int, S: float) -> np.ndarray:
    return np.full(N, S / N)

def _init_random(N: int, S: float) -> np.ndarray:
    x = np.random.rand(N)
    return x / x.sum() * S

def _init_sine(N: int, S: float) -> np.ndarray:
    x = np.sin(np.linspace(0, np.pi, N)) ** 2
    return x / x.sum() * S

def _init_gaussian(N: int, S: float) -> np.ndarray:
    i = np.arange(N)
    sigma = N / 6.0
    x = np.exp(-0.5 * ((i - N / 2) / sigma) ** 2)
    return x / x.sum() * S

def _init_from_base(N: int, S: float) -> np.ndarray:
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        # linear interpolation to required length
        x = np.linspace(0, len(base) - 1, N)
        base_resampled = np.interp(x, np.arange(len(base)), base)
        base_resampled = np.maximum(base_resampled, 0)
        if base_resampled.sum() == 0:
            raise ValueError("empty base")
        return base_resampled / base_resampled.sum() * S
    except Exception:
        return _init_random(N, S)

# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that minimises
    `evaluate_sequence`.  Returns the best sequence found before the time
    budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    try:
        base_len = len(height_sequence_1)  # type: ignore
    except Exception:
        base_len = 2000
    # Target length: about twice the previous length, capped at 30 000
    N = min(30000, max(10000, base_len * 2))
    S = math.sqrt(2.0 * N)          # enforced ℓ1‑norm (score = max(conv))

    # --------------------------------------------------------------
    # Prepare a list of initialisers
    # --------------------------------------------------------------
    init_funs = [
        _init_from_base,
        _init_uniform,
        _init_random,
        _init_sine,
        _init_gaussian,
    ]

    best_seq = None
    best_score = float("inf")
    target = 1.503                # stop early if we reach this level

    # --------------------------------------------------------------
    # Loop over initialisers (restart strategy)
    # --------------------------------------------------------------
    for init_idx, init_fun in enumerate(init_funs):
        if time.time() >= deadline:
            break
        # ------------------- initialise a -------------------
        a = init_fun(N, S)
        # tiny random jitter to break symmetry
        a += 1e-4 * S * np.random.rand(N) / N
        a = _project_onto_simplex(a, S)

        # ------------------------------------------------------
        # Adam parameters for projected gradient descent
        # ------------------------------------------------------
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # iteration budget for this restart (will adapt to remaining time)
        # we aim for ~15000 gradient steps per restart, but respect the deadline
        max_grad_iters = 15000
        it = 0

        while time.time() < deadline and it < max_grad_iters:
            # ----- compute convolution -----
            conv = _fft_convolve(a)
            max_conv = conv.max()

            # ----- record best -----
            if max_conv < best_score:
                best_score = float(max_conv)
                best_seq = a.copy()
                print(f"[Restart {init_idx}] iter {it:05d} new best = {best_score:.6f}")
                if best_score <= target:
                    print("Target reached – exiting early.")
                    return best_seq.tolist()

            # ----- build weight vector w (top‑K approximation) -----
            # K grows slowly with N, but never exceeds 30
            K = max(1, min(30, int(N * 0.0005)))
            # use argpartition for speed
            top_idx = np.argpartition(conv, -K)[-K:]
            w = np.zeros_like(conv)
            w[top_idx] = 1.0 / K

            # ----- gradient of max(conv) (sub‑gradient) -----
            grad = 2.0 * _fft_correlate(w, a)

            # ----- Adam update (learning rate schedule) -----
            it += 1
            # linear decay from ~0.4 to 0.02
            lr = 0.4 * (1.0 - it / max_grad_iters) + 0.02
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- projection onto the feasible simplex -----
            a_new = np.maximum(a_new, 0.0)
            a = _project_onto_simplex(a_new, S)

        # ------------------------------------------------------
        # Polishing phase – random pair swaps (simulated annealing)
        # ------------------------------------------------------
        polish_iters = 8000
        it = 0
        while time.time() < deadline and it < polish_iters:
            it += 1
            # linear cooling for the swap amplitude
            sigma = 0.5 * (1.0 - it / polish_iters) + 0.01
            cand = _random_pair_move(best_seq, sigma)
            conv = _fft_convolve(cand)
            score = conv.max()
            if score < best_score:
                best_score = float(score)
                best_seq = cand.copy()
                print(f"[Polish {init_idx}] iter {it:05d} new best = {best_score:.6f}")
                if best_score <= target:
                    print("Target reached during polishing – exiting early.")
                    return best_seq.tolist()

        # occasionally try a fresh random perturbation of the current best
        # to jump out of a possible local basin (only if time permits)
        if time.time() < deadline:
            perturb = best_seq + 1e-3 * S * np.random.randn(N) / N
            perturb = np.maximum(perturb, 0.0)
            perturb = _project_onto_simplex(perturb, S)
            # a quick check – keep if it improves
            conv = _fft_convolve(perturb)
            if conv.max() < best_score:
                best_score = float(conv.max())
                best_seq = perturb.copy()
                print(f"[Perturb] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but fallback to a uniform vector
        best_seq = _init_uniform(N, S)
    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
