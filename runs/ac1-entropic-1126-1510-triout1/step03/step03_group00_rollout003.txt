"""search algorithm for low evaluate_sequence score

The algorithm proceeds in three phases:
1. Coarse shape search – sample symmetric Beta distributions
   (including the arcsine case) for several lengths and keep the best.
2. Refine the best candidate(s) with a projected Adam optimiser
   that smooths the max‑convolution using a soft‑max temperature which
   is gradually increased.
3. If time permits, run a few random restarts from perturbed copies
   of the current best solution.

All iterates stay on the probability simplex (sum = 1, non‑negative)
by Euclidean projection after each update.
The returned list is a plain Python list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # Inspired by the algorithm of Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]

def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log∑ exp(α·b_k) w.r.t. a.
    For large α the max entry is used as a one‑hot sub‑gradient.
    """
    n = a.shape[0]
    # Sub‑gradient when α is huge – avoid overflow
    if alpha > 2000:
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g

    # Soft‑max weights
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution w * a_rev
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(alpha, alpha) distribution sampled at cell mid‑points."""
    # density ∝ x^{alpha-1} (1-x)^{alpha-1}
    i = np.arange(n, dtype=np.float64)
    # map cell centre to (0,1)
    x = (i + 0.5) / n
    # avoid underflow for very small alpha
    vals = np.exp((alpha - 1) * (np.log(x) + np.log(1.0 - x)))
    # for alpha < 1 the expression may underflow at exact 0 or 1,
    # but our x ∈ (0,1) so it is safe.
    vals_sum = vals.sum()
    if vals_sum == 0.0 or not np.isfinite(vals_sum):
        # fallback to uniform
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= vals_sum
    return vals

def _adam_optimize_one_length(init_a: np.ndarray,
                              n: int,
                              deadline: float):

    a = init_a.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr0 = 0.6

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for w (2n‑1) + a_rev

    t = 0
    while time.time() < deadline:
        t += 1
        # current convolution & score
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={t} score={best_score:.6f}")

        # gradient (soft‑max or sub‑gradient)
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + epsilon)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny random walk to escape plateaus
        if t % 4000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

The algorithm works in three stages:

1️⃣ **Coarse shape search** – sample symmetric Beta(alpha,alpha)
   distributions (including the arcsine case) for several lengths and
   keep the best one.

2️⃣ **Refinement** – run a projected Adam optimiser on the simplex.
   The optimiser smooths the max‑convolution by a soft‑max temperature
   that is gradually increased, then finishes with a true sub‑gradient.

3️⃣ **Random restarts** – while time permits, perturb the current best,

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
