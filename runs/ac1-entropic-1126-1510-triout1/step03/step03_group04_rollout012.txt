"""
Simulated‑annealing optimizer for the max–self‑convolution problem.

The algorithm works in three stages:

1. **Initialisation** – start from the known best construction (if any) or
   from a random point on the ℓ¹‑simplex Σa = √(2 n).

2. **Simulated annealing with incremental convolution updates** – repeatedly
   move a small amount of mass from one index to another.  The convolution
   (a*a) is updated in O(n) time using the exact algebraic formula
   (no FFT per move).  Moves are accepted if they lower the current
   maximum or with a Metropolis probability that depends on a slowly
   decreasing temperature.  The best solution seen is recorded.

3. **Finishing refinements** – a cheap random block‑search (piecewise
   constant patterns) and a final pair‑swap hill‑climb are performed while
   time permits.

The returned sequence is a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}.
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv_fft(vec: np.ndarray) -> float:
    """
    Compute max entry of the self‑convolution using a real FFT.
    """
    n = vec.shape[0]
    # next power of two ≥ 2n‑1
    n2 = 1 << (2 * n - 1).bit_length()
    f = np.fft.rfft(vec, n2)
    conv = np.fft.irfft(f * f, n2)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _apply_delta(conv: np.ndarray, a: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Incrementally update the convolution array when `delta` is moved
    from position i to position j (i != j).  `conv` and `a` are the
    current convolution and sequence *before* the move.
    The update is performed in‑place.
    """
    n = a.shape[0]
    # ordered‑pair contribution change  – 2 δ a shifted
    conv[i : i + n] -= 2.0 * delta * a
    conv[j : j + n] += 2.0 * delta * a

    # quadratic δ² terms
    idx2i = 2 * i
    if idx2i < conv.shape[0]:
        conv[idx2i] += delta * delta
    idx2j = 2 * j
    if idx2j < conv.shape[0]:
        conv[idx2j] += delta * delta
    idxij = i + j
    if idxij < conv.shape[0]:
        conv[idxij] -= 2.0 * delta * delta


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects a wall‑clock budget and returns the best sequence
    found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)   # Σa = √(2 n)

    # --------------------------------------------------------------
    # initialise a
    if have_ref:
        a = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a = a * (target_sum / max(a.sum(), 1e-12))
        a = _project_to_simplex(a, target_sum)
    else:
        a = np.random.rand(n)
        a = _project_to_simplex(a, target_sum)

    # initial convolution and best record
    conv = np.convolve(a, a)                     # O(n²) – acceptable once
    cur_max = float(conv.max())
    best_a = a.copy()
    best_score = cur_max
    print(f"[init] n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – simulated annealing with cheap O(n) updates
    temp = best_score * 0.5           # initial temperature
    temp_end = best_score * 1e-5      # final temperature
    decay = 0.99994                   # exponential cooling
    it = 0

    while time.time() < deadline - 30.0:   # keep ~30 s for refinements
        # pick a source index with non‑zero mass
        nz = np.flatnonzero(a > 1e-12)
        if nz.size == 0:
            break
        i = int(np.random.choice(nz))
        j = int(np.random.randint(0, n))
        if i == j:
            continue

        max_delta = a[i] * 0.5                # move at most 50 % of source mass
        if max_delta <= 0.0:
            continue
        delta = max_delta * np.random.rand()  # random step size

        # keep a copy of the old convolution (tiny array, cheap)
        conv_old = conv.copy()
        cur_max_before = float(conv.max())

        # apply the delta (conv updated in‑place)
        _apply_delta(conv, a, i, j, delta)

        # update the sequence itself
        a[i] -= delta
        a[j] += delta

        new_max = float(conv.max())

        # record global best
        if new_max < best_score:
            best_score = new_max
            best_a = a.copy()
            print(f"[{it:08d}] new best = {best_score:.6f}")

        # Metropolis acceptance
        accept = False
        if new_max <= cur_max_before:
            accept = True
        else:
            prob = math.exp(-(new_max - cur_max_before) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if not accept:
            # revert both sequence and convolution
            a[i] += delta
            a[j] -= delta
            conv[:] = conv_old

        # temperature schedule
        temp = max(temp * decay, temp_end)
        it += 1

    # --------------------------------------------------------------
    # Phase 2 – cheap random block search (piecewise‑constant patterns)
    block_trials = 5000
    for _ in range(block_trials):
        if time.time() >= deadline - 10.0:
            break
        nb = np.random.randint(2, 13)                     # 2‑12 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _max_conv_fft(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final pair‑swap hill climb (still uses full convolution)
    while time.time() < deadline:
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size == 0:
            break
        i = int(np.random.choice(nz))
        j = int(np.random.randint(0, n))
        if i == j:
            continue
        delta = best_a[i] * np.random.uniform(0.05, 0.35)
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        sc = _max_conv_fft(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[swap] new best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
