"""
Propose a step‑function sequence that minimizes the evaluation score.

Algorithm
---------
1.  Choose a problem size N (≈15k‑30k) and enforce the ℓ¹–norm
    S = sqrt(2·N) (the score equals max(conv) under this normalization).

2.  Run several *restarts*:
    * Initialise a random feasible vector on the simplex.
    * Optimise it with Adam on a smooth approximation of the
      max‑convolution objective:
          w_i ∝ exp(β·(conv_i – max(conv)))   (soft‑max)
          ∇ = 2·corr(w, a)   (FFT‑based correlation)
    * The step is taken with Adam‑style moments and then projected
      back onto the non‑negative simplex.
    * The best vector of the restart is kept.

3.  Polishing via simulated‑annealing style random pair moves:
    * Propose a small transfer of mass between two positions,
      preserving the ℓ¹‑norm.
    * Accept if the new max‑convolution is lower, otherwise accept
      with probability exp(-(Δ)/T) where T slowly cools.
    * Keep the overall best sequence.

The whole routine respects the 1000 s budget, uses only NumPy
(and the already‑imported `evaluate_sequence`) and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho_arr = u - cssv / ind > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.where(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (length 2n‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based valid correlation (w ⋆ a)  → length n
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                     # = 2n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length n (valid mode)


# ----------------------------------------------------------------------
# Random pairwise mass transfer (keeps ℓ¹‑norm exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # delta ∈ (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        # moving from j → i, cannot exceed what j holds
        delta = min(delta, seq[j])
    else:
        # moving from i → j
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Compute the objective value (max of the full convolution)
# ----------------------------------------------------------------------
def _max_conv(seq: np.ndarray) -> float:
    conv = _fft_convolve(seq)
    return float(conv.max())

        beta2 = 0.999
        eps = 1e-8

        for it in range(1, max_iters + 1):
            # Early‑exit if time is out
            if time.time() >= deadline:
                break

            # 1) convolution and soft‑max weights
            conv = _fft_convolve(a)
            max_c = conv.max()
            t_frac = it / max_iters
            beta = beta_start + (beta_end - beta_start) * t_frac
            lr = lr_start + (lr_end - lr_start) * t_frac

            # stable soft‑max
            exp_vals = np.exp(beta * (conv - max_c))
            w = exp_vals / exp_vals.sum()

            # 2) gradient = 2 * corr(w, a)
            grad = 2.0 * _fft_correlate(w, a)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            # Project back onto the simplex
            a_new = _project_onto_simplex(a_new, S)

            a = a_new

            # periodic evaluation
            if it % 200 == 0:
                cur_score = _max_conv(a)
                nonlocal best_seq, best_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(f"[Adam] it={it:05d}, new best={best_score:.6f}")

        return a, _max_conv(a)

    # --------------------------------------------------------------
    # Perform several random restarts.
    # --------------------------------------------------------------
    RESTARTS = 4
    ADAM_ITERS = 8000
    for r in range(RESTARTS):
        if time.time() >= deadline:
            break

        # Initialise: either perturb the known solution or random.
        if base is not None and r == 0:
            # use existing construction (padded / trimmed to N)
            a0 = np.zeros(N, dtype=float)
            L = min(N, base_len)
            a0[:L] = base[:L]
            a0 = _project_onto_simplex(a0, S)
            print("[Restart] Using provided height_sequence_1 as start.")
        else:
            # random Dirichlet‑like start (exponential variables)
            raw = -np.log(np.random.rand(N) + 1e-12)
            a0 = _project_onto_simplex(raw, S)

        # Run Adam optimisation
        print(f"[Restart {r+1}/{RESTARTS}] Starting Adam optimisation.")
        seq_opt, sc_opt = _adam_run(
            a0,
            max_iters=ADAM_ITERS,
            beta_start=0.5,
            beta_end=80.0,
            lr_start=0.6,
            lr_end=0.02,
        )
        # Update global best if needed
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = seq_opt.copy()
            print(f"[Restart {r+1}] Adam finished with best={best_score:.6f}")

    # --------------------------------------------------------------
    # Polishing phase – simulated annealing with random pair moves.
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform distribution
        best_seq = np.full(N, S / N, dtype=float)

    current_seq = best_seq.copy()
    cur_score = best_score

    # Annealing schedule parameters
    total_polish = 25000
    start_T = 0.5 * cur_score   # modest initial temperature
    end_T = 1e-4

    print("[Polish] Starting simulated‑annealing refinement.")
    for it in range(1, total_polish + 1):
        if time.time() >= deadline:
            break

        # Linear cooling for temperature and move magnitude
        T = start_T + (end_T - start_T) * (it / total_polish)
        sigma = 0.6 * (1.0 - it / total_polish) + 0.02

        cand = _random_pair_move(current_seq, sigma)
        cand_score = _max_conv(cand)

        if cand_score < cur_score:
            # improvement – accept unconditionally
            current_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[Polish {it:05d}] New best={best_score:.6f}")
        else:
            # Metropolis acceptance
            delta = cand_score - cur_score
            prob = math.exp(-delta / (T + 1e-12))
            if np.random.rand() < prob:
                current_seq = cand
                cur_score = cand_score

    # --------------------------------------------------------------
    # Return the best sequence found (as plain Python list).
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # safety clamp
    print(f"[Done] Best score achieved: {best_score:.6f}")
    return best_seq.tolist()
