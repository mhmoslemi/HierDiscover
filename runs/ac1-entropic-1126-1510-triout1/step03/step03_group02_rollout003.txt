"""
Propose a non‑negative step function that minimizes
    2·n·max(a*a) / (sum(a)²)

The algorithm works in three phases:

1) **Initial exploration** – generate a handful of simple parametric families
   (uniform, inverse‑sqrt, power‑law, exponential, sinusoidal, random blocks)
   and keep the best few candidates (exact convolution score).

2) **Linear‑programming refinement** – for the current best shape `f`,
   solve the LP
        maximize   Σ b_j
        subject to b ≥ 0,
                    (f * b)_k ≤ M   for all k,
   where M = max(f*f).  The solution `b` is scaled so that Σb = √(2n) and
   mixed with `f` (convex combination).  This step is repeated while time
   permits; each iteration guarantees the convolution bound does not increase
   and usually drives the peak down.

3) **Optional smooth gradient polish** – a few thousand iterations of the
   soft‑max gradient descent from the original code are run to escape tiny
   plateaus.

All steps respect the simplex constraint Σa = √(2n) and non‑negativity,
so the final list can be handed directly to `evaluate_sequence`.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution through FFT (real inputs)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    # keep only the non‑zero part (len = len(u)+len(v)-1)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Return a * a (full convolution)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray,
               target_sum: float,
               size_fft: int,
               deadline: float,
               max_lp_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based refinement described in the statement.
    Returns the best vector encountered and its evaluation score.
    """
    import cvxpy as cp

    n = f.shape[0]

    # CVXPY objects (constructed once, then warmed‑started)
    b = cp.Variable(n)
    f_param = cp.Parameter(n)
    M_param = cp.Parameter()
    conv_expr = cp.conv(f_param, b)            # length = 2n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(b)),
                      [b >= 0,
                       conv_expr <= M_param])

    best_vec = f.copy()
    # exact score uses the fixed normalization √(2n)
    best_score = 2.0 * n * _self_convolution(f, size_fft).max() / (target_sum ** 2)

    t = 0.5          # initial mixing weight
    for it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        # current bound M
        conv_f = _self_convolution(f, size_fft)
        M = conv_f.max()

        # solve LP
        f_param.value = f
        M_param.value = M
        try:
            prob.solve(solver='ECOS', warm_start=True, max_iters=2000, verbose=False)
        except Exception as e:
            print("[LP] solver exception:", e)
            break

        if prob.status not in ("optimal", "optimal_inaccurate"):
            print("[LP] status not optimal:", prob.status)
            break

        b_opt = b.value
        if b_opt is None:
            break
        b_opt = np.maximum(b_opt, 0.0)
        sum_b = b_opt.sum()
        if sum_b <= 1e-12:
            # nothing useful can be added
            break

        # scale to the required sum
        g = b_opt * (target_sum / sum_b)

        # convex update
        f_new = (1.0 - t) * f + t * g

        # evaluate
        conv_new = _self_convolution(f_new, size_fft)
        new_score = 2.0 * n * conv_new.max() / (target_sum ** 2)

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_vec = f_new.copy()
            print(f"[LP] iter {it} improved score = {best_score:.6f}")

        # accept the update only if the peak does not increase
        if conv_new.max() <= M:
            f = f_new
        else:
            # no progress – shrink mixing factor and try again
            t *= 0.5
            if t < 1e-3:
                break

        # slowly decay the mixing factor to fine‑tune later
        t = max(0.1, t * 0.95)

    return best_vec, best_score
