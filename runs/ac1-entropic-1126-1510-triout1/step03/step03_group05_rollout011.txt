   np.random.seed(seed)
   start = time.time()
   deadline = start + budget_s - 0.5

   # Determine n
   try: n = len(height_sequence_1)
   except: n = 3000  # use moderate size
   target_sum = math.sqrt(2.0 * n)
   size_fft = 1 << ((3*n-2).bit_length())
   # functions defined: projection, self_convolution_fft, smooth_max_and_grad

   best_a = None
   best_score = float('inf')

   # Sample initial candidates
   init_candidates = []

   # Uniform
   a = np.ones(n) / n * target_sum
   init_candidates.append(a.copy())

   # Inverse sqrt shape
   for beta in np.linspace(0.1,5,5):
       a = 1.0/np.sqrt(np.arange(n)+beta)
       a = _project_to_simplex(a, target_sum)
       init_candidates.append(a)

   # Double-ended power law
   for p in [0.5,1,1.5,2]:
       a = (1.0/(np.arange(n)+1)**p + 1.0/(np.arange(n)[::-1]+1)**p)
       a = _project_to_simplex(a, target_sum)
       init_candidates.append(a)

   # Random block candidates (20)
   for _ in range(20):

       cuts = np.sort(np.random.choice(n-1, nb-1, replace=False)+1)
       blocks = np.split(np.arange(n), cuts)
       vals = np.random.rand(nb)+0.1
       cand = np.zeros(n)
       for b,v in zip(blocks, vals):
           cand[b] = v
       cand = _project_to_simplex(cand, target_sum)
       init_candidates.append(cand)

   # Evaluate each init candidate and track best
   for a in init_candidates:
       sc = evaluate_sequence(a.tolist())
       if sc < best_score:
           best_score = sc
           best_a = a.copy()
   # Print best after init

   # For each restart (maybe 6) from top k candidates
   restarts = [best_a] + [candidate for candidate in init_candidates[:5]]

       a = _project_to_simplex(a, target_sum)
       # Adam states

       v = np.zeros_like(a)
       beta1, beta2 = 0.9, 0.999

       lr0 = 0.15

       tau_end = 1e-3

"""
Optimise a non‑negative step function to minimise
evaluate_sequence(sequence).

Strategy
--------
1. **Initial sampling** – generate a diverse set of families
   (uniform, exponential, power‑law, inverse‑sqrt, random
   monotone, random block).  Each candidate is projected onto the
   simplex Σa = √(2 n) and the best is kept.

2. **Smooth max Adam descent** – minimise the true objective
   2 n·max(a∗a)/(∑a)².  The “max” is approximated by a
   temperature‑controlled soft‑max:
        w_k ∝ exp( ( (a∗a)_k – max(a∗a) ) / τ )
   The gradient of the smoothed objective is
        ∇ = 2·( w_rev ∗ a )[2 n‑2−i] .
   Adam with bias‑correction updates the parameters, followed by a
   Euclidean projection onto the simplex.  τ is annealed from a
   large value (≈10) to a tiny one (≈10⁻³) so the optimisation first
   sees a smooth landscape and later concentrates on the true
   maximum.

3. **Random‑block refinement** – after the main phase a handful of
   fresh block‑structured candidates are evaluated; the best one
   replaces the current solution.

The whole routine respects the 1000 s budget, prints intermediate
scores and finally returns the best sequence found (as plain
Python list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all zero, put the whole mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1) via FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    n = a.shape[0]
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w_rev: np.ndarray,
                          size_fft: int) -> np.ndarray:
    """
    Compute ∇ = 2·(w_rev ∗ a)[2 n‑2‑i]  (i = 0…n‑1)
    where w_rev is the reversed soft‑max weight vector.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, n=size_fft)
    fw = np.fft.rfft(w_rev, n=size_fft)
    conv_rev = np.fft.irfft(fa * fw, n=size_fft)
    # keep only the first 3·n‑2 entries (exact linear convolution length)
    conv_rev = conv_rev[: 3 * n - 2]
    idx = 2 * n - 2 - np.arange(n)          # indices for the gradient slice
    grad = 2.0 * conv_rev[idx]
    return grad


# ----------------------------------------------------------------------
def _evaluate_score(a: np.ndarray, n: int) -> float:
    """
    Return the exact evaluate_sequence score for a (assumes a is non‑negative).
    The formula is 2·n·max(a∗a) / (Σa)².
    """
    size_fft = 1 << ((3 * n - 2).bit_length())
    conv = _self_convolution_fft(a, size_fft)
    max_val = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_val / (s * s)


# ----------------------------------------------------------------------
def _initial_candidates(n: int, target_sum: float):
    """Yield a collection of diverse initial vectors (projected onto the simplex)."""
    # uniform
    a = np.ones(n, dtype=float)
    yield _project_to_simplex(a, target_sum)

    # exponential decay (single‑ended)
    for decay in np.linspace(5e-4, 1.5e-2, 5):
        a = np.exp(-decay * np.arange(n, dtype=float))
        yield _project_to_simplex(a, target_sum)

    # inverse‑sqrt (single‑ended)
    for beta in np.linspace(0.1, 5.0, 5):
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        yield _project_to_simplex(a, target_sum)

    # power‑law (single‑ended)
    for p in np.linspace(0.3, 2.5, 5):
        beta = np.random.rand() * 3.0
        a = 1.0 / (np.arange(n, dtype=float) + beta + 1.0) ** p
        yield _project_to_simplex(a, target_sum)

    # monotone random (sorted decreasing)
    for _ in range(5):
        a = np.random.rand(n)
        a.sort()
        a = a[::-1]                     # decreasing
        yield _project_to_simplex(a, target_sum)

    # random block (few large constant blocks)
    for _ in range(30):
        nb = np.random.randint(2, 7)      # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        yield _project_to_simplex(cand, target_sum)

    """Search for a low‑score non‑negative step function within the time budget."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    # ------------------------------------------------------------------
    # problem size (feel free to override the fallback)
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]

    except Exception:
        n = 4000                                    # a moderately large size
        have_ref = False

    target_sum = math.sqrt(2.0 * n)                 # normalisation Σa = √(2 n)
    size_fft = 1 << ((3 * n - 2).bit_length())     # power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # keep track of the best solution seen so far
    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    init_pool = list(_initial_candidates(n, target_sum))

    print(f"[init] best score after sampling = {best_score:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
