def _targeted_move(seq, sigma):
    n = len(seq)
    conv = _fft_convolve(seq)
    k = conv.argmax()
    # Overlap indices: i from max(0, k-(n-1)) to min(n-1, k)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_range = np.arange(i_start, i_end + 1)
    j_range = k - i_range
    # product contributions
    prod = seq[i_range] * seq[j_range]
    # weight for selecting i: product; if both i and j same, product = a_i^2.
    probs = prod.copy()
    if probs.sum() == 0:
        # fallback random
        return _random_pair_move(seq, sigma)
    probs /= probs.sum()
    # sample i according to probs
    i = np.random.choice(i_range, p=probs)
    # also get corresponding j
    j = k - i
    # Choose to move mass from i (or j) to random index l not overlapping
    # Determine which has larger value maybe
    if seq[i] > seq[j]:
        src = i
    else:
        src = j
    # choose destination random
    dst = np.random.choice(n)
    # move delta
    delta = min(seq[src], sigma) * np.random.rand()
    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta

"""
Propose a candidate step‐function with a low evaluation score.

The algorithm works in three stages:

1. **Initialisation** – start from the previously discovered best sequence
   (`height_sequence_1`) if it exists, otherwise from a random point on the
   ℓ₁‑simplex.  The ℓ₁‑norm is fixed to S = √(2·N) so that the evaluation
   function reduces to max(conv(a,a)).

2. **Projected Adam optimisation** – minimise a smooth approximation of
   max(conv) using a soft‑max weighting (β is annealed from 0.5 to 200)
   and the Adam optimiser on the simplex.  After 60 % of the iterations the
   smooth approximation is replaced by the true sub‑gradient of the
   max‑convolution (i.e. the gradient of the single most violating
   convolution entry).  All updates are projected back onto the non‑negative
   simplex.

3. **Simulated‑annealing polishing** – perform many cheap “mass‑swap”
   moves that keep the ℓ₁‑norm exact.  Two move types are used:
   *random pair move* (the same as in the reference code) and a
   *targeted move* that steals a little mass from the indices that
   directly contribute to the current maximal convolution term.
   Moves are accepted with a temperature‑controlled Metropolis rule;
   the best ever sequence is kept.

The whole routine runs until the supplied time budget (default 1000 s)
is exhausted and finally returns the best sequence found as a plain
Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based full convolution a*a (length 2n‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Random pairwise mass swap (preserves the ℓ₁‑norm exactly)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a small amount of mass from one index to another."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # draw delta from (-σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Targeted move that reduces the current maximum convolution entry.
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """
    Identify the convolution index k that attains the maximum,
    then move a little mass from one of the two operands participating
    in that term to a random location.
    """
    n = seq.shape[0]
    conv = _fft_convolve(seq)
    k = int(conv.argmax())

    # Indices that contribute to conv[k]: i + j = k, 0≤i,j<n
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i_vals = np.arange(i_start, i_end + 1, dtype=int)
    j_vals = k - i_vals

    # Contribution of each pair (i,j)
    contrib = seq[i_vals] * seq[j_vals]

    # Pick a source index proportionally to its contribution
    total = contrib.sum()
    if total == 0.0:
        return _random_pair_move(seq, sigma)

    probs = contrib / total
    src_i = np.random.choice(i_vals, p=probs)
    src_j = k - src_i

    # Choose which side to take mass from (the larger one)
    src = src_i if seq[src_i] >= seq[src_j] else src_j

    # Destination is any index (could be the same as src – then nothing happens)
    dst = np.random.randint(0, n)

    # Amount to transfer – bounded by sigma and source mass
    delta = min(seq[src], sigma) * np.random.rand()
    if delta <= 0.0:
        return seq.copy()

    new_seq = seq.copy()
    new_seq[src] -= delta
    new_seq[dst] += delta
    return new_seq


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (soft‑max + sub‑gradient phases)
# ----------------------------------------------------------------------
def _adam_optimize(a: np.ndarray,
                   target_sum: float,
                   deadline: float,
                   max_iters: int = 60000) -> tuple[np.ndarray, float]:
    """
    Run projected Adam on the sequence a.
    Returns (best_sequence, best_score).
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _fft_convolve(a).max()

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Schedules
    beta_start, beta_end = 0.5, 200.0   # soft‑max temperature
    lr_start, lr_end = 0.5, 0.01       # learning rate

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution -----
        conv = _fft_convolve(a)
        max_conv = conv.max()
        # ----- decide gradient type (soft‑max or true sub‑gradient) -----
        if it <= int(0.6 * max_iters):
            # Soft‑max (smooth approximation of max)
            frac = it / max_iters
            beta = beta_start + (beta_end - beta_start) * frac
            exp_vals = np.exp(beta * (conv - max_conv))
            w = exp_vals / exp_vals.sum()
            # Correlation w * a (valid mode) via FFT
            # w length = 2n-1, a length = n
            # correlation = conv(w, reverse(a)) at offsets n-1..2n-2 --> length n
            a_rev = a[::-1]
            nfft = 1 << ((len(w) + n - 1).bit_length())
            W = np.fft.rfft(w, nfft)
            A_rev = np.fft.rfft(a_rev, nfft)
            full = np.fft.irfft(W * A_rev, nfft)[: len(w) + n - 1]
            start = n - 1
            corr = full[start:start + n]
            grad = 2.0 * corr
        else:
            # Exact sub‑gradient of the max‑convolution entry
            k = int(conv.argmax())
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_vals = np.arange(i_start, i_end + 1, dtype=int)
            j_vals = k - i_vals
            grad = np.zeros_like(a)
            grad[i_vals] = 2.0 * a[j_vals]

        # ----- Adam update -----
        lr = lr_start + (lr_end - lr_start) * (it / max_iters)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.clip(a_new, 0.0, None)
        a_new = _project_onto_simplex(a_new, target_sum)

        a = a_new

        # ----- keep the best ever -----
        if it % 20 == 0:
            cur_score = _fft_convolve(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Adam it {it:05d}] new best score {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (random & targeted moves)
# ----------------------------------------------------------------------
def _polish_sequence(seq: np.ndarray,
                     target_sum: float,
                     deadline: float,
                     max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Greedy + Metropolis local search based on mass‑swap moves.
    Returns (best_sequence, best_score).
    """
    n = seq.shape[0]
    cur_seq = seq.copy()
    cur_score = _fft_convolve(cur_seq).max()
    best_seq = cur_seq.copy()
    best_score = cur_score

    # cooling schedule: start relatively hot, end cold
    T_start = max(0.1 * cur_score, 0.01)
    T_end = 0.0005

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        frac = it / max_iters
        T = T_start + (T_end - T_start) * frac

        # choose move type
        if np.random.rand() < 0.3:
            sigma = 0.4 * (1 - frac) + 0.02
            cand = _targeted_move(cur_seq, sigma)
        else:
            sigma = 0.4 * (1 - frac) + 0.02
            cand = _random_pair_move(cur_seq, sigma)

        cand_score = _fft_convolve(cand).max()
        delta = cand_score - cur_score

        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            # accept
            cur_seq = cand
            cur_score = cand_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[Polish it {it:05d}] new best score {best_score:.6f}")

    # final projection to be safe
    best_seq = _project_onto_simplex(np.maximum(best_seq, 0.0), target_sum)
    return best_seq, best_score
