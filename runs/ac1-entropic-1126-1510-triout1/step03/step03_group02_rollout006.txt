"""
Propose a step‑function sequence with a low self‑convolution peak.

Algorithm
---------
1. **Exploratory parametric sweep** – generate a diverse set of
   symmetric/non‑symmetric families (inverse‑sqrt, power‑law,
   exponential, cosine, sine, double‑ended mixtures, random block)
   and keep the K best raw candidates (according to the exact
   evaluation score).

2. **Intensive refinement** – starting from the best raw candidate,
   minimise a smooth approximation of the max‑convolution using the
   Adam optimiser, a soft‑max (log‑sum‑exp) weighting of the
   convolution entries and a projection onto the simplex
   {a ≥ 0, Σa = √(2 n)}.  The temperature of the soft‑max is
   gradually cooled and occasional random kicks help escape plateaus.
   Symmetrisation of the sequence is performed each iteration because
   the best solutions observed so far are symmetric.

3. **Secondary refinements** – if time permits, a few of the next
   best raw candidates are refined for a shorter budget.

4. **Final random‑block sweep** – a quick pass of symmetric random‑block
   candidates is performed with the remaining time.

All operations respect the security checks performed later by
`evaluate_sequence`.  The function returns the best sequence it has
found before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    fu = np.fft.rfft(u, n=fft_len)
    fv = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(fu * fv, n=fft_len)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, fft_len)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a (weights w sum to 1).
    Exact expression: 2·(w ↔ a_rev) restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, fft_len)          # length = 3·n‑2
    start = len(w) - len(a)                      # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, T: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights focusing on large convolution entries."""
    shifted = conv - conv.max()          # ≤ 0
    exp_arg = shifted / max(T, 1e-12)
    exp_arg = np.clip(exp_arg, -50, 0)   # protect from overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce exact mirror symmetry."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
def _refine_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    target_sum: float,
    lr0: float = None,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    temp_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based minimisation of the smoothed max‑convolution.
    Returns the best sequence encountered and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    # exact score (no security checks)
    conv = _self_convolution(a, fft_len)
    best_score = 2.0 * n * conv.max() / (a.sum() ** 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # initialise temperature from current spread
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    # learning‑rate normalisation (if not supplied)
    if lr0 is None:
        lr0 = 0.2 * target_sum / n   # a modest step size

    no_improve = 0
    it = 0
    while time.time() < deadline:
        # ---- current objective ----
        conv = _self_convolution(a, fft_len)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[adam] it {it:,} new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ---- soft‑max weights ----
        w = _softmax_weights(conv, T)

        # ---- gradient of smoothed objective ----
        grad = _weighted_gradient(a, w, fft_len)

        # ---- Adam update ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # ---- enforce symmetry & feasibility ----
        a_new = _symmetrize(a_new)
        a_new = _project_to_simplex(a_new, target_sum)

        # ---- occasional random kick to escape plateaus ----
        if (it % 5000 == 0 and it > 0) or no_improve > 5000:
            noise = np.random.randn(n) * target_sum * 1e-3
            a_new = a_new + noise
            a_new = _symmetrize(a_new)
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[adam] it {it:,} random perturbation")

        a = a_new
        T *= temp_decay               # cool temperature
        it += 1

    return best_a, best_score
