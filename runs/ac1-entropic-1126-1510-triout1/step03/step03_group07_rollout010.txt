"""
Simple multi‑restart sub‑gradient optimizer for the step‑function problem.

The evaluation metric is invariant under scaling, so we fix the ℓ1‑norm
∑a_i = √(2·N).  For a given sequence a the objective equals the
maximum entry of the (full) auto‑convolution of a.  The optimizer works
as follows:

* Choose a problem size N (larger N gives more degrees of freedom).
* Initialise a with one of several simple seeds (uniform, decreasing
  linear, random Dirichlet) and scale it onto the ℓ1‑simplex.
* Run a projected sub‑gradient descent on the function  max(conv(a)).
  The sub‑gradient of max(conv) at a point where the maximum occurs at
  index k is   g_i = 2·a_{k-i}  (if the index is valid).  When several
  indices share the maximal value we average their contributions.
* Keep the best sequence seen during the descent.
* After the gradient phase a short random‑pair mass‑swap polishing
  (simulated‑annealing style) is performed.
* The whole procedure is repeated with fresh random restarts until the
  time budget expires.  The best sequence across all restarts is
  returned.

The implementation uses only NumPy and the provided `evaluate_sequence`
function (called implicitly by the scoring harness) and respects the
1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: project a vector onto the non‑negative simplex with prescribed sum.
# Used only for the very first scaling; later we simply renormalise.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (real‑valued).
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) at the current a.
# Returns (max_value, gradient_vector, list_of_active_indices)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> tuple:
    """Compute sub‑gradient of max(conv) w.r.t. a."""
    N = a.shape[0]
    max_val = conv.max()
    # treat all indices whose value is within a tiny tolerance as active
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (N - 1))
        high = min(N - 1, k)
        i = np.arange(low, high + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    grad /= max(1, len(active))  # average if several active indices
    return max_val, grad, active


# ----------------------------------------------------------------------
# Simple random pairwise mass‑swap (preserves ℓ1‑norm exactly).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Multi‑restart sub‑gradient optimiser for the step‑function attack.
    Returns a list of non‑negative floats (the candidate sequence).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a 1‑second safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 1.0)

    # --------------------------------------------------------------
    # Determine a sensible problem size.
    # --------------------------------------------------------------
    base_len = 0
    if "height_sequence_1" in globals():
        try:
            base_len = len(height_sequence_1)  # type: ignore
        except Exception:
            base_len = 0
    # we aim for at least 8 k variables and preferably twice the base length
    N_min = max(8000, base_len * 2)
    N_max = 40000                     # hard cap – still cheap for FFT
    # start with a moderately large size; later restarts may vary it
    N = int(min(N_max, max(N_min, base_len * 2 if base_len else 12000)))

    # --------------------------------------------------------------
    # Best‑so‑far bookkeeping.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Helper: scale a vector onto the ℓ1‑simplex of sum S.
    # --------------------------------------------------------------
    def _renormalise(v, S):
        v = np.maximum(v, 0.0)
        total = v.sum()
        if total > 0:
            v *= (S / total)
        else:
            v[:] = S / len(v)
        return v

    # --------------------------------------------------------------
    # Main optimisation loop – many independent restarts.
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline - 5.0:
        restart += 1

        # ---- possibly vary N a bit (log‑uniform) to explore sizes ----
        # keep N within the allowed bounds
        factor = np.random.rand() ** 2          # bias towards smaller factor
        N_cur = int(N_min + factor * (N_max - N_min))
        S = math.sqrt(2.0 * N_cur)               # target ℓ1‑norm

        # ---- initialise a ------------------------------------------------
        init_mode = np.random.choice(["uniform", "linear", "random"])
        if init_mode == "uniform":
            a = np.full(N_cur, S / N_cur, dtype=float)
        elif init_mode == "linear":
            # decreasing linear profile (N_cur … 1)
            idx = np.arange(N_cur, dtype=float)
            a = (N_cur - idx)                     # values from N_cur‑1 down to 0
            a = _renormalise(a, S)
        else:   # random Dirichlet‑like start
            a = np.random.rand(N_cur)
            a = _renormalise(a, S)

        # ----------------------------------------------------------
        # Sub‑gradient descent phase
        # ----------------------------------------------------------
        max_iters = 3000
        # learning‑rate schedule (slow decrease)
        lr_start = 0.25
        lr_end = 0.015

        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # 1) compute full convolution and its maximum
            conv = _fft_convolve(a)
            cur_score, grad, _ = _subgradient_max_conv(a, conv)

            # 2) keep global best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[Restart {restart:02d}] New best {best_score:.6f} "
                    f"(N={N_cur}, it={it})"
                )

            # 3) step size (linear annealing)
            lr = lr_start + (lr_end - lr_start) * (it / max_iters)

            # 4) projected sub‑gradient step
            a = a - lr * grad
            a = _renormalise(a, S)

        # ----------------------------------------------------------
        # Polishing phase – random pairwise mass swaps
        # ----------------------------------------------------------
        polish_iters = 2500
        for it in range(polish_iters):
            if time.time() >= deadline:
                break
            # decreasing swap amplitude (simple linear cooling)
            sigma = 0.3 * (1.0 - it / polish_iters) + 0.02
            cand = _random_pair_move(best_seq if best_seq is not None else a, sigma)
            cand = _renormalise(cand, S)          # keep on the simplex
            conv = _fft_convolve(cand)
            cand_score = conv.max()
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(
                    f"[Polish {restart:02d}] New best {best_score:.6f} "
                    f"(N={N_cur}, it={it})"
                )
                # continue polishing from the improved candidate
                a = cand.copy()

        # End of this restart – loop again if time permits
        # ----------------------------------------------------------

    # --------------------------------------------------------------
    # Fallback in the extremely unlikely event that nothing was found.
    # --------------------------------------------------------------
    if best_seq is None:
        # Uniform fallback with a modest size
        N_fallback = max(8000, base_len * 2)
        S_fallback = math.sqrt(2.0 * N_fallback)
        best_seq = np.full(N_fallback, S_fallback / N_fallback, dtype=float)

    # Ensure non‑negativity before returning (as required by the scorer)
    best_seq = np.maximum(best_seq, 0.0)

    return best_seq.tolist()
