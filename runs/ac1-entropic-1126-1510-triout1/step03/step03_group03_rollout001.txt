"""
Propose a low‑score step‑function using a Beta(α,α) (U‑shaped) density.
The continuous arcsine law (α=0.5) makes the sum of two i.i.d. draws
uniform on [0,2]; after discretisation this yields a convolution that is
almost flat, i.e. max ≈ average ≈ 2 n/(2 n‑1) ≈ 1.  The routine tries a few
α‑values around 0.5, keeps the best, and then performs cheap random
perturbations with simplex projection.  All operations stay within the
1000 s budget and avoid any filesystem or network access.
"""

import math
import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_density(alpha: float, n: int) -> np.ndarray:
    """
    Returns a probability vector proportional to the Beta(α,α) density
    on the interval (0,1) using the centre of each of the n bins.
    """
    # mid‑points avoid the singularities at 0 and 1
    i = np.arange(n, dtype=float)
    x = (i + 0.5) / n            # 0 < x < 1
    # raw weight = x^{α‑1} * (1‑x)^{α‑1}
    # using logarithms prevents under/overflow for extreme α
    if alpha == 1.0:
        w = np.ones_like(x)
    else:
        w = np.exp((alpha - 1.0) * (np.log(x) + np.log(1.0 - x)))
    # normalise to a probability vector
    p = w / w.sum()
    return p


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a very low evaluation score.
    The core idea is to use a discretised arcsine (Beta(0.5,0.5))
    distribution whose self‑convolution is (almost) uniform.
    A small local random‑perturbation phase is added for robustness.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # slight safety margin

    # ------------------------------------------------------------------
    # problem size: reuse the reference length if available
    ref = globals().get("height_sequence_1")
    if isinstance(ref, (list, tuple)):
        n = len(ref)
    else:
        n = 3000                     # reasonable default

    S = math.sqrt(2.0 * n)           # target ℓ¹‑norm (makes score = max conv)

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # try a handful of α‑values around the arcsine law
    alpha_candidates = [0.5, 0.45, 0.55, 0.4, 0.6, 0.35, 0.65, 0.3, 0.7]
    for α in alpha_candidates:
        if time.time() > deadline:
            break
        p = _beta_density(α, n)          # probability vector (∑p = 1)
        a = S * p                        # scale to the required ℓ¹‑norm
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[alpha] α={α:.3f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # if the β‑family failed (should not happen) fall back to uniform
    if best_a is None:
        best_a = np.full(n, S / n, dtype=float)
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform candidate, score =", best_score)

    # ------------------------------------------------------------------
    # cheap random perturbation phase: Gaussian noise + projection
    it = 0
    while time.time() < deadline and it < 2000:
        # Gaussian perturbation; scale relative to typical entry size
        noise_scale = 0.02 * S
        cand = best_a + np.random.randn(n) * noise_scale
        cand = _project_to_simplex(cand, S)   # enforce non‑negativity & sum=S
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            print(f"[perturb] iter={it:04d} → score={score:.6f}")
        it += 1

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} (n = {n})")
    return best_a.tolist()
