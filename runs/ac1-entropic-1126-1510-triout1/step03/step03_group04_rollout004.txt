"""
Simulated annealing with incremental convolution updates.

The algorithm works on the simplex (nonnegative, fixed sum) by repeatedly
moving a random amount of mass from a relatively large component to a
random other component.  The change of the self‑convolution can be
updated in O(n) time without recomputing the whole convolution, which
allows millions of iterations in the 1000 s budget.  The temperature is
gradually cooled and occasional full recomputation of the convolution
keeps numerical errors under control.  The best sequence found is
returned.
"""

import time, math, random, numpy as np

def _compute_conv(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())
    f = np.fft.rfft(a, m)
    c = np.fft.irfft(f * f, m)
    return c[:2*n-1]

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Initial point

        a = ref.copy()

        a = a / a.sum() * target_sum

    conv = _compute_conv(a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_max = cur_max
    print(f"[init] n={n}, score={cur_max:.6f}")

    # Simulated annealing parameters
    T = cur_max * 0.2          # initial temperature
    cooling = 0.999985         # per‑iteration multiplicative cooling
    min_T = 1e-12
    iter_count = 0
    recompute_every = 5000

    # Pre‑compute total sum value for later possible renorm
    while time.time() < deadline - 5.0:
        # Pick source index i with probability proportional to current value
        # (to move mass from a large entry)
        probs = a / a.sum()

    print(f"[final] best score = {best_max:.6f}")
    # Return as plain Python list
    return best_a.tolist()

"""
Simulated‑annealing optimizer for the self‑convolution supremum.

The algorithm works on the simplex {a_i ≥ 0, Σa_i = √(2 n)}.
It repeatedly moves a random amount of mass from one index to another.
The effect of the move on the full self‑convolution is updated in O(n)
time by an exact additive formula, so millions of trials fit into the
1000 s budget.  A temperature that is slowly cooled accepts occasional
up‑hill moves, allowing the search to escape shallow minima.  Periodic
full recomputation of the convolution (FFT) keeps numerical drift
under control.  After the annealing phase a cheap block‑search and a
final tiny‑swap hill‑climb are performed to polish the result.

The returned list of heights is a plain Python list; the evaluation
score equals the maximum entry of the (scaled) self‑convolution.
"""

import time
import math
import random
import numpy as np


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Exact self‑convolution using a real FFT (length 2 n‑1)."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(a, m)
    c = np.fft.irfft(f * f, m)
    return c[: 2 * n - 1]


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step function with a low evaluation score."""
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.1      # safety margin

    # ------------------------------------------------------------------
    # Length of the problem (use previous construction if available)
    try:
        ref = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = len(ref)
        have_ref = True
    except Exception:        # no reference sequence present
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)               # we keep Σa = √(2 n)

    # ------------------------------------------------------------------
    # Initialise a feasible point on the simplex
    if have_ref:
        a = ref.copy()
        # Rescale to the required ℓ¹‑norm and project onto the simplex
        a = a * (target_sum / max(a.sum(), 1e-12))
        a = np.maximum(a, 0.0)
    else:
        a = np.random.rand(n)
        a = a / a.sum() * target_sum

    # ------------------------------------------------------------------
    # Initial convolution and best‑so‑far bookkeeping
    conv = _conv_self(a)
    cur_max = float(conv.max())
    best_a = a.copy()
    best_max = cur_max
    print(f"[init] n={n}, score={cur_max:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    T = cur_max * 0.20                # initial temperature
    cooling = 0.999985                # per‑iteration multiplicative cooling
    min_T = 1e-12
    iter_cnt = 0
    recompute_every = 5000           # full FFT recompute to avoid drift

    # ------------------------------------------------------------------
    # Main annealing loop
    while time.time() < deadline - 5.0:
        # --- pick a source index i (must have some mass) and a destination j ---
        i = random.randrange(n)
        if a[i] < 1e-12:
            continue
        j = random.randrange(n - 1)
        if j >= i:
            j += 1                      # guarantee j != i

        # --- how much mass to move -------------------------------------------------
        frac = random.uniform(0.05, 0.5)          # 5‑% … 50‑% of the source mass
        delta = a[i] * frac

        # --- trial convolution (exact incremental update) ---------------------------
        conv_trial = conv.copy()
        # contribution of the removed mass from position i
        conv_trial[i:i + n] -= 2.0 * delta * a
        # contribution of the added mass at position j
        conv_trial[j:j + n] += 2.0 * delta * a
        # quadratic correction term (Δ⋆Δ)
        conv_trial[2 * i] += delta * delta
        conv_trial[2 * j] += delta * delta
        conv_trial[i + j] -= 2.0 * delta * delta
