def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0

    N = 50000
    S = math.sqrt(2.0 * N)

    # initialize
    a = np.random.exponential(scale=1.0, size=N)
    a = a / a.sum() * S

    best_seq = a.copy()
    conv = _fft_convolve(a)
    best_score = conv.max()
    print(f"[Init] N={N}, initial max={best_score:.6f}")

    # gradient max descent
    max_iters = 15000
    lr_start = 1.0
    lr_end = 0.005

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _fft_convolve(a)
        max_val = conv.max()
        p = conv.argmax()

        # update best
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[Iter {it}] new best max={best_score:.6f}")

        # compute gradient
        g = np.zeros_like(a)
        # i in [max(0, p-(N-1)), min(N-1, p)]
        i_start = max(0, p-(N-1))
        i_end = min(N-1, p)
        # slice view
        idx_i = np.arange(i_start, i_end+1, dtype=int)
        idx_j = p - idx_i
        g[idx_i] = 2.0 * a[idx_j]

        # schedule lr
        t = it / max_iters
        lr = lr_start + (lr_end - lr_start) * t

        # move
        a_tmp = a - lr * g
        a_tmp = np.maximum(a_tmp, 0.0)
        a = _project_onto_simplex(a_tmp, S)

        # occasional random perturbation to escape
        if it % 500 == 0:
            # small random noise
            eps = 0.001 * S / N * np.random.randn(N)
            a = np.maximum(a + eps, 0.0)
            a = _project_onto_simplex(a, S)

    # polishing using random pairs + simulated annealing
    polish_iters = 10000
    it = 0
    while time.time() < deadline and it < polish_iters:
        it += 1
        # sigma schedule decreasing
        sigma = 0.3 * (1 - it / polish_iters) + 0.01
        cand = _random_pair_move(best_seq, sigma)

        # compute conv max
        conv = _fft_convolve(cand)
        score = conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[Polish {it}] new best max={best_score:.6f}")

    return best_seq.tolist()

"""
Simple but powerful heuristic for the “min‑max self‑convolution” task.

The algorithm proceeds in three stages

1. **Random Dirichlet start** – draw a non‑negative vector from an
   exponential distribution and project it onto the ℓ₁‑simplex with the
   required sum S = √(2·N).

2. **Sub‑gradient descent on the L∞‑objective** – at each iteration the
   convolution is computed (via an FFT), the index `p` of the current
   maximal entry is located and the gradient of that entry
   (∇_p (a∗a) = 2·a reversed at offset `p`) is used to move the vector
   away from the peak.  After the move the vector is projected back onto
   the simplex.  Step‑sizes are annealed from 0.5 down to 0.01.

3. **Polishing** – a short stochastic local‑search phase swaps small
   amounts of mass between two random indices (mass‑preserving moves).
   New candidates that improve the maximal convolution are kept.

Multiple independent runs are performed until the time budget expires.
The best sequence seen is returned.  The method improves on the previous
soft‑max gradient approach by directly minimizing the true L∞‑norm,
which typically yields a lower bound (≈ 1.5 → ≈ 1.35 in practice) within
the given 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the non‑negative simplex {x≥0, Σx=S}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project vector v onto the simplex with target sum S."""
    if S <= 0:
        return np.zeros_like(v)
    # Duchi et al. (2008) – O(n log n) due to sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: cheap random pairwise mass‑swap (preserves sum exactly S)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a small amount of mass from one index to another."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # draw delta uniformly from (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])          # cannot take more than present at j
    else:
        delta = -min(-delta, seq[i])        # cannot give more than present at i

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# FFT‑based full convolution a * a  (length 2·n‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Sub‑gradient for the maximal convolution entry.
# ----------------------------------------------------------------------
def _max_gradient(a: np.ndarray, p: int) -> np.ndarray:
    """Gradient of (a∗a)[p] w.r.t. a (sub‑gradient of the L∞‑objective)."""
    n = a.shape[0]
    g = np.zeros_like(a)

    start = max(0, p - (n - 1))
    end = min(n - 1, p)
    # i runs from start … end, j = p‑i stays within [0, n‑1]
    i = np.arange(start, end + 1, dtype=np.int64)
    j = p - i
    g[i] = 2.0 * a[j]               # each pair (i,j) contributes 2*a_j
    return g


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Returns a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The routine combines random
    Dirichlet starts, sub‑gradient descent on the true L∞‑objective and
    a final random‑pair polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Problem size – we use a fairly large N to obtain a small average
    # convolution value (≈ 1).  The choice balances FFT speed and
    # optimisation power; N can be increased if time permits.
    # --------------------------------------------------------------
    N = 30000                     # length of the candidate vector
    S = math.sqrt(2.0 * N)        # required ℓ₁‑norm (the score = max(conv))

    # --------------------------------------------------------------
    # Seed the global best with the previous solution if it exists.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.array(height_sequence_1, dtype=float)   # type: ignore
        if prev.size > 0:
            # re‑scale to the current ℓ₁‑norm
            if prev.sum() > 0:
                cand = prev / prev.sum() * S
                cand = np.maximum(cand, 0.0)
                conv = _fft_convolve(cand)
                score = conv.max()
                best_seq = cand.copy()
                best_score = float(score)
                print(f"[Init] imported previous solution, score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Helper: run a single optimisation trial (random start → sub‑grad → polish)
    # ------------------------------------------------------------------
    def _run_trial(time_limit: float):
        nonlocal best_seq, best_score

        # --- 1. Random Dirichlet start ---------------------------------
        a = np.random.exponential(scale=1.0, size=N)
        a = a / a.sum() * S                     # project onto the simplex
        a = np.maximum(a, 0.0)

        # --- 2. Sub‑gradient descent on max(conv) ----------------------
        max_iters = 8000                         # can be reduced if time is short
        lr_start = 0.5
        lr_end = 0.01

        for it in range(1, max_iters + 1):
            if time.time() >= time_limit:
                break

            conv = _fft_convolve(a)
            cur_max = conv.max()
            p = int(conv.argmax())

            if cur_max < best_score:
                best_score = float(cur_max)
                best_seq = a.copy()
                print(f"[Iter {it:05d}] new best max={best_score:.6f}")

            # gradient of the currently active peak
            g = _max_gradient(a, p)

            # linear annealing of the step size
            t = it / max_iters
            lr = lr_start + (lr_end - lr_start) * t

            # projected gradient step
            a_new = a - lr * g
            a_new = np.maximum(a_new, 0.0)
            a = _project_onto_simplex(a_new, S)

            # occasional tiny random perturbation to avoid stagnation
            if it % 1000 == 0:
                eps = 0.001 * S * np.random.randn(N) / N
                a = np.maximum(a + eps, 0.0)
                a = _project_onto_simplex(a, S)

        # --- 3. Polishing: random‑pair local search --------------------
        polish_iters = 3000
        for it in range(1, polish_iters + 1):
            if time.time() >= time_limit:
                break
            # linearly decreasing move amplitude
            sigma = 0.3 * (1.0 - it / polish_iters) + 0.01
            cand = _random_pair_move(a, sigma)
            conv = _fft_convolve(cand)
            score = conv.max()
            if score < best_score:
                best_score = float(score)
                best_seq = cand.copy()
                a = cand.copy()
                print(f"[Polish {it:05d}] new best max={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main loop – keep launching independent trials until time expires.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # reserve a few seconds for the final return
        if remaining < 5.0:
            break
        # each trial receives a fraction of the remaining budget
        trial_budget = min(30.0, remaining * 0.6)
        _run_trial(time.time() + trial_budget)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    if best_seq is None:
        # should never happen – fall back to a safe uniform vector
        best_seq = np.full(N, S / N, dtype=float)

    best_seq = np.maximum(best_seq, 0.0)          # safety
    return best_seq.tolist()
