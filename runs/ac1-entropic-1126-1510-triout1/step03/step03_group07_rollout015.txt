    for it in range(max_iters_gd):
         # compute conv and weight w
         conv = _fft_convolve(a)
         max_conv = conv.max()
         # Use beta schedule linearly increasing
         beta = 0.5 + (80-0.5) * it / max_iters_gd
         # Softmax
         # maintain numeric stability: exp(beta*(conv - max_conv)) may overflow if big
         exp_vals = np.exp(beta * (conv - max_conv))
         w = exp_vals / exp_vals.sum()
         # gradient = 2 * correlation

         # step size decreasing
         lr = 0.5 * (1 - it / max_iters_gd) + 0.01 * (it / max_iters_gd)

"""
Propose candidate step‐function via multi‑restart projected gradient + simulated annealing.

The algorithm works as follows:
1. Choose a problem size N (length of the sequence).  If the previous best
   construction (global variable ``height_sequence_1``) is available we reuse its
   length, otherwise we test a few different lengths.
2. Initialise a feasible point on the non‑negative simplex with ℓ₁‑norm
   S = √(2 N).  The initial point is either the previous construction (rescaled)
   or a random Dirichlet‑like vector.
3. Run a projected gradient descent on a smooth approximation of the
   max‑convolution objective.  The smoothness is introduced via a soft‑max
   weighting with an annealed temperature β.  After each gradient step the
   vector is projected onto the simplex by a simple renormalisation (clipping
   negatives and scaling to sum S).  The best iterate is remembered.
4. Polish the best iterate with a simulated‑annealing style random‑pair
   mass‑swap.  A move transfers a small amount of mass from one index to
   another while preserving the total sum.  Improvements are always accepted;
   uphill moves are accepted with probability exp(−Δ/T) where the temperature T
   is linearly cooled.
5. Several independent restarts are performed until the time budget expires.
   The globally best sequence found is returned as a plain list of floats.

The evaluation function in the competition is scale‑invariant, therefore we
fix the ℓ₁‑norm to S = √(2 N) and minimise max(conv) directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: clip negatives, then renormalise to exact ℓ₁‑norm S.
# ----------------------------------------------------------------------
def _clip_and_rescale(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative and scale to have Σv = target_sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        v = np.full_like(v, target_sum / v.size)
    return v


# ----------------------------------------------------------------------
# Helper: cheap random pairwise mass‑swap (preserves the ℓ₁‑norm).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a random amount of mass from one index to another (preserves sum)."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    # draw delta from (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    # keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# FFT‑based full convolution (length 2 N‑1).
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based correlation: corr = np.correlate(w, a, mode='valid')
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Correlation of length‑2N‑1 weight vector w with a (valid mode ⇒ length N)."""
    n = a.shape[0]
    a_rev = a[::-1]
    nfft = 1 << ((w.size + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: w.size + n - 1]
    start = n - 1
    return full[start : start + n]  # length N


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length N.
# ----------------------------------------------------------------------
def _optimise_fixed_length(N: int,
                           S: float,
                           init_seq: np.ndarray,
                           time_limit: float) -> tuple[np.ndarray, float]:
    """Projected gradient descent + simulated annealing for a given N."""
    a = init_seq.astype(float).copy()
    # ensure feasibility
    a = _clip_and_rescale(a, S)

    # keep track of best solution
    conv = _fft_convolve(a)
    best_score = conv.max()
    best_seq = a.copy()
    print(f"[Init] N={N}, initial max(conv)={best_score:.6f}")

    # ------------------------------------------------------------------
    # Gradient‑descent phase (soft‑max approximation of max).
    # ------------------------------------------------------------------
    max_iter = 20000                     # feel free to adjust
    it = 0
    while time.time() < time_limit and it < max_iter:
        it += 1
        conv = _fft_convolve(a)
        max_c = conv.max()

        # annealing schedules
        t_frac = it / max_iter
        beta = 0.5 + (150.0 - 0.5) * t_frac   # temperature for soft‑max
        lr = 0.5 * (1.0 - t_frac) + 0.01 * t_frac  # learning‑rate decay

        # soft‑max weights (stable)
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # gradient = 2 * corr(w, a)
        grad = 2.0 * _fft_correlate(w, a)

        # gradient step + projection onto simplex (clip & renormalise)
        a = a - lr * grad
        a = _clip_and_rescale(a, S)

        # occasional evaluation
        if it % 10 == 0:
            score = _fft_convolve(a).max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[GD] iter {it:05d} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing polishing phase.
    # ------------------------------------------------------------------
    # Temperature is a fraction of the current best value.
    T0 = max(best_score * 0.1, 1e-4)
    sa_iter = 0
    max_sa_iter = 15000
    while time.time() < time_limit and sa_iter < max_sa_iter:
        sa_iter += 1
        # linearly cooled move amplitude
        sigma = 0.4 * max(0.0, 1.0 - sa_iter / max_sa_iter) + 0.02
        cand = _random_pair_move(best_seq, sigma)
        cand_score = _fft_convolve(cand).max()

        # linearly cooled temperature
        T = T0 * max(0.0, 1.0 - sa_iter / max_sa_iter)

        # Metropolis acceptance
        if cand_score < best_score or np.random.rand() < math.exp(
                -(cand_score - best_score) / max(1e-12, T)):
            best_seq = cand
            best_score = cand_score
            print(f"[SA] iter {sa_iter:05d} new best {best_score:.6f}")

    return best_seq, best_score

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Multi‑restart optimisation of a non‑negative step function.
    Returns the best sequence (list of floats) found before the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # ------------------------------------------------------------------
    # Determine a few candidate lengths.
    # ------------------------------------------------------------------
    lengths = []
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        lengths.append(base.size)
    except Exception:
        base = None

    # some additional lengths to diversify the search
    for n in (8000, 12000, 16000, 20000, 24000):
        if n not in lengths:
            lengths.append(n)

    global_best_seq = None
    global_best_score = float("inf")
    restart_id = 0

    # ------------------------------------------------------------------
    # Perform independent restarts until time runs out.
    # ------------------------------------------------------------------
    for N in lengths:
        if time.time() >= deadline:
            break
        restart_id += 1
        S = math.sqrt(2.0 * N)

        # initialise sequence for this restart
        if restart_id == 1 and base is not None and base.size == N:
            init_seq = base.astype(float) * (S / base.sum())
        else:
            # Dirichlet‑like random start (exponential variables)
            rnd = np.random.rand(N)
            rnd = rnd / rnd.sum()
            init_seq = rnd * S

        # allocate a slice of the remaining budget for this run
        remaining = deadline - time.time()
        if remaining < 5.0:
            break
        # give this run at most 80 % of the remaining budget
        sub_deadline = time.time() + remaining * 0.80

        seq, score = _optimise_fixed_length(N, S, init_seq, sub_deadline)

        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(f"[Restart {restart_id}] N={N} new global best {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final quick polishing on the overall best sequence (if any time left).
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        while time.time() < deadline:
            sigma = 0.2
            cand = _random_pair_move(global_best_seq, sigma)
            cand_score = _fft_convolve(cand).max()
            if cand_score < global_best_score:
                global_best_seq = cand
                global_best_score = cand_score
                print(f"[Final polish] improved to {global_best_score:.6f}")

    # safeguard: ensure non‑negative output
    if global_best_seq is None:
        # fallback – uniform distribution
        N = lengths[0] if lengths else 1000
        S = math.sqrt(2.0 * N)
        global_best_seq = np.full(N, S / N, dtype=float)

    global_best_seq = np.maximum(global_best_seq, 0.0)
    return global_best_seq.tolist()
