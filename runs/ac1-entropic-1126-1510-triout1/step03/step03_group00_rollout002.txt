"""
propose_candidate
=================

Search for a non‑negative step‑function (list of heights) that minimises the
provided ``evaluate_sequence`` score.

Algorithm
---------
1. **Simplex representation** – the objective is homogeneous,
   therefore we keep all candidate vectors on the probability simplex:
   ``a_i ≥ 0`` and ``∑ a_i = 1``.  After every update we project back onto the
   simplex with the Euclidean projection.

2. **Progressive lengths** – we start with a modest length (1024) and
   gradually increase it up to 16 384 (never exceeding 20 000).  The best
   vector for a shorter length is interpolated to initialise the next,
   larger problem; this provides a good warm‑start.

3. **Smoothed max‑convolution** – the loss
   ``L(a) = 2·n·max(a*a)`` is non‑smooth.  We replace the hard maximum by a
   soft‑max with temperature ``α``:

   ``S_α(a) = (1/α)·log(∑_k exp(α·(a*a)_k))``

   For large ``α`` the soft‑max approaches the true max, while for small
   ``α`` the gradient is a smooth average.  The gradient of ``S_α`` w.r.t.
   ``a`` is
   ``∇S_α = 2· (w * a_rev)`` where ``w`` are the soft‑max weights and
   ``a_rev`` is ``a`` reversed.  Convolutions are performed with FFT.

4. **Adam optimiser** – we use the Adam update rule (first‑ and second‑moment
   adaptation) instead of plain momentum.  A diminishing learning‑rate
   factor is built‑in.  After each step we project onto the simplex.

5. **Annealing schedule** – ``α`` starts at 1.0 and is multiplied by
   ``α_mul = 1.02`` each iteration until a cap ``α_max = 50000``.  This
   lets the optimiser first move smoothly and later focus on the true
   maximum.

6. **Exploration** – every few thousand iterations a tiny Gaussian jitter
   is added (and the result projected) to help escape plateaus.
   For each length we run a few random restarts (different initial
   distributions) and keep the best result.

7. **Final local refinement** – after the main optimisation a short
   stochastic hill‑climb (random Gaussian perturbations + projection) is
   performed on the globally best vector to squeeze out any remaining
   improvement.

The whole routine respects the supplied time budget (default 1000 s),
printing intermediate scores so that progress can be observed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _semicircle_distribution(n: int) -> np.ndarray:
    """Discrete (scaled) semicircle distribution of length n."""
    i = np.arange(n, dtype=np.float64)
    # shifted to avoid zeros at the ends
    vals = np.sqrt((i + 0.5) * (n - i - 0.5))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str, fallback_seq: np.ndarray = None) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "semicircle":
        a = _semicircle_distribution(n)
    elif kind == "prev" and fallback_seq is not None:
        # interpolate previous best to length n
        old = fallback_seq
        old_idx = np.arange(len(old))
        new_idx = np.linspace(0, len(old) - 1, n)
        a = np.interp(new_idx, old_idx, old)
    else:   # default
        a = _arcsine_distribution(n)

    # ensure exact simplex membership
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    # next power of two at least 2·n‑1
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of the soft‑max smoothed max function.
    For large alpha the gradient reduces to the true sub‑gradient.
    """
    n = a.shape[0]

    # very large temperature → treat max entry as one‑hot
    if alpha > 20000.0:
        k = int(np.argmax(b))
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[idx[mask]]
        return g

    # soft‑max weights
    shift = b - b.max()               # for numerical stability
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution w * a_rev
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g

def _adam_optimize_one_length(init_a: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑based projected optimisation of the smoothed max‑convolution.
    Returns the best vector found (still on the simplex) and its
    internal score = 2·n·max(a*a) (i.e. evaluate_sequence result for sum=1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05               # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 50000.0

    # FFT size for the gradient step (must accommodate w of length 2n‑1)
    size_fft = 1 << ((2 * n - 1).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # evaluate current loss
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        # keep global best for this length
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient (soft‑max or sub‑gradient)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # increase temperature

            alpha = min(alpha_max, alpha * alpha_mul)

def _local_refinement(seq: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing on the simplex.

    re‑projects and accepts the move if it improves the loss.
    """

    b = _fft_self_convolve(a)

    best_seq = a.copy()

    while time.time() < deadline:

            best_score = cand_score
            best_seq = cand.copy()
            a = cand                 # move to the better point
            print(f"[{time.time():.2f}] refinement n={n} score={best_score:.6f}")

    return best_seq, best_score
