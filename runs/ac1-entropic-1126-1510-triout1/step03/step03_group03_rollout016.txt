"""
A fast, gradient‑free optimizer for the step‑function problem.

The algorithm works on the simplex (fixed ℓ¹‑norm) and improves a candidate
by repeatedly moving a random amount of mass from one coordinate to another.
The convolution is updated analytically in O(n) using the identity

    (a+Δe_j-Δe_i)*(a+Δe_j-Δe_i)
      = a*a  + 2Δ·(shift_j – shift_i)  + Δ²·(e_{2j}+e_{2i}–2e_{i+j})

so each trial only needs a few vectorised NumPy operations.
Accept/reject follows a simulated‑annealing rule, which lets the search
escape shallow plateaus.  After the stochastic phase a few projected‑gradient
steps (sub‑gradient of the max‑convolution) polish the result.

The routine respects the 1000 s budget, works with any provided
`height_sequence_1` (or falls back to a default size), and returns the
best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        {x ≥ 0, Σ x = z}.
    """
    # ensure non‑negative
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        # uniform distribution if everything is zero
        return np.full_like(v, z / v.shape[0])
    # classic O(n log n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score non‑negative step‑function within the time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (defaults to 1000).

    Returns
    -------
    list[float]
        The best sequence of heights discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
        ref_seq = np.asarray(height_sequence_1, dtype=float)
    except Exception:
        n = 3000
        have_ref = False
        ref_seq = None

    # we work with a fixed ℓ¹‑norm equal to 1 (scale‑invariant)
    target_sum = 1.0

    # ------------------------------------------------------------------
    # helper to compute the objective quickly
    def _score_from_conv(conv: np.ndarray, total: float) -> float:
        return 2.0 * n * conv.max() / (total * total)

    # ------------------------------------------------------------------
    # generate a few diverse starting points
    starts = []

    # uniform
    starts.append(np.full(n, target_sum / n))

    # V‑shaped (large at the ends, small in the centre)
    idx = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vshape = np.abs(idx - mid) + 1e-3
    vshape = vshape / vshape.sum() * target_sum
    starts.append(vshape)

    # a few random Dirichlet draws
    for _ in range(4):
        r = np.random.rand(n)
        starts.append(r / r.sum() * target_sum)

    # reference sequence, if available
    if have_ref:
        ref = np.maximum(ref_seq, 0.0)
        if ref.sum() == 0.0:
            ref = np.full_like(ref, target_sum / n)
        else:
            ref = ref / ref.sum() * target_sum
        starts.append(ref)

    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # main loop over initial candidates
    for start_id, a0 in enumerate(starts):
        if time.time() > deadline:
            break

        a = a0.copy()
        total = a.sum()                     # should be 1.0 (up to round‑off)
        conv = np.convolve(a, a)            # length 2·n‑1
        cur_score = _score_from_conv(conv, total)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[init {start_id}] initial score = {cur_score:.6f}")

        # --------------------------------------------------------------
        # stochastic mass‑transfer search
        step_factor = 0.5                    # relative size of a move
        min_step = 0.001
        step_decay = 0.9996                  # exponential cooling of step size
        temperature = cur_score * 0.05 + 1e-12
        temp_decay = 0.999

        it = 0
        while time.time() < deadline:
            it += 1

            # pick a source index with non‑zero mass
            i = np.random.randint(0, n)
            if a[i] <= 1e-14:
                continue

            # pick a different target index
            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1

            # amount of mass to move (≤ a[i])
            delta = a[i] * step_factor * np.random.rand()
            if delta <= 1e-15:
                continue

            # ---- propose a new state (copy‑on‑write) -----------------
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            conv_new = conv.copy()
            # linear part: 2·Δ·(shift_j – shift_i)
            conv_new[i:i + n] -= 2.0 * delta * a
            conv_new[j:j + n] += 2.0 * delta * a
            # quadratic correction
            conv_new[2 * i] += delta * delta
            conv_new[2 * j] += delta * delta
            conv_new[i + j] -= 2.0 * delta * delta

            new_score = _score_from_conv(conv_new, total)

            accept = False
            if new_score < cur_score:
                accept = True
            else:
                # Metropolis criterion
                prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
                if np.random.rand() < prob:
                    accept = True

            if accept:

                conv = conv_new
                cur_score = new_score

                    best_score = new_score
                    best_a = a.copy()

                temperature *= temp_decay
                step_factor = max(min_step, step_factor * step_decay)
            else:
                # even on rejection we slowly shrink the step size
                step_factor = max(min_step, step_factor * step_decay)

            # occasional full recomputation to avoid accumulation of rounding errors
            if it % 20000 == 0:
                conv = np.convolve(a, a)
                cur_score = _score_from_conv(conv, total)

        # --------------------------------------------------------------
        # final projected sub‑gradient polishing (as in the reference code)
        # ------------------------------------------------------------------
        # we start polishing from the best point found so far
        a_pg = best_a if best_a is not None else a.copy()
        total_pg = a_pg.sum()
        conv_pg = np.convolve(a_pg, a_pg)
        eta0 = (total_pg * 0.5) / n
        decay = 1e-4

        for sub_it in range(2000):
            if time.time() > deadline:
                break

            conv_pg = np.convolve(a_pg, a_pg)
            max_c = conv_pg.max()
            mask = conv_pg == max_c
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                continue
            w = np.zeros_like(conv_pg, dtype=float)
            w[mask] = 1.0 / cnt

            # sub‑gradient w.r.t. a (2·conv(w, a[::-1]) on the central window)
            conv_wr = np.convolve(w, a_pg[::-1])
            g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

            eta = eta0 / (1.0 + decay * sub_it)
            a_new_pg = a_pg - eta * g_a

            # project back onto the probability simplex
            a_new_pg = _project_to_simplex(a_new_pg, z=target_sum)

            new_score_pg = _score_from_conv(np.convolve(a_new_pg, a_new_pg), target_sum)

            if new_score_pg < best_score:
                best_score = new_score_pg
                best_a = a_new_pg.copy()
                print(f"[grad improve] new best = {best_score:.6f} (subit {sub_it})")
            a_pg = a_new_pg

    # ------------------------------------------------------------------
    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, target_sum / n)

    # safety clip (the evaluation routine caps at 1000 anyway)
    best_a = np.clip(best_a, 0.0, 1000.0)

    # final verification (optional – cheap because we already have conv)
    # final_score = evaluate_sequence(best_a.tolist())
    # print(f"[final] best score = {final_score:.6f}")

    return best_a.tolist()
