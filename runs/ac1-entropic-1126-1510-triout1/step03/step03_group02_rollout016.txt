def anneal(a, deadline, n, fft_size, target_sum):

    best = a.copy()
    best_score = quick_score(a)
    cur = a.copy()
    cur_score = best_score

    # initial temperature: maybe 0.1 * cur_score
    T0 = max(0.01, cur_score * 0.1)
    # cooling factor per iteration: say 0.999995 (slow)
    T = T0

    it = 0
    while time.time() < deadline:
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # epsilon fraction of a[i] or based on target_sum/n
        # choose epsilon uniformly small
        eps = np.random.rand() * target_sum * 0.01  # up to 1% of average
        # ensure we don't go negative
        if cur[i] < eps:
            eps = cur[i] * 0.5
        new = cur.copy()
        new[i] -= eps
        new[j] += eps
        # projection not needed because sum unchanged, but may get negative rounding.
        # Clip small negatives
        if new[i] < 0:
            new[i] = 0.0
        # compute score
        conv = _self_convolution(new, fft_size)
        new_score = 2.0 * n * conv.max() / (new.sum() ** 2)  # sum constant anyway

        # acceptance
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
        # cooling
        T *= 0.99999
        it += 1
        if it % 10000 == 0:
            # optional log
            print(f"[anneal] it {it:,} best {best_score:.6f} cur {cur_score:.6f} T {T:.6e}")

    return best, best_score

"""
A two‑stage optimizer for the self‑convolution peak.

Stage 1 – generate a large set of simple parametric families (exponential,
inverse‑sqrt, power‑law, Gaussian bumps, double‑ended shapes, random blocks)
and keep the few with the lowest exact score.

Stage 2 – starting from the best raw shape perform a long Adam‑type gradient
descent on a soft‑max approximation of the maximum convolution value,
projecting onto the simplex {a≥0, Σa = √(2 n)} after each step.

Stage 3 – a simulated‑annealing “transfer” walk makes larger stochastic
jumps that can escape remaining plateaus.  The final sequence is returned.

All work respects the security checks of `evaluate_sequence`; the function
`propose_candidate` respects the 1000 s time budget and returns the best
list of non‑negative heights found.
"""
import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution using FFT (real valued)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a, given weights w that sum to 1.
    The exact expression is 2·(w ↔ a_rev) restricted to the n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact score 2*n*max(conv)/(sum(a)^2) without security checks."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_candidate_adam(a0: np.ndarray,
                           sub_deadline: float,
                           n: int,
                           size_fft: int,
                           target_sum: float,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Adam‑type minimisation of the soft‑max of the self‑convolution.
    Returns the best sequence encountered and its exact score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning rate – a few percent of the average entry
    eta0 = target_sum * 0.2 / n

        eps = np.random.rand() * target_sum * 0.01
        if cur[i] < eps:
            eps = cur[i] * 0.5   # keep non‑negative

        new = cur.copy()
        new[i] -= eps
        new[j] += eps

        # occasional re‑projection to fix tiny rounding errors
        if it % 20000 == 0:
            new = _project_to_simplex(new, target_sum)

        new_score = _quick_score(new, n, size_fft)

        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # geometric cooling
        T *= 0.99995

        it += 1
        if it % 10000 == 0:
            print(f"[anneal] it {it:,} best {best_score:.6f} cur {cur_score:.6f} T {T:.2e}")

    return best, best_score

Stage 1 – generate a large set of simple parametric families (exponential,
inverse‑sqrt, power‑law, Gaussian bumps, double‑ended shapes, random blocks)
and keep the few with the lowest exact score.

Stage 2 – starting from the best raw shape perform a long Adam‑type gradient
descent on a soft‑max approximation of the maximum convolution value,
projecting onto the simplex {a≥0, Σa = √(2 n)} after each step.

Stage 3 – a simulated‑annealing “transfer” walk makes larger stochastic
jumps that can escape remaining plateaus.  The final sequence is returned.

All work respects the security checks of `evaluate_sequence`; the function
`propose_candidate` respects the 1000 s time budget and returns the best

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution using FFT (real valued)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a, given weights w that sum to 1.
    The exact expression is 2·(w ↔ a_rev) restricted to the n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad

    """Exact score 2·n·max(conv)/(sum(a)²) without security checks."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

        anneal_deadline = time.time() + remaining * 0.95
        best_a, best_score = _anneal(global_best_a, anneal_deadline, n, size_fft, target_sum)
        if best_score < global_best_score:
            global_best_a = best_a
            global_best_score = best_score
            print(f"[anneal] new global best = {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
