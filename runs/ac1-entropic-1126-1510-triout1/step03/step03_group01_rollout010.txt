"""
propose_candidate
-----------------

Search for a non‑negative step function (vector of heights) that minimizes

    score = 2·n·max(conv(a, a)) / (sum(a)**2)

(the denominator is invariant to scaling).  The routine combines three ideas:

1.  A **parameter sweep** over simple analytic families (uniform,
    arcsine, power‑law / beta‑like shapes, mixtures and random Dirichlet
    draws).  These candidates are evaluated very rapidly with an FFT‑based
    convolution and the best few are kept.

2.  A **two‑phase refinement** starting from each kept candidate.
    *Phase 1* performs a smooth gradient descent on a soft‑max surrogate
    of the peak using a temperature that is slowly annealed.
    *Phase 2* switches to a hard sub‑gradient that only looks at the
    indices attaining the current maximum – this directly reduces the
    L∞‑norm of the self‑convolution.
    Momentum, an adaptive step‑size and occasional tiny jitter are
    employed, and every iterate is projected back onto the probability
    simplex.

3.  The whole search respects the supplied time budget (default 1000 s);
    as soon as the deadline is passed the best sequence found so far is
    returned.  All numbers are non‑negative and the sequence is returned
    as a plain Python list, ready to be passed to ``evaluate_sequence``.

The algorithm is deliberately lightweight (only NumPy and the standard
library) and does not rely on any external optimisation packages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFTs."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (symmetrical U‑shaped) of length n, sum=1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric beta‑like distribution: a_i ∝ (i+0.5)^p·(n-i-0.5)^p, p can be negative.
    Returned vector is projected onto the simplex (sum=1).
    """
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** p * (n - i - 0.5) ** p
    # Guard against possible overflow/underflow for extreme p
    vals = np.where(np.isfinite(vals), vals, 0.0)
    if vals.sum() == 0:
        # fallback to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _mixture_distribution(n: int, w_arc: float) -> np.ndarray:
    """Mixture w·arcsine + (1‑w)·uniform, projected onto simplex."""
    arc = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    mix = w_arc * arc + (1.0 - w_arc) * uni
    return mix / mix.sum()


# ----------------------------------------------------------------------
def _dirichlet_distribution(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `alpha` (α<1 gives U‑shape)."""
    if alpha <= 0:
        alpha = 0.1
    g = np.random.gamma(shape=alpha, scale=1.0, size=n)
    if g.sum() == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return g / g.sum()


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _refine_sequence(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Two‑phase refinement of a candidate sequence.

    Phase 1 – smooth optimisation with a soft‑max surrogate.
    Phase 2 – hard sub‑gradient descent on the actual maximal entry.
    Returns the best sequence encountered (as a NumPy array) and its
    fast‑score (2·n·max(conv)).
    """
    a = a0.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_val = b.max()
    best_score = 2.0 * n * best_val

    # Momentum state
    velocity = np.zeros_like(a)
    beta = 0.9

    # Phase 1 settings
    alpha = 1.0               # soft‑max temperature
    alpha_factor = 1.004
    alpha_max = 200.0
    eta0 = 0.5                 # base learning rate

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # current self‑convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_max < best_val:
            best_val = cur_max
            best_score = cur_score
            best_seq = a.copy()

        # ------------------------------------------------------------------
        # Choose gradient source
        if alpha < 20.0:                     # smooth phase
            exp_term = np.exp(alpha * (b - cur_max))
            w = exp_term / exp_term.sum()
        else:                                # hard sub‑gradient (max only)
            # mask of entries very close to the maximum
            mask = (b >= cur_max - 1e-14)
            w = mask.astype(float)
            if w.sum() == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w.sum()

        # gradient of the smoothed/max objective
        a_rev = a[::-1]
        conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # SGD step + projection
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to avoid plateaus
        if it % 1000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

        # anneal / increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step function within the allotted time budget.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    # Reserve a couple of seconds for finalisation / printing.
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Existing construction (if any) -------------------------------------------------
    try:
        known_seq = np.asarray(height_sequence_1, dtype=np.float64).flatten()
        known_seq = known_seq[np.where(~np.isnan(known_seq))]
        if known_seq.size == 0:
            known_seq = None
    except Exception:
        known_seq = None

    # ------------------------------------------------------------------
    # Choose a few lengths to explore (all >= 500)
    base_n = known_seq.size if (known_seq is not None) else 8000
    length_factors = (0.6, 0.8, 1.0, 1.25, 1.5, 2.0)
    lengths = [
        max(500, int(base_n * f))
        for f in length_factors
        if int(base_n * f) >= 500
    ]
    # cap the largest length to keep FFT work reasonable
    max_allowed = 15000
    lengths = [min(l, max_allowed) for l in lengths]
    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Store all generated candidates together with a quick score
    all_candidates = []   # list of (fast_score, seq_array)

    for n in lengths:
        # --------- deterministic families ---------------------------------
        # uniform
        a_uni = np.full(n, 1.0 / n, dtype=np.float64)
        all_candidates.append((_fast_score(a_uni), a_uni))

        # arcsine
        a_arc = _arcsine_distribution(n)
        all_candidates.append((_fast_score(a_arc), a_arc))

        # power‑law / beta‑like shapes
        p_grid = np.linspace(-0.48, -0.10, 9)       # from strongly U‑shaped to mildly
        for p in p_grid:
            a_pow = _power_distribution(n, p)
            all_candidates.append((_fast_score(a_pow), a_pow))

        # mixtures of arcsine and uniform
        for w in (0.25, 0.5, 0.75):
            a_mix = _mixture_distribution(n, w)
            all_candidates.append((_fast_score(a_mix), a_mix))

        # some Dirichlet draws (U‑shaped and moderate)
        for alpha in (0.3, 0.7):
            for _ in range(3):
                a_dir = _dirichlet_distribution(n, alpha)
                all_candidates.append((_fast_score(a_dir), a_dir))

        # --------- known sequence, interpolated if needed -----------------
        if known_seq is not None:
            if known_seq.size == n:
                a_k = known_seq.copy()
            else:
                # linear interpolation to desired length
                x_old = np.linspace(0, 1, known_seq.size)
                x_new = np.linspace(0, 1, n)
                a_k = np.interp(x_new, x_old, known_seq)
                a_k = _project_onto_simplex(a_k, s=1.0)
            all_candidates.append((_fast_score(a_k), a_k))

        # early break if we are out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Keep the best few candidates for refined optimisation
    all_candidates.sort(key=lambda tup: tup[0])   # ascending by fast_score
    top_k = 7
    top_candidates = all_candidates[:top_k]

    # ------------------------------------------------------------------
    # Global best tracking
    best_overall_seq = None
    best_overall_score = float("inf")

    # ------------------------------------------------------------------
    for idx, (init_score, a_init) in enumerate(top_candidates):
        n = a_init.shape[0]
        # Allocate a slice of remaining time for this refinement
        time_left = deadline - time.time()
        if time_left < 5.0:
            break
        slice_deadline = time.time() + time_left * 0.4   # use up to 40 % of remaining

        print(f"[{time.time():.1f}] Refinement {idx+1}/{len(top_candidates)} "
              f"len={n} init_score≈{init_score:.6f}")

        refined_seq, refined_fast_score = _refine_sequence(
            a_init, n, deadline=slice_deadline
        )
        # Evaluate the final candidate with the official black‑box (exact)
        try:
            final_score = evaluate_sequence(refined_seq.tolist())
        except Exception:
            # fallback to fast score (should be close)
            final_score = refined_fast_score

        print(f"    refined fast score={refined_fast_score:.6f} "
              f"final evaluated score={final_score:.6f}")

        if final_score < best_overall_score:
            best_overall_score = final_score
            best_overall_seq = refined_seq.copy()
            print(f"*** New global best: score={best_overall_score:.6f}")

        # early exit if we already beat the target
        if best_overall_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded (should not happen) return a uniform vector.
    if best_overall_seq is None:
        n = 1000
        best_overall_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_overall_score = evaluate_sequence(best_overall_seq.tolist())

    print(
        f"Final best score: {best_overall_score:.6f} "
        f"(length = {len(best_overall_seq)})"
    )
    # Return a plain Python list as required by the harness.
    return best_overall_seq.tolist()
