def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import time, numpy as np, math, random

    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 5.0)
    np.random.seed(seed)

    # Determine problem size: if height_sequence_1 exists, use its length * 2 or else default.
    try:
        base = np.array(height_sequence_1, dtype=float)
        n0 = base.size
    except Exception:
        n0 = 15000
    N = min(30000, max(12000, n0*2))   # maybe 30000
    S = math.sqrt(2.0 * N)

    best_seq = None
    best_score = float('inf')

    # Predefine modes
    init_modes = ['uniform', 'exponential', 'linear', 'inverse', 'random']

    # ADAM hyperparameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # outer loop: restarts while time
    restart = 0
    while time.time() < deadline:
        restart += 1
        mode = init_modes[ restart % len(init_modes) ]
        # Generate initial sequence on simplex
        if mode == 'uniform':
            a = np.full(N, S/N, dtype=float)
            a += 0.001 * S * np.random.rand(N) / N
        elif mode == 'exponential':
            r = np.random.uniform(0.5,0.99)
            a = r ** np.arange(N, dtype=float)
            a = a / a.sum() * S
        elif mode == 'linear':
            a = np.arange(1, N+1, dtype=float)
            a = a / a.sum() * S
        elif mode == 'inverse':
            a = 1.0 / np.arange(1, N+1, dtype=float)
            a = a / a.sum() * S
        elif mode == 'random':
            a = np.random.rand(N)
            a = a / a.sum() * S
        else:
            a = np.full(N, S/N, dtype=float)

        a = _project_onto_simplex(a, S) # ensure exact sum

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # iteration limit per restart based on remaining time
        max_iter_restart = 4000
        for it in range(max_iter_restart):
            if time.time() >= deadline:
                break
            # compute convolution
            conv = _fft_convolve(a)
            # compute max and indices
            max_conv = conv.max()
            # compute subgradient mask
            # consider values within small tolerance to avoid jitter
            tol = 1e-12
            mask = (conv >= max_conv - tol)
            # number of max positions
            cnt = mask.sum()
            if cnt == 0:
                w = np.zeros_like(conv)
            else:
                w = mask.astype(float) / cnt
            # gradient
            grad = 2.0 * _fft_correlate(w, a)

            # Adam updates
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))
            lr = lr0 * (1.0 / (1.0 + 0.0005 * it))  # decay
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            # enforce nonnegativity
            a_new = np.maximum(a_new, 0.0)
            # project onto simplex
            a_new = _project_onto_simplex(a_new, S)

            a = a_new

            # occasional evaluation
            if it % 20 == 0:
                conv = _fft_convolve(a)
                # compute score (since sum = S)
                score = conv.max()
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[Restart {restart} it {it}] New best score {best_score:.6f}")

        # after inner loops, perform polishing on best_seq (maybe)
        # but only if time left
        # We'll do some random pair moves after each restart
        if best_seq is not None and time.time() < deadline:
            polish_iters = 2000
            for p in range(polish_iters):
                if time.time() >= deadline:
                    break
                sigma = 0.4 * (1.0 - p / polish_iters) + 0.02
                cand = _random_pair_move(best_seq, sigma)
                conv = _fft_convolve(cand)
                score = conv.max()
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    print(f"[Polish restart {restart} step {p}] New best score {best_score:.6f}")

    # fallback: if no best_seq found (unlikely) use uniform
    if best_seq is None:
        best_seq = np.full(N, S/N)

    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()

"""
Improved optimizer for the step‑function evaluation problem.

Main ideas
----------
* Work on a fixed ℓ₁‑norm S = √(2·N) (the score becomes simply
  max(conv) when the sum equals S, see the evaluation routine).
* Use several random restarts with different sensible initial shapes
  (uniform, exponential, linear, inverse, random).
* Inside each restart run an Adam‑style projected gradient descent.
  The gradient is taken w.r.t. a *sub‑gradient* of the max‑convolution:
  we put weight 1 on every index where the current convolution attains
  its maximum (or is within a tiny tolerance) and zero elsewhere.
  This directly targets the true objective “minimise the peak”.
* After each Adam step the iterate is projected onto the non‑negative
  simplex of sum S (Duchi et al. 2008 algorithm).
* Whenever a better sequence is found we keep it as the global best.
* When time permits we perform a short polishing phase consisting of
  random pairwise mass swaps (still respecting the ℓ₁‑norm) – a cheap
  simulated‑annealing step.
* All heavy work (convolution / correlation) is performed with FFTs,
  giving O(N log N) per iteration and allowing N up to a few × 10⁴
  within the 1000 s budget.

The routine respects the required signature `propose_candidate(...)`
and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of `v` onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho_arr = u - cssv / ind > 0
    if not np.any(rho_arr):
        theta = 0.0
    else:
        rho = np.where(rho_arr)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two random indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """Valid‑mode correlation w ⋆ a (length = len(a))."""
    n = a.shape[0]
    L = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start:start + n]

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    The algorithm consists of several random restarts of an Adam‑type
    projected gradient descent, followed by a short random‑pair polishing
    phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the wrapper
    deadline = start_time + max(0.0, budget_s - 2.0)

    # ------------------------------------------------------------------
    # Problem size – try to beat the previous construction
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n0 = base.size
    except Exception:
        n0 = 15000
    # use a length comfortably larger than the previous best but still
    # cheap enough for many FFTs
    N = min(30000, max(12000, n0 * 2))
    S = math.sqrt(2.0 * N)               # ℓ₁‑norm we keep fixed

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    init_modes = ['uniform', 'exponential', 'linear', 'inverse', 'random']
    restart = 0

    # ------------------------------------------------------------------
    # Main optimisation loop (restarts until time runs out)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restart += 1
        mode = init_modes[restart % len(init_modes)]

        # ---------- initialise a new feasible point ----------
        if mode == 'uniform':
            a = np.full(N, S / N, dtype=float)
            a += 0.001 * S * np.random.rand(N) / N
        elif mode == 'exponential':
            r = np.random.uniform(0.5, 0.99)
            a = r ** np.arange(N, dtype=float)
            a = a / a.sum() * S
        elif mode == 'linear':
            a = np.arange(1, N + 1, dtype=float)
            a = a / a.sum() * S
        elif mode == 'inverse':
            a = 1.0 / np.arange(1, N + 1, dtype=float)
            a = a / a.sum() * S
        else:  # 'random'
            a = np.random.rand(N)
            a = a / a.sum() * S

        a = _project_onto_simplex(a, S)

        # ---------- Adam state ----------
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # set per‑restart iteration budget (adapt to remaining time)
        max_iter_restart = 4000
        for it in range(max_iter_restart):
            if time.time() >= deadline:
                break

            # compute convolution and the current peak
            conv = _fft_convolve(a)
            max_conv = conv.max()

            # sub‑gradient of the max‑function:
            # put equal weight on all indices that attain the maximum
            mask = (conv >= max_conv - 1e-12)
            k = mask.sum()
            w = np.zeros_like(conv)
            if k > 0:
                w[mask] = 1.0 / k

            # gradient of  max‑approximation  (2 ⋅ corr(w, a))
            grad = 2.0 * _fft_correlate(w, a)

            # -------- Adam update (projected) ----------
            beta1, beta2 = 0.9, 0.999
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)

            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))

            # learning‑rate schedule – slowly decays
            lr0 = 0.5
            lr = lr0 / (1.0 + 0.001 * it)

            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            a_new = a - step
            a_new = np.maximum(a_new, 0.0)
            a = _project_onto_simplex(a_new, S)

            # occasional evaluation of the true score (exact max)
            if it % 20 == 0:
                conv = _fft_convolve(a)
                score = conv.max()            # because Σa = S
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[Restart {restart:02d} it {it:04d}] "
                          f"New best score {best_score:.6f}")

        # ---------- local polishing on the current global best ----------
        if best_seq is not None and time.time() < deadline:
            polish_iters = 2000
            for p in range(polish_iters):
                if time.time() >= deadline:
                    break
                sigma = 0.4 * (1.0 - p / polish_iters) + 0.02
                cand = _random_pair_move(best_seq, sigma)
                conv = _fft_convolve(cand)
                score = conv.max()
                if score < best_score:
                    best_score = score
                    best_seq = cand.copy()
                    print(f"[Polish R{restart:02d} step {p:04d}] "
                          f"New best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case no sequence was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(N, S / N, dtype=float)

    best_seq = np.maximum(best_seq, 0.0)
    return best_seq.tolist()
