"""
Improved step‑function search.

The algorithm consists of three phases:

1. **Soft‑max L‑BFGS continuation** – the heights a_i are written as
   a = S·softmax(y) with S = sqrt(2·n).  We minimise a smooth surrogate of
   max(a*a) using a log‑sum‑exp smoothing parameter μ that is geometrically
   cooled (μ ← μ·0.85) down to 1e‑4.  Several diverse initialisations (uniform,
   V‑shaped, random Gaussian, random Dirichlet, reference) are tried.

2. **Hard‑max projected sub‑gradient descent** – starting from the best
   sequence obtained above we perform a sub‑gradient step for the true
   objective max(a*a).  Step size follows a simple diminishing schedule and a
   back‑tracking line‑search guarantees monotone improvement.  Projection onto
   the simplex keeps the ℓ¹‑norm fixed (S).

3. **Random block perturbations** – finally a cheap stochastic block search
   explores the neighbourhood of the best solution: the index set is split
   into a few random blocks, each block receives a random height, the vector
   is projected onto the simplex and evaluated.  Improvements are kept.

All phases respect the global time‑budget (default 1000 s) and update the
best sequence whenever a lower score is found.  The returned list contains
the final heights (non‑negative, sum ≈ sqrt(2·n)).
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smoothed objective f_μ(y) = max_k (a*a)[k] + μ·log Σ_k exp((a*a)[k]/μ)
    together with its gradient w.r.t. y.
    """
    # probabilities and actual heights
    p = _softmax(y)                # Σp = 1, p_i ≥ 0
    a = S * p                      # Σa = S

    # full convolution a*a
    conv = np.convolve(a, a)       # length 2·n‑1
    max_c = conv.max()

    # soft‑max weights
    diff = (conv - max_c) / mu
    diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
    exp_diff = np.exp(diff)
    Z = exp_diff.sum()
    w = exp_diff / Z                # Σ w = 1, w_i ≥ 0

    f_val = max_c + mu * math.log(Z)

    # gradient w.r.t. the heights a (size n)
    # g_a[i] = 2 * Σ_k w_k * a_{k-i}
    conv_wr = np.convolve(w, a[::-1])      # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]   # slice of length n

    # chain rule to obtain gradient w.r.t. y
    inner = np.dot(g_a, p)                 # scalar = Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObj:
    """
    Callable for scipy.optimize.minimize.  Returns (value, gradient)
    for the current smoothing parameter μ.
    """

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _smooth_obj_and_grad(y, self.mu, self.n, self.S)
        return f, g

    init_y.append(np.zeros(n, dtype=np.float64))

    # 1. V‑shaped (large at ends, tiny in centre)
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    vshape = np.abs(idx - mid) + 1e-3
    init_y.append(np.log(vshape))

    # 2. random Gaussian (moderate variance)
    init_y.append(np.random.randn(n) * 0.5)

    # 3. random Gaussian (higher variance)
    init_y.append(np.random.randn(n) * 1.0)

    # 4. random Dirichlet‑like (log of uniform positives)
    init_y.append(np.log(np.random.rand(n) + 1e-12))

    # 5. reference if available
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        eps = 1e-12
        init_y.append(np.log(a_ref + eps))

    # ------------------------------------------------------------------
    # continuation parameters
    mu_factor = 0.85
    mu_min = 1e-4
    maxiter_per_mu = 250

    for start_id, y0 in enumerate(init_y):
        if time.time() > deadline:
            break

        print(f"[cont] start {start_id}/{len(init_y)} – initializing continuation")
        y = y0.copy()

        # initialise temperature from the current max‑convolution
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        if mu <= 0.0:
            mu = 1.0

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObj(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x

            # evaluate the *hard* objective
            a = S * _softmax(y)
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[improve] new best = {best_score:.6f}")

            # cool
            mu *= mu_factor

    # ------------------------------------------------------------------
    # Hard‑max projected sub‑gradient descent (with simple line‑search)
    a = best_a.copy()
    current_max = np.convolve(a, a).max()
    step0 = S * 0.5 / n                # initial step size (roughly average entry)
    decay = 1e-5
    it = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        # sub‑gradient of max‑conv w.r.t. a
        g = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

        # diminishing step size
        eta = step0 / (1.0 + decay * it)

        # back‑tracking line‑search (max‑conv based, cheaper than evaluate_sequence)
        improved = False
        for _ in range(12):
            a_new = a - eta * g
            a_new = _project_to_simplex(a_new, S)
            max_new = np.convolve(a_new, a_new).max()
            if max_new < max_c - 1e-12:
                improved = True
                a = a_new
                max_c = max_new
                if max_new < best_score:
                    best_score = max_new
                    best_a = a_new.copy()
                    print(f"[sgd improve] new best = {best_score:.6f}")
                break
            eta *= 0.5

        it += 1
        if not improved and eta < 1e-12:
            # no progress any more
            break
        if it > 5000:
            break

    # ------------------------------------------------------------------
    # cheap random block search
    remaining = deadline - time.time()
    if remaining > 2.0:
        block_trials = int(min(5000, remaining * 150))
        print(f"[block] performing {block_trials} random block trials")
        for _ in range(block_trials):
            nb = np.random.randint(2, 13)                     # 2‑12 blocks
            cuts = np.sort(
                np.random.choice(n - 1, nb - 1, replace=False) + 1
            )
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            sc = evaluate_sequence(cand.tolist())
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[block improve] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final polishing with tiny random perturbations (if time permits)
    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = np.random.uniform(0.0, best_a[i])
        if delta <= 0.0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, S)
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[swap improve] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
