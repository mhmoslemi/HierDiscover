def projected_gradient_descent(a0, S, budget_s, start_time):
   a = a0.copy()
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   eps=1e-8
   beta1=0.9
   beta2=0.999
   max_iters = big number e.g., 100000
   for it in range(max_iters):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 5: break
        conv = _fft_convolve(a)
        max_conv = conv.max()
        # compute weight vector w: identify top K max conv indices,
        # maybe top_k = max(1, int(0.001 * len(conv)))  # e.g., 0.1% of indices.
        # or softmax:
        beta = 0.5 + (80 - 0.5)* (it / max_iters)
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        # gradient
        grad = 2.0 * _fft_correlate(w, a)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        lr = 0.5 * (1 - it/max_iters) + 0.01  # linearly decreasing
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # projection onto simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, S)
        a = a_new
        # track best
        conv = _fft_convolve(a)
        cur = conv.max()
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
   return best_seq, best_score

"""
Multi‑start projected Adam optimization for the step‑function problem.

* Each restart begins from a different feasible point (uniform,
  interpolated previous best, random Dirichlet).
* The objective  max(conv) is approximated by a smooth soft‑max.
  The gradient of  Σ w_k·(a*a)_k  w.r.t. a is 2·corr(w,a), which is
  computed via FFT.
* An Adam‑style update is taken and the result is projected onto the
  non‑negative simplex with prescribed ℓ¹‑norm S = √(2·N).
* After the gradient phase a polishing stage performs many random
  pair‑wise mass swaps (simulated‑annealing style) while keeping the
  sum exactly S.
* The whole routine respects the global time budget (default 1000 s)
  and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto the set {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based self‑convolution (full length 2N‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based correlation: (w ⋆ a)[i] = Σ_k w[k+i]·a[k]  (valid mode)
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                     # = 2N‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length N (valid correlation)


# ----------------------------------------------------------------------
# Random pairwise mass‑swap (keeps exact ℓ¹‑norm)
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a tiny amount of mass between two random positions."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # delta ∈ (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Interpolation helper (used for the previous best sequence)
# ----------------------------------------------------------------------
def _interp_to_length(src: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation of src to a new length."""
    if src.size == target_len:
        return src.copy()
    x_src = np.arange(src.size)
    x_tgt = np.linspace(0, src.size - 1, target_len)
    return np.interp(x_tgt, x_src, src)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Returns a non‑negative sequence that (heuristically) minimises
    evaluate_sequence().  The routine runs multiple restarts of projected
    Adam optimisation with a soft‑max approximation of max(conv) and a
    final random‑pair polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 2.0)   # 2 s safety margin

    # --------------------------------------------------------------
    # Problem size (choose a power‑of‑two length for fast FFTs)
    # --------------------------------------------------------------
    try:
        base_seq = np.array(height_sequence_1, dtype=float)   # type: ignore
        n0 = base_seq.size
    except Exception:
        base_seq = None
        n0 = 0

    # Target length – use the maximum allowed (20000) but keep a power of two.
    # If the previous best is already large, keep it; otherwise go up.
    N = 20000
    if N & (N - 1):
        # round up to the next power of two for efficiency
        N = 1 << ((N - 1).bit_length())
    S = math.sqrt(2.0 * N)          # ℓ¹‑norm we enforce (score = max(conv) after scaling)

    # --------------------------------------------------------------
    # Helper to evaluate the current sequence (quick internal version)
    # --------------------------------------------------------------
    def _current_score(seq: np.ndarray) -> float:
        """Score = max(conv) for a sequence already normalised to sum=S."""
        conv = _fft_convolve(seq)
        return conv.max()

    # --------------------------------------------------------------
    # Prepare a list of initial points for the restarts
    # --------------------------------------------------------------
    init_points = []

    # 1) Uniform + tiny noise
    a0 = np.full(N, S / N, dtype=float)
    a0 += 1e-4 * S * np.random.rand(N) / N
    a0 = _project_onto_simplex(a0, S)
    init_points.append(a0)

    # 2) Interpolated previous best (if available)
    if base_seq is not None and base_seq.size > 0:
        a1 = _interp_to_length(base_seq, N)
        a1 = _project_onto_simplex(a1, S)
        init_points.append(a1)

    # 3) Several random Dirichlet points (will be generated on the fly)
    # We'll generate them lazily inside the restart loop.

    # --------------------------------------------------------------
    # Global best tracker
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Restart loop
    # --------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        # If we have pre‑computed initial points left, pop one; otherwise create a random start.
        if restart < len(init_points):
            a = init_points[restart].copy()
        else:
            # Random Dirichlet‐like start
            a = np.random.rand(N)
            a = _project_onto_simplex(a, S)

        # Adam parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8

        # Hyper‑parameters for the soft‑max approximation
        max_iters = 25000               # safe upper bound; will be cut by time
        beta_start, beta_end = 0.5, 80.0
        lr_start, lr_end = 0.8, 0.02

        # Track local best for this restart
        local_best_seq = a.copy()
        local_best_sc = _current_score(a)

        # ----------------------------------------------------------
        # Projected Adam optimisation (soft‑max of convolution)
        # ----------------------------------------------------------
        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break

            # 1) compute convolution and the soft‑max weights
            conv = _fft_convolve(a)
            max_conv = conv.max()

            t_frac = it / max_iters
            beta = beta_start + (beta_end - beta_start) * t_frac
            lr = lr_start + (lr_end - lr_start) * (1.0 - t_frac)

            # soft‑max (stable)
            exp_vals = np.exp(beta * (conv - max_conv))
            w = exp_vals / exp_vals.sum()

            # 2) gradient of Σ w_k·(a*a)_k  = 2·corr(w,a)
            grad = 2.0 * _fft_correlate(w, a)

            # 3) Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step

            # 4) projection onto the non‑negative simplex Σ= S
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_onto_simplex(a_new, S)

            a = a_new

            # occasional evaluation
            if it % 20 == 0:
                sc = _current_score(a)
                if sc < local_best_sc:
                    local_best_sc = sc
                    local_best_seq = a.copy()
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[Restart {restart:02d}  Iter {it:05d}] New global best: {best_score:.6f}")

        # ----------------------------------------------------------
        # Polishing phase for this restart (random pair swaps)
        # ----------------------------------------------------------
        polish_iters = 5000
        for p_it in range(1, polish_iters + 1):
            if time.time() >= deadline:
                break
            # linear cooling of swap amplitude
            sigma = 0.4 * (1.0 - p_it / polish_iters) + 0.02
            cand = _random_pair_move(local_best_seq, sigma)
            sc = _current_score(cand)
            if sc < local_best_sc:
                local_best_sc = sc
                local_best_seq = cand.copy()
                if sc < best_score:
                    best_score = sc
                    best_seq = cand.copy()
                    print(f"[Polish R{restart:02d} {p_it:05d}] New global best: {best_score:.6f}")

        # store result of this restart
        restart += 1

    # --------------------------------------------------------------
    # Final polishing on the globally best sequence (if any)
    # --------------------------------------------------------------
    if best_seq is None:
        # shouldn't happen, but fallback to uniform
        best_seq = np.full(N, S / N, dtype=float)

    final_polish_iters = 8000
    seq = best_seq.copy()
    for it in range(1, final_polish_iters + 1):
        if time.time() >= deadline:
            break
        sigma = 0.3 * (1.0 - it / final_polish_iters) + 0.01
        cand = _random_pair_move(seq, sigma)
        sc = _current_score(cand)
        if sc < best_score:
            best_score = sc
            seq = cand.copy()
            print(f"[Final polish {it:05d}] New best score: {best_score:.6f}")

    # Ensure non‑negativity (tiny negative rounding artefacts)
    seq = np.maximum(seq, 0.0)

    print(f"[Done] Best score achieved: {best_score:.6f}  Length: {len(seq)}")
    return seq.tolist()
