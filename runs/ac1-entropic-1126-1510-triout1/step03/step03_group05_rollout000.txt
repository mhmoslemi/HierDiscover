"""
Smooth‑max Adam optimisation for the self‑convolution bound.

The objective 2·n·max(a*a)/(∑a)² is convex in the non‑negative vector a
with a fixed ℓ₁‑norm.  We approximate the non‑smooth max with a
log‑sum‑exp (soft‑max) smoothing parameter λ and minimise the smoothed
objective by Adam while repeatedly projecting onto the simplex
{a≥0, Σa = √(2n)}.  λ is increased during optimisation (continuation) so
the algorithm first follows a smooth landscape and then focuses on the
true maximum.  The routine starts from a set of parametric candidates,
keeps the best, and finally refines it by random block perturbations.

The whole search respects a 1000 s wall‑clock budget; intermediate
scores are printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n−1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _eval_score(a: np.ndarray, target_sum: float, size_fft: int) -> float:
    """
    Fast evaluation of the objective:
        2·n·max(a*a) / target_sum²
    (identical to evaluate_sequence for admissible a).
    """
    n = a.shape[0]
    conv = _self_convolution(a, size_fft)
    return float(2.0 * n * conv.max() / (target_sum ** 2))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score non‑negative step function.
    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1) Determine problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3130
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())   # >= 3n‑2 for linear‑conv steps

    # --------------------------------------------------------------
    # 2) Helper to keep the current global best
    best_a = None
    best_score = float("inf")

    def _maybe_update(candidate: np.ndarray):
        nonlocal best_a, best_score
        sc = _eval_score(candidate, target_sum, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            print(f"[init] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Parametric shape search (light version – enough to get a good start)
    # uniform baseline
    a0 = np.ones(n, dtype=float)
    a0 = _project_to_simplex(a0, target_sum)
    _maybe_update(a0)

    # reference (if present)
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        ref = np.maximum(ref, 0.0)
        ref = _project_to_simplex(ref, target_sum)
        _maybe_update(ref)

    # families – limited sampling to stay within the time budget
    N_SAMPLES = 80

    # ----- inverse‑sqrt (single‑ended) -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        _maybe_update(a)

    # ----- double‑ended inverse‑sqrt -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        i = np.arange(n, dtype=float)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        _maybe_update(a)

    # ----- power‑law (double‑ended) -----
    for _ in range(N_SAMPLES):

        p = np.random.rand() * 2.5 + 0.1

             1.0 / ((n - 1 - i) + beta + 1.0) ** p)
        a = _project_to_simplex(a, target_sum)
        _maybe_update(a)

    # ----- exponential decay (double‑ended) -----
    for _ in range(N_SAMPLES):

        i = np.arange(n, dtype=float)
        a = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
        a = _project_to_simplex(a, target_sum)
        _maybe_update(a)

    for _ in range(N_SAMPLES):

        a = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        _maybe_update(a)

    # ----- random block candidates -----
    for _ in range(200):
        nb = np.random.randint(2, 7)            # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)

            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        _maybe_update(cand)

    print(f"[shape] best score after families = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Smooth‑max Adam optimisation
    a = best_a.copy()
    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lr0 = 0.12

    # smoothing schedule
    lam = 2.0          # start with a gentle smoothing
    lam_max = 250.0    # final (almost exact) max
    lam_mul = 1.001    # multiplicative increase each iteration

    # control loops
    it = 0
    max_iter = 400_000      # hard ceiling; real stop comes from time budget
    no_improve = 0

    while time.time() < deadline and it < max_iter:
        # ---------- compute convolution ----------
        fa = np.fft.rfft(a, n=size_fft)
        conv_full = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        max_c = conv_full.max()

        # ---------- smooth max (softmax) ----------
        shifted = conv_full - max_c               # ≤ 0
        exp_terms = np.exp(lam * shifted)
        w = exp_terms / exp_terms.sum()           # weights sum to 1

        # ---------- gradient of smoothed objective ----------
        fw = np.fft.rfft(w, n=size_fft)
        corr = np.fft.irfft(np.conj(fa) * fw, n=size_fft)[:n]   # correlation for lags 0…n‑1
        grad = 2.0 * corr

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / (1.0 + 1e-5 * it)          # very gentle decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick to escape plateaus
        if (it % 3500 == 0 and it > 0) or no_improve >= 4000:
            noise = np.random.randn(n) * target_sum * 4e-4
            a_new = _project_to_simplex(a_new + noise, target_sum)
            no_improve = 0
            print(f"[{it:07d}] random perturbation")

        # ---------- evaluate exact score ----------
        conv_new = np.fft.irfft(np.fft.rfft(a_new, n=size_fft) *
                               np.fft.rfft(a_new, n=size_fft),
                               n=size_fft)[: 2 * n - 1]
        cur_score = 2.0 * n * conv_new.max() / (target_sum ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            no_improve = 0
            print(f"[{it:07d}] new best = {best_score:.6f}")
        else:
            no_improve += 1

        a = a_new
        it += 1
        lam = min(lam_max, lam * lam_mul)   # increase smoothing gradually

    print(f"[adam] finished after {it} iterations, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Final random block refinement (local search)
    for _ in range(250):
        if time.time() >= deadline:
            break
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _eval_score(cand, target_sum, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
