def _refine_adam(a0, deadline, n, size_fft, target_sum):
    a = a0.copy()
    best = a.copy()
    best_score = _score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    # initial temperature maybe based on conv range
    conv = _self_convolution(a, size_fft)
    M = conv.max()
    m_range = M - conv.min()
    T = max(m_range * 0.5, 1e-4)
    while time.time() < deadline and t < 200000:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum()**2)
        if cur_score < best_score:
            best_score = cur_score; best = a.copy()
        # compute softmax weights
        shifted = conv - conv.max()
        # Temperature schedule: T = max(initial_T * 0.99**t, 1e-5)
        # Actually we can adapt T gradually down.
        w = np.exp(shifted / (T+1e-9))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad = _weighted_gradient(a, w, size_fft)
        # adam update
        t += 1
        lr = 0.5 * target_sum / n / (1 + 0.0001*t) # decaying step size
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a + step
        a_new = _project_to_simplex(a_new, target_sum)
        # optionally random jitter every few iterations
        if t % 5000 == 0:
            a_new += np.random.randn(n) * target_sum * 0.0005
            a_new = _project_to_simplex(a_new, target_sum)
        a = a_new
        T *= 0.9995
    return best, best_score

"""
Propose a step‑function sequence whose self‑convolution peak is as low as
possible (the evaluation score equals that peak when the ℓ¹‑norm is fixed
to √(2 n)).  The algorithm combines three ideas:

1. **Rich initialisation** – a handful of parametric families (including an
   arcsine‑like shape, inverse‑sqrt, exponential, sinusoidal and random
   block patterns) and, if available, the previously discovered best
   sequence ``height_sequence_1``.

2. **Projected Adam optimisation** – a smooth approximation of the max‑convolution
   (soft‑max with an annealed temperature) is minimised with a
   Adam‑style update, after each step projecting the vector onto the
   simplex {a ≥ 0, Σa = √(2 n)}.  The gradient of the smoothed objective is
   obtained analytically via FFT‑based convolution.

3. **Late‑stage jittering** – once the time budget is almost exhausted we
   perform cheap random perturbations of the current best vector and keep
   any improvement.

All heavy work is done with NumPy’s FFT (linear convolution) which makes
each iteration O(n log n) and easily fits into the 1000 s budget for n≈3000.
The function ``propose_candidate`` respects the required signature and
returns the best list of floats found.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    For a fixed weight vector w (sum w = 1) this equals
        2·(w ⋆ rev(a))   restricted to the first n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = (2n-1)+(n-1) = 3n-2
    start = len(w) - len(a)                       # = n-1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, size_fft: int) -> float:
    """Return the maximum entry of the full self‑convolution of a."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _refine_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the smoothed max‑convolution.
    Returns the best vector encountered and its (exact) score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _conv_max(best, size_fft)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # temperature schedule (soft‑max smoothing)
    conv = _self_convolution(a, size_fft)
    conv_range = conv.max() - conv.min()
    T = max(conv_range * 0.5, 1e-4)   # initial temperature

    it = 0
    while time.time() < deadline and it < max_iters:
        # evaluate exact score occasionally
        if it % 1000 == 0:
            cur_score = _conv_max(a, size_fft)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                # helpful progress output
                print(f"[adam] it {it:,} new best = {best_score:.6f}")

        # ----- compute soft‑max weights -----
        conv = _self_convolution(a, size_fft)
        shifted = conv - conv.max()                 # ≤ 0
        w = np.exp(shifted / max(T, 1e-12))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        it += 1
        lr = 0.5 * target_sum / n / (1.0 + 1e-5 * it)   # slowly decaying
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a + step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick to escape plateaus
        if it % 5000 == 0:
            a_new += np.random.randn(n) * target_sum * 0.0008
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        # cool the temperature very gently
        T *= 0.9999

    # final exact evaluation of the best we have seen
    final_score = _conv_max(best, size_fft)
    return best, final_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock budget
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.05   # small safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)
    size_fft = 1 << ((2 * n - 1).bit_length())   # power‑of‑two ≥ 2n‑1

    # ------------------------------------------------------------------
    # helper to evaluate the exact score (max convolution)
    def exact_score(a: np.ndarray) -> float:
        return _conv_max(a, size_fft)

    # ------------------------------------------------------------------
    # generate a diverse set of initial candidates
    candidates = []

    idx = np.arange(n, dtype=float)

    # 1) arcsine‑like shape (largest at the edges, smallest in the centre)
    a = 1.0 / np.sqrt((idx + 0.5) * (n - idx - 0.5))
    a = _project_to_simplex(a, target_sum)
    candidates.append(a)

    # 2) double‑ended inverse‑sqrt (as in the previous code)
    for _ in range(80):
        beta = np.random.rand() * 10.0
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        candidates.append(a)

    # 3) pure inverse‑sqrt (single‑ended)
    for _ in range(80):
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        candidates.append(a)

    # 4) exponential decay
    for _ in range(80):
        decay = np.random.rand() * 0.01 + 0.0005
        a = np.exp(-decay * idx)
        a = _project_to_simplex(a, target_sum)
        candidates.append(a)

    # 5) cosine‑based (large at ends)
    for _ in range(80):
        p = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        candidates.append(a)

    # 6) sine‑based (peak in the middle)
    for _ in range(80):
        p = np.random.rand() * 5.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** p
        a = _project_to_simplex(a, target_sum)
        candidates.append(a)

    # 7) random block patterns (2–6 blocks)
    for _ in range(200):
        nb = np.random.randint(2, 7)                     # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        candidates.append(cand)

    # 8) use the previous best sequence if available
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        candidates.append(a_ref)

    # ------------------------------------------------------------------
    # keep a few top raw candidates (by exact score)
    K = 12
    top_candidates = []   # list of (score, array)

    def consider(score: float, arr: np.ndarray) -> None:
        """Maintain a sorted list of the best K candidates (lower is better)."""
        if len(top_candidates) < K:
            top_candidates.append((score, arr.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if score < top_candidates[-1][0]:
                top_candidates[-1] = (score, arr.copy())
                top_candidates.sort(key=lambda x: x[0])

    for a in candidates:
        sc = exact_score(a)
        consider(sc, a)

    print(f"[initial] best raw score = {top_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2) REFINEMENT – projected Adam optimiser on the best K candidates
    global_best_a = top_candidates[0][1].copy()
    global_best_score = top_candidates[0][0]

    for rank, (raw_score, a0) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank + 1} (raw {raw_score:.6f})")
        # allocate remaining time proportionally
        remaining = deadline - time.time()
        # allow about 0.75 s per 1000 iterations as a rough guide
        max_iters = int(min(300_000, remaining * 1500))
        refined_a, refined_score = _refine_adam(
            a0, deadline, n, size_fft, target_sum, max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after Adam = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) FINAL jitter phase – cheap random perturbations around the best
    while time.time() < deadline:
        # perturb a small random subset (≈1 % of entries)
        a = global_best_a.copy()
        nb_perturb = max(1, int(0.01 * n))
        idxs = np.random.choice(n, size=nb_perturb, replace=False)
        a[idxs] += np.random.randn(nb_perturb) * target_sum * 0.001
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
        sc = exact_score(a)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = a.copy()
            print(f"[jitter] improved to {global_best_score:.6f}")

        # quick exit if not much time left
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
