def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # parameter for f
    f_param = cp.Parameter(n, value=f)
    conv = cp.conv(f_param, g) # length 2n-1
    constraints = [conv <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver='ECOS', warm_start=True, verbose=False, max_iters=2000)
    except cp.error.SolverError:
        return None
    if g.value is None:
        return None
    return np.array(g.value, dtype=float)

The algorithm starts from the known best construction (if available) or a
random simplex point.  It repeatedly solves a linear program

   max  Σ g_i   s.t.   f * g ≤ M ,  g ≥ 0

where f is the current sequence, * denotes full convolution and M is the
current maximal self‑convolution of f.  The LP pushes mass to positions
that produce small convolution values.  The solution is renormalised to
the original ℓ¹‑norm and mixed with f.  After a few LP steps a cheap
random‑swap hill‑climbing refines the result.

The whole routine respects the 1000 s budget and returns the best list
found.
"""
import time
import numpy as np
import math

def _max_conv(a: np.ndarray) -> float:
    """Return max entry of the full self‑convolution of a."""
    n = a.shape[0]
    # FFT based O(n log n) convolution
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return float(conv[: 2 * n - 1].max())

def _conv_matrix(f: np.ndarray):
    """Build the (2n‑1)×n convolution matrix C with C·g = f*g."""
    n = f.shape[0]
    # each diagonal d (0 ≤ d < n) contains the constant f[d]
    # shape (2n‑1, n)
    import scipy.sparse as sp
    diag_data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)          # 0,1,…,n‑1
    C = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')
    return C

def _lp_step(f: np.ndarray, M: float):
    """Solve   max Σ g_i   s.t.   f*g ≤ M , g ≥ 0   and return g."""
    import scipy.optimize as opt
    import scipy.sparse as sp

    n = f.shape[0]
    C = _conv_matrix(f)                     # (2n‑1)×n sparse matrix
    b = np.full(2 * n - 1, M, dtype=float)  # RHS of each inequality
    c = -np.ones(n)                         # minimise –Σg_i  (i.e. maximise Σg_i)
    bounds = [(0.0, None)] * n

    res = opt.linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return None
    g = np.maximum(res.x, 0.0)       # numerical safety
    return g

3. Renormalise the LP solution to ℓ¹‑norm = 1 and mix it with the
