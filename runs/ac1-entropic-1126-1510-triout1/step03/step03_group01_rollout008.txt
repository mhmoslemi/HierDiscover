```python
def _subgradient_descent(a, n, time_limit):
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_max = b.max()
    best_score = 2 * n * best_max

    it = 0
    eta0 = 0.5
    momentum = np.zeros_like(a)
    beta = 0.9
    while time.time() < time_limit:
        it += 1
        # compute convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
        # compute gradient using argmax
        argmax_idx = np.argmax(b)
        # create mask with 1 at argmax
        mask = np.zeros(2 * n - 1, dtype=np.float64)
        mask[argmax_idx] = 1.0
        # gradient = 2 * (rev(a) * mask) slice n elements
        grad = 2.0 * _fft_convolve(a[::-1], mask)[n - 1: n - 1 + n]
        # momentum
        momentum = beta * momentum + (1 - beta) * grad
        # step size diminishing
        eta = eta0 / np.sqrt(it)
        # update
        a = a - eta * momentum
        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # jitter occasionally
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)
    return best_seq
```

```python
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_max = b.max()
    best_score = 2.0 * n * best_max
    velocity = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    alpha = 5.0
    alpha_factor = 1.01
    alpha_max = 2000.0
    it = 0

        it += 1
```

The objective is  

\[
\min_{a\ge 0,\;\sum a = 1}\; \max_k (a*a)_k ,
\qquad a\in\mathbb R^n .
\]

`evaluate_sequence` returns  

\[
\text{score}=2\,n\;\max_k (a*a)_k .
\]

The problem is non‑convex, but a good local optimum can be reached by
alternating two complementary optimisation stages:

1. **Smooth‑max gradient descent** – the same idea as the reference
   implementation, but with a more aggressive annealing schedule,
   momentum and occasional jitter.  
   It quickly drives the distribution to a low‑peak region.

2. **Exact‑max sub‑gradient descent** – once the smoothed objective is
   close to the true peak we switch to the true sub‑gradient of the
   `max` (the gradient of the active convolution entry).  This fine‑tunes
   the shape and usually yields a small additional reduction of the
   peak.

Both stages keep the vector on the probability simplex by Euclidean
projection.  The algorithm is run for several lengths (the length of the
previous best construction and a few scaled variants) and for a few
initialisations (arcsine, uniform, random, the previous construction).
The best sequence seen over the whole run is returned.

The whole search respects the 1000 s budget, checks the deadline
regularly and stops as soon as the target score ≤ 1.503 is reached.

```python
"""
Propose a low‑score step function.

The search combines two phases:
  * A smoothed‑max gradient descent (soft‑max surrogate) with momentum,
    aggressive temperature increase and occasional jitter.
  * A fine‑tuning phase that uses the exact sub‑gradient of the
    max‑convolution (arg‑max of the self‑convolution) together with
    momentum.

Both phases operate on vectors normalised to sum = 1 and are projected
back onto the simplex after each update.  Several lengths (derived from
the previously found construction) and several initialisations are
explored within the 1000 s time budget.  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path for already feasible vectors
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Chebyshev) distribution of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b (real FFT, length a+b‑1)."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """Smoothed‑max descent (soft‑max surrogate) for a fixed time slice."""
    a = a0.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_max = b.max()

    velocity = np.zeros_like(a)
    beta = 0.9                     # momentum coefficient
    eta0 = 0.5                     # base step size
    alpha = 5.0                    # starting temperature
    alpha_factor = 1.010           # how fast we increase the temperature
    alpha_max = 2000.0

    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()
        # soft‑max weights (w sums to 1)
        exp_term = np.exp(alpha * (b - cur_max))
        w = exp_term / exp_term.sum()
        # gradient = 2 * (rev(a) * w) restricted to n entries
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    return best_seq


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """Exact‑max sub‑gradient descent (arg‑max of the convolution)."""
    a = a0.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_max = b.max()

    velocity = np.zeros_like(a)
    beta = 0.9          # momentum
    eta0 = 0.5          # base step size
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        if cur_max < best_max:
            best_max = cur_max
            best_seq = a.copy()

        # index of the active (max) convolution entry
        k_star = np.argmax(b)

        # mask with a single 1 at the active position
        mask = np.zeros(2 * n - 1, dtype=np.float64)
        mask[k_star] = 1.0

        # exact sub‑gradient: 2 * (rev(a) * mask) restricted to n entries
        grad = 2.0 * _fft_convolve(a[::-1], mask)[n - 1:n - 1 + n]

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # tiny jitter every few hundred iterations
        if it % 500 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    `evaluate_sequence` score.  The routine respects the time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    # leave a small safety margin for final printing
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Base length: try to reuse the previously discovered construction.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Lengths to explore – 0.8 ×, 1.0 ×, 1.2 ×, 1.5 ×, 2.0 × the base,
    # never smaller than 500.
    factors = (0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({max(500, int(base_n * f)) for f in factors})

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # give roughly 30 % of the remaining budget to this length
        slice_deadline = time.time() + max(0.1, (deadline - time.time()) * 0.30)

        # try a handful of initialisations
        for init_kind in ("prev", "arcsine", "uniform", "random"):
            if time.time() >= slice_deadline:
                break

            # ------------------- initialise ----------------------------
            if init_kind == "prev":
                try:
                    a = np.asarray(height_sequence_1, dtype=np.float64)
                    if a.size != n:
                        raise ValueError
                except Exception:
                    a = _arcsine_distribution(n)
                a = a / a.sum()
            elif init_kind == "arcsine":
                a = _arcsine_distribution(n)
            elif init_kind == "uniform":
                a = np.full(n, 1.0 / n, dtype=np.float64)
            elif init_kind == "random":
                a = np.random.rand(n)
                a /= a.sum()
            else:
                a = _arcsine_distribution(n)

            a = _project_onto_simplex(a, s=1.0)

            # ------------------- phase 1 : smooth descent -------------
            # split the slice time roughly half‑half between the two phases
            now = time.time()
            remaining_slice = slice_deadline - now
            phase1_deadline = now + remaining_slice * 0.45
            best_after_smooth = _smooth_gradient_descent(a.copy(), n, phase1_deadline)

            # ------------------- phase 2 : exact sub‑gradient ----------
            # use the global deadline (minus a tiny safety margin)
            phase2_deadline = min(slice_deadline, deadline - 0.5)
            candidate_seq = _subgradient_descent(best_after_smooth.copy(), n, phase2_deadline)

            # ------------------- evaluate -----------------------------
            try:
                cand_score = evaluate_sequence(candidate_seq.tolist())
            except Exception:
                # fallback – compute the score from the peak we stored
                cand_score = float("inf")

            if cand_score < overall_best_score:
                overall_best_score = cand_score
                overall_best_seq = candidate_seq.copy()
                print(
                    f"*** New global best – len={n} init={init_kind} "
                    f"score={overall_best_score:.6f}"
                )
                # early exit if we have already beaten the target
                if overall_best_score <= 1.503:
                    break

        # early break if target already reached
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    # Return plain Python list as required by the harness.
    return overall_best_seq.tolist()
```
