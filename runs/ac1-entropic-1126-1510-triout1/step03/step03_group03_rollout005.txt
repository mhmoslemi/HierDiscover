"""
Linear‐programming alternating descent for the step‑function problem.

We minimise 2·n·max(conv(a,a))/sum(a)² by keeping the ℓ¹‑norm of the
sequence equal to S = √(2·n).  Starting from a feasible a (taken from the
reference construction, a uniform vector, a V‑shape and a random vector)
the algorithm repeatedly:

1. computes M = max(a⋆a);
2. solves the linear programme
        maximise   1ᵀg
        subject to (a⋆g) ≤ M ,  g ≥ 0
   which is linear because a is frozen;
3. rescales the LP optimum g₀ to have ℓ¹‑norm S,
4. replaces a ← (1‑t)·a + t·g  with a small mixing factor t
   (the mixing keeps the ℓ¹‑norm unchanged).

The LP can be written as   A·g ≤ M·1   where A is the full convolution
matrix of a (Toeplitz).  For modest n (≈800–1200) the dense matrix fits easily
in memory and the HiGHS interior‑point solver finds g₀ within a fraction of a
second.  After a handful of mixing steps the maximum convolution decreases
monotonically; the best sequence found is returned.

A cheap random‑block search is performed at the end to polish the result.
All numbers are non‑negative; any tiny negatives introduced by numerical error
are projected back to the simplex.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.linalg as la

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the (2·n‑1)×n matrix B such that B @ x = np.convolve(a, x).
    Uses a dense Toeplitz representation.
    """
    n = a.size
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    return la.toeplitz(col, row)


# ----------------------------------------------------------------------
def _solve_lp_fixed_a(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise   1ᵀg   subject to   (a⋆g) ≤ M ,  g ≥ 0 .
    Returns the optimal g (not yet rescaled) or None if the LP fails.
    """
    n = a.size
    A = _conv_matrix(a)                # shape (2·n‑1, n)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises; maximise 1ᵀg  → minimise –1ᵀg
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # use HiGHS (fast for dense problems of this size)
    res = opt.linprog(
        c=c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return None
    return res.x
