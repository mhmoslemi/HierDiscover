"""
Propose a low‑score step‑function sequence.

This implementation builds on the previous reference algorithm but
adds a richer set of initial candidates, longer and more aggressive
exponentiated‑gradient (mirror‑descent) runs with a much larger soft‑max
parameter, a higher‑iteration sub‑gradient phase and an expanded hill‑climb.
The schedule for the temperature λ grows exponentially from a modest
starting value up to a large λ_max (≈200) which makes the smooth
approximation converge closely to the true max‑convolution peak.
All optimisation respects the simplex constraint Σa_i = 1 via the
entropy‑mirror (multiplicative) update and a final Euclidean projection.
The search explores several sequence lengths (4000‑10000) and stops
as soon as the target score 1.503 is reached or the time budget (≈1000 s)
is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    # Use log‑space to avoid overflow for very small α
    log_vals = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    vals = np.exp(log_vals)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑two length."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using an FFT of length m.
    w – weight vector (length 2·n‑1) from the soft‑max,
    a – current sequence (length n),
    m – power‑of‑two ≥ len(w)+len(a)-1.
    Returns g of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised (Σa = 1) vector."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())

def _initial_candidates(n: int) -> list[np.ndarray]:
    """
    Produce a diversified pool of probability vectors of length n.
    The pool contains arcsine, uniform, many symmetric Betas,
    mixtures of arcsine/uniform, Dirichlet draws and a few‑spike vectors.
    """
    cand = []

    # 1) arcsine
    cand.append(_arcsine_distribution(n))

    # 2) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) mixtures arcsine–uniform (weight on arcsine from 0.2 to 0.8)
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    for mix in (0.2, 0.4, 0.6, 0.8):
        cand.append(mix * arcs + (1.0 - mix) * uni)

    # 4) symmetric Betas on a fine grid of α (including very small α)
    alphas = [0.05, 0.08, 0.10, 0.12, 0.15, 0.18,
              0.22, 0.28, 0.35, 0.45, 0.55, 0.70,
              0.90, 1.20, 1.50]
    for alpha in alphas:
        cand.append(_beta_distribution(n, alpha))

    # 5) random symmetric Betas
    for _ in range(3):
        alpha = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, alpha))

    # 6) sparse spike vectors (≈0.2 % non‑zero entries)
    k = max(2, n // 500)
    idx = np.random.choice(n, size=k, replace=False)
    spike = np.zeros(n, dtype=np.float64)
    spike[idx] = np.random.rand(k) + 0.1
    cand.append(spike / spike.sum())

    # 7) Dirichlet draws with low concentration
    for conc in (0.05, 0.10, 0.30):
        a = np.random.dirichlet(np.full(n, conc))
        cand.append(a)

    # Normalise all candidates (numerical safety)
    normed = []
    for v in cand:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            normed.append(v / s)
    return normed

def _optimise_exponential_advanced(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 15000,
    eta0: float = 0.20,
    lambda0: float = 0.5,
    lambda_max: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Entropy‑mirror descent on a soft‑max approximation of max(conv(a,a)).
    The temperature λ grows exponentially from λ0 to λ_max, making the
    approximation increasingly sharp.  The update is multiplicative
    (exponentiated‑gradient) with a diminishing step size η_t = η0 / √t.
    Returns the best vector (sum 1) encountered and its exact score.
    """
    a = init_seq.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    # length needed for correlation: w (2n‑1) + a (n) ⇒ 3n‑2
    fft_len = 1 << ((3 * n - 2).bit_length())

    # Pre‑compute the λ growth factor
    if max_iters > 1:
        grow_factor = (lambda_max / lambda0) ** (1.0 / (max_iters - 1))
    else:
        grow_factor = 1.0
    lamb = lambda0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} EG it={it:5d} score={best_score:.6f}")

        # soft‑max weights (log‑sum‑exp trick)
        maxb = b.max()
        exp_arg = lamb * (b - maxb)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # gradient via correlation
        g = _gradient_via_correlation(w, a, fft_len)

        # step size
        eta = eta0 / np.sqrt(it)

        # exponentiated gradient update
        a = a * np.exp(-eta * g)

        # projection onto simplex (renormalisation)
        total = a.sum()
        if total > 0.0:
            a = a / total
        else:
            a = np.full(n, 1.0 / n)

        # occasional jitter to escape plateaus
        if it % 1200 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        # increase λ
        lamb = lamb * grow_factor
        if lamb > lambda_max:
            lamb = lambda_max

    return best_seq, best_score

def _subgradient_descent_advanced(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
    eta0: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Uses the sub‑gradient g_i = 2·a_{k*−i} where k* is an arg‑max of the
    convolution.  The step size decays as η0/√t.
    """
    a = init_seq.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # build sub‑gradient: g_i = 2·a_{k*−i} where defined
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = a[k_star - idx]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} SG it={it:5d} score={best_score:.6f}")

        # jitter
        if it % 800 == 0 and np.random.rand() < 0.18:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score

def _hill_climb_advanced(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 40000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random amount of mass
    from a donor index to a receiver index (up to 20 % of donor) and keep
    the change if the exact objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move between 0 % and 20 % of the donor (capped by 0.02 absolute)
        delta = np.random.rand() * min(best_seq[i] * 0.20, 0.02)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score

                best_n_seq = seq.copy()
                print(f"  >>> New best for n={n}: score={best_n_score:.6f}")

        # ------------------------------------------------------------------
        # b) sub‑gradient refinement (if time permits)
        if best_n_seq is not None and time.time() < deadline:
            sub_deadline = deadline - 2.0
            seq, sc = _subgradient_descent_advanced(
                n=n,
                init_seq=best_n_seq,
                deadline=sub_deadline,
                max_iters=5000,
                eta0=0.30,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  *** Sub‑grad refined n={n} score={best_n_score:.6f}")

        # ------------------------------------------------------------------
        # c) update global best
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # early exit if target already reached
        if overall_best_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # 3️⃣  final local hill‑climbing refinement on the global best
    if overall_best_seq is not None and time.time() < deadline:
        hill_deadline = deadline
        seq, sc = _hill_climb_advanced(overall_best_seq, hill_deadline, max_trials=40000)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** After hill‑climb: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  a last short EG polishing with an even larger λ_max (≈500)
    if overall_best_seq is not None and time.time() < deadline:
        polish_deadline = deadline
        seq, sc = _optimise_exponential_advanced(
            n=len(overall_best_seq),
            init_seq=overall_best_seq,
            deadline=polish_deadline,
            max_iters=8000,
            eta0=0.10,
            lambda0=1.0,
            lambda_max=500.0,
        )
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** After final polishing: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  fallback (should never trigger)
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"Final best: length={len(overall_best_seq)} score={overall_best_score:.6f}")

    # Convert to plain list for the evaluation harness
    return overall_best_seq.tolist()
