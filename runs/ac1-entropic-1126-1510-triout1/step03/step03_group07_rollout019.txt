"""
propose_candidate
-----------------
Solve the scale‑invariant optimisation problem

    minimise   2·n·max(conv(a,a)) / (∑a)²

with a ≥ 0, by fixing the ℓ₁‑norm to S = √(2·n) (then the objective reduces
to max(conv)).  The algorithm consists of

* a warm start (uniform or a previous solution, padded/truncated to the chosen
  length);
* a soft‑max approximation of the max‑convolution together with an Adam optimiser
  on the simplex (projection after each step);
* a polishing phase that mixes cheap random‑pair mass swaps with a *targeted*
  move that explicitly reduces the contribution to the current peak of the
  convolution.

The whole search respects the 1000 s wall‑time budget and returns the best
sequence found (as a plain list of floats).  All numbers are non‑negative,
well below the 1000‑clip, and the ℓ₁‑norm is exactly S, so the final
`evaluate_sequence` score equals the maximal convolution value reported during
optimisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple random mass‑swap that conserves the ℓ₁ norm
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two random indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() - 0.5) * sigma
    if delta > 0:                     # move from j to i
        delta = min(delta, seq[j])
    else:                             # move from i to j
        delta = -min(-delta, seq[i])
    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# FFT‑based full convolution (linear, not circular)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based correlation:  corr = np.correlate(w, a, mode='valid')
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                     # = 2n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length n (valid correlation)


# ----------------------------------------------------------------------
# Targeted polishing move – reduces the dominant contribution to the peak.
# ----------------------------------------------------------------------
def _targeted_move(seq: np.ndarray, sigma: float, peak_idx: int) -> np.ndarray:
    """
    Identify the i,j whose product a[i]*a[j] contributes most to the
    current maximum convolution entry `peak_idx` and move a small amount of
    mass away from those indices.
    """
    n = seq.shape[0]
    # indices that actually appear in this convolution entry
    if peak_idx < n:
        i_vals = np.arange(peak_idx + 1)
        j_vals = peak_idx - i_vals
    else:
        i_start = peak_idx - n + 1
        i_vals = np.arange(i_start, n)
        j_vals = peak_idx - i_vals
    contrib = seq[i_vals] * seq[j_vals]
    # index of the largest contribution
    best = np.argmax(contrib)
    i = int(i_vals[best])
    j = int(j_vals[best])

    # choose a third location far from i and j
    l = np.random.choice(n)
    while l == i or l == j:
        l = np.random.choice(n)

    # amount of mass to be moved – a small fraction of the smaller partner
    delta = sigma * min(seq[i], seq[j])

    new_seq = seq.copy()
    if i == j:                # diagonal term a[i]²
        new_seq[i] -= delta
        new_seq[l] += delta          # keep total sum unchanged
    else:
        new_seq[i] -= delta
        new_seq[j] -= delta
        new_seq[l] += 2.0 * delta    # preserve the ℓ₁ norm
    # keep non‑negativity (tiny rounding errors are later fixed by the projection)
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq
