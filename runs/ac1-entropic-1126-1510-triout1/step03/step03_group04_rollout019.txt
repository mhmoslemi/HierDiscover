def propose_candidate(seed:int=0,budget_s: int=1000):
    import time, numpy as np, math, random
    np.random.seed(seed)
    sleep...

    n = 4000
    target_sum = math.sqrt(2.0*n)
    best_max = float('inf')
    best_seq = None

    def max_conv(vec):
        n = len(vec)
        # Use next power of 2 >= 2n-1
        size = 1 << (2*n-1).bit_length()
        f = np.fft.rfft(vec, size)
        conv = np.fft.irfft(f * f, size)
        return float(conv[:2*n-1].max())

    # Phase 1: random block search
    while time.time() < deadline - 30:
        nb = np.random.randint(2, 30)  # number of blocks
        block_len = n // nb
        # For simplicity allocate equal block lengths; remainder assign to last block
        lens = np.full(nb, block_len, dtype=int)
        lens[-1] = n - block_len*(nb-1)
        # random heights
        heights = np.random.rand(nb)  # uniform random [0,1)
        # Scale to target sum
        cur_sum = (lens * heights).sum()
        if cur_sum == 0:
            continue
        factor = target_sum / cur_sum
        heights *= factor
        # Build sequence
        seq = np.repeat(heights, lens)
        # ensure length n
        seq = seq[:n]
        # compute max conv
        cur_max = max_conv(seq)
        if cur_max < best_max:

"""
Propose a step‐function with a low evaluation score.

The search works in three stages:

1. ***Exploratory block search***
   Randomly sample many piecewise–constant patterns (2‑30 blocks),
   project them onto the ℓ¹‑simplex Σa = √(2 n) and keep the best
   (the objective of the challenge equals the maximal self‑convolution
   when the sum constraint holds).

2. ***Simulated‑annealing refinement***
   Starting from the best block pattern, repeatedly move a random
   fraction of mass from one position to another.
   Moves are accepted if they improve the maximum convolution or with
   a Metropolis probability that slowly cools down.  Periodic global
   “shakes” (small Gaussian noise + projection) help escape plateaus.

3. ***Final greedy hill‑climb***
   A short pure‑improvement loop swaps a random amount of mass between
   two indices while keeping the ℓ¹‑norm fixed.  Any improvement is kept.

All operations respect the non‑negative simplex, use FFT‑based O(n log n)
convolution for speed, and stay well within the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max entry of the (full) self‑convolution of a using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # the convolution length is 2·len(a)‑1; only the first part matters
    return float(conv[: 2 * a.shape[0] - 1].max())

        before the limit to avoid a hard timeout.

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    DEFAULT_N = 5000                      # a size large enough for a good bound
    try:
        ref_len = len(height_sequence_1)   # type: ignore[name-defined]
        n = max(DEFAULT_N, ref_len)
    except Exception:
        n = DEFAULT_N

    target_sum = math.sqrt(2.0 * n)       # we keep Σa = √(2 n)
    fft_len = 1 << ((2 * n - 1).bit_length())   # next power of two ≥ 2n‑1

    # ------------------------------------------------------------------
    # Helper ------------------------------------------------------------
    def max_conv(vec: np.ndarray) -> float:
        return _max_convolution(vec, fft_len)

    # ------------------------------------------------------------------
    # 1) Random block search (many cheap samples) ----------------------
    # start from uniform (M = 2) just to have a baseline
    a = np.full(n, target_sum / n, dtype=float)
    best_a = a.copy()
    best_score = max_conv(best_a)
    print(f"[init] uniform start, score = {best_score:.6f}")

    block_trials = 20000
    for _ in range(block_trials):
        # number of constant blocks
        nb = np.random.randint(2, 30)
        # random cut points that define the blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        heights = np.random.rand(nb)                     # raw heights
        cand = np.empty(n, dtype=float)
        for blk, h in zip(blocks, heights):
            cand[blk] = h
        # project onto the simplex (preserves feasibility)
        cand = cand * (target_sum / cand.sum())
        sc = max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing refinement ---------------------------------
    a = best_a.copy()
    cur_score = best_score
    # temperature schedule – start relatively hot and cool slowly
    temperature = best_score * 0.5
    cooling = 0.99995          # per‑iteration cooling factor
    it = 0

    while time.time() < deadline - 5:
        # pick a source index with positive mass
        nz = np.flatnonzero(a > 1e-12)
        if nz.size == 0:
            break
        i = np.random.choice(nz)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # transfer a random fraction of the source mass
        frac = np.random.uniform(0.05, 0.30)
        delta = a[i] * frac
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = max_conv(cand)

        # accept if better, otherwise with Metropolis probability
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / (temperature + 1e-12)):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[anneal] it={it:07d} new best = {best_score:.6f}")

        # temperature cooling
        temperature = max(temperature * cooling, 1e-9)
        it += 1

        # occasional global “shake” to escape shallow minima
        if it % 5000 == 0:
            noise = np.random.randn(n) * target_sum * 1e-4
            shaken = a + noise
            shaken = _project_to_simplex(shaken, target_sum)
            a = shaken
            cur_score = max_conv(a)

    # ------------------------------------------------------------------
    # 3) Final greedy hill‑climb (pure improvements) --------------------
    while time.time() < deadline:
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size == 0:
            break
        i = np.random.choice(nz)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # move a modest fraction of mass
        frac = np.random.uniform(0.10, 0.40)
        delta = best_a[i] * frac
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = max_conv(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[final] improved best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
