   m = np.zeros_like(a)
   v = np.zeros_like(a)
   lam = 0.5
   lam_max = 100
   lam_inc = (lam_max - lam) / max_iters
   for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute convolution
        conv = _conv_full_fft(a)
        maxb = conv.max()
        # compute softmax weights
        exp_arg = lam * (conv - maxb)
        w = np.exp(exp_arg - np.max(exp_arg))  # to avoid overflow
        w_sum = w.sum()
        w = w / w_sum
        # gradient
        g = _gradient_via_correlation(w, a, m_fft)  # returns 2 * corr
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # maybe add jitter small probability
        # update lam
        lam = min(lam_max, lam + lam_inc)
        # track best sequence
        score = _score(a)
        if score < best_score: update.

def _optimise_adam(n, init_seq, deadline,
    max_iters=8000, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
    lambda0=0.5, lambda_max=100.0):
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)
    # Precompute FFT size
    m_fft = 1 << ((3 * n - 2).bit_length())
    m = np.zeros_like(a)  # momentum
    v = np.zeros_like(a)  # second moment
    lam = lambda0
    lam_inc = (lambda_max - lambda0) / max_iters
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = _conv_full_fft(a)
        maxb = conv.max()
        # softmax weights
        exp_arg = lam * (conv - maxb)   # <=0
        # numerical stability
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum
        # gradient via correlation (2 * corr)
        g = _gradient_via_correlation(w, a, m_fft)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # track best
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} score={best_score:.6f}")
        # jitter occasional
        if it % 1200 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
        lam = min(lambda_max, lam + lam_inc)
    return best_seq, best_score

def _hill_climb(a, deadline, max_trials=40000):

    best_score = _score(best_seq)
    for t in range(max_trials):

        i = np.argmax(best_seq)

        j = np.random.randint(n-1)

            j += 1

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

            best_seq = trial
            # print occasional

"""
Propose a low‑score step‑function.

Algorithm (≈ 1000 s budget)
--------------------------------
1. Load historic sequence (if present) → baseline.
2. For several lengths (≈ 5‑15 k) build a diverse pool of start vectors:
   • arcsine law, uniform.
   • symmetric and asymmetric Beta(α,β) densities (many (α,β)).
   • power‑law i⁻ᵖ, exponential e^{‑γi}, triangular.
   • random Dirichlet (low concentration) and a few‑spike vectors.
3. Refine each start vector with a **Adam** optimiser on a soft‑max
   approximation of `max(conv(a,a))`.  λ (soft‑max temperature) is
   linearly increased up to a large value (≈ 100) so that the objective
   approaches the true ℓ\_∞ norm.
4. A short projected sub‑gradient descent on the exact peak is applied
   for further polishing.
5. Keep the best vector for the current length; update the global best.
6. After all lengths (or after reaching the target) run an aggressive
   hill‑climbing phase that repeatedly moves a small amount of mass from
   the largest coordinate to a random one, accepting only improvements.
7. Return the best sequence (as a plain list).  All vectors are kept
   non‑negative and normalised to sum = 1; the evaluator’s scaling invariance
   guarantees that the returned list is valid.

The implementation uses only NumPy, works in O(n log n) per iteration via
FFT, and respects the global time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g = 2·corr(w, a) where w has length 2·n‑1,
    a has length n and m is an FFT length (power of two).
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector a with Σa = 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points, normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float = None) -> np.ndarray:
    """Symmetric if beta=None, otherwise asymmetric Beta(α,β)."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

    i = np.arange(n, dtype=np.float64)

    vals /= vals.sum()
    return vals

    i = np.arange(n, dtype=np.float64)

def _optimise_adam(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 6000,
    lr: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    lambda0: float = 0.5,
    lambda_max: float = 120.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of max(conv(a,a)).
    Returns the best vector (Σa = 1) observed and its true unsmoothed score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT size – a safe upper bound for both conv and correlation
    m_fft = 1 << ((3 * n - 2).bit_length())

    # Adam moment buffers
    m_buf = np.zeros_like(a)
    v_buf = np.zeros_like(a)

    lam = lambda0
    lam_inc = (lambda_max - lambda0) / max_iters

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a)
        maxb = conv.max()

        # soft‑max weights (numerically stable)
        exp_arg = lam * (conv - maxb)          # ≤ 0, max = 0
        w = np.exp(exp_arg)                   # max weight is 1
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        g = _gradient_via_correlation(w, a, m_fft)

        # Adam update
        m_buf = beta1 * m_buf + (1.0 - beta1) * g
        v_buf = beta2 * v_buf + (1.0 - beta2) * (g * g)
        m_hat = m_buf / (1.0 - beta1 ** it)
        v_hat = v_buf / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the probability simplex
        a = _project_onto_simplex(a, s=1.0)

        # bookkeeping
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 1200 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # increase temperature
        lam = min(lambda_max, lam + lam_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 3000,
    eta0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact peak of the self‑convolution.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a)
        k_star = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k*‑i} where the index is valid
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = a[k_star - idx]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        # jitter
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Heuristic local search: repeatedly move a tiny amount of mass from the
    largest coordinate to a random other coordinate, accepting only improvements.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.argmax(best_seq)                # donor (largest entry)
        if best_seq[i] < 1e-14:
            continue

        # pick a different receiver uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # amount to transfer – up to 5 % of donor (capped)
        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.02)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            if t % 2000 == 0:
                print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a very low `evaluate_sequence`
    score.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣ historic sequence, if available
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣ candidate lengths (larger lengths often give a slightly smaller score)
    candidate_lengths = [5000, 7000, 9000, 11000, 13000, 15000]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        print(f"--- Exploring length n={n} ---")
        init_pool = _initial_candidates(n)

        best_n_seq = None
        best_n_score = float("inf")

        # a) Adam optimisation on many starts
        for init in init_pool:
            if time.time() >= deadline:
                break
            seq, sc = _optimise_adam(
                n=n,
                init_seq=init,
                deadline=deadline,
                max_iters=6000,
                lr=0.08,
                lambda0=0.5,
                lambda_max=120.0,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"   >>> New best for n={n}: score={best_n_score:.6f}")

        # b) sub‑gradient refinement (if time permits)
        if best_n_seq is not None and time.time() < deadline:
            seq, sc = _subgradient_descent(
                n=n,
                init_seq=best_n_seq,
                deadline=deadline,
                max_iters=3000,
                eta0=0.3,
            )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"   *** Sub‑grad refined n={n} score={best_n_score:.6f}")

        # c) update global best
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # early exit if we already beat the target
        if overall_best_score <= 1.503:
            print("Target reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # 3️⃣ final hill‑climbing refinement on the global best
    if overall_best_seq is not None and time.time() < deadline:
        hill_deadline = deadline
        seq, sc = _hill_climb(overall_best_seq, hill_deadline, max_trials=30000)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** After hill‑climb: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ fallback (should never be needed)
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"Final best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    # Return as a plain Python list for the evaluation harness
    return overall_best_seq.tolist()
