   otherwise from a uniform vector, and also evaluate a small collection
   of simple parametric families (inverse‑sqrt, power‑law, exponential,
   cosine, random block) projected onto the simplex Σa = √(2·n).

2. **Smooth max optimisation** – minimise a smooth approximation of the
   L∞‑norm of the self‑convolution.  For a temperature τ we use the
   soft‑max   f_τ(a) = τ·log Σ_k exp((a∗a)_k / τ).  Its gradient is a
   correlation of the soft‑max weights with the current vector and can be
   computed with two FFT‑based convolutions per iteration.  An Adam optimiser
   with Euclidean projection onto the simplex is applied, while τ is
   exponentially cooled towards 0 (the sub‑gradient of the true max).

3. **Escapes & restarts** – every few thousand steps a small Gaussian kick
   is added; if no improvement is seen for many iterations a random
   block‑structured restart is performed.

4. **Final block refinement** – a handful of random block candidates are
   evaluated around the best solution found.

The routine respects a 1000 s wall‑clock budget, prints intermediate scores
and returns the best sequence as a plain Python list of floats.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n−1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _max_convolution_value(a: np.ndarray, size_fft: int) -> float:
    """Return max entry of a*a (the quantity returned by evaluate_sequence after normalisation)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Return the soft‑max probabilities of conv / tau (stable version)."""
    # conv is a 1‑D array of length 2n‑1
    max_val = conv.max()
    shifted = conv - max_val            # now max(shifted) == 0
    exp_shift = np.exp(shifted / tau)
    total = exp_shift.sum()
    if total == 0:
        # degenerate case – fall back to a hard max weight
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w
    return exp_shift / total


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     tau: float,
                     size_fft_corr: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate f_τ(a) = τ·log Σ exp(conv/τ)
    w.r.t. a.  It equals 2·correlation(p, a), where p are the soft‑max
    weights.  Returned gradient has the same shape as a.
    """
    n = a.shape[0]
    p = _softmax_weights(conv, tau)                     # shape (2n‑1,)
    a_rev = a[::-1]                                     # reversed a
    conv_pa = _conv_fft(p, a_rev, size_fft_corr)        # length 3n‑2
    conv_len = (2 * n - 1) + n - 1                       # = 3n‑2
    conv_pa = conv_pa[:conv_len]
    # grad_i = 2 * conv_pa[n‑1‑i]  for i = 0,…,n‑1
    grad = 2.0 * conv_pa[(n - 1) - np.arange(n)]
    return grad

        sc = _true_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    # --- random block candidates ---
    for _ in range(80):
        cand = _random_block_candidate(n, target_sum)
        sc = _true_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    print(f"[seed] best after families = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. SMOOTH‑MAX ADAM OPTIMISATION
    # ------------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    lr0 = 0.25                     # initial learning rate
    tau0 = 0.30                    # initial temperature for soft‑max
    tau_min = 1e-7
    tau_decay = 0.9998            # exponential cooling per iteration

    it = 0
    no_improve = 0
    max_iters = 250_000           # safety cap; loop also stops on time

    while time.time() < deadline and it < max_iters:
        # current true score
        conv_full = _self_convolution(a, size_fft_self)
        cur_score = float(conv_full.max())

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[{it:07d}] new best  = {best_score:.6f}")
        else:
            no_improve += 1

        # temperature for this iteration
        tau = max(tau_min, tau0 * (tau_decay ** it))

        # gradient of the smooth objective
        grad = _gradient_softmax(a, conv_full, tau, size_fft_corr)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / (1.0 + 1e-4 * it)          # slowly decaying step size
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick to escape plateaus
        if (it % 6000 == 0 and it > 0) or no_improve > 8000:
            eps_noise = target_sum * 5e-4
            a_new = a_new + np.random.randn(n) * eps_noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[{it:07d}] random perturbation")

        # restart if we have been stuck for very long
        if no_improve > 12000:
            a_new = _random_block_candidate(n, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0
            print(f"[{it:07d}] block restart")

        a = a_new
        it += 1

    print(f"[adam] finished after {it} iterations, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM‑BLOCK REFINEMENT (pure evaluation)
    # ------------------------------------------------------------------
    for _ in range(300):
        cand = _random_block_candidate(n, target_sum)
        sc = _true_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. LOCAL SMALL‑PERTURBATION REFINEMENT
    # ------------------------------------------------------------------
    a = best_a.copy()
    for _ in range(200):
        # pick a handful of indices and move a tiny amount of mass
        idxs = np.random.choice(n, size=8, replace=False)
        delta = np.random.randn(8) * target_sum * 2e-4
        a2 = a.copy()
        a2[idxs] += delta
        a2 = _project_to_simplex(a2, target_sum)
        sc = _true_score(a2)
        if sc < best_score:
            best_score = sc
            best_a = a2.copy()
            a = a2
            print(f"[local] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
