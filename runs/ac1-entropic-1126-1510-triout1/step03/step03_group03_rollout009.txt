def _run_continuation(y0, deadline):
    global best_a, best_score
    y = y0.copy()
    # initial mu
    a_cur = S * _softmax(y)
    mu = float(np.convolve(a_cur, a_cur).max())
    if mu <=0: mu = 1.0
    mu_factor = 0.8

    maxiter_per_mu = 200
    while mu > mu_min:
        if time.time() > deadline: break
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(obj, y, method='L-BFGS-B', jac=True,
                options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False})
        except Exception as e:
            print("[error] L-BFGS failed:", e)
            break
        y = res.x
        # evaluate
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print("[new best] score = {:.6f}".format(best_score))
        # cool
        mu *= mu_factor
    return y

"""
Propose a low‐score non‑negative step function.

The algorithm combines several ideas:
* many independent starts (uniform, monotone, V‑shaped, random, reference);
* a continuation scheme that minimises a smooth approximation of the
  hard max convolution using L‑BFGS‑B on the softmax parametrisation;
* cheap random block constructions for extra diversity;
* a long final projected sub‑gradient descent on the true max‑convolution.
All work is performed on the simplex Σa_i = √(2·n) (the scaling does not
change the evaluation value).  The best sequence found before the time
budget expires is returned.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (or hard max if mu==0) and its gradient w.r.t. y.
    The underlying variable is a = S·softmax(y) which is already on the
    required simplex (ℓ¹‑norm = S).
    """
    # probabilities and heights
    p = _softmax(y)                # Σp = 1, p ≥ 0
    a = S * p                      # Σa = S

    # full (linear) convolution a*a
    conv = np.convolve(a, a)       # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:                     # smooth max
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)     # overflow guard
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                            # hard max: uniform weight on all maximisers
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a (see derivation in the description)
    # conv_wr = w * a (convolution) with a reversed
    conv_wr = np.convolve(w, a[::-1])           # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]        # slice length n

    # chain rule: ∂f/∂y = S * p * (g_a - (p·g_a))
    inner = np.dot(g_a, p)                      # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize: returns (value, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The search consists of many restarts of a
    continuation L‑BFGS‑B optimiser, a cheap random block exploration and
    a long projected sub‑gradient polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # problem size (use the reference if present)
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    S = math.sqrt(2.0 * n)                 # ℓ¹‑norm that cancels the outer factor

    # bookkeeping
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # function to update the global best
    def _maybe_update(a_vec: np.ndarray):
        nonlocal best_a, best_score
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # generate a pool of starting points (y‑vectors for the soft‑max)
    init_ys = []

    # 0. uniform (y = 0)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1. V‑shaped (large at ends, tiny in the centre)
    idx = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    vshape = np.abs(idx - mid) + 1e-3
    init_ys.append(np.log(vshape))

    # 2. monotone decreasing (large left, small right)
    init_ys.append(-np.arange(n, dtype=np.float64))

    # 3. monotone increasing (large right, small left)
    init_ys.append(np.arange(n, dtype=np.float64))

    # 4. random Gaussian
    init_ys.append(np.random.randn(n) * 0.5)

    # 5. random Dirichlet‑like (log of a random simplex point)
    p_rand = np.random.rand(n)
    p_rand /= p_rand.sum()
    init_ys.append(np.log(p_rand + 1e-12))

    # 6. reference if available
    if have_ref:

        eps = 1e-12
        init_ys.append(np.log(a_ref + eps))

    # ------------------------------------------------------------------

    start_counter = 0

        a_cur = S * _softmax(y)
        mu = float(np.convolve(a_cur, a_cur).max())
        if mu <= 0.0:

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:

                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print("[error] L‑BFGS failed:", exc)
                break

            y = res.x

            # evaluate the *hard* max objective
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # ------------------------------------------------------------------
    # If nothing good yet, fall back to uniform (should never happen)
    if best_score == float("inf"):
        best_a = S * np.ones(n) / n
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # Random block search – cheap but adds diversity
    remaining = deadline - time.time()
    if remaining > 3.0:
        block_trials = int(min(4000, remaining * 150))
        print(f"[block] performing {block_trials} random block trials")
        for _ in range(block_trials):
            nb = np.random.randint(2, 9)                     # 2‑8 blocks
            cuts = np.sort(
                np.random.choice(n - 1, nb - 1, replace=False) + 1
            )
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)

            # a few cheap sub‑gradient steps to polish the block candidate
            a_tmp = cand
            for __ in range(5):
                conv = np.convolve(a_tmp, a_tmp)
                max_c = conv.max()
                mask = conv == max_c
                cnt = mask.sum()
                w = np.zeros_like(conv)
                w[mask] = 1.0 / cnt
                conv_wr = np.convolve(w, a_tmp[::-1])
                g_a = 2.0 * conv_wr[n - 1:n - 1 + n]
                eta = S * 0.2 / n
                a_tmp = a_tmp - eta * g_a
                a_tmp = _project_to_simplex(a_tmp, S)
            _maybe_update(a_tmp)

    # ------------------------------------------------------------------
    # Final polishing using projected sub‑gradient descent on the hard max
    print("[polish] starting final projected sub‑gradient refinement")
    a = best_a.copy()
    eta0 = S * 0.5 / n     # initial step size roughly the average entry
    decay = 5e-5
    it = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        conv_wr = np.convolve(w, a[::-1])
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        _maybe_update(a_new)

        a = a_new
        it += 1
        if it % 1000 == 0:
            print(f"[polish] iteration {it}, best score {best_score:.6f}")
        if it > 10000:          # safety cap
            break

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
