def upsample(seq, n_new):
    xx = (np.arange(len(seq)) + 0.5) / len(seq)
    yy = seq
    xx_new = (np.arange(n_new) + 0.5) / n_new
    new_seq = np.interp(xx_new, xx, yy, left=0, right=0)
    new_seq = np.maximum(new_seq, 0)
    new_seq /= new_seq.sum()

def _adam_optimize(n, init_seq, deadline, max_iters=3000, lam_start=0.5, lam_end=40.0, alpha=0.02):
    a = np.maximum(init_seq.astype(np.float64), 0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    lam = lam_start
    lam_inc = (lam_end - lam_start) / max_iters
    # compute FFT length for gradient correlation
    len_w = 2*n - 1
    fft_len = 1 << ((len_w + n - 1).bit_length())  # next power of two >= 3n-2
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = _conv_full_fft(a)
        maxc = conv.max()
        # compute softmax weights
        exp_arg = lam * (conv - maxc)
        # safe exponent
        # To avoid overflow, we can shift by max; already done
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum
        # gradient
        g = _gradient_via_correlation(w, a, fft_len)
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # update lambda
        lam = min(lam_end, lam + lam_inc)
        # evaluate exact score occasionally
        if it % 10 == 0:
            cur_score = _score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n} it={it} lam={lam:.2f} score={best_score:.6f}")

def _upsample(seq, n_new):
    n_old = len(seq)
    if n_old == n_new:
        return seq.copy()
    old_x = (np.arange(n_old) + 0.5) / n_old
    new_x = (np.arange(n_new) + 0.5) / n_new
    new_seq = np.interp(new_x, old_x, seq, left=0.0, right=0.0)
    # ensure non-negative
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0:
        new_seq = np.full(n_new, 1.0 / n_new)
    else:
        new_seq /= s
    return new_seq
```

def _project_onto_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64)
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
```

def _gradient_via_correlation(w, a, m):
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start+len(a)]

"""
Search for a non‑negative step‑function that minimises the
`evaluate_sequence` score.

The algorithm works in three stages:

1. **Diverse initialisations** – arcsine law, symmetric Beta(α,α) with a
   fine grid of α (including very small values), asymmetric Beta,
   Dirichlet draws with low concentration, sparse spike vectors and a
   few mixtures.

2. **Adam optimisation** – minimise a smooth soft‑max approximation of
   the peak of the self‑convolution.  The smoothing parameter λ is
   increased linearly from λ₀ to λ_max while an Adam optimiser with
   simplex projection updates the probabilities.

3. **Refinement** – a cheap projected sub‑gradient descent on the exact
   peak of the convolution followed by a stochastic hill‑climbing
   phase.  The best vector found over several problem sizes (up to
   15 000 entries) is returned as a plain Python list.

All helper functions are defined at the top level, there are no
closures, and the search respects the global time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    v = np.asarray(v, dtype=np.float64)
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns an array of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum 1."""
    n = len(a)
    conv = _conv_full_fft(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution_general(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on the same grid (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a small, diverse set of initial probability vectors for size n."""
    cand = []

    # 1) arcsine
    cand.append(_arcsine_distribution(n))

    # 2) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3) symmetric betas on a fine grid, including very small alphas
    for alpha in (0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.45,
                  0.55, 0.65, 0.75, 0.9, 1.2, 1.5):
        cand.append(_beta_distribution(n, alpha))

    # 4) a few random asymmetric betas
    for _ in range(4):
        a = np.random.uniform(0.05, 2.0)
        b = np.random.uniform(0.05, 2.0)
        cand.append(_beta_distribution_general(n, a, b))

    # 5) Dirichlet draws with low concentration (many tiny entries)
    for conc in (0.2, 0.3, 0.5):
        cand.append(np.random.dirichlet(np.full(n, conc)))

    # 6) a sparse “spike’’ vector (very few non‑zero entries)
    k = max(2, n // 500)               # roughly 0.2 % of positions
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1   # avoid exact zeros on selected spots
    cand.append(a / a.sum())

    # 7) mixture of arcsine and uniform (0.7/0.3)
    mix = 0.7 * _arcsine_distribution(n) + 0.3 * np.full(n, 1.0 / n)
    mix /= mix.sum()
    cand.append(mix)

    # 8) another sparse vector with a slightly different pattern
    k2 = max(3, n // 400)
    idx2 = np.random.choice(n, size=k2, replace=False)
    a2 = np.zeros(n, dtype=np.float64)
    a2[idx2] = np.random.rand(k2) + 0.05
    cand.append(a2 / a2.sum())

    return cand


# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a normalised distribution onto a finer grid."""
    n_old = len(seq)
    if n_old == n_new:
        return seq.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    new_seq = np.interp(x_new, x_old, seq, left=0.0, right=0.0)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        new_seq = np.full(n_new, 1.0 / n_new, dtype=np.float64)
    else:
        new_seq /= s
    return new_seq


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2500,
    lam_start: float = 0.5,
    lam_end: float = 35.0,
    alpha: float = 0.025,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a soft‑max approximation of max(conv(a,a)).
    Returns the best probability vector (sum 1) seen and its exact score.
    """
    a = np.maximum(init_seq.astype(np.float64), 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing schedule
    lam = lam_start
    lam_inc = (lam_end - lam_start) / max_iters

    # FFT size for the correlation (w length 2n‑1, a length n)
    len_w = 2 * n - 1
    fft_len = 1 << ((len_w + n - 1).bit_length())   # next power of two ≥ 3n‑2

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- soft‑max of the convolution ---
        conv = _conv_full_fft(a)
        maxc = conv.max()
        exp_arg = lam * (conv - maxc)          # ≤0, avoids overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.shape[0])
        else:
            w /= w_sum

        # --- gradient of the smooth objective ---
        g = _gradient_via_correlation(w, a, fft_len)

        # --- Adam update ---
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # projection onto the simplex (keeps sum 1 and non‑negativity)
        a = _project_onto_simplex(a, s=1.0)

        # increase smoothing parameter
        lam = min(lam_end, lam + lam_inc)

        # occasional tiny jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, s=1.0)

        # evaluate exact objective every few iterations
        if it % 50 == 0:
            cur = _score(a)
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] n={n:5d} it={it:5d} λ={lam:.2f} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 800,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Returns the best sequence seen and its exact score.
    """
    a = np.maximum(init_seq.astype(np.float64), 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a.sum()

    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a)
        k_star = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k*−i} for indices where it is defined
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = a[k_star - idx] * 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur = _score(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        # jitter occasionally
        if it % 200 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny mass from one index
    to another and keep the change if the (unsmoothed) objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a very low
    `evaluate_sequence` score.

    The routine tries several problem sizes (up to 15000), runs a
    multi‑restart Adam optimiser on many initial distributions,
    refines the best vectors with a projected sub‑gradient method and a
    final hill‑climbing phase, and returns the best candidate as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # keep a small safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 0) historic sequence, if it exists
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1) explore a few problem sizes
    candidate_lengths = [6000, 10000, 15000]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        print(f"--- Exploring length n={n} ---")

        # pool of starting points
        pool = _initial_candidates(n)

        # also try to up‑sample the current global best (if any)
        if overall_best_seq is not None and len(overall_best_seq) != n:
            pool.append(_upsample(overall_best_seq, n))

        best_local_seq: np.ndarray | None = None
        best_local_score = float("inf")

        # adaption of iteration budget depending on n
        max_iters = 2500 if n <= 8000 else 1500

        # ------------------------------------------------------------------
        # a) Adam optimisation from each start point
        for init in pool:
            if time.time() >= deadline:
                break
            seq, sc = _adam_optimize(
                n=n,
                init_seq=init,
                deadline=deadline,
                max_iters=max_iters,
                lam_start=0.5,
                lam_end=35.0,
                alpha=0.025,
            )
            if sc < best_local_score:
                best_local_score = sc
                best_local_seq = seq.copy()
                print(f"   >>> new local best for n={n}: score={best_local_score:.6f}")

        # ------------------------------------------------------------------
        # b) optional sub‑gradient refinement (quick, cheap)
        if best_local_seq is not None and time.time() < deadline:
            sub_deadline = deadline - 0.5
            seq, sc = _subgradient_descent(
                n=n,
                init_seq=best_local_seq,
                deadline=sub_deadline,
                max_iters=800,
                eta0=0.25,
            )
            if sc < best_local_score:
                best_local_score = sc
                best_local_seq = seq.copy()
                print(f"   *** sub‑grad refined for n={n}: score={best_local_score:.6f}")

        # ------------------------------------------------------------------
        # c) update global best
        if best_local_seq is not None and best_local_score < overall_best_score:
            overall_best_score = best_local_score
            overall_best_seq = best_local_seq.copy()
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # early exit if target is already met
        if overall_best_score <= 1.503:
            print("Target reached – exiting early.")
            break

        # ------------------------------------------------------------------
        # d) final cheap hill‑climb on the current global best
        if overall_best_seq is not None and time.time() < deadline:
            seq, sc = _hill_climb(overall_best_seq, deadline, max_trials=12000)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = seq.copy()
                print(f"*** After hill‑climb: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback (should never be needed)
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"Final best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    return overall_best_seq.tolist()
