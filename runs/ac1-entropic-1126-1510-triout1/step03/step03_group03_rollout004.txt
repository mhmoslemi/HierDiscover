scaled to ‖a‖₁ = √(2n) (the scaling does not change the score).  The
procedure works for any n; if a reference sequence ``height_sequence_1``
is present we ignore its length and use a larger problem size (≥ 5000)
to get a tighter theoretical bound.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fast_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a power‑of‑2 FFT (real inputs)."""
    n = a.shape[0] + b.shape[0] - 1
    size = 1 << (n - 1).bit_length()          # next power of two
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    fc = fa * fb
    conv = np.fft.irfft(fc, size)[:n]
    return conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # choose problem size (larger n gives a tighter theoretical lower bound)
    try:
        n_ref = len(height_sequence_1)        # type: ignore[name-defined]
    except Exception:
        n_ref = 0
    n = max(5000, n_ref)                     # at least 5 000 entries
    print(f"[setup] problem size n = {n}")

    # Target ℓ¹‑norm (used only for final scaling)
    S_target = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # bookkeeping
    best_p = None               # probabilities (sum = 1)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # ---------- Phase 1: variance (flattening) optimisation ------------
    restart = 0
    while time.time() < deadline and restart < 30:
        restart += 1

        # initialise a candidate on the simplex
        if restart % 5 == 0:
            # a simple V‑shape (large at the ends, small in the centre)
            idx = np.arange(n, dtype=float)
            mid = (n - 1) / 2.0
            vshape = np.abs(idx - mid) + 1e-3
            p = vshape / vshape.sum()
        else:
            p = np.random.rand(n)
            p = p / p.sum()

        # gradient descent on variance
        for it in range(2000):
            if time.time() > deadline:
                break

            # convolution a*a
            conv = _fast_convolve(p, p)               # length 2n‑1
            max_c = conv.max()
            score = 2.0 * n * max_c                    # exact objective for Σa=1

            # record best
            if score < best_score:
                best_score = score
                best_p = p.copy()
                print(f"[restart {restart}] iter {it} new best = {best_score:.6f}")

            # variance gradient
            avg = 1.0 / (2 * n - 1)
            r = conv - avg
            grad = 2.0 * _fast_convolve(r, p[::-1])[n - 1:n - 1 + n]

            # simple diminishing step size
            step = 0.6 / math.sqrt(it + 1.0)
            p_new = p - step * grad
            p_new = _project_to_simplex(p_new, 1.0)

            # stop if change is negligible
            if np.linalg.norm(p_new - p) < 1e-12:
                break
            p = p_new

    # ------------------------------------------------------------------
    # If we never found anything, fall back to uniform distribution
    if best_p is None:
        best_p = np.ones(n, dtype=float) / n
        best_score = 2.0 * n * (_fast_convolve(best_p, best_p).max())
        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # ---------- Phase 2: sub‑gradient descent on the hard max ----------
    p = best_p.copy()
    it = 0
    while time.time() < deadline and it < 5000:
        conv = _fast_convolve(p, p)
        max_c = conv.max()
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt

        # sub‑gradient of the max‑convolution
        grad_max = 2.0 * _fast_convolve(w, p[::-1])[n - 1:n - 1 + n]

        # decreasing step size
        eta = 0.5 / (1.0 + it * 0.001)
        p_new = p - eta * grad_max
        p_new = _project_to_simplex(p_new, 1.0)

        # evaluate new score
        new_score = 2.0 * n * max_c
        if new_score < best_score:
            best_score = new_score
            best_p = p_new.copy()
            print(f"[refine] iter {it} new best = {best_score:.6f}")

        p = p_new
        it += 1

    # ------------------------------------------------------------------
    # ---------- Phase 3: cheap random block search (if time left) -----
    remaining = deadline - time.time()
    if remaining > 2.0:
        trials = int(min(3000, remaining * 120))
        print(f"[block] performing {trials} random block trials")
        for _ in range(trials):
            nb = np.random.randint(2, 8)  # between 2 and 7 blocks
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, 1.0)
            conv = _fast_convolve(cand, cand)
            cand_score = 2.0 * n * conv.max()
            if cand_score < best_score:
                best_score = cand_score
                best_p = cand.copy()
                print(f"[block improve] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the required ℓ¹‑norm (does not change the score)
    best_a = best_p * S_target
    # Ensure the values respect the evaluator's clipping bounds
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"[final] best score (theoretical) = {best_score:.6f}")
    return best_a.tolist()
