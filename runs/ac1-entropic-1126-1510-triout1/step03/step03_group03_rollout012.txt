"""
Hybrid optimizer for the step‑function bound.

The algorithm proceeds in several phases:

1. **Warm‑start** – the provided reference sequence (if any) is projected
   onto the simplex Σa_i = √(2 n). Its score is taken as the current best.

2. **Soft‑max L‑BFGS continuation** – a smooth‑max surrogate of the
   ℓ∞‑convolution is minimised from a few diverse initial points.
   This quickly descends to a good local optimum.

3. **Projected Adam (sub‑gradient) descent** – the exact hard‑max objective
   (the maximal convolution entry) is non‑smooth but its sub‑gradient is
   known analytically.  Adam‑style adaptive steps are applied, always
   re‑projected onto the simplex.

4. **Periodic LP improvement** – given the current shape a, we solve a
   linear program (SciPy `linprog` with the HiGHS solver) that maximises
   Σb_j subject to the convolution bound a∗b ≤ M (with M the current
   maximal entry).  The optimum b is rescaled to the required ℓ¹‑norm and
   mixed with a.  This step is cheap (a single LP) but often yields a
   noticeable drop in the objective.

5. **Stochastic hill‑climbing** – occasional random pairwise transfers
   of mass are tried; improvements are kept.

All phases respect the time budget (default 1000 s).  The returned list
contains non‑negative heights whose ℓ¹‑norm equals √(2 n); the evaluation
score equals the maximal convolution entry after this normalisation.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate and gradient w.r.t. the softmax parameters y.
    """
    p = _softmax(y)                # Σp = 1, p ≥ 0
    a = S * p                      # Σa = S

    conv = np.convolve(a, a)       # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)                 # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (value, gradient)."""

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


def _max_convolution(a: np.ndarray) -> float:
    """Return max_k (a*a)[k]; a is already on the simplex."""
    return np.convolve(a, a).max()


def _lp_improve(a: np.ndarray, M: float, S: float):
    """
    Solve   max Σ b_j   s.t.  b ≥ 0  and  a∗b ≤ M.
    Return the rescaled vector g (Σg = S) if successful, else None.
    """
    n = a.shape[0]
    rows = 2 * n - 1

    # Build the full (2n‑1)×n Toeplitz matrix representing convolution a∗b.
    # Only non‑negative diagonals are needed (offset d = 0…n‑1).
    diag_vals = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(0, n)                 # row‑col = d
    A = sp.diags(diag_vals, offsets, shape=(rows, n), format='csr')

    c = -np.ones(n)                           # maximise Σb  → minimise −Σb
    b_ub = np.full(rows, M)
    bounds = [(0, None)] * n

    # HiGHS (the default for method='highs') handles sparse matrices efficiently.
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success or res.x is None:
        return None

    b = res.x
    total = b.sum()
    if total == 0:
        return None
    # Rescale to the required ℓ¹‑norm.
    g = b * (S / total)
    return g

                    print(f"[lp improve] new best = {best_score:.6f}")
                a = a_mix
                # reset Adam moments after a jump
                m = np.zeros_like(a)
                v_vec = np.zeros_like(a)
                t_iter = 1

        # ----------------------------------------------------------
        # occasional random hill‑climb (every 500 iterations)
        if t_iter % 500 == 0:
            for _ in range(10):
                i, j = np.random.choice(n, 2, replace=False)
                # transfer a fraction of mass from i to j
                delta = np.random.uniform(0.0, 0.5) * a[i]
                if delta == 0.0:
                    continue
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta
                a_tmp = _project_to_simplex(a_tmp, S)
                sc_tmp = _max_convolution(a_tmp)
                if sc_tmp < best_score:
                    best_score = sc_tmp
                    best_a = a_tmp.copy()
                    print(f"[rand climb] new best = {best_score:.6f}")
                # accept the move if it improves the current a
                if sc_tmp < sc_new:
                    a = a_tmp
                    break

    # --------------------------------------------------------------
    # 3) Final cheap random block search (as a safety net)
    remaining = deadline - time.time()
    if remaining > 1.0:
        trials = int(min(1500, remaining * 80))
        print(f"[block search] performing {trials} trials")
        for _ in range(trials):
            nb = np.random.randint(2, 7)                     # 2‑6 blocks
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb)
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            sc = _max_convolution(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[block improve] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    # Return a plain Python list (evaluate_sequence will accept it directly)
    return best_a.tolist()
