"""
Proposes a step‑function sequence with a low self‑convolution peak.

The algorithm consists of two stages:
1. Broad parametric sweep – a large collection of simple symmetric families
   (inverse‑sqrt, power‑law, exponential, cosine, sine, product‑type) is
   generated, projected onto the simplex Σa = √(2 n) and the best raw
   candidate (according to the exact score) is kept.
2. Long‑run smooth‑max descent – starting from the best raw candidate a
   gradient descent on a soft‑max approximation of the peak is performed.
   The gradient is computed via FFTs, a Nesterov momentum update is used,
   and after every step the iterate is projected back onto the simplex.
   Temperature (soft‑max “sharpness”) is slowly decreased to approach the
   true max.  The loop runs until the time budget expires, keeping the
   best sequence seen.

The returned list satisfies the security checks of `evaluate_sequence`
and aims at a score ≤ 1.503 (the target).
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k · (a * a)_k  with respect to a.
    w must be a weight vector that sums to 1.
    """
    # The convolution a*a is symmetric, gradient formula reduces to:
    # grad = 2 * (w ↔ a_rev) where ↔ denotes linear convolution.
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = len(w)+len(a)-1 = 3n-2
    offset = len(w) - len(a)                     # = n-1
    grad = 2.0 * conv[offset: offset + len(a)]
    return grad


# ----------------------------------------------------------------------

Proposes a step‑function sequence with a low self‑convolution peak.

Stage 1 – Broad parametric sweep:
    Generates many symmetric families (inverse‑sqrt, power‑law, exponential,
    cosine, sine, product‑type, random blocks), projects each onto the simplex
    Σa = √(2 n) and keeps the best raw candidate (exact score).

Stage 2 – Long‑run smooth‑max descent:
    Starting from the best raw candidate a, performs gradient descent on a
    soft‑max approximation of max (a∗a).  The gradient is computed via FFTs,
    a Nesterov momentum update is applied, and after each step the iterate is
    projected back onto the simplex.  Temperature (soft‑max “sharpness”)
    slowly decreases so that the objective converges to the true max.
    The loop runs until the time budget expires, remembering the best
    sequence seen.

The returned list satisfies the security checks of `evaluate_sequence`
and is aimed at a score ≤ 1.503 (the target).
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using real FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. a  (weights w sum to 1).
    The exact expression is   2·(w ↔ a_rev)  restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    offset = len(w) - len(a)                     # = n‑1
    grad = 2.0 * conv[offset: offset + len(a)]
    return grad
