"""
Propose a step‐function with a low self‑convolution peak.

Algorithm
---------
1. Choose a length N (between 8 000 and 20 000).  The ℓ¹‑norm is fixed to
   S = √(2·N) so that the evaluation score equals max(conv(a)).
2. Initialise a with a Dirichlet draw (uniform random on the simplex) and
   optionally mix in the provided `height_sequence_1`.
3. Run a two‑phase first‑order optimisation:
   * **Soft‑max gradient phase** – 70 % of the iterations use a
     temperature‑controlled soft‑max weight (β slowly grows from 0.5 to
     120) and an Adam‑style step.  This spreads the mass away from the
     currently most active convolution lags.
   * **Exact sub‑gradient phase** – the last 30 % use the true
     sub‑gradient of `max(conv)` (the indicator of the maximal lags).
   After each step the vector is projected onto the non‑negative simplex
   simply by clipping negatives and renormalising to the target sum S.
4. Polishing with simulated‑annealing style random pair moves: a small
   amount of mass is transferred from one index to another; the move is
   always accepted if it improves the score and otherwise accepted with
   probability `exp(-(Δ)/T)`.  The temperature T is linearly cooled.
5. The whole routine is repeated for a few different lengths (if time
   permits).  The best sequence found during the run is returned.

The code stays within the 1000 s time budget, uses only NumPy and the
provided `evaluate_sequence`, and prints progress messages.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: simple scaling projection onto the non‑negative simplex.
# ----------------------------------------------------------------------
def _scale_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip negatives, then rescale to have the prescribed ℓ¹‑sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        # degenerate case – fill uniformly
        v = np.full_like(v, target_sum / v.size)
    return v


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based correlation:   corr_i = Σ_k w_{i+k}·a_k   (valid mode, length n)
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                     # == 2·n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length n (valid correlation)

    # ------------------------------------------------------------------
