* Work on the probability simplex (sum of heights = 1).  The score then equals
  `2 * n * max(a * a)`, i.e. two times the length times the maximum entry of
  the self‑convolution.
* Initialise a candidate with several simple shapes (arcsine, symmetric
  Beta‑distribution, exponential decay, uniform or random).  When a previous
  best sequence is available it is up‑sampled to the new length.
* Optimise the candidate by **projected Adam** using an exact sub‑gradient of the
  `max` of the convolution:
      – Compute the full self‑convolution with a real FFT.
      – Collect all (or almost all) indices whose value is within a tiny
        tolerance of the maximum; this is the active set.
      – The sub‑gradient of the max is `2 * (mask * a_rev)` where `mask` is the
        active‑set indicator and `a_rev` is a reversed copy of `a`.  The
        convolution is also evaluated with FFT.
      – Adam’s adaptive step‑size and momentum accelerate convergence.
      – After each update the vector is projected back onto the simplex.
      – A small random jitter is added occasionally to escape plateaus.
* The optimisation is run for a short time slice on many lengths (geometric
  progression, up to 20000).  The best sequence seen over the whole budget is
  returned.
* All code is written with top‑level helper functions (no closures) and uses
  only `numpy` and `scipy`.  It respects the 1000 s time limit.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (U‑shaped) distribution, already normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) distribution on a discrete grid, normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use np.nan_to_num to silence potential NaNs when α<1
    vals = np.nan_to_num(x ** (alpha - 1) * (1 - x) ** (alpha - 1), nan=0.0, posinf=0.0)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam * i), normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_full_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full discrete convolution of a and b using a real FFT."""
    n = len(a) + len(b) - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    C = A * B
    conv = np.fft.irfft(C, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    return _fft_full_convolve(a, a)


# ----------------------------------------------------------------------
def _gradient_from_active_set(mask: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(a*a) for an active set given by `mask`
    (mask[i]=1 if b[i] is (almost) maximal, else 0).

    The gradient is 2 * conv(mask, a_rev) restricted to the original indices.
    """
    n = a.shape[0]
    a_rev = a[::-1]                     # reversed copy
    size = 1 << ((len(mask) + n - 1).bit_length())
    M = np.fft.rfft(mask, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(M * A_rev, n=size)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _optimize_one_length(init_seq: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Adam‑based projected optimisation of a single length.
    Returns the best sequence found and its internal score (2·n·max_b).
    """
    a = init_seq.copy()
    n = a.shape[0]

    best_seq = a.copy()
    b = _self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # tolerance for the active set – starts relatively loose and tightens
    eps_factor = 1e-3

    while time.time() < deadline:
        t += 1

        # current convolution and score
        b = _self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={t} score={best_score:.6f}")

        # build the active mask – all entries within a tiny fraction of the max
        eps_now = max(eps_factor * cur_max, 1e-12)
        mask = (b >= cur_max - eps_now).astype(np.float64)

        # sub‑gradient of the max‑convolution
        grad = _gradient_from_active_set(mask, a)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – a modest constant works well in practice
        step = 0.5 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # project back onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        # slowly tighten the active‑set tolerance
        eps_factor = max(1e-12, eps_factor * 0.995)

        # occasional tiny random jitter to leave plateaus
        if t % 2000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑vector to a larger size."""
    old_n = seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    return _project_onto_simplex(new_seq, s=1.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function with a low `evaluate_sequence` score.
    The routine respects the supplied time budget and returns the best
    sequence as a plain Python list.
    """
    np.random.seed(seed)
    # small safety margin for the final printing step
    deadline = time.time() + budget_s - 3.0

    # ------------------------------------------------------------------
    # Candidate lengths – geometric series, capped at 20000 for speed
    base_len = 1024
    lengths = []
    for factor in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0):
        n = int(base_len * factor)
        if n >= 500 and n <= 20000:
            lengths.append(n)
    lengths = sorted(set(lengths))

    overall_best_seq = None
    overall_best_score = float("inf")
    prev_best_seq = None

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # time slice for this length (≈20 % of what is left, but at most 30 s)
        slice_deadline = time.time() + min(30.0, max(0.5, remaining * 0.2))

        # initialise a few candidates for this length
        init_candidates = []

        # 1) arcsine distribution
        init_candidates.append(_arcsine_distribution(n))

        # 2) up‑sample the best sequence from the previous length (if any)
        if prev_best_seq is not None:
            init_candidates.append(_upsample_sequence(prev_best_seq, n))

        # 3) a handful of random / parametric starts
        for _ in range(2):
            init_candidates.append(np.random.rand(n).astype(np.float64))
        # Beta‑family with a few α‑values
        for alpha in (0.3, 0.6, 1.2):
            init_candidates.append(_beta_distribution(n, alpha))
        # Exponential decay with a random rate
        lam = np.random.uniform(0.005, 0.03)
        init_candidates.append(_exp_decay_distribution(n, lam))

        # normalise every candidate onto the simplex (just in case)
        init_candidates = [_project_onto_simplex(v, s=1.0) for v in init_candidates]

        # ------------------------------------------------------------------
        for init_idx, a0 in enumerate(init_candidates):
            now = time.time()
            if now >= slice_deadline:
                break

            # give each start a modest slice of the remaining time for this length
            sub_deadline = time.time() + min(8.0, max(0.2, slice_deadline - now))

            seq, internal_score = _optimize_one_length(a0, sub_deadline)

            # evaluate with the official black‑box (more reliable)
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = internal_score          # fallback

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq.copy()
                print(
                    f"*** New global best → len={n} init={init_idx} "
                    f"score={overall_best_score:.6f}"
                )

            # early exit if target already reached
            if overall_best_score <= 1.503:
                print("Target reached – exiting early.")
                return overall_best_seq.tolist()

        # keep the best sequence of this length as a seed for the next length
        prev_best_seq = overall_best_seq if overall_best_seq is not None else None

        # stop early if time is gone
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # safety net – should never be triggered
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
