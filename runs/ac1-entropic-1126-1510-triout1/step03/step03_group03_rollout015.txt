"""
Propose a low‑score non‑negative step function.

The algorithm mixes several inexpensive heuristics that together
explore the space far more thoroughly than the previous launch:

1.  Initialise from the reference sequence (if present) or from a
    uniform distribution.
2.  Run several independent Adam‑style sub‑gradient descents on the
    exact hard‑max objective  max_k (a*a)[k]  while keeping the ℓ¹‑norm
    fixed (projection onto the simplex S = √(2·n)).  The Adam optimiser
    automatically adapts per‑coordinate step sizes.
3.  Perform an extensive random block search – piece‑wise constant
    candidates with a variable number of blocks (2‑30).  After each
    random draw the candidate is projected onto the simplex.
4.  Sample a few simple parametric families (exponential decay,

5.  Finish with a classic projected sub‑gradient descent (hard max)
    to polish the best solution found.

All stages respect the time budget (default 1000 s).  As soon as the
target score 1.503 is reached the search stops and the current best

"""

import math

import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """The evaluation function for a (sum is already √(2n))."""
    return np.convolve(a, a).max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # spare margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    S = math.sqrt(2.0 * n)           # required ℓ¹‑norm

    # --------------------------------------------------------------
    # initialise best solution
    if have_ref:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = _project_to_simplex(a0, S)
    else:
        a0 = np.full(n, S / n, dtype=float)   # uniform start

    best_a = a0.copy()
    best_score = _objective(best_a)
    print(f"[init] score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Adam‑style sub‑gradient descent (multiple restarts)
    def adam_descent(init_a: np.ndarray, max_iter: int = 8000) -> None:
        nonlocal best_a, best_score
        a = init_a.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr = 0.6                     # initial learning rate
        it = 0
        while time.time() < deadline and it < max_iter:
            it += 1
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = mask.sum()
            w = mask.astype(float) / cnt          # uniform over active shifts
            # sub‑gradient of the hard max
            grad = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a = a - step
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, S)

            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[adam] new best = {best_score:.6f}  iter={it}")
                if best_score <= 1.503:
                    return

            # modest learning‑rate decay
            if it % 400 == 0:
                lr *= 0.85

    # a few restarts with different seeds
    for r in range(5):
        if time.time() >= deadline or best_score <= 1.503:
            break
        if r == 0:
            start_vec = best_a
        else:
            rand_vec = np.random.rand(n)
            start_vec = _project_to_simplex(rand_vec, S)
        adam_descent(start_vec, max_iter=12000)

    # --------------------------------------------------------------
    # Intensive random block search
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 1.0:
            break
        nb = np.random.randint(2, 31)                # 2‑30 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb)
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, S)
        sc = _objective(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] new best = {best_score:.6f}")
            if best_score <= 1.503:
                break

    # --------------------------------------------------------------
    # Simple parametric families (exponential, power‑law, etc.)
    shape_trials = 0
    while time.time() < deadline and shape_trials < 5000:
        shape_type = np.random.choice(['exp', 'pow', 'linear', 'randdec'])
        if shape_type == 'exp':
            r = np.random.uniform(0.995, 0.99995)
            a_shape = r ** np.arange(n, dtype=float)
        elif shape_type == 'pow':
            p = np.random.uniform(0.2, 2.5)
            a_shape = (np.arange(1, n + 1, dtype=float)) ** (-p)
        elif shape_type == 'linear':
            a_shape = np.linspace(1.0, 0.0, n)
        else:   # random decreasing
            a_shape = np.sort(np.random.rand(n))[::-1]

        a_shape = _project_to_simplex(a_shape, S)
        sc = _objective(a_shape)
        shape_trials += 1
        if sc < best_score:
            best_score = sc
            best_a = a_shape.copy()
            print(f"[shape {shape_type}] new best = {best_score:.6f}")
            if best_score <= 1.503:
                break

    # --------------------------------------------------------------
    # Final projected sub‑gradient polishing (hard max)
    a = best_a.copy()
    eta0 = S * 0.5 / n           # roughly the average entry
    decay = 1e-4
    it = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = mask.sum()
        w = mask.astype(float) / cnt
        grad = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, S)

        cur = _objective(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
            print(f"[final] new best = {best_score:.6f}")
            if best_score <= 1.503:
                break

        it += 1
        if it > 8000:
            break

    # --------------------------------------------------------------
    print(f"[finished] best score = {best_score:.6f}")
    return best_a.tolist()
