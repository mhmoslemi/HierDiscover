        conv = np.fft.irfft(f * f, n2)
        return conv[:2*n-1].max()
    
    # initialize best
    if have_ref:

    # Phase 1: gradient descent (softmax)
    # hyperparameters:
    eta = target_sum * 0.2 / n
    beta = 0.9

    mu_decay = 0.9995

        conv = np.convolve(a, a)   # could use FFT for speed

        # compute softmax weights

        diff = conv - cur_max

        # gradient wrt a

        grad = 2.0 * conv_wr[n-1:n-1+n]

"""
Propose a low‑score step function.

The algorithm consists of three stages:

1. **Fast gradient descent on a smooth approximation** – the max‑convolution
   is smoothed by a soft‑max (temperature µ) and the parameters are updated
   in the exponential (soft‑max) parametrisation.  This quickly drives the
   solution away from the uniform distribution.

2. **Massive random search** – after the gradient phase we spend the bulk of
   the remaining budget generating candidate sequences.  Candidates are drawn
   either from a Dirichlet distribution (Gamma‑generated weights) with a
   random concentration parameter or as a piece‑wise constant sequence with a
   random number of blocks.  Each candidate is scaled to the required ℓ¹‑norm
   and its max‑convolution is evaluated via an FFT‑based O(n log n) routine.
   The best candidate found so far is kept.

3. **Local pairwise swaps** – a final short hill‑climbing phase swaps a
   fraction of mass between two positions; the move is accepted only if it
   strictly improves the objective.

All operations respect the non‑negative constraint and the required
ℓ¹‑norm (the usual normalisation Σa = √(2 n) makes the evaluation function
return exactly the maximal self‑convolution).  The code respects the 1000 s
wall‑clock limit and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _scale_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have the given ℓ¹‑norm."""
    s = v.sum()
    if s == 0:
        return np.full_like(v, target / v.size)
    return v * (target / s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Approximate wall‑clock budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered (plain Python list).
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]

        n = 2500
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # ensures objective = max‑conv

    # ------------------------------------------------------------------
    # FFT helpers --------------------------------------------------------
    # length for convolution (next power of two ≥ 2n‑1)
    n2 = 1 << ((2 * n - 1).bit_length())

    def _conv_self(vec: np.ndarray) -> np.ndarray:
        """Full self‑convolution using a real FFT, returns length 2n‑1."""
        f = np.fft.rfft(vec, n2)
        conv = np.fft.irfft(f * f, n2)
        return conv[: 2 * n - 1]

    def _conv_w_a(wvec: np.ndarray, arev: np.ndarray) -> np.ndarray:
        """Convolution w * a_rev (both length 2n‑1 and n respectively)."""
        fw = np.fft.rfft(wvec, n2)
        fa = np.fft.rfft(arev, n2)
        conv = np.fft.irfft(fw * fa, n2)
        return conv[: 2 * n - 1]

    def _max_conv(vec: np.ndarray) -> float:
        """Maximum entry of the self‑convolution."""
        return float(_conv_self(vec).max())

    # ------------------------------------------------------------------

    if have_ref:

    best_a = a.copy()

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max projected gradient descent --------------------
    # hyper‑parameters (chosen to be aggressive yet stable)
    eta0 = target_sum * 0.4 / n                # base step size

    mu_decay = 0.9997                          # very slow cooling

            cand = _scale_to_sum(cand, target_sum)

        sc = _max_conv(cand)
        if sc < best_score:
            best_score = sc

            print(f"[rand] improved score = {best_score:.6f}")

    while time.time() < time.time() + budget_s - 0.1:
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size < 2:
            break

            best_a = cand
            print(f"[swap] new best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
