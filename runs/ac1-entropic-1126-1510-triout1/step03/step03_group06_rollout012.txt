    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # j varies from max(0, k-(n-1)) to min(k, n-1)
        low_j = max(0, k-(n-1))
        high_j = min(k, n-1)
        # b_j corresponds to column j (0-indexed)
        # coefficient a_{k-j}

    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

Algorithm overview
------------------
1.  Choose moderate problem sizes (n≈1500‑2500) – small enough for a dense
    linear program, large enough to approximate the continuous problem.
2.  Generate a few diverse initial probability vectors:
        * discrete arcsine law,
        * symmetric Beta(α,α) for several small α (U‑shaped densities),
        * uniform,
        * a sparse “spike” vector,

        •  Let M = max (conv(a,a)).
        •  Solve the LP   maximise  ∑ b   subject to   a∗b ≤ M , b ≥ 0 .
        •  Normalise the LP solution b to a probability vector g.
        •  Mix a←(1‑t)·a+t·g with several candidate step‑sizes t
          and keep the mixture that yields the smallest objective.
        •  Repeat until no improvement or the time budget runs out.
   This is the “LP‑step” described in the statement and is known to
   strictly decrease the peak of the self‑convolution.
4.  Refine the best candidate with exponentiated‑gradient mirror descent
    on a soft‑max approximation of the peak (large temperature λ≈80).
5.  Finish with a cheap stochastic hill‑climbing phase that moves tiny

6.  Return the best sequence found (as a plain Python list).

All operations respect the simplex constraint (non‑negative, sum = 1),

import time

# ----------------------------------------------------------------------
# Basic convolution / scoring utilities
# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised (sum = 1) vector."""
    n = a.shape[0]
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
# Distributions for initial candidates
# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (soft‑max of the peak)
# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector (length 2·n‑1),
    a – current sequence (length n),
    m – FFT length (power of two, ≥ len(w)+len(a)-1).
    Returns gradient of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = a.shape[0] - 1
    return 2.0 * corr[start:start + a.shape[0]]


def _optimise_exponential(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 8000,
    eta0: float = 0.12,
    lambda0: float = 0.5,
    lambda_max: float = 80.0,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on a soft‑max approximation of max(conv)."""
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT size for the correlation (needs to accommodate vectors of length 2n‑1 and n)
    m = 1 << ((3 * n - 2).bit_length())

    # linear schedule for the temperature λ
    lamb = lambda0
    lambda_inc = (lambda_max - lambda0) / max_iters

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and (soft) max
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] exp‑gd n={n:5d} it={it:5d} score={best_score:.6f}")

        # soft‑max weights over the convolution entries
        maxb = b.max()
        exp_arg = lamb * (b - maxb)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # project back onto the simplex (renormalise)
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)

        # occasional tiny jitter to escape plateaus
        if it % 800 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        lamb = min(lambda_max, lamb + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (tiny mass transfers)
# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 20000,
) -> tuple[np.ndarray, float]:
    """Move a tiny amount of mass from one index to another if it improves the score."""
    n = a.shape[0]
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming step (maximise total mass under a convolution bound)
# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, M: float, t_values: list[float]) -> tuple[np.ndarray | None, float]:
    """
    Solve  maximise Σ b  subject to  a∗b ≤ M , b ≥ 0.
    Return the best mixed vector (a←(1‑t)a+t g) among the supplied t values.
    """
    n = a.shape[0]

    # Build the dense convolution matrix A (size (2n‑1)×n)
    A = np.empty((2 * n - 1, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a

    # Linear‑program formulation (maximise → minimise –Σb)
    c = -np.ones(n)                     # minimise –sum(b)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True}
    )
    if not res.success:
        return None, np.inf

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1e-12:
        return None, np.inf

    # Normalise the LP solution to a probability vector
    g = g / sum_g

    best_mix = None
    best_score = np.inf

    # Try several mixing factors and keep the best improvement
    for t in t_values:
        mix = (1.0 - t) * a + t * g
        # Numerical cleanup
        mix = np.clip(mix, 0.0, None)
        mix = mix / mix.sum()
        cur_score = _score(mix)
        if cur_score < best_score:
            best_score = cur_score
            best_mix = mix.copy()

    return best_mix, best_score


def _lp_optimize(a: np.ndarray, deadline: float, max_outer: int = 8) -> tuple[np.ndarray, float]:
    """Repeated LP‑step + mixing until no further improvement."""
    best_a = a.copy()
    best_score = _score(best_a)
    t_candidates = [0.05, 0.1, 0.2, 0.3, 0.4]

    while time.time() < deadline and max_outer > 0:
        M = _conv_full_fft(best_a).max()
        new_a, new_score = _lp_step(best_a, M, t_candidates)
        if new_a is None or new_score >= best_score - 1e-12:
            break
        best_a = new_a
        best_score = new_score
        print(f"[{time.time():.1f}] LP improvement: score={best_score:.6f}")
        max_outer -= 1

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a very low
    `evaluate_sequence` score.  Returns the best sequence found as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  historic sequence (if available)
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  problem sizes to explore (moderate n for LP feasibility)
    candidate_lengths = [1500, 2000, 2500]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Exploring length n={n} ===")
        # --------------------------------------------------------------
        #   a) build a small pool of starter vectors
        # --------------------------------------------------------------
        init_pool = [
            _arcsine_distribution(n),
            np.full(n, 1.0 / n, dtype=np.float64),               # uniform
        ]

        # a few symmetric Betas with small α (U‑shaped)
        for alpha in (0.10, 0.15, 0.20, 0.25, 0.35):
            init_pool.append(_beta_distribution(n, alpha))

        # random Dirichlet (low concentration ⇒ many tiny entries)
        init_pool.append(np.random.dirichlet(np.full(n, 0.2)))

        # a sparse “spike” vector
        k = max(2, n // 500)
        idx = np.random.choice(n, size=k, replace=False)
        a_spike = np.zeros(n, dtype=np.float64)
        a_spike[idx] = np.random.rand(k) + 0.1
        init_pool.append(a_spike / a_spike.sum())

        # --------------------------------------------------------------
        #   b) optimise each candidate
        # --------------------------------------------------------------
        for init in init_pool:
            if time.time() >= deadline:
                break

            a0 = init.astype(np.float64)
            a0 = a0 / a0.sum()                     # safety normalisation

            # ---- LP‑based improvement ----
            a_lp, score_lp = _lp_optimize(a0, deadline, max_outer=8)

            # ---- Mirror‑descent fine tuning ----
            a_opt, score_opt = _optimise_exponential(
                n=n,
                init_seq=a_lp,
                deadline=deadline,
                max_iters=8000,
                eta0=0.12,
                lambda0=0.5,
                lambda_max=80.0,
            )

            if score_opt < overall_best_score:
                overall_best_score = score_opt
                overall_best_seq = a_opt.copy()
                print(f"*** New global best: length={n} score={overall_best_score:.6f}")

            # early exit if we already reach the target
            if overall_best_score <= 1.503:
                print("Target reached – exiting early.")
                break

        # abort outer loop early if target met
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # 3️⃣  final stochastic hill‑climbing refinement
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline:
        seq, sc = _hill_climb(overall_best_seq, deadline, max_trials=20000)
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = seq.copy()
            print(f"*** After hill‑climb: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 1500
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"\nFinal best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    return overall_best_seq.tolist()
