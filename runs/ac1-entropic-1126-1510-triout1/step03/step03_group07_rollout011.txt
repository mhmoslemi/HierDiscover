def _adam_optimize(a0, S, deadline, max_iters=5000):
    a = a0.copy()
    best_seq = a.copy()
    conv = _fft_convolve(a)
    best_score = conv.max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_start = 0.1
    lr_end = 0.01
    beta_start = 0.5
    beta_end = 80.0
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute gradient
        conv = _fft_convolve(a)
        max_conv = conv.max()
        t_frac = it / max_iters
        beta = beta_start + (beta_end - beta_start) * t_frac
        # compute softmax weights
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad = 2.0 * _fft_correlate(w, a)
        # update Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = lr_start + (lr_end - lr_start) * t_frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # projection
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, S)
        # evaluate occasionally
        if (it % 10) == 0:
            conv = _fft_convolve(a)
            score = conv.max()
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                # print message
    return best_seq, best_score

def _simulated_annealing(seq, S, deadline, max_iters=20000):
    cur = seq.copy()
    cur_score = _fft_convolve(cur).max()
    best_seq = cur.copy()
    best_score = cur_score
    # schedule: temperature decreasing from initial T0 (maybe based on current_score range)
    # Let's set T0 = (cur_score - 1.0) maybe.
    # We'll adapt sigma for moves linearly decreasing.
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute sigma
        sigma = 0.5 * (1 - it / max_iters) + 0.01
        cand = _random_pair_move(cur, sigma)
        # enforce sum S (random pair keeps sum exactly) but ensure projection if small negative after due to rounding? Already ensured nonneg.
        # compute score
        conv = _fft_convolve(cand)
        cand_score = conv.max()
        # acceptance
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                # print
        else:
            # compute temperature
            T0 = (cur_score - best_score) if cur_score != best_score else 0.001
            # maybe set temperature schedule: T = T0 * (1 - it / max_iters)
            temp = T0 * (1 - it / max_iters) + 0.0001
            # accept with probability exp(-(cand_score - cur_score)/temp)
            if np.random.rand() < math.exp(-(cand_score - cur_score) / temp):
                cur = cand
                cur_score = cand_score
        # optionally print best occasionally
    return best_seq, best_score

    idx = 0
    for s, c in zip(sums, counts):
        avg = s / c
        result[idx: idx+c] = avg

"""
Multi‑restart Adam optimiser with monotone projection + simulated‑annealing polish.

The objective (after fixing the ℓ₁‑norm to S = √(2·N)) is to minimise
    max_k (a * a)[k] .
The code works with any length N (chosen automatically from the
previous best sequence if available) and respects the non‑negative
simplex constraint Σ a_i = S.

Key ideas
---------
* Adam (adaptive moment estimation) on the smooth “soft‑max” surrogate
  of the max‑convolution, with a projection onto the ℓ₁‑simplex.
* After every gradient step we enforce a *decreasing* shape via an
  isotonic regression (pool‑adjacent‑violators).  By the rearrangement
  inequality a decreasing sequence minimises the inner product with its
  reversal, which is the dominant term of the convolution.
* Several diverse initial points (uniform, noisy uniform, Dirichlet,
  tiled previous construction) are optimised; the best result is kept.
* A final simulated‑annealing stage performs random pairwise mass‑swaps
  (which keep the ℓ₁‑norm exactly) to escape possible local minima.
  The temperature and the swap amplitude are linearly cooled.

The routine respects the global time‑budget (default 1000 s) and returns
the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (non‑negative, prescribed sum) – Duchi et al. (2008)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (length 2n‑1)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, nfft)
    conv = np.fft.irfft(A * A, nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based "valid" correlation: corr = np.correlate(w, a, mode='valid')
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]                # length of a
    L = w.shape[0]                # length of w  (= 2n‑1)
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, nfft)
    A_rev = np.fft.rfft(a_rev, nfft)
    full = np.fft.irfft(W * A_rev, nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length n (valid region)


# ----------------------------------------------------------------------
# Random pairwise mass‑swap that exactly preserves the ℓ₁‑norm.
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Transfer a small amount of mass between two distinct indices."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # delta uniformly drawn from (-σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    # keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# Isotonic regression (non‑increasing) – L₂ optimal via PAVA.
# ----------------------------------------------------------------------
def _isotonic_decreasing(v: np.ndarray) -> np.ndarray:
    """Return the monotone non‑increasing L₂‑projection of v."""
    # Work on -v to obtain a non‑decreasing regression, then flip sign back.
    x = -v.astype(float).copy()
    n = len(x)
    # PAVA data structures
    sums = list(x)          # sum of each block
    counts = [1] * n        # size of each block
    i = 0
    while i < len(sums) - 1:
        if sums[i] / counts[i] > sums[i + 1] / counts[i + 1]:
            # merge blocks i and i+1
            total_sum = sums[i] + sums[i + 1]
            total_cnt = counts[i] + counts[i + 1]
            sums[i] = total_sum
            counts[i] = total_cnt
            del sums[i + 1]
            del counts[i + 1]
            if i > 0:
                i -= 1
        else:
            i += 1
    # Expand the block averages back to an array
    result = np.empty(n, dtype=float)
    pos = 0
    for s, c in zip(sums, counts):
        avg = s / c
        result[pos: pos + c] = avg
        pos += c
    return -result


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max surrogate) with monotone projection.
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 25000,
) -> tuple[np.ndarray, float]:
    """Return (best_sequence, best_score) after a projected‑Adam run."""
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Schedules (linear annealing)
    lr_start, lr_end = 0.3, 0.02
    beta_start, beta_end = 0.5, 80.0

    best_seq = a.copy()
    best_score = _fft_convolve(a).max()

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # ----- gradient of soft‑max max(conv) -----
        conv = _fft_convolve(a)
        max_conv = conv.max()
        progress = it / max_iters
        beta = beta_start + (beta_end - beta_start) * progress

        # soft‑max weights (stable because conv ≤ max_conv)
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = 2.0 * _fft_correlate(w, a)        # shape (n,)

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        lr = lr_start + (lr_end - lr_start) * progress
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # ----- enforce constraints -----
        a = np.maximum(a, 0.0)                     # non‑negative
        a = _project_onto_simplex(a, target_sum)   # ℓ₁‑norm
        a = _isotonic_decreasing(a)                # monotone decreasing
        a = _project_onto_simplex(a, target_sum)   # re‑normalise after isotonic

        # ----- occasional evaluation -----
        if it % 20 == 0:
            cur_score = _fft_convolve(a).max()
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[Adam iter {it:05d}] new best = {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (random pair moves, temperature schedule)
# ----------------------------------------------------------------------
def _simulated_annealing(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
    max_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    """Return (best_sequence, best_score) after SA polishing."""
    cur = seq.copy()
    cur_score = _fft_convolve(cur).max()
    best_seq = cur.copy()
    best_score = cur_score

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # cooling schedules
        sigma = 0.4 * (1 - it / max_iters) + 0.02          # move amplitude
        temperature = max(0.0005, (cur_score - best_score) * (1 - it / max_iters) + 0.0005)

        cand = _random_pair_move(cur, sigma)
        cand_score = _fft_convolve(cand).max()

        if cand_score < cur_score:
            cur, cur_score = cand, cand_score
            if cand_score < best_score:
                best_seq, best_score = cand.copy(), cand_score
                print(f"[SA  {it:05d}] new best = {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_score - cur_score) / temperature)
            if np.random.rand() < prob:
                cur, cur_score = cand, cand_score

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Multi‑restart Adam optimiser + SA polishing.
    Returns a list of non‑negative heights (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Choose problem size N and target ℓ₁‑norm S = √(2·N)
    # --------------------------------------------------------------
    base_seq = globals().get("height_sequence_1")
    if isinstance(base_seq, (list, tuple)):
        n0 = len(base_seq)
    else:
        n0 = 2000
    N = min(20000, max(8000, n0 * 2))
    S = math.sqrt(2.0 * N)

    # --------------------------------------------------------------
    # Assemble a handful of diverse starting points.
    # --------------------------------------------------------------
    starters = []

    # 1) Uniform simplex point
    a_uniform = np.full(N, S / N, dtype=float)
    starters.append(a_uniform)

    # 2) Uniform + tiny random perturbation
    a_noise = a_uniform + 0.001 * S * np.random.rand(N) / N
    a_noise = _project_onto_simplex(a_noise, S)
    starters.append(a_noise)

    # 3) Random Dirichlet (Gamma(1,1))
    rnd = np.random.gamma(1.0, 1.0, N)
    a_dir = S * rnd / rnd.sum()
    starters.append(a_dir)

    # 4) Tiled version of the previous best construction (if available)
    if isinstance(base_seq, (list, tuple)):
        base_arr = np.asarray(base_seq, dtype=float)
        rep = int(np.ceil(N / base_arr.size))
        a_tile = np.tile(base_arr, rep)[:N]
        a_tile = S * a_tile / a_tile.sum()
        starters.append(a_tile)

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Run Adam optimiser on each starter (time‑aware)
    # --------------------------------------------------------------
    for idx, a0 in enumerate(starters):
        if time.time() >= deadline:
            break
        seq_opt, score_opt = _adam_optimize(
            a0, S, deadline, max_iters=25000
        )
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print(f"[Adam start {idx}] best = {best_score:.6f}")

    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(N, S / N, dtype=float)

    # --------------------------------------------------------------
    # Final polishing with simulated annealing
    # --------------------------------------------------------------
    seq_final, score_final = _simulated_annealing(
        best_seq, S, deadline, max_iters=50000
    )
    if score_final < best_score:
        best_score = score_final
        best_seq = seq_final.copy()
        print(f"[SA final] improved best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the answer as a plain Python list.
    # --------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)   # just in case
    return best_seq.tolist()
