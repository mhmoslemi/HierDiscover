"""
Propose a low‑score step‑function.

The search is a pure NumPy implementation that works inside the
1000 s budget.  It proceeds in three stages:

1️⃣  Choose a problem size *n* (up to 20 000) and initialise the
    distribution with a few analytically‑motivated candidates
    (arcsine, Beta, uniform, Dirichlet).

2️⃣  Refine each candidate with a long exponentiated‑gradient (soft‑max)
    optimisation of the peak of the self‑convolution, followed by a
    projected sub‑gradient descent on the exact peak.  Both methods run
    until the global deadline is reached.

3️⃣  Finish with a cheap stochastic hill‑climbing phase that moves tiny
    bits of mass between positions and accepts improvements.

All vectors are kept non‑negative and normalised to sum 1 – the
evaluation function `evaluate_sequence` is invariant under this scaling.
The routine prints intermediate scores and returns the best sequence
found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=False)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute gradient of  Σ_k w_k·(a*a)_k   w.r.t. a.
    Equivalent to 2·corr(w, a).
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a normalised sequence (sum=1)."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1.0 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _optimise_exponential(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 20000,
    eta0: float = 0.5,
    lambda0: float = 0.2,
    lambda_max: float = 120.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror (exponentiated) descent on a soft‑max approximation of the peak
    of the self‑convolution.  Returns the best sequence (sum 1) seen and
    its exact unsmoothed score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT length for the correlation (w length 2n‑1, a length n)
    m = 1 << ((3 * n - 2).bit_length())
    lamb = lambda0
    lambda_inc = (lambda_max - lambda0) / max_iters

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exact convolution and score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} exp it={it:5d} score={best_score:.6f}")

        # soft‑max weights (log‑sum‑exp trick)
        maxb = b.max()
        exp_arg = lamb * (b - maxb)           # ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient of the smooth objective
        g = _gradient_via_correlation(w, a, m)

        # multiplicative (exponentiated) update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)

        # occasional tiny jitter to leave plateaus
        if it % 1000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

        # increase λ
        lamb = min(lambda_max, lamb + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 8000,
    eta0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # sub‑gradient g_i = 2·a_{k*−i}  (where defined)
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        # jitter every few hundred steps
        if it % 500 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny mass from one index
    to another and keep the change if the (unsmoothed) objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t:5d} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # tiny safety margin

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣ historic sequence, if any
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hist_score = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hist_score
            print(f"Historic sequence: length={len(hist)} score={hist_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣ problem sizes – we focus on a few large values (up to 20 000)
    candidate_lengths = [2000, 5000, 10000, 20000]
    for n in candidate_lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Exploring length n={n} ===")

        # --------------------------------------------------------------
        # generate a modest pool of diverse start points
        init_pool = []

        # deterministic arcsine
        init_pool.append(_arcsine_distribution(n))

        # a handful of symmetric Betas
        for alpha in (0.3, 0.6, 1.0, 1.5):
            init_pool.append(_beta_distribution(n, alpha))

        # random Beta draws
        for _ in range(2):
            alpha = np.random.uniform(0.2, 2.5)
            init_pool.append(_beta_distribution(n, alpha))

        # uniform
        init_pool.append(np.full(n, 1.0 / n, dtype=np.float64))

        # low‑concentration Dirichlet (many tiny entries)
        init_pool.append(np.random.dirichlet(np.full(n, 0.3)))

        best_n_seq = None
        best_n_score = float("inf")

        # --------------------------------------------------------------
        # a) exponential mirror‑descent (soft‑max) on every start
        for init in init_pool:
            if time.time() >= deadline:
                break
            seq, sc = _optimise_exponential(
                n=n,
                init_seq=init,
                deadline=deadline,
                max_iters=20000,
                eta0=0.5,
                lambda0=0.2,
                lambda_max=120.0,
            )
            # follow up with a sub‑gradient pass
            if time.time() < deadline:
                seq, sc = _subgradient_descent(
                    n=n,
                    init_seq=seq,
                    deadline=deadline,
                    max_iters=8000,
                    eta0=0.3,
                )
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  >>> New best for n={n}: score={best_n_score:.6f}")

        # --------------------------------------------------------------
        # b) final hill‑climbing refinement on the per‑n best
        if best_n_seq is not None and time.time() < deadline:
            seq, sc = _hill_climb(best_n_seq, deadline, max_trials=30000)
            if sc < best_n_score:
                best_n_score = sc
                best_n_seq = seq.copy()
                print(f"  *** After hill‑climb n={n}: score={best_n_score:.6f}")

        # --------------------------------------------------------------
        # c) update global optimum
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"*** Global best updated: length={n} score={overall_best_score:.6f}")

        # early exit if target is already met
        if overall_best_score <= 1.503:
            print("Target threshold reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # 3️⃣ fallback (should never be needed)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"\nFinal best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    # Return a plain Python list for the evaluation harness
    return overall_best_seq.tolist()
