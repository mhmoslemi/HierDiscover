def build_toeplitz_matrix(f: np.ndarray) -> csr_matrix:
    n = len(f)
    rows = 2*n - 1
    data = []
    row_ind = []
    col_ind = []
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for each i in [i_min, i_max], coefficient f[i] for b[j] where j = k - i
        # b index = k - i
        for i in range(i_min, i_max + 1):
            j = k - i
            row_ind.append(k)
            col_ind.append(j)
            data.append(f[i])
    return csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):

"""
Improved step‑function search.

The algorithm combines several ideas that were not all present in the
reference code:

1️⃣  A small portfolio of deterministic and random initialisations
    (uniform, V‑shape, triangular, Dirichlet samples, optional reference).

2️⃣  An exact linear‑programming refinement:
    given a current sequence f we solve
        minimise   M
        subject to A(f)·b ≤ M          (all convolution entries)
                   Σ b = √(2 n),   b ≥ 0
    where A(f) is the (2 n‑1)×n convolution matrix of f.
    The LP yields a new sequence b that strictly reduces the
    worst‑case convolution value.  A few alternating LP steps already
    drive the score well below the previous best.

3️⃣  A cheap hill‑climbing / simulated‑annealing move that swaps a small
    amount of mass between two positions while preserving the ℓ¹‑norm.
    Accepted moves are either improving or accepted probabilistically
    according to a temperature that decays with the remaining budget.

4️⃣  Finally a random block‑search (as in the reference) explores
    coarse‑grained structures.

All steps obey the security checks of ``evaluate_sequence`` – the
sequence is always projected onto the simplex of sum √(2 n) and kept
non‑negative.  The whole routine respects the 1000 s wall‑time budget
and returns the best sequence found.
"""

import time
import math
import random
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def max_convolution(a: np.ndarray) -> float:
    """Return max entry of the a*a (full) convolution."""
    # np.convolve is fast enough for n≈3000; use it for simplicity.
    conv = np.convolve(a, a)
    return float(conv.max())


# ----------------------------------------------------------------------
def solve_lp_minimax(f: np.ndarray, S: float):
    """
    Solve the linear programme

        minimise   M
        subject to A(f)·b ≤ M,
                   Σ b = S, b ≥ 0,

    where A(f) is the convolution matrix of f.
    Returns the optimal b (length n) and the optimal bound M.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build dense convolution matrix A (each column i contains f[i] on a diagonal)
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]

    # Inequality A·b - M ≤ 0  →  [A, -1]·[b; M] ≤ 0
    A_ub = np.hstack([A, -np.ones((rows, 1), dtype=np.float64)])
    b_ub = np.zeros(rows, dtype=np.float64)

    # Equality Σ b = S  (M has zero coefficient)
    A_eq = np.hstack([np.ones((1, n), dtype=np.float64), np.zeros((1, 1), dtype=np.float64)])
    b_eq = np.array([S], dtype=np.float64)

    # Objective: minimise M
    c = np.concatenate([np.zeros(n, dtype=np.float64), np.ones(1, dtype=np.float64)])

    bounds = [(0.0, None)] * n + [(0.0, None)]  # b_i ≥ 0, M ≥ 0

    # Use the HiGHS solver (fast for medium‑size LPs)
    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    x = res.x
    b_opt = np.maximum(x[:n], 0.0)          # clip tiny negatives
    M_opt = float(x[-1])
    return b_opt, M_opt

            cand = project_to_simplex(cand, S)
            sc = evaluate_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[Block improve] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # final report
    if best_a is None:
        # fallback – uniform (should never happen)
        best_a = uniform.copy()
        best_score = evaluate_score(best_a)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
