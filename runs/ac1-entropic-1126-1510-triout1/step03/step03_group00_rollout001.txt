"""
Propose a non‑negative step function that minimizes
`evaluate_sequence`.  The algorithm is a multi‑restart projected Adam
optimizer on the probability simplex (sum of coefficients = 1).  It
starts the optimisation with a smooth soft‑max approximation of the
max‑convolution and, after a few thousand iterations, switches to a
true sub‑gradient (the active max entry only).  Several sequence lengths
are tried and the best sequence found before the time budget expires is
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Compute soft‑max weights w_k = exp(alpha * (b_k - max(b))) / sum."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        # fallback – pure max
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """Gradient of the soft‑maxed max‑convolution w.r.t a."""
    n = a.shape[0]
    a_rev = a[::-1]
    # FFT of w and reversed a
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _gradient_submax(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max‑entry (index k) of a*a w.r.t a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    # valid range for j where a_{k-j} exists
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)               # inclusive
    js = np.arange(i_start, i_end + 1)
    g[js] = 2.0 * a[k - js]
    return g


# ----------------------------------------------------------------------
def _optimize_one_length(a0: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for a fixed length `n`.
    Returns the best sequence discovered and its exact evaluation score.
    """
    # initialise
    a = _project_onto_simplex(a0.copy(), s=1.0)

    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.15                # learning rate (on probability simplex)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters for the soft‑max phase
    softmax_iters = 2000
    alpha = 1.0
    alpha_mul = 1.04
    alpha_max = 20000.0

    # size for FFT used in soft‑max gradient
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # compute convolution and current score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # -------- gradient computation ----------
        if it <= softmax_iters:
            # soft‑max of the convolution values
            w = _softmax_weights(b, alpha)
            grad = _gradient_softmax(a, w, size_fft)
            # anneal temperature
            if alpha < alpha_max:
                alpha = min(alpha_max, alpha * alpha_mul)
        else:
            # pure sub‑gradient (active max entry only)
            k = int(np.argmax(b))
            grad = _gradient_submax(a, k)

        # -------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a = a - step                     # move opposite to gradient
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to escape possible plateaus
        if it % 4000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score
