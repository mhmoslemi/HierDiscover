"""
Propose a low‑score step‑function sequence.

Algorithm Overview
------------------
1. **Candidate lengths** – explore several lengths (2000‑8000).  
2. **Initial pool** – for each length generate a diverse set of start‑vectors:
   arcsine, uniform, many symmetric Beta(α,α) laws, Dirichlet draws and sparse
   spike vectors.
3. **AdaGrad exponentiated‑gradient** – minimise a soft‑max approximation of
   `max(conv(a,a))`.  The temperature λ is increased exponentially from a
   small value to a large one (≈150).  Per‑coordinate AdaGrad step‑sizes
   accelerate convergence.
4. **Projected sub‑gradient refinement** – after the smooth phase run a
   classic sub‑gradient descent on the exact `max` using a simplex projection.
5. **Stochastic hill‑climbing** – cheap local moves that transfer a tiny
   amount of mass between two positions; keep the move if it improves the
   true objective.
6. **Time‑aware loop** – all loops check the global deadline (default 1000 s)
   and stop early if necessary.
7. Return the best sequence (as a plain Python list) found within the budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid (i+½)/n, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using an FFT of power‑of‑two length."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector (length 2·n‑1), a – current sequence (length n),
    m – FFT length (power of two, ≥ len(w)+len(a)-1).
    Returns gradient for indices 0…n‑1.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum 1."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Return a modest, diverse pool of probability vectors of length n."""
    cand = []

    # arcsine and uniform
    cand.append(_arcsine_distribution(n))
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a dense grid of symmetric Beta laws
    for alpha in (0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.9, 1.2, 1.5, 2.0):
        cand.append(_beta_distribution(n, alpha))

    # a few random Betas
    for _ in range(3):
        alpha = np.random.uniform(0.1, 2.5)
        cand.append(_beta_distribution(n, alpha))

    # sparse spikes (≈0.2 % non‑zero)
    k = max(2, n // 500)
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    cand.append(a / a.sum())

    # low‑concentration Dirichlet (many tiny entries)
    cand.append(np.random.dirichlet(np.full(n, 0.3)))

    # very low concentration Dirichlet (more spikes)
    cand.append(np.random.dirichlet(np.full(n, 0.05)))

    return cand


# ----------------------------------------------------------------------
def _exp_mirror_adagrad(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 8000,
    eta0: float = 0.6,
    lambda0: float = 0.3,
    lambda_max: float = 150.0,
) -> tuple[np.ndarray, float]:
    """
    AdaGrad‑accelerated exponentiated‑gradient descent on a soft‑max
    approximation of `max(conv(a,a))`.  Returns the best vector (∑=1)
    seen and its true (unsmoothed) score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT length (enough for convolution of w and a)
    m = 1 << ((3 * n - 2).bit_length())

    # AdaGrad accumulator
    grad_sq_acc = np.zeros_like(a)

    # exponential schedule for λ
    log_ratio = np.log(lambda_max / lambda0) if lambda_max > lambda0 else 0.0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full convolution and true score
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # soft‑max weights
        maxb = b.max()
        lam = lambda0 * np.exp(log_ratio * (it - 1) / max_iters)  # exponential rise
        w = np.exp(lam * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient of the smoothed objective
        g = _gradient_via_correlation(w, a, m)

        # AdaGrad update inside the multiplicative mirror map
        grad_sq_acc += g * g
        step = eta0 / (np.sqrt(grad_sq_acc) + 1e-12)   # element‑wise
        a = a * np.exp(-step * g)

        # re‑normalise onto the simplex
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        # occasional tiny jitter to escape plateaus
        if it % 1000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 5e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 4000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact `max(conv(a,a))`.
    Uses the standard sub‑gradient g_i = 2·a_{k*−i} where k* is a max index.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # build sub‑gradient
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        # jitter now and then
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 5e-4
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny mass from one index
    to another and keep the move if the (unsmoothed) objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # transfer at most 4 % of the donor mass (or 0.01 absolute)
        delta = np.random.rand() * min(best_seq[i] * 0.04, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score
