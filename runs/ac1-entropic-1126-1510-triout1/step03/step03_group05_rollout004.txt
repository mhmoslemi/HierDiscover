def simulated_annealing(a_start, score_start, time_limit):
    a = a_start.copy()
    cur_score = score_start
    best_local_a = a.copy()
    best_local_score = cur_score
    # schedule
    total_iters = 300_000  # or compute based on remaining time
    # scaling for delta
    base_step = target_sum * 0.02  # max delta proportion
    for it in range(total_iters):
        now = time.time()
        if now >= time_limit:
            break
        # temperature schedule:
        t_frac = it / total_iters
        T = max(0.01, (1 - t_frac) * 0.5)  # linear cooling
        # propose move:
        i = np.random.randint(n)
        # ensure we pick i with positive mass
        if a[i] <= 0:
            continue
        # choose j different
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # delta magnitude
        # consider scaling with a[i] to keep relative
        max_delta = a[i]
        delta = np.random.rand() * max_delta  # up to entire mass maybe large; we can reduce
        # optionally limit delta
        delta = min(delta, max_step)
        # move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute score
        new_score = _score(a_new, size_fft)
        # decide acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local_a = a_new.copy()
                print(f"SA iteration {it} new best {best_local_score:.6f}")
    return best_local_a, best_local_score

def _simulate_annealing(a0, cur_score0, size_fft, target_sum, deadline):
    a = a0.copy()
    cur_score = cur_score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0
    # set total iterations maximum based on time left.
    # We'll run e.g., while time < deadline and it < 500000:
    max_iters = 500000
    while time.time() < deadline and it < max_iters:
        # compute temperature based on proportion of time used
        frac = (it / max_iters)
        T = max(0.001, (1.0 - frac) * 0.5)  # linear decreasing
        # decide move type
        if np.random.rand() < 0.9:
            # local transfer move
            i = np.random.randint(n)
            if a[i] <= 0:
                # try again after some loops? just continue
                it += 1
                continue
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            # delta proportion up to 10% of a[i] scaled by (1-frac) for annealing
            max_delta = a[i]
            delta = max_delta * np.random.rand() * (0.2 * (1 - frac) + 0.01)  # up to ~20% early, 1% late
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            # global noise
            sigma = target_sum * (0.01 * (1 - frac) + 0.001)
            noise = np.random.randn(n) * sigma
            a_new = a + noise
            a_new = _project_to_simplex(a_new, target_sum)

        # evaluate
        new_score = _score(a_new, size_fft)

        # accept
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[SA] iter {it:07d} new best {best_score:.6f}")
        else:
            # metropolis
            if np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
                a = a_new
                cur_score = new_score
        it += 1
    return best_a, best_score

def _adam_optimize(a0, cur_score0, size_fft, target_sum, deadline):
    a = a0.copy()
    cur_score = cur_score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    # Adam hyperparams
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    # initial learning rate
    lr0 = 0.15
    # bounds for momentum adaptation? not needed
    while time.time() < deadline and it < 800000:
        # compute full conv
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter {it:07d} new best {best_score:.6f}")

        # compute indices of current max (tolerance)
        max_val = conv.max()
        eps_t = max(1e-12, max_val * 1e-9)
        max_idxs = np.where(conv >= max_val - eps_t)[0]

        grad = _gradient_at_max(a, max_idxs)

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        lr = lr0 / (1 + it * 1e-5)  # decay

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional perturbation to escape stagnation
        if (it % 4000 == 0 and it > 0) or np.random.rand() < 0.001:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1
    return best_a, best_score

"""
Propose a non‑negative step function (list of heights) that minimizes
evaluate_sequence.  The algorithm consists of four stages:

1.  Simple parametric families (inverse‑sqrt, power‑law, exponential,
    cosine, beta‑distribution, random block) are sampled and projected
    onto the simplex Σa = √(2 n).  The best candidate from this stage is
    kept.

2.  Simulated annealing with local mass transfers and occasional global
    Gaussian perturbations explores the neighbourhood of the best
    candidate while respecting the simplex constraints.

3.  Adam‑based sub‑gradient descent (the same sub‑gradient that was used
    in the reference solution) fine‑tunes the solution found by SA.

4.  A short random‑block refinement tries a few fresh block‑structured
    candidates to see if they improve the current best.

All convolutions are evaluated with FFT for speed, and every candidate
is kept on the simplex so the denominator of evaluate_sequence is
constant (Σa² = 2 n).  The function respects the 1000 s time budget and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using FFT."""
    fu = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fu * fu, n=size_fft)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _score(a: np.ndarray, size_fft: int) -> float:
    """Score = max entry of the self‑convolution (equal to evaluate_sequence)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _gradient_at_max(a: np.ndarray, max_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Σ_{k∈max_idxs}(a*a)_k w.r.t. a.
    Implements grad_i = 2· Σ_{k∈max_idxs} a_{k−i}  (indices outside [0,n) ignored).
    The gradient is averaged over the active set.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    if max_idxs.size == 0:
        return grad
    weight = 1.0 / max_idxs.size
    for k in max_idxs:
        i0 = max(0, k - (n - 1))
        i1 = min(n - 1, k)
        i_range = np.arange(i0, i1 + 1, dtype=int)
        grad[i_range] += 2.0 * a[k - i_range]
    grad *= weight
    return grad


# ----------------------------------------------------------------------
def _simulate_annealing(a0: np.ndarray,
                        cur_score0: float,
                        size_fft: int,
                        target_sum: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk on the simplex.
    Local moves transfer a small amount of mass between two indices;
    occasional global Gaussian perturbations are added.
    """
    a = a0.copy()
    cur_score = cur_score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    it = 0
    # maximum number of iterations – high enough to fill the time budget
    max_iters = 500_000

    while time.time() < deadline and it < max_iters:
        # linear cooling schedule (temperature in [0.5, 0.001])
        frac = it / max_iters
        T = max(0.001, (1.0 - frac) * 0.5)

        # decide move type
        if np.random.rand() < 0.9:               # local transfer
            i = np.random.randint(n)
            if a[i] <= 0.0:                      # nothing to move
                it += 1
                continue
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            # transfer up to ~20 % of the source mass at the start,
            # shrinking to ~1 % near the end of the schedule.
            max_delta = a[i]
            delta = max_delta * np.random.rand() * (0.2 * (1.0 - frac) + 0.01)
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:                                     # global perturbation
            sigma = target_sum * (0.01 * (1.0 - frac) + 0.001)
            noise = np.random.randn(n) * sigma
            a_new = a + noise
            a_new = _project_to_simplex(a_new, target_sum)

        new_score = _score(a_new, size_fft)

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[SA] iter {it:07d}  new best {best_score:.6f}")

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   cur_score0: float,
                   size_fft: int,
                   target_sum: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the true max‑convolution (no soft‑max).
    Uses the exact sub‑gradient on the active set of maximal indices.
    """
    a = a0.copy()
    cur_score = cur_score0
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    # learning‑rate schedule
    lr0 = 0.15
    while time.time() < deadline and it < 250_000:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] iter {it:07d}  new best {best_score:.6f}")

        # active set of maximal entries
        max_val = conv.max()
        eps_tol = max(1e-12, max_val * 1e-9)
        max_idxs = np.where(conv >= max_val - eps_tol)[0]

        grad = _gradient_at_max(a, max_idxs)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        lr = lr0 / (1.0 + it * 1e-5)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick to escape plateaus
        if (it % 4000 == 0 and it > 0) or np.random.rand() < 0.001:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step function within the allotted time.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3130                                    # fallback length
        have_ref = False

    target_sum = math.sqrt(2.0 * n)
    size_fft = 1 << ((2 * n - 1).bit_length())    # power of two ≥ 2n‑1

    # ------------------------------------------------------------------
    # Phase 1 – parametric shape search
    best_a = None
    best_score = float("inf")

    # uniform baseline
    a = np.full(n, target_sum / n, dtype=float)
    sc = _score(a, size_fft)
    best_a, best_score = a.copy(), sc
    print(f"[phase1] uniform score = {sc:.6f}")

    # reference (if available)
    if have_ref:
        ref = np.maximum(np.array(height_sequence_1, dtype=float), 0.0)
        ref = _project_to_simplex(ref, target_sum)
        sc = _score(ref, size_fft)
        if sc < best_score:
            best_a, best_score = ref.copy(), sc
        print(f"[phase1] reference score = {sc:.6f}")

    N_SAMPLES = 120

    # ----- inverse‑sqrt (single‑ended) -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        a = 1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        sc = _score(a, size_fft)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- double‑ended inverse‑sqrt -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        i = np.arange(n, dtype=float)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _score(a, size_fft)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- power‑law (double‑ended) -----
    for _ in range(N_SAMPLES):
        beta = np.random.rand() * 10.0
        p = np.random.rand() * 2.5 + 0.05
        i = np.arange(n, dtype=float)
        a = (1.0 / (i + beta + 1.0) ** p +
             1.0 / ((n - 1 - i) + beta + 1.0) ** p)
        a = _project_to_simplex(a, target_sum)
        sc = _score(a, size_fft)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- exponential decay (double‑ended) -----
    for _ in range(N_SAMPLES):
        decay = np.random.rand() * 0.012 + 0.0003
        i = np.arange(n, dtype=float)
        a = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
        a = _project_to_simplex(a, target_sum)
        sc = _score(a, size_fft)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- cosine‑based (large at ends) -----
    for _ in range(N_SAMPLES):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        sc = _score(a, size_fft)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- beta‑distribution shapes (U‑shaped) -----
    for _ in range(N_SAMPLES):
        a_par = np.random.rand() * 2.0 + 0.1
        b_par = np.random.rand() * 2.0 + 0.1
        x = (np.arange(n, dtype=float) + 0.5) / n
        a = (x ** (a_par - 1)) * ((1.0 - x) ** (b_par - 1))
        a = _project_to_simplex(a, target_sum)
        sc = _score(a, size_fft)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- random block candidates (few large blocks) -----
    for _ in range(300):
        nb = np.random.randint(2, 7)               # 2‑6 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _score(cand, size_fft)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    print(f"[phase1] best score after families = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing
    if time.time() < deadline:
        print("[phase2] simulated annealing start")
        best_a, best_score = _simulate_annealing(best_a, best_score,
                                                 size_fft, target_sum,
                                                 deadline)
        print(f"[phase2] best score after SA = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam sub‑gradient descent
    if time.time() < deadline:
        print("[phase3] Adam optimization start")
        best_a, best_score = _adam_optimize(best_a, best_score,
                                            size_fft, target_sum,
                                            deadline)
        print(f"[phase3] best score after Adam = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final random‑block refinement
    if time.time() < deadline:
        print("[phase4] final block refinement")
        for _ in range(200):
            nb = np.random.randint(2, 7)
            cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = np.random.rand(nb) + 0.1
            cand = np.zeros(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, target_sum)
            sc = _score(cand, size_fft)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[final block] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    result = best_a.tolist()
    if not result:                     # defensive fallback
        result = [target_sum]
    print(f"[final] best max‑convolution = {best_score:.6f}")
    return result
