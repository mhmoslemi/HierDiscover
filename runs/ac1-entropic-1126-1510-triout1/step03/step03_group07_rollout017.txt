"""
A fast non‑convex optimiser for the step‑function evaluation problem.

The objective 2 n·max(conv)/(∑a)² is scale‑invariant.
We fix the ℓ₁‑norm to S = √(2 n) (so that the score = max(conv)).
The optimiser uses

* an Adam‑style projected gradient descent on a smooth soft‑max
  approximation of the max‑convolution,
* an annealed soft‑max temperature (β) and learning rate,
* a final random‑pair polishing phase (simple simulated‑annealing).

Multiple problem sizes are tried (including a larger size than the
previous best) and the best candidate found before the time budget
expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the non‑negative simplex Σx = s.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {x≥0, Σx = target_sum}."""
    if target_sum <= 0:
        return np.zeros_like(v)
    # Duchi et al. (2008) – O(n log n) sort based algorithm.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(1, v.size + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: cheap random pairwise mass‑swap (preserves the ℓ₁‑norm).
# ----------------------------------------------------------------------
def _random_pair_move(seq: np.ndarray, sigma: float) -> np.ndarray:
    """Move a small amount of mass from one index to another (preserves sum)."""
    n = seq.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)

    # draw delta uniformly from (‑σ/2, σ/2)
    delta = (np.random.rand() - 0.5) * sigma
    # keep both entries non‑negative
    if delta > 0:
        delta = min(delta, seq[j])
    else:
        delta = -min(-delta, seq[i])

    new_seq = seq.copy()
    new_seq[i] += delta
    new_seq[j] -= delta
    return new_seq


# ----------------------------------------------------------------------
# FFT‑based full convolution a*a (length 2 n‑1).
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    nfft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(A * A, n=nfft)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# FFT‑based “valid” correlation: corr = np.correlate(w, a, mode='valid').
# ----------------------------------------------------------------------
def _fft_correlate(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    L = w.shape[0]                     # = 2 n‑1
    a_rev = a[::-1]
    nfft = 1 << ((L + n - 1).bit_length())
    W = np.fft.rfft(w, n=nfft)
    A_rev = np.fft.rfft(a_rev, n=nfft)
    full = np.fft.irfft(W * A_rev, n=nfft)[: L + n - 1]
    start = n - 1
    return full[start : start + n]    # length n (valid correlation)


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Optimise a non‑negative step‑function under the evaluation function.
    Returns the best sequence (list of floats) discovered before the
    time limit expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, budget_s - 2.0)

    # --------------------------------------------------------------
    # Problem sizes to try (starting from the previously known sequence).
    # --------------------------------------------------------------
    try:
        base_n = len(height_sequence_1)  # type: ignore
    except Exception:
        base_n = 2000

    # We will attempt three sizes – the last one is deliberately larger.
    candidate_lengths = [
        max(8000, int(base_n * 2)),
        30000,
        50000,
    ]
    # Clamp to a reasonable maximum (memory / time constraints).
    candidate_lengths = [min(60000, max(5000, L)) for L in candidate_lengths]
    candidate_lengths = sorted(set(candidate_lengths))

    best_seq = None
    best_score = float('inf')

    # --------------------------------------------------------------
    # Optimisation loop over the chosen lengths.
    # --------------------------------------------------------------
    for N in candidate_lengths:
        # Respect the overall time budget.
        if time.time() >= deadline:
            break

        # Target ℓ₁‑norm (scale‑invariant normalisation).
        target_sum = math.sqrt(2.0 * N)

        # ----- initialise a feasible point (uniform + tiny noise) -----
        a = np.full(N, target_sum / N, dtype=float)
        a += 0.01 * target_sum * np.random.rand(N) / N
        a = _project_onto_simplex(a, target_sum)

        # ----- Adam state -----
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        beta1_pow, beta2_pow = 1.0, 1.0

        # ----- schedules -----
        max_iters = 30000
        beta_start, beta_end = 0.5, 400.0          # soft‑max temperature
        lr_start, lr_end = 0.3, 0.02               # learning rate

        # ----- main gradient‑descent phase -----
        for it in range(1, max_iters + 1):
            if time.time() >= deadline:
                break

            # convolution of the current iterate
            conv = _fft_convolve(a)
            max_conv = conv.max()

            # keep track of the globally best sequence seen so far
            if max_conv < best_score:
                best_score = max_conv
                best_seq = a.copy()
                print(f"[N={N} iter={it:05d}] New global best score: {best_score:.6f}")

            # linear schedules for β (soft‑max sharpness) and learning rate
            t_frac = it / max_iters
            beta = beta_start + (beta_end - beta_start) * t_frac
            lr = lr_start + (lr_end - lr_start) * t_frac

            # soft‑max weights concentrating on the current peak(s)
            exp_vals = np.exp(beta * (conv - max_conv))
            w = exp_vals / exp_vals.sum()

            # gradient of the smooth max‑objective (2 · correlation)
            grad = 2.0 * _fft_correlate(w, a)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = np.maximum(a_new, 0.0)          # non‑negativity
            a = _project_onto_simplex(a_new, target_sum)

            # occasional re‑evaluation (cheap, already have conv)
            if it % 200 == 0:
                conv = _fft_convolve(a)
                max_conv = conv.max()
                if max_conv < best_score:
                    best_score = max_conv
                    best_seq = a.copy()
                    print(f"[N={N} eval={it:05d}] New global best score: {best_score:.6f}")

        # --------------------------------------------------------------
        # Polishing phase – random pairwise mass swaps (simulated annealing).
        # --------------------------------------------------------------
        if best_seq is None:
            continue

        polish_iters = 12000
        for p in range(1, polish_iters + 1):
            if time.time() >= deadline:
                break
            # linear cooling for the swap amplitude
            sigma = 0.35 * (1.0 - p / polish_iters) + 0.02
            cand = _random_pair_move(best_seq, sigma)

            conv = _fft_convolve(cand)
            max_conv = conv.max()
            if max_conv < best_score:
                best_score = max_conv
                best_seq = cand.copy()
                print(f"[Polish N={N} iter={p:05d}] New global best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: if nothing was found (should never happen), use uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        N = 20000
        target_sum = math.sqrt(2.0 * N)
        best_seq = np.full(N, target_sum / N, dtype=float)

    # Clip any tiny negatives that may appear due to numerical noise.
    best_seq = np.maximum(best_seq, 0.0)

    print(f"Finished optimisation – best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
