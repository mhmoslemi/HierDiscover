def propose_candidate(seed=0, budget_s=1000):

    start = time.time()
    deadline = start + budget_s - 5.0

    overall_best_seq = None
    overall_best_score = float('inf')
    # define candidate lengths array
    lengths = [20000, 30000, 40000, 50000, 60000, 80000]  # ensure <= some limit maybe disc space.

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"--- n={n} ---")
        candidates = []
        # arcsine
        candidates.append(_arcsine_distribution(n))
        # uniform
        candidates.append(np.full(n, 1.0/n))
        # beta set
        for alpha in [0.1,0.15,0.2,0.25,0.3]:
            candidates.append(_beta_distribution(n, alpha))
        # mixture arcsine+uniform
        arcs = _arcsine_distribution(n)
        uniform = np.full(n, 1.0/n)
        for w in [0.2,0.4,0.6,0.8]:
            mixture = w*arcs + (1-w)*uniform
            mixture /= mixture.sum()
            candidates.append(mixture)
        # few random low-concentration Dirichlet
        a = np.random.dirichlet(np.full(n, 0.3))
        candidates.append(a)

        best_n_seq = None
        best_n_score = float('inf')
        for init in candidates:
            if time.time() >= deadline: break
            seq = init.copy()
            # Ensure nonnegative
            seq = np.clip(seq, 0, None)
            seq_sum = seq.sum()
            if seq_sum == 0:
                continue
            seq /= seq_sum
            score = _score(seq)
            if score < best_n_score:
                best_n_score = score
                best_n_seq = seq.copy()
                print(f" init candidate score {score:.6f}")
            # optional quick refinement: subgradient descent 200 it
            seq_ref, score_ref = _subgradient_descent(
                n=n,
                init_seq=seq,
                deadline=deadline,
                max_iters=300,
                eta0=0.25
            )
            if score_ref < best_n_score:
                best_n_score = score_ref
                best_n_seq = seq_ref.copy()
                print(f"   refined score {score_ref:.6f}")

        # after all candidates for n, update global best
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"--- Global best update for n={n}: score {overall_best_score:.6f}")

        # early break if below target 1.503
        if overall_best_score <= 1.503:
            print("Reached target; exiting.")
            break

    # final refinement on global best
    if overall_best_seq is not None and time.time() < deadline:
        n = len(overall_best_seq)
        # Use exponentiated gradient for perhaps 5000 it
        seq_ref, score_ref = _optimise_exponential(
            n=n,
            init_seq=overall_best_seq,
            deadline=deadline,
            max_iters=8000,
            eta0=0.15,
            lambda0=0.3,
            lambda_max=30.0
        )
        if score_ref < overall_best_score:
            overall_best_score = score_ref
            overall_best_seq = seq_ref.copy()
            print(f"After exponential refine: score {overall_best_score:.6f}")

        # hill climb
        seq_ref, score_ref = _hill_climb(
            overall_best_seq,
            deadline,
            max_trials=25000
        )
        if score_ref < overall_best_score:
            overall_best_score = score_ref
            overall_best_seq = seq_ref.copy()
            print(f"After hill‑climb: score {overall_best_score:.6f}")

    if overall_best_seq is None:
        # fallback
        n = 1000
        overall_best_seq = np.full(n, 1.0/n)
        overall_best_score = _score(overall_best_seq)

    print(f"Final best: n={len(overall_best_seq)} score={overall_best_score:.6f}")
    return overall_best_seq.tolist()

"""
Search for a step‑function minimizing 2·n·max(conv(a,a))/sum(a)^2.

Approach
--------
* Scan several problem sizes (up to ≈80 k points) and evaluate a
  collection of analytically‑motivated candidate densities
  (arcsine, symmetric Beta with various α, uniform, mixtures).
* Keep the best candidate across all sizes.
* Refine the winner with an exponentiated‑gradient (soft‑max) descent
  that works directly on the max‑convolution, enforcing symmetry and
  simplex constraints after every step.
* Finish with a cheap stochastic hill‑climber that moves a tiny amount
  of mass between positions.
* All operations are O(n log n) using FFTs and run well within the
  1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path – already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (full length)."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector (length 2·n‑1),
    a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a sum‑1 vector."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α=½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _optimise_exponential(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 6000,
    eta0: float = 0.15,
    lambda0: float = 0.3,
    lambda_max: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of max(conv).
    Returns the best (sum‑1, symmetric) vector encountered and its true score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    # enforce symmetry from the start
    a = 0.5 * (a + a[::-1])

    best_seq = a.copy()
    best_score = _score(a)

    # FFT size for correlation (w length 2n‑1, a length n)
    m = 1 << ((3 * n - 2).bit_length())

    # linear schedule for soft‑max temperature λ
    lambda_inc = (lambda_max - lambda0) / max_iters
    lamb = lambda0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} exp it={it} score={best_score:.6f}")

        # soft‑max weights
        maxb = b.max()
        exp_arg = lamb * (b - maxb)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)

        # exponentiated‑gradient step
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # enforce symmetry and simplex constraint
        a = 0.5 * (a + a[::-1])
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)

        # occasional jitter to escape plateaus
        if it % 800 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = 0.5 * (a + a[::-1])
            a = a / a.sum()

        # increase λ
        lamb = min(lambda_max, lamb + lambda_inc)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Uses the sub‑gradient g_i = 2·a_{k*−i}.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    a = 0.5 * (a + a[::-1])

    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # build sub‑gradient g_i = 2·a_{k*−i}
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        # enforce symmetry again (helps stability)
        a = 0.5 * (a + a[::-1])

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} sub it={it} score={best_score:.6f}")

        # jitter occasionally
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = np.clip(a, 0.0, None)
            a = 0.5 * (a + a[::-1])
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny mass from one index
    to another and keep the change if the (unsmoothed) objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta
        # No projection needed – sum unchanged & non‑negative by construction
        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score

"""
Search for a step‑function minimizing 2·n·max(conv(a,a))/sum(a)^2.

Approach
--------
* Scan several problem sizes (up to ≈80 k points) and evaluate a
  collection of analytically‑motivated candidate densities
  (arcsine, symmetric Beta with various α, uniform, mixtures).
* Keep the best candidate across all sizes.
* Refine the winner with an exponentiated‑gradient (soft‑max) descent
  that works directly on the max‑convolution, enforcing symmetry and
  simplex constraints after every step.
* Finish with a cheap stochastic hill‑climber that moves a tiny amount
  of mass between positions.
* All operations are O(n log n) using FFTs and run well within the
  1000 s budget.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT (full length)."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g_i = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector (length 2·n‑1),
    a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns g[0…n‑1].
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]

    n = len(a)

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the grid x_i = (i+½)/n (normalised)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals

    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

                print(f"    sub‑refined score={score_ref:.6f}")

        # ----- update global best -----
        if best_n_seq is not None and best_n_score < overall_best_score:
            overall_best_score = best_n_score
            overall_best_seq = best_n_seq.copy()
            print(f"*** Global best updated: n={n} score={overall_best_score:.6f}")

        if overall_best_score <= 1.503:
            print("Target achieved – exiting early.")
            break

    # ------------------------------------------------------------------
    # 2️⃣ intensive refinement on the current global best
    if overall_best_seq is not None and time.time() < deadline:
        n = len(overall_best_seq)
        print(f"\n=== Intensive refinement on length {n} ===")
        seq_ref, score_ref = _optimise_exponential(
            n=n,
            init_seq=overall_best_seq,
            deadline=deadline,
            max_iters=8000,
            eta0=0.15,
            lambda0=0.3,
            lambda_max=30.0,
        )
        if score_ref < overall_best_score:
            overall_best_score = score_ref
            overall_best_seq = seq_ref.copy()
            print(f"After exponential refine: score={overall_best_score:.6f}")

        seq_ref, score_ref = _hill_climb(
            overall_best_seq,
            deadline,
            max_trials=25000,
        )
        if score_ref < overall_best_score:
            overall_best_score = score_ref
            overall_best_seq = seq_ref.copy()
            print(f"After hill‑climb: score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ fallback (should never happen)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"\nFinal best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    return overall_best_seq.tolist()
