    i = np.arange(n, dtype=np.float64)

def _optimize_params(n, deadline):
    # start with beta=0.5, eps=0.5
    beta = 0.5
    eps = 0.5
    best_seq = _distribution_beta_eps(n, beta, eps)
    best_score = _score_of_distribution(best_seq)
    # Step size schedule
    delta_beta = 0.1
    delta_eps = 1.0
    while time.time() < deadline and (delta_beta > 1e-4 or delta_eps > 1e-2):
        improved = False
        for db in ( -delta_beta, 0, delta_beta):
            for de in ( -delta_eps, 0, delta_eps):
                if db==0 and de==0:
                    continue
                nb = beta + db
                ne = eps + de
                if nb < 0.0 or nb > 0.5 or ne <= 0.0:
                    continue
                seq = _distribution_beta_eps(n, nb, ne)
                score = _score_of_distribution(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    beta, eps = nb, ne
                    improved = True
        if not improved:
            delta_beta *= 0.5
            delta_eps *= 0.5
    return best_seq, best_score
```

def _param_optimize(n, deadline):
    # start with arcsine
    best_beta = 0.5
    best_eps = 0.5
    best_seq = _dist_beta_eps(n, best_beta, best_eps)
    best_score = _score_of_seq(best_seq)
    # parameters for search
    beta_step = 0.1
    eps_step = 1.0
    # random restarts: maybe every few iterations do new random start
    while time.time() < deadline:
        # Try neighbor moves
        improved = False
        for db in (-beta_step, 0, beta_step):
            nb = best_beta + db
            if nb < 0: continue
            if nb > 0.5: continue
            for de in (-eps_step,0,eps_step):
                ne = best_eps + de
                if ne <= 0: continue
                if ne >= n/2: continue  # ensure positivity in denominator
                if db == 0 and de == 0:
                    continue
                seq = _dist_beta_eps(n, nb, ne)
                score = _score_of_seq(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    best_beta, best_eps = nb, ne
                    improved = True
        if not improved:
            # shrink step sizes
            beta_step *= 0.5
            eps_step *= 0.5
        # random restart occasional if steps too small
        if beta_step < 1e-4 and eps_step < 1e-2 and np.random.rand() < 0.3:

                best_beta, best_eps = nb, ne
        # early exit if step sizes small enough

def _dist_beta_eps(n, beta, eps):
    i = np.arange(n, dtype=np.float64)
    # Ensure denominators remain positive
    # Might clip lower bound for i+eps (should be >0)
    log_vals = -beta * (np.log(i + eps) + np.log(n - i - eps))
    # Subtract max for stability
    log_vals -= np.max(log_vals)
    vals = np.exp(log_vals)
    total = np.sum(vals)
    if total == 0:
        # fallback uniform
        return np.full(n, 1.0/n, dtype=np.float64)
    vals /= total
    return vals
```

"""
Propose a non‑negative step function that minimizes `evaluate_sequence`.
The algorithm consists of two phases:

1. **Parametric search** – a family of symmetric distributions
   a_i ∝ (i+ε)^{-β} (n‑i‑ε)^{-β} (β∈[0,0.5], ε>0) is explored.
   The parameters (β,ε) are tuned by a simple hill‑climbing
   (geometric step reduction + occasional random restarts).  The
   candidate is always normalised to sum‑to‑1, so the evaluation
   reduces to `2·n·max(a*a)`.

2. **Sub‑gradient refinement** – starting from the best parametric
   candidate a projected‑gradient descent (sub‑gradient of the current
   maximal convolution entry) with momentum and a diminishing step
   size is performed.  After each step the vector is projected back onto
   the probability simplex.

Both phases run within the global time budget (default 1000 s).  The best
sequence found across a few different lengths is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Exact internal score 2·n·max(a*a) for a normalised (sum=1) vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _beta_eps_distribution(n: int, beta: float, eps: float) -> np.ndarray:
    """
    Symmetric distribution a_i ∝ (i+eps)^{-beta} (n-i-eps)^{-beta},
    returned normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    # avoid log(0)
    log_vals = -beta * (np.log(i + eps) + np.log(n - i - eps))
    # stabilise the exponentiation
    log_vals -= np.max(log_vals)
    vals = np.exp(log_vals)
    total = np.sum(vals)
    if total == 0.0:
        # fallback to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    vals /= total
    return vals


# ----------------------------------------------------------------------
def _param_optimize(n: int, deadline: float):
    """
    Hill‑climbing on (beta, eps).  Returns the best distribution found
    and its internal score.
    """
    # start from the classic arcsine distribution
    best_beta, best_eps = 0.5, 0.5
    best_seq = _beta_eps_distribution(n, best_beta, best_eps)
    best_score = _score_of_seq(best_seq)

    beta_step = 0.12
    eps_step = max(0.5, n * 0.02)   # start with a few percent of n

    while time.time() < deadline:
        improved = False
        for db in (-beta_step, 0.0, beta_step):
            nb = best_beta + db
            if nb < 0.0 or nb > 0.5:
                continue
            for de in (-eps_step, 0.0, eps_step):
                ne = best_eps + de
                # keep denominators positive
                if ne <= 0.0 or ne >= n - 1.0:
                    continue
                if db == 0.0 and de == 0.0:
                    continue
                seq = _beta_eps_distribution(n, nb, ne)
                sc = _score_of_seq(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq
                    best_beta, best_eps = nb, ne
                    improved = True
        if not improved:
            beta_step *= 0.5
            eps_step *= 0.5
        # occasional random restart when we are already fine‑grained
        if beta_step < 1e-4 and eps_step < 1e-3 and np.random.rand() < 0.25:
            nb = np.random.rand() * 0.5
            ne = 0.5 + np.random.rand() * (n * 0.25)
            seq = _beta_eps_distribution(n, nb, ne)
            sc = _score_of_seq(seq)
            if sc < best_score:
                best_score, best_seq = sc, seq
                best_beta, best_eps = nb, ne
        if beta_step < 1e-5 and eps_step < 1e-4:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _refine_subgradient(a0: np.ndarray, n: int, deadline: float):
    """
    Sub‑gradient descent on the true max‑convolution objective.
    Uses a simple momentum scheme and Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_of_seq(a)

    beta_mom = 0.9          # momentum coefficient
    eta0 = 0.8              # initial step factor
    velocity = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        # index of the current maximal convolution entry
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2·a_{k‑i} for valid indices
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # momentum update
        velocity = beta_mom * velocity + (1.0 - beta_mom) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + simplex projection
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

    return best_a, best_score
