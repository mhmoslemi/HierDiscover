def _beta_distribution(alpha: float, n: int) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    vals = np.power(x, alpha-1) * np.power(1 - x, alpha-1)
    # may cause overflow, clip to large number; compute using log:
    # better compute in log to avoid overflow: log vals = (alpha-1)*(log(x)+log(1-x)).
    # Then exponentiate after subtract max log.
    log_vals = (alpha-1)*(np.log(x) + np.log(1-x))
    # subtract max for numerical stability
    max_log = np.max(log_vals)
    vals = np.exp(log_vals - max_log)
    vals /= vals.sum()
    return vals

"""
Proposed solution
-----------------
A projected Adam optimiser is used to minimise the peak of the self‑convolution
a*a.  The objective  max_k (a*a)_k  is approximated by a smooth soft‑max
Lα(a) = (1/α)·log Σ_k exp(α·(b_k−max(b)))  where b = a*a.
The gradient of Lα is a convolution of the reversed a with the soft‑max
weights; after each step the vector is projected onto the probability simplex.
Several sensible initialisations are tried (uniform, arcsine, a family of
Beta‑distributions and the previous best sequence).  The total time budget
(≈1000 s) is split among these starts, and the overall best sequence is
returned.  A final refinement is performed on the current best for any
remaining time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density on {0,…,n‑1}, normalised to sum‑1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in the log‑domain to avoid overflow for α<1
    log_vals = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    log_vals -= np.max(log_vals)          # shift for stability
    vals = np.exp(log_vals)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Arcsine distribution = Beta(0.5,0.5)."""
    return _beta_distribution(0.5, n)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·len(a)−1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(v: np.ndarray, target_len: int) -> np.ndarray:
    """Resize a 1‑D vector to target_len by linear interpolation."""
    if v.size == target_len:
        return v.copy()
    x_old = np.linspace(0, 1, v.size)
    x_new = np.linspace(0, 1, target_len)
    return np.interp(x_new, x_old, v)


# ----------------------------------------------------------------------
def _local_optimize(a: np.ndarray, n: int, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the soft‑max surrogate.
    Returns the best vector seen and its max‑convolution value.
    """
    a = _project_onto_simplex(a, s=1.0)
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_max = b.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # temperature (soft‑max sharpness)
    alpha = 2.0
    alpha_factor = 1.025
    alpha_max = 5000.0

    it = 0
    while time.time() < time_limit:
        it += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()

        # record improvement
        if cur_max < best_max - 1e-12:
            best_max = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={it} maxB={best_max:.8e}")

        # soft‑max weights
        exp_term = np.exp(alpha * (b - cur_max))
        w_sum = exp_term.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = exp_term / w_sum

        # gradient = 2 * (rev(a) * w) sliced to length n
        a_rev = a[::-1]
        conv_rev = _fft_convolve(a_rev, w)               # length = 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam update with a decaying learning rate
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = 0.6 / np.sqrt(it)                           # base step size
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

    return best_seq, best_max
