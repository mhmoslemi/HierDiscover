Propose a step‑function (non‑negative sequence) with a low value of

Algorithm
---------
1. **Parametric initialization** – generate many candidates from a few
   simple families (inverse‑sqrt, double‑ended power‑law, exponential,
   cosine, mixtures, random block).  Every candidate is projected onto the
   simplex {a_i ≥ 0, Σ a_i = √(2 n)} and its true score (max‑convolution) is
   evaluated.  The best of them becomes the starting point.

2. **Smoothed gradient descent** – minimise a smooth approximation of the
   max‑convolution:
        f_T(a) = (1/T)·log Σ_k exp(T·(a∗a)_k) .

        ∇f_T(a)_i = 2·Σ_k w_k·a_{k‑i}   with   w_k ∝ exp(T·(a∗a)_k) .

   The temperature T is decreased exponentially from 5 to ≈2·10⁻³,

   After each step the vector is projected back onto the simplex.
   The best objective value seen (the true max‑convolution) is recorded.

3. **Random‑block refinement** – after the descent a handful of
   random block‑structured perturbations and small Gaussian kicks are tried.
   Any improvement is kept.

The routine respects the 1000 s time budget, prints progress
messages, and finally returns the best sequence found as a plain list of
floats.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n−1)."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, T: float) -> np.ndarray:
    """Stable soft‑max weights for a given temperature T."""
    # shift by the maximum to avoid overflow
    shifted = conv - conv.max()
    # when T is tiny the exponent may under‑flow → all zeros;
    # in that case fall back to a uniform distribution
    with np.errstate(over='ignore', under='ignore'):
        w = np.exp(shifted / T)
    s = w.sum()
    if s == 0.0 or not np.isfinite(s):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = w / s
    return w

        sc = _true_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- double‑ended inverse‑sqrt -----
    for _ in range(NSAMPLE):
        beta = np.random.rand() * 20.0
        i = np.arange(n, dtype=float)
        a = (1.0 / np.sqrt(i + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - i) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        sc = _true_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- double‑ended power‑law -----
    for _ in range(NSAMPLE):
        beta = np.random.rand() * 20.0
        p = np.random.rand() * 2.0 + 0.1
        i = np.arange(n, dtype=float)
        a = (1.0 / (i + beta + 1.0) ** p +
             1.0 / ((n - 1 - i) + beta + 1.0) ** p)
        a = _project_to_simplex(a, target_sum)
        sc = _true_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- exponential decay (double‑ended) -----
    for _ in range(NSAMPLE):
        decay = np.random.rand() * 0.02 + 1e-4
        i = np.arange(n, dtype=float)
        a = np.exp(-decay * i) + np.exp(-decay * (n - 1 - i))
        a = _project_to_simplex(a, target_sum)
        sc = _true_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- cosine‑based (large at ends) -----
    for _ in range(NSAMPLE):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        sc = _true_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- mixture of uniform & inverse‑sqrt -----
    for _ in range(NSAMPLE):
        beta = np.random.rand() * 20.0
        lam = np.random.rand()
        a = lam * np.ones(n, dtype=float) + (1.0 - lam) * (
            1.0 / np.sqrt(np.arange(n, dtype=float) + beta + 1.0)
        )
        a = _project_to_simplex(a, target_sum)
        sc = _true_score(a)
        if sc < best_score:
            best_a, best_score = a.copy(), sc

    # ----- random block candidates (few large blocks) -----
    for _ in range(2000):
        nb = np.random.randint(2, 8)                # 2‑7 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.zeros(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _true_score(cand)
        if sc < best_score:
            best_a, best_score = cand.copy(), sc

    print(f"[init] best score after families = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. SMOOTHED GRADIENT DESCENT
    # --------------------------------------------------------------
    a = best_a.copy()

    # schedule parameters
    max_iters = 8000
    it = 0
    lr_start, lr_end = 0.20, 0.005
    T_start, T_end = 5.0, 0.002

    while time.time() < deadline and it < max_iters:
        # current true score
        conv = _self_convolution(a, size_fft_self)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[grad] iter {it:05d}  new best = {best_score:.6f}")

        # temperature (exponential decay)
        T = T_start * (T_end / T_start) ** (it / max_iters)

        # soft‑max weights (stable)
        w = _softmax_weights(conv, T)

        # gradient = 2·(w * a_rev)  (linear convolution)
        a_rev = a[::-1]
        grad_full = _conv_fft(w, a_rev, size_fft_grad)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # learning‑rate schedule (exponential decay)
        lr = lr_start * (lr_end / lr_start) ** (it / max_iters)

        # projected gradient step
        a_new = a - lr * grad
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional stochastic kick to escape plateaus
        if it % 2000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 1e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    print(f"[grad] finished after {it} iterations, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. FINAL RANDOM‑BLOCK REFINEMENT
    # --------------------------------------------------------------
    for _ in range(5000):
        cand = best_a.copy()
        # generate a new block‑structured perturbation
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = np.random.rand(nb) + 0.05
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _true_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[refine] improved to {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
