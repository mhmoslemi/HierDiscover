"""
Propose a low‑score step‑function sequence.

Algorithm
---------
* Phase 1 – coarse search:
  – Try several lengths (up to 50 000) and a wide family of
    analytic candidates (arcsine with a tunable offset, symmetric
    Beta(α,α) with a dense grid of α, uniform, a few Dirichlet draws,
    sparse “spike” vectors).
  – Keep the best few candidates for each length (exact objective is
    computed via FFT‑based convolution, which is O(n log n) and very
    fast).

* Phase 2 – refinement:
  – Starting from the globally best candidate run exponentiated‑gradient
    descent on a soft‑max approximation of the peak of the self‑convolution.
    The update uses a momentum term, a linearly‑increasing temperature
    λ and a √t step‑size schedule.
  – A short projected sub‑gradient phase (exact max‑convolution peak) is
    applied afterwards.

* Phase 3 – cheap stochastic hill‑climbing:
  – Repeatedly move a tiny amount of mass from one index to another and
    keep the change if the true objective improves.

The whole procedure respects the 1000 s budget and returns a plain
Python list ready for evaluation.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two FFT."""
    n = len(a)
    m = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A * A, m)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_correlation(w: np.ndarray, a: np.ndarray, m: int) -> np.ndarray:
    """
    Compute g = 2·Σ_k w_k·a_{k−i} (correlation) using FFT.
    w – weight vector of length 2·n‑1, a – current sequence (length n),
    m – FFT length (power of two, at least len(w)+len(a)-1).
    Returns vector of length n.
    """
    W = np.fft.rfft(w, m)
    A = np.fft.rfft(a, m)
    corr = np.fft.irfft(np.conj(W) * A, m)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a).max())


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the grid (i+½)/n, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_offset_distribution(n: int, offset: float) -> np.ndarray:
    """
    Discrete arcsine‑like law with a tunable offset:
        a_i ∝ 1 / sqrt((i+offset) * (n-i-offset))
    offset must be in (0, 1).  The distribution is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    denom1 = i + offset
    denom2 = n - i - offset
    # protect against zero / negative values (offset < 1 ensures positivity)
    vals = 1.0 / np.sqrt(np.maximum(denom1 * denom2, np.finfo(np.float64).tiny))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _optimise_exponential_adv(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 12000,
    eta0: float = 0.2,
    lambda0: float = 0.1,
    lambda_max: float = 60.0,
    beta1: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on a soft‑max approximation of
    max(conv(a,a)).  Momentum (parameter beta1) is used to accelerate
    convergence.  Returns the best vector (sum = 1) seen and its exact score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    # FFT size for correlation (w length 2n‑1, a length n)
    m = 1 << ((3 * n - 2).bit_length())

    # λ schedule – exponential growth from lambda0 to lambda_max
    lam = lambda0
    lam_factor = (lambda_max / lambda0) ** (1.0 / max_iters)

    # momentum buffer
    m_buf = np.zeros_like(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current objective
        b = _conv_full_fft(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # soft‑max weights
        maxb = b.max()
        exp_arg = lam * (b - maxb)          # ≤ 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(b, 1.0 / b.shape[0])
        else:
            w /= w_sum

        # gradient via correlation
        g = _gradient_via_correlation(w, a, m)

        # momentum update
        m_buf = beta1 * m_buf + (1.0 - beta1) * g

        # step size
        eta = eta0 / np.sqrt(it)

        # mirror step (exponentiated gradient)
        a = a * np.exp(-eta * m_buf)

        # projection onto simplex (renormalisation)
        a_sum = a.sum()
        if a_sum > 0.0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)

        # increase λ
        lam = min(lambda_max, lam * lam_factor)

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.3:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 4000,
    eta0: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    Uses the sub‑gradient g_i = 2·a_{k*−i} where k* is an argmax of conv.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        b = _conv_full_fft(a)
        k_star = int(np.argmax(b))

        # sub‑gradient: g_i = 2·a_{k*−i} where the index is inside [0,n)
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub‑it={it:5d} score={best_score:.6f}")

        # jitter to avoid stagnation
        if it % 800 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 25000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search.  Move a tiny amount of mass from
    a random donor index to a random receiver and keep the move if the
    exact objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # transfer at most 5 % of the donor mass, but cap to 0.01
        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    overall_best_n = None

    # ------------------------------------------------------------------
    # 0️⃣  Historic sequence, if present
    try:
        hist = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        hist = np.clip(hist, 0.0, None)
        if hist.sum() > 0.0:
            hist = hist / hist.sum()
            hs = _score(hist)
            overall_best_seq = hist.copy()
            overall_best_score = hs
            overall_best_n = len(hist)
            print(f"Historic sequence: length={overall_best_n} score={hs:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Coarse search over many lengths and simple analytic families
    length_candidates = [50000, 40000, 30000, 20000, 15000,
                         12000, 10000, 8000, 6000, 4000]

    for n in length_candidates:
        if time.time() >= deadline:
            break
        print(f"--- Coarse search for length n={n} ---")

        # keep up to K best local candidates for a later refinement
        K = 4
        local_best = []                 # list of (score, seq)

        def _maybe_store(seq):
            s = _score(seq)
            if len(local_best) < K:
                local_best.append((s, seq.copy()))
                local_best.sort(key=lambda x: x[0])
            else:
                if s < local_best[-1][0]:
                    local_best[-1] = (s, seq.copy())
                    local_best.sort(key=lambda x: x[0])

        # Uniform
        _maybe_store(np.full(n, 1.0 / n, dtype=np.float64))

        # Arcsine with tunable offset
        for off in np.linspace(0.15, 0.85, 15):
            _maybe_store(_arcsine_offset_distribution(n, off))

        # Symmetric Beta(α,α) with a fine grid of α
        for α in np.linspace(0.35, 0.85, 21):
            _maybe_store(_beta_distribution(n, α))

        # Low‑concentration Dirichlet draw
        try:
            _maybe_store(np.random.dirichlet(np.full(n, 0.3)))
        except Exception:
            pass

        # Sparse random “spike” vector
        k = max(2, n // 500)    # about 0.2 % non‑zeros
        idx = np.random.choice(n, size=k, replace=False)
        a_spike = np.zeros(n, dtype=np.float64)
        a_spike[idx] = np.random.rand(k) + 0.1
        _maybe_store(a_spike / a_spike.sum())

        # Mix arcsine with uniform (convex combinations)
        arcs = _arcsine_offset_distribution(n, 0.5)
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        for λ in np.linspace(0.1, 0.9, 9):
            mix = λ * arcs + (1 - λ) * uni
            _maybe_store(mix / mix.sum())

        # Update global best with the locally best candidates
        for s, seq in local_best:
            if s < overall_best_score:
                overall_best_score = s
                overall_best_seq = seq.copy()
                overall_best_n = n
                print(f"*** New global best (coarse) length={n} score={s:.6f}")

        # early exit if target already reached
        if overall_best_score <= 1.503:
            print("Target reached during coarse phase.")
            break

    # ------------------------------------------------------------------
    # 2️⃣  Refinement of the best candidate found so far
    if overall_best_seq is not None and time.time() < deadline:
        n = len(overall_best_seq)
        print(f"Refining best sequence of length {n}")

        # Exponentiated‑gradient with momentum
        seq, sc = _optimise_exponential_adv(
            n=n,
            init_seq=overall_best_seq,
            deadline=deadline,
            max_iters=12000,
            eta0=0.2,
            lambda0=0.1,
            lambda_max=60.0,
            beta1=0.9,
        )
        if sc < overall_best_score:
            overall_best_seq = seq.copy()
            overall_best_score = sc
            print(f"After EG refinement score={sc:.6f}")

        # Projected sub‑gradient descent
        if time.time() < deadline:
            seq, sc = _subgradient_descent(
                n=n,
                init_seq=overall_best_seq,
                deadline=deadline,
                max_iters=4000,
                eta0=0.3,
            )
            if sc < overall_best_score:
                overall_best_seq = seq.copy()
                overall_best_score = sc
                print(f"After sub‑gradient refinement score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final stochastic hill‑climbing
    if overall_best_seq is not None and time.time() < deadline:
        seq, sc = _hill_climb(overall_best_seq, deadline, max_trials=25000)
        if sc < overall_best_score:
            overall_best_seq = seq.copy()
            overall_best_score = sc
            print(f"After hill‑climb final score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Fallback (should never trigger)
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _score(overall_best_seq)
        print("Fallback uniform sequence used.")

    print(f"Final best: length={len(overall_best_seq)} score={overall_best_score:.6f}")
    # Return a plain Python list for the evaluation harness
    return overall_best_seq.tolist()
