"""
propose_candidate
-----------------

Searches for a non‑negative step function (a list of floats) that minimizes
the evaluation function

    2 · n · max(self‑convolution) / (sum(sequence)²)

The algorithm consists of three phases:

1. **Broad random‑shape sampling** – many families of analytically
   parametrised shapes (inverse‑sqrt, power‑law, exponential,
   cosine, sine, random blocks, Dirichlet noise, mixtures, …) are
   generated.  Each candidate is projected onto the simplex
   {a ≥ 0, Σa = √(2 n)} and its maximal convolution entry is measured
   with an FFT.  The best candidate seen so far is retained.

2. **Simple simulated‑annealing hill‑climbing** – starting from the best
   candidate, random mass transfers between two coordinates are proposed.
   Moves are accepted if they improve the score or with a probability
   governed by a slowly cooling temperature.  The step size is tied to
   the current temperature.  This explores the neighbourhood of the
   best shape much more aggressively than a pure sub‑gradient descent.

3. **Final random‑block sweep** – a last batch of fresh block‑structured
   candidates is examined to see whether they beat the current optimum.

The routine respects the 1000 s time budget, prints intermediate scores,
and finally returns the best sequence as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    return _conv_fft(a, a, size_fft)


def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Maximum entry of a*a (the quantity we minimise)."""
    return float(_self_convolution(a, size_fft).max())


def _project_and_score(raw: np.ndarray, target_sum: float,
                       size_fft: int) -> tuple[np.ndarray, float]:
    """Project a raw vector onto the simplex and return (proj, max‑conv)."""
    proj = _project_to_simplex(raw, target_sum)
    score = _max_convolution(proj, size_fft)
    return proj, score

# ----- shape generators (return an un‑normalised np.ndarray) -------------
def _shape_uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=float)


def _shape_dirichlet(n: int) -> np.ndarray:
    # Positive random values; Dirichlet(1,…,1) is equivalent to normalising
    # iid exponential draws, but uniform(0,1) is cheaper and works as well.
    return np.random.rand(n)


def _shape_inverse_sqrt(n: int) -> np.ndarray:
    beta = np.random.rand() * 10.0
    i = np.arange(n, dtype=float)
    return 1.0 / np.sqrt(i + beta + 1.0)


def _shape_double_inverse_sqrt(n: int) -> np.ndarray:
    beta = np.random.rand() * 10.0
    i = np.arange(n, dtype=float)
    left = 1.0 / np.sqrt(i + beta + 1.0)
    right = 1.0 / np.sqrt((n - 1 - i) + beta + 1.0)
    return left + right


def _shape_power_law(n: int) -> np.ndarray:
    beta = np.random.rand() * 10.0
    p = np.random.rand() * 2.5 + 0.1   # exponent in (0.1, 2.6)
    i = np.arange(n, dtype=float)
    left = 1.0 / (i + beta + 1.0) ** p
    right = 1.0 / ((n - 1 - i) + beta + 1.0) ** p
    return left + right


def _shape_exponential(n: int) -> np.ndarray:
    decay = np.random.rand() * 0.02 + 0.0005   # [5e‑4, 2.05e‑2]
    i = np.arange(n, dtype=float)
    left = np.exp(-decay * i)
    right = np.exp(-decay * (n - 1 - i))
    return left + right


def _shape_cosine_U(n: int) -> np.ndarray:
    # U‑shaped: large at the ends, small in the middle
    p = np.random.rand() * 5.0 + 0.5
    a = np.abs(np.cos(np.pi * np.arange(n, dtype=float) / (n - 1))) ** p
    return a


def _shape_sine_M(n: int) -> np.ndarray:
    # Mountain‑shaped: zero at ends, peak in the centre
    p = np.random.rand() * 4.0 + 0.5
    a = np.sin(np.pi * np.arange(n, dtype=float) / (n - 1)) ** p
    return a


def _shape_random_blocks(n: int) -> np.ndarray:
    nb = np.random.randint(2, 21)          # 2‑20 blocks
    cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = np.random.rand(nb) + 0.1        # avoid exact zeros
    out = np.empty(n, dtype=float)
    for blk, v in zip(blocks, vals):
        out[blk] = v
    return out


def _shape_symmetric_blocks(n: int) -> np.ndarray:
    # high blocks at both ends, low block in the centre
    left_len = np.random.randint(1, n // 3 + 1)
    mid_len = n - 2 * left_len
    high = np.random.rand() + 0.5          # >0.5
    low = np.random.rand() * high * 0.5   # definitely lower than high
    left = np.full(left_len, high, dtype=float)
    middle = np.full(mid_len, low, dtype=float)
    right = left.copy()
    return np.concatenate([left, middle, right])


def _shape_mixture(n: int) -> np.ndarray:
    # Random convex combination of uniform and a double inverse sqrt
    lam = np.random.rand()
    uniform = np.ones(n, dtype=float)
    double = _shape_double_inverse_sqrt(n)
    return lam * uniform + (1.0 - lam) * double

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a list of non‑negative heights (a step function) that achieves a
    low value of `evaluate_sequence`.  The routine runs for at most
    `budget_s` seconds (default 1000) and prints intermediate progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # small safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3130                                   # fallback length
        have_ref = False

    target_sum = math.sqrt(2.0 * n)                # normalisation used in the paper
    size_fft = 1 << ((2 * n - 1).bit_length())    # power‑of‑two >= 2·n‑1

    # --------------------------------------------------------------
    # keep the best candidate seen so far
    best_seq = None
    best_score = float("inf")

    def _update_best(candidate: np.ndarray, score: float) -> None:
        nonlocal best_seq, best_score
        if score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[new best] score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1. EXTENSIVE RANDOM SHAPE SEARCH
    # --------------------------------------------------------------

    uniform_raw = _shape_uniform(n)
    uniform_proj, uniform_score = _project_and_score(uniform_raw, target_sum, size_fft)
    _update_best(uniform_proj, uniform_score)

    # if a reference sequence is available, evaluate it as well
    if have_ref:
        ref = np.maximum(np.array(height_sequence_1, dtype=float), 0.0)
        ref_proj, ref_score = _project_and_score(ref, target_sum, size_fft)
        _update_best(ref_proj, ref_score)
        print(f"[reference] score = {ref_score:.6f}")

    # Main random sampling loop – we keep generating candidates until
    # 70 % of the overall time budget is consumed.
    phase1_deadline = time.time() + (budget_s * 0.7)
    while time.time() < phase1_deadline:
        gen = np.random.choice(_SHAPE_GENERATORS)
        raw = gen(n)                               # un‑normalised positive vector
        proj, sc = _project_and_score(raw, target_sum, size_fft)
        _update_best(proj, sc)

    print(f"[phase 1] best score after random families = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. SIMULATED‑ANNEALING HILL‑CLIMBING FROM THE BEST SO FAR
    # --------------------------------------------------------------
    # Start from the best candidate found in phase 1.
    cur_seq = best_seq.copy()
    cur_score = best_score

    # Temperature schedule – start relatively high and cool slowly.
    init_temp = cur_score * 0.05 + 1e-6
    temp = init_temp
    temp_decay = 0.99995

    # Step size is a fraction of the total mass; it shrinks together with
    # the temperature.
    base_step = target_sum * 0.03

    it = 0
    while time.time() < deadline:
        # choose two distinct indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a small amount of mass from i to j
        max_move = cur_seq[i]
        if max_move <= 0.0:
            continue
        # the fraction of mass to move depends on the temperature
        frac = np.random.rand() * (temp / init_temp)
        delta = max_move * frac
        if delta <= 0.0:
            continue

        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        # score the candidate (projection is not needed – sum is preserved)
        cand_score = _max_convolution(cand, size_fft)

        # Metropolis acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = cand
            cur_score = cand_score
            if cur_score < best_score:
                _update_best(cur_seq, cur_score)

        # anneal temperature and occasionally inject a larger random perturbation
        temp *= temp_decay
        it += 1
        if it % 50000 == 0:
            # a gentle shake to avoid getting stuck
            shake = np.random.randn(n) * target_sum * 1e-3
            cur_seq = _project_to_simplex(cur_seq + shake, target_sum)
            cur_score = _max_convolution(cur_seq, size_fft)
            print(f"[shake] iteration {it:,} score = {cur_score:.6f}")

    print(f"[phase 2] hill‑climb finished (iterations {it:,}), best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. FINAL BLOCK‑BASED REFINEMENT (lightweight)
    # --------------------------------------------------------------
    final_deadline = deadline
    while time.time() < final_deadline:
        raw = _shape_random_blocks(n)
        proj, sc = _project_and_score(raw, target_sum, size_fft)
        _update_best(proj, sc)

    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()
