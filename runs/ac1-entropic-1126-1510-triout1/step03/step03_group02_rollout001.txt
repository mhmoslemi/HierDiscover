"""
Propose a step‑function sequence with a low self‑convolution peak.

Algorithm
---------
1. **Exploratory parametric search** – generate a diverse set of
   analytic families (uniform, inverse‑sqrt, double‑ended,
   power‑law, exponential, cosine, sine, Gaussian bumps, random
   block patterns) and keep the K best raw candidates according to
   the exact score (no security checks).

2. **Intensive refinement** – take the best raw candidate and run a
   long soft‑max gradient descent (Nesterov‑type momentum with
   Adam‑style adaptation).  The objective is a smooth approximation
   of the max‑convolution; after each step the iterate is projected
   onto the simplex {a≥0, Σa = √(2 n)}.

3. **Simulated‑annealing polish** – starting from the refined
   solution perform a cheap Metropolis‑Hastings walk that perturbs
   the vector with Gaussian noise, projects onto the simplex and
   accepts moves with probability exp(−Δscore/T).  The temperature
   is slowly cooled.

4. **Final random‑block sweep** – a quick pass of random block
   candidates to capture any missed improvement.

All steps respect the 1000 s time budget and print progress.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution (full) of u and v using FFT."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Formula: 2·conv(w, a_rev) restricted to n entries.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact evaluation score (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_candidate(a0: np.ndarray,
                      deadline: float,
                      n: int,
                      size_fft: int,
                      target_sum: float,
                      max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Soft‑max gradient descent with Adam momentum.
    Returns the best sequence seen and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _score(a, n, size_fft)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # learning‑rate schedule (based on target sum)
    lr0 = target_sum * 0.8 / n
    decay = 1e-5

    # initialise temperature for soft‑max
    conv = _self_convolution(a, size_fft)
    T = max(1e-5, (conv.max() - conv.min()) * 0.5)

    it = 0
    no_improve = 0
    while time.time() < deadline and it < max_iters:
        # current convolution and exact score
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[refine] it {it:,} new best = {best_score:.6f}")
        else:
            no_improve += 1

        # ----- soft‑max weights -----
        shifted = conv - conv.max()
        exp_arg = shifted / max(T, 1e-9)
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smooth objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        lr = lr0 / (1.0 + decay * it)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random kick -----
        if (it % 7000 == 0 and it > 0) or no_improve > 6000:
            noise = np.random.randn(n) * target_sum * 5e-4
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)
            no_improve = 0
            print(f"[refine] it {it:,} random perturbation")

        a = a_new
        T *= 0.9999          # slow cooling
        it += 1

    return best_a, best_score

def _anneal_candidate(a0: np.ndarray,
                     deadline: float,
                     n: int,
                     size_fft: int,
                     target_sum: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing neighbourhood search.
    Uses a Gaussian perturbation, projects onto the simplex,
    and accepts with Metropolis probability.
    """
    cur = a0.copy()
    cur_score = _score(cur, n, size_fft)
    best = cur.copy()
    best_score = cur_score

    T = 1.0
    step_scale = target_sum * 0.004
    it = 0

    while time.time() < deadline:
        noise = np.random.randn(n) * step_scale
        cand = cur + noise
        cand = _project_to_simplex(cand, target_sum)
        cand_score = _score(cand, n, size_fft)

        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[anneal] it {it:,} new best = {best_score:.6f}")

        # cooling schedules
        T *= 0.99995
        step_scale *= 0.99995
        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation score.  The function respects a 1000 s budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # the normalisation used in the literature
    size_fft = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    def quick_score(a: np.ndarray) -> float:
        """Fast exact score (no security checks)."""
        conv = _self_convolution(a, size_fft)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    # Keep top‑K raw families
    K = 8
    best_raw = []   # list of (score, array)

    def register(score: float, a: np.ndarray) -> None:
        """Maintain the ascending top‑K list."""
        if len(best_raw) < K:
            best_raw.append((score, a.copy()))
            best_raw.sort(key=lambda x: x[0])
        else:
            if score < best_raw[-1][0]:
                best_raw[-1] = (score, a.copy())
                best_raw.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 1. PARAMETRIC SHAPE SEARCH
    idx = np.arange(n)

    # Uniform baseline
    a = np.ones(n)
    a = _project_to_simplex(a, target_sum)
    sc = quick_score(a)
    register(sc, a)
    print(f"[init] uniform score = {sc:.6f}")

    # Inverse‑sqrt (single‑ended)
    for _ in range(100):
        beta = np.random.rand() * 12.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        a = _project_to_simplex(a, target_sum)
        register(quick_score(a), a)

    # Double‑ended inverse‑sqrt
    for _ in range(100):
        beta = np.random.rand() * 12.0
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        a = _project_to_simplex(a, target_sum)
        register(quick_score(a), a)

    # Power‑law
    for _ in range(100):
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 1.7 + 0.3
        a = 1.0 / (idx + beta + 1.0) ** pexp
        a = _project_to_simplex(a, target_sum)
        register(quick_score(a), a)

    # Exponential decay
    for _ in range(100):
        decay = np.random.rand() * 0.012 + 0.0003
        a = np.exp(-decay * idx)
        a = _project_to_simplex(a, target_sum)
        register(quick_score(a), a)

    # Cosine‑based (large at ends)
    for _ in range(80):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        a = _project_to_simplex(a, target_sum)
        register(quick_score(a), a)

    # Sine‑based (zero at ends)
    for _ in range(80):
        p = np.random.rand() * 5.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** p
        a = _project_to_simplex(a, target_sum)
        register(quick_score(a), a)

    # Gaussian bump (single)
    for _ in range(80):
        mu = np.random.rand() * n
        sigma = np.random.rand() * n * 0.15 + n * 0.02
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        a = _project_to_simplex(a, target_sum)
        register(quick_score(a), a)

    # Double Gaussian bumps (mirrored)
    for _ in range(80):
        mu1 = np.random.rand() * n * 0.4
        mu2 = n - 1 - mu1
        sigma = np.random.rand() * n * 0.08 + n * 0.02
        a = np.exp(-0.5 * ((idx - mu1) / sigma) ** 2) \
            + np.exp(-0.5 * ((idx - mu2) / sigma) ** 2)
        a = _project_to_simplex(a, target_sum)
        register(quick_score(a), a)

    # Random block patterns
    for _ in range(300):
        nb = np.random.randint(2, 9)                # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        register(quick_score(cand), cand)

    # Report best raw candidate
    best_raw.sort(key=lambda x: x[0])
    raw_best_score, raw_best_a = best_raw[0]
    print(f"[shape search] best raw score = {raw_best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. INTENSIVE REFINEMENT (gradient descent on best raw candidate)
    remaining = deadline - time.time()
    if remaining <= 0:
        # time exhausted – return the raw best
        return raw_best_a.tolist()

    # allocate ~80 % of remaining time to refinement
    refine_deadline = time.time() + remaining * 0.80
    max_iters = int(min(3_000_000, (refine_deadline - time.time()) * 1500))
    refined_a, refined_score = _refine_candidate(
        raw_best_a, refine_deadline, n, size_fft, target_sum, max_iters
    )
    print(f"[refine] finished with score = {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3. SIMULATED‑ANNEALING POLISH
    remaining = deadline - time.time()
    if remaining > 0.5:
        anneal_deadline = time.time() + remaining * 0.90
        polished_a, polished_score = _anneal_candidate(
            refined_a, anneal_deadline, n, size_fft, target_sum
        )
        print(f"[anneal] finished with score = {polished_score:.6f}")
    else:
        polished_a, polished_score = refined_a, refined_score

    # ------------------------------------------------------------------
    # 4. FINAL RANDOM BLOCK SWEEP (quick)
    best_a = polished_a
    best_score = polished_score
    while time.time() < deadline:
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[final block] improved to {best_score:.6f}")
        # break early if time is almost out
        if time.time() + 0.03 > deadline:
            break

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
