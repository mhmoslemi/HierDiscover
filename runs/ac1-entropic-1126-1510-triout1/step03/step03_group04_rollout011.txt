Algorithm
---------
1. **Initialisation**

     otherwise from a random point on the simplex
     {a ≥ 0, Σ a = √(2 n)}.

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return a * a (full linear convolution) using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _exact_subgradient(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    Parameters
    ----------
    a : np.ndarray
        Current vector (length n).
    conv : np.ndarray
        Full convolution a*a (length 2n‑1).
    eps : float
        Numerical tolerance for determining active set.
    Returns
    -------
    grad : np.ndarray
        Gradient of shape (n,).
    """
    n = a.shape[0]
    max_val = conv.max()
    # active mask (indices where the maximum is attained)
    w = (conv >= max_val - eps).astype(float)

    # convolution w * a_rev (FFT version)
    size = 1 << ((len(w) + n - 1).bit_length())
    fw = np.fft.rfft(w, size)
    fa_rev = np.fft.rfft(a[::-1], size)
    conv_wr = np.fft.irfft(fw * fa_rev, size)
    start = n - 1
    sum_term = conv_wr[start : start + n]

    grad = 2.0 * sum_term

    # diagonal correction (i == j terms appear only once)
    double_idx = 2 * np.arange(n)
    mask_diag = double_idx < conv.shape[0]
    diag_w = w[double_idx[mask_diag]]
    grad[mask_diag] -= diag_w * a[mask_diag]

    return grad

        print("[init] using reference sequence")
    else:
        a = _random_initial_point(n, target_sum, rng)
        print("[init] random start")

    best_a = a.copy()
    conv = _full_convolution(a)
    best_score = conv.max()
    print(f"[init] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    # initial learning rate – a few percent of the typical simplex step
    lr0 = target_sum * 0.5 / n
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Restart schedule (up to six restarts)
    max_restarts = 6
    restart_interval = (deadline - time.time()) / (max_restarts + 1)
    next_restart = time.time() + restart_interval
    restarts_done = 0
    it = 1

    # ------------------------------------------------------------------
    # Phase 1 – exact sub‑gradient descent with Adam
    while time.time() < deadline - 30.0:      # reserve ~30 s for later phases
        conv = _full_convolution(a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            print(f"[{it:08d}] new best = {best_score:.6f}")

        # Compute exact sub‑gradient of the maximum convolution entry.
        grad = _exact_subgradient(a, conv)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # Bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Parameter update (descending)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = a - step

        # Projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

        # Occasional random shake to avoid stagnation
        if it % 4000 == 0:
            a += rng.normal(scale=target_sum * 2e-4, size=n)
            a = _project_to_simplex(a, target_sum)

        # Restart handling
        if time.time() > next_restart and restarts_done < max_restarts:
            restarts_done += 1
            print(f"[{it:08d}] restart #{restarts_done}")
            a = _random_initial_point(n, target_sum, rng)
            m.fill(0.0)
            v.fill(0.0)
            next_restart = time.time() + restart_interval

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – cheap block search (piecewise constant patterns)
    block_trials = 6000
    for _ in range(block_trials):
        nb = rng.integers(2, 13)                     # 2‑12 blocks
        cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = rng.random(nb)
        cand = np.zeros(n, dtype=float)
        for blk, vblk in zip(blocks, vals):
            cand[blk] = vblk
        cand = _project_to_simplex(cand, target_sum)
        sc = _full_convolution(cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[block] improved block score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final tiny hill‑climb (swap mass between two positions)
    while time.time() < deadline:
        nz = np.flatnonzero(best_a > 1e-12)
        if nz.size == 0:
            break
        i = rng.choice(nz)
        j = rng.integers(0, n)
        if i == j:
            continue
        # Move a random fraction of a[i] to position j
        frac = rng.uniform(0.1, 0.5)
        delta = best_a[i] * frac
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_to_simplex(cand, target_sum)
        sc = _full_convolution(cand).max()
        if sc < best_score:
            best_score = sc
            best_a = cand
            print(f"[swap] new best = {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
