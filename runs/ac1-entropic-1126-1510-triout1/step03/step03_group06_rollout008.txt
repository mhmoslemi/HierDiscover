Implement function _adam_optimize(n, a0, deadline, max_iters=8000, λ_max=80, init_lr=0.2).

Pseudo:

a = a0.astype(np.float64).copy(); a /= a.sum()
m_vec = np.zeros_like(a)
v_vec = np.zeros_like(a)

best_seq = a.copy(); best_score = _score(a)

for it in range(1, max_iters+1):
    if time.time() >= deadline: break

    # compute conv and gradient
    # conv
    b = conv_fft(a, mfft)
    cur_score = 2.0 * n * b.max()
    if cur_score < best_score:
        best_score = cur_score
        best_seq = a.copy()
        print(f"{time.time():.1f} n={n} it={it} score={best_score:.6f}")

    # compute lambda schedule
    lamb = λ_max * it / max_iters  # linear
    maxb = b.max()
    # compute softmax
    exp_arg = lamb * (b - maxb)  # <=0
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(b, 1.0 / b.shape[0])

    # gradient via correlation
    g = gradient_corr(w, a, mfft)

    # Adam update
    m_vec = 0.9 * m_vec + 0.1 * g
    v_vec = 0.999 * v_vec + 0.001 * (g * g)
    m_hat = m_vec / (1 - 0.9**it)
    v_hat = v_vec / (1 - 0.999**it)
    lr = init_lr / (np.sqrt(v_hat) + 1e-8)
    a = a - lr * m_hat

    # clip negative
    a = np.clip(a, 0.0, None)

    # renorm onto simplex
    sum_a = a.sum()
    if sum_a > 0:
        a = a / sum_a
    else:
        a = np.full(n, 1.0 / n)

    # occasional random jitter

"""
Hybrid Adam / sub‑gradient / hill‑climb search for the minimal
evaluate_sequence score.

The algorithm works on the equivalent objective

    F(a) = 2·n·max_k (a*a)_k ,    Σ a_i = 1 , a_i ≥ 0

where (a*a) is the linear convolution of the step‑function with itself.
A smooth approximation of the max is the soft‑max
    f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) .
Its gradient is 2·correlation(w, a) with soft‑max weights w.
We minimise f_λ by Adam (adaptive first‑order) while λ grows from 0 to a
large value (≈80).  After this phase a projected sub‑gradient descent
fine‑tunes the exact max‑convolution peak, and a cheap stochastic
hill‑climb makes final local improvements.

Multiple starting distributions (arcsine, Beta, Dirichlet, spikes,
uniform, random) are tried for several lengths (4000‑12000).  The best
sequence found before the time budget (default 1000 s) is returned as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float = None) -> np.ndarray:
    """Symmetric Beta(α,α) (or Beta(α,β) if β given) on a regular grid."""
    if beta is None:
        beta = alpha
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = np.finfo(np.float64).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (α = ½) on n points, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _conv_full_fft(a: np.ndarray, mfft: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length mfft (power‑of‑2)."""
    A = np.fft.rfft(a, mfft)
    C = np.fft.irfft(A * A, mfft)
    n = len(a)
    return C[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_correlation(w: np.ndarray, a: np.ndarray, mfft: int) -> np.ndarray:
    """Gradient = 2·correlation(w, a) using FFT of length mfft."""
    W = np.fft.rfft(w, mfft)
    A = np.fft.rfft(a, mfft)
    corr = np.fft.irfft(np.conj(W) * A, mfft)
    start = len(a) - 1
    return 2.0 * corr[start:start + len(a)]


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (Σa=1)."""
    n = len(a)
    return float(2.0 * n * _conv_full_fft(a, 1 << ((2 * n - 1).bit_length())).max())


# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    """Diverse pool of starting vectors for a given length."""
    cand = []

    # arcsine
    cand.append(_arcsine_distribution(n))

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric betas on a grid of α
    for alpha in (0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85):
        cand.append(_beta_distribution(n, alpha))

    # random symmetric betas
    for _ in range(3):
        a = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, a))

    # asymmetric betas (α ≠ β)
    for _ in range(2):
        a = np.random.uniform(0.1, 2.0)
        b = np.random.uniform(0.1, 2.0)
        cand.append(_beta_distribution(n, a, b))

    # sparse spike vectors
    k = max(2, n // 400)               # a few spikes (≈0.25 %)
    idx = np.random.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = np.random.rand(k) + 0.1
    cand.append(a / a.sum())

    # low‑concentration Dirichlet
    cand.append(np.random.dirichlet(np.full(n, 0.25)))

    # completely random (Gaussian then rectify)
    a = np.abs(np.random.randn(n))
    cand.append(a / a.sum())

    return cand


# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
    lam_max: float = 80.0,
    lr0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the soft‑max approximation of the max‑convolution.
    Returns the best sequence (Σ=1) seen and its exact score.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)

    best_seq = a.copy()
    best_score = _score(a)

    # FFT length sufficient for both convolution and correlation
    mfft = 1 << ((3 * n - 2).bit_length())

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exact score (used only for bookkeeping)
        conv = _conv_full_fft(a, mfft)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} it={it:5d} score={best_score:.6f}")

        # soft‑max temperature (linear schedule)
        lam = lam_max * it / max_iters

        maxc = conv.max()
        exp_arg = lam * (conv - maxc)               # ≤0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.shape[0])

        # gradient of smoothed objective
        g = _gradient_correlation(w, a, mfft)

        # Adam update
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * (g * g)

        m_hat = m_vec / (1 - beta1 ** it)
        v_hat = v_vec / (1 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # enforce feasibility
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)

        # occasional random jitter to escape tiny plateaus
        if it % 1000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(
    n: int,
    init_seq: np.ndarray,
    deadline: float,
    max_iters: int = 2000,
    eta0: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution peak.
    """
    a = init_seq.astype(np.float64).copy()
    a /= a.sum()
    best_seq = a.copy()
    best_score = _score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_full_fft(a, 1 << ((2 * len(a) - 1).bit_length()))
        k_star = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k*−i} where index inside array
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        i_range = np.arange(low, high + 1, dtype=int)
        g[i_range] = a[k_star - i_range]
        g *= 2.0

        eta = eta0 / np.sqrt(it)
        a = _project_onto_simplex(a - eta * g, s=1.0)

        cur = _score(a)
        if cur < best_score:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n:5d} sub it={it:5d} score={best_score:.6f}")

        # jitter
        if it % 500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _hill_climb(
    a: np.ndarray,
    deadline: float,
    max_trials: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a tiny mass from one index
    to another and keep the change if the (unsmoothed) objective improves.
    """
    n = len(a)
    best_seq = a.copy()
    best_score = _score(best_seq)

    for t in range(max_trials):
        if time.time() >= deadline:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = np.random.rand() * min(best_seq[i] * 0.05, 0.01)

        trial = best_seq.copy()
        trial[i] -= delta
        trial[j] += delta

        cur = _score(trial)
        if cur < best_score:
            best_score = cur
            best_seq = trial
            print(f"[{time.time():.1f}] hill‑climb trial={t} score={best_score:.6f}")

    return best_seq, best_score
