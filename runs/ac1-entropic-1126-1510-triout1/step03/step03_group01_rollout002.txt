Hybrid gradient descent for minimizing the peak of the self‑convolution.

The objective 2·n·max(a*a) / (sum(a)**2) is invariant under scaling,
hence we fix sum(a)=1 and search on the probability simplex.
The algorithm combines three ideas that differ from the previous
submission:

* **Symmetry enforcement** – the optimal step function is expected to be
  symmetric (a_i = a_{n‑1‑i}).  After each update we mirror the vector and
  average, which reduces the dimensionality of the search space and
  improves the flatness of the convolution.

* **Active‑set gradient** – instead of a slowly increasing soft‑max we
  switch to the exact sub‑gradient once the temperature becomes large.
  For a given convolution `b` we set the weight vector `w` uniform on the
  indices attaining (or almost attaining) the current maximum.
  The gradient of the (smoothed) peak is `grad = 2·rev(a) * w`,
  computed efficiently with FFTs.

  scheme (`velocity`) together with a step size `η = η0 / sqrt(iter)`
  provides fast descent while staying on the simplex via Euclidean
  projection.  Small random jitter is added occasionally to escape
  plateaus.

The search explores a few sequence lengths around the length of the
previous best (`height_sequence_1`), runs several random restarts per
length and keeps the globally best vector found within the supplied
time budget (default 1000 s).  The returned value is a plain Python list
compatible with `evaluate_sequence`.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast path – already feasible
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Return a * a via real FFT (full length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT (real output)."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _make_symmetric(a: np.ndarray) -> np.ndarray:
    """Force exact symmetry a[i] = a[-1-i] (average with its reverse)."""
    return (a + a[::-1]) * 0.5

    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The implementation follows the hybrid algorithm
    described in the module docstring.
    """
    # ------------------------------------------------------------------
    # Randomness
    np.random.seed(seed)

            for f in (0.7, 0.85, 1.0, 1.15, 1.3)
            if int(base_n * f) >= 500
        }
    )
    # Ensure we have at least one length.
    if not lengths:
        lengths = [8000]

    # ------------------------------------------------------------------
    # Global best tracking
    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        # Allow ~30 % of remaining time for this length.
        slice_deadline = time.time() + max(0.1, (deadline - time.time()) * 0.3)

        # ------------------------------------------------------------------
        # Several restarts for this length.
        for restart_id, init_kind in enumerate(
            ("prev", "arcsine", "uniform", "random")
        ):
            # ------------------------------------------------------------------
            # Initialise a on the simplex.
            if init_kind == "prev":
                try:
                    a = np.asarray(height_sequence_1, dtype=np.float64)
                    if a.size != n:
                        raise ValueError
                except Exception:
                    a = _arcsine_distribution(n)
            elif init_kind == "arcsine":
                a = _arcsine_distribution(n)
            elif init_kind == "uniform":
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:  # "random"
                a = np.random.rand(n)
                a /= a.sum()

            # Guarantee exact feasibility.
            a = _project_onto_simplex(a, s=1.0)
            # Enforce symmetry immediately – the search space is symmetric.
            a = _make_symmetric(a)

            # ------------------------------------------------------------------
            # Local optimisation state.
            best_local_seq = a.copy()
            cur_b = _fft_self_convolve(a)
            best_local_max = cur_b.max()
            best_local_score = 2.0 * n * best_local_max

            # Momentum variables.
            velocity = np.zeros_like(a)
            beta = 0.9

            # Step‑size schedule.
            eta0 = 0.8

            # Soft‑max temperature.
            alpha = 1.0
            alpha_factor = 1.04
            alpha_switch = 200.0  # after this we use the exact active set.
            alpha_max = 5000.0

            # Convergence monitoring.
            iter_no_improve = 0
            max_iter_without_improve = 3000

            it = 0

            # ------------------------------------------------------------------
            while time.time() < slice_deadline:
                it += 1

                # 1) Self‑convolution and current max.
                b = _fft_self_convolve(a)
                cur_max = b.max()
                cur_score = 2.0 * n * cur_max

                # Record improvement.
                if cur_max < best_local_max - 1e-14:
                    best_local_max = cur_max
                    best_local_seq = a.copy()
                    best_local_score = cur_score
                    iter_no_improve = 0
                    print(
                        f"[{time.time():.1f}] len={n} restart={restart_id} "
                        f"init={init_kind} iter={it} maxB={best_local_max:.10f} "
                        f"score≈{best_local_score:.6f}"
                    )
                else:
                    iter_no_improve += 1

                # 2) Determine weight vector w.
                if alpha < alpha_switch:
                    # Smoothed soft‑max.
                    exp_term = np.exp(alpha * (b - cur_max))
                    w = exp_term / exp_term.sum()
                else:
                    # Exact active‑set (all indices within a tiny tolerance).
                    tol = max(1e-15, cur_max * 1e-12)
                    active = np.where(b >= cur_max - tol)[0]
                    if active.size == 0:
                        # fallback – should not happen
                        w = np.full_like(b, 1.0 / b.size)
                    else:
                        w = np.zeros_like(b)
                        w[active] = 1.0 / active.size

                # 3) Gradient of the (smoothed) peak.
                a_rev = a[::-1]
                conv_rev = _fft_convolve(a_rev, w)      # length = 3n‑2
                grad = 2.0 * conv_rev[n - 1 : n - 1 + n]  # slice of length n

                # 4) Momentum update.
                velocity = beta * velocity + (1.0 - beta) * grad

                # 5) Adaptive step size.
                eta = eta0 / math.sqrt(it)

                # 6) Gradient step + projection onto simplex.
                a = a - eta * velocity
                # Enforce symmetry after the step.
                a = _make_symmetric(a)
                # Project back onto the probability simplex.
                a = _project_onto_simplex(a, s=1.0)

                # 7) Occasional jitter to escape plateaus.
                if it % 800 == 0 and np.random.rand() < 0.06:
                    a += np.random.randn(n) * 1e-5
                    a = _make_symmetric(a)
                    a = _project_onto_simplex(a, s=1.0)

                # 8) Anneal / increase temperature.
                if alpha < alpha_max:
                    alpha = min(alpha_max, alpha * alpha_factor)

                # 9) Early termination of this restart if stagnating.
                if iter_no_improve >= max_iter_without_improve:
                    # Re‑inject a small random perturbation and continue.
                    a += np.random.randn(n) * 1e-4
                    a = _make_symmetric(a)
                    a = _project_onto_simplex(a, s=1.0)
                    iter_no_improve = 0

                # ------------------------------------------------------------------
                # End of iteration.
            
            # ------------------------------------------------------------------
            # Evaluate the best exact candidate from this restart.
            try:
                final_score = evaluate_sequence(best_local_seq.tolist())
            except Exception:
                # Should not happen; fall back to our own estimate.
                final_score = best_local_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_seq.copy()
                print(
                    f"*** New global best ***** len={n} restart={restart_id} "
                    f"init={init_kind} score={overall_best_score:.6f}"
                )

            # Early exit if we already beat the target.
            if overall_best_score <= 1.503:
                break

        if time.time() >= deadline:
            break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded (unlikely) return a uniform vector.
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    # Return plain Python list (required by the evaluation harness).
    return overall_best_seq.tolist()
