def solve_lp(a, M, S):
    n = len(a)
    # Build constraint matrix A_ub (2n-1) constraints: conv <= M
    # rows 0..2n-2
    # data per row: a_i for each pair (i,j) with i+j=row
    i = np.repeat(np.arange(n), n)
    j = np.tile(np.arange(n), n)
    rows = i + j
    cols = j
    data = a[i]
    # Use csr matrix
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # bounds for variables: (0, None)
    bounds = [(0, None)] * n
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    # right hand side: M for each row
    b = np.full(2*n-1, M)
    # solve: linprog
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero vector
        return np.zeros_like(a)
    g = res.x
    # must scale to sum = S (if sum(g)==0, keep zeros)
    sum_g = g.sum()
    if sum_g > 1e-12:
        g = g * (S / sum_g)
    else:
        g = np.zeros_like(g)
    return g
```

def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 3000
    S = math.sqrt(2.0 * n)
    best_score = float('inf')
    best_a = None

    # precompute rows, cols
    i_vals = np.repeat(np.arange(n, dtype=np.int32), n)
    j_vals = np.tile(np.arange(n, dtype=np.int32), n)
    rows = i_vals + j_vals
    cols = j_vals

    # set up initial candidates
    init_candidates = []
    # uniform
    init_candidates.append(np.full(n, S / n))
    # V-shaped
    idx = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    vshape = np.abs(idx - mid) + 1e-6
    vshape = vshape / vshape.sum() * S
    init_candidates.append(vshape)
    # random
    rnd = np.random.rand(n)
    rnd = rnd / rnd.sum() * S
    init_candidates.append(rnd)
    # reference if available
    try:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        init_candidates.append(a_ref)
    except Exception:
        pass
    # iterate over candidates
    for idx_cand, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break
        a = a0.copy()
        # project to simplex to guarantee sum
        a = _project_to_simplex(a, S)
        # start mixing loop
        t = 0.5   # mixing weight
        it = 0
        while time.time() < deadline and it < 20:
            conv = np.convolve(a, a)
            M = conv.max()
            # solve LP:
            g = solve_lp_given_a(a, M, n, rows, cols)
            # scale g to norm S
            sum_g = g.sum()
            if sum_g > 1e-12:
                g = g * (S / sum_g)
            else:
                g = np.zeros_like(g)
            # mix
            a = (1 - t) * a + t * g
            # project onto simplex
            a = _project_to_simplex(a, S)
            # evaluate
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[cand {idx_cand} iter {it}] new best = {best_score:.6f}")
            # adapt t maybe gradually
            # reduce t over iterations
            t = max(0.05, t * 0.9)
            it += 1
        # after mixing, optionally refined by gradient descent
    # final polishing using gradient descent as previously
    if best_a is None:
        # fallback uniform
        best_a = np.full(n, S / n)
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform used")
    # gradient polishing
    a = best_a.copy()
    it = 0
    eta0 = S * 0.5 / n
    decay = 2e-4
    while time.time() < deadline and it < 2000:
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        conv_wr = np.convolve(w, a[::-1])
        g_a = 2.0 * conv_wr[n-1:n-1+n]
        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

            best_a = a_new.copy()

        a = a_new
        it += 1

"""
Propose a low‑score step‑function using a linear‑programming mixing scheme.

The algorithm works in three stages:

1. **Initialisation** – build a few reasonable starting vectors on the
   simplex (uniform, V‑shaped, random and the supplied reference).

2. **LP‑mixing iterations** – given the current vector `a` we compute its
   maximum autocorrelation `M = max(a*a)`.  We then solve a linear program
   (via ``scipy.optimize.linprog``) that maximises the ℓ¹‑norm of a
   non‑negative vector `g` under the linear constraints
   ``conv(a, g) ≤ M``.  After scaling `g` to the required ℓ¹‑norm we
   mix it with `a` and project back onto the simplex.
   Repeating this step strictly non‑increases the objective
   ``2·n·max(conv)/sum(a)²`` and quickly drives the sequence towards a
   better configuration.

3. **Final polishing** – a few sub‑gradient steps (projected gradient
   descent) fine‑tune the result.  The best sequence encountered is
   returned.

The whole search respects the 1000 s time budget, prints progress and
stores the best sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _solve_lp_for_g(a: np.ndarray, M: float,
                    rows: np.ndarray, cols: np.ndarray) -> np.ndarray:
    """
    Solve the linear program:

        maximise   sum(g)
        subject to conv(a, g) ≤ M   (element‑wise)
                    g ≥ 0

    The constraint matrix is built from the pre‑computed ``rows`` and
    ``cols`` arrays (the convolution pattern).  ``M`` is a scalar upper
    bound for all convolution entries.
    """
    n = a.shape[0]
    # data for the sparse constraint matrix: each entry uses the
    # coefficient a[i] where i is the row‑index of the convolution pair.
    data = np.repeat(a, n)          # shape (n*n,)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Linear program: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    # SciPy's HiGHS solver handles large sparse LPs efficiently.
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success:
        # In the unlikely event of failure return the zero vector.
        return np.zeros_like(a)
    return res.x
