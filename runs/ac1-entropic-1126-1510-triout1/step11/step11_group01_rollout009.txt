    full_len = len(w_rev) + len(a) - 1

    idx = (2 * len(a) - 2) - np.arange(len(a))
    grad = 2.0 * conv_rev[idx]
    return grad

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm Overview
------------------
1. **Generate a diverse pool of seeds** (uniform, arcsine, symmetric Beta,
   edge‑weighted, random Dirichlet and sparse vectors) for several
   lengths (≈5k–20k).  Keep the ~8 best shapes (by the exact objective).

2. **Smooth‑max mirror descent**.
   The true objective 2·n·max(conv) is non‑smooth.
   We replace the max by a temperature‑controlled soft‑max:
        Φβ(a)= (1/β)·log∑_k exp(β·(a*a)[k]).
   Its gradient can be obtained analytically via a second convolution.
   Starting with a low temperature we increase β gradually (continuation
   method) while performing exponentiated‑gradient updates and re‑projecting
   onto the probability simplex.  Each seed is refined for as many
   iterations as the remaining time allows.

3. **Multiscale refinement** – once a good shape for a moderate length has
   been found, it is linearly interpolated to a larger length (≈80 k) and
   the smooth‑max descent is run again.  Larger n lowers the theoretical
   lower bound 2·n/(2n−1)≈1.

4. **Final greedy polishing** – random pairwise mass transfers that are
   kept only when they improve the exact objective.

The whole routine respects the 1000 s budget (a 2 s safety margin) and
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input is assumed non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1}.  p>0 (p<1 gives edge‑heavy shape)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to stay stable for extreme p
    logw = (p - 1.0) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Soft‑max based gradient -----------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max Φβ(a)= (1/β)·log∑_k exp(β·conv_k)
    w.r.t. a.  Returns a vector of length n.
    """
    n = a.shape[0]
    # soft‑max weights (stable)
    max_c = conv.max()
    exp_terms = np.exp(beta * (conv - max_c))
    w = exp_terms / exp_terms.sum()                     # length 2n‑1

    # convolution w_rev * a (full) via FFT
    w_rev = w[::-1]
    size = 1 << ((len(w_rev) + n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_rev = np.fft.irfft(A * W, n=size)
    full_len = len(w_rev) + n - 1                     # = 3n‑2
    conv_rev = conv_rev[:full_len]

    # gradient_i = 2 * Σ_j w_{i+j} a_j   =>   2 * conv_rev[2n‑2‑i]
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_rev[idx]
    return grad


def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    global_best_seq: np.ndarray,
    global_best_score: float,
    eta0: float = 0.5,
    beta0: float = 0.5,
    beta_max: float = 30.0,
    max_iter: int = 20000,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the soft‑max surrogate.
    Returns an improved sequence and its exact objective.
    """
    n = a.shape[0]
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a.sum()

    best_local = a.copy()
    best_local_score = _objective(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (linear increase)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / max_iter)

        # convolution and gradient
        conv = _fft_self_convolve(a)
        grad = _softmax_gradient(a, conv, beta)

        # exponentiated‑gradient update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # periodic exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_local_score - 1e-14:
                best_local_score = cur
                best_local = a.copy()
                print(
                    f"[{time.time():.1f}] smooth descent it={t} new local best {cur:.6f}"
                )
            # early exit if we are already at the theoretical lower bound
            lower = 2.0 * n / (2 * n - 1)  # = 1 + 1/(2n‑1)
            if best_local_score <= lower * 1.005:
                break

    # final check
    final_score = _objective(a)
    if final_score < best_local_score:
        best_local_score = final_score
        best_local = a.copy()

    # update global best if improved
    if best_local_score < global_best_score:
        return best_local, best_local_score
    else:
        return global_best_seq, global_best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random local moves that are kept only when they improve the exact objective."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.4   # up to 40 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    seed_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Build a diverse pool of analytic seeds (lengths 5k–20k)
    # ------------------------------------------------------------------
    lengths = [5000, 10000, 20000]   # moderate sizes – fast enough for many seeds
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametric families
        # symmetric Beta(p,p) – edge‑heavy when p<1
        for p in np.linspace(0.05, 1.5, 13):
            if time.time() >= deadline:
                break
            cand = _distribution_beta_sym(n, p)
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} p={p:.3g} score={sc:.6f}")

        # edge‑weighted power law
        for p in np.linspace(0.01, 1.5, 13):
            if time.time() >= deadline:
                break
            cand = _distribution_edge_weighted(n, p)
            sc = _objective(cand)
            seed_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} pw={p:.3g} score={sc:.6f}")

        # random Dirichlet
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            seed_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best random Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            seed_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

    # keep only the very best few seeds for intensive refinement
    seed_pool.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in seed_pool[:8]]

    # ------------------------------------------------------------------
    # 2.  Refine each seed with smooth‑max mirror descent
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_seeds:
        per_seed = remaining / max(1, len(top_seeds))
        print(f"[{time.time():.1f}] starting smooth‑max descent on {len(top_seeds)} seeds")
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            overall_best_seq, overall_best_score = _smooth_descent(
                seed_seq.copy(),
                sub_deadline,
                overall_best_seq,
                overall_best_score,
                eta0=0.6,
                beta0=0.4,
                beta_max=35.0,
                max_iter=15000,
                eval_interval=150,
            )
            print(
                f"[{time.time():.1f}] after descent seed {idx} global best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 3.  Multiscale up‑sampling (if time permits)
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline - 30.0:
        target_n = 80000
        if target_n > len(overall_best_seq):
            print(f"[{time.time():.1f}] up‑sampling to {target_n}")
            old_n = len(overall_best_seq)
            xs_old = np.linspace(0, 1, old_n, endpoint=False)
            xs_new = np.linspace(0, 1, target_n, endpoint=False)
            up_seq = np.interp(xs_new, xs_old, overall_best_seq)
            up_seq = np.maximum(up_seq, 0.0)
            up_seq /= up_seq.sum()
            overall_best_seq, overall_best_score = _smooth_descent(
                up_seq,
                deadline,
                overall_best_seq,
                overall_best_score,
                eta0=0.5,
                beta0=0.3,
                beta_max=40.0,
                max_iter=20000,
                eval_interval=200,
            )
            print(
                f"[{time.time():.1f}] after up‑sample refinement global best {overall_best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 4.  Final greedy polishing
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise‑swap polishing")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
        )

    # ------------------------------------------------------------------
    # 5.  Return result
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # fallback: uniform of length 5000
        overall_best_seq = _distribution_uniform(5000)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` handles any necessary normalisation, we simply return the list.
    return overall_best_seq.tolist()
