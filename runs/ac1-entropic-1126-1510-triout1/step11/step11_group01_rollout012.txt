def _gradient_max_conv(a, w):
    n = a.shape[0]
    size = 1 << ((n + (2*n-1) - 1).bit_length())
    # compute rfft of a and w to size
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w, n=size)
    conv_full = np.fft.irfft(A * W, n=size)
    # conv_full length = size; we need slice from n-1 to n-1+n
    # Since size >= n + (2n-1) -1 = 3n-2. Use size accordingly.
    # We'll take conv_full[n-1:n-1+n]
    start = n - 1
    grad = 2.0 * conv_full[start:start + n]
    return grad

def _subgrad_descent_exact(a, deadline, best_seq, best_score, lr0=0.8, eval_interval=100, max_iter=10**7):
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        M = conv.max()
        # compute gradient
        w = np.zeros(conv.shape[0], dtype=np.float64)
        # include indices with abs(conv-M) <= 1e-12
        w[(np.abs(conv - M) <= 1e-12)] = 1.0
        # compute gradient as described
        size_fft = 1 << ((len(a) + len(w) - 1).bit_length())
        A = np.fft.rfft(a, n=size_fft)
        W = np.fft.rfft(w, n=size_fft)
        conv_full = np.fft.irfft(A * W, n=size_fft)
        grad = 2.0 * conv_full[n-1:n-1 + n]

        eta = lr0 / math.sqrt(t)  # decaying step
        # exponential update
        a = a * np.exp(-eta * grad)
        # renorm onto simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence```.  

The algorithm proceeds in four stages:

1. **Rich family scan** – generate a variety of analytic windows
   (uniform, arcsine, Beta‑symmetric, edge‑weighted, Hann, Hamming,
   Chebyshev, Kaiser, triangular, random Dirichlet/sparse).  For each
   candidate compute the exact objective
   `2·n·max(conv(a,a))` (the denominator is 1 because the vectors are
   normalised).  Keep the ~30 best seeds.

2. **Exact sub‑gradient (mirror) descent** – for every top seed run a
   long mirror‑descent using the *full* sub‑gradient of the max‑convolution
   (the gradient is the convolution of the vector with the indicator of
   the indices attaining the maximum).  The update is exponentiated‑gradient
   with a decaying stepsize and a projection onto the simplex (simple
   normalisation).  The loop stops when the global time budget is exhausted.

3. **Greedy pairwise mass transfer** – a cheap local optimiser that moves
   a random fraction of the mass from one coordinate to another and keeps
   the move only if it improves the exact objective.

4. **Return the best vector** – as a plain list; ``evaluate_sequence`` will
   perform the required clamping, scaling etc.

The routine respects the 1000 s budget (a small safety margin) and is
written to run on up to two CPU cores using only NumPy and the optional
SciPy signal windows.
"""

import time
import math

# ----------------------------------------------------------------------
# Utility: Euclidean projection onto the probability simplex
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, fast for all lengths we use)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
# Exact objective for a vector that is already normalised (sum == 1)
def _objective(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv(a,a)) assuming Σa = 1 and a_i ≥ 0.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a normalised probability vector)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta density x^{p‑1}(1‑x)^{p‑1} on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·(n‑i))^{‑p} (indices are 0‑based, shifted by ½)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangle, height 2/n at the centre, zero at the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = np.minimum(i, n - i + 1)
    w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    w = np.hanning(n).astype(np.float64)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_hamming(n: int) -> np.ndarray:
    """Hamming window."""
    w = np.hamming(n).astype(np.float64)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, alpha))
    a /= a.sum()
    return a


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mix λ·uniform + (1‑λ)·base, already normalised."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Exact sub‑gradient for max‑convolution (full set of active indices)
def _max_conv_and_indicator(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Compute max(conv(a,a)) and a binary indicator vector of length 2·n‑1
    marking the indices where the maximum is attained.
    """
    conv = _fft_self_convolve(a)
    M = conv.max()
    # use an absolute tolerance to catch ties caused by floating‑point noise
    tol = 1e-12
    ind = (np.abs(conv - M) <= tol).astype(np.float64)
    return float(M), ind


def _gradient_of_max(a: np.ndarray, indicator: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of the max‑convolution at a.
    ``indicator`` is the binary mask from ``_max_conv_and_indicator``.
    Returns a vector of length n.
    """
    n = a.shape[0]
    # full convolution a * indicator  (length n + 2n‑1 – 1 = 3n‑2)
    # we use FFT for speed.
    size = 1 << ((n + len(indicator) - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(indicator, n=size)
    full = np.fft.irfft(A * B, n=size)
    # slice the central n entries (indices n‑1 … n‑1+n‑1)
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad

# Mirror‑descent on the exact max‑convolution objective
def _mirror_descent_exact(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    eval_interval: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using the *exact* sub‑gradient
    (all active indices).  The iterate is always projected onto the simplex
    by simple renormalisation.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1

        M, ind = _max_conv_and_indicator(a)
        grad = _gradient_of_max(a, ind)

        eta = lr0 / math.sqrt(t)      # classic decaying stepsize
        # avoid overflow/underflow in the exponent
        expo = -eta * grad
        expo = np.clip(expo, -30, 30)
        a = a * np.exp(expo)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = 2.0 * n * M
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] mirror it={t} new best {best_score:.6f}")

        # safety cap to avoid pathological infinite loops
        if t >= 500_000:
            break

    return best_seq, best_score

# Greedy pairwise mass transfer (local polishing)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_cand = np.where(a > 1e-14)[0]
        if src_cand.size == 0:
            break
        i = np.random.choice(src_cand)
        # pick a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35  # move up to 35 % of the source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score
