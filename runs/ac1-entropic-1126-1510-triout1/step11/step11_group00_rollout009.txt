"""
Propose a non‑negative step‑function (list of heights) that minimizes
`evaluate_sequence`.  

The algorithm consists of three phases

1. **Diverse seed generation** – Analytic families (uniform, arcsine,
   symmetric Beta, edge‑weighted, sine‑weighted, exponential tails) are sampled
   for a few lengths together with random Dirichlet / sparse vectors.
   The ~30 best seeds (according to the exact objective) are kept.

2. **Soft‑max mirror descent** – For each seed we run a projected
   exponentiated‑gradient method that minimizes a smooth surrogate of the
   max‑convolution:
       g(a)= (1/β)·log Σ exp(β·conv(a,a)).
   The gradient of *g* is the weighted sum of the individual convolution
   gradients, which can be computed in O(n log n) by FFTs.  β is increased
   gradually so that the optimisation finally targets the true max.
   An exponential‑moving‑average of the gradient provides a cheap momentum
   term.  The iterate is kept on the probability simplex (positivity &
   unit‑sum) by a simple renormalisation.

3. **Local refinement** – A deterministic pairwise‑mass‑transfer walk and a
   cheap simulated‑annealing style stochastic walk polish the best vector
   found.

The whole search respects the supplied time budget (default 1000 s) and
returns a plain Python list.  All numbers are non‑negative; the caller
(`evaluate_sequence`) will perform the final clipping and normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers -------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities -------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (real FFT, length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _grad_from_softmax_weights(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate w.r.t. a.
    grad_i = 2·Σ_k w_k·a_{k-i}  (indices out of range are ignored).
    Implemented via FFT as a convolution of w_rev and a.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # reversed weight sequence
    need = len(w_rev) + n - 1            # = 3·n‑2
    size = 1 << ((need - 1).bit_length())
    Wr = np.fft.rfft(w_rev, n=size)
    Ar = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(Wr * Ar, n=size)    # linear convolution
    start = len(w_rev) - 1                 # index of lag 0
    grad = 2.0 * conv[start:start + n]
    # Numerical safety
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return grad


# ----------------------------------------------------------------------
# Exact objective (unit‑sum normalisation) -------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(a,a)) for a unit‑sum vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p) on the grid i+0.5 / n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n-i-0.5))^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π·(i+½)/n)^p."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exp_tails(n: int, alpha: float) -> np.ndarray:
    """Exponential decay from the centre: w_i ∝ exp(-α·|i‑mid|)."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-alpha * np.abs(i - mid))
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max mirror descent -----------------------------------------------
def _softmax_mirror_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.55,
    beta0: float = 0.5,
    beta_max: float = 80.0,
    beta_mul: float = 1.001,
    eval_interval: int = 120,
    patience: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the smooth surrogate
    g(a)= (1/β)·log Σ exp(β·conv(a,a)).  β grows steadily until beta_max.
    The update uses an exponential‑moving‑average of the gradient as a simple
    momentum term.
    """
    n = a.shape[0]
    t = 0
    beta = beta0
    grad_ma = np.zeros_like(a)           # momentum (moving average)
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 300_000:
        t += 1

        # 1) self‑convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # 2) soft‑max weights (numerically stable)
        shifted = conv - max_c               # max_c is subtracted to avoid overflow
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # 3) gradient of the surrogate
        grad = _grad_from_softmax_weights(w, a)

        # 4) momentum (exponential moving average)
        grad_ma = 0.9 * grad_ma + 0.1 * grad

        # 5) exponentiated update (log‑space for stability)
        log_a = np.log(np.maximum(a, 1e-30))
        eta = lr0 / math.sqrt(t)
        log_a -= eta * grad_ma
        a = np.exp(log_a)

        # 6) project back onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # 7) increase β (slowly approach the true max)
        beta = min(beta_max, beta * beta_mul)

        # 8) occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] softmax it={t} new global best {best_score:.6f}")
            else:
                no_improve += 1

        # 9) jitter to escape stagnation
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass from i→j if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
