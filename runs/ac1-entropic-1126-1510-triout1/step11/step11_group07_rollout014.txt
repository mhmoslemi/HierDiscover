"""
Search for a non‑negative step function that minimises `evaluate_sequence`.
The algorithm consists of:
1) A broadened analytic‑family exploration over several lengths,
   keeping the globally best vector.
2) A long smooth‑max mirror‑descent phase with an aggressive
   temperature schedule (α up to 1e6) and many iterations.
3) A projected Adam‑style sub‑gradient phase.
4) A final fine‑grained mass‑transfer hill‑climber.
Multiple random restarts from the current best are performed during
mirror descent to alleviate local minima.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Assumes a is normalised to sum = 1. Returns the exact objective
    2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (all return probability vectors with sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangle(n: int) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    w = np.minimum(idx, n + 1 - idx)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """Deterministic edge‑biased power law:
       w_i ∝ ((i+½)(n‑i‑½))^{-α} + c."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """Random Dirichlet draw with edge‑biased concentration."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec

# Optimisation primitives
# ----------------------------------------------------------------------


def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float, eps: float = 1e-12):
    """One exponentiated‑gradient step on the smooth‑max proxy.
    Returns (new_a, max_conv)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of smooth‑max w.r.t. a
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """Projected Adam step using the true max‑convolution sub‑gradient."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """Tiny deterministic hill‑climber that moves mass from the
    most‑responsible pair to a minimal entry."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):

"""
Search for a non‑negative step function that minimises `evaluate_sequence`.

The algorithm consists of four stages:
1) **Broad analytic exploration** over several lengths and many
   parameterised families (Beta, edge‑biased power‐law, triangle,
   arcsine, mixtures, Dirichlet draws).  The globally best vector is kept.
2) **Long smooth‑max mirror descent** on that best vector.  The smooth‑max
   temperature α is increased geometrically up to 1e6 while the learning
   rate decays, and small random jitters are injected every few thousand
   steps to escape shallow minima.
3) **Projected Adam sub‑gradient descent** on the true max‑convolution.
4) **Fine‑grained mass‑transfer hill‑climber** that moves tiny probability
   mass from the pair responsible for the current peak towards the smallest
   entry.  The routine respects the 1000‑second budget and returns the best
   sequence found (as a plain list of floats).
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Assuming a is normalised to sum = 1, returns the exact value produced by
    `evaluate_sequence` (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangle(n: int) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    w = np.minimum(idx, n + 1 - idx)
    w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """Deterministic edge‑biased power law:
       w_i ∝ ((i+½)(n‑i‑½))^{-α} + c."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float,

                       step0: float, beta1: float, beta2: float,

    w[active] = 1.0
