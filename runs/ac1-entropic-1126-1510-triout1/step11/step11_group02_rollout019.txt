"""
Hybrid optimiser for the “max‑convolution’’ problem.

The search proceeds in three stages:

1. **Broad seed generation** – a large collection of analytically‑motivated
   distributions (uniform, arcsine, symmetric / asymmetric Beta, power‑law,
   exponential, two‑block spikes, Dirichlet, random monotone, etc.) is created
   for several lengths (up to 30 k).  Each seed is normalised to sum‑one and its
   exact objective 2·n·max(conv)/sum² is evaluated via an FFT‑based convolution.
   The best few seeds (global ranking) are retained.

2. **Refinement** – each retained seed undergoes a long smooth‑max descent
   (exponentiated‑gradient on the log‑sum‑exp surrogate) with a geometric
   temperature schedule (α from 1 to ≈200) followed by the classic multimax
   sub‑gradient descent.  Both phases keep the iterate on the probability simplex
   (Euclidean projection).  The number of iterations is chosen dynamically from
   the remaining time budget.

3. **Polishing** – a cheap greedy pairwise‑swap local search moves mass from a
   randomly chosen source index to a random destination if it reduces the
   objective.  This final phase runs until the global deadline.

The routine returns the best sequence (as a plain ``list[float]``) found
within the allotted time (default 1000 s).  It is completely self‑contained
and uses only ``numpy`` and the provided ``evaluate_sequence`` for final
validation.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes sum(a) > 0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# --- Analytic families -------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p).  p>0, p=0.5 gives the arcsine law."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_power_decreasing(n: int, p: float) -> np.ndarray:
    """a_i ∝ (n-i)^{p}, p>0 => decreasing shape."""
    i = np.arange(n, 0, -1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_exponential_decreasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_monotone(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random decreasing vector (sorted Gamma draws)."""
    vals = rng.gamma(1.0, size=n)
    vals = np.sort(vals)[::-1]   # decreasing
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _generate_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Return a rich list of seed vectors for a given length."""
    cand: List[np.ndarray] = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric Beta – dense grid (includes p=0.5 = arcsine)
    for p in np.linspace(0.05, 0.95, 19):
        cand.append(_distribution_beta_sym(n, p))

    # asymmetric Beta – random draws
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power law families
    for p in (0.3, 0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))
        cand.append(_distribution_power_decreasing(n, p))

    # exponential families
    for lam in (0.001, 0.003, 0.006, 0.012):
        cand.append(_distribution_exponential_increasing(n, lam))
        cand.append(_distribution_exponential_decreasing(n, lam))

    # two‑block spikes
    for block_frac in (0.04, 0.07, 0.12):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random Dirichlet draws
    for _ in range(6):
        alpha = rng.uniform(0.1, 1.0)
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # random monotone patterns
    for _ in range(4):
        cand.append(_distribution_random_monotone(n, rng))

    # random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.015))

    # completely random vectors (renormalised)
    for _ in range(5):
        v = rng.random(n)
        v /= v.sum()
        cand.append(v)

    return cand

def _smooth_max_descent(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        eta0: float = 0.7,
                        alpha0: float = 1.0,
                        alpha_max: float = 200.0,
                        max_iter: int = 30000,
                        eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth max:
        f_α(a) = (1/α) log Σ_i exp(α·conv_i(a)).
    α grows geometrically up to ``alpha_max``.
    The iterate is always re‑projected onto the probability simplex.
    """
    n = a.shape[0]
    t = 0
    alpha = alpha0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # keep on the simplex (project if needed)
        a = _project_onto_simplex(a, s=1.0)

        # increase temperature slowly
        if t % 400 == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.25, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.1f} new best {best_score:.6f}")

    return best_seq, best_score

def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.6,
                                 max_iter: int = 30000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classic multimax sub‑gradient descent: the gradient is the average of
    the active set (indices attaining the maximal convolution value).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """
    Simple greedy mass‑transfer walk: pick a source with positive mass,
    move a random fraction (≤30 %) to a random destination and keep the move
    only if it improves the objective.
    """
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # pick a distinct destination uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _smooth_max_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.9,
                alpha0=0.8,
                alpha_max=250.0,
                max_iter=50000,
                eval_interval=800,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smooth descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Second refinement: classic multimax sub‑gradient descent
    remaining = deadline - time.time()
    if remaining > 40:
        # always keep the current overall best plus a few others for diversity
        seeds = [overall_best_seq.copy()]
        seeds += [seq for _, seq in sorted(top_candidates,
                                            key=lambda x: x[0])[:3]
                  if not np.may_share_memory(seq, overall_best_seq)]

        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds))
            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=60000,
                eval_interval=800,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after multimax descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final polishing with cheap local swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # ensure plain Python list of floats (evaluate_sequence accepts that)
    return overall_best_seq.tolist()
