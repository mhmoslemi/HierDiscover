"""
SMOOTH‑MAX GRADIENT SEARCH FOR A NON‑NEGATIVE STEP‑FUNCTION
==========================================================

The evaluation metric is

    score = 2·n·max(conv(a,a))          (∑a = 1)

where ``conv`` is the ordinary linear convolution.
For a fixed length the uniform distribution yields ``score = 2``.
The best known construction so far reaches ≈1.51.

This implementation improves the previous pipeline by

* adding several new analytic families (exponential, logistic,
  SciPy windows such as Hann, Blackman, Welch …),
* using a *smoothed‑max* objective – the maximum is approximated by a
  soft‑max (log‑sum‑exp) with a temperature that is increased during the
  run,
* performing an exponentiated‑gradient (mirror‑descent on the simplex)
  that uses the full soft‑max gradient (instead of only the top‑k
  entries),
* a cheap FFT based implementation of the gradient, and
* a final greedy pairwise‑mass‑transfer polish.

The whole search respects a 1000 s budget (2 s safety margin).
The function ``propose_candidate`` is the entry point used by the
evaluation harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# 1.  Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project the vector `v` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  FFT based convolutions (real, fast)
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution x*y via a real FFT."""
    lx, ly = x.shape[0], y.shape[0]
    size = 1 << ((lx + ly - 2) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[: lx + ly - 1]


# ----------------------------------------------------------------------
# 3.  Objective (exact) – assumes Σa = 1
# ----------------------------------------------------------------------
def _objective_score(a: np.ndarray) -> float:
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = x ** (p - 1) * (1.0 - x) ** (p - 1)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i).  lam ≥ 0."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i)
    w /= w.sum()
    return w


def _distribution_logistic(n: int, scale: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i-mid)/scale)).  scale > 0."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / (1.0 + np.exp((i - mid) / scale))
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    """Sparse Dirichlet‑like seed."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base·(1−lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _distribution_window(name: str, n: int, **kw) -> np.ndarray:

# ----------------------------------------------------------------------
# 5.  Smoothed‑max (soft‑max) gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_max_descent(a0: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        lr0: float = 0.8,
                        beta_start: float = 0.2,
                        beta_end: float = 150.0,
                        max_iters: int = 250000,
                        eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using a soft‑max approximation of the L∞ norm.
    The update is:
        a ← a ⊙ exp(−η·∇),   then renormalise onto the simplex.
    Gradient is the exact gradient of the weighted sum of convolutions,
    where the weights are a soft‑max over the current convolution values.
    """
    a = a0.copy()
    n = a.shape[0]
    a_rev = a[::-1]                 # constant for the whole run (updated each step)
    t = 0
    # linear schedule for the soft‑max temperature
    beta_ramp = max(1, max_iters // 20)   # first 5 % of iterations ramp up
    while time.time() < deadline and t < max_iters:
        t += 1

        # --- convolution and current max ---
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_obj = 2.0 * n * cur_max
        if cur_obj < best_score - 1e-14:
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] new global best (soft‑max) {best_score:.6f}")

        # --- temperature (β) ---
        if t <= beta_ramp:
            beta = beta_start + (beta_end - beta_start) * (t / beta_ramp)
        else:
            beta = beta_end

        # --- soft‑max weights (normalized) ---
        # subtraction of cur_max prevents overflow
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # --- gradient of Σ w_i·conv_i  (exact) ---
        # g_j = 2·∑_i w_i·a_{i−j}
        # This is a (partial) correlation of w with a_rev.
        # Use FFT for speed.
        grad_full = _fft_convolve(w, a_rev)          # length 3n‑2
        start = n - 1
        grad = 2.0 * grad_full[start:start + n]

        # --- exponentiated‑gradient update ---
        eta = lr0 / math.sqrt(t)                     # decaying stepsize
        a = a * np.exp(-eta * grad)

        # --- project back onto the probability simplex ---
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # --- occasional exact evaluation for logging ---
        if t % eval_interval == 0:
            # (objective already computed above, we just keep the interval)
            pass

    return best_seq, best_score

# 6.  Greedy pairwise mass‑transfer polish (identical to the previous code)
# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep the move only if it improves the exact objective."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.4
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur_obj = _objective_score(new_a)
        if cur_obj < best_score - 1e-14:
            a = new_a
            best_score = cur_obj
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 7.  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.
    The routine mixes analytic families, a soft‑max mirror‑descent
    optimiser and a final greedy polishing, all respecting the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Scan analytic families (dense coverage)
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6000, 8000, 12000, 20000]
    p_grid = np.concatenate([
        np.logspace(-2, -0.5, 12),      # heavy edge
        np.logspace(-0.5, 0.0, 12),    # up to uniform
        np.logspace(0.0, 0.7, 10),     # mild centre bias
        np.logspace(0.7, 1.4, 8),      # peaked centre
    ])
    lam_grid = np.linspace(0.0, 1.0, 6)       # mix with uniform
    exp_grid = np.logspace(-3, 0, 6)         # λ for exponential
    log_grid = np.logspace(-2, -0.2, 6)      # scale for logistic (small → steep)

    # optional SciPy windows
    window_names = ["hann", "hamming", "blackman", "welch", "parzen"]
    window_params = {
        "parzen": {"sym": True},
    }

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ---- deterministic families ----
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective_score(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # ---- parameterised families ----
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective_score(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}")

        # ---- exponential and logistic families ----
        for lam_exp in exp_grid:
            cand = _distribution_exponential(n, lam_exp)
            sc = _objective_score(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best exp len={n} λ={lam_exp:.3g} score={sc:.6f}")

        for log_sc in log_grid:
            cand = _distribution_logistic(n, log_sc)
            sc = _objective_score(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best logistic len={n} scale={log_sc:.3g} score={sc:.6f}")

        # ---- SciPy windows (if available) ----
        if _HAS_SCI:
            for wname in window_names:
                if time.time() >= deadline:
                    break
                kw = window_params.get(wname, {})
                cand = _distribution_window(wname, n, **kw)
                sc = _objective_score(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new global best window {wname} len={n} score={sc:.6f}")

        # ---- random Dirichlet seeds ----
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective_score(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best random Dirichlet len={n} α={alpha:.3g} score={sc:.6f}")

        # ---- sparse random seeds ----
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective_score(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the ~30 best seeds seen so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Re‑use a previously discovered construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective_score(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (should never trigger)
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective_score(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smoothed‑max mirror‑descent on a few top seeds
    # ------------------------------------------------------------------
    # Only run the expensive descent on moderate lengths (≤8000)
    max_len_for_descent = 8000
    seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]
             if seq.shape[0] <= max_len_for_descent]

    remaining = deadline - time.time()
    if remaining > 60.0 and seeds:
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting soft‑max descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_max_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                beta_start=0.2,
                beta_end=200.0,
                max_iters=200000,
                eval_interval=250,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final greedy pairwise polishing (fast, works for any length)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise‑swap polishing")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200000,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Report and return
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective = 2·n·max(conv)) : {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
