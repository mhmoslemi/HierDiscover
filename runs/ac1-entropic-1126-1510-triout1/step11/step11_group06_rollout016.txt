"""
Propose a non‑negative step‑function (list of floats) that minimizes
`evaluate_sequence`.  The search proceeds in three phases:

1. **Broad exploration** – generate many candidate vectors for several
   lengths (2000 … 32000) using a handful of parametric families
   (uniform, arcsine, symmetric Beta, edge‑power, Dirichlet draws).  The
   best three candidates for each length (according to the exact
   objective 2·n·max(a*a)) are kept.

2. **Smooth‑max mirror descent with Adam** – each kept candidate is
   refined by minimising a soft‑max approximation of the max‑convolution.
   A temperature `alpha` is slowly increased while Adam–style momentum
   updates move the vector on the probability simplex.  After each step
   the vector is forced to be symmetric (a_i = a_{n‑1‑i}) and projected
   onto the simplex.

3. **Hard‑max sub‑gradient polishing + stochastic hill climbing** – the
   refined vector is further improved using genuine sub‑gradients of the
   max‑convolution, followed by cheap mass‑transfer moves that try to
   reduce the current peak.  Throughout the optimisation the best
   sequence (lowest score) is stored.

The algorithm respects the given time budget (`budget_s`), performs all
operations with NumPy and uses only FFT‑based convolutions (O(n log n))
so even the largest lengths finish comfortably within the limit.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x ≥ 0, sum x = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (full length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b (full length len(a)+len(b)-1) using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective for a vector already normalised to sum=1:
       2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Simple symmetric families (all return a vector with sum = 1) -------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    v = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    v /= v.sum()
    return v


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15
    w /= w.sum()
    return w


def _distribution_dirichlet(n: int, concentration: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, concentration))


# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force a_i = a_{n-1-i} and renormalise."""
    v = (v + v[::-1]) * 0.5
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return v

                step *= 0.5

            a = a_cand
            if cand_sc < cur_best - 1e-12:
                cur_best = cand_sc
                print(f"[{time.time():.2f}] subgrad it={sub_it} new best {cand_sc:.6f}")

            # occasional jitter to escape plateaus
            if sub_it % 1200 == 0 and sub_it > 0:
                jitter = np.random.randn(n) * 5e-6
                a = np.maximum(a + jitter, 0.0)
                a = _project_onto_simplex(a, 1.0)
                a = _symmetrize(a)

            sub_it += 1
        return a, cur_best

    # --------------------------------------------------------------
    # Helper: tiny stochastic hill‑climber (mass transfer)
    def _hill_climb(a: np.ndarray, n: int, deadline: float,
                    cur_best: float) -> tuple[np.ndarray, float]:
        """Mass‑transfer polishing – move a little probability mass from a
        strong‑product pair to the current smallest coordinate."""
        # pre‑compute convolution for speed
        conv = _fft_self_convolve(a)
        M = conv.max()
        hill_it = 0
        while time.time() < deadline:
            hill_it += 1
            # locate the index k where the max occurs
            k = int(conv.argmax())
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # find the pair (i,j) on this anti‑diagonal with largest product
            best_i, best_j = i_start, k - i_start
            best_prod = a[best_i] * a[best_j]
            for i in range(i_start, i_end + 1):
                j = k - i
                prod = a[i] * a[j]
                if prod > best_prod:
                    best_prod = prod
                    best_i, best_j = i, j

            # source = index holding more mass, target = smallest entry
            source = best_i if a[best_i] >= a[best_j] else best_j
            target = int(np.argmin(a))
            if target == source:
                # second smallest if identical
                target = int(np.argsort(a)[1])

            # propose a modest transfer
            max_delta = min(a[source] * 0.5, 0.03)
            delta = max_delta * random.random()
            if delta < eps:
                continue

            a2 = a.copy()
            a2[source] -= delta
            a2[target] += delta
            a2 = np.maximum(a2, eps)
            a2 /= a2.sum()
            a2 = _symmetrize(a2)

            new_M = _fft_self_convolve(a2).max()
            if new_M < M - 1e-12:
                a = a2
                conv = _fft_self_convolve(a)
                M = new_M
                cur_best = 2.0 * n * M
                best_seq = a.copy()
                if hill_it % 100 == 0:
                    print(f"[{time.time():.2f}] hill it={hill_it} new best {cur_best:.6f}")

            # periodic jitter
            if hill_it % 2500 == 0:
                jitter = np.random.randn(n) * 1e-5
                a = np.maximum(a + jitter, eps)
                a /= a.sum()
                a = _symmetrize(a)
                conv = _fft_self_convolve(a)
                M = conv.max()
        return a, cur_best

    # --------------------------------------------------------------
    # 3️⃣  Refine the best candidate found so far (can be a list of start points)
    # We'll keep at most a few starts to stay within the time budget.
    start_vectors = [best_seq]           # already the best from phase‑1
    # optionally add a few random Dirichlet start points (different shape)
    for _ in range(2):
        vec = np.random.dirichlet(np.full(best_n, 0.3))
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        vec = _symmetrize(vec)
        start_vectors.append(vec)

    for idx, init_a in enumerate(start_vectors):
        if time.time() >= deadline:
            break
        a = init_a.copy()
        n = a.shape[0]

        # Phase‑2a : smooth‑max Adam
        a, best_score = _adam_smoothmax(a, n, deadline, best_score)

        # Phase‑2b : hard‑max sub‑gradient descent
        a, best_score = _hard_subgrad(a, n, deadline, best_score)

        # Phase‑2c : stochastic hill‑climbing
        a, best_score = _hill_climb(a, n, deadline, best_score)

        best_seq = a.copy()
        best_n = n

    # --------------------------------------------------------------
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_seq.tolist()
