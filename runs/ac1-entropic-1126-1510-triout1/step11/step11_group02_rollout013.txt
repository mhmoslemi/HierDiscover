def _max_adam(a0, deadline, best_seq, best_score, lr=0.02, max_iter=50000, eval_interval=200):
    n = a0.shape[0]
    x = np.log(a0 + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    while t < max_iter and time.time() < deadline:
        t+=1
        a = np.exp(x - np.max(x))
        a = a / a.sum()
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt
        grad_a = _gradient_from_weights(a,w)
        # gradient for logits
        dot = np.dot(grad_a,a)

                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] max adam it={t} new best {best_score:.6f}")

"""
Hybrid random‑seed + Adam‐optimised search for the min‑max self‑convolution
problem.

The algorithm works in two main stages:

1. **Seeding** – a rich collection of analytic families (uniform, arcsine,
   beta, power‑law, exponential, two‑block, Dirichlet, random sparse etc.) is
   generated for a modest length (default 4000).  A few hundred random Dirichlet
   draws are added and the best few candidates are kept.

2. **Refinement** – each kept seed is refined by two Adam‑based mirror‑descent
   phases:
   * a smooth‑max phase (log‑sum‑exp surrogate) that gradually raises the
     temperature α, allowing stable gradients over all convolution entries;
   * a true‑max phase that uses the exact active‑set gradient.
   Both phases operate on the log‑probability vector (soft‑max parametrisation)
   to keep the iterate on the simplex without an explicit projection.
   Momentum (Adam) dramatically speeds up convergence compared to the simple
   1/√t step‑size used previously.

If time permits the best distribution is up‑sampled to a larger length
(e.g. 16000) by linear interpolation and refined again.  Finally a cheap
pairwise‑swap local search finishes the run.  The routine respects the 1000 s
budget and returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum(a) assumed >0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic families (used for seeding)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a broad variety of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.10, 0.20, 0.30, 0.40, 0.60, 0.80, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac))

    for _ in range(5):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    for _ in range(3):
        v = rng.random(n)
        v /= v.sum()
        cand.append(v)

    return cand


# ----------------------------------------------------------------------
# Adam based descent on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _smooth_max_adam(a0: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     lr: float = 0.02,
                     alpha0: float = 1.0,
                     alpha_max: float = 200.0,
                     max_iter: int = 60000,
                     eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑softmax parametrisation of a.
    Objective is the smooth max
        f_α(a) = (1/α) log Σ_i exp(α·conv_i(a)).
    α is increased geometrically up to ``alpha_max``.
    """
    n = a0.shape[0]
    # initialise logits (soft‑max parametrisation)
    x = np.log(a0 + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # recompute a from logits (numerically stable)
        a = np.exp(x - np.max(x))
        a /= a.sum()

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = _gradient_from_weights(a, w)

        # chain rule to logits (soft‑max Jacobian)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update of logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        if t % 200 == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.2, alpha_max)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur_score = 2.0 * n * conv.max()   # sum(a) is 1
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑adam it={t} α={alpha:.2f} "
                      f"new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam based descent on the true max (active‑set)
# ----------------------------------------------------------------------
def _max_adam(a0: np.ndarray,
              deadline: float,
              best_seq: np.ndarray,
              best_score: float,
              lr: float = 0.02,
              max_iter: int = 60000,
              eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser where the gradient is built from the active set of
    convolution entries attaining the maximal value.
    """
    n = a0.shape[0]
    x = np.log(a0 + 1e-16)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1

        a = np.exp(x - np.max(x))
        a /= a.sum()

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad_a = _gradient_from_weights(a, w)

        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % eval_interval == 0:
            cur_score = 2.0 * n * max_val
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] max‑adam it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple greedy pairwise swap for final polishing
# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk, accepting only improving moves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upscaling utility (interpolates a probability vector to a larger size)
# ----------------------------------------------------------------------
def _scale_distribution(a_small: np.ndarray, N: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length N."""
    n = a_small.shape[0]
    pos_small = (np.arange(n, dtype=np.float64) + 0.5) / n
    pos_big = (np.arange(N, dtype=np.float64) + 0.5) / N
    a_big = np.interp(pos_big, pos_small, a_small, left=0.0, right=0.0)
    a_big = np.maximum(a_big, 0.0)
    s = a_big.sum()
    if s == 0.0:
        a_big = np.full(N, 1.0 / N, dtype=np.float64)
    else:
        a_big /= s
    return a_big

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Runs for up to ``budget_s`` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose a moderate length for the fast exploratory phase
    n_mid = 4000
    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    candidates: List[Tuple[float, np.ndarray]] = []

    print(f"[{time.time():.1f}] Generating analytic seeds (n={n_mid})")
    for cand in _initial_candidates_extended(n_mid, rng):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        sc = _objective(cand)
        candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] **** new global best seed len={n_mid} score={sc:.6f}")

    # add a handful of random Dirichlet draws for diversity
    for _ in range(30):
        alpha = 0.5 + 0.5 * rng.random()      # α ∈ [0.5,1.0]
        cand = _distribution_dirichlet(n_mid, rng, alpha=alpha)
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective(cand)
        candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] **** new best (Dirichlet) score={sc:.6f}")

    # keep only the most promising seeds
    candidates.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in candidates[:6]]

    # ------------------------------------------------------------------
    # 2. First refinement: smooth‑max Adam on the moderate length
    remaining = deadline - time.time()
    if remaining > 30.0:

        for seed_seq in top_seeds:
            if time.time() >= deadline:

            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(top_seeds))

            )
            if best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after smooth‑Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Second refinement: true‑max Adam on the same length
    remaining = deadline - time.time()
    if remaining > 30.0:
        print(f"[{time.time():.1f}] Starting max‑Adam (n={n_mid})")
        for seed_seq in [overall_best_seq] + top_seeds[:2]:
            if time.time() >= deadline:

            sub_deadline = time.time() + (deadline - time.time()) / max(1, 3)

                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.02,
                max_iter=60000,
                eval_interval=500,

            max_iter=80000,
            eval_interval=500,

            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr=0.02,
            max_iter=80000,
            eval_interval=500,

        )

    return overall_best_seq.tolist()
