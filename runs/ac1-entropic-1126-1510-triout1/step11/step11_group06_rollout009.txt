"""
A robust, time‑bounded optimizer for the step‑function evaluation problem.

Algorithm
---------
1. **Coarse exploration** – test a handful of lengths (2 k … 32 k) and a rich set
   of parametric families (uniform, arcsine, symmetric/asymmetric Beta,
   deterministic edge‑power, mixtures with uniform).  The best shape (sum=1)
   is kept.

2. **Mirror‑descent refinement** – minimise a soft‑max (log‑sum‑exp) approximation
   of the maximal self‑convolution.  The update is exponentiated‑gradient
   (KL‑mirror descent) with a temperature α that grows from 1 to 1e6 and a
   learning‑rate that decays from 0.5 to 1e‑4.  The iterate is projected onto
   the simplex after each step.

3. **Simulated‑annealing hill‑climbing** – repeatedly move a small amount of
   probability mass between two coordinates.  Moves are accepted if they
   improve the true objective or with a Metropolis probability that
   decreases with a temperature schedule.  This helps escape shallow plateaus.

The whole procedure respects the supplied time budget (default 1000 s) and
always returns a valid list of non‑negative floats.  The internal score
`2·n·max(conv(a,a))` equals the value returned by `evaluate_sequence`
for a vector normalised to sum = 1, so minimising it is equivalent to the
original task.
"""

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v, s=1.0):
    """Project a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Score for a vector already normalised to sum = 1: 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Parametric families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n, p):
    """Discrete symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w

def _distribution_beta(n, p, q):
    """Discrete asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w

def _distribution_edge_power(n, p, offset=0.0):
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5)+offset)^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5) + offset) ** (-p)
    w = np.maximum(w, 1e-20)
    w /= w.sum()
    return w

def _mix(v1, v2, lam):
    """Convex combination lam·v1 + (1‑lam)·v2, renormalised."""
    mix = lam * v1 + (1.0 - lam) * v2
    mix = np.maximum(mix, 1e-20)
    mix /= mix.sum()
    return mix

        sc = _max_convolution_score(vec)

            print(f"[{time.time():.1f}] n={n:5d} arcsine            score={sc:.6f}")

            # left‑heavy
            vec = _distribution_beta(n, p, 0.5)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} beta(p={p:.3f},q=0.5) score={sc:.6f}")

            # right‑heavy
            vec = _distribution_beta(n, 0.5, p)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} beta(p=0.5,q={p:.3f}) score={sc:.6f}")

        for offset in offsets:
            for p in edge_ps:
                if time.time() >= deadline:
                    break
                vec = _distribution_edge_power(n, p, offset=offset)
                vec = np.maximum(vec, eps); vec /= vec.sum()
                sc = _max_convolution_score(vec)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec.copy(), n
                    print(f"[{time.time():.1f}] n={n:5d} edge_power(p={p:.2f},off={offset}) score={sc:.6f}")

        # ----- mixtures of uniform and edge‑power -------------------------------
        uniform_vec = _distribution_uniform(n)
        for offset in offsets:
            for p in edge_ps:

                for lam in mix_lams:
                    if time.time() >= deadline:
                        break
                    vec = _mix(edge_vec, uniform_vec, lam)   # lam*edge + (1‑lam)*uni
                    sc = _max_convolution_score(vec)
                    if sc < best_score:
                        best_score, best_seq, best_n = sc, vec.copy(), n
                        print(f"[{time.time():.1f}] n={n:5d} mix(lam={lam:.2f},p={p:.2f}) score={sc:.6f}")

        # ----- a few random Dirichlet draws (concentration = 1) ----------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.ones(n))
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} dirichlet(1)       score={sc:.6f}")

            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), prev.size
                print(f"[{time.time():.1f}] previous best            score={sc:.6f}")

        pass

    # Guard against an empty result

        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform          score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Mirror‑descent refinement (soft‑max surrogate)
    a = best_seq.copy()
    n = a.shape[0]

    remaining = deadline - time.time()
    est_per_iter = max(0.003, 0.001 * (n / 4000.0))    # crude estimate
    max_iters = int(max(0, remaining / est_per_iter))

    total_iters = max(8000, min(60000, max_iters))

    # temperature (soft‑max) schedule
    alpha_start, alpha_end = 1.0, 1e6
    # learning‑rate schedule
    lr_start, lr_end = 0.5, 1e-4

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        maxc = conv.max()

        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        # bookkeeping

            if cur_sc < best_score - 1e-12:

                print(f"[{time.time():.1f}] mirror it={it}/{total_iters} new best {cur_sc:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Simulated‑annealing hill‑climbing polish
    a = best_seq.copy()
    cur_score = best_score
    temperature_start = 0.02
    temperature_end = 1e-6

    steps_remaining = int(max(0, (deadline - time.time()) / est_per_iter))
    if steps_remaining < 1:
        steps_remaining = 1

    for step in range(steps_remaining):
        if time.time() >= deadline:
            break
        # linear temperature schedule
        T = temperature_start * (temperature_end / temperature_start) ** (step / steps_remaining)

        # propose a random mass transfer
        i, j = random.sample(range(n), 2)
        if a[i] < eps:
            continue
        delta = min(a[i] * 0.2, 0.03) * random.random()
        if delta <= eps:
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_score = _max_convolution_score(a2)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = a2
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] anneal step={step}/{steps_remaining} new best {cur_score:.6f}")

        # occasional jitter to avoid stagnation
        if step % max(1, steps_remaining // 20) == 0 and step > 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            cur_score = _max_convolution_score(a)

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
