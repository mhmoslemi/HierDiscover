def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input is assumed non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta( p , p ) – p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric centred Gaussian window."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_tukey(n: int, alpha: float) -> np.ndarray:
    """Tukey window (implemented directly, no SciPy required)."""
    # alpha ∈ [0,1]; alpha=0 → rectangular, alpha=1 → Hann.
    if alpha <= 0:
        return _distribution_uniform(n)
    if alpha >= 1:
        # Hann window
        i = np.arange(n, dtype=np.float64)
        w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        w /= w.sum()
        return w
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    # first cosine taper
    idx1 = i < alpha / 2
    w[idx1] = 0.5 *
        (1 + np.cos(2 * np.pi / alpha * (i[idx1] - alpha / 2)))
    # second taper
    idx3 = i > 1 - alpha / 2
    w[idx3] = 0.5 *
        (1 + np.cos(2 * np.pi / alpha * (i[idx3] - 1 + alpha / 2)))
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex blend base with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _subgradient_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         lr0: float = 0.8,
                         top_k: int = 10,
                         eval_interval: int = 60,
                         max_iters: int = 500000) -> tuple[np.ndarray, float]:
    """Mirror‑descent on max‑convolution (exponentiated‑gradient)."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _fft_self_convolve(a)
        # indices of the top‑k entries (including the zero shift)
        idxs = np.argpartition(-conv, range(top_k))[:top_k]
        grad = np.zeros_like(a)
        for idx in idxs:
            # contribution of a to conv at shift idx
            # conv[idx] = Σ a[j] * a[idx−j]; derivative w.r.t a[i] = 2*a[idx−i] (if in bounds)
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)        # exponentiated‑gradient update
        a = _project_onto_simplex(a)       # enforce Σa=1, a≥0

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _lp_refine(f: np.ndarray, M: float, deadline: float,
               best_seq: np.ndarray, best_score: float,
               mix_frac: float = 0.2) -> tuple[np.ndarray, float]:
    """One step of LP‑based refinement (SciPy required)."""
    try:
        from scipy.optimize import linprog
        from scipy.sparse import diags, csr_matrix
    except Exception:  # pragma: no cover
        return best_seq, best_score

    n = f.shape[0]
    rows = 2 * n - 1

    # Build the lower‑triangular Toeplitz matrix A (shape rows×n)
    # A[k, j] = f[k - j] for k>=j, else 0.
    diags_data = []
    offsets = []
    for d in range(n):
        diag = np.full(n - d, f[d], dtype=np.float64)   # length = n‑d
        diags_data.append(diag)
        offsets.append(d)
    A = diags(diags_data, offsets, shape=(rows, n), format='csr')

    # Linear program: maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if not res.success:
        return best_seq, best_score

    g_raw = np.maximum(res.x, 0.0)  # just in case
    S = g_raw.sum()
    if S == 0.0:
        return best_seq, best_score

    g_norm = g_raw / S          # Σ g_norm = 1
    candidate = (1 - mix_frac) * f + mix_frac * g_norm
    candidate = _project_onto_simplex(candidate)

    cur_score = _objective(candidate)
    if cur_score < best_score - 1e-14:
        print(f"[{time.time():.1f}] LP‑refine improved score {cur_score:.6f}")
        return candidate, cur_score
    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Simple annealing on the exact objective."""
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective(cur)
    T = 0.05                     # initial temperature (objective units)
    cooling = 0.9995
    it = 0
    while time.time() < deadline:
        i = np.random.choice(np.where(cur > 1e-12)[0])
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # move up to 30 % of source mass
        delta = cur[i] * np.random.rand() * 0.3
        if delta < 1e-14:
            continue
        new = cur.copy()
        new[i] -= delta
        new[j] += delta

        new_score = _objective(new)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur = new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"[{time.time():.1f}] anneal it={it} new best {best_score:.6f}")
        T *= cooling
        if T < 1e-8:
            T = 0.05       # occasional reheating to escape plateaus
        it += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list (the caller normalises internally).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Rich pool of analytic / random seeds
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]   # diverse discretisations
    # grid for exponent p (Beta / edge families)
    p_grid = np.concatenate((
        np.logspace(-2, -0.5, 10, base=10),
        np.logspace(-0.5, 0.0, 10, base=10),
        np.logspace(0.0, 0.6, 8, base=10),
    ))
    lam_grid = np.linspace(0.0, 1.0, 6)               # mixing with uniform
    sigma_grid = np.linspace(0.2, 0.6, 7)             # fraction of length for Gaussian
    tukey_grid = np.linspace(0.0, 1.0, 6)            # α for Tukey
    # SciPy window parameters (if SciPy is present)
    scip_window_params = {
        "hann": None,
        "hamming": None,
        "blackman": None,
        "nuttall": None,
        "kaiser": np.linspace(0.0, 14.0, 6),
        "chebwin": np.linspace(30, 150, 5),
    }

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # --- deterministic families ------------------------------------
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new global best (uniform) len={n} score={sc:.6f}")

        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] new global best (arcsine) len={n} score={sc:.6f}")

        # --- parametric families ---------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)

            for base in (beta, edge):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new global best p={p:.3g} lam={lam:.2f} len={n} score={sc:.6f}")

        # --- Gaussian windows -----------------------------------------
        for sigma_frac in sigma_grid:
            if time.time() >= deadline:
                break
            sigma = sigma_frac * n
            cand = _distribution_gaussian(n, sigma)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (gaussian σ={sigma_frac:.2f}) len={n} score={sc:.6f}")

        # --- Tukey windows --------------------------------------------
        for alpha in tukey_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_tukey(n, alpha)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best (tukey α={alpha:.2f}) len={n} score={sc:.6f}")

        # --- SciPy windows (optional) ---------------------------------
        if _HAS_SCI:
            for win_name, param_grid in scip_window_params.items():
                if param_grid is None:
                    # simple windows without extra parameter
                    cand = _distribution_scipy_window(win_name, n)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(f"[{time.time():.1f}] new global best ({win_name}) len={n} score={sc:.6f}")
                else:
                    # windows requiring a numeric parameter
                    for par in param_grid:
                        if time.time() >= deadline:
                            break
                        cand = _distribution_scipy_window(win_name, n, param=par)
                        sc = _objective(cand)
                        top_candidates.append((sc, cand.copy()))
                        if sc < overall_best_score:
                            overall_best_score = sc
                            overall_best_seq = cand.copy()
                            print(f"[{time.time():.1f}] new global best ({win_name}={par}) len={n} score={sc:.6f}")

        # --- random Dirichlet seeds ------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best (rand Dirichlet) len={n} score={sc:.6f}")

        # --- sparse random seeds ----------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.01))
            idx = rng.choice(n, size=k, replace=False)
            a[idx] = rng.random(k)
            a /= a.sum()
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best (sparse) len={n} score={sc:.6f}")

        # Retain only the best candidates seen so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Safety fallback
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Mirror‑descent refinement on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] sub‑gradient refinement on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            cand_seq, cand_score = _subgradient_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                top_k=10,
                eval_interval=60,
            )
            if cand_score < overall_best_score - 1e-12:
                overall_best_score = cand_score
                overall_best_seq = cand_seq.copy()
                print(f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  LP‑based refinement (if SciPy available)
    # ------------------------------------------------------------------
    if _HAS_SCI and time.time() < deadline - 5.0:
        M_cur = _fft_self_convolve(overall_best_seq).max()
        # try at most three LP refinement steps
        for lp_iter in range(3):
            if time.time() >= deadline:
                break
            overall_best_seq, overall_best_score = _lp_refine(
                overall_best_seq, M_cur, deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                mix_frac=0.2,
            )
            M_cur = _fft_self_convolve(overall_best_seq).max()
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # 6.  Simulated annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polishing")
        overall_best_seq, overall_best_score = _simulated_annealing(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 7.  Report result
    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
