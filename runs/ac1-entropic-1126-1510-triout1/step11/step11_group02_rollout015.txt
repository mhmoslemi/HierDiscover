        # beta symmetric p grid including lower than 0.05

                    x_old = np.linspace(0,1,prev.size)
                    x_new = np.linspace(0,1,n)

            pass

        # evaluate seeds quickly and keep top few

def _adam_smooth_descent(a, deadline, best_seq, best_score,
                         lr=0.1, beta1=0.9, beta2=0.999,
                         alpha0=1.0, alpha_max=200.0, alpha_factor=1.05,
                         epochs=None):
    a = a.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = alpha0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)   # additive descent
        a = a - step

        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # increase alpha gradually
        if t % 200 == 0:
            alpha = min(alpha * alpha_factor, alpha_max)

        if t % 500 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] adam iter={t} α={alpha:.2f} new best {cur:.6f}")

        # optionally break after many iterations
        if epochs is not None and t >= epochs:
            break
    return best_seq, best_score

"""
Hybrid optimizer for the “max‑convolution” step‑function problem.

The algorithm proceeds in three stages:

1. **Seeding** – a palette of analytic distributions (uniform,
   arcsine, symmetric Beta with a wide range of exponents, Dirichlet,
   plus the previously discovered best vector if available) is generated
   for a single length (chosen from the previous solution or a default
   of 4000).  The best few seeds (by the exact objective) are kept.

2. **Adam‑driven smooth‑max descent** – starting from each seed we minimise
   the smooth approximation
          f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a))
   using Adam (first‑order adaptive moments) and a Euclidean projection onto
   the simplex (∑a_i = 1, a_i ≥ 0).  The temperature α is ramped from a
   modest value to a large one (≈400) so that the surrogate gradually
   approaches the true max‑convolution.  The best vector found during this
   phase is recorded.

3. **Finishing refinements** – a short multimax sub‑gradient phase (the
   classic “active‑set” descent from the reference) and a cheap
   pair‑wise mass‑transfer polishing step are applied.

All computation is performed with FFT‑based convolutions (O(n log n)).
The routine respects a 1000 s wall‑clock budget and returns the best
sequence found as a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_i w_i·conv_i(a) with respect to a.
    It equals 2·a⋆w where ⋆ denotes cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv(a,a)) / (sum(a)²).  Assumes sum(a)>0."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic candidate families -----------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0)) +
               (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _adam_smooth_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr: float = 0.05,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         alpha0: float = 1.0,
                         alpha_max: float = 400.0,
                         alpha_factor: float = 1.02,
                         max_iter: int = 200_000,
                         eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The temperature α starts at ``alpha0`` and is increased geometrically
    (multiplying by ``alpha_factor``) until ``alpha_max``.
    The iterates are projected onto the probability simplex after every step.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # ----- smooth‑max weights -----
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))      # safe because shift = max(conv)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # ----- gradient -----
        grad = _gradient_from_weights(a, w)

        # ----- Adam update (additive) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step

        # ----- projection onto the simplex (∑a = 1, a ≥ 0) -----
        a = _project_onto_simplex(a, s=1.0)

        # ----- temperature schedule -----
        if t % 100 == 0:
            alpha = min(alpha * alpha_factor, alpha_max)

        # ----- occasional evaluation -----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {cur:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.5,
                                 max_iter: int = 50_000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classic sub‑gradient descent with the exact max‑convolution (active set)
    as in the reference implementation.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Multimax it={t} new best {cur:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy local mass‑transfer moves that only keep improvements."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src_idx = np.where(a > 1e-12)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)
        # pick a distinct destination index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30  # move up to 30 % of source mass
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {cur:.6f}")

    return best_seq, best_score

            seed_vec.copy(),
            slice_deadline,
            best_seq if best_seq is not None else seed_vec,
            best_score,
            lr=0.05,
            alpha0=2.0,
            alpha_max=400.0,
            alpha_factor=1.02,
            max_iter=200_000,
            eval_interval=2000,
        )
        if score_out < best_score:
            best_score = score_out

            print(f"[{time.time():.1f}] after multimax new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Small random perturbation restarts (very cheap) – can escape
    #      shallow local minima.
    if best_seq is not None and time.time() < deadline:
        for _ in range(5):
            if time.time() >= deadline:
                break
            # add a tiny Gaussian noise, re‑normalise
            pert = best_seq * (1.0 + 0.01 * rng.standard_normal(n))
            pert = np.maximum(pert, 0.0)
            ss = pert.sum()
            if ss == 0.0:
                continue
            pert = pert / ss
            # run a short Adam burst (few thousand iters)
            sub_dead = time.time() + min(4.0, deadline - time.time())
            seq_out, score_out = _adam_smooth_descent(
                pert,
                sub_dead,
                best_seq,
                best_score,
                lr=0.05,
                alpha0=5.0,
                alpha_max=200.0,
                alpha_factor=1.02,
                max_iter=30_000,
                eval_interval=3000,
            )
            if score_out < best_score:
                best_score = score_out
                best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] perturb restart improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final cheap local‑swap polishing
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
        )

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    print(f"[{time.time():.1f}] Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
