Implementation of simulated annealing:

def simulated_annealing(a, deadline, initial_temp=0.01, final_temp=1e-6, max_iter=50000):
    T = initial_temp
    a_curr = a.copy()
    sc_curr = _max_convolution_score(a_curr)
    best = a_curr.copy()
    best_sc = sc_curr
    n = len(a_curr)
    it = 0
    while time.time() < deadline and it < max_iter:
        # schedule temperature using exponential cooling based on remaining time
        # compute fraction of total SA steps based on elapsed time proportion.
        # For simplicity: reduce T each iteration by factor r = (final_temp/initial_temp)^(1/max_iter)
        # but we can also adapt to remaining time.
        # Perform trial:
        src = np.random.randint(n)
        trg = np.random.randint(n)
        if src == trg: continue
        # choose delta up to a fraction of src mass
        max_delta = a_curr[src] * 0.5  # move at most half
        if max_delta <= 0: continue
        delta = np.random.rand() * max_delta * 0.1  # small move effect
        a_new = a_curr.copy()
        a_new[src] -= delta
        a_new[trg] += delta
        # ensure nonnegativity
        if a_new[src] < 0: continue
        a_new = a_new / a_new.sum()
        sc_new = _max_convolution_score(a_new)
        delta_sc = sc_new - sc_curr
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / T):
            a_curr = a_new
            sc_curr = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best = a_new.copy()
                # print improvement
                print(f"[{time.time():.1f}] SA it={it} new best {best_sc:.6f}")
        # update temperature
        T = max(final_temp, T * 0.9995)  # cooling factor
        it += 1
    return best, best_sc

def distribution_beta_sym(n, p):
    x = (np.arange(n)+0.5)/n
    logw = (p-1)*np.log(x) + (p-1)*np.log(1-x)
    w = np.exp(logw)
    w_sum = w.sum()
    if not np.isfinite(w_sum) or w_sum == 0:
        w = np.full_like(w, 1/n)
    else:
        w = w/w_sum
    return w

"""
search for a non‑negative step function that minimises `evaluate_sequence`.

The algorithm consists of three stages:

1. **Broad exploration** – a set of analytic families (uniform, arcsine,
   symmetric Beta(p,p), mixtures with the uniform distribution, Dirichlet draws
   and random soft‑max vectors) are evaluated for several lengths (including
   the length of a historic solution, if present).  The best few vectors are
   kept in an *elite* list.

2. **Massive random search** – while time permits we draw random candidates
   (random length, random Beta‑parameter, random Dirichlet concentration or a
   random log‑softmax) and keep any improvement.

3. **Local refinement** – each elite vector is polished by
   *smooth‑max mirror descent* (exponentiated‑gradient on a log‑sum‑exp
   surrogate), followed by a short *simulated‑annealing* walk and a final
   projected sub‑gradient step.  The best refined vector is returned.

All heavy loops stop as soon as the deadline (1000 s minus a safety margin)
is reached, guaranteeing a result is produced.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT helpers ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective for a normalised vector (sum = 1): 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_random_softmax(n):
    """Draw a random vector from a soft‑max of Gaussian logits."""
    logits = np.random.randn(n)
    # subtract max for numerical stability
    logits -= np.max(logits)
    exp_logits = np.exp(logits)
    vec = exp_logits / exp_logits.sum()
    return vec


# ----------------------------------------------------------------------
# Local optimisation – smooth‑max mirror descent -----------------------
def _smooth_max_mirror_descent(a0, deadline):
    """
    Exponentiated‑gradient descent on the smooth‑max surrogate.
    Three temperature stages, each with a decreasing learning rate.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    schedule = [
        (1.0, 1e3, 3000, 0.15, 0.015),
        (1e3, 1e5, 3000, 0.015, 0.0015),
        (1e5, 1e6, 4000, 0.0015, 0.00015),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated annealing --------------------------------------------------
def _simulated_annealing(a0, deadline,
                         initial_temp=2e-3,
                         final_temp=1e-7,
                         max_iter=200000):
    """
    Classic Metropolis walk on the objective.
    Moves consist of transferring a small amount of mass from one index to another.
    Temperature decays exponentially.  Returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    cur_sc = _max_convolution_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    # cooling factor chosen to reach final_temp after max_iter steps
    cooling = (final_temp / initial_temp) ** (1.0 / max_iter)
    T = initial_temp

    it = 0
    while time.time() < deadline and it < max_iter:
        # propose a move
        src = random.randint(0, n - 1)
        trg = random.randint(0, n - 1)
        if src == trg:
            continue
        # move at most half of the source mass, scaled down to keep moves tiny
        max_delta = a[src] * 0.5
        if max_delta <= eps:
            continue
        delta = random.random() * max_delta * 0.2   # 0‑20 % of allowed move
        a_new = a.copy()
        a_new[src] -= delta
        a_new[trg] += delta
        # re‑normalise (numerical drift)
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_sc = _max_convolution_score(a_new)

        delta_sc = new_sc - cur_sc
        if delta_sc < 0.0 or random.random() < math.exp(-delta_sc / T):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA it={it} new best {best_sc:.6f}")

        # geometric cooling
        T = max(final_temp, T * cooling)
        it += 1

    return best_a, best_sc

# Projected sub‑gradient descent (short version) -----------------------
def _projected_subgradient(a0, deadline, max_iters=20000):
    """True sub‑gradient descent with simple back‑tracking line search."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    step0 = 0.7
    best_sc = _max_convolution_score(a)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (indices attaining the max, up to numerical tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # back‑tracking line search
        step = step0
        while True:
            a_candidate = a - step * grad
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0 and time.time() + 0.5 >= deadline:
            break

    return best_a, best_sc

# Candidate generator ----------------------------------------------------
def _generate_candidates_for_n(n):
    """Return a list of normalised candidate vectors for a given length."""
    cand = []

    # Basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))  # arcsine

    # Grid of symmetric betas
    grid_ps = np.concatenate(
        (np.logspace(-2, -0.2, 12), np.linspace(0.1, 2.0, 20))
    )
    for p in grid_ps:
        cand.append(_distribution_beta_sym(n, float(p)))

    # Mixtures with the uniform distribution
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.2, 0.5, 0.8):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # Random soft‑max (Gaussian logits)
    for _ in range(3):
        cand.append(_distribution_random_softmax(n))

    # Small random shuffles to break symmetry
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand
