"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  The algorithm works in three phases:

1.  Broad exploration – a few analytic families (arcsine, symmetric
    Beta(p,p), asymmetric Beta(p,q), edge‑biased power‑law) are instantiated
    for several modest lengths (2000–16000).  Additionally, simple convex
    mixtures of arcsine with uniform or edge‑biased distributions are tried.
    The best few candidates are kept.

2.  Local refinement – each retained candidate is optimised with a
    smooth‑max mirror‑descent (exponentiated‑gradient) loop whose temperature
    grows to a very large value, followed by a projected‑Adam sub‑gradient
    phase.  The number of iterations is adapted to the remaining time.

3.  Final polishing – an aggressive mass‑transfer hill‑climber runs for as
    long as time permits, moving tiny probability mass from the indices that
    contribute most to the current convolution peak to the smallest entry.
    The best sequence found overall is returned.

All vectors are kept on the probability simplex (sum = 1) and values are
clamped to a tiny ε>0 to avoid divide‑by‑zero issues.
"""

import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT utilities (real)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a probability vector (sum = 1) returns the exact value produced by
    `evaluate_sequence`: 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# Analytic families (return probability vectors, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density to avoid overflow
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """Deterministic edge‑biased distribution."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_mixture(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam*a + (1-lam)*b, normalised."""
    m = lam * a + (1.0 - lam) * b
    m = np.maximum(m, 1e-12)
    m /= m.sum()
    return m

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float, eps: float = 1e-12):
    """
    One exponentiated‑gradient step on the smooth‑max approximation of
    max(conv).  Returns the updated probability vector and the exact
    maximal convolution entry (before smoothing).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights for the smooth max
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of smooth max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()

def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """Projected Adam‑style step on the true max‑convolution."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()

def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Simple hill‑climber that repeatedly moves a tiny amount of probability
    mass from the (i,j) pair that dominates the current peak towards the
    globally smallest entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        # location of the current peak
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) that yields the largest product for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # pick source (the larger of the two) and target (global min)
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]  # second‑smallest

        # move a tiny chunk
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional random jitter to escape plateaus
        if (it + 1) % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

            if time.time() >= deadline:
                break
            try_candidate(_distribution_beta(n, p, 0.5), f"Beta(p={p:.3f},q=0.5)")
            try_candidate(_distribution_beta(n, 0.5, p), f"Beta(p=0.5,q={p:.3f})")

        # edge‑biased deterministic power‑law
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            try_candidate(_distribution_edge_power(n, alpha), f"EdgePower(alpha={alpha:.2f})")

        # mixtures with uniform and arcsine
        base_arc = _distribution_arcsine(n)
        base_uni = _distribution_uniform(n)
        for lam in mix_lam:
            if time.time() >= deadline:
                break
            vec = _distribution_mixture(base_arc, base_uni, lam)
            try_candidate(vec, f"MixArcUni(lam={lam:.2f})")

        # mixtures with edge‑power and arcsine
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            base_edge = _distribution_edge_power(n, alpha)
            for lam in mix_lam:
                if time.time() >= deadline:
                    break
                vec = _distribution_mixture(base_arc, base_edge, lam)
                try_candidate(vec, f"MixArcEdge(alpha={alpha:.2f},lam={lam:.2f})")

    # keep only the top K global candidates
    K = 10
    global_top.sort(key=lambda x: x[0])
    global_top = global_top[:K]

    # ------------------------------------------------------------------
    # 2️⃣  Local refinement of each top candidate
    # ------------------------------------------------------------------
    for idx, (sc, cand) in enumerate(global_top):
        if time.time() >= deadline - 5.0:
            break
        a = cand.copy()
        n = a.shape[0]

        # Decide iteration budget for this candidate (roughly proportional to remaining time)
        remaining = deadline - time.time()
        est_it = max(2000, int(remaining / (0.0004 * len(global_top))))   # very rough estimate

        # ----- smooth‑max mirror descent -----
        total_iters = min(30000, est_it)
        alpha_start, alpha_end = 0.5, 5e5
        lr_start, lr_end = 0.4, 0.0005
        for it in range(total_iters):
            if time.time() >= deadline - 3.0:
                break
            t = it / total_iters
            alpha = alpha_start * (alpha_end / alpha_start) ** t
            lr = lr_start * (lr_end / lr_start) ** t
            a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)
            # occasional exact check
            if (it + 1) % max(1, total_iters // 8) == 0:
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score - 1e-12:
                    best_seq, best_score = a.copy(), cur_score
                    print(f"[{time.time():.1f}] cand {idx} mirror it={it}/{total_iters} new best {cur_score:.6f}")

        # ----- Adam‑style sub‑gradient descent -----
        velocity = np.zeros_like(a)
        step0 = 0.6
        beta1, beta2 = 0.9, 0.999
        sub_it = 0
        while time.time() < deadline - 2.0:
            sub_it += 1
            a, velocity, cur_max = _adam_subgrad_step(a, velocity, step0,
                                                      beta1, beta2, sub_it, eps)
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] cand {idx} subgrad it={sub_it} new best {cur_score:.6f}")

            if sub_it % 500 == 0:
                # small jitter to avoid being stuck
                jitter = np.random.randn(n) * 1e-5
                a = np.maximum(a + jitter, eps)
                a /= a.sum()

        # ----- aggressive hill climbing -----
        # use as many iterations as time permits (but not too many)
        hill_iters = 8000
        a = _mass_transfer_hill_climb(a, hill_iters, eps)
        final_sc = _max_convolution_score(a)
        if final_sc < best_score - 1e-12:
            best_seq, best_score = a.copy(), final_sc
            print(f"[{time.time():.1f}] cand {idx} hill‑climber improved to {final_sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing on the overall best sequence
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple arcsine of moderate size
        n0 = 4000
        best_seq = _distribution_arcsine(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    a = best_seq.copy()
    n = a.shape[0]
    remaining = deadline - time.time()
    # allocate the rest of the budget to a very long hill‑climber run
    max_hill_iters = int(max(2000, remaining / 0.00005))
    a = _mass_transfer_hill_climb(a, max_hill_iters, eps)
    final_score = _max_convolution_score(a)
    if final_score < best_score:
        best_seq, best_score = a.copy(), final_score
        print(f"[{time.time():.1f}] final polishing improved score to {final_score:.6f}")

    # ------------------------------------------------------------------
    # Return best sequence as a plain Python list
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
