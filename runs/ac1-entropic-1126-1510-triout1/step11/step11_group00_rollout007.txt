def _lp_step(f, M):
    n = len(f)
    # Build A matrix shape (2*n-1, n) as described.
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(2*n-1):
        # j range: max(0, i-(n-1)) .. min(n-1, i)
        j_min = max(0, i-(n-1))
        j_max = min(n-1, i)
        # f index k = i - j
        # fill part
        idxs = np.arange(j_min, j_max+1)
        A[i, idxs] = f[i - idxs]  # vectorized
    # bounds
    bounds = [(0, None) for _ in range(n)]
    # constraint upper bound vector
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    # objective: maximize sum b => minimize -sum b
    c = -np.ones(n, dtype=np.float64)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return f unchanged
        return f, M, np.inf
    b = res.x
    sum_b = b.sum()
    if sum_b == 0:
        g = np.full_like(b, 1.0/n)
    else:
        g = b / sum_b

def _lp_improve(f, t, max_iter, deadline):
    n = len(f)
    best_f = f.copy()
    best_ratio = _objective(f) (where sum=1). Actually objective function returns 2n * max(conv). Use ratio_ = _objective(f).
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        M = _fft_self_convolve(f).max()
        # Build A matrix
        # Use scipy.sparse.diags; each diagonal constant = f[d]; offset d
        data = np.vstack([np.full(n, f[d]) for d in range(n)])  # shape (n,n)
        offsets = np.arange(n) # non-negative offsets
        A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
        b_ub = np.full(2*n-1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            # If LP fails, break early
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            # fallback to uniform
            g = np.full(n, 1.0/n)
        else:
            g = b / sum_b
        # update
        f_new = (1 - t) * f + t * g
        # normalize to sum=1 due to numerical errors
        f_new = f_new / f_new.sum()
        # evaluate
        ratio = _objective(f_new)  # sum=1
        if ratio < best_ratio - 1e-12:
            best_ratio, best_f = ratio, f_new.copy()
            # maybe reduce t if improvement small? Not needed.
        # keep iterative
        f = f_new
        # optionally adapt t = t * 0.99 for later? Not needed.
    return best_f, best_ratio

"""
Propose a non‑negative step function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Rich initial pool** – Generate a few analytic families
   (uniform, arcsine, symmetric beta, edge‑weighted, sine‑weighted)
   for several lengths, plus a handful of Dirichlet / sparse random vectors.
   Keep the ~15 best (lowest 2·n·max(conv)) as seeds.

2. **Active‑set linear‑programming descent** – For every seed run a small
   number of iterations of the following LP:
       maximise   Σ b_j
       subject to  (f * b)_k ≤ M   for all “active’’ convolution indices k,
                   b_j ≥ 0,
   where M = max (f * f) and only the indices whose value is within a
   fraction ``active_frac`` of M are kept (this makes the LP tiny).
   The optimiser returns a distribution ``b``; after normalising it
   we move the current vector towards ``b`` by a step size ``t``.
   A tiny random jitter is injected if the improvement stalls.

   The LP is solved with ``scipy.optimize.linprog`` (method = 'highs')
   on a dense constraint matrix whose size is at most
   ``max_active × n`` (typically a few hundred rows).

3. **Local polishing** – A cheap pairwise‑mass‑transfer walk followed by a
   simulated‑annealing style random walk (exactly as in the previous
   script) are applied to the best vector found so far.

The whole search respects the supplied time budget (default 1000 s) and
returns a plain ``list[float]``; ``evaluate_sequence`` will handle the
final clipping and normalisation.
"""

import numpy as np

def _project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Helper for active‑set LP ------------------------------------------------
def _active_constraint_matrix(f: np.ndarray, act_idx: np.ndarray) -> np.ndarray:
    """
    Build the (|act_idx| × n) matrix A where
    A[k, j] = f[act_idx[k] - j] if 0 ≤ act_idx[k]‑j < n, else 0.
    The rows correspond exactly to the convolution constraints (f * b)[k] ≤ M.
    """
    n = f.shape[0]
    m = act_idx.shape[0]
    A = np.empty((m, n), dtype=np.float64)
    for row, k in enumerate(act_idx):
        # range of j that gives a valid index
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        # zero‑initialise the whole row first
        A[row, :] = 0.0
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1, dtype=np.int64)
            A[row, js] = f[k - js]
    return A

# Active‑set LP improvement -----------------------------------------------
def _lp_improve(
    f: np.ndarray,
    t: float = 0.2,
    max_iter: int = 20,
    deadline: float = 1e9,
    active_frac: float = 0.1,
    max_active: int = 250,
    patience: int = 400,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient style descent where at each iteration a linear
    program (maximise Σb_j under (f*b)_k ≤ M for all *active* k) is solved.
    The solution b is renormalised and blended into f.  Small random jitter
    is injected when the improvement stalls.
    """
    if _opt is None:                     # SciPy not available – skip LP
        return f, _objective(f)

    n = f.shape[0]
    best_f = f.copy()
    best_score = _objective(f)
    no_improve = 0
    rng = np.random.default_rng()

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(f)
        M = conv.max()

        # ---- active set -------------------------------------------------
        thresh = M * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.shape[0] > max_active:
            # keep only the largest entries
            top = np.argsort(conv[act])[-max_active:]
            act = act[top]

        # ---- build LP ---------------------------------------------------
        A = _active_constraint_matrix(f, act)
        b_ub = np.full(act.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)               # maximise Σb
        bounds = [(0.0, None)] * n

        # ---- solve -------------------------------------------------------
        res = _opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # if the LP fails we abort the LP phase for this seed
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            g = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            g = b / sum_b

        # ---- blend -------------------------------------------------------
        f_new = (1.0 - t) * f + t * g
        f_new = f_new / f_new.sum()                     # keep Σ = 1

        score = _objective(f_new)

        if score < best_score - 1e-14:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new
            no_improve += 1

        # ---- jitter on stagnation ----------------------------------------
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            f += jitter
            f = np.maximum(f, 0.0)
            f /= f.sum()
            no_improve = 0

    return best_f, best_score

# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
