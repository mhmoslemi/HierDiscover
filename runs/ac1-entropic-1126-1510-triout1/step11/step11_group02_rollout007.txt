"""
Hybrid Adam‑based optimiser for the max‑convolution step‑function problem.

The algorithm proceeds in three stages:

1. **Broad seeding** – a variety of analytic families (uniform,
   arcsine, beta, power‑law, exponential, sinusoidal, triangular,
   Gaussian, random Dirichlet, sparse patterns) are generated for several
   lengths.  The best few (by the exact objective) are kept as seeds.

2. **Adam smooth‑max descent** – each seed is optimised on the
   smooth‑max surrogate
        f_α(a) = (1/α) log Σ_i exp(α·conv_i(a))
   using an Adam optimiser on the soft‑max parametrisation
   a = softmax(z).  The temperature α starts small and is increased
   geometrically up to a large value, so the surrogate converges to the
   true max.  The exact objective is evaluated periodically and the best
   vector is retained.

3. **Polishing** – a short multimax sub‑gradient descent followed by a
   cheap greedy pairwise‑swap local search are applied to the current
   best solution.

All vectors are kept on the probability simplex (non‑negative,
sum = 1), which is invariant under the soft‑max parametrisation.
The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes sum(a) > 0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    """Positive half‑sine shape, symmetric, zero‑free."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    """Peak in the centre, linear rise/fall."""
    i = np.arange(n, dtype=np.float64)
    vals = np.minimum(i + 1, n - i)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates_extended(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a broad variety of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_gaussian(n))

    # symmetric Beta family with a grid of exponents
    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric Beta draws
    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law families
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential families
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes
    for block_frac in (0.05, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random Dirichlet draws
    for _ in range(4):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # a few completely random probability vectors
    for _ in range(3):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        if v.sum() == 0.0:
            v[0] = 1.0
        v /= v.sum()
        cand.append(v)

    return cand


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    shift = np.max(z)
    exp_z = np.exp(z - shift)
    s = exp_z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / len(z))
    return exp_z / s


# ----------------------------------------------------------------------
def _adam_optimize(z0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   alpha0: float = 1.0,
                   alpha_max: float = 500.0,
                   growth_factor: float = 1.02,
                   lr: float = 0.1,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8,
                   max_iter: int = 200000,
                   eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Parameters follow standard Adam defaults; the temperature α grows
    geometrically until alpha_max.
    """
    n = z0.shape[0]
    z = z0.copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # --- probability vector on the simplex ---
        a = _softmax(z)                     # a_i ≥ 0, sum = 1

        # --- convolution and smooth‑max weights ---
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # --- gradient wrt a (exact for the smooth surrogate) ---
        grad_a = _gradient_from_weights(a, w)

        # --- chain rule to obtain gradient wrt z (softmax Jacobian) ---
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # --- temperature schedule ---
        if t % 500 == 0:
            alpha = min(alpha * growth_factor, alpha_max)

        # --- periodic exact evaluation ---
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.5,
                                 max_iter: int = 20000,
                                 eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Classic multimax sub‑gradient descent (exponentiated‑gradient on the
    simplex).  Used for polishing after Adam.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk, accepting only objective‑improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    The routine spends the allotted time budget on (i) broad seeding,
    (ii) Adam smooth‑max descent from the most promising seeds,
    (iii) a short multimax polishing phase,
    (iv) a greedy pairwise‑swap local walk.
    The best sequence (as a plain ``list[float]``) is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Choose a handful of lengths (moderate n keeps each iteration cheap)
    lengths = [2000, 4000, 8000, 12000]

    overall_best_seq: Optional[np.ndarray] = None
    overall_best_score = float("inf")
    candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length (cheap evaluations)
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")

        # retain only the most promising seeds across all lengths
        candidates.sort(key=lambda x: x[0])
        candidates = candidates[:15]

    # ------------------------------------------------------------------
    # 3. Incorporate a previous solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback if nothing found
    if overall_best_seq is None:
        n = lengths[0]
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Adam smooth‑max descent on a handful of top seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        # take the best few distinct seeds
        top_seeds = [seq for _, seq in sorted(candidates, key=lambda x: x[0])[:8]]
        for seed_seq in top_seeds:
            if time.time() >= deadline:
                break
            # allocate a time slice (at least 10 s, respecting the overall deadline)
            slice_time = min(40.0, (deadline - time.time()) / max(1, len(top_seeds)))
            sub_deadline = time.time() + slice_time

            # initialise Adam on the soft‑max parametrisation
            z0 = np.log(np.maximum(seed_seq, 1e-300))

            seq_out, score_out = _adam_optimize(
                z0,
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                alpha0=1.0,
                alpha_max=500.0,
                growth_factor=1.02,
                lr=0.12,
                max_iter=200000,
                eval_interval=250,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after Adam new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Polishing: short multimax descent on the current best
    remaining = deadline - time.time()
    if remaining > 25.0:
        sub_deadline = time.time() + remaining * 0.6
        overall_best_seq, overall_best_score = _subgradient_multimax_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_iter=30000,
            eval_interval=300,
        )
        print(f"[{time.time():.1f}] after multimax best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final greedy pairwise‑swap walk (cheap but often yields tiny gains)
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] after final swaps best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
