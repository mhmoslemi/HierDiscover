def _random_hill_climb(a, deadline, best_seq, best_score, rng):
    n = a.shape[0]
    while time.time() < deadline:
        # choose number of indices (2..5)
        s = rng.integers(2, 6)
        idx = rng.choice(n, size=s, replace=False)

        new_weights = rng.random(s)
        new_weights = np.maximum(new_weights, 0)
        if new_weights.sum() == 0:

        new_weights = new_weights / new_weights.sum()
        new_a = a.copy()
        new_a[idx] = total * new_weights
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] hillclimb improved to {best_score:.6f}")
    return best_seq, best_score

"""
Hybrid search for a low‑max‑convolution step function.

* Scan a rich library of analytic families (beta, power, sine, cosine,
  Blackman, random Dirichlet, sparse, etc.) for several lengths.
* Keep the best few seeds and refine them with two stages of
  exponentiated‑gradient descent:
    1) smooth‑max (log‑sum‑exp) surrogate with a slowly increasing
       temperature,
    2) multimax sub‑gradient (weight on the active set where the
       convolution attains its maximum).
* Finish with a cheap stochastic hill‑climbing that redistributes mass
  among a few random indices.
* The whole routine respects the given time budget and returns the best
  sequence it has found as a plain list of floats.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of 2·max(conv) w.r.t. a when the weight vector w (sum 1)
    is used:  grad = 2·a⋆w.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) assuming a is a probability vector."""
    s = a.sum()
    if s == 0.0:
        return float("inf")
    a = a / s
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_sine_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_cosine_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))) ** p
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_blackman(n: int, p: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    vals = np.maximum(w, 0.0) ** p
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Return a fairly large list of plausible probability vectors."""
    seeds: List[np.ndarray] = []

    # Basic families
    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_arcsine(n))

    # Symmetric Beta family (dense grid)
    for p in np.linspace(0.05, 3.0, 31):
        seeds.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta random draws
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 2)   # 0.01 – 100
        beta = 10 ** rng.uniform(-2, 2)
        seeds.append(_distribution_beta_asym(n, alpha, beta))

    # Power‑law increasing
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        seeds.append(_distribution_power_increasing(n, p))

    # Exponential increasing
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        seeds.append(_distribution_exponential_increasing(n, lam))

    # Two‑block spikes
    for block_frac in (0.05, 0.10, 0.15):
        seeds.append(_distribution_two_blocks(n, block_frac))

    # Sine‑power and cosine‑power windows
    for p in (0.5, 1.0, 2.0, 3.0):
        seeds.append(_distribution_sine_power(n, p))
        seeds.append(_distribution_cosine_power(n, p))

    # Blackman window (different exponents)
    for p in (0.5, 1.0, 2.0):
        seeds.append(_distribution_blackman(n, p))

    # Dirichlet draws with varying concentration
    for _ in range(6):
        alpha = rng.uniform(0.1, 2.0)
        seeds.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # Random sparse patterns
    seeds.append(_distribution_random_sparse(n, rng, density=0.02))

    # Completely random dense vectors
    for _ in range(5):
        v = rng.random(n)
        v /= v.sum()
        seeds.append(v)

    # Mixtures of two random seeds
    for _ in range(8):
        d1 = seeds[rng.integers(len(seeds))]
        d2 = seeds[rng.integers(len(seeds))]
        w = rng.random()
        mix = w * d1 + (1.0 - w) * d2
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        seeds.append(mix)

    return seeds


# ----------------------------------------------------------------------
def _smooth_max_descent(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        eta0: float = 1.0,
                        alpha0: float = 1.0,
                        alpha_max: float = 200.0,
                        eval_interval: int = 1000) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth (log‑sum‑exp) surrogate.
    The temperature alpha is increased geometrically.
    """
    n = a.shape[0]
    t = 0
    alpha = alpha0
    while time.time() < deadline:
        t += 1
        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # gradient (averaged over all entries)
        grad = _gradient_from_weights(a, w)

        # step‑size schedule η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(t)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad)

        # project back onto simplex (keeps sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        # slowly increase temperature
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.25, alpha_max)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.2f} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.7,
                                 eval_interval: int = 1000) -> Tuple[np.ndarray, float]:
    """
    Classical multimax sub‑gradient descent.
    The weight vector is uniform on the active set (indices attaining the maximum).
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} best={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_hill_climb(a: np.ndarray,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float,
                       rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """Greedy mass‑re‑distribution among a few random indices."""
    n = a.shape[0]
    while time.time() < deadline:
        # pick a random subset size (2‑5)
        s = rng.integers(2, 6)
        idx = rng.choice(n, size=s, replace=False)
        total = a[idx].sum()
        if total == 0.0:
            continue
        # propose a new Dirichlet splitted mass
        new_weights = rng.random(s)
        new_weights /= new_weights.sum()
        new_a = a.copy()
        new_a[idx] = total * new_weights

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] hill‑climb improved to {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine runs for up to ``budget_s`` seconds
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a few lengths.  Larger n gives a finer discretisation,
    #    but each iteration costs a bit more.
    lengths = [6000, 8000, 12000]

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length (very cheap)
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        seeds = _generate_seeds(n, rng)
        for cand in seeds:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep a limited pool of promising seeds across all lengths
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 3. Incorporate any previously known solution (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 0.0:
                prev = prev / s
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = prev.copy()
                    print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
                top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback (should never be needed)
    if overall_best_seq is None:
        n = 6000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. First refinement: smooth‑max descent on a handful of the best seeds
    remaining = deadline - time.time()
    if remaining > 30:
        # keep the very best 8 seeds for refinement
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
        for i, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            # equal time slice for each seed
            sub_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds) - i)
            seq_out, score_out = _smooth_max_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=1.0,

                alpha_max=300.0,
                eval_interval=1500,

            if score_out < overall_best_score:

                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smooth descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------

    remaining = deadline - time.time()

        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
