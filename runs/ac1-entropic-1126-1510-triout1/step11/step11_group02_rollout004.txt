def smooth_obj(z, alpha):
    # compute a = softmax(z) or better a = exp(z) / sum(exp(z)).
    expz = np.exp(z - np.max(z)) # for numerical stability
    a = expz / expz.sum()
    conv = _fft_self_convolve(a) (size 2n-1)
    shift = conv.max()
    # Actually we could compute w and then f = (1/alpha) * (np.log(np.sum(np.exp(alpha * (conv - shift)))) + alpha*shift)
    # Because log-sum-exp: log Σ exp(alpha*conv) = shift*alpha + log Σ exp(alpha*(conv-shift)).
    # So f = (1/alpha)*(shift*alpha + log Σ exp(alpha*(conv - shift))) = shift + (1/alpha)*log Σ exp(alpha*(conv - shift)).
    # But we can compute direct for numeric stability.
    # Actually we need to compute the objective itself: smooth max approx = shift + log_sum_exp(alpha*(conv - shift))/alpha.
    # Use stable computing.
    cmax = conv.max()
    exps = np.exp(alpha * (conv - cmax))
    w_sum = exps.sum()
    f = cmax + np.log(w_sum) / alpha
    # gradient w.r.t a: w_i normalized
    w = exps / w_sum
    grad_a = _gradient_from_weights(a, w) # shape (n,)
    # gradient w.r.t z: as above
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return f, grad_z

"""
Hybrid L‑BFGS‑B / smooth‑max optimiser for the max‑convolution step‑function.

The algorithm works on a probability vector `a` (sum = 1, a_i ≥ 0)
parameterised by logits `z` (soft‑max).  For a sequence of increasing
soft‑max temperatures `α` we minimise the smooth surrogate

        f_α(z) = (1/α)·log Σ_i exp(α·conv_i(a(z)))

where `conv = a * a` is the full self‑convolution.
The gradient w.r.t. the logits is obtained analytically via the
convolution theorem (FFT) and the Jacobian of the soft‑max.
A continuation scheme (α from 0.5 up to 150) lets us approach the true
`max(conv)` while keeping the optimisation well‑behaved.
After the L‑BFGS stages a cheap greedy pair‑wise mass‑transfer is
performed for final polishing.

The routine respects the 1000 s budget, works for any length ≤ ≈30 000,
and returns a plain list of floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from typing import List, Tuple
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a (length 2·n‑1) using a real FFT.
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of Σ_i w_i·conv_i(a) w.r.t. a.
    The weights `w` must have length 2·n‑1 and sum to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Exact objective (used for bookkeeping) -------------------------------

def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) (the denominator is 1 because a sums to 1).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv

# Analytic candidate families -------------------------------------------

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int,
                                rng: np.random.Generator) -> List[np.ndarray]:
    """Diverse pool of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta family – dense grid of exponents
    for p in np.linspace(0.02, 2.5, 30):
        cand.append(_distribution_beta_sym(n, float(p)))

    # random asymmetric beta draws
    for _ in range(10):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law families
    for p in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential families
    for lam in (0.0005, 0.001, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block spikes
    for block_frac in (0.04, 0.07, 0.10, 0.13):
        cand.append(_distribution_two_blocks(n, block_frac))

    # random Dirichlet draws
    for _ in range(12):
        alpha = 0.1 + 0.9 * rng.random()
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # random sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.008))

    # completely random probability vectors (few)
    for _ in range(6):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand


# ----------------------------------------------------------------------
# Smooth‑max objective / gradient (log‑sum‑exp) -------------------------

def _smooth_objective_and_grad(z: np.ndarray,
                               alpha: float) -> Tuple[float, np.ndarray]:
    """
    Returns (value, gradient) of the smooth surrogate
        f_α(z) = (1/α)·log Σ_i exp(α·conv_i(a(z)))
    where a(z) = softmax(z).  Gradient is w.r.t. the logits `z`.
    """
    # softmax
    z_max = np.max(z)
    expz = np.exp(z - z_max)
    a = expz / expz.sum()

    # full self‑convolution
    conv = _fft_self_convolve(a)

    # stable log‑sum‑exp
    cmax = conv.max()
    exp_term = np.exp(alpha * (conv - cmax))
    w_sum = exp_term.sum()
    if w_sum == 0.0:                     # pathological – fall back to uniform weights
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exp_term / w_sum

    # smooth value
    val = cmax + np.log(w_sum) / alpha

    # gradient w.r.t. a : Σ_i w_i ∂conv_i/∂a
    grad_a = _gradient_from_weights(a, w)

    # chain rule through soft‑max (Jacobian = diag(a) - a·a^T)
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)

    return float(val), grad_z.astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS continuation ---------------------------------------------------

def _l_bfgs_refine(seed_a: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float) -> Tuple[np.ndarray, float]:
    """
    Runs a continuation of L‑BFGS‑B on the smooth surrogate for a single seed.
    The temperature `α` starts small and is increased geometrically.
    Returns the best exact (non‑smooth) sequence seen on this branch.
    """
    eps = 1e-12
    z = np.log(seed_a + eps)          # initialise logits

    # temperature schedule – enough to approach the true max
    alphas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0]

    for alpha in alphas:
        # each stage gets at most 250 L‑BFGS iterations (adjustable)
        max_iter_stage = 250

        # early‑stop callback – abort L‑BFGS when we run out of budget
        def _callback(_z):
            if time.time() >= deadline:
                raise StopIteration()

        # minimise the smooth objective for the current temperature
        try:
            res = minimize(fun=lambda zz: _smooth_objective_and_grad(zz, alpha),
                           x0=z,
                           method='L-BFGS-B',
                           jac=True,
                           callback=_callback,
                           options={'maxiter': max_iter_stage,
                                    'gtol': 1e-12,
                                    'disp': False})
        except StopIteration:
            # out of time – quit the whole refinement
            break

        # keep the last point (whether converged or not)
        z = res.x

        # evaluate the exact (non‑smooth) objective
        expz = np.exp(z - np.max(z))
        a = expz / expz.sum()
        cur_score = _objective(a)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] L‑BFGS α={alpha:.1f} ⇒ new best {best_score:.6f}")

        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise‑swap polishing (identical to the reference implementation) -

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk, only accepting strict improvements."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = int(np.random.choice(src))
        # pick a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search routine that returns a non‑negative step‑function (list of floats)
    minimising `evaluate_sequence`.  It respects the supplied time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.7       # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a single, FFT‑friendly length.
    n = 8192                         # 2ⁿ⁺¹ gives fast FFTs and enough resolution

    # ------------------------------------------------------------------
    # 2. Assemble a broad collection of seed vectors.
    seed_vectors: List[np.ndarray] = _initial_candidates_extended(n, rng)

    # optionally prepend the previously submitted solution (if it exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)    # type: ignore[name-defined]
        if prev.ndim == 1:
            # adapt size if necessary (simple truncation / zero‑padding)
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            elif prev.size > n:
                prev = prev[:n]
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(n)
                s = prev.sum()
            prev = prev / s
            seed_vectors.append(prev.copy())
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Quick exact evaluation of every seed, keep the most promising.
    evaluated: List[Tuple[float, np.ndarray]] = []
    for vec in seed_vectors:
        a = np.maximum(vec, 0.0)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        score = _objective(a)
        evaluated.append((score, a.copy()))
    evaluated.sort(key=lambda x: x[0])
    top_k = min(12, len(evaluated))
    top_seeds = [seq for _, seq in evaluated[:top_k]]

    # initialise the global best
    if top_seeds:
        overall_best_seq = top_seeds[0].copy()
        overall_best_score = _objective(overall_best_seq)
    else:
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    # ------------------------------------------------------------------
    # 4. Allocation of the remaining budget to L‑BFGS refinements.
    while time.time() < deadline and top_seeds:
        # split the remaining time equally among the unfinished seeds
        remaining = deadline - time.time()
        per_seed = remaining / len(top_seeds)
        seq = top_seeds.pop(0)
        sub_deadline = time.time() + per_seed

        overall_best_seq, overall_best_score = _l_bfgs_refine(
            seq,
            sub_deadline,
            overall_best_seq,
            overall_best_score,
        )

    # ------------------------------------------------------------------
    # 5. Final cheap polishing with pairwise swaps (if any time left).
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score (approx.): {overall_best_score:.6f}  length={len(overall_best_seq)}")
    # Convert to plain Python list – this is what the evaluator expects.
    return overall_best_seq.tolist()
