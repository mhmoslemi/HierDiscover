"""
Hybrid search for a non‑negative step function that minimizes
``evaluate_sequence``.

Algorithm
---------
1.  Build a rich library of analytic families (uniform, arcsine,
    symmetric beta, edge‑weighted, semicircle, triangular, cosine,
    Chebyshev/Kaiser windows) on several lengths (2 k – 80 k).
    Mix each shape with the uniform vector for extra flexibility
    and keep the ~50 best candidates (objective = 2·n·max(conv) after
    normalisation Σa = 1).

2.  Refine the top‑few seeds with two complementary gradient methods:
    *Additive sub‑gradient* – a simple Euclidean‑projected gradient
    step based on the (single) maximal convolution entry.
    *Exponential (mirror) descent* – the exponentiated‑gradient update
    from the original baseline.

    The remaining budget is split equally among the seeds.

3.  Final greedy polish: random pairwise mass transfers that are kept
    only when the exact objective improves.

4.  Optional random‑restart hill‑climbing: repeatedly perturb the
    current best vector, run a short additive descent and accept any
    improvement while time permits.

The routine respects the 1000 s budget (a tiny safety margin) and
returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input is assumed non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_semi_circle(n: int) -> np.ndarray:
    """Semicircle (non‑negative) density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = (i - n / 2.0) / (n / 2.0)
    w = np.sqrt(np.maximum(0.0, 1.0 - x * x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    """Triangular (tent) shape."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.minimum(i, n - i)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    """Absolute cosine window (non‑negative)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.abs(np.cos(np.pi * (i / n - 0.5)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination base ⊙ (1‑lam) + uniform ⊙ lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Gradient – additive (Euclidean) descent ---------------------------------
def _additive_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    top_k: int = 1,
    report_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """Projected Euclidean sub‑gradient descent on max‑convolution."""
    n = a.shape[0]
    cur_obj = _objective(a)
    t = 0
    no_improve = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)

        # indices of the k largest entries (k=1 → the maximum)
        if top_k == 1:
            idxs = [int(conv.argmax())]
        else:
            idxs = np.argpartition(-conv, np.arange(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s

        new_obj = _objective(a_new)

        if new_obj < cur_obj - 1e-14:
            a = a_new
            cur_obj = new_obj
            no_improve = 0
            if new_obj < best_score - 1e-14:
                best_score = new_obj
                best_seq = a.copy()
                print(f"[{time.time():.1f}] additive it={t} new global best {best_score:.6f}")
        else:
            no_improve += 1
            # if we have been stuck for a while we can reduce step‑size more aggressively
            if no_improve > 5000:
                # restart a tiny bit to shake out of a flat region
                a = a * np.exp(-eta * grad)  # fall back to multiplicative step
                a = _project_onto_simplex(a)
                cur_obj = _objective(a)

        if t % report_interval == 0:
            print(f"[{time.time():.1f}] additive it={t} cur obj {cur_obj:.6f}")

        if t >= 400000:   # safety cap
            break

    return best_seq, best_score

# Gradient – exponential (mirror) descent (same as baseline) -------------
def _exp_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    top_k: int = 3,
    eval_interval: int = 120,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using exponentiated‑gradient updates.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        idxs = np.argpartition(-conv, np.arange(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp it={t} new global best {best_score:.6f}")

        if t >= 300000:
            break

    return best_seq, best_score

# Greedy pairwise mass transfer -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep it only when objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidate = np.where(a > 1e-12)[0]
        if src_candidate.size == 0:
            break
        i = rng.choice(src_candidate)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.4  # up to 40 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Random‑restart hill climbing -------------------------------------------
def _random_restart_hill(
    best_seq: np.ndarray,
    deadline: float,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Perturb the current best and run a short additive descent."""
    n = best_seq.shape[0]
    rng = np.random.default_rng()
    while time.time() + 5.0 < deadline:
        # multiplicative Gaussian perturbation (keeps positivity)
        noise = rng.normal(loc=0.0, scale=0.08, size=n)
        a = best_seq * np.exp(noise)
        a /= a.sum()
        sub_deadline = time.time() + 4.5
        a_opt, sc_opt = _additive_descent(
            a,
            sub_deadline,
            best_seq,
            best_score,
            lr0=0.4,
            top_k=1,
            report_interval=1000,
        )
        if sc_opt < best_score - 1e-14:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] random restart new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine blends analytic families, two types of sub‑gradient descent,
    greedy polishing and a final random‑restart phase, all within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Analytic families (rich library)
    # ------------------------------------------------------------------
    lengths = [2000, 5000, 10000, 20000, 40000, 80000]
    # p‑grid: extremely fine near zero where optimum tends to live
    p_grid = np.concatenate(
        [
            np.logspace(-4, -2, 6),
            np.logspace(-2, -0.5, 12),
            np.logspace(-0.5, 0.0, 12),
            np.logspace(0.0, 0.7, 10),
        ]
    )
    lambda_grid = np.linspace(0.0, 1.0, 5)  # 0, .25, .5, .75, 1

    # optional SciPy windows
    atten_grid = np.linspace(30, 150, 7) if _HAS_SCI_SIX else []
    beta_grid = np.linspace(0.0, 14.0, 8) if _HAS_SCI_SIX else []

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for base in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parameterised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)

            for base in (beta, edge):
                for lam in lambda_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # additional shapes
        for extra in (
            _distribution_semi_circle(n),
            _distribution_triangle(n),
            _distribution_cosine(n),
        ):
            if time.time() >= deadline:
                break
            sc = _objective(extra)
            top_candidates.append((sc, extra.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = extra.copy()
                print(f"[{time.time():.1f}] new global best extra len={n} score={sc:.6f}")

        # SciPy windows
        if _HAS_SCI_SIX:
            for atten in atten_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_chebyshev(n, atten)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] Cheb len={n} att={atten:.0f} score={sc:.6f}")

            for beta_par in beta_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_kaiser(n, beta_par)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] Kaiser len={n} beta={beta_par:.2f} score={sc:.6f}")

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] random Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = np.zeros(n, dtype=np.float64)
            k = max(2, int(n * 0.01))
            idx = rng.choice(n, size=k, replace=False)
            a[idx] = rng.random(k)
            a /= a.sum()
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the very best seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:50]

    # ------------------------------------------------------------------
    # 2. Incorporate any previously discovered construction (if present)
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback if nothing was generated
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient refinement (additive + exponential) on top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] refining {len(seeds)} seeds, ~{per_seed:.1f}s each")

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed * 0.6
            # additive descent (fast, cheap)
            a_opt, sc_opt = _additive_descent(
                seed_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                top_k=1,
                report_interval=1000,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after additive seed {idx} new global best {overall_best_score:.6f}"
                )
            # exponential (mirror) descent for a finer tweak
            sub_dead = time.time() + per_seed * 0.3
            a_opt, sc_opt = _exp_descent_multi(
                overall_best_seq.copy(),
                sub_dead,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                top_k=3,
                eval_interval=80,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after exp‑descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Greedy pairwise polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise‑swap polishing")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6. Random‑restart hill climbing (if time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] random‑restart hill climbing")
        overall_best_seq, overall_best_score = _random_restart_hill(
            overall_best_seq.copy(),
            deadline,
            overall_best_score,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
