The problem is to find a non‑negative vector `a = (a₀,…,a_{n‑1})` that minimizes  

\[
F(a)=\frac{2\,n\;\max_k (a\star a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a⋆a` is the full self‑convolution.  
Because the objective is homogeneous, we may (and do) keep the vector on the
simplex  

\[
\sum_i a_i = 1,\qquad a_i\ge 0 .
\]

If the convolution were perfectly flat, its maximum would be
\(1/(2n-1)\) and the ideal value of `F` would be `≈1`.  The algorithm
below tries to approach this limit by three stages

1. **Flattening stage** – minimise the squared‑error between the
   convolution and the flat target distribution.
2. **Smooth‑max stage** – minimise a smooth approximation of the true
   max (log‑sum‑exp) with an Adam optimiser.
3. **Local polishing** – cheap greedy mass‑transfer moves that only
   accept improvements.

All steps work on the simplex (Euclidean projection after each update)
and use FFT‑based convolutions, so each iteration costs  
\(O(n\log n)\).  The code tries a handful of sequence lengths
(`4000 … 80000`) and a variety of initial seeds (analytic families,
random Dirichlet draws, sparse patterns, the previously known best
vector).  The whole search stops after the supplied time‑budget
(≤ 1000 s) and returns the best sequence found.

The implementation respects the platform limits (no closures,
no file I/O, only NumPy and the already‑available `evaluate_sequence`
function).

```python
"""
Hybrid optimiser for the max‑convolution step‑function problem.

Three phases:
  1. Flatten the self‑convolution towards a uniform distribution
     (quadratic loss) using Adam on the simplex.
  2. Refine the solution with a smooth‑max (log‑sum‑exp) surrogate,
     also with Adam and a temperature schedule.
  3. Final greedy pairwise mass‑transfer local search.

The routine explores several lengths (up to 80 000) and many random
seeds.  It runs for at most ``budget_s`` seconds and returns the best
list of non‑negative floats found.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    # Sort decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Returns 2·a⋆w where ``w`` may be any real vector of length 2·n‑1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/sum(a)^2 (sum must be >0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families (used as seeds)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _adam_flatten(a: np.ndarray,
                  deadline: float,
                  lr: float = 0.6,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_iter: int = 30000,
                  eval_interval: int = 400) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the squared error
        ‖a⋆a – u‖² ,   u = uniform distribution on {0,…,2n‑2}.
    """
    n = a.shape[0]
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        w = conv - target                       # residual (may be negative)

        grad = 4.0 * _gradient_from_weights(a, w)   # d/da ‖conv‑target‖²

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # Simple step‑size clipping for stability
        max_norm = 0.3
        step_norm = np.linalg.norm(step)
        if step_norm > max_norm:
            step = step * (max_norm / step_norm)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_val - 1e-13:
                best_val = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] flatten it={t} new best {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
def _adam_smoothmax(a: np.ndarray,
                    deadline: float,
                    eta0: float = 0.5,
                    alpha0: float = 1.0,
                    alpha_max: float = 80.0,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8,
                    max_iter: int = 25000,
                    eval_interval: int = 300) -> Tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate
        f_α(a) = (1/α) log Σ_i exp(α·(a⋆a)_i)
    with a geometric temperature schedule.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective(a)

    alpha = alpha0
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum                     # soft‑max weights (sum to 1)

        grad = _gradient_from_weights(a, w)   # ∂f_α/∂a

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        eta = eta0 / math.sqrt(t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # clip very large steps (helps stability on large n)
        max_norm = 0.3
        step_norm = np.linalg.norm(step)
        if step_norm > max_norm:
            step = step * (max_norm / step_norm)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # increase temperature every few hundred steps
        if t % 300 == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.25, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_val - 1e-13:
                best_val = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.2f} new best {best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk on the simplex (accepts only improvements)."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # pick a distinct destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the source mass
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

    if remaining > 30:
        # pick up to five promising seeds (including the current best)
        seed_arrays = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        if overall_best_seq is not None:
            seed_arrays.insert(0, overall_best_seq.copy())

        for seed_seq in seed_arrays:
            if time.time() >= deadline:
                break

            # allocate a slice of the remaining budget to this seed
            per_seed_deadline = time.time() + (deadline - time.time()) / max(1, len(seed_arrays))

            # Phase A – flattening
            seq_a, score_a = _adam_flatten(seed_seq.copy(),
                                            per_seed_deadline,
                                            lr=0.6,
                                            max_iter=25000,
                                            eval_interval=400)
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after flatten new best {overall_best_score:.6f}")

            # Phase B – smooth‑max descent
            seq_b, score_b = _adam_smoothmax(seq_a.copy(),
                                              per_seed_deadline,
                                              eta0=0.5,
                                              max_iter=30000,
                                              eval_interval=300)
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after smoothmax new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final cheap local polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
```
