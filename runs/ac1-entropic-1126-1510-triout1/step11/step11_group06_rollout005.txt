All stages respect the supplied time budget (≈1000 s).  The routine

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT based convolutions -------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution a*b using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
# Scoring (sum of a is 1 → score = 2·n·max(a*a)) -------------------------
def _max_convolution_score(a):
    """Return 2·n·max(a*a) for a normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families – all return a vector with sum = 1 ----------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n, p):
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15                      # avoid exact zeros
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin
    eps = 1e-15

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration – generate a modest set of candidates.
    lengths = [2000, 4000, 8000, 16000, 32000, 64000]

    # for each length we keep the best vector (few families)
    candidates = []   # entries are (vector, n, label, initial_score)

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine
        vec = _distribution_arcsine(n)
        vec = np.maximum(vec, eps); vec /= vec.sum()
        candidates.append((vec, n, "arcsine"))

        # symmetric Beta with a few p values
        for p in (0.2, 0.4, 0.6):
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            candidates.append((vec, n, f"beta_{p:.1f}"))

        # edge‑power laws
        for p in (0.4, 0.6, 0.8):
            vec = _distribution_edge_power(n, p)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            candidates.append((vec, n, f"edge_{p:.1f}"))

        # a quick Dirichlet draw (uniform concentration)
        vec = np.random.dirichlet(np.full(n, 1.0))
        vec = np.maximum(vec, eps); vec /= vec.sum()
        candidates.append((vec, n, "dirichlet"))

    # ------------------------------------------------------------------
    # 1.5️⃣  If a previous solution exists, add it as a candidate.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            candidates.append((prev, prev.size, "previous"))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Compute initial scores and keep the best candidate **per length**.
    best_per_len = {}
    for vec, n, label in candidates:
        if time.time() >= deadline:
            break
        sc = _max_convolution_score(vec)
        # keep smallest score for this length
        if n not in best_per_len or sc < best_per_len[n][0]:
            best_per_len[n] = (sc, vec, label)

    # ------------------------------------------------------------------
    # Create a short list (≈ one per length) to be refined.
    refine_list = []
    for n, (sc, vec, label) in best_per_len.items():
        refine_list.append((sc, vec.copy(), n, label))
    # Sort by initial score – we will give a little more time to the
    # promising lengths but still treat all of them.
    refine_list.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Global best tracking
    best_score = float("inf")
    best_seq = None
    best_n = None

    # ------------------------------------------------------------------
    # 2️⃣  Refine each candidate with smooth‑max mirror descent.
    remaining = deadline - time.time()
    if remaining <= 0:
        # out of time, return the best we have so far (if any)
        if best_seq is not None:
            return best_seq.tolist()
        # fallback to a uniform vector of modest size
        fallback = _distribution_uniform(2000)
        return fallback.tolist()

    # allocate time roughly equally among candidates
    time_per_cand = remaining / max(1, len(refine_list))

    for init_sc, a0, n, label in refine_list:
        if time.time() >= deadline:
            break

        # estimate number of iterations we can afford for this candidate
        approx_iter = int(max(2000, min(35000, (time_per_cand) / 0.004)))
        a = a0.copy()
        best_local_sc = init_sc
        best_local_a = a.copy()

        # schedule parameters
        alpha_start, alpha_end = 1.0, 1e6
        lr_start, lr_end = 0.5, 0.001

        for it in range(approx_iter):
            if time.time() >= deadline:
                break
            # linear interpolation in [0,1]
            t = it / approx_iter
            alpha = alpha_start * (alpha_end / alpha_start) ** t
            lr = lr_start * (lr_end / lr_start) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()

            # stable soft‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the soft‑max approximation
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient (mirror descent) step
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            # bookkeeping
            if it % max(1, approx_iter // 10) == 0 or it == approx_iter - 1:
                cur_sc = _max_convolution_score(a)
                if cur_sc < best_local_sc - 1e-12:
                    best_local_sc = cur_sc
                    best_local_a = a.copy()
                    print(
                        f"[{time.time():.1f}] n={n} {label} it={it}/{approx_iter} "
                        f"new best {cur_sc:.6f}"
                    )

        # ------------------------------------------------------------------
        # 3️⃣  Short Adam‑style sub‑gradient polishing (true max‑conv)
        a_sub = best_local_a.copy()
        m = np.zeros_like(a_sub)
        beta1 = 0.9
        sub_it = 0
        while time.time() < deadline and sub_it < 400:
            sub_it += 1
            conv = _fft_self_convolve(a_sub)
            cur_max = conv.max()
            tol = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(conv >= cur_max - tol)

            w = np.zeros_like(conv)
            w[active] = 1.0
            grad = 2.0 * _fft_convolve(a_sub[::-1], w)[n - 1 : n - 1 + n]

            m = beta1 * m + (1.0 - beta1) * grad
            step = 0.5 / np.sqrt(sub_it)   # decaying step
            a_cand = a_sub - step * m
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()

            cand_sc = _max_convolution_score(a_cand)
            if cand_sc < best_local_sc - 1e-12:
                best_local_sc = cand_sc
                best_local_a = a_cand.copy()
                print(
                    f"[{time.time():.1f}] n={n} {label} subgrad it={sub_it} new best {cand_sc:.6f}"
                )
                a_sub = a_cand
            else:
                a_sub = a_cand

        # Update global best if improved
        if best_local_sc < best_score - 1e-12:
            best_score = best_local_sc
            best_seq = best_local_a.copy()
            best_n = n
            print(
                f"[{time.time():.1f}] Global new best from n={n} {label}: {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 4️⃣  Final stochastic hill‑climbing polish on the overall best.
    if best_seq is None:
        # very unlikely, but keep a fallback
        best_seq = _distribution_uniform(2000)
        best_seq = best_seq / best_seq.sum()
        best_n = best_seq.size
        best_score = _max_convolution_score(best_seq)

    a = best_seq.copy()
    n = a.shape[0]
    hill_it = 0

    while time.time() < deadline:
        hill_it += 1
        # random mass transfer move
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # always move from the larger entry to the smaller one
        if a[i] < a[j]:
            i, j = j, i
        max_delta = min(a[i] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        cand_sc = _max_convolution_score(a2)
        if cand_sc < best_score - 1e-12:
            a = a2
            best_seq = a2.copy()
            best_score = cand_sc
            print(
                f"[{time.time():.1f}] hill‑climb it={hill_it} new best {best_score:.6f}"
            )
        # occasional jitter to escape plateaus
        if hill_it % 4000 == 0:
            a = np.maximum(a + np.random.randn(n) * 1e-5, eps)
            a /= a.sum()

    print(f"Final best score ≈ {best_score:.6f} (length={best_n})")
    return best_seq.tolist()
