"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Broad exploratory phase** – sample several lengths (2k‑20k) and a
   rich set of analytic families (uniform, arcsine, edge‑weighted with
   many p‑values, mixing with the uniform distribution) together with a
   few random Dirichlet / sparse seeds.  The best ~30 candidates are kept.
   A scalar optimisation (SciPy) refines the edge‑weighted exponent p for
   each length.

2. **Smooth mirror descent** – starting from the globally best seed we
   minimise the exact objective 2·n·max(conv(a,a)) on the simplex.
   A *soft‑max* weighting of all convolution entries (temperature β)
   yields a smooth sub‑gradient:
        grad = 2·(a * w_rev)[L‑1‑i]   with w = exp(β·(conv‑max)).
   An exponentiated‑gradient (mirror) update keeps the vector non‑negative,
   after which it is re‑normalised to sum 1.  β grows gradually so the
   method first smooths many indices and later concentrates on the true
   maximum.  Jitter is injected when progress stalls.

3. **Local refinements** – a deterministic pairwise‑swap walk followed by
   a cheap simulated‑annealing style random walk is applied while time
   allows.

All helpers are top‑level; the only entry point is ``propose_candidate``.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_full_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    n, m = len(x), len(y)
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.ifft(X * Y, n=size).real
    return conv[:need]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size).real
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Smooth mirror descent (soft‑max weighted sub‑gradient) -------------
def _smooth_mirror_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    eval_interval: int = 200,
    beta0: float = 5.0,
    beta_factor: float = 1.02,
    beta_max: float = 2000.0,
    patience: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using a soft‑max weighting of all
    convolution entries.  ``a`` is assumed to sum to 1.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    beta = beta0
    rng = np.random.default_rng()

    while time.time() < deadline and t < 200_000:
        t += 1

        conv = _fft_self_convolve(a)
        M = conv.max()

        # soft‑max weights (values ∈ (0,1])
        w = np.exp(beta * (conv - M))

        # gradient = 2·(a * w_rev)[L‑1‑i]
        w_rev = w[::-1]
        conv_rev = _fft_full_convolve(a, w_rev)          # length 3n‑2
        L = len(w)                                        # 2n‑1
        grad = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # exponentiated update with clipping of the exponent to avoid overflow
        eta = lr0 / math.sqrt(t)
        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -50, 50)
        a = a * np.exp(exp_arg)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # increase temperature
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # occasional evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # jitter when stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Pairwise swap local refinement ----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score

# Simulated‑annealing style random walk ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

            if prev.sum() > 0.0:
                prev = prev / prev.sum()
                sc = _objective(prev)
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, prev.copy()
                    print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1.  Broad exploratory phase (many analytic families)
    lengths = [2000, 4000, 8000, 12000, 20000]
    lam_grid = np.linspace(0.0, 0.5, 7)          # 0.0 … 0.5
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ----- uniform baseline -------------------------------------------------
        cand = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best uniform len={n} score={sc:.6f}")

        # ----- arcsine -----------------------------------------------------------
        cand = _distribution_arcsine(n)
        sc = _objective(cand)
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best arcsine len={n} score={sc:.6f}")

        # ----- edge‑weighted coarse grid -----------------------------------------
        p_grid = np.logspace(-2, 1.0, 15)   # 0.01 … 10
        local_best_seq = None
        local_best_score = float("inf")
        for p in p_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_edge_weighted(n, p)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best edge p={p:.4g} len={n} score={sc:.6f}")
            if sc < local_best_score:
                local_best_score, local_best_seq = sc, cand.copy()

        # ----- edge‑weighted scalar optimisation ---------------------------------
        if time.time() < deadline:
            try:
                def edge_obj(p: float) -> float:
                    if p <= 0.0:
                        return np.inf
                    cand = _distribution_edge_weighted(n, p)
                    return _objective(cand)

                res = opt.minimize_scalar(
                    edge_obj,
                    bounds=(1e-3, 5.0),
                    method="bounded",
                    options={"maxiter": 30},
                )
                if res.success:
                    p_opt = res.x
                    cand = _distribution_edge_weighted(n, p_opt)
                    sc = _objective(cand)
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(f"[{time.time():.1f}] new global best edge opt p={p_opt:.6g} len={n} score={sc:.6f}")
                    if sc < local_best_score:
                        local_best_score, local_best_seq = sc, cand.copy()
            except Exception:
                pass

        # ----- mixing with uniform (if we have a local best) --------------------
        if local_best_seq is not None:
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(local_best_seq, lam)
                sc = _objective(cand)
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(f"[{time.time():.1f}] new global best mix lam={lam:.2f} len={n} score={sc:.6f}")

        # ----- random Dirichlet seeds -------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # ----- sparse random seeds -----------------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(a)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2.  Fallback in the unlikely case that nothing was found
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Smooth mirror descent on the current global best
    remaining = deadline - time.time()
    if remaining > 30.0 and overall_best_seq is not None:
        print(f"[{time.time():.1f}] starting smooth mirror descent (len={len(overall_best_seq)})")
        a_opt, sc_opt = _smooth_mirror_descent(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq.copy(),
            overall_best_score,
            lr0=0.8,
            eval_interval=200,
            beta0=5.0,
            beta_factor=1.02,
            beta_max=2000.0,
            patience=2000,
        )
        if sc_opt < overall_best_score:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after descent new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Pairwise swap local refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise swap refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq.copy(),
            overall_best_score,
            max_trials=300_000,
        )

    # ------------------------------------------------------------------
    # 5.  Final cheap simulated‑annealing style search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq.copy(),
            overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will perform its own checks.
    return overall_best_seq.tolist()
