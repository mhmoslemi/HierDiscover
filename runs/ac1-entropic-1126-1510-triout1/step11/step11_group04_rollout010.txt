def _hill_climb_pair(a, deadline):
    n = len(a)
    best = a.copy()
    best_score = _objective(a)
    # Precompute conv
    conv = _fft_self_convolve(a)
    M = conv.max()
    stagnation = 0
    while time.time() < deadline:
        # Find max pair indexes
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # Find pair with max product
        best_pair = None
        max_prod = -1
        for i in range(i_low, i_high+1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_pair = (i, j)
        i, j = best_pair
        # Choose two target indices distinct from i,j
        target1 = random.randint(0, n-1)
        while target1 == i or target1 == j:
            target1 = random.randint(0, n-1)
        target2 = random.randint(0, n-1)
        while target2 == i or target2 == j or target2 == target1:
            target2 = random.randint(0, n-1)
        # Transfer small amount from i and j to targets
        delta_i = min(a[i], 0.02) * random.random() * 0.5
        delta_j = min(a[j], 0.02) * random.random() * 0.5
        a2 = a.copy()
        a2[i] -= delta_i
        a2[j] -= delta_j
        a2[target1] += delta_i
        a2[target2] += delta_j
        # Normalize (should already sum to 1)
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-14:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            sc = _objective(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                stagnation = 0
        else:
            stagnation += 1
        if stagnation > 2000:
            # random perturbation
            idxs = np.random.choice(n, size=5, replace=False)
            perturb = np.random.randn(5) * 1e-4
            a[idxs] += perturb
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            stagnation = 0
    return best, best_score

2. **Enhanced mirror descent** – a softened max‑convolution is minimised by
   exponentiated‑gradient updates.  The temperature grows from 1 up to 1e9
   while the learning‑rate decays, allowing global exploration followed by
   fine‑grained local tuning.

3. **Long sub‑gradient polishing** – true sub‑gradient steps on the exact
   max are performed with an Adam‑style momentum and a back‑tracking line
   search.  The loop runs until the time budget is almost exhausted.

4. **Aggressive hill‑climbing & pruning** – repeatedly reduces the dominant
   product ‑ a_i·a_j that creates the current maximal convolution term by
   moving a small amount of mass from both i and j to random locations.
   Tiny entries are periodically pruned to keep the support sparse.

The routine respects the 1000 s budget, prints intermediate scores and
returns the best sequence found.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers -------------------------------------------------

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Returns 2·n·max(conv) for a vector a that already satisfies sum(a)=1.
    The scaling is irrelevant because the true objective is homogeneous.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families – each returns a vector with sum = 1 -----------------

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Candidate generator ----------------------------------------------------

def _generate_candidates_for_length(n):
    """Return a list of candidate vectors (sum = 1) for a given length."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta
    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta
    for p, q in ((0.2, 1.0), (1.0, 0.2), (0.5, 1.5), (1.5, 0.5)):
        cand.append(_distribution_beta_asym(n, p, q))

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws with various concentrations
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Sparse random start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Shuffled versions of a few bases to add asymmetry
    for base in (cand[0], cand[1], cand[4]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Local optimisation utilities -------------------------------------------

def _smooth_max_mirror_descent(a, deadline):
    """
    Exponentiated‑gradient mirror descent on the smooth‑max version.
    The temperature α grows to 1e9 and the learning‑rate decays.
    """
    n = a.shape[0]
    eps = 1e-12

    # Extended schedule (more phases, higher final temperature)
    schedule = [
        (1.0, 1e2, 12000, 1.0, 0.2),
        (1e2, 1e4, 12000, 0.2, 0.04),
        (1e4, 1e6, 12000, 0.04, 0.008),
        (1e6, 1e8, 12000, 0.008, 0.0016),
        (1e8, 1e9, 12000, 0.0016, 0.00032),
    ]

    best_a = a.copy()
    best_sc = _objective(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # stabilized soft‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed max
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

            # exponentiated‑gradient update (mirror descent)
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 3000 == 0:
                sc = _objective(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a, deadline, max_iters=None):
    """
    True sub‑gradient descent with Adam‑style momentum and back‑tracking.
    Runs until the deadline (or a supplied iteration cap) is reached.
    """
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_step = 0.4
    it = 0

    best_a = a.copy()
    best_sc = _objective(best_a)

    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # Adam momentum update
        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_step
        # back‑tracking line search
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_and_prune(a, deadline):
    """
    Aggressive hill‑climber that reduces the dominant product a_i·a_j,
    occasionally prunes tiny entries and injects jitter.
    """
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- locate the pair (i,j) that gives the maximal convolution entry -----
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]

        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # ----- move a small amount from both i and j to two random locations -----
        delta_i = min(a[best_i] * 0.5, 0.02) * random.random()
        delta_j = min(a[best_j] * 0.5, 0.02) * random.random()
        if delta_i > eps or delta_j > eps:
            a2 = a.copy()
            a2[best_i] -= delta_i
            a2[best_j] -= delta_j

            # choose two distinct random targets (could coincide with each other)
            tgt1 = random.randint(0, n - 1)
            while tgt1 == best_i or tgt1 == best_j:
                tgt1 = random.randint(0, n - 1)
            tgt2 = random.randint(0, n - 1)
            while tgt2 == best_i or tgt2 == best_j or tgt2 == tgt1:
                tgt2 = random.randint(0, n - 1)

            a2[tgt1] += delta_i
            a2[tgt2] += delta_j
            a2 = np.maximum(a2, eps)
            a2 /= a2.sum()

            new_M = _fft_self_convolve(a2).max()
            if new_M < M - 1e-14:
                a = a2
                conv = _fft_self_convolve(a)
                M = new_M
                sc = _objective(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    if it % 200 == 0:
                        print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ----- pruning of very small entries every few thousand steps -----
        if it % 3000 == 0:
            thresh = 1e-6
            mask = a > thresh
            if mask.sum() < n and mask.sum() >= 2:
                a = a * mask
                a = np.maximum(a, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
                M = conv.max()
                sc = _objective(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Prune it={it} new best {best_sc:.6f}")

        # ----- occasional jitter to keep the dynamics alive -----
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin
    eps = 1e-12

    best_seq = None
    best_score = float("inf")
    elite = []                                 # (score, vector)

    # ------------------------------------------------------------------
    # 0. try to reuse a previous construction, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            elite.append((sc, prev.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Loaded previous best score {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1. Broad exploration (keep a tiny elite)
    lengths = [2000, 3000, 4000, 6000, 8000, 12000, 16000]
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score - 1e-12:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")
            # maintain elite of size ≤ 10
            if len(elite) < 10:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, vec.copy())
                elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. If elite is empty, fall back to a uniform vector
    if not elite:
        n0 = 4000
        v0 = _distribution_uniform(n0)
        sc = _objective(v0)
        elite.append((sc, v0.copy()))
        best_seq = v0.copy()
        best_score = sc
        print(f"[{time.time():.1f}] Fallback uniform score {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. Local optimisation from the elite (up to 5 restarts)
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx + 1}/{restarts} length={n0} start score {elite[idx][0]:.6f}")

        # ---- smooth‑max mirror descent with extended schedule ----
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # ---- projected sub‑gradient polishing (run as long as time allows) ----
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # ---- final aggressive hill‑climber + pruning ----
        a_mid, sc_mid = _hill_climb_and_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 4. final sanitisation
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
