"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

The algorithm works in three stages:

1. **Broad exploration** – generate many analytic families (uniform,
   symmetric Beta(p,p) for a dense grid of p, mixtures with the uniform,
   random Dirichlet draws, sparse starters) for several lengths.
   Keep the best few candidates (the “elite”).

2. **LP‑based refinement** – for each elite vector solve a linear program
   that maximises the total mass of a new vector `b` under the constraint
   `f * b ≤ M` (where `M` is the current maximal self‑convolution of `f`).
   Blend `f` with the normalised `b` (line‑search on the blending weight)
   and repeat a few times.  This step is the discrete analogue of the
   LP described in the statement and often reduces the peak of the
   convolution appreciably.

3. **Polishing** – run a longer mirror‑descent on a smooth‑max surrogate,
   a projected sub‑gradient descent and a final hill‑climber/pruner
   (identical to the previous state‑of‑the‑art code).  The best vector
   found across all restarts is returned.

All heavy loops respect a deadline (`budget_s` seconds, minus a safety
margin).  The code uses only NumPy, SciPy and the standard library and
produces a plain Python list of floats.
"""

import time
import random
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow/underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform (both already normalised)."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

# Candidate generator ----------------------------------------------------
def _generate_candidates(n, rng):
    """Return a list of normalised candidate vectors for a given length."""
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))   # arcsine

    # finer grid of symmetric betas
    p_grid = np.concatenate((
        np.logspace(-3, -0.3, 15),   # very spiky end
        np.linspace(0.05, 2.0, 30)   # medium range
    ))
    for p in p_grid:
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with the uniform distribution
    for p in (0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.1, 0.3, 0.5, 0.7, 0.9):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # several Dirichlet draws
    for alpha in (0.02, 0.05, 0.1, 0.3, 0.8, 2.0):
        for _ in range(3):
            cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start (few non‑zeros)
    k = max(2, n // 500)
    cand.append(_distribution_sparse_random(n, k))

    # random permutations of a few bases to introduce asymmetry
    base_choices = [cand[0], cand[2], cand[-3]]
    for base in base_choices:
        sh = base.copy()
        rng.shuffle(sh)
        cand.append(sh)

    return cand

# Local optimisation ----------------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """Exponentiated‑gradient (mirror) descent on a smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # schedule - more steps than the previous script
    schedule = [
        (1.0, 1e3, 4000, 0.15, 0.015),
        (1e3, 1e5, 4000, 0.015, 0.0015),
        (1e5, 1e6, 6000, 0.0015, 0.00015),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline):
    """True sub‑gradient descent with Adam‑style momentum and back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a0, deadline):
    """Hill‑climbing with aggressive pruning of tiny entries."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # move mass away from the current maximal entry
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j

        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # aggressive pruning every few thousand steps
        if it % 2000 == 0:
            for keep_frac in (0.97, 0.94, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# LP‑based refinement ----------------------------------------------------
def _lp_refine(a0, deadline, max_iter=3):
    """Iteratively maximise total mass of a vector b under the constraint
    f * b ≤ M (M = max convolution of f).  Blend the result back into f."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current bound M
        conv = _fft_self_convolve(a)
        M = conv.max()

        # Build convolution matrix A (shape (2n‑1, n))
        # A_{k,j} = a_{k-j} for k‑j in [0,n‑1], else 0.
        col = np.concatenate([a, np.zeros(n - 1)])
        row = np.concatenate([a[:1], np.zeros(n - 1)])
        A = toeplitz(col, row)   # dense, fine for n ≤ ~2000

        # Linear programme: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            # If the LP fails, abort the refinement.
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        b_norm = b / sum_b

        # line search on blending weight t
        best_local_a = a.copy()
        best_local_sc = best_sc
        for t in (0.1, 0.2, 0.3, 0.5, 0.7, 0.9):
            cand = (1.0 - t) * a + t * b_norm
            cand = _project_onto_simplex(cand, 1.0)
            sc = _max_convolution_score(cand)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local_a = cand.copy()

        if best_local_sc < best_sc - 1e-12:
            a = best_local_a
            best_sc = best_local_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP refine it={it} new best {best_sc:.6f}")
        else:
            # no improvement – stop early
            break

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []               # (score, vector)

    # ------------------------------------------------------------------
    # 1. Broad exploration -------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]   # powers of two – FFT friendly

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates(n, rng)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")

            # keep a small elite set (≤6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Include any historic solution (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, prev.copy())
                elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never trigger)
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement of elite candidates
    restarts = min(4, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) init score {elite[idx][0]:.6f}")

        # ----- LP based refinement (may be skipped if n too large) -----
        if n0 <= 2000:
            a0, _ = _lp_refine(a0, deadline, max_iter=3)

        # ----- mirror‑descent smoothing ---------------------------------
        a0, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # ----- sub‑gradient polishing -----------------------------------
        a0, sc_mid = _projected_subgradient(a0, deadline)

        # ----- final hill‑climber / pruning ------------------------------
        a0, sc_mid = _hill_climb_prune(a0, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a0.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity, normalise and report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
