def propose_candidate(seed=42,budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    start=time.time()
    deadline=start+budget_s-2
    best_seq=None
    best_score=inf

    # step 1: broad search
    lengths = [4096,8192,16384,32768]
    elite = []  # list of (score, vec)
    for n in lengths:
        if time.time()>=deadline: break
        candidates = _generate_initial_candidates(n)
        for v in candidates:
            if time.time()>=deadline: break
            v = np.maximum(v, 1e-12)
            v/=v.sum()
            sc = _objective(v)
            # update global best
            if sc<best_score:
                best_score=sc
                best_seq=v.copy()
                print(f"[{time.time():.1f}] n={n} new global best {sc:.6f}")
            # maintain elite of fixed size 8
            if len(elite)<8:
                elite.append((sc,v.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc<elite[-1][0]:
                elite[-1]=(sc,v.copy())
                elite.sort(key=lambda x: x[0])
    # incorporate previous best if exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim==1 and prev.size>0:
            prev = np.maximum(prev,1e-12)
            prev/=prev.sum()
            sc=_objective(prev)
            if sc<best_score:
                best_score=sc
                best_seq=prev.copy()
                print(f"[{time.time():.1f}] using previous best {sc:.6f}")
            if len(elite)<8:
                elite.append((sc,prev.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc<elite[-1][0]:
                elite[-1]=(sc,prev.copy())
                elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # local optimization from elite
    max_restarts=min(5,len(elite))
    for idx in range(max_restarts):
        if time.time()>=deadline: break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Restart {idx+1}/{max_restarts} len={len(a0)} start score {elite[idx][0]:.6f}")
        # Mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline, max_iters=20000)
        # Subgradient
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline, max_iters=4000)
        # Hill climbing
        a_mid, sc_mid = _hill_climb_and_prune(a_mid, deadline, max_iter=5000)
        if sc_mid<best_score-1e-12:
            best_score=sc_mid
            best_seq=a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # final sanitization
    best_seq = np.maximum(best_seq, 1e-12)
    best_seq/=best_seq.sum()
    # final check
    final_score=_objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} length={best_seq.shape[0]}")
    return best_seq.tolist()

def _generate_initial_candidates(n):
    # produce list of np.ndarray vectors sum=1
    cand = []

    # Uniform
    cand.append(_distribution_uniform(n))

    # arcsine
    cand.append(_distribution_arcsine(n))

    # triangular
    cand.append(_distribution_triangular(n))

    # exponential
    for lam in (0.001,0.005,0.02,0.05):
        cand.append(_distribution_exponential(n, lam))

    # symmetric Beta fine grid
    for p in np.linspace(0.02,0.3,15):  # about 15 values
        cand.append(_distribution_beta_sym(n, p))

    # extreme Beta
    for p in (0.01,0.015,0.025):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta combos
    for p,q in [(0.2,0.8),(0.8,0.2),(0.5,1.2),(1.2,0.5)]:
        cand.append(_distribution_beta_asym(n,p,q))

    # mixture random
    for _ in range(5):
        p1 = random.uniform(0.01,0.3)
        p2 = random.uniform(0.01,0.3)
        lam = random.random()
        v1 = _distribution_beta_sym(n,p1)
        v2 = _distribution_beta_sym(n,p2)
        mix = lam * v1 + (1 - lam) * v2
        mix /= mix.sum()
        cand.append(mix)

    # random Dirichlet
    for alpha in (0.2,0.5,1.0,2.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # sparse random start
    k = max(2, n // 500)
    spars = np.zeros(n)
    idx = np.random.choice(n, k, replace=False)
    spars[idx] = np.random.rand(k)
    spars /= spars.sum()
    cand.append(spars)

    # noise-infused uniform
    u = _distribution_uniform(n)
    noise = np.random.randn(n) * 0.02
    u2 = np.maximum(u + noise, 0)
    u2 /= u2.sum()
    cand.append(u2)

    return cand

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Broad exploration** – generate a large variety of analytic families
   (uniform, arcsine, triangular, exponential, symmetric and asymmetric
   Beta, mixtures of two Betas, Dirichlet draws, sparse starts).  For a
   selection of lengths (powers of two up to 32768) the best few candidates
   are kept in an elite set.

2. **Smooth‑max mirror descent** – minimise a smooth approximation of the
   infinity‑norm of the auto‑convolution:
   M_α(a)=log‑sum‑exp(α·(a∗a))/α.  The update is an exponentiated‑gradient
   step on the probability simplex (mirror descent).  The temperature
   ``α`` is increased exponentially from 1 to 1e8 while the learning rate
   decays.  The best iterate is stored.

3. **Projected sub‑gradient polishing** – using the exact max‑convolution
   a true sub‑gradient step with Adam‑style momentum and a back‑tracking
   line‑search refines the solution.

4. **Hill‑climbing & pruning** – local mass‑transfers from the pair that
   creates the current maximal convolution term, occasional pruning of
   negligible entries and jitter keep the search alive.

5. **Multiple restarts** – the elite candidates are used as starting points

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:

    else:

    return vals

    return vals

    """Symmetric Beta(p,p) on the discrete grid, stable for tiny p."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    max_log = np.max(logw)
    w = np.exp(logw - max_log)
    s = w.sum()
    if s == 0 or not np.isfinite(s):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w

            prev /= prev.sum()
            previous = prev
    except Exception:
        pass

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_initial_candidates(n, previous)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain elite (max size 8)
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, vec.copy())
                elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate the previous best (if any) – already done in step‑1
    # ------------------------------------------------------------------
    # Fallback in the unlikely case nothing was generated
    if best_seq is None:
        best_n = 4096
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local optimisation from the elite (up to 5 restarts)
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} "
              f"len={a0.shape[0]} start score {elite[idx][0]:.6f}")

        # --- smooth‑max mirror descent ---
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline, max_iters=20000)

        # --- projected sub‑gradient polishing ---
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline, max_iters=5000)

        # --- hill‑climbing & pruning ---
        a_mid, sc_mid = _hill_climb_and_prune(a_mid, deadline, max_iter=5000)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 4. Final sanitisation
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
