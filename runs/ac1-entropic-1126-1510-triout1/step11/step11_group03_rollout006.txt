def _simulated_anneal_polish(a, deadline, cur_best_score):
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = cur_best_score
    start = time.time()
    while time.time() < deadline:
        # compute fraction elapsed
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total
        # temperature schedule decays linearly from 0.01 to 0
        T = max(1e-4, 0.02 * (1 - frac))
        # choose indices
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.6  # up to 60% of source mass
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # keep sum unchanged
        score = _objective(new_a)
        if score < best_score:
            best_score = score
            best_a = new_a.copy()
            a = new_a
            print(f"[{time.time():.1f}] SA move i->{j} δ={delta:.2g} new best {best_score:.6f}")
            continue
        # acceptance probability
        if T > 0 and rng.random() < math.exp(-(score - best_score) / T):
            a = new_a
    return best_a, best_score

"""
Hybrid analytic‑seed → Adam on the exact max‑convolution sub‑gradient →
temperature‑controlled simulated‑annealing polish.

The algorithm proceeds in three phases:

1. **Seed pool** – A diverse collection of analytic distributions
   (uniform, arcsine, Beta, power‑law, exponential, sparse, two‑block,
   Dirichlet) for a few lengths.  All seeds are normalised to sum = 1
   (the objective is homogeneous, so any scale works).  The best few
   seeds are kept for the intensive optimisation.

2. **Exact‑max Adam** – For each seed we minimise
   ``2·n·max(conv(a,a))/sum(a)²`` (with ``sum(a)=1``) using Adam on the
   log‑variables ``x = log(a)``.  The gradient is obtained from the
   sub‑gradient of the *maximum* convolution entry:
   ``w`` is the uniform weight over all indices attaining the current
   maximum and ``grad_a = 2·a⋆w`` (cross‑correlation via FFT).  Adam
   updates are performed on ``x`` and the simplex constraint is enforced
   by the soft‑max (exponentiate → normalise).  The best sequence found
   during the run is retained.

3. **Simulated‑annealing polish** – A cheap stochastic walk moves a
   random fraction of mass from one index to another and accepts the
   move if it improves the true (non‑smoothed) objective or with a
   probability depending on a linearly decreasing temperature.  This
   often escapes the tiny “plateaus’’ left by the gradient stage.

The whole procedure respects the global time budget (default 1000 s)
and prints occasional progress messages.  The returned list of floats
is ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2·n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Analytic distributions ------------------------------------------------

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) *
               np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) *
               np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int,
                                rng: np.random.Generator,
                                density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int,
                            rng: np.random.Generator,
                            alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int,
                                 rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # a few Dirichlet draws with random α
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))
    return cand


# ----------------------------------------------------------------------
# Adam optimiser on the exact max (sub‑gradient) -------------------------

def _adam_optimize(a_init: np.ndarray,
                   deadline: float,
                   max_iter: int = 25000) -> tuple[np.ndarray, float]:
    """
    Adam on log‑variables, using the exact max‑convolution sub‑gradient.
    Returns the best sequence seen (sum=1) and its objective value.
    """
    a = a_init.copy()
    eps = 1e-12
    x = np.log(a + eps)                     # start in log‑space
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    eps_adam = 1e-8

    best_a = a.copy()
    best_score = _objective(a)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # soft‑max to obtain a on the simplex (stable)
        shift = x.max()
        exp_x = np.exp(x - shift)
        a = exp_x / exp_x.sum()

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # uniform weight over all maximising indices (exact sub‑gradient)
        mask = np.isclose(conv, max_val, atol=1e-12)
        if not mask.any():
            continue
        w = mask.astype(np.float64) / mask.sum()

        grad_a = _gradient_from_weights(a, w)          # 2·a⋆w
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)                    # chain rule through soft‑max

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # periodic exact evaluation
        if t % 200 == 0:
            cur_score = _objective(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Adam iter={t} len={len(a)} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polish (cheap mass‑transfer walk) -----------------

def _simulated_anneal_polish(a: np.ndarray,
                            deadline: float,
                            cur_best_score: float) -> tuple[np.ndarray, float]:
    """
    Randomly transfers a fraction of mass between positions.
    Accepts worsening moves with a temperature that cools linearly.
    Returns the best sequence seen.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    best_a = a.copy()
    best_score = cur_best_score
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total
        # temperature tapers from ~0.02 to ~0.0001
        T = max(1e-4, 0.02 * (1 - frac))

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        # move up to 60 % of the source mass
        delta = a[i] * rng.random() * 0.60
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur_score = _objective(new_a)

        if cur_score < best_score:
            best_score = cur_score
            best_a = new_a.copy()
            a = new_a
            print(f"[{time.time():.1f}] SA move {i}->{j} δ={delta:.2g} new best {best_score:.6f}")
            continue

        # accept worse moves with probability exp(-Δ/T)
        if T > 0 and rng.random() < math.exp(-(cur_score - best_score) / T):
            a = new_a

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths (larger = more freedom, but slower)
    lengths = [3000, 5000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    candidate_pool = []  # (score, seq)

    # ------------------------------------------------------------------
    # 2. Generate analytic candidates for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- generating seeds, length={n} ---")
        seeds = _initial_candidates_extended(n, rng)

        # add a few pure Dirichlet draws for extra diversity
        for _ in range(3):
            seeds.append(_distribution_dirichlet(n, rng,
                                                 alpha=0.3 + 0.7 * rng.random()))

        for seq in seeds:
            if time.time() >= deadline:
                break
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s == 0.0:
                continue
            seq = seq / s                     # enforce sum = 1  (scale‑invariant)
            sc = _objective(seq)
            candidate_pool.append((sc, seq.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = seq.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")

    # keep only a small elite set (across all lengths)
    candidate_pool.sort(key=lambda x: x[0])
    elite = candidate_pool[:8]

    # ------------------------------------------------------------------
    # 3. Incorporate the previous run if the variable exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved to {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:8]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Adam optimisation on the elite seeds
    remaining = deadline - time.time()
    if remaining > 20.0 and elite:
        per_seed_budget = remaining / len(elite)
        for seed_score, seed_seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_budget
            opt_seq, opt_score = _adam_optimize(seed_seq.copy(), sub_deadline)
            if opt_score < overall_best_score:
                overall_best_score = opt_score
                overall_best_seq = opt_seq.copy()
                print(f"[{time.time():.1f}] after Adam new best score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing with a simulated‑annealing walk
    if time.time() < deadline and overall_best_seq is not None:
        overall_best_seq, overall_best_score = _simulated_anneal_polish(
            overall_best_seq.copy(),
            deadline,
            overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6. Return the best sequence (as plain python list)
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
