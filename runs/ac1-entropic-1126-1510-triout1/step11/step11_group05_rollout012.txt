"""
Advanced search for a non‑negative step function minimizing evaluate_sequence.

The algorithm combines a richer pool of analytic families (including asymmetric
Beta, exponential and power‑law shapes), a broader set of lengths (up to 32768),
and an extended local optimisation phase (smooth‑max mirror descent,
projected sub‑gradient descent, and a random‑move hill‑climber).  All loops are
time‑aware and respect the 1000 s budget.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_beta_asym(n, alpha, beta):
    """Asymmetric Beta(alpha,beta) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_arcsine(n):
    return _distribution_beta_sym(n, 0.5)

def _distribution_exponential(n, lam=1.0, reverse=False):
    """a_i ∝ exp(-lam * i) (or reversed)."""
    i = np.arange(n, dtype=np.float64)
    if reverse:
        i = n - 1 - i
    w = np.exp(-lam * i)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = _distribution_uniform(n)
    else:
        w /= w_sum
    return w

def _distribution_power_law(n, p):
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = _distribution_uniform(n)
    else:
        w /= w_sum
    return w

def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec

def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = _distribution_uniform(n)
    else:
        a /= a_sum
    return a

def _mix_vectors(v1, v2, w):
    """Return w·v1+(1‑w)·v2, re‑normalised."""
    mix = w * v1 + (1.0 - w) * v2
    mix = np.maximum(mix, 0.0)
    s = mix.sum()
    if s == 0.0:
        mix = _distribution_uniform(len(v1))
    else:
        mix /= s
    return mix

# Candidate generation ----------------------------------------------------
def _generate_candidates(n):
    """Return a list of candidate vectors (already normalised to sum = 1)."""
    cand = []

    # Basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric beta grid
    for p in np.concatenate(
        (np.logspace(-2, -0.2, 12), np.linspace(0.1, 2.5, 20))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Asymmetric beta families (left‑heavy and right‑heavy)
    for alpha, beta in [(0.1, 1.0), (0.2, 1.0), (0.5, 1.5), (0.5, 2.0), (1.5, 0.5), (2.0, 0.5), (2.0, 1.0)]:
        cand.append(_distribution_beta_asym(n, alpha, beta))
        cand.append(_distribution_beta_asym(n, beta, alpha))

    # Exponential tails
    for lam in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_exponential(n, lam=lam, reverse=False))
        cand.append(_distribution_exponential(n, lam=lam, reverse=True))

    # Power‑law decays
    for p in (0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_power_law(n, p))

    # Random Dirichlet draws (different concentration)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # Sparse start
    k = max(2, n // 500)
    cand.append(_distribution_sparse_random(n, k))

    # Mixtures of a beta with uniform
    for p in (0.1, 0.2, 0.5, 0.8):
        beta = _distribution_beta_sym(n, p)
        uniform = _distribution_uniform(n)
        for w in (0.2, 0.5, 0.8):
            cand.append(_mix_vectors(beta, uniform, w))

    # Shuffled versions of a few bases (to break symmetry)
    for base in (cand[0], cand[2], cand[-3]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Local optimisation ----------------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """
    Exponentiated‑gradient (mirror) descent on a smooth‑max surrogate.
    The temperature α is ramped from 1 to 1e7 while the learning rate
    decays; the schedule is deliberately long to give the optimizer time
    to escape shallow plateaux.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # schedule: (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e3, 4000, 0.2, 0.02),
        (1e3, 1e5, 4000, 0.02, 0.002),
        (1e5, 1e7, 5000, 0.002, 0.0002),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # soft‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed max
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline, max_iters=40000):
    """
    Classical sub‑gradient descent with Adam‑style momentum and back‑tracking.
    The loop stops at the deadline or after max_iters iterations.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active indices (within tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search on the true objective
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-15 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-15:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter to escape plateaus
        if it % 1200 == 0:
            jitter = np.random.randn(n) * 5e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        # break early if only little time left
        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _random_hill_climb(a0, deadline, max_iters=8000):
    """
    Very cheap stochastic hill‑climber that moves a small amount of mass
    between two random positions and accepts improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        src = random.randrange(n)
        if a[src] < eps:
            continue
        tgt = random.randrange(n - 1)
        if tgt >= src:
            tgt += 1
        # propose a transfer that is at most a fraction of the source mass
        max_delta = min(a[src] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        sc = _max_convolution_score(a2)

        if sc < best_sc - 1e-15:
            best_sc = sc
            a = a2
            best_a = a2.copy()
            if it % 1000 == 0:
                print(f"[{time.time():.1f}] hill it={it} new best {sc:.6f}")

        # every few thousand iterations try a pruning step (remove tiny entries)
        if it % 3000 == 0:
            # keep only the biggest 95 % of entries (by count)
            keep = max(2, int(0.95 * n))
            idx = np.argpartition(a, -keep)[-keep:]
            a_pruned = np.zeros_like(a)
            a_pruned[idx] = a[idx]
            a_pruned /= a_pruned.sum()
            sc_pruned = _max_convolution_score(a_pruned)
            if sc_pruned < best_sc - 1e-15:
                a = a_pruned
                best_a = a.copy()
                best_sc = sc_pruned
                print(f"[{time.time():.1f}] prune it={it} new best {sc_pruned:.6f}")

    return best_a, best_sc

"""
Advanced search for a non‑negative step function minimising `evaluate_sequence`.

The algorithm expands the candidate pool (asymmetric Beta, exponentials,
power‑law and common window shapes), explores a broader set of
lengths (up to 32768) and performs a longer local optimisation phase:
smooth‑max mirror descent → Adam‑style projected sub‑gradient → a cheap
random/targeted hill‑climber with occasional pruning.  All loops respect
the 1000 s budget.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, alpha, beta):
    """Asymmetric Beta(alpha,beta) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_arcsine(n):
    return _distribution_beta_sym(n, 0.5)


def _distribution_exponential(n, lam=1.0, reverse=False):
    """a_i ∝ exp(-lam·i) (or reversed)."""
    i = np.arange(n, dtype=np.float64)
    if reverse:
        i = n - 1 - i
    w = np.exp(-lam * i)
    s = w.sum()
    if s == 0.0:
        w = _distribution_uniform(n)
    else:
        w /= s
    return w


def _distribution_power_law(n, p):
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    s = w.sum()
    if s == 0.0:
        w = _distribution_uniform(n)
    else:
        w /= s
    return w


def _distribution_power_law_offset(n, p, offset):
    """a_i ∝ (i+offset)^{-p}."""
    i = np.arange(n, dtype=np.float64) + offset
    w = i ** (-p)
    s = w.sum()
    if s == 0.0:
        w = _distribution_uniform(n)
    else:
        w /= s
    return w


def _distribution_hann(n):
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_hamming(n):
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_blackman(n):
    i = np.arange(n, dtype=np.float64)
    w = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    w /= s
    return w


def _distribution_random_dirichlet(n, alpha):
    return np.random.dirichlet(np.full(n, alpha))


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    s = a.sum()
    if s == 0.0:
        return _distribution_uniform(n)
    a /= s
    return a


def _mix_vectors(v1, v2, w):
    """Return w·v1 + (1‑w)·v2 and renormalise."""
    mix = w * v1 + (1.0 - w) * v2
    mix = np.maximum(mix, 0.0)
    s = mix.sum()
    if s == 0.0:
        mix = _distribution_uniform(len(v1))
    else:
        mix /= s
    return mix

# Candidate generation ----------------------------------------------------
def _generate_candidates(n):
    """Return a list of normalised candidate vectors for length n."""
    cand = []

    # 1. Simple families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # 2. Symmetric beta grid (p from 0.01 to 2.5)
    for p in np.concatenate((np.logspace(-2, -0.2, 12), np.linspace(0.1, 2.5, 20))):
        cand.append(_distribution_beta_sym(n, float(p)))

    # 3. Asymmetric beta families (left‑heavy / right‑heavy)
    asym_params = [
        (0.1, 1.0), (0.2, 1.0), (0.5, 1.5), (0.5, 2.0),
        (1.0, 0.1), (1.0, 0.2), (1.5, 0.5), (2.0, 0.5),
        (2.0, 1.0), (3.0, 0.5)
    ]
    for a, b in asym_params:
        cand.append(_distribution_beta_asym(n, a, b))

    # 4. Exponential families (both tails)
    for lam in (0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_exponential(n, lam=lam, reverse=False))
        cand.append(_distribution_exponential(n, lam=lam, reverse=True))

    # 5. Power‑law families (plain and offset)
    for p in (0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_power_law(n, p))
        for off in (0.5, 1.0, 2.0):
            cand.append(_distribution_power_law_offset(n, p, off))

    # 6. Classical DSP windows
    cand.append(_distribution_hann(n))
    cand.append(_distribution_hamming(n))
    cand.append(_distribution_blackman(n))

    # 7. Random Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # 8. Sparse start (few non‑zeros)
    k = max(2, n // 500)
    cand.append(_distribution_sparse_random(n, k))

    # 9. Mixtures of a symmetric beta with the uniform distribution
    for p in (0.1, 0.2, 0.5, 0.8):
        beta = _distribution_beta_sym(n, p)
        uniform = _distribution_uniform(n)
        for w in (0.2, 0.5, 0.8):
            cand.append(_mix_vectors(beta, uniform, w))

    # 10. Shuffled versions of a few bases to break symmetry
    for base in (cand[0], cand[2], cand[-3]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Local optimisation ----------------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """
    Exponentiated‑gradient (mirror) descent on a smooth‑max surrogate.
    Temperature α grows from 1 to 1e7 while the learning rate decays.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e3, 4000, 0.2, 0.02),
        (1e3, 1e5, 4000, 0.02, 0.002),
        (1e5, 1e7, 5000, 0.002, 0.0002),
    ]

    for a0_alpha, a1_alpha, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline, max_iters=40000):
    """
    Adam‑style projected sub‑gradient descent with back‑tracking.
    Stops at the deadline or after max_iters iterations.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # Simple back‑tracking line search
        step = 0.7
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-15 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-15:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter
        if it % 1200 == 0:
            jitter = np.random.randn(n) * 5e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _random_hill_climb(a0, deadline, max_iters=8000):
    """
    Cheap stochastic hill‑climber that moves a small amount of mass
    between two random positions and accepts strictly improving moves.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        src = random.randrange(n)
        if a[src] < eps:
            continue
        tgt = random.randrange(n - 1)
        if tgt >= src:
            tgt += 1

        # transfer up to 50 % of the source mass, limited by 0.02 absolute amount
        max_delta = min(a[src] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        sc = _max_convolution_score(a2)

        if sc < best_sc - 1e-15:
            best_sc = sc
            a = a2
            best_a = a2.copy()
            if it % 1000 == 0:
                print(f"[{time.time():.1f}] hill it={it} new best {sc:.6f}")

        # occasional pruning of tiny entries
        if it % 3000 == 0:
            keep = max(2, int(0.95 * n))
            idx = np.argpartition(a, -keep)[-keep:]
            a_pruned = np.zeros_like(a)
            a_pruned[idx] = a[idx]
            a_pruned /= a_pruned.sum()
            sc_pruned = _max_convolution_score(a_pruned)
            if sc_pruned < best_sc - 1e-15:
                a = a_pruned
                best_a = a.copy()
                best_sc = sc_pruned
                print(f"[{time.time():.1f}] prune it={it} new best {sc_pruned:.6f}")

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence discovered before the timeout.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # safety margin – we stop a couple of seconds before the hard limit
    deadline = start + budget_s - 2.0

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []                       # (score, vector), kept sorted ascending

    # ------------------------------------------------------------------
    # 1. Broad exploration on several lengths
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")
            # maintain a short elite list (max size 8)
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Use a historic solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(4, len(elite))          # limit restarts to keep within budget
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.6f}")

        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)
        a_mid, sc_mid = _random_hill_climb(a_mid, deadline)

        if sc_mid < best_score - 1e-15:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity, normalise and report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
