"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

The algorithm builds on the previous gradient‑based method but adds a
lightweight simulated‑annealing phase that works directly on the
non‑smooth objective (the maximal convolution entry).  After an
extensive mirror‑descent and Adam‑style sub‑gradient polishing we
perform many random “move‑mass” steps that are accepted either if they
decrease the current worst convolution value or with a temperature‑
controlled probability.  The temperature slowly cools, allowing the
search to escape shallow plateaus and to make fine adjustments near the
end.  The whole routine is time‑aware (budget ≈ 1000 s) and keeps the
best vector found over a few different lengths.

The returned list sums to 1, so the value returned by `evaluate_sequence`
is exactly `2·n·max(conv)`.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform (both already normalised)."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    return np.random.dirichlet(np.full(n, alpha))


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

# Local optimisation ----------------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """Exponentiated‑gradient descent on a smoothed max (soft‑max)."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    schedule = [
        (1.0, 1e3, 3000, 0.15, 0.015),
        (1e3, 1e5, 3000, 0.015, 0.0015),
        (1e5, 1e6, 4000, 0.0015, 0.00015),
    ]

    for a0_alpha, a1_alpha, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline):
    """Adam‑style sub‑gradient descent with back‑tracking line‑search."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _simulated_annealing(a0, deadline):
    """Simple SA that moves a tiny amount of mass away from the worst offset."""
    a = a0.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * cur_max

    # initial temperature – 5 % of the current max (scale‑free)
    T = max(cur_max * 0.05, 1e-12)
    it = 0
    eps = 1e-12

    while time.time() < deadline:
        it += 1
        # gentle exponential cooling
        T = max(T * 0.99999, 1e-12)

        # which convolution index is currently the worst?
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idxs = np.arange(i_start, i_end + 1, dtype=np.int64)
        partners = k - idxs
        prod = a[idxs] * a[partners]
        total = prod.sum()

        if total > 0:
            probs = prod / total
            src = np.random.choice(idxs, p=probs)
        else:
            src = np.random.randint(0, n)

        # pick a different target index
        trg = np.random.randint(0, n - 1)
        if trg >= src:
            trg += 1

        # propose a small transfer of probability mass
        max_delta = min(a[src] * 0.5, 0.02)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()

        a_new = a.copy()
        a_new[src] -= delta
        a_new[trg] += delta
        # (sum stays 1, non‑negativity guaranteed by construction)

        conv_new = _fft_self_convolve(a_new)
        new_max = conv_new.max()

        if new_max < cur_max - 1e-15:
            # deterministic improvement
            a = a_new
            conv = conv_new
            cur_max = new_max
            if 2.0 * n * cur_max < best_sc - 1e-12:
                best_sc = 2.0 * n * cur_max
                best_a = a.copy()
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] SA it={it} new best {best_sc:.6f}")
        else:
            # accept with Metropolis probability
            if T > 0 and np.random.rand() < math.exp(-(new_max - cur_max) / T):
                a = a_new
                conv = conv_new
                cur_max = new_max

        # every few thousand steps bring the walk back to the best seen so far
        if it % 5000 == 0 and cur_max > best_sc / (2.0 * n) * 1.05:
            a = best_a.copy()
            conv = _fft_self_convolve(a)
            cur_max = conv.max()

    return best_a, best_sc

# Candidate generator ----------------------------------------------------
def _generate_candidates(n):
    """Return a diverse list of normalised vectors for the given length."""
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))   # arcsine

    # dense grid of symmetric betas
    for p in np.concatenate(
        (np.logspace(-2, -0.2, 12), np.linspace(0.1, 2.0, 20))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with the uniform distribution
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.2, 0.5, 0.8):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # several Dirichlet draws
    for alpha in (0.05, 0.2, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start (few non‑zeros)
    k = max(2, n // 400)
    cand.append(_distribution_sparse_random(n, k))

    # shuffled versions to break symmetry
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence (plain Python list) found within the time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # a safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []               # (score, vector) sorted increasingly

    # ------------------------------------------------------------------
    # 1. Broad exploration over a few lengths
    lengths = [4096, 8192]   # powers of two – FFT friendly and reasonably fast

    for n in lengths:
        if time.time() >= deadline:
            break
        cand_vecs = _generate_candidates(n)

        for vec in cand_vecs:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)

            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n} score={sc:.6f}")

            # maintain a tiny elite list (≤ 6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Accept any historic solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – should never happen
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(4, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} len={n0} start score={elite[idx][0]:.6f}")

        # a) smooth‑max mirror descent (still fast)
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # b) long Adam‑style sub‑gradient polishing – give it most of the remaining time
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # c) simulated‑annealing fine‑tuning on whatever time is left
        if time.time() < deadline - 1.0:
            a_mid, sc_mid = _simulated_annealing(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – ensure non‑negativity, normalise and report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
