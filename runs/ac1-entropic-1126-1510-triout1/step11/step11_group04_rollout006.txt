"""
Improved search for a non‑negative step function minimizing
evaluate_sequence.

The algorithm works in three phases:

1. **Broad exploration** – generate analytic families (uniform,
   arcsine, beta, power‑law, etc.) for several lengths and keep a very
   small elite (≤4) per length.  All candidates are symmetrised
   (a_i = a_{n‑1‑i}) because the optimal solution is believed to be
   symmetric.

2. **Mix‑and‑refine** – repeatedly form random convex combinations of
   elite members of the same length, evaluate the cheap surrogate
   objective (2·n·max(a∗a)), and keep any improvement in the elite.
   Occasionally a short mirrored‑gradient (exponentiated‑gradient)
   refinement is applied to the mixture.

3. **Heavy polishing** – from the global best vector run a long
   exponentiated‑gradient (smooth‑max mirror descent) followed by a
   projected Adam‑style sub‑gradient step and a final hill‑climber /
   pruning phase.  All steps respect the simplex and non‑negativity
   constraints.

The total running time is limited to the supplied budget (default 1000 s);
a 2 s safety margin is kept.  The returned list is a plain Python list
of non‑negative floats whose sum ≈ 1.0, ready for ``evaluate_sequence``.
"""

# Basic linear‑algebra utilities -----------------------------------------

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence``.
    ``a`` is assumed to be normalised to sum = 1.
    Returns 2·n·max(a*a) (the same value up to the scaling factor used in
    ``evaluate_sequence``).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry a[i] = a[n‑1‑i]."""
    return (v + v[::-1]) * 0.5


def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return _symmetrize(vals)


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals = vals.astype(np.float64)
    vals /= vals.sum()
    return _symmetrize(vals)


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return _symmetrize(vals)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return _symmetrize(w)


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    # keep asymmetry – do not symmetrise here
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return _symmetrize(vals)


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return _symmetrize(vals)


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return _symmetrize(vals)


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return _symmetrize(vals)

def _generate_candidates_for_length(n):

    cand = []

    # Analytic families (already symmetric where appropriate)
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta – scanning p over a wide range
    for p in (0.005, 0.01, 0.02, 0.04, 0.06, 0.08, 0.12, 0.15,
              0.2, 0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta – keep them unsymmetrised (they can still be useful)
    for p, q in ((0.2, 1.0), (1.0, 0.2), (0.5, 1.5), (1.5, 0.5)):
        cand.append(_distribution_beta_asym(n, p, q))

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law and its reverse (reverse is automatically symmetric after symmetrisation)
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws (different concentrations)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(_symmetrize(vec))

    # Sparse random start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(_symmetrize(sparse))

    # Shuffle a few base families to introduce asymmetry
    for base in (cand[0], cand[1], cand[4]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    # Ensure all candidates are properly normalised and non‑negative
    processed = []
    eps = 1e-12
    for v in cand:
        v = np.maximum(v, eps)
        v /= v.sum()
        processed.append(v)
    return processed

# Smooth‑max mirror descent (exponentiated‑gradient) --------------------
def _smooth_max_mirror_descent(a: np.ndarray, deadline: float):
    """
    Exponentiated‑gradient mirror descent on the smooth‑max surrogate.
    The algorithm enforces symmetry after every update.
    Returns (best_vector, best_score).
    """
    n = a.shape[0]
    eps = 1e-12

    # Schedule: temperature grows exponentially, learning rate decays.
    # We use a long single schedule covering the whole allowed time.
    max_steps = 50000
    # Determine actual number of steps allowed by the time budget
    est_step_time = 0.002  # rough estimate
    max_allowed = int((deadline - time.time()) / est_step_time)
    steps = min(max_steps, max_allowed)

    best_a = a.copy()
    best_sc = _objective(best_a)

    for it in range(steps):
        if time.time() >= deadline:
            break

        # temperature α ranging from 1 to 1e7 (log‑linear)
        t = it / steps
        alpha = 1.0 * (1e7) ** t
        # learning rate ranging from 0.8 to 1e-5 (log‑linear)
        lr = 0.8 * (1e-5 / 0.8) ** t

        # compute smoothed max
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
        a = _symmetrize(a)

        if it % 4000 == 0:
            sc = _objective(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc

# Projected sub‑gradient (Adam‑style) ------------------------------------
def _projected_subgradient(a: np.ndarray, deadline: float):
    """
    True sub‑gradient descent on the exact max, with Adam‑style momentum
    and back‑tracking line search. Symmetry is enforced after each step.
    """
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_step = 0.4

    best_a = a.copy()
    best_sc = _objective(best_a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum update
        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_step
        # back‑tracking line search
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand = _symmetrize(cand)
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        # occasional random jitter to escape plateaus
        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            a = _symmetrize(a)

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc

# Hill‑climbing & pruning -------------------------------------------------
def _hill_climb_and_prune(a: np.ndarray, deadline: float):
    """
    Light hill‑climber that moves tiny masses away from the dominant
    convolution pair, together with occasional pruning of negligible entries.
    Symmetry is kept throughout.
    """
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- mass transfer from the pair that creates the maximal conv -----
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # Transfer a little mass from the larger of the two to a random partner
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            max_delta = min(a[source] * 0.5, 0.03)
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                a2 = _symmetrize(a2)
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-13:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    new_sc = _objective(a)
                    if new_sc < best_sc - 1e-12:
                        best_sc = new_sc
                        best_a = a.copy()
                        if it % 200 == 0:
                            print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ----- pruning every few thousand steps -----
        if it % 2500 == 0:
            # keep a high‑mass fraction
            for keep_frac in (0.98, 0.95, 0.92):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                pruned = np.zeros_like(a)
                pruned[idx] = a[idx]
                pruned /= pruned.sum()
                pruned = _symmetrize(pruned)
                sc = _objective(pruned)
                if sc < best_sc - 1e-12:
                    a = pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ----- jitter -----
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            a = _symmetrize(a)
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Mixing utilities --------------------------------------------------------
def _mix_vectors(vectors, n):
    """
    Produce a random convex combination of the supplied vectors (all of length n).
    The result is symmetrised and normalised.
    """
    if len(vectors) == 1:
        return vectors[0].copy()
    k = random.randint(2, min(3, len(vectors)))
    chosen = random.sample(vectors, k)
    weights = np.random.dirichlet(np.ones(k))
    mix = np.zeros(n, dtype=np.float64)
    for w, vec in zip(weights, chosen):
        mix += w * vec
    mix = np.maximum(mix, 1e-12)
    mix /= mix.sum()
    mix = _symmetrize(mix)
    return mix

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin
    eps = 1e-12

    lengths = [2000, 3500, 5000, 8000, 12000, 16000, 20000]

    for n in lengths:
        if time.time() >= deadline:

        candidates = _generate_candidates_for_length(n)
        elite = []
        for vec in candidates:
            if time.time() >= deadline:
                break
            sc = _objective(vec)
            # keep up to 4 elite vectors per length
            if len(elite) < 4:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])
        elite_by_len[n] = elite
        best_local = elite[0][0] if elite else float('inf')
        print(f"[{time.time():.1f}] Length {n} best local {best_local:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate a previous solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            prev = _symmetrize(prev)
            n_prev = prev.size
            sc = _objective(prev)
            # insert into elite of its length (create list if missing)
            if n_prev not in elite_by_len:
                elite_by_len[n_prev] = []
            elite = elite_by_len[n_pred := n_prev]
            if len(elite) < 4:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
            print(f"[{time.time():.1f}] Integrated previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Mixing phase – random convex combos of elites (short refinements)
    mix_deadline = deadline - 30.0  # preserve ~30 s for final polishing
    while time.time() < mix_deadline:
        # pick a length that currently has at least 2 elite members
        possible_lengths = [ln for ln, el in elite_by_len.items() if len(el) >= 2]
        if not possible_lengths:
            break
        n = random.choice(possible_lengths)
        elite = [vec for (_, vec) in elite_by_len[n]]
        new_vec = _mix_vectors(elite, n)

        # quick short mirror‑descent refinement (≈5 s) to polish the mixture
        sub_dead = min(time.time() + 5.0, deadline)
        new_vec, _ = _smooth_max_mirror_descent(new_vec, sub_dead)

        sc = _objective(new_vec)
        # attempt to insert into elite list
        cur_elite = elite_by_len[n]
        if len(cur_elite) < 4:
            cur_elite.append((sc, new_vec.copy()))
            cur_elite.sort(key=lambda x: x[0])
        else:
            if sc < cur_elite[-1][0]:
                cur_elite[-1] = (sc, new_vec.copy())
                cur_elite.sort(key=lambda x: x[0])
        # occasional status
        if random.random() < 0.01:
            print(f"[{time.time():.1f}] Mix len={n} new score {sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Select global best and run heavy polishing
    global_best_vec = None
    global_best_score = float('inf')
    for elite in elite_by_len.values():
        for sc, vec in elite:
            if sc < global_best_score:
                global_best_score = sc
                global_best_vec = vec.copy()
    if global_best_vec is None:
        # safety fallback
        n = 4000
        global_best_vec = _distribution_uniform(n)

    # a) long mirror descent
    global_best_vec, sc_mid = _smooth_max_mirror_descent(global_best_vec, deadline)
    global_best_score = sc_mid
    print(f"[{time.time():.1f}] After mirror descent score {global_best_score:.6f}")

    # b) Adam‑style sub‑gradient polishing
    if time.time() < deadline:
        global_best_vec, sc_mid = _projected_subgradient(global_best_vec, deadline)
        if sc_mid < global_best_score - 1e-12:

    if time.time() < deadline:

        if sc_mid < global_best_score - 1e-12:

        print(f"[{time.time():.1f}] After hill‑climb score {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanitisation & return
    global_best_vec = np.maximum(global_best_vec, eps)
    global_best_vec /= global_best_vec.sum()
    final_score = _objective(global_best_vec)
    print(f"[{time.time():.1f}] Final best score ≈ {final_score:.6f} (length={global_best_vec.shape[0]})")
    return global_best_vec.tolist()
