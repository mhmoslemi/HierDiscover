"""
Propose a non‑negative step function via alternating linear‑program refinements.

Algorithm
---------

1. **Initial families** – generate a modest set of analytic probability
   vectors (uniform, arcsine, symmetric / asymmetric beta, sine‑power,
   cosine window, Gaussian, power‑law, random Dirichlet).  Several
   lengths around 3000–4000 are tried.

2. **Iterative LP improvement** – for a current vector `f` (∑f = 1):

   * Compute the self‑convolution `c = f*f` and its maximum `M`.
   * Take all indices where `c` is within a tiny tolerance of `M` –
     these are the *active* constraints.
   * Build a linear programme
         maximise   Σ g_j
         subject to (f * g)_k ≤ M   for every active k
                     0 ≤ g_j ≤ 1000
     (the convolution is linear in `g`).  Solve it with SciPy’s
     high‑performance ‘highs’ solver.
   * Rescale the solution to a probability vector `g` (∑g = 1).

   The LP direction `g` is guaranteed not to increase the maximal
   convolution when mixed with `f`.

3. **Line‑search mixing** – instead of the tiny convex step used in
   the reference code we perform an explicit line search on
   `a(t) = (1‑t)·f + t·g` for `t∈[0,1]`.  A coarse grid is examined,
   then a finer grid around the best point.  The best `t` is kept.

4. **Stochastic jitter** – every few iterations a small random perturbation
   is added to avoid stagnation.

5. **Global bookkeeping** – the best vector seen over all lengths and
   restarts is stored, sanitised (non‑negative, sum = 1) and returned.

The whole search stops well before the 1000 s budget (2 s safety
margin).  The method is completely self‑contained, uses only NumPy and
SciPy and respects the security checks of `evaluate_sequence`.
"""

# Convolution utilities (FFT based, O(n log n))

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Homogenised objective: 2 * n * max(a*a)  (a already normalised to sum = 1)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Analytic families – each returns a probability vector (sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _generate_initial_candidates(n: int):
    """Create a modest collection of analytic families and random draws."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta
    for p in (0.02, 0.04, 0.08, 0.12, 0.2, 0.4, 0.6, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta
    for p, q in ((0.2, 1.0), (1.0, 0.2), (0.5, 1.5), (1.5, 0.5)):
        cand.append(_distribution_beta_asym(n, p, q))

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.20, 0.30, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (and its reverse)
    for p in (1.2, 1.5, 2.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws
    for alpha in (0.2, 0.5, 1.0, 2.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # Sparse start (few non‑zeros)
    k = max(2, n // 250)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    return cand

# Linear‑program utilities

def _active_indices(conv: np.ndarray, M: float, rel_tol: float = 1e-12) -> np.ndarray:
    """Return indices where conv is equal to its maximum up to a tiny tolerance."""
    # absolute tolerance works fine because conv values are O(1/n)
    tol = rel_tol * (abs(M) + 1.0)
    idx = np.where(conv >= M - tol)[0]
    if idx.size == 0:          # safety fallback
        idx = np.array([int(np.argmax(conv))], dtype=int)
    return idx


def _build_constraint_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (|active| × n) such that (f * g)[k] = A[k,:]·g.
    Rows correspond to the active convolution indices.
    """
    n = f.shape[0]
    m = active_idx.shape[0]
    A = np.zeros((m, n), dtype=np.float64)

    for r, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        length = i_end - i_start + 1
        # indices of g that appear in the convolution for this k
        f_slice = f[k - np.arange(i_start, i_end + 1, dtype=int)]
        A[r, i_start:i_end + 1] = f_slice
    return A


def _lp_direction(f: np.ndarray, M: float, active_idx: np.ndarray):
    """
    Solve the LP  maximise Σ g_j   subject to   (f * g)_k ≤ M  on active rows,
    0 ≤ g_j ≤ 1000.  Return the normalised vector g (sum = 1).  If the LP
    fails or is unbounded, fall back to the current f.
    """
    if not _SCIPY_OK:          # pragma: no cover
        return f.copy()        # simple fallback

    n = f.shape[0]
    A = _build_constraint_matrix(f, active_idx)
    m = active_idx.size

    # maximise Σ g_j  → minimise -Σ g_j
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # Use the high‑performance HiGHS solver
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 5.0})

    if not res.success or res.x is None or np.all(res.x == 0):
        # Anything reasonable is better than crashing – keep f.
        return f.copy()

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return f.copy()
    return g / s

# Local improvement: LP direction + line search + jitter

def _improve_sequence(f0: np.ndarray, deadline: float, max_iters: int = 60):
    """
    Starting from a probability vector `f0`, run alternating LP+line‑search
    steps until time runs out or `max_iters` is reached.  Returns the
    best vector seen and its objective value.
    """
    eps = 1e-12
    f = np.maximum(f0, eps)
    f /= f.sum()
    n = f.shape[0]

    best_vec = f.copy()
    best_sc = _objective(f)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _fft_self_convolve(f)
        M = conv.max()
        active = _active_indices(conv, M)

        # ----- LP direction -------------------------------------------------
        g = _lp_direction(f, M, active)

        # ----- line search --------------------------------------------------
        # start with a coarse grid, then zoom in
        best_local = f.copy()
        best_local_sc = best_sc

        def _eval_t(t):
            a = (1.0 - t) * f + t * g
            a = np.maximum(a, eps)
            a /= a.sum()
            return _objective(a), a

        # coarse grid (11 points)
        t_grid = np.linspace(0.0, 1.0, 11)
        for t in t_grid:
            sc, a = _eval_t(t)
            if sc < best_local_sc - 1e-14:
                best_local_sc = sc
                best_local = a

        # identify best t from the coarse grid
        best_t = min(t_grid, key=lambda tt: _eval_t(tt)[0])

        # refinement phases
        for delta in (0.07, 0.03, 0.015, 0.007):
            lo = max(0.0, best_t - delta)
            hi = min(1.0, best_t + delta)
            t_grid = np.linspace(lo, hi, 9)
            for t in t_grid:
                sc, a = _eval_t(t)
                if sc < best_local_sc - 1e-14:
                    best_local_sc = sc
                    best_local = a
                    best_t = t

        f = best_local
        if best_local_sc < best_sc - 1e-14:
            best_sc = best_local_sc
            best_vec = f.copy()
            print(f"[{time.time():.1f}] Improvement it={it} score={best_sc:.6f}")

        # ----- occasional jitter to escape a flat region --------------------
        if it % 12 == 0:
            jitter = np.random.randn(n) * 1e-6
            f = np.maximum(f + jitter, eps)
            f /= f.sum()

    return best_vec, best_sc

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising `evaluate_sequence`.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # Lengths around the region that gave the previous best (≈3300)
    lengths = [2500, 3000, 3500, 4000, 4500]

    for n in lengths:
        if time.time() >= deadline:
            break

        # --- generate a small elite of analytic candidates ----------------
        candidates = _generate_initial_candidates(n)
        # keep the top 4 according to the plain objective
        scored = []
        for vec in candidates:
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()
            sc = _objective(vec)
            scored.append((sc, vec))
        scored.sort(key=lambda x: x[0])
        elite = scored[:4]

        # --- improve each elite candidate ---------------------------------
        for init_sc, init_vec in elite:
            if time.time() >= deadline:
                break
            # Run the LP‑based optimiser
            improved_vec, improved_sc = _improve_sequence(init_vec, deadline,
                                                          max_iters=70)
            if improved_sc < best_score - 1e-14:
                best_score = improved_sc
                best_seq = improved_vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] New global best length={n} score={best_score:.6f}")

    # --- fallback (should never be needed) --------------------------------
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score={best_score:.6f}")

    # final sanitisation
    best_seq = np.maximum(best_seq, 1e-12)
    best_seq /= best_seq.sum()
    final_sc = _objective(best_seq)
    print(f"Final best score ≈ {final_sc:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
