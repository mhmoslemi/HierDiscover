def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2 # safety

    # lengths list
    lengths = [1000,2000,4000,8000,12000,20000]
    # initial best
    overall_best_seq = None
    overall_best_score = float('inf')
    top_candidates = [] # store tuples (score,array)

    # scanning
    for n in lengths:
        if time.time() >= deadline:
            break
        # generate static candidates
        candidates = _generate_static_candidates(n, rng)
        for arr in candidates:
            if time.time() >= deadline:
                break
            # ensure positivity, renorm
            arr = np.maximum(arr, 0.0)
            if arr.sum() == 0:
                continue
            arr = arr / arr.sum()
            sc = _objective(arr)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = arr.copy()
                print(f"New global best n={n}, score={sc:.6f}")
            top_candidates.append((sc, arr.copy()))
    # keep top candidates across all lengths
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:10]

    # incorporate previous solution if any
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim ==1 and prev.size >0:
            prev = np.maximum(prev,0.0)
            if prev.sum() ==0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"Previous solution improved: score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
            top_candidates.sort(key=lambda x: x[0])
            top_candidates = top_candidates[:10]
    except Exception as e:
        pass

    # gradient descent stage
    remaining = deadline - time.time()
    if remaining > 30: # enough time
        # allocate time per seed equally
        seeds = [cand for _,cand in top_candidates]
        per_seed_deadline = remaining / max(1,len(seeds))
        for seed_seq in seeds:
            sub_deadline = time.time() + per_seed_deadline
            # run descent

                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"After descent new best {overall_best_score:.6f}")
            if time.time() >= deadline:
                break

    # final polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(overall_best_seq.copy(), deadline, overall_best_seq, overall_best_score)

    # final output: optionally scale to sum not needed but ok.
    # evaluate final using evaluate_sequence?
    # convert to list

    return overall_best_seq.tolist()

"""
Hybrid multi‑start optimizer for the step‑function attack.

The algorithm proceeds in four phases:

1. **Broad seed generation** – Analytic families (uniform, arcsine,
   beta, power law, exponential, two‑block, Dirichlet, custom
   (i+1)^a·(n−i)^b) are instantiated for several lengths.
   Each seed is normalised to a probability distribution and its exact
   objective `2·n·max(conv)/sum^2` is evaluated.  The best few seeds are
   kept.

2. **Warm‑up stochastic descent** – For the elite seeds we run a long
   exponentiated‑gradient descent on a *soft‑max* approximation of the
   maximal auto‑convolution.  The temperature `α` is increased linearly
   (0.5 → 60) while the learning rate decays like `1/√t`.  After each
   iteration the vector is projected back onto the simplex.

3. **Targeted sub‑gradient descent** – Starting from the result of the
   previous phase we perform a short descent using the *hard* max
   (weights are 1 on the current maximal convolution entries, 0 else).
   This sharpens the solution.

4. **Greedy mass‑transfer polishing** – A cheap local walk moves a
   random fraction of mass from one index to another and keeps the
   change only if the true (non‑smoothed) objective improves.

All phases respect the overall time budget (default 1000 s).  The best
sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow_two(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow_two(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the cross‑correlation gradient 2·a⋆w for a weight vector w.
    Both a and w are real non‑negative; len(w) = 2·n‑1.
    """
    n = a.shape[0]
    m = w.shape[0]  # = 2·n‑1
    a_rev = a[::-1]
    size = _next_pow_two(n + m - 1)
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_custom_powerproduct(n: int,
                                     a_exp: float,
                                     b_exp: float) -> np.ndarray:
    """Shape proportional to (i+1)^a_exp * (n-i)^b_exp."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 1) ** a_exp * (n - i) ** b_exp
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _generate_static_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse set of analytically‑defined candidates for length n."""
    cand = []

    # Simple families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # Symmetric beta (p in [0.05,5])
    for p in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 3.0, 5.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric beta – grid in log‑space
    for _ in range(6):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # Power law (both increasing and decreasing)
    for p in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        cand.append(_distribution_power(n, p))

    # Exponential (different rates)
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # Two blocks
    for bf in (0.05, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, bf))

    # Dirichlet draws
    for _ in range(4):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random()))

    # Sparse random
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # Custom power‑product shapes – grid a_exp, b_exp ∈ [-2,2] step 1
    for a_exp in (-2.0, -1.0, 0.0, 1.0, 2.0):
        for b_exp in (-2.0, -1.0, 0.0, 1.0, 2.0):
            cand.append(_distribution_custom_powerproduct(n, a_exp, b_exp))

    return cand

def _softmax_expo_gradient_descent(a: np.ndarray,
                                   deadline: float,
                                   best_seq: np.ndarray,
                                   best_score: float,
                                   lr0: float = 1.0,
                                   max_iter: int = 60000,
                                   eval_interval: int = 400,
                                   alpha_start: float = 0.5,
                                   alpha_end: float = 60.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the max‑conv.
    Returns possibly improved sequence and its objective.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # simplex projection
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.2f}] soft‑max it={t} α={alpha:.2f} new best {best_score:.6f}")

        # occasional tiny random jitter to escape stagnation
        if t % (eval_interval * 10) == 0:
            jitter = rng.normal(scale=0.02, size=n)
            a = np.maximum(a + jitter * a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a_sum

    return best_seq, best_score

def _hardmax_gradient_descent(a: np.ndarray,
                              deadline: float,
                              best_seq: np.ndarray,
                              best_score: float,
                              lr0: float = 0.5,
                              max_iter: int = 15000,
                              eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Short descent using the actual max (hard‑max) instead of a soft‑max.
    The weight vector w is 1 on all indices attaining the current maximum
    (equally split), otherwise 0.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # weight on all maximal entries
        mask = (conv >= max_conv - 1e-15)
        w = mask.astype(np.float64)
        w /= w.sum()               # uniform on the active set

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # projection onto simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.2f}] hard‑max it={t} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer walk: repeatedly move a random fraction of mass
    from one index to another and keep the move only if the true
    (non‑smoothed) objective improves.
    """
    n = a.shape[0]
    rng_local = np.random.default_rng()
    while time.time() < deadline:
        # pick a source with non‑trivial mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng_local.choice(src_candidates)
        j = rng_local.integers(n)
        if j == i:
            continue

        # propose moving a random fraction (up to 50 %) of the source mass
        delta = a[i] * rng_local.random() * 0.5
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.2f}] swap i={i}→j={j} δ={delta:.2g} new best {best_score:.6f}")

    return best_seq, best_score
