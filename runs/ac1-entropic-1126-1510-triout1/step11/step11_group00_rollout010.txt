def _grad_softmax(a, conv_weights, temperature):
    # conv_weights = conv (full self-convolution), shape (2n-1)
    w = np.exp(conv_weights / temperature)
    w = w / w.sum()
    # reversed w
    w_rev = w[::-1]
    # convolution a * w_rev using FFT
    n = a.shape[0]
    need = n + (2*n-1) - 1 # = 3n-2
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(A * W, n=size)
    conv_full = conv_full[:need]
    L = 2*n-1
    g = conv_full[L-1 - np.arange(n)]
    return 2.0 * g

   matches ``evaluate_sequence`` up to the clipping that never triggers for the
   generated vectors.

3. **Multi‑active mirror descent** – run an exponentiated‑gradient descent
   on the simplex for each top seed.  The sub‑gradient is the sum of the
   gradients of **all** convolution indices whose value is within a small
   fraction (``active_frac``) of the current maximum.  This “active set”
   approach flattens the self‑convolution far more reliably than using only
   the single maximiser.

4. **Soft‑max (smoothed) gradient descent** – starting from the best vector
   obtained in step 3, minimise a smooth surrogate of the max:
   ``f_T(a)=log‑sum‑exp(conv(a,a)/T)``.
   The gradient of this surrogate is obtained by a weighted cross‑correlation
   (computed with a single FFT).  The temperature ``T`` is slowly decreased
   (geometric schedule) so that the algorithm gradually focuses on the true
   maximum.  Multiplicative updates with simplex projection keep the iterates
   feasible.

5. **Local polishing** – cheap pairwise mass‑transfer swaps followed by a
   short simulated‑annealing walk.

All stages respect the supplied time budget (default 1000 s).  The returned
list is a plain Python list; ``evaluate_sequence`` will take care of any
additional clipping or normalisation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
    2 · n · max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle,  σ = sigma_frac·n."""
    sigma = max(sigma_frac * n, 1e-12)
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.008
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    return lam * (1.0 / n) + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Multi‑active mirror descent -------------------------------------------
def _subgradient_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    eval_interval: int = 200,
    active_frac: float = 0.005,
    max_active: int = 20,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent with a multi‑active sub‑gradient.
    All iterates stay on the probability simplex.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]

        if act.shape[0] > max_active:
            top_idx = np.argsort(conv[act])[-max_active:]
            act = act[top_idx]

        # accumulate gradient from all active indices
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for idx in act:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        # multiplicative update (mirror descent with KL divergence)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] subgrad_multi it={t} new global best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # jitter if stuck
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) gradient descent -------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    T0: float = 0.12,
    decay_T: float = 0.998,
    eval_interval: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of the max using a temperature ``T``.
    The gradient of  log‑sum‑exp(conv/T)  is a weighted cross‑correlation.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # length of self‑convolution
    need = n + L - 1                  # length needed for the cross‑correlation
    size = 1 << ((need - 1).bit_length())  # power‑of‑two FFT size
    t = 0
    T = T0
    while time.time() < deadline and t < 150_000:
        t += 1

        conv = _fft_self_convolve(a)          # shape (2n‑1,)
        # softmax weights
        w = np.exp(conv / T)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / (2 * n - 1))
        else:
            w /= w_sum

        # gradient = 2 * (a ⋆ w)   (cross‑correlation)
        w_rev = w[::-1]
        A = np.fft.rfft(a, n=size)
        W = np.fft.rfft(w_rev, n=size)
        corr_full = np.fft.irfft(A * W, n=size)
        corr_full = corr_full[:need]          # length 3n‑2
        # cross‑correlation index mapping
        grad = 2.0 * corr_full[L - 1 - np.arange(n)]

        # multiplicative update
        eta = lr0 / math.sqrt(t)
        # clip gradient to avoid overflow of exp
        grad = np.clip(grad, -1e3, 1e3)
        a = a * np.exp(-eta * grad)

        # project back onto simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # temperature schedule
        T = max(T * decay_T, 1e-8)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] smooth_descent it={t} T={T:.3e} new best {best_score:.6f}"
                )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the supplied time budget (default 1000 s) and
    returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0           # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.   Build a diverse pool of analytic / random seeds
    lengths = [2500, 3000, 4000, 6000, 8000, 12000, 16000, 20000]
    # exponent grids (very wide)
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.5, 8, base=10),   # 0.01 … 0.316
            np.logspace(-0.5, 0, 8, base=10),    # 0.316 … 1
            np.logspace(0, 0.5, 8, base=10),    # 1 … 3.16
            np.logspace(0.5, 1, 8, base=10),    # 3.16 … 10
            np.logspace(1, 2, 6, base=10),      # 10 … 100
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 9)          # 0, 0.125,…,1
    sigma_grid = [0.04, 0.07, 0.11, 0.18, 0.30]  # fraction of n for Gaussian

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families + mixing
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)

            for base in (beta, edge, sine):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # Gaussian family
        for sigma_frac in sigma_grid:
            if time.time() >= deadline:
                break
            gau = _distribution_gaussian(n, sigma_frac)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(gau, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best (Gaussian) len={n} σ={sigma_frac:.3f} lam={lam:.2f} score={sc:.6f}"
                    )

        # random Dirichlet seeds
        for _ in range(10):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.006)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the strongest seeds overall (cull after each length)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:60]

    # ------------------------------------------------------------------
    # 2.   Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.   Guard in case nothing was found
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.   Multi‑active mirror descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:7]]
        per_seed = remaining / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] start multi‑active descent on {len(seeds)} seeds"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_multi(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.55,
                eval_interval=120,
                active_frac=0.005,
                max_active=20,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.   Smoothed gradient descent (log‑sum‑exp surrogate)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] smoothed gradient descent")
        overall_best_seq, overall_best_score = _smooth_gradient_descent(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.5,
            T0=0.12,
            decay_T=0.998,
            eval_interval=150,
        )

    # ------------------------------------------------------------------
    # 6.   Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 7.   Final stochastic local search (simulated annealing style)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return plain Python list for ``evaluate_sequence``.
    return overall_best_seq.tolist()
