"""
Propose a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
-------------------------------------------------------------
1.  **Analytic seeding** – generate many candidate shapes at a few
    lengths (2000 … 20000) using simple closed‑form families:
    uniform, arcsine, symmetric beta, edge‑weighted, triangular,
    semicircle, Gaussian and a few random Dirichlet/sparse vectors.
    Keep the ~30 best (objective = 2·n·max(conv) with Σa=1).

2.  **Active‑set mirror descent** – for each top seed run a
    projected exponentiated‑gradient scheme.
    * At every iteration the exact sub‑gradient is obtained from the
      *active set* of convolution entries that attain the maximum
      (or are within 1e‑12 of it).  This gradient is the correlation
      of the current vector with a sparse weight vector.
    * Update:   a ← a·exp(−η·grad)  (multiplicative EG),  then
      renormalise onto the simplex.
    * η decays like 1/√t; the loop runs until the remaining time budget
      is exhausted (or a hard iteration limit).

3.  **Final greedy polish** – random pairwise mass transfers (up to
    40 % of the source mass).  A move is kept only if it strictly
    improves the exact objective.

The code respects the 2 s safety margin required by the judging
environment and returns a plain Python list ready for ``evaluate_sequence``.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid (i+0.5)/n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i+½)^{‑p}·(n‑i‑½)^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangular bump (peak at centre)."""
    i = np.arange(n, dtype=np.float64)
    peak = (n - 1) / 2.0
    w = 1.0 - np.abs(i - peak) / peak
    w = np.clip(w, 0, None)
    w /= w.sum()
    return w


def _distribution_semicircle(n: int) -> np.ndarray:
    """Wigner‑semicircle pdf on [0,1]."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sqrt(1.0 - (2 * x - 1) ** 2)
    w = np.clip(w, 0, None)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Truncated Gaussian centred at ½ with std = sigma."""
    pts = (np.arange(n, dtype=np.float64) + 0.5) / n
    center = 0.5
    w = np.exp(-0.5 * ((pts - center) / sigma) ** 2)
    w = np.clip(w, 0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Gradient descent using the *active set* of maximal convolution entries.
# ----------------------------------------------------------------------
def _active_set_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    eval_interval: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) using the exact sub‑gradient
    obtained from all entries of the auto‑convolution that attain the
    maximum (within a tiny tolerance).
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # Active set – indices whose value is essentially maximal.
        mask = conv >= max_val - 1e-12
        weight = mask.astype(np.float64)
        wsum = weight.sum()
        if wsum == 0.0:               # extremely unlikely, fall back to exact equality
            weight = np.isclose(conv, max_val).astype(np.float64)
            wsum = weight.sum()
        weight /= wsum

        # Gradient = correlation of a with the sparse weight vector.
        grad = np.correlate(a, weight, mode="full")[-n:]

        # Decaying stepsize (standard 1/√t schedule)
        eta = lr0 / math.sqrt(t)
        # Multiplicative EG update + re‑normalisation
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] descent it={t} new global best {best_score:.6f}"
                )
        # safety cap (prevent endless loops on pathological seeds)
        if t >= 400_000:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random pairwise mass moves kept only when they improve the objective."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.4   # up to 40 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
