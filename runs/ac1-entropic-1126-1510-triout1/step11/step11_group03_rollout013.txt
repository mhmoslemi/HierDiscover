"""
optimise_step_sequence.py

A fast, pure‑numpy optimiser for the step‑function “max‑convolution” problem.
It builds a small portfolio of analytic seeds, evaluates them,
runs a long exponentiated‑gradient descent on the most promising ones
(using a temperature‑controlled soft‑max approximation of the max),
and finishes with an inexpensive greedy mass‑transfer walk.

The code is deliberately written to stay within the 1000 s budget,
to make heavy use of FFT based convolutions (O(n log n)) and to keep all
data in RAM (no file I/O).  The final sequence is returned as a plain
list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import math
from typing import List
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)**2; assumes sum(a)>0."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the soft‑max smoothed objective.
    Gradient = 2·a ⋆ w (cross‑correlation) implemented by FFTs.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + len(w) - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Analytic seed families -------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a diverse set of analytic seeds for a given length."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        _distribution_two_blocks(n, block_frac=0.05),
        _distribution_two_blocks(n, block_frac=0.10),
        _distribution_exponential_increasing(n, lam=0.001),
        _distribution_exponential_increasing(n, lam=0.01),
        _distribution_power_increasing(n, p=0.5),
        _distribution_power_increasing(n, p=2.0),
    ]
    # a few beta‑sym and random sparse seeds
    for p in (0.1, 0.3, 0.6):
        cand.append(_distribution_beta_sym(n, p))
    for _ in range(3):
        cand.append(_distribution_random_sparse(n, rng, density=0.015))
    return cand


# ----------------------------------------------------------------------
def _smooth_grad_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr0: float = 0.8,
                         max_iter: int = 200_000,
                         eval_interval: int = 200,
                         alpha_start: float = 0.5,
                         alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the max‑convolution.
    Returns the best sequence (and its score) seen during the run.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # soft‑max weights (stable via subtraction of max)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)            # diminishing step‑size
        a = a * np.exp(-eta * grad)         # exponentiated‑gradient step

        # re‑project onto the probability simplex
        a = _project_onto_simplex(a)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] iter={t} α={alpha:.1f} new best {best_score:.6f}")

        # early stop if the improvement is vanishingly small
        if t % (10 * eval_interval) == 0 and best_score < 1e-12:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap greedy walk: move a random fraction of mass from one index to another
    and keep the change only if the true objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if i == j:
            continue

        delta = a[i] * rng.random() * 0.30   # up‑to‑30 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> List[float]:
    """
    Main entry point for the competition.  It respects the supplied time budget
    and returns a list of non‑negative floats (the step‑function heights) that
    minimise ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. pick a handful of problem sizes; larger sizes give a better lower bound
    lengths = [4000, 8000, 12000]  # enough to be challenging but still cheap for FFT

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates = []                 # (score, array) – will keep a tiny elite

    # ------------------------------------------------------------------
    # 2. generate and evaluate analytic seeds for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length = {n}")

        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand = _project_onto_simplex(cand)   # guarantee sum = 1
            sc = _objective(cand)

            # keep global best
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")

            top_candidates.append((sc, cand))

        # keep only the very best across all lengths (tiny elite)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:8]

    # ------------------------------------------------------------------
    # 3. add the previous solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = _project_onto_simplex(prev)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_candidates.append((sc, prev))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. fallback – should never happen
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. intensive smooth‑gradient descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # use the few absolute best seeds
        elite = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]]
        per_seed = max(1.0, remaining / len(elite))
        for seed_seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _smooth_grad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                max_iter=200_000,
                eval_interval=200,
                alpha_start=0.5,
                alpha_end=80.0,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. final cheap polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (len={len(overall_best_seq)})")
    return overall_best_seq.tolist()
