    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# Distribution families (all give vectors with sum = 1) -----------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) discretised on an equally spaced grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """Mixture w·Beta(p,p) + (1‑w)·Uniform."""
    beta = _distribution_beta_sym(n, p)
    uni = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uni
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    return np.random.dirichlet(np.full(n, alpha))


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a


def _distribution_two_spike(n, gap):
    """Two equal spikes separated by `gap` zeroes (if possible)."""
    a = np.zeros(n, dtype=np.float64)
    i = n // 4
    j = i + gap
    if j >= n:
        j = n // 2
    a[i] = a[j] = 0.5
    return a


def _generate_initial_vectors(n):
    """A rich pool of start points (already normalised to sum = 1)."""
    out = []

    # very basic families
    out.append(_distribution_uniform(n))
    out.append(_distribution_beta_sym(n, 0.5))          # arcsine

    # dense grid of symmetric Betas
    for p in np.concatenate(
        (np.logspace(-2, -0.2, 15), np.linspace(0.1, 2.0, 25))
    ):
        out.append(_distribution_beta_sym(n, float(p)))

    # mixtures with uniform
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.2, 0.5, 0.8):
            out.append(_distribution_mixture_beta_uniform(n, p, w))

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.8, 2.0, 5.0):
        out.append(_distribution_random_dirichlet(n, alpha))

    # sparse start (few non‑zeros)
    k = max(2, n // 500)
    out.append(_distribution_sparse_random(n, k))

    # two‑spike patterns – a classic candidate that often reduces the
    # maximal self‑overlap
    for gap in (int(0.1 * n), int(0.2 * n), int(0.3 * n)):
        out.append(_distribution_two_spike(n, gap))

    # random shuffles of a few base patterns to break symmetry
    for base in (out[0], out[2], out[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        out.append(sh)

    # ensure positivity (tiny epsilon) and perfect normalisation
    eps = 1e-12
    clean = []
    for v in out:
        v = np.maximum(v, eps)
        v /= v.sum()
        clean.append(v)
    return clean

# Smooth‑max objective & its gradient (w.r.t. the softmax parameters) --
def _smooth_max_objective(y, alpha):
    """
    Given parameters y (log‑weights) return (value, gradient) of the
    smooth‑max surrogate φ_α(a) where a = softmax(y).
    """
    # a = exp(y) / sum exp(y)   – always on the simplex
    max_y = np.max(y)
    exp_y = np.exp(y - max_y)          # shift for numerical stability
    Z = exp_y.sum()
    a = exp_y / Z

    # convolution of a with itself
    conv = _fft_self_convolve(a)

    # smooth‑max:  φ = (1/α)·log Σ exp(α·conv)
    max_c = conv.max()
    # use the classic log‑sum‑exp trick
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    phi = max_c + (1.0 / alpha) * np.log(w_sum / w.sum())  # equals (max_c + 1/α·log w_sum) but numeric stable

    # normalised weights (softmax over the α·conv values)
    ws = w / w_sum

    # gradient of (max_c + 1/α·log w_sum) w.r.t. a:
    #   ∂φ/∂a = 2 * (a_rev * ws)   (as in the reference code)
    grad_a = 2.0 * _fft_convolve(a[::-1], ws)[a.size - 1 : a.size - 1 + a.size]

    # chain rule through the softmax:
    #   ∂f/∂y_j = a_j * (grad_a_j - Σ_i grad_a_i * a_i)
    dot = np.dot(grad_a, a)
    grad_y = a * (grad_a - dot)

    return phi, grad_y

# Optimisation wrapper ---------------------------------------------------
def _optimize_vector(initial_a, deadline, max_iter=5000):
    """
    Optimise a starting vector `initial_a` (already on the simplex) using
    a continuation on α = 1 → 1e6 and L‑BFGS‑B on the smooth‑max surrogate.
    Returns (best_a, best_score) where `best_score` is the true objective
    2·n·max(conv).
    """
    n = initial_a.shape[0]
    eps = 1e-12
    # initialise y = log a (tiny epsilon to avoid log 0)
    y = np.log(np.maximum(initial_a, eps))

    # continuation schedule for the smoothing parameter α
    alphas = [1.0, 1e2, 1e4, 1e6]

    best_a = initial_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_a).max()

    for alpha in alphas:
        if time.time() >= deadline:
            break

        # L‑BFGS‑B options – we limit the number of function evaluations
        # to keep the total runtime modest.
        opts = {
            "maxiter": max_iter,
            "disp": False,
            "ftol": 1e-12,
        }

        def func(y_vec):
            phi, grad = _smooth_max_objective(y_vec, alpha)
            return phi, grad

        res = minimize(
            fun=func,
            x0=y,
            method="L-BFGS-B",
            jac=True,
            options=opts,
        )
        y = res.x  # start next phase from where we ended

        # project back onto the simplex (softmax already does it)
        a = np.exp(y - np.max(y))
        a /= a.sum()
        a = np.maximum(a, eps)
        a /= a.sum()

        # evaluate true objective
        cur_score = 2.0 * n * _fft_self_convolve(a).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] α={alpha:.0e} new best {cur_score:.6f}")

    return best_a, best_score

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence (plain Python list of floats).
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    best_seq = None
    best_score = float("inf")
    elite = []                                 # (score, vector)

    # ------------------------------------------------------------------
    # 1. Scan a range of lengths (powers of two – FFT‑friendly)
    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")

        candidates = _generate_initial_vectors(n)

        for vec in candidates:
            if time.time() >= deadline:
                break
            # quick evaluation (exact objective)
            sc = 2.0 * n * _fft_self_convolve(vec).max()

            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n} score={sc:.6f}")

            # keep a small elite set (≤8)
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Try to reuse a previously discovered solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev /= prev.sum()
            sc = 2.0 * prev.shape[0] * _fft_self_convolve(prev).max()
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Imported previous solution score={sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback to uniform if nothing was discovered (should never happen)
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_score = 2.0 * n0 * _fft_self_convolve(best_seq).max()
        print(f"[{time.time():.1f}] Fallback uniform (len={n0}) score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local refinement from elite pool
    restarts = min(len(elite), 6)
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score={elite[idx][0]:.6f}")

        a_opt, sc_opt = _optimize_vector(a0, deadline, max_iter=2500)

        if sc_opt < best_score - 1e-12:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_opt:.6f}")

        # very cheap hill‑climber as a final polish
        # (move a tiny amount of mass from the entry contributing most
        #  to the max‑convolution to a random neighbour)
        a = a_opt.copy()
        n = a.shape[0]
        conv = _fft_self_convolve(a)
        for _ in range(200):
            if time.time() >= deadline:
                break
            k = int(conv.argmax())
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # locate the pair (i,j) that actually realises the max
            best_pair = (i_start, k - i_start)
            best_val = a[best_pair[0]] * a[best_pair[1]]
            for i in range(i_start, i_end + 1):
                j = k - i
                val = a[i] * a[j]
                if val > best_val:
                    best_val = val
                    best_pair = (i, j)
            src = best_pair[0] if a[best_pair[0]] >= a[best_pair[1]] else best_pair[1]
            trg = random.randint(0, n - 1)
            if trg == src:
                continue
            delta = min(a[src] * 0.3, 0.01) * random.random()
            a2 = a.copy()
            a2[src] -= delta
            a2[trg] += delta
            a2 = np.maximum(a2, 1e-12)
            a2 /= a2.sum()
            new_score = 2.0 * n * _fft_self_convolve(a2).max()
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a2.copy()
                a = a2
                conv = _fft_self_convolve(a)
                print(f"[{time.time():.1f}] hill‑climb improvement {new_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity and proper normalisation
    best_seq = np.maximum(best_seq, 1e-12)
    best_seq /= best_seq.sum()
    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
