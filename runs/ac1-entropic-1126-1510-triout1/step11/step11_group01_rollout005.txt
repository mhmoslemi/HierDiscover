assistantfinal**Solution Overview**

The function `propose_candidate` searches for a non‑negative step‑function (a probability
vector) that minimizes  

\[
\text{score}=2\,n\;\max\bigl\{ (a*a)[k]\bigr\},
\qquad\text{with}\quad \sum_i a_i=1 .
\]

The search consists of three stages

1. **Broad family scan** – a rich collection of analytic families
   (uniform, arcsine, symmetric β, edge‑weighted, sine, cosine, quadratic,
   Chebyshev/Kaiser windows, random Dirichlet and sparse seeds) is evaluated
   on several lengths.  
   For each family a small grid of shape parameters (`p`) and a mixing weight
   `λ∈[0,1]` with the uniform distribution is tried.  The best ~10 candidates
   are kept.

2. **Exponential‑gradient refinement** – each kept seed is improved by a
   mirror‑descent on the exact *max‑convolution* objective.  
   The sub‑gradient is built from the `top_k` largest convolution entries
   (default 5).  An exponentiated‑gradient step maintains non‑negativity and a
   Euclidean projection onto the simplex guarantees the sum‑to‑one constraint.
   The routine runs until the allocated time slice expires (or a hard
   iteration limit is hit).

3. **Greedy pairwise polishing** – the current best vector is repeatedly
   altered by moving a random fraction of mass from one index to another.
   A move is kept only if it strictly improves the exact objective; this
   very cheap local search removes any remaining “rough edges”.

All loops respect the global deadline (`budget_s` seconds).  Whenever a new
global best is found its score and the elapsed time are printed so that the
progress is visible.

The code below follows the described algorithm and returns a plain Python
`list[float]` ready for `evaluate_sequence`.  It works with only `numpy`
(and optionally `scipy` for a few window families).  

```python
"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The algorithm mixes an analytic family scan,
exponential‑gradient (mirror) descent on the exact max‑convolution
objective and a final greedy pairwise polishing.  It respects a 1000 s
budget, prints progress and returns a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # size = next power of two ≥ 2*n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input must be non‑negative; the function does **not** perform the
    safety clipping present in evaluate_sequence().
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a grid {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) when p<1
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """
    Edge‑weighted family: w_i ∝ ((i+½)*(n‑i‑½))^{‑p}.
    For p>0 the distribution is heavy at the two ends.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine(n: int) -> np.ndarray:
    """Sine window (always positive)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    """Raised‑cosine window: 1‑cos(π·i/n) (non‑negative)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 - np.cos(np.pi * i / n)
    w /= w.sum()
    return w


def _distribution_quadratic(n: int) -> np.ndarray:
    """Parabolic shape peaking in the middle."""
    i = np.arange(n, dtype=np.float64)
    # scale to [-1,1]
    x = (2.0 * i) / (n - 1) - 1.0
    w = 1.0 - x * x
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# optional SciPy windows (chebwin, kaiser) --------------------------------
try:
    from scipy.signal import chebwin, kaiser
    _HAS_SCI = True
except Exception:  # pragma: no cover
    _HAS_SCI = False


def _distribution_chebyshev(n: int, atten: float) -> np.ndarray:
    w = chebwin(n, at=atten, sym=True).astype(np.float64)
    w = np.maximum(w, 0.0)        # safety, although chebwin is non‑negative
    w /= w.sum()
    return w


def _distribution_kaiser(n: int, beta: float) -> np.ndarray:
    w = kaiser(n, beta, sym=True).astype(np.float64)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

# Sub‑gradient / mirror descent ----------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    top_k: int = 5,
    eval_every: int = 200,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on the exact max‑convolution
    objective.  The gradient is assembled from the `top_k` largest
    convolution entries.  Returns the best sequence ever seen.
    """
    a = a0.copy()
    n = a.shape[0]
    t = 0
    while time.time() < time_limit and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        # indices of the largest entries (including the centre)
        idxs = np.argpartition(-conv, range(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            # contribution 2·a shifted by idx
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)          # decaying step size
        a = a * np.exp(-eta * grad)       # EG update (keeps positivity)
        # simplex projection
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap(
    a0: np.ndarray,
    time_limit: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass from one index to another.
    A move is kept only if it strictly improves the exact objective.
    """
    a = a0.copy()
    n = a.shape[0]
    trials = 0
    while time.time() < time_limit:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-14)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.45   # up to 45 % of source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes evaluate_sequence().
    The routine:
      1. scans a large variety of analytic families on several lengths,
      2. refines the most promising seeds with exponentiated‑gradient descent,
      3. finishes with a greedy pairwise polishing.
    It runs within the supplied time budget and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []   # (score, seq)

    # ------------------------------------------------------------------
    # 1.  Rich family scan
    # ------------------------------------------------------------------
    # lengths – a mix of moderate and large sizes (still cheap enough)
    lengths = [2000, 4000, 6000, 8000, 12000, 20000, 30000]

    # parameter grids
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.3, 12),      # 0.01 … 0.5 (edge‑heavy)
            np.linspace(0.5, 2.0, 10),      # moderate
            np.linspace(2.0, 5.0, 6),       # more centre‑heavy
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 6)        # 0, .2, …, 1

    # optional SciPy windows
    atten_grid = np.linspace(30, 150, 7)      # Chebyshev attenuation (dB)
    beta_grid = np.linspace(0.0, 14.0, 8)     # Kaiser beta

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ---- deterministic families ------------------------------------
        for base_fn in (
            _distribution_uniform,
            _distribution_arcsine,
            _distribution_sine,
            _distribution_cosine,
            _distribution_quadratic,
        ):
            cand = base_fn(n)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # ---- parametric families (beta / edge‑weighted) -----------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta_seq = _distribution_beta_sym(n, p)
            edge_seq = _distribution_edge_weighted(n, p)

            for base in (beta_seq, edge_seq):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # ---- optional SciPy windows ------------------------------------
        if _HAS_SCI:
            for atten in atten_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_chebyshev(n, atten)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] Cheb len={n} att={atten:.0f} score={sc:.6f}")

            for bet in beta_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_kaiser(n, bet)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] Kaiser len={n} beta={bet:.2f} score={sc:.6f}")

        # ---- random Dirichlet seeds ------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] Dirichlet len={n} score={sc:.6f}")

        # ---- sparse random seeds ---------------------------------------
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] sparse len={n} score={sc:.6f}")

        # keep only the best ~30 seeds overall (to bound memory / time)
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Include any previously discovered construction, if present.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous construction score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform length={n} score={overall_best_score:.6f}"
        )

    # ------------------------------------------------------------------
    # 4.  Mirror‑descent refinement on the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        # keep the 5 best seeds for deeper refinement
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] start subgradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sg_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent(
                seed_seq,
                sg_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.8,
                top_k=5,
                eval_every=150,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy pairwise polishing (pure hill‑climbing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swap polishing")
        a_opt, sc_opt = _pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)): {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will do its own normalisation.
    return overall_best_seq.tolist()
```
