"""
Propose a step‑function that (hopefully) beats the previous best.
Algorithm
---------

1. **Broad parametric sweep** – try several lengths (2 k … 32 k) and a
   rich set of families (uniform, arcsine, edge‑power with many
   exponents, symmetric/asymmetric Beta).  The best vector (L1‑normalised
   to 1) is kept.

2. **Continuation Lp‑gradient descent** – minimise the Lp‑norm of the
   self‑convolution for a sequence of p values increasing from 2 to
   200.  For each p we perform a few projected exponentiated‑gradient
   steps (mirror descent) on the simplex.  This smoothes the max‑norm
   and pushes the distribution towards a flatter convolution.

3. **Targeted simulated‑annealing polish** – starting from the best
   vector, repeatedly move a small amount of mass from the index that
   contributes most to the current peak towards the globally smallest
   entry.  Moves are accepted according to a geometric temperature schedule,
   allowing occasional uphill steps to escape local plateaus.

All stages respect the time budget (`budget_s`).  The function returns a
plain Python list; the external `evaluate_sequence` will be called
later to obtain the final score.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions -----------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    Score for a vector already normalised to sum = 1:
    2·n·max(a*a).  This equals the value returned by `evaluate_sequence`
    for the same shape (the scaling cancels out).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Parametric families (all return a vector with sum = 1) ---------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    return _distribution_beta(n, p, p)


def _distribution_edge_power(n, p):
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15  # avoid exact zeros
    w /= w.sum()
    return w

# Continuation Lp gradient descent --------------------------------------
def _lp_descent(a, p_start, p_end, steps, lr_start, lr_end):
    """
    Minimise the Lp‑norm of the self‑convolution by a projected
    exponentiated‑gradient scheme while gradually increasing p.
    """
    n = a.shape[0]
    eps = 1e-15
    a = a.copy()
    for it in range(steps):
        t = it / steps
        p = p_start * (p_end / p_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        conv = np.maximum(conv, eps)

        # weight for Lp‑norm gradient (proportional to conv^{p‑1})
        w = conv ** (p - 1.0)
        # optional normalisation (does not affect direction)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum

        # gradient of the smooth objective (mirror‑descent direction)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()
    return a

# Targeted simulated‑annealing -----------------------------------------
def _simulated_annealing(a, max_iters, deadline):
    """
    Starting from a probability vector `a` (sum = 1), move small chunks
    of mass from the index most responsible for the current peak to
    the globally smallest entry.  Accept moves with a geometric temperature
    schedule, allowing occasional uphill steps.
    """
    n = a.shape[0]
    eps = 1e-15
    a = a.copy()
    b = _fft_self_convolve(a)
    cur_score = 2.0 * n * b.max()
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule (geometric decay)
    T = 0.001
    T_min = 1e-8
    decay = 0.9995

    max_move_frac = 0.12  # at most 12 % of the source weight per move

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # locate the peak of the convolution and a contributing pair
        k = int(b.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # the pair (i,j) with maximal product for this peak
        best_i, best_j = i_start, k - i_start
        best_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i, best_j = i, j

        src = best_i if a[best_i] >= a[best_j] else best_j
        tgt = np.argmin(a)
        if src == tgt:
            # pick a random different target if the smallest coincides
            cand = np.argsort(a)
            tgt = cand[1] if cand[0] == src else cand[0]

        # propose a modest transfer
        max_delta = min(a[src] * max_move_frac, 0.025)
        if max_delta <= eps:
            continue
        delta = max_delta * random.random()
        if delta < eps:
            continue

        # build candidates
        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta

        # incremental update of the convolution
        b2 = b.copy()
        # subtract contribution of src
        b2[src:src + n] -= 2.0 * delta * a
        b2[2 * src] += delta * delta
        # add contribution of tgt
        b2[tgt:tgt + n] += 2.0 * delta * a
        b2[2 * tgt] += delta * delta

        new_score = 2.0 * n * b2.max()

        # acceptance test
        improve = new_score < cur_score
        accept = improve or (random.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)))
        if accept:
            a = a2
            b = b2
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a2.copy()
                print(f"[{time.time():.1f}] SA it={it} new best {best_score:.6f}")

        # temperature decay
        T = max(T_min, T * decay)

        # occasional re‑normalisation to curb numerical drift
        if (it & 0xFFF) == 0:      # every 4096 steps
            a = np.maximum(a, eps)
            s = a.sum()
            if not np.isclose(s, 1.0):
                a /= s
                b = _fft_self_convolve(a)
                cur_score = 2.0 * n * b.max()
    return best_a, best_score
