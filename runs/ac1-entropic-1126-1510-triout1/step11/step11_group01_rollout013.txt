def _subgradient_descent(a, deadline, best_seq, best_score, initial_lr=0.5, eval_interval=200, max_iter=200000):

    t=0

    while time.time() < deadline and t < max_iter:

        t+=1

        conv = _fft_self_convolve(a)

        max_val = conv.max()

        if max_val == 0:
            break  # all zeros, not possible since sum=1

        # find indices with near max
        # tolerance small relative to max
        eps = max_val * 1e-12
        idxs = np.where(conv >= max_val - eps)[0]

        # compute gradient
        grad = np.zeros_like(a)

        n = a.shape[0]
        # shift indices relative to center index n-1
        for idx in idxs:
            s = idx - (n-1)
            shift = abs(s)
            if shift == 0:
                # derivative 2*a
                grad += 2.0 * a
            else:
                # add contributions
                # for positions shift..n-1: grad[pos] += a[pos-shift]
                grad[shift:] += a[: n - shift]
                # positions 0..n-1-shift: grad[pos] += a[pos+shift]
                grad[: n - shift] += a[shift:]

        # step size
        eta = initial_lr / math.sqrt(t)

        # gradient step
        a = a - eta * grad

        # projection onto simplex (and nonnegativity implicitly)
        a = _project_onto_simplex(a, s=1.0)

        # optionally break if not improving

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad iter={t} new best {best_score:.6f}")

    return best_seq, best_score

"""
Propose a candidate step function minimizing evaluate_sequence.

Algorithm:
1. Generate candidate vectors from analytic families (uniform, arcsine,
   Beta, exponential, power‑law, Gaussian, common window functions) and
   random Dirichlet / sparse seeds for several lengths.
2. Keep the top‑K candidates (by the exact 2·n·max(conv) objective) and
   refine each of them using a projected sub‑gradient method that
   minimizes the infinity‑norm of the self‑convolution.  The sub‑gradient
   is computed from all active shifts where the convolution attains its
   maximum; the update uses a decreasing step‑size and Euclidean projection
   onto the probability simplex.
3. Finish with a fast greedy pairwise mass‑transfer local search.
4. Return the best vector found as a plain Python list.
"""

import time
import math
import numpy as np

# optional SciPy windows
try:
    from scipy.signal import get_window
    _HAS_SCIWIN = True
except Exception:
    _HAS_SCIWIN = False

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x>=0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# ------------ Analytic families -----------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) when p<1
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / np.power(i, p)
    w /= w.sum()
    return w

def _distribution_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w

def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    center = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    w /= w.sum()
    return w

def _distribution_window(name: str, n: int, **kwargs) -> np.ndarray:
    # wrappers for scipy.signal.get_window; ensures non‑negative and normalised.
    w = get_window(name, n, **kwargs).astype(np.float64)
    if w.sum() == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= w.sum()
    return w

def _distribution_random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# ----------- Subgradient descent -----------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    eval_interval: int = 200,
    momentum: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Projected subgradient method for min max(conv(a,a)).
    Parameters:
        a          – initial vector (already normalised to sum = 1)
        deadline   – time when the optimisation must stop
        best_seq   – current best vector (for global reporting)
        best_score – its objective value
        lr0        – initial learning‑rate multiplier
        eval_interval – how often we compute the exact objective
        momentum   – if >0 uses simple momentum v←βv+grad
    Returns the best vector seen during the run and its objective.
    """
    n = a.shape[0]
    v = np.zeros_like(a) if momentum > 0 else None
    t = 0
    while time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # indices where conv is within tiny tolerance of the max
        idxs = np.where(conv >= max_val - 1e-14)[0]
        # compute subgradient of max(conv)
        grad = np.zeros_like(a)
        center = n - 1
        for idx in idxs:
            shift = idx - center
            s = abs(shift)
            if s == 0:
                grad += 2.0 * a
            else:
                # contribution from a[j-s] and a[j+s]
                grad[s:] += a[: n - s]
                grad[: n - s] += a[s:]

        # step size ‑ decaying 1/√t
        eta = lr0 / math.sqrt(t)

        if momentum > 0:
            v = momentum * v + grad
            step = eta * v
        else:
            step = eta * grad

        a = a - step

        # projection onto simplex Σa = 1, a≥0
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")

        # safety cap
        if t >= 300000:
            break

    return best_seq, best_score

# ----------- Greedy pairwise swap ---------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass from one entry to another."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # choose destination different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a fraction of source mass
        delta = a[i] * np.random.rand() * 0.4
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

"""
Propose a candidate step function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
1. Generate many seeds from analytic families (uniform, arcsine, beta,
   power‑law, exponential, Gaussian, common window functions) and from
   random Dirichlet / sparse vectors for several lengths.
2. Keep the ~30 best seeds (objective = 2·n·max(conv) with Σa = 1).
3. Refine each of them with a projected sub‑gradient method that
   minimises the infinity‑norm of the self‑convolution.  The sub‑gradient
   is computed from all active shifts where the convolution attains its
   maximum; a decaying step‑size and Euclidean projection onto the
   probability simplex keep the iterate feasible.
4. Finish with a fast greedy pairwise mass‑transfer local search.
5. Return the best vector found as a plain Python list – ``evaluate_sequence``
   will handle any required normalisation.
"""

import time
import math
import numpy as np

# Optional SciPy windows
try:
    from scipy.signal import get_window

    _HAS_SCIWIN = True
except Exception:  # pragma: no cover
    _HAS_SCIWIN = False


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = 1.0 / np.power(i, p)
    w /= w.sum()
    return w


def _distribution_exponential(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-rate * i)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_window(name: str, n: int, **kwargs) -> np.ndarray:
    """
    Helper that builds a SciPy window, clips negatives, and normalises.
    """
    w = get_window(name, n, **kwargs).astype(np.float64)
    w = np.maximum(w, 0.0)  # all entries must be non‑negative
    if w.sum() == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a prescribed density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture of `base` with the uniform distribution."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient descent (full gradient of max‑convolution) -------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    eval_interval: int = 200,
    momentum: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient optimiser for min max(conv(a,a)).
    Parameters
    ----------
    a : np.ndarray
        Initial vector (already normalised to sum = 1).
    deadline : float
        Time (epoch seconds) after which the optimisation must stop.
    best_seq, best_score : current global best (updated on improvement).
    lr0 : float
        Initial learning‑rate multiplier (step size = lr0/√t).
    eval_interval : int
        How often the exact objective is evaluated.
    momentum : float
        If >0, a simple momentum term v←βv+grad is used.
    Returns
    -------
    Tuple (best_vector, best_objective) seen during the run.
    """
    n = a.shape[0]
    v = np.zeros_like(a) if momentum > 0 else None
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        # indices attaining (approximately) the maximum
        idxs = np.where(conv >= max_val - 1e-14)[0]

        # sub‑gradient of the supremum
        grad = np.zeros_like(a)
        centre = n - 1
        for idx in idxs:
            shift = idx - centre
            s = abs(shift)
            if s == 0:
                grad += 2.0 * a
            else:
                grad[s:] += a[: n - s]
                grad[: n - s] += a[s:]

        eta = lr0 / math.sqrt(t)
        if momentum > 0:
            v = momentum * v + grad
            step = eta * v
        else:
            step = eta * grad

        a = a - step
        a = _project_onto_simplex(a, s=1.0)  # enforce Σa = 1, a≥0

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

        if t >= 300000:  # safety cap
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass‑transfer polishing -----------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass; keep the move if it improves the objective."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.4  # up to 40 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

                momentum=0.5,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after subgrad seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise‑swap polishing")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    print(
        f"Final best score (2·n·max(conv) for Σ=1): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
