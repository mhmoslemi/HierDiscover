def simulated_annealing(a, time_deadline):
    n = a.shape[0]
    M = _max_convolution_score(a)
    best = a.copy()
    bestM = M
    T0 = 0.1 * M
    while time.time() < deadline:
        # propose move
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # step size factor
        delta = min(a[i], np.random.rand() * 0.01) # 1% of source mass
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, 1e-12)
        a_new /= a_new.sum()
        M_new = _max_convolution_score(a_new)
        dM = M_new - M
        # accept if better or with probability exp(-dM/T)
        if dM < 0 or np.exp(-dM / T) > np.random.rand():
            a = a_new
            M = M_new
            if M_new < bestM:
                bestM = M_new
                best = a_new.copy()
        # temperature schedule: T decays
        T = max(1e-8, T * 0.99999)  # adjust factor times iterations maybe.
    return best, bestM
```

def _simulated_annealing(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _max_convolution_score(a)
    # temperature schedule
    T0 = best_sc * 0.1  # start at 10% of current objective
    T = T0
    # set parameters
    move_scale = 0.02  # max fraction of source mass to move

    while time.time() < deadline:
        # propose a random move
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # move mass
        max_move = a[i] * move_scale
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, 1e-12)
        a_new /= a_new.sum()
        sc_new = _max_convolution_score(a_new)
        d = sc_new - best_sc
        if d < 0:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            # accept with prob
            prob = math.exp(-d / max(T, 1e-12))
            if prob > np.random.rand():
                a = a_new
        # cool down
        T *= 0.9999  # adjust factor
        if T < 1e-9:
            T = 1e-9
    return best_a, best_sc

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection onto {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow/underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    beta = _distribution_beta_sym(n, p)
    uni = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uni
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

# Deterministic local move (gradient‑guided mass transfer) -------------
def _deterministic_move(a0, deadline):
    """
    Move mass from the index with the largest sub‑gradient to the index
    with the smallest sub‑gradient.  Repeated until no improvement or
    time runs out.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active constraints (within tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0
        # sub‑gradient of the max‑convolution
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if i_max == i_min:
            break

        # move at most 5 % of the source entry (but not more than 0.02)
        delta = min(a[i_max] * 0.05, 0.02) * random.random()
        if delta < eps:
            break

        a2 = a.copy()
        a2[i_max] -= delta
        a2[i_min] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        sc2 = _max_convolution_score(a2)
        if sc2 < best_sc - 1e-12:
            best_sc = sc2
            best_a = a2.copy()
            a = a2
            # optional: print progress
            print(f"[{time.time():.1f}] deterministic move new best {sc2:.6f}")
        else:
            # if the move does not improve we quit – the gradient points
            # to the direction of steepest descent already.
            break

    return best_a, best_sc

# Simulated annealing (light‑weight) -----------------------------------
def _simulated_annealing(a0, deadline):
    """
    Simple SA that repeatedly moves a small amount of probability mass.
    Acceptance follows the Metropolis rule with a geometric cooling.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    cur_sc = _max_convolution_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    # start temperature as 10 % of the current objective
    T = max(1e-6, cur_sc * 0.1)
    min_T = 1e-9
    cooling = 0.99997   # multiplicative cooling factor

    while time.time() < deadline:
        # propose a random transfer of mass
        i = random.randrange(n)
        if a[i] < eps:
            continue
        j = random.randrange(n - 1)
        if j >= i:
            j += 1
        # amount to move – up to 3 % of the source entry
        delta = min(a[i] * 0.03, 0.02) * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_sc = _max_convolution_score(a_new)
        diff = new_sc - cur_sc
        accept = False
        if diff < 0:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-diff / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc - 1e-12:
                best_sc = new_sc
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] SA new best {new_sc:.6f}")

        # cool down
        T = max(T * cooling, min_T)

        # safety break if temperature already tiny
        if T < 1e-12:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Interpolation / up‑sampling -------------------------------------------
def _interpolate_sequence(seq, new_n):
    """Linear interpolation of a simplex‑vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 1e-12)
    new_seq /= new_seq.sum()
    return new_seq

# Local optimisation (mirrored from the reference implementation) ------
def _smooth_max_mirror_descent(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    schedule = [
        (1.0, 1e3, 3000, 0.12, 0.012),
        (1e3, 1e5, 3000, 0.012, 0.0012),
        (1e5, 1e6, 4000, 0.0012, 0.00012),
    ]

    for a0_alpha, a1_alpha, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # move mass away from the pair creating the current max
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j
        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # occasional aggressive pruning
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Candidate generator ----------------------------------------------------
def _generate_initial_candidates(n):
    """A small but diverse pool of starting points (all summed to 1)."""
    cand = []

    # simple families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))   # arcsine

    # a more exhaustive grid of symmetric betas
    for p in np.concatenate(
        (np.logspace(-2, -0.2, 12), np.linspace(0.1, 2.0, 20))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixture of a Beta with the uniform – w is explored later more finely
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.0, 0.25, 0.5, 0.75, 1.0):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # several Dirichlet draws
    for alpha in (0.02, 0.05, 0.2, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start – k ≈ sqrt(n)
    k = max(2, int(np.sqrt(n)))
    cand.append(_distribution_sparse_random(n, k))

    # random shuffles of a few bases (break symmetry)
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    # ensure numeric safety
    out = []
    eps = 1e-12
    for v in cand:
        v = np.maximum(v, eps)
        v /= v.sum()
        out.append(v)
    return out

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain Python list) found before the deadline.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # a small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []          # list of (score, vector)

    # ------------------------------------------------------------------
    # 1. Multi‑scale exploration – we go from small to larger n
    scales = [256, 512, 1024, 2048, 4096, 8192]

    for n in scales:
        if time.time() >= deadline:
            break

        print(f"[{time.time():.1f}] >>> exploring length {n}")

        # a) generate a modest set of analytic candidates
        for vec in _generate_initial_candidates(n):
            if time.time() >= deadline:
                break
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] new global best len={n}  score={sc:.6f}")

            # keep a tiny elite pool (size ≤6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

        # b) polish the elite members with the three standard local stages
        for idx in range(min(3, len(elite))):
            if time.time() >= deadline:
                break
            a0 = elite[idx][1].copy()
            print(f"[{time.time():.1f}]  local polish of elite #{idx} (len={n})")

            a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)
            a_mid, sc_mid = _projected_subgradient(a_mid, deadline)
            a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

            if sc_mid < best_score - 1e-12:
                best_score = sc_mid
                best_seq = a_mid.copy()
                print(f"[{time.time():.1f}]  improved global best via elite #{idx} -> {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 2. Hierarchical up‑sampling (use the current best as a seed)
    if best_seq is not None:
        current = best_seq.copy()
        for n in (16384, 32768):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] up‑sampling to length {n}")
            up = _interpolate_sequence(current, n)

            # a short round of the three local stages
            up, sc_up = _smooth_max_mirror_descent(up, deadline)
            up, sc_up = _projected_subgradient(up, deadline)
            up, sc_up = _hill_climb_prune(up, deadline)

            # deterministic gradient‑guided move
            up, sc_up = _deterministic_move(up, deadline)

            if sc_up < best_score - 1e-12:
                best_score = sc_up
                best_seq = up.copy()
                print(f"[{time.time():.1f}] new best after up‑sampling length {n} -> {sc_up:.6f}")

            current = up

    # ------------------------------------------------------------------
    # 3. Final stochastic fine‑tuning (simulated annealing)
    if best_seq is not None:
        print(f"[{time.time():.1f}] final simulated‑annealing polishing")
        best_seq, best_score = _simulated_annealing(best_seq, deadline)

    # ------------------------------------------------------------------
    # 4. Clean‑up and return
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
