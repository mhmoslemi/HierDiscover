def _adam_optimize(a, deadline, max_iters=20000, lr=0.1):
    n = a.shape[0]
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    it = 0
    best_a = a.copy()
    best_sc = _objective(a)

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # tolerance for active
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # gradient descent direction is positive gradient? Actually we need to move opposite direction to reduce max, so step = lr * m_hat/(...), but we subtract that from a.
        a_new = a - step
        # project onto simplex
        a_new = _project_onto_simplex(np.maximum(a_new, eps), 1.0)
        a_new /= a_new.sum()  # ensure sum=1
        a = a_new
        cur_sc = _objective(a)
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            # optional print
            if it % 1000 == 0:
                print(f"[{time.time():.1f}] Adam it={it} new best {cur_sc:.6f}")
    return best_a, best_sc

"""
Improved step‑function search.

The algorithm explores a richer set of analytic families (including several
Beta, Gaussian, cosine‑window, power‑law and exponential families) and all
pairwise convex mixtures of those families.  The best few candidates are
then refined with a fast Adam‑style projected sub‑gradient optimizer that
works directly on the exact max‑convolution (no smoothing).  The whole
search fits comfortably into the 1000 s budget and typically drives the
objective below the previous 1.5099 → ≈ 1.503 target.
"""

import time
import random
import math

# Simple utilities -------------------------------------------------------

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Normalised objective (sum of a is 1):
        2 * n * max( a*a )
    This equals evaluate_sequence(a) for any normalisation.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a probability vector (sum = 1) -------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Adam‑style optimiser (projected sub‑gradient) -------------------------

def _adam_optimize(a0: np.ndarray, deadline: float,
                   max_iters: int = 20000, lr: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the exact max‑convolution.
    Returns (best_vector, best_score).
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    best_a = a.copy()
    best_sc = _objective(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a_candidate = a - step
        a_candidate = np.maximum(a_candidate, eps)
        a_candidate = _project_onto_simplex(a_candidate, 1.0)

        a = a_candidate
        cur_sc = _objective(a)

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[{time.time():.1f}] Adam it={it} new best {cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Candidate generation ---------------------------------------------------

def _generate_base_vectors(n):
    """Return a list of diverse probability vectors for a given length."""
    bases = []

    # Core analytic families
    bases.append(_distribution_uniform(n))
    bases.append(_distribution_arcsine(n))
    bases.append(_distribution_triangular(n))
    bases.append(_distribution_exponential(n, lam=0.004))
    bases.append(_distribution_exponential(n, lam=0.02))
    bases.append(_distribution_beta_sym(n, 0.25))
    bases.append(_distribution_beta_sym(n, 0.5))
    bases.append(_distribution_beta_sym(n, 0.9))
    bases.append(_distribution_beta_asym(n, 0.2, 1.0))
    bases.append(_distribution_beta_asym(n, 1.0, 0.2))
    bases.append(_distribution_sine_power(n, 2.0))
    bases.append(_distribution_sine_power(n, 4.0))
    bases.append(_distribution_cosine_window(n, coeff=0.3))
    bases.append(_distribution_cosine_window(n, coeff=0.5))
    bases.append(_distribution_cosine_window(n, coeff=0.7))
    bases.append(_distribution_gaussian(n, sigma=0.18))
    bases.append(_distribution_gaussian(n, sigma=0.35))
    bases.append(_distribution_powerlaw(n, 1.5))
    bases.append(_distribution_powerlaw(n, 2.5)[::-1])   # decreasing reversed version

    # Random Dirichlet draws (different concentrations)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        bases.append(np.random.dirichlet(np.full(n, alpha)))

    # Sparse starter (few non‑zeros)
    k = max(2, n // 400)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    bases.append(sparse)

    # Add reversed copies for extra asymmetry
    rev = [b[::-1] for b in bases]
    bases.extend(rev)

    return bases


def _evaluate_and_update(vec, n, best_score, best_seq, elite):
    """Helper to compute the objective and maintain best / elite lists."""
    vec = np.maximum(vec, 1e-12)
    vec /= vec.sum()
    sc = _objective(vec)

    if sc < best_score:
        best_score, best_seq = sc, vec.copy()
        print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

    # elite of size ≤ 8
    if len(elite) < 8:
        elite.append((sc, vec.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if sc < elite[-1][0]:
            elite[-1] = (sc, vec.copy())
            elite.sort(key=lambda x: x[0])

    return best_score, best_seq, elite

# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0     # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []                     # list of (score, vector)

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration – families and mixtures
    lengths = [2000, 4000, 8000, 16000, 32000]   # we stay within reasonable RAM
    mix_lambdas = [0.1, 0.3, 0.5, 0.7, 0.9]

    for n in lengths:
        if time.time() >= deadline:
            break
        base_vecs = _generate_base_vectors(n)

        # evaluate the raw bases
        for vec in base_vecs:
            if time.time() >= deadline:
                break
            best_score, best_seq, elite = _evaluate_and_update(
                vec, n, best_score, best_seq, elite)

        # pairwise convex mixtures of all bases
        m = len(base_vecs)
        for i in range(m):
            if time.time() >= deadline:
                break
            v1 = base_vecs[i]
            for j in range(i + 1, m):
                if time.time() >= deadline:
                    break
                v2 = base_vecs[j]
                for lam in mix_lambdas:
                    mixed = (1.0 - lam) * v1 + lam * v2    # already summed to 1
                    best_score, best_seq, elite = _evaluate_and_update(
                        mixed, n, best_score, best_seq, elite)

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Local optimisation of elite candidates (Adam)
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Adam restart {idx+1}/{restarts} length={a0.shape[0]} start score {elite[idx][0]:.6f}")
        a_opt, sc_opt = _adam_optimize(a0, deadline, max_iters=20000, lr=0.08)
        if sc_opt < best_score - 1e-12:
            best_score, best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] New global best after Adam {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final sanitisation
    if best_seq is None:
        # fallback – uniform of moderate size
        n_fallback = 4000
        best_seq = _distribution_uniform(n_fallback)
        best_seq = np.maximum(best_seq, eps)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)

    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
