    data = []
    rows = []

        end_i = min(shift, n - 1)
        # i from start_i to end_i inclusive
        # set A[row_idx, i] = f[i]
        num_entries = end_i - start_i + 1
        data.extend(f[start_i:end_i+1].tolist())
        rows.extend([row_idx] * num_entries)

        g0 = res.x
        if g0.sum() <= 0:
            return None
        g = g0 / g0.sum()
        return g
    else:
        return None

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input is assumed non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i·(n−i))^{−p} , i indexed from 0."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(α,β) density on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Dirichlet vector with symmetric concentration α."""
    a = rng.dirichlet(np.full(n, alpha))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a_sum
    return a


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mix lam·uniform + (1−lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# SciPy‐based windows (if available)
# ----------------------------------------------------------------------
def _distribution_window(name: str, n: int, param) -> np.ndarray:
    """Generate a window via ``scipy.signal.get_window`` and normalize."""
    if param is None:
        w = get_window(name, n, fftbins=False)
    else:
        w = get_window((name, param), n, fftbins=False)
    w = np.maximum(np.asarray(w, dtype=np.float64), 0.0)  # safety
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w


def _distribution_dpss(n: int, nw: float) -> np.ndarray:
    """First Slepian (DPSS) window, normalised to sum = 1."""
    # dpss returns (Kmax, n); we request only the 0‑th eigenvector.
    w = dpss(n, nw, Kmax=1, sym=True)[0].astype(np.float64)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        w = w / s
    return w


# ----------------------------------------------------------------------
# Linear‑programming boost (requires SciPy)
# ----------------------------------------------------------------------
def _lp_boost(f: np.ndarray, M: float, active_k: int = 5) -> np.ndarray:
    """
    Solve a tiny LP that maximises Σg subject to g≥0 and (f·g)≤M on the
    ``active_k`` largest self‑convolution entries of ``f``.
    Returns the improved vector (still normalised to sum = 1) or ``f``
    if the LP fails or yields no improvement.
    """
    if not _HAS_OPT:
        return f

    n = f.shape[0]
    # --------------------------------------------------------------
    # 1. Identify the most binding shifts of the current self‑conv.
    # --------------------------------------------------------------
    conv_self = _fft_self_convolve(f)
    idxs = np.argpartition(-conv_self, range(active_k))[:active_k]
    idxs = np.unique(idxs)

    # --------------------------------------------------------------
    # 2. Build a sparse constraint matrix A·g ≤ M.
    #    Row r corresponds to shift s = idxs[r].
    # --------------------------------------------------------------
    data, rows, cols = [], [], []
    for r, s in enumerate(idxs):
        i_start = max(0, s - (n - 1))
        i_end = min(s, n - 1)
        # contribution of f[i] to (f·g)[s] is f[i]·g[s‑i]
        segment = f[i_start : i_end + 1]
        data.extend(segment.tolist())
        rows.extend([r] * segment.size)
        cols.extend(list(range(i_start, i_end + 1)))
    A_ub = _sp.coo_matrix((data, (rows, cols)), shape=(len(idxs), n)).tocsr()

    # --------------------------------------------------------------
    # 3. Linear programme:   maximise 1ᵀg   ⇔   minimise –1ᵀg
    # --------------------------------------------------------------
    c = -np.ones(n)                 # minimise –sum(g)
    bounds = [(0, None)] * n
    res = _opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=np.full(len(idxs), M),
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success or res.x.sum() <= 0.0:
        return f

    g0 = res.x
    # --------------------------------------------------------------
    # 4. Mix the new vector with the original one.
    #    Simple convex combination (t = 0.4) followed by renormalisation.
    # --------------------------------------------------------------
    t = 0.4
    g = g0 / g0.sum()
    new = (1.0 - t) * f + t * g
    new = new / new.sum()
    # verify that the LP constraints are still respected (optional)
    if _fft_self_convolve(f)[: 2 * n - 1].max() < M * 1.001:
        # keep improvement only if objective really got better
        if _objective(new) < _objective(f):
            return new
    return f


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑conv objective
# ----------------------------------------------------------------------
def _subgradient_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    top_k: int = 5,
    eval_interval: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on 2·n·max(conv(a,a)).
    Gradient is accumulated from the ``top_k`` largest convolution entries.
    The iterate is always projected onto the probability simplex.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        # indices of the largest conv entries (including zero shift)
        idxs = np.argpartition(-conv, range(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            inds = idx - np.arange(n)
            mask = (inds >= 0) & (inds < n)
            grad[mask] += 2.0 * a[inds[mask]]

        eta = lr0 / math.sqrt(t)            # decaying stepsize
        a = a * np.exp(-eta * grad)         # exponentiated‑gradient update
        a = _project_onto_simplex(a, s=1.0)  # maintain Σa = 1

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}"
                )
        # safety cap – an iteration budget that is generous but still finite
        if t >= 500_000:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy local polishing (pairwise mass transfers)
# ----------------------------------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from i→j; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.4   # up to 40 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a vector of non‑negative heights that minimises
    ``evaluate_sequence``.  The function respects the supplied time budget
    and returns a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0            # leave a small safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Scan a diverse pool of analytic / window families.
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 16000, 32000, 50000]  # increasing discretisation
    # grids for the different parametric families
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.5, 12, base=10),  # 0.01 … 0.316
            np.logspace(-0.5, 0.0, 12, base=10),  # 0.316 … 1
            np.logspace(0.0, 0.7, 10, base=10),   # 1 … ~5
        ]
    )
    lambda_grid = np.linspace(0.0, 1.0, 6)                # 0 … 1
    # windows that accept a single real parameter (or none)
    window_specs = [
        ("hann", None),
        ("hamming", None),
        ("blackman", None),
        ("tukey", np.linspace(0.0, 1.0, 5)),
        ("kaiser", np.linspace(0.0, 15.0, 5)),
        ("chebwin", np.linspace(30, 120, 5)),
        ("gaussian", np.linspace(0.3, 5.0, 5)),
        ("exponential", np.linspace(0.3, 5.0, 5)),
    ]
    dpss_params = np.linspace(2.0, 6.0, 5) if _HAS_SCI else []

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # ---- basic deterministic families ------------------------------------
        uniform = _distribution_uniform(n)
        arcsine = _distribution_arcsine(n)
        for cand in (uniform, arcsine):
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] new global best len={n} score={sc:.6f}"
                )

        # ---- symmetric beta & edge‑weighted families ------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)

            for base in (beta, edge):
                for lam in lambda_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # ---- asymmetric beta (randomly sampled) ---------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.1, 3.0)
            beta = rng.uniform(0.1, 3.0)
            cand = _distribution_beta_asym(n, alpha, beta)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asym‑beta len={n} α={alpha:.2f} β={beta:.2f} score={sc:.6f}"
                )

        # ---- SciPy windows -------------------------------------------------
        if _HAS_SCI:
            for name, grid in window_specs:
                if time.time() >= deadline:
                    break
                if grid is None:
                    cand = _distribution_window(name, n, None)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best win={name} len={n} score={sc:.6f}"
                        )
                else:
                    for par in grid:
                        if time.time() >= deadline:
                            break
                        cand = _distribution_window(name, n, par)
                        sc = _objective(cand)
                        top_candidates.append((sc, cand.copy()))
                        if sc < overall_best_score:
                            overall_best_score = sc
                            overall_best_seq = cand.copy()
                            print(
                                f"[{time.time():.1f}] new win={name} param={par:.2f} len={n} score={sc:.6f}"
                            )

            # ---- DPSS (Slepian) windows ------------------------------------
            for nw in dpss_params:
                if time.time() >= deadline:
                    break
                cand = _distribution_dpss(n, nw)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(
                        f"[{time.time():.1f}] new DPSS nw={nw:.2f} len={n} score={sc:.6f}"
                    )

        # ---- random Dirichlet seeds ----------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = _distribution_random_dirichlet(n, rng, alpha)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # ---- sparse random seeds --------------------------------------------
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the best ~30 candidates overall to limit memory/time
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Include a previously known construction if it exists in the env.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback in the unlikely case no candidate was generated.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 8000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  LP‑based refinement on a handful of top seeds.
    # ------------------------------------------------------------------
    if _HAS_OPT:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        for idx, seed in enumerate(seeds):
            if time.time() >= deadline:
                break
            M = _fft_self_convolve(seed).max()
            improved = _lp_boost(seed, M, active_k=7)
            sc = _objective(improved)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = improved.copy()
                print(
                    f"[{time.time():.1f}] LP boost seed {idx} new best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Sub‑gradient descent on the most promising seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # Pick the 5 currently best seeds
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(
            f"[{time.time():.1f}] launching sub‑gradient descent on {len(seeds)} seeds (≈{per_seed:.1f}s each)"
        )
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_multi(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                top_k=5,
                eval_interval=50,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 6.  Final greedy pairwise polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise‑swap polishing")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # Return a plain Python list – ``evaluate_sequence`` will handle any needed normalisation.
    return overall_best_seq.tolist()
