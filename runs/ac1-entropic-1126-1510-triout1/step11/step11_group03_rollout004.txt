def _adam_smooth_descent(a, deadline, best_seq, best_score, ...):
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.5  # base learning rate
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        alpha = alpha_start * ((alpha_end/alpha_start) ** (t / max_iter))  # exponential schedule
        # compute softmax weights
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad = _gradient_from_weights(a, w)
        # adam update for "log a" (multiplicative)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)  # shape (n,)
        # maybe scale step using sqrt of iteration index to keep small
        step = step / np.sqrt(t)  # similar to earlier schedule
        # compute multiplicative factor
        a = a * np.exp(-step)
        # enforce nonneg and renormalize
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam iter={t} α={alpha:.2f} new best {best_score:.6f}")
    return best_seq, best_score

"""
Step‑function optimizer (≈ 100 s per seed).

The algorithm builds on the previous “soft‑max gradient descent’’ but
adds a few decisive changes that helped push the bound from 1.5099
towards the target 1.503:

* **Adam‑style adaptive step‑size** – we keep first‑ and second‑moment
  estimates of the exponentiated‑gradient direction and scale the
  multiplicative update per‑coordinate.  This stabilises the early
  rapid descent while still allowing fine adjustments later on.

* **Exponential temperature schedule** for the soft‑max that replaces
  the linear schedule.  The temperature grows from a modest value up
  to ~50, making the approximation of the max sharper only when the
  iterate is already well‑behaved.

* **Longer multi‑start phase** – a rich library of analytic families
  (uniform, arcsine, Beta, power‑law, exponential, two‑block, sparse
  and Dirichlet) is sampled for several lengths (2 k – 16 k).  The 6
  best seeds are refined with Adam.

* **Final greedy polishing** – a cheap random mass‑transfer walk that
  accepts only improving moves.  It is executed after the intensive
  descent and usually yields a small extra gain.

All heavy work (convolutions, gradients) is performed with real FFTs,
so each iteration costs O(n log n).  The code respects the global time
budget (default 1000 s) and always returns a valid non‑negative list
of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic seed families ------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # a few Dirichlet draws with random α
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand

def _adam_smooth_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr: float = 0.5,
                         max_iter: int = 200_000,
                         eval_interval: int = 500,
                         alpha_start: float = 0.2,
                         alpha_end: float = 50.0,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on a soft‑max
    approximation of the max‑convolution term.
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # exponential temperature schedule
        alpha = alpha_start * ((alpha_end / alpha_start) ** (t / max_iter))

        # stable soft‑max weights
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        # Adam updates (on log‑space, i.e. multiplicative)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step = step / math.sqrt(t)          # gentle decay like 1/√t

        # multiplicative (exponentiated) update & renormalisation
        a = a * np.exp(-step)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer walk that keeps a move only if the
    true objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

            top_candidates.append((sc, cand.copy()))

        # retain a small elite set overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 2️⃣  Include previous solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Adam‑smooth descent on the elite seeds
    if best_seq is None:                # fallback – should never happen
        n = 2048
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)

    top_candidates.sort(key=lambda x: x[0])
    elite = [seq for _, seq in top_candidates[:6]]

    remaining = deadline - time.time()
    if remaining > 30.0 and elite:
        per_seed = max(5.0, remaining / len(elite))
        for seed_seq in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            seq_out, score_out = _adam_smooth_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq,
                best_score,
                lr=0.5,
                max_iter=200_000,
                eval_interval=500,
                alpha_start=0.2,
                alpha_end=50.0,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
            )
            if score_out < best_score:
                best_score, best_seq = score_out, seq_out.copy()
                print(f"[{time.time():.1f}] after Adam new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final greedy polishing
    if time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
        )

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
