def _multimax_descent_adagrad(a, deadline, best_seq, best_score, eta0=0.5, max_iter=50000, eval_interval=200):
    n = a.shape[0]
    sum_sq = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-12
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt
        grad = _gradient_from_weights(a, w)
        sum_sq += grad*grad
        eta = eta0 / np.sqrt(sum_sq + 1e-12)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum

                best_score = cur
                best_seq = a.copy()

def _annealing_random_moves(a, deadline, best_seq, best_score, T0=1.0, cooling=0.999):
    T = T0
    while time.time() < deadline:
        # propose random transfer:
        i = np.random.choice(np.where(a>1e-12)[0])
        j = np.random.randint(len(a))
        if i==j: continue
        delta = a[i] * np.random.rand() * 0.1
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        # accept with probability:
        if cur < best_score or np.exp(-(cur - best_score)/T) > np.random.rand():
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] anneal i={i} j={j} cur={cur:.6f}")
        T *= cooling
    return best_seq, best_score

"""
Hybrid optimiser for the max‑convolution step‑function problem.

The optimiser works in three stages:

1. **Broad seeding** – a large collection of analytic families (uniform,
   arcsine, symmetric β‑distributions, random Dirichlet, sparse draws, …)
   is generated for several lengths (≈2k–12k).  Their exact objective is
   evaluated with an FFT‑based convolution and the best few seeds are kept.

2. **Continuous refinement** – each kept seed is processed by two
   gradient‑descent phases that use a *soft‑max* surrogate of the max‑convolution.
   The gradient is computed via FFT cross‑correlation and an AdaGrad‑style
   adaptive step‑size is applied with an exponentiated‑gradient (mirror‑descent)
   update, followed by a Euclidean projection onto the probability simplex.
   The temperature α of the soft‑max is increased from 1 to a large value
   (≈200) so that the surrogate gradually becomes the true max.

   After the smooth phase a classic multimax sub‑gradient descent (active‑set
   averaging) further polishes the solution.

3. **Local polishing** – a cheap greedy “mass‑transfer” walk swaps a random
   amount of weight between two positions only when it improves the exact
   objective.

All phases respect the global time budget (≈1000 s).  The best sequence found
is returned as a plain list of floats (non‑negative, sum normalised to 1).
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a∗a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes sum>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# 2.   Analytic families (probability vectors, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((p - 1) * np.log(np.clip(x, eps, 1.0))
               + (p - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# 3.   Gradient‑descent phases (smooth‑max & multimax) – AdaGrad version
# ----------------------------------------------------------------------
def _smooth_max_descent_adagrad(a: np.ndarray,
                                deadline: float,
                                best_seq: np.ndarray,
                                best_score: float,
                                eta0: float = 0.5,
                                alpha_start: float = 1.0,
                                alpha_end: float = 200.0,
                                max_iter: int = 50000,
                                eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    AdaGrad exponentiated‑gradient descent on the smooth‑max surrogate:
        f_α(a) = (1/α) log Σ_i exp(α·conv_i(a)).
    α is increased geometrically from `alpha_start` to `alpha_end`.
    """
    n = a.shape[0]
    t = 0
    sum_sq = np.zeros_like(a) + 1e-12
    while t < max_iter and time.time() < deadline:
        t += 1
        # temperature schedule (geometric)
        alpha = alpha_start * ((alpha_end / alpha_start) ** (t / max_iter))

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        sum_sq += grad * grad
        eta = eta0 / np.sqrt(sum_sq)

        a = a * np.exp(-eta * grad)

        # simplex projection (keeps sum = 1)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth  it={t} α={alpha:.2f}  obj={cur:.6f}")

    return best_seq, best_score


def _multimax_descent_adagrad(a: np.ndarray,
                              deadline: float,
                              best_seq: np.ndarray,
                              best_score: float,
                              eta0: float = 0.5,
                              max_iter: int = 50000,
                              eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    AdaGrad exponentiated‑gradient descent using the exact active set
    (indices where the convolution attains its maximum).
    """
    n = a.shape[0]
    t = 0
    sum_sq = np.zeros_like(a) + 1e-12
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)
        sum_sq += grad * grad
        eta = eta0 / np.sqrt(sum_sq)

        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t}  obj={cur:.6f}")

    return best_seq, best_score


def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """
    Simple greedy walk: transfer a random fraction of mass from a
    source index to a destination index, keep the move only if the exact
    objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src_idx = np.where(a > 1e-12)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)
        j = rng.integers(0, n - 1)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g}  obj={cur:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
# 4.   Main optimisation driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine runs for up to ``budget_s`` seconds and returns the best
    sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 3.0           # safety margin

    # --------------------------------------------------------------
    # 4.1  Broad seeding – a few analytic families for several lengths
    # --------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000]                # moderate sizes
    seed_vectors: List[np.ndarray] = []
    for n in lengths:
        # uniform
        seed_vectors.append(_distribution_uniform(n))
        # arcsine (β(0.5,0.5))
        seed_vectors.append(_distribution_arcsine(n))

        # symmetric beta – dense grid of shape parameters
        for p in np.linspace(0.05, 2.0, 22):           # 22 values gives fine granularity
            seed_vectors.append(_distribution_beta_sym(n, p))

        # random Dirichlet draws
        for _ in range(3):
            alpha = 0.1 + 0.9 * rng.random()
            seed_vectors.append(_distribution_dirichlet(n, rng, alpha=alpha))

        # sparse random patterns (few spikes)
        seed_vectors.append(_distribution_random_sparse(n, rng, density=0.02))

    # Include the previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            seed_vectors.append(prev.copy())
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4.2  Evaluate all seeds, keep the most promising few
    # --------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_score = float("inf")
    scored_seeds: List[Tuple[float, np.ndarray]] = []

    for vec in seed_vectors:
        if time.time() >= deadline:
            break
        a = np.maximum(vec, 0.0)
        total = a.sum()
        if total == 0.0:
            continue
        a = a / total
        score = _objective(a)
        scored_seeds.append((score, a.copy()))
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] initial best len={len(a)}  obj={score:.6f}")

    # keep only a handful of top seeds for the costly gradient phases
    scored_seeds.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in scored_seeds[:12]]

    # --------------------------------------------------------------
    # 4.3  Continuous refinement – smooth‑max descent → multimax → swaps
    # --------------------------------------------------------------
    # Allocate remaining time roughly uniformly among the seeds
    while time.time() < deadline and top_seeds:
        # choose the next seed (best‑first order)
        a = top_seeds.pop(0)
        # ------------------------------------------------------------------
        # smooth‑max phase
        sub_dead = time.time() + (deadline - time.time()) * 0.5
        a, sc = _smooth_max_descent_adagrad(
            a.copy(),
            sub_dead,
            best_seq if best_seq is not None else a,
            best_score,
            eta0=0.7,
            alpha_start=1.0,
            alpha_end=200.0,
            max_iter=50000,
            eval_interval=400,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] after smooth descent new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # multimax phase
        sub_dead = time.time() + (deadline - time.time()) * 0.4
        a, sc = _multimax_descent_adagrad(
            a.copy(),
            sub_dead,
            best_seq if best_seq is not None else a,
            best_score,
            eta0=0.7,
            max_iter=50000,
            eval_interval=500,
        )
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] after multimax descent new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # cheap local polishing
        if time.time() < deadline:
            a, sc = _pairwise_swap_local(
                a.copy(),
                deadline,
                best_seq if best_seq is not None else a,
                best_score,
            )
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time():.1f}] after local swaps new best {best_score:.6f}")

        # re‑insert the refined seed at the end if we still have time
        if time.time() < deadline:
            top_seeds.append(a)   # allow another pass if budget permits

    # --------------------------------------------------------------
    # 4.4  Fallback – should never be needed
    # --------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform  obj={best_score:.6f}")

    # --------------------------------------------------------------
    # 5.   Return the best sequence (as a plain list of floats)
    # --------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
