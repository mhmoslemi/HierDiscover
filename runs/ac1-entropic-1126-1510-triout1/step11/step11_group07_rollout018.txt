"""
Propose a step‑function (non‑negative list of floats) that minimizes
`evaluate_sequence`.  The search proceeds in three phases:

1. **Broad exploration** – many analytic families (symmetric / asymmetric
   Beta, edge‑biased power laws with an added constant, a few Dirichlet
   draws) are instantiated for several lengths up to 2·10⁵.  The best
   few candidates are retained.

2. **Refinement** – the current overall best candidate is improved with
   a smooth‑max mirror‑descent (exponentiated‑gradient) phase followed
   by a projected Adam‑style sub‑gradient phase.  Both respect the
   simplex constraint (sum = 1, non‑negative).

3. **Polishing** – a tiny mass‑transfer hill‑climber performs many
   local moves, moving a little probability mass from the most
   contributing pair to the smallest entry.  The best sequence found
   before the deadline is returned as a plain Python list.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1,
    returns the exact objective value produced by `evaluate_sequence`
    (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on the discrete grid (both parameters >0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # protect against log(0)
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)            # for numerical stability
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Convenient wrapper for symmetric Beta(p,p)."""
    return _distribution_beta(n, p, p)


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """Random Dirichlet draw whose concentration is larger at the edges."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _smooth_max_mirror_step(a: np.ndarray,
                            alpha: float,
                            lr: float,
                            eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective  f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector and the raw maximal convolution.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max over the convolution entries
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # grad = ∂f_α/∂a = 2 * a (convolved with w)   (see derivation in the docstring)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]
    a_new = a * np.exp(-lr * grad)          # exponentiated‑gradient update
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, m


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One Adam‑style projected sub‑gradient step on the true max‑convolution.
    Returns (new_a, new_velocity, current_max_before_step).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices achieving the current max, tolerance for numerics)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

    # Adam update (bias‑corrected first moment, ignore second moment for simplicity)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search
    step = step0
    while True:
        cand = a - step * v_hat
        cand = _project_onto_simplex(cand, 1.0)
        cand = np.maximum(cand, eps)
        cand /= cand.sum()
        cand_score = _max_convolution_score(cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return cand, vel, cur_max


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Hill‑climber that repeatedly moves a tiny amount of probability mass
    from the index that participates in the current peak toward the
    globally smallest entry.  The loop stops after `max_iters` or when
    the deadline is exceeded (checked internally).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) that realises the maximal product for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]  # second smallest

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional jitter to avoid getting stuck
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a hard time budget (default
    1000 s) and returns the best sequence found before the deadline.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 1.0  # 1 s safety margin

    eps = 1e-12

    # ------------------------------------------------------------------
    # 0️⃣  Load a previously saved construction if present (may be better)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            best_seq = prev.copy()
            best_score = _max_convolution_score(best_seq)
            print(f"[{time.time():.1f}] loaded previous best score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration over several lengths and families
    # ------------------------------------------------------------------
    lengths = [20000, 50000, 100000, 200000]   # longer sequences give finer resolution
    # symmetric Beta parameters (very fine near the singular end)
    p_grid_fine = np.concatenate(
        [np.linspace(0.02, 0.12, 11), np.linspace(0.12, 0.5, 20)]
    )
    # edge‑biased power‑law exponents and small additive constants
    alpha_grid = [0.30, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70]
    c_grid = [0.0, 1e-5, 5e-5, 1e-4, 5e-4]

    # number of random asymmetric Beta draws per length
    n_rand_beta = 12

    # keep the top K candidates globally (small K to stay cheap)
    K = 10
    top_candidates = []  # list of (score, vector)

    def _consider(vec, name):
        """Normalise `vec`, evaluate, and possibly update the global best."""
        nonlocal best_seq, best_score, top_candidates
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score - 1e-12:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] {name}  new best {sc:.6f}")
        top_candidates.append((sc, vec.copy()))

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- deterministic families ------------------------------------------------
        # arcsine (Beta(½,½))
        _consider(_distribution_arcsine(n), f"n={n} arcsine")

        # symmetric Beta(p,p) – fine grid
        for p in p_grid_fine:
            if time.time() >= deadline:
                break
            _consider(_distribution_beta_sym(n, p), f"n={n} Beta(p={p:.3f})")

        # edge‑biased power‑law with optional flat floor
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            for c in c_grid:
                if time.time() >= deadline:
                    break
                _consider(_distribution_edge_power(n, alpha, c),
                          f"n={n} edgePower(α={alpha:.2f},c={c:.5f})")

        # -------------------------------------------------
        # random asymmetric Beta draws (explore a 2‑D parameter space)
        for _ in range(n_rand_beta):
            if time.time() >= deadline:
                break
            p = random.uniform(0.02, 0.6)
            q = random.uniform(0.02, 0.6)
            _consider(_distribution_beta(n, p, q),
                      f"n={n} Beta(p={p:.3f},q={q:.3f})")

        # a few edge‑biased Dirichlet draws
        for _ in range(4):
            if time.time() >= deadline:
                break
            bex = random.uniform(0.2, 0.8)
            _consider(_distribution_edge_dirichlet(n, bex),
                      f"n={n} edge‑Dirichlet(β={bex:.3f})")

    # keep only the K best candidates (including the current best)
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:K]

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max mirror descent on the overall best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of medium size
        n0 = 50000
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    a = best_seq.copy()
    n = a.shape[0]

    # allocate remaining time for mirror descent
    remaining = deadline - time.time()
    # rough cost per iteration (empirical)
    est_per_iter = 0.006 if n > 80000 else 0.003
    max_iters = max(0, int(remaining / est_per_iter))

    # we never want to exceed a few ten‑thousands iterations
    total_iters = max(2000, min(18000, max_iters))
    print(f"[{time.time():.1f}] Mirror‑descent phase: n={n}, iters={total_iters}")

    # temperature and learning‑rate schedules (more gradual than before)
    alpha_start, alpha_end = 0.5, 1e5
    lr_start, lr_end = 0.25, 0.0008

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)

        # occasional exact score check
        if (it + 1) % max(1, total_iters // 10) == 0 or it == total_iters - 1:
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] mirror it={it}/{total_iters} new best {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected Adam‑style sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.6
    sub_it = 0

    while time.time() < deadline - 0.5:
        sub_it += 1
        a, velocity, cur_max = _adam_subgrad_step(
            a, velocity, step0, beta1, beta2, sub_it, eps
        )
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cur_score:.6f}")

        # occasional jitter to escape shallow plateaus
        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 4️⃣  Tiny mass‑transfer hill‑climber for final polishing
    # ------------------------------------------------------------------
    a = best_seq.copy()
    max_hill_iters = 10000
    a = _mass_transfer_hill_climb(a, max_hill_iters, eps)
    final_score = _max_convolution_score(a)
    if final_score < best_score:
        best_seq, best_score = a.copy(), final_score
        print(f"[{time.time():.1f}] final hill‑climber improved score to {final_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
