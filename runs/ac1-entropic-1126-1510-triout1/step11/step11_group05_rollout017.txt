def _conv_matrix_sparse(f):
    from scipy.sparse import diags
    n = f.shape[0]
    data = [np.full(n, val) for val in f]
    offsets = np.arange(n)
    return diags(data, offsets, shape=(2*n-1, n), format='csr')

def _lp_mix_step(a, deadline):
    n = a.shape[0]
    # compute max conv
    conv = _fft_self_convolve(a)
    M = conv.max()
    A = _conv_matrix_sparse(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # Solve
    try:
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            raise Exception('LP failed')
        g0 = res.x
        s = g0.sum()
        if s <= 0:
            g = a
        else:
            g = g0 / s
    except Exception:
        # fallback
        g = a
    # mixing
    best_f = a
    best_M = M
    for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):
        f_new = (1 - t) * a + t * g
        conv_new = _fft_self_convolve(f_new)
        M_new = conv_new.max()
        if M_new < best_M - 1e-15:
            best_f = f_new
            best_M = M_new
    return best_f, best_M

def _lp_iterative_optimize(a0, max_passes, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for i in range(max_passes):
        if time.time() >= deadline:
            break
        a_new, M_new = _lp_mix_step(a, deadline)
        score_new = 2*len(a_new)*M_new

            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            break
    return best_a, best_score

"""
A multi‑phase optimizer for the “max‑convolution” problem.

1.  Global exploration – generate many analytic families (uniform,
    arcsine, symmetric Beta(p,p), mixtures with the uniform,
    Dirichlet draws, sparse starts) for a variety of lengths and keep a
    small elite list of the best candidates (objective = 2·n·maxconv
    for a vector normalised to sum = 1).

2.  Local refinement – for each elite element we apply a sequence of
    optimisers:
        * A few passes of a linear‑programming “mix” step (maximise
          ∑g subject to f∗g ≤ M) followed by a short line‑search on the
          mixing factor.
        * Smooth‑max mirror descent (exponential‑gradient) with a
          temperature schedule up to α≈1e⁶.
        * Projected sub‑gradient descent with Adam‑style momentum and a
          back‑tracking line‑search.
        * Simulated‑annealing random transfers of mass.
        * Simple hill‑climber with aggressive pruning of tiny entries.

    Each phase checks the remaining time and stops early if needed.

3.  The best vector (still normalised to sum = 1) is finally scaled to
    the required normalisation ∑a = √(2 n) – the evaluation function
    then returns exactly the maximal convolution value, which we have
    minimised.

The routine returns a plain Python list of non‑negative floats.
"""

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform (both already normalised)."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    return np.random.dirichlet(np.full(n, alpha))


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a


def _generate_candidates(n):
    """Return a list of candidate vectors (already normalised to sum = 1)."""
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))   # arcsine

    # dense grid of symmetric Betas, extended towards the edges
    p_vals = np.concatenate((
        np.logspace(-4, -0.2, 14),          # very U‑shaped
        np.linspace(0.1, 2.0, 20)           # moderate shapes
    ))
    for p in p_vals:
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with the uniform distribution, finer weight grid
    for p in (0.01, 0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.0, 0.1, 0.25, 0.5, 0.75, 0.9, 1.0):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # several Dirichlet draws
    for alpha in (0.01, 0.05, 0.2, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start (few non‑zeros)
    k = max(2, n // 300)
    cand.append(_distribution_sparse_random(n, k))

    # random shuffles of a few bases to introduce asymmetry
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Mirror descent on smooth max ------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """Exponentiated‑gradient descent on a smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e4, 4000, 0.12, 0.012),
        (1e4, 1e5, 4000, 0.012, 0.0012),
        (1e5, 1e6, 5000, 0.0012, 0.00012),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc

def _projected_subgradient(a0, deadline):
    """True sub‑gradient descent with Adam‑style momentum and back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    step0 = 0.7
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc

def _hill_climb_prune(a0, deadline):
    """Simple hill‑climbing moves plus aggressive pruning of tiny entries."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # move mass away from the pair generating the current max
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j

        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # aggressive pruning every few thousand steps
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # occasional random jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Simulated annealing ---------------------------------------------------
def _simulated_annealing(a0, deadline, max_iter=20000):
    """Random mass transfers with Metropolis acceptance."""
    a = a0.copy()
    n = a.shape[0]
    cur_sc = _max_convolution_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    # temperature schedule: start relatively high and cool down linearly
    T0 = cur_sc * 0.1 + 1e-6

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # propose a move
        src = random.randrange(n)
        if a[src] <= 1e-15:
            continue
        tgt = random.randrange(n - 1)
        if tgt >= src:
            tgt += 1
        delta = a[src] * (0.1 * random.random())
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta

        sc_new = _max_convolution_score(a_new)

        # accept according to Metropolis rule
        if sc_new < cur_sc:
            accept = True
        else:
            T = T0 * (1.0 - it / max_iter)
            accept = random.random() < math.exp(-(sc_new - cur_sc) / max(T, 1e-12))

        if accept:
            a = a_new
            cur_sc = sc_new
            if cur_sc < best_sc - 1e-12:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA it={it} new best {best_sc:.6f}")

        if it % 5000 == 0:
            # small jitter to keep the walk lively
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, 1e-12)
            a /= a.sum()
            cur_sc = _max_convolution_score(a)

    return best_a, best_sc

# Linear‑programming mixing step -----------------------------------------
def _conv_matrix_sparse(f):
    """Build the (2n‑1)×n convolution matrix for a fixed f (sum=1)."""
    n = f.shape[0]
    # each diagonal offset i has constant f[i] and length n
    data = [np.full(n, val, dtype=np.float64) for val in f]
    offsets = np.arange(n)
    return diags(data, offsets, shape=(2 * n - 1, n), format='csr')


def _lp_mix_step(a, deadline):
    """
    One LP‑mix iteration:
      * maximise ∑g subject to f∗g ≤ M,
      * rescale to sum = 1,
      * line‑search over a few mixing factors.
    Returns the improved vector and its max‑convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()

    A = _conv_matrix_sparse(a)
    c = -np.ones(n)                         # maximise sum(g)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            raise RuntimeError('LP failed')
        g0 = res.x
        s = g0.sum()
        if s <= 0:
            g = a.copy()
        else:
            g = g0 / s                         # re‑normalise to sum = 1
    except Exception:
        # fallback: keep the current vector
        g = a.copy()

    best_f = a
    best_M = M
    for t in (0.5, 0.25, 0.125, 0.0625, 0.03125):
        f_new = (1.0 - t) * a + t * g
        conv_new = _fft_self_convolve(f_new)
        M_new = conv_new.max()
        if M_new < best_M - 1e-15:
            best_f = f_new
            best_M = M_new
    return best_f, best_M


def _lp_iterative_optimize(a0, max_passes, deadline):
    """Repeated LP‑mix steps (up to max_passes) with early stop."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    for _ in range(max_passes):
        if time.time() >= deadline:
            break
        a_new, M_new = _lp_mix_step(a, deadline)
        sc_new = 2.0 * len(a_new) * M_new
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            a = a_new
        else:
            break
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    best_seq = None
    best_score = float("inf")
    elite = []                                 # (score, vector)

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths
    lengths = [1024, 2048, 4096, 8192]
    # add a few random non‑power‑of‑two lengths
    for _ in range(3):
        lengths.append(random.randint(2000, 8000))
    lengths = sorted(set(lengths))

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()                 # normalise to sum = 1
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")

            # maintain a tiny elite list (size ≤6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Include any previously discovered solution
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case we have nothing
    if best_seq is None:
        n0 = 2048
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(6, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a = elite[idx][1].copy()
        n = a.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} len={n} start score {elite[idx][0]:.6f}")

        # a) LP based mixing (few passes)
        a, sc_lp = _lp_iterative_optimize(a, max_passes=4, deadline=deadline)
        if sc_lp < best_score - 1e-12:
            best_score = sc_lp
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best after LP {sc_lp:.6f}")

        # b) smooth‑max mirror descent
        a, sc_mid = _smooth_max_mirror_descent(a, deadline)
        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best after mirror {sc_mid:.6f}")

        # c) projected sub‑gradient polishing
        a, sc_mid = _projected_subgradient(a, deadline)
        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best after subgrad {sc_mid:.6f}")

        # d) simulated annealing
        a, sc_mid = _simulated_annealing(a, deadline)
        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best after SA {sc_mid:.6f}")

        # e) final hill‑climb and pruning
        a, sc_mid = _hill_climb_prune(a, deadline)
        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a.copy()
            print(f"[{time.time():.1f}] New best after hill {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce positivity, renormalise to sum=√(2·n)
    n = best_seq.shape[0]
    eps = 1e-12
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    # Scale to the normalisation used by evaluate_sequence
    scale = math.sqrt(2.0 * n)
    best_seq *= scale

    final_score = _max_convolution_score(best_seq / scale)   # back to sum‑=1 scale
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
