def _distribution_beta_general(n, alpha, beta, c):
    i = np.arange(n, dtype=np.float64)
    # compute logs to handle extremes
    logw = (alpha-1.0) * np.log(i+0.5) + (beta-1.0) * np.log(n - i - 0.5)
    # shift to avoid overflow
    max_log = np.max(logw)
    w = np.exp(logw - max_log) + c  # adding c after scaling may shift, but okay
    w = np.maximum(w, 0.0)
    total = w.sum()
    if total == 0 or not np.isfinite(total):
        w = np.full(n, 1.0/n)
    else:
        w = w/total
    return w

def _refine_params(a,b,c,n, iters=200):
    best_params = (a,b,c)
    best_vec = _distribution_beta_general(n,a,b,c)
    best_score = _max_convolution_score(best_vec)
    step_factor = 0.2
    for it in range(iters):
        # random relative perturbation
        a_new = max(0.01, best_params[0] * (1 + step_factor * random.uniform(-1,1)))
        b_new = max(0.01, best_params[1] * (1 + step_factor * random.uniform(-1,1)))
        c_new = max(0.0, best_params[2] + step_factor * random.uniform(-0.1,0.1))
        # optionally clamp c to [0, max_c]
        vec = _distribution_beta_general(n, a_new, b_new, c_new)
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score = sc
            best_params = (a_new,b_new,c_new)
            best_vec = vec
        # gradually decrease step factor
        step_factor *= 0.995
    return best_vec, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  The algorithm combines three ideas that were not
used together before:

1️⃣  A low‑dimensional parametric family (a discrete Beta‑like shape):
        w_i ∝ (i+½)^{α‑1} (n‑i‑½)^{β‑1} + c ,   α,β>0 , c≥0 .
    Random sampling of (α,β,c) for several lengths yields many good
    starting points.  The best few candidates are refined by a tiny
    stochastic coordinate descent on the three parameters.

2️⃣  A global exponentiated‑gradient / smooth‑max mirror‑descent phase
    (soft‑max of the self‑convolution) that fine‑tunes every entry of the
    vector while staying on the probability simplex.

3️⃣  A projected Adam sub‑gradient step on the true max‑convolution,
    followed by a cheap mass‑transfer hill‑climber.  The whole pipeline
    is run for several lengths (up to 16000) and the best result
    across all runs is returned.

All heavy work is performed with FFT‑based convolutions (O(n log n)) and
the code respects a 1000 s wall‑clock budget.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / FFT utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1,
    returns the exact value produced by `evaluate_sequence`
    (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of distributions
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_general(n: int, alpha: float, beta: float, c: float) -> np.ndarray:
    """
    Discrete Beta‑like family:
        w_i ∝ (i+½)^{α‑1} (n‑i‑½)^{β‑1} + c
    Parameters are kept inside safe bounds before use.
    """
    # clamp to avoid pathological values
    alpha = max(0.01, alpha)
    beta = max(0.01, beta)
    c = max(0.0, c)

    idx = np.arange(n, dtype=np.float64)
    left = (idx + 0.5) ** (alpha - 1.0)
    right = (n - idx - 0.5) ** (beta - 1.0)
    w = left * right + c
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0 or not np.isfinite(s):
        return _distribution_uniform(n)
    w /= s
    return w


# ----------------------------------------------------------------------
# Optimisation primitives (smooth max, Adam, hill‑climber)
# ----------------------------------------------------------------------


def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float, eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective  f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector and the current max conv value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # Soft‑max weights
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # Gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    # Exponentiated‑gradient update (mirror w.r.t. KL)
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One Adam‑style projected sub‑gradient step on the true max‑convolution.
    Returns the updated vector, the new velocity and the new maximal
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices attaining the maximum, within tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # Simple back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Tiny hill‑climber that moves a little probability mass from the pair
    contributing most to the current peak towards the globally smallest
    entry.  Stops after `max_iters` iterations.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())                # position of the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) giving the largest product a_i*a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]   # second‑smallest

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional jitter to avoid stagnation
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

def _search_parametric_family(n: int,
                              n_samples: int = 400,
                              top_k: int = 5):
    """
    Randomly sample (α,β,c) for the Beta‑like family, evaluate,
    and return the `top_k` candidates (score, (α,β,c), vector).
    """
    candidates = []
    for _ in range(n_samples):
        alpha = random.uniform(0.05, 0.90)      # avoid extremes causing overflow
        beta = random.uniform(0.05, 0.90)
        c = random.uniform(0.0, 0.2)
        vec = _distribution_beta_general(n, alpha, beta, c)
        sc = _max_convolution_score(vec)
        candidates.append((sc, (alpha, beta, c), vec))
    candidates.sort(key=lambda x: x[0])
    return candidates[:top_k]


def _refine_parameters(alpha: float,
                       beta: float,
                       c: float,
                       n: int,
                       iters: int = 200,
                       init_step: float = 0.20):
    """
    Stochastic coordinate descent on (α,β,c).  Returns an improved
    probability vector and its score.
    """
    best_alpha, best_beta, best_c = alpha, beta, c
    best_vec = _distribution_beta_general(n, best_alpha, best_beta, best_c)
    best_score = _max_convolution_score(best_vec)
    step = init_step

    for it in range(iters):
        # propose a small relative perturbation
        a_new = max(0.01, best_alpha * (1.0 + step * random.uniform(-1.0, 1.0)))
        b_new = max(0.01, best_beta  * (1.0 + step * random.uniform(-1.0, 1.0)))
        c_new = max(0.0, best_c + step * random.uniform(-0.1, 0.1))
        vec = _distribution_beta_general(n, a_new, b_new, c_new)
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score = sc
            best_alpha, best_beta, best_c = a_new, b_new, c_new
            best_vec = vec
        # slowly shrink the step size
        step *= 0.995
    return best_vec, best_score

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the provided time budget (default 1000 s) and
    returns the best list of floats found.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 1.0          # a tiny safety margin

    eps = 1e-12

    # ------------------------------------------------------------------
    # Load previous construction if it is present (height_sequence_1)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            best_seq = prev.copy()
            best_score = _max_convolution_score(best_seq)
            print(f"[{time.time():.1f}] loaded previous best score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Exploration over several lengths using the Beta‑like family
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000]   # moderate sizes – easy to handle with FFT
    top_global = []                       # (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        # Combine a few deterministic families as fall‑backs
        for fam_name, maker in [
            ("uniform", _distribution_uniform),
            ("arcsine", _distribution_arcsine)
        ]:
            vec = maker(n)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} {fam_name:8s} score={sc:.6f}")
            top_global.append((sc, vec.copy()))

        # Random sampling in the three‑parameter family
        top_candidates = _search_parametric_family(n, n_samples=500, top_k=5)

        for sc, (a, b, c), vec in top_candidates:
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} param‑init α={a:.3f} β={b:.3f} c={c:.3f} score={sc:.6f}")
            top_global.append((sc, vec.copy()))

        # Local refinement of each of the best 5 parametric candidates
        for _, (a, b, c), _ in top_candidates:
            if time.time() >= deadline:
                break
            vec_ref, sc_ref = _refine_parameters(a, b, c, n, iters=250)
            if sc_ref < best_score:
                best_seq, best_score = vec_ref.copy(), sc_ref
                print(f"[{time.time():.1f}] n={n} param‑refined α={a:.3f}→{vec_ref[:3]} score={sc_ref:.6f}")
            top_global.append((sc_ref, vec_ref.copy()))

        # Keep only a modest number of global elites to limit memory
        top_global.sort(key=lambda x: x[0])
        top_global = top_global[:12]   # keep the 12 best across all lengths so far

    # ------------------------------------------------------------------
    # If no candidate was found (unlikely), fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n0 = 4000
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max mirror descent on the current overall best vector
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    remaining = deadline - time.time()
    # Estimate one iteration cost roughly (depends on n)
    est_one = 0.003 if n > 8000 else 0.0015
    max_iters = max(2000, int(min(20000, remaining / est_one)))

    print(f"[{time.time():.1f}] Mirror‑descent phase: n={n}, iters={max_iters}")

    alpha_start, alpha_end = 0.5, 2e5          # smooth‑max temperature schedule
    lr_start, lr_end = 0.35, 0.001            # learning‑rate schedule

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        t = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)

        if (it + 1) % max(1, max_iters // 10) == 0 or it == max_iters - 1:
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:

    # ------------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.6
    sub_it = 0

    while time.time() < deadline - 0.6:
        sub_it += 1
        a, velocity, cur_max = _adam_subgrad_step(a, velocity, step0,
                                                 beta1, beta2, sub_it, eps)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cur_score:.6f}")

        if sub_it % 800 == 0:

            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny mass‑transfer hill‑climber
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
