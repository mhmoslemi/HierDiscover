"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Approach
--------
1.  Generate a diverse pool of candidate shapes for several lengths.
    • Analytic families (uniform, arcsine, symmetric β, edge‑weighted,
      cosine, tent, exponential) and a few signal‑processing windows.
    • Random Dirichlet / sparse seeds.
2.  Keep the ~30 best candidates (objective = 2·n·max(conv) because we
    normalise the sum to 1).
3.  Refine each of the top‑few seeds with a *smoothed* mirror‑descent.
    The non‑smooth max in the objective is replaced by a
    log‑sum‑exp with a gradually increasing smoothing parameter
    ``α``.  The gradient is obtained analytically and applied via an
    exponentiated‑gradient update, followed by a Euclidean projection
    onto the probability simplex.
4.  Finish with a cheap greedy pairwise‑mass‑transfer polishing.
5.  Return the best found vector as a plain Python list.

The algorithm respects the 1000 s budget (leaving a 2 s safety
margin) and works for any length up to a few 10⁴ without exceeding the
time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
def _self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a."""
    return _fft_convolve(a, a)


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used in ``evaluate_sequence`` when Σa = 1:
        2·n·max(conv(a,a))
    The input must be non‑negative and already normalised.
    """
    n = a.shape[0]
    conv = _self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta( p , p ) density discretised on the middle of each cell."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·(n‑i))^(‑p) evaluated at cell centres."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    """Cosine window (non‑negative, symmetric)."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.cos(2.0 * math.pi * (i + 0.5) / n)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_tent(n: int) -> np.ndarray:
    """Triangular (tent) shape, peak in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = 1.0 - np.abs(i - centre) / centre
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_exponential(n: int, scale: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-scale * np.abs(i - centre))
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination λ·uniform + (1‑λ)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

# Smoothed gradient descent ---------------------------------------------
def _smooth_grad_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    alpha_start: float = 5.0,
    alpha_end: float = 50.0,
    lr0: float = 0.8,
    eval_interval: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a smoothed version of the max‑convolution objective.
    The smoothing parameter α grows linearly from ``alpha_start`` to
    ``alpha_end``.  Exponentiated‑gradient updates keep the iterate inside
    the simplex.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        # current smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * (t / 5000.0)
        if alpha > alpha_end:
            alpha = alpha_end

        # ----- convolution and softmax weights -----
        conv = _self_convolve(a)                     # length 2n‑1
        # stable softmax
        max_c = conv.max()
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

        # ----- gradient of the smoothed max -----
        # w_rev is w reversed (corresponding to shift −k)
        w_rev = w[::-1]

        # convolution a * w and a * w_rev (full)
        conv_w = _fft_convolve(a, w)                 # length 3n‑2
        conv_w_rev = _fft_convolve(a, w_rev)

        # slice the middle n entries (indices n‑1 … 2n‑2)
        start = n - 1
        grad = conv_w[start:start + n] + conv_w_rev[start:start + n]

        # ----- exponentiated‑gradient update -----
        eta = lr0 / math.sqrt(t)                     # decaying stepsize
        a = a * np.exp(-eta * grad)                  # multiplicative update
        a = _project_onto_simplex(a)                 # back to the simplex

        # ----- occasional evaluation -----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] grad it={t} α={alpha:.1f} "
                    f"new best {best_score:.6f}"
                )
        # safety cap
        if t >= 300000:
            break
    return best_seq, best_score

# Greedy pairwise mass transfer -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly shift a fraction of mass; keep the change only when it improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.4   # move up to 40 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after polishing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will handle the exact scaling.
    return overall_best_seq.tolist()
