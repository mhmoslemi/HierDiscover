def _smooth_exponential_descent(a: np.ndarray, deadline: float,
    best_seq: np.ndarray, best_score: float,
    eta0: float = 0.6,
    tau0: float = 5.0,
    tau_mult: float = 1.001,
    eval_interval: int = 100,
    active_k: int = 12,
    patience: int = 2000,
):
    t=0
    tau = tau0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 300_000:
        t += 1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # get top_k indices
        top_idx = np.argpartition(conv, -active_k)[-active_k:]
        # compute weights from softmax
        w_raw = np.exp(tau * (conv[top_idx] - max_c))  # values <= 1
        w = w_raw / w_raw.sum()
        # zero weight for others
        grad = np.zeros_like(a)
        ind_grid = np.arange(a.shape[0], dtype=int)
        for weight, idx in zip(w, top_idx):
            ind = idx - ind_grid
            mask = (ind >=0) & (ind < a.shape[0])
            # add scaled gradient
            grad[mask] += 2.0 * weight * a[ind[mask]]
        # step size
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # renorm
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / a.shape[0])
        else:
            a /= sum_a
        # update tau
        tau *= tau_mult
        # evaluate periodically
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} best {best_score:.6f}")
            else:
                no_improve += 1
        # jitter if stagnated
        if no_improve >= patience:
            jitter = rng.normal(scale=0.01, size=a.shape)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
            no_improve = 0
    return best_seq, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Rich initial pool** – generate sequences from several analytic families
   (uniform, arcsine, symmetric/asymmetric Beta, edge‑weighted, sine‑weighted)
   for a few different lengths, plus random Dirichlet and sparse vectors.
   Keep the best ~50 seeds.

2. **Smooth exponential descent** – for each promising seed run an
   exponentiated‑gradient descent on the exact objective.
   Instead of a hard active set we use a *soft* max:
        w_i ∝ exp(τ·conv_i)               (τ grows during optimisation)
   The gradient is a weighted sum of the usual sub‑gradients.
   Step‑size η decays like 1/√t; a small jitter is injected after long stalls.

3. **Greedy pairwise swaps** – move a small amount of mass between two
   positions if the objective improves.

4. **Final stochastic local search** – a cheap simulated‑annealing walk that
   occasionally accepts a worsening move.

The whole routine respects the supplied time budget (default 1000 s).  The
returned list is a plain Python list; ``evaluate_sequence`` will perform the
final normalisation, clipping, etc.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (Beta(a,b)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Beta PDF up to a normalising constant
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth exponential descent -------------------------------------------
def _smooth_exponential_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    tau0: float = 3.0,
    tau_mult: float = 1.001,
    eval_interval: int = 120,
    active_k: int = 12,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true objective.
    Uses a softmax weighting of the convolution entries:
        w_i ∝ exp(τ·conv_i)   (τ slowly increases).
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    tau = tau0

    while time.time() < deadline and t < 400_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ---- soft‑max weighting of the top active_k entries ----
        if active_k >= len(conv):
            top_idx = np.arange(len(conv))
        else:
            top_idx = np.argpartition(conv, -active_k)[-active_k:]

        # scale by the current temperature
        # (subtract max_c to avoid overflow)
        w_raw = np.exp(tau * (conv[top_idx] - max_c))
        w = w_raw / w_raw.sum()

        # ---- compute weighted gradient ----
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for weight, idx in zip(w, top_idx):
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * weight * a[ind[mask]]

        # ---- exponentiated‑gradient update ----
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # increase temperature slowly (focuses more on the true max)
        tau *= tau_mult

        # ---- occasional evaluation ----
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # ---- jitter on stagnation ----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score
