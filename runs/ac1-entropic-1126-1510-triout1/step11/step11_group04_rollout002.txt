# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT‑based convolutions ------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# Objective (max of self‑convolution, sum normalised to 1) -------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv) for a probability vector a (sum = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# Analytic families – each returns a probability vector (sum = 1) -----
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_alternating(n):
    """High‑low alternating pattern to disperse large masses."""
    base = np.random.rand(n)
    base.sort()
    high = base[::-1]  # descending
    low = base
    out = np.empty_like(base)
    left, right = 0, n - 1
    toggle = True
    for h in high:
        if toggle:
            out[left] = h
            left += 1
        else:
            out[right] = h
            right -= 1
        toggle = not toggle
    out /= out.sum()
    return out

def _generate_candidates_for_length(n):

    cand = []

    # Analytic families

    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    for lam in (0.001, 0.003, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential(n, lam=lam))

    # Symmetric Beta
    for p in (0.01, 0.02, 0.04, 0.06, 0.08, 0.12, 0.15,

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0, 16.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.10, 0.15, 0.20, 0.25, 0.30, 0.40):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0, 4.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws (different concentration)
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Sparse random start
    k = max(2, n // 200)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)

    cand.append(_distribution_alternating(n))

    # Shuffle a few base families
    for base in (cand[0], cand[1], cand[4]):

# Smooth‑max mirror descent --------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """Exponentiated‑gradient mirror descent on the smoothed max."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _objective(best_a)

    # Temperature / learning‑rate schedule (more iterations than reference)
    schedule = [
        (1.0, 1e5, 15000, 0.6, 0.04),
        (1e5, 1e7, 15000, 0.04, 0.004),
        (1e7, 1e9, 15000, 0.004, 0.0004),
    ]

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))               # smoothed weights
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smoothed max (see derivation in reference code)
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient update
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if (it & 0x3FF) == 0:       # every 1024 iterations
                sc = _objective(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc

def _projected_subgradient(a0, deadline):
    """True sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_step = 0.4
    it = 0
    best_a = a.copy()
    best_sc = _objective(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum
        vel = beta1 * vel + (1 - beta1) * grad
        v_hat = vel / (1 - beta1 ** it)

        step = base_step
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-14 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-14:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc

                    M = conv.max()
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ----- simple simulated annealing move (random mass shift) -----
        if it % 500 == 0:
            i, j = random.sample(range(n), 2)
            if a[i] > eps:
                delta = min(a[i] * 0.2, random.random() * 0.01)
                cand = a.copy()
                cand[i] -= delta
                cand[j] += delta
                cand = np.maximum(cand, eps)
                cand /= cand.sum()
                cand_sc = _objective(cand)
                accept = False
                if cand_sc < best_sc:
                    accept = True
                else:
                    prob = math.exp(-(cand_sc - best_sc) / max(temperature, 1e-12))
                    if random.random() < prob:
                        accept = True
                if accept:
                    a = cand
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    if cand_sc < best_sc - 1e-14:
                        best_sc = cand_sc
                        best_a = a.copy()
                temperature *= 0.9995   # cool slowly

        # safety break
        if it % 5000 == 0 and time.time() + 0.5 > deadline:
            break

    return best_a, best_sc

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain Python list) found within the time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # keep a safety margin (2 s) to avoid forced termination
    deadline = start + budget_s - 2.0

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []                     # (score, vector) sorted increasingly
    # --------------------------------------------------------------
    # 1. Broad exploration
    lengths = [2000, 3000, 4000, 5000, 8000, 12000, 16000, 20000]
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain a tiny elite set (max size 10)
            if len(elite) < 10:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2. Incorporate any previously known best (if the variable exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 10:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Fallback if nothing has been found (should never happen)
    if best_seq is None:
        best_seq = _distribution_uniform(4000)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Local optimisation from the elite (max 5 restarts)
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} length={a0.shape[0]} start score={elite[idx][0]:.6f}")

        # ---- smoothed‑max mirror descent ----
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # ---- projected sub‑gradient polishing ----
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # ---- hill‑climbing + pruning + annealing ----
        a_mid, sc_mid = _hill_climb_and_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-14:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # --------------------------------------------------------------
    # 5. Final sanitisation
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
