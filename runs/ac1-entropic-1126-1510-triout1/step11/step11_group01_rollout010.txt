def _softmax_exponential_update(a, beta, eta):
    conv = _fft_self_convolve(a)
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # Compute weighted gradient
    grad = _weighted_gradient(a, w)

    # exponentiated gradient update:
    a_new = a * np.exp(-eta * grad)
    # renormalize
    a_new_sum = a_new.sum()
    if a_new_sum == 0.0:
        a_new = np.full_like(a, 1.0 / a.size)
    else:
        a_new = a_new / a_new_sum
    return a_new

def _fft_convolve(a, b):
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]

Improved search for a non‑negative step‑function that minimises
``evaluate_sequence``.

Algorithm
---------
1.  Generate a diverse pool of analytic candidates for several lengths.
    Families: uniform, arcsine, symmetric beta, edge‑weighted,
    exponential taper, Gaussian, Hann/Hamming/Blackman windows,

3.  Refine the top few seeds with a *soft‑max* mirror‑descent.

      sub‑gradient is the weighted sum of all convolution shifts.
    - Updates are performed by exponentiated‑gradient (mirror descent)
      on the simplex with a decaying stepsize.
    - β is increased slowly so that the optimisation first sees a smooth
      landscape and later focuses on the true ‑max‑.

4.  Final greedy polishing: random pairwise mass transfers that are
    accepted only when they improve the exact objective.

All work respects the 1000 s budget (a 2 s safety margin).  The function

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution a*b using a real FFT."""
    n, m = a.shape[0], b.shape[0]
    size = 1 << ((n + m - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a probability vector (sum=1).
    The input is assumed non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta density ∝ x^{p‑1}(1‑x)^{p‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^‑p   (i,j are distance to the two edges)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, r: float) -> np.ndarray:
    """Symmetric exponential taper w_i = r**i + r**(n‑1‑i)."""
    i = np.arange(n, dtype=np.float64)
    w = np.power(r, i) + np.power(r, n - 1 - i)
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle, σ = sigma_frac·n."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


def _distribution_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _distribution_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * base
    mixed /= mixed.sum()
    return mixed

# Gradient machinery – smooth max via soft‑max weighting -------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_s w_s·(a*a)[s]  (the convolution entry at shift s).
    Uses two FFT‑based convolutions to obtain the linear combination.
    """
    n = a.shape[0]

    # a * w   (full length 3n‑2) → first n entries give Σ_i a_i w[j‑i]
    conv_aw = _fft_convolve(a, w)
    g2 = conv_aw[:n]

    # a * w_rev   (full) → slice n‑1 … 2n‑1 and reverse gives Σ_i a_i w[i+j]
    w_rev = w[::-1]
    conv_aw_rev = _fft_convolve(a, w_rev)
    g1 = conv_aw_rev[n - 1 : 2 * n - 1][::-1]

    grad = 2.0 * (g1 + g2)  # each convolution contributes a factor 2
    return grad


def _softmax_exponential_update(a: np.ndarray, beta: float, eta: float) -> np.ndarray:
    """
    One mirror‑descent step based on the soft‑max approximation of the max.
    """
    conv = _fft_self_convolve(a)
    max_c = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    grad = _weighted_gradient(a, w)

    # exponentiated‑gradient update (guarantees non‑negativity)
    factor = np.exp(np.clip(-eta * grad, -50, 50))
    a_new = a * factor
    s = a_new.sum()
    if s == 0.0:
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


def _smooth_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.8,
    beta0: float = 5.0,
    beta_max: float = 300.0,
    eval_interval: int = 500,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent with a soft‑max smoothing of the max‑convolution.
    The temperature β is increased linearly during the first few thousand
    iterations so that the algorithm first sees a smooth landscape.
    """
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # linear β ramp (first 5000 iters go from beta0 → beta_max)
        beta = beta0 + (beta_max - beta0) * min(1.0, t / 5000.0)
        eta = eta0 / math.sqrt(t)  # classic decaying stepsize
        a = _softmax_exponential_update(a, beta, eta)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} new best {best_score:.6f}")

    return best_seq, best_score

# Greedy pairwise mass transfer ----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass; keep the move only if it improves the exact objective."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # choose a different destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move up to 30 % of the source mass (finer than the previous 40 %)
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

        print(f"[{time.time():.1f}] smooth descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.8,
                beta0=5.0,
                beta_max=300.0,
                eval_interval=500,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise‑swap polishing")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=300_000,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (objective=2·n·max(conv)): {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will normalise internally; we just return a list.
    return overall_best_seq.tolist()
