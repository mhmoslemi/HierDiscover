"""
Propose a step‑function (list of non‑negative floats) that minimizes
`evaluate_sequence`.

Algorithm
---------
1. **Parametric warm‑starts** – generate a few deterministic families
   (uniform, arcsine, edge‑power, symmetric beta) for several lengths.
   Keep the best candidate.

2. **Smooth‑max L‑BFGS optimisation** – minimise a smooth
   approximation of the max‑convolution:
        f_α(a) = (1/α)·log∑_k exp(α·(a*a)_k)
   for a probability vector a.  The variables are the logits y with
   a = softmax(y).  For a schedule of increasing temperatures
   α = 10,30,100,300,1000 we run a limited L‑BFGS‑B optimisation
   (gradient supplied analytically via FFT).  Several random restarts
   (including a good edge‑power seed) are performed for each length.
   The best vector over all lengths is retained.

3. **Final hill‑climbing polish** – repeatedly move a small amount of
   probability mass from the largest entry to the smallest entry;
   accept the move only if the true max‑convolution strictly improves.
   Occasional jitter avoids stagnation.

All heavy work uses FFT‑based convolutions, the whole search respects the
time budget (default 1000 s) and returns a plain Python list.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y):
    """Numerically stable softmax."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# FFT based convolutions -----------------------------------------------


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# Parametric families (all return a vector with sum = 1) ---------------


def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n, p):
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15            # avoid exact zeros
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective used by L‑BFGS -----------------------------------


class _SmoothMaxObj:
    """
    Callable returning (value, gradient) for the smooth‑max objective
    f_α(a) = max_{k} (a*a)_k approximated by log‑sum‑exp.
    The optimisation variable is y (logits) with a = softmax(y).
    """

    def __init__(self, n, alpha):
        self.n = n
        self.alpha = float(alpha)

    def __call__(self, y):
        a = _softmax(y)
        conv = _fft_self_convolve(a)               # length 2n‑1
        maxc = conv.max()
        # stable soft‑max weights over the convolution entries
        exp_alpha = np.exp(self.alpha * (conv - maxc))
        sum_exp = exp_alpha.sum()
        w = exp_alpha / sum_exp                     # shape (2n‑1,)

        # gradient w.r.t. a (same formula as in the baseline code)
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[self.n - 1:self.n - 1 + self.n]

        # chain rule through softmax
        dot = np.dot(a, grad_a)
        grad_y = a * (grad_a - dot)

        # smooth max value
        val = maxc + (1.0 / self.alpha) * np.log(sum_exp)
        return val, grad_y


# ----------------------------------------------------------------------
# Main proposal function -----------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list.
    """
    # ------------------------------------------------------------------
    # Seed & time budget
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Helper to compute the true evaluation (without the final scaling)
    def _true_score(a):
        n = a.shape[0]
        return 2.0 * n * _fft_self_convolve(a).max()

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration of a few lengths & deterministic families
    candidate_lengths = [2000, 4000, 8000, 12000]

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        # Uniform
        a = _distribution_uniform(n)
        sc = _true_score(a)
        if sc < best_score:
            best_score, best_seq, best_n = sc, a.copy(), n
            print(f"[{time.time():.1f}] n={n} uniform            score={sc:.6f}")

        # Arcsine
        a = _distribution_arcsine(n)
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        sc = _true_score(a)
        if sc < best_score:
            best_score, best_seq, best_n = sc, a.copy(), n
            print(f"[{time.time():.1f}] n={n} arcsine            score={sc:.6f}")

        # Edge‑power (p tuned a little)
        for p in (0.3, 0.5, 0.7):
            if time.time() >= deadline:
                break
            a = _distribution_edge_power(n, p)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            sc = _true_score(a)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a.copy(), n
                print(f"[{time.time():.1f}] n={n} edge_power(p={p:.1f}) score={sc:.6f}")

        # Symmetric Beta (p around 0.2 … 0.6)
        for p in (0.15, 0.25, 0.35, 0.45):
            if time.time() >= deadline:
                break
            a = _distribution_beta_sym(n, p)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            sc = _true_score(a)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a.copy(), n
                print(f"[{time.time():.1f}] n={n} beta_sym(p={p:.2f}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max L‑BFGS refinements (multiple restarts per length)
    alphas = [10.0, 30.0, 100.0, 300.0, 1000.0]   # temperature schedule
    max_restarts_per_n = 5

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        # build a collection of sensible starting vectors
        start_vectors = [
            _distribution_edge_power(n, 0.4),
            _distribution_beta_sym(n, 0.25),
            np.random.dirichlet(np.full(n, 0.7)),
            np.random.dirichlet(np.full(n, 1.0)),
            np.random.dirichlet(np.full(n, 0.3)),
        ]

        for init_a in start_vectors[:max_restarts_per_n]:
            if time.time() >= deadline:
                break

            a0 = np.maximum(init_a, 1e-15)
            a0 /= a0.sum()
            y = np.log(a0)                 # initialise logits

            # cascade through the temperature schedule
            for alpha in alphas:
                if time.time() >= deadline:
                    break
                obj = _SmoothMaxObj(n, alpha)

                # L‑BFGS‑B with analytic gradient
                res = _opt.minimize(
                    fun=obj,
                    x0=y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": 150, "ftol": 1e-12, "gtol": 1e-12}
                )
                y = res.x

            # final probability vector (symmetrise – optional but helpful)
            a_opt = _softmax(y)
            a_opt = (a_opt + a_opt[::-1]) * 0.5
            a_opt /= a_opt.sum()

            sc = _true_score(a_opt)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a_opt.copy(), n
                print(f"[{time.time():.1f}] n={n} L‑BFGS restart  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Fallback – use previous best if any global variable is present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-15)
            prev /= prev.sum()
            sc = _true_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best            score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard against no candidate (should never happen)
    if best_seq is None:
        n = 4000
        best_seq = _distribution_uniform(n)
        best_seq /= best_seq.sum()
        best_score = _true_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform          score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Targeted stochastic hill‑climbing polish
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0

    while time.time() < deadline:
        hill_it += 1

        # source = current largest entry, target = smallest entry (different)
        src = int(np.argmax(a))
        tgt = int(np.argmin(a))
        if tgt == src:
            tgt = int(np.argsort(a)[1])

        max_delta = min(a[src] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= 1e-16:
            continue

        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta
        # (sum unchanged, positivity preserved)
        conv2 = _fft_self_convolve(a2)
        M2 = conv2.max()
        if M2 < M - 1e-12:
            a = a2
            conv = conv2
            M = M2
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 100 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if hill_it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, 1e-15)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_it}")

    # ------------------------------------------------------------------
    final_score = 2.0 * best_n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_seq.tolist()
