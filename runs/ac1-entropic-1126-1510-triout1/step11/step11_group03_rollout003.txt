def _subgrad_descent(a, deadline, best_seq, best_score, lr0=1.0, max_iter=50000, eval_interval=500):
    n = len(a)
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # find all indices where conv close to max (within tolerance)
        tol = 1e-12
        max_idx = np.where(conv >= max_conv - tol)[0]
        # weight vector w length = 2n-1
        w = np.zeros_like(conv)
        w[max_idx] = 1.0 / len(max_idx)  # uniform among max indices
        grad = _gradient_from_weights(a, w)  # 2*a⋆w
        # learning rate schedule
        eta = lr0 / math.sqrt(t)
        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        # renorm to simplex
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"Iter {t} new best {cur:.6f}")
    return best_seq, best_score

"""
Propose a step‑function sequence using mirror‑descent on the exact
sub‑gradient of the maximal self‑convolution.  The algorithm

1. builds a small pool of analytic distributions (uniform, arcsine,
   symmetric/asymmetric Beta, sparse Dirichlet) for a relatively large
   length (≈20 000);
2. evaluates them and keeps the best few;
3. runs a long exponentiated‑gradient (KL‑mirror) descent on the true
   max‑convolution (sub‑gradient is 2·a⋆w where w is a one‑hot vector on
   the maximal indices);
4. finishes with a cheap greedy mass‑transfer walk that keeps moves only
   when the true (non‑smoothed) objective improves.

The whole procedure respects the given time budget (default 1000 s) and
returns the best sequence it found.  All values are non‑negative and
renormalised to sum to 1, which makes the clipping performed inside
``evaluate_sequence`` irrelevant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the weighted sum Σ_k w_k (a*a)_k.
    Implemented via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _subgrad_descent(a: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     lr0: float = 0.5,
                     max_iter: int = 120000,
                     eval_interval: int = 300) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Uses the sub‑gradient 2·a⋆w where w is a uniform weight on all
    maximal convolution entries.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # indices where the maximum is attained (allow tiny tolerance)
        idx = np.where(conv >= max_conv - 1e-15)[0]
        w = np.zeros_like(conv)
        w[idx] = 1.0 / len(idx)

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)          # diminishing step size
        a = a * np.exp(-eta * grad)       # mirror‑descent (KL) step

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic mass‑transfer walk.  Randomly moves a small fraction
    of mass from one index to another; the move is kept only if the true
    (non‑smoothed) objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        # random fraction of the source mass (up to 30 %)
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = prev / prev.sum()
            sc = _objective(prev)
            elite.append((sc, prev.copy()))
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] imported previous best score={sc:.6f}")
    except Exception:
        pass

    # keep only a handful of elite seeds for the intensive phase
    elite.sort(key=lambda x: x[0])
    elite = elite[:6]

    # --------------------------------------------------------------
    # 5. Main optimisation – exact sub‑gradient exponentiated descent
    remaining = deadline - time.time()
    if remaining > 30.0 and best_seq is not None:
        per_seed = max(1.0, remaining / len(elite))
        for seq in [e[1] for e in elite]:
            if time.time() >= deadline:
                break
            sub_deadline = min(deadline, time.time() + per_seed)
            seq_out, score_out = _subgrad_descent(
                seq.copy(),
                sub_deadline,
                best_seq,
                best_score,
                lr0=0.5,
                max_iter=120000,
                eval_interval=300,
            )
            if score_out < best_score:
                best_score = score_out
                best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after subgrad new best {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final polishing with greedy local swaps
    if best_seq is not None and time.time() < deadline:
        best_seq, best_score = _pairwise_swap_local(
            best_seq.copy(),
            deadline,
            best_seq,
            best_score,
        )

    # --------------------------------------------------------------
    # 7. Return the best sequence (as plain Python list)
    if best_seq is None:
        # fallback – a uniform distribution of a modest size
        n_fallback = 4000
        best_seq = _distribution_uniform(n_fallback)
        best_score = _objective(best_seq)

    print(f"Final best objective: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
