def propose_candidate(seed=123, budget_s=1000):
    import time, numpy as np, math
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.1

    # candidate parameters
    lengths = [2000, 4000, 8000, 15000, 20000]
    alphas = np.linspace(0.5, 4.0, 71)  # e.g., step 0.05 approx
    best_score = float('inf')
    best_seq = None

    for n in lengths:
        if time.time() > deadline: break
        print(f"Scanning n={n}")
        for alpha in alphas:
            if time.time() > deadline: break
            a = np.exp(-alpha * np.arange(n) / n, dtype=np.float64)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            # compute objective quickly using fft
            conv = _fft_self_convolve(a)
            max_conv = conv.max()
            score = 2.0 * n * max_conv
            # notice denominator sum^2 =1 because we normalized to 1.
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best {best_score:.6f} n={n} alpha={alpha:.3f}")

    # try to improve with local swaps (few steps)
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        while time.time() < deadline:
            # pick random pair
            i,j = rng.integers(n, size=2)
            if i==j:
                continue
            # randomly transfer small fraction from i to j
            delta = a[i] * rng.random() * 0.05
            if delta <= 0:
                continue
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            # renormalize to sum=1 (may spurious rounding)
            # actually sum unchanged, since we added same amount.
            conv = _fft_self_convolve(new_a)
            score = 2.0 * n * conv.max()
            if score < best_score - 1e-12:
                best_score = score
                a = new_a
                best_seq = a
                print(f"Local swap improved {best_score:.6f}")
    # final printing
    print(f"Final best score {best_score:.6f} length={len(best_seq)}")
    # Convert to list of floats; ensure non-negative and not too large (they are small <1)
    return best_seq.tolist()

def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A*A, n=size)
    return conv[:2*n-1]
```

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The design is deliberately simple but explores a richer set of analytic
families than the previous submission.  It works in three phases:

1. **Analytic seeding** – generate many candidates (uniform,
   reversed arcsine, decreasing exponential, decreasing power‑law,
   Dirichlet draws …) for several lengths.  Each candidate is projected
   onto the probability simplex and its exact objective is evaluated.
   The best few are kept.

2. **Hard‑max (sub‑gradient) descent** – starting from each elite seed we
   repeatedly identify the convolution index where the maximum occurs,
   compute the exact sub‑gradient of that maximum (2·a shifted) and
   take a diminishing step, projecting back onto the simplex.  This
   removes the temperature smoothing of the original soft‑max method
   and often drives the objective lower.

3. **Local random mass‑transfer polishing** – cheap random moves that
   transfer a small fraction of mass from one index to another are
   accepted only if the true (unsmoothed) objective improves.

All work respects the supplied time budget (default 1000 s).  The
function returns the best sequence found as a plain list of floats.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT of length power‑of‑2."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum forced to 1)."""
    # a is assumed to be non‑negative and already normalised to sum = 1
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    return 2.0 * n * max_conv   # denominator = 1 because sum(a)=1

# Analytic families – many are the same as the previous code, plus a few
# new decreasing families.
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_decreasing_exponential(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-alpha·i/n) (decreasing exponential)."""
    vals = np.exp(-alpha * np.arange(n) / n)
    vals /= vals.sum()
    return vals


def _distribution_decreasing_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (decreasing power‑law)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds (including the new decreasing families)."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # decreasing exponentials
    for alpha in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_decreasing_exponential(n, alpha))

    # decreasing power‑laws
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_decreasing_power(n, p))

    # a few Dirichlet draws
    for _ in range(4):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.2 + 0.8 * rng.random()))

    # a random sparse draw (helps escaping plateaus)
    k = max(2, int(n * 0.01))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    cand.append(vals)

    return cand

def _hardmax_subgrad_descent(a: np.ndarray,
                             max_iter: int,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution objective.
    At each step the index k* with maximal convolution entry is located,
    the exact sub‑gradient (2·a shifted) is taken, and the iterate is
    projected back onto the probability simplex.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k_star = int(conv.argmax())
        max_val = conv[k_star]

        # sub‑gradient of the max‑entry w.r.t. a
        grad = np.zeros_like(a)
        idx = k_star - np.arange(n)                     # idx = k* - i
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # step size with a simple 1/√t schedule
        eta = 0.6 / math.sqrt(t)
        a = a - eta * grad
        a = np.maximum(a, 0.0)
        # re‑project onto the simplex (ensures sum = 1)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # occasional exact evaluation
        if t % 500 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Cheap random mass‑transfer walk that accepts only improvements."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move up to 30 % of source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point used by the checker.

    The routine respects the overall time budget, builds a diverse pool of
    analytic seeds (including the new decreasing families), refines the
    most promising ones with a hard‑max sub‑gradient descent, and finally
    applies a cheap local‑swap polishing phase.  The best sequence (as a
    plain list of floats) is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Scan a handful of lengths with many analytic seeds
    lengths = [2000, 4000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    elite = []   # list of (score, array)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] *** scanning length={n} ***")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] >>> new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep a tiny elite set (the very best seeds) across all lengths
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # ------------------------------------------------------------------
    # 2. If a previous solution exists, inject it into the elite pool
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Main optimisation – hard‑max sub‑gradient descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_seed_budget = max(5.0, remaining / len(elite))   # seconds per seed
        for score, seed_seq in elite[:5]:                     # only the very best
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_budget
            seq_out, score_out = _hardmax_subgrad_descent(
                seed_seq.copy(),
                max_iter=40000,
                deadline=sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after sub‑grad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final polishing with cheap local swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list – evaluate_sequence will apply its own clipping.
    return overall_best_seq.tolist()
