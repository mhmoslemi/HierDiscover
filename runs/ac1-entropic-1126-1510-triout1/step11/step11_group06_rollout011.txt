Implementation details of gradient descent for flattening:

def flatten_step(a, n, target, lr, m, v, beta1, beta2, t):
   conv = _fft_self_convolve(a)
   diff = conv - target
   w = 2 * diff   # weight
   grad = 2 * _fft_convolve(a[::-1], w)[n-1:n-1+n]   # shape n
   m = beta1 * m + (1 - beta1) * grad
   v = beta2 * v + (1 - beta2) * (grad**2)
   m_hat = m / (1 - beta1**t)
   v_hat = v / (1 - beta2**t)
   step = lr / (np.sqrt(v_hat) + 1e-8)
   a = a - step * m_hat
   # projection onto simplex
   a = _project_onto_simplex(np.maximum(a, 0.0), 1.0)
   return a, m, v

"""
Propose a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  

The algorithm proceeds in three stages:
1️⃣  **Exploration** – a few classic parametric families (uniform,
    arcsine, edge‑power) are sampled for several lengths.  The best
    candidate found is kept as a warm start.
2️⃣  **Flattening phase** – we minimise the L² distance between the
    self‑convolution and the flat target `1/(2·n‑1)` using Adam on the
    squared‑error loss.  The loss forces the whole autocorrelation to
    become as constant as possible, which drives down the maximum
    value.
3️⃣  **Polishing phase** – once the autocorrelation is almost flat we
    switch to a direct sub‑gradient descent on the true `max`
    objective (active‑set method).  A small amount of momentum and
    occasional jitter help to escape plateaus.

All updates are projected onto the simplex (non‑negative, sum = 1)
using the Euclidean projection.  The routine respects a 1000 s time
budget, prints progress, and returns the best sequence found.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v, s=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT based convolutions (real → real)
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# Simple parametric families that return a probability vector (sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_power(n, p):
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15          # avoid exact zeros
    w /= w.sum()
    return w
