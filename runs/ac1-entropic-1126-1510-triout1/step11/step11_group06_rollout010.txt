`evaluate_sequence`.  The algorithm consists of four stages:

1. **Broad exploration** – generate many candidate shapes for several
   lengths (uniform, arcsine, edge‑power, symmetric Beta, Dirichlet

   staying on the simplex.  The temperature grows from 1 to a very large

   increase the maximum are sometimes accepted according to an annealing
   schedule, allowing the search to escape shallow plateaus.

All stages respect the time budget (`budget_s`).  The final best vector is
returned as a plain Python list.  Only NumPy (and the built‑in `random`)
is used.

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT based convolutions -----------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Score for a vector already normalised to sum = 1:
        2·n·max(a*a)
    This equals the value returned by `evaluate_sequence` for the same
    shape (the scaling cancels out).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families (all return a vector with sum = 1) ---------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n: int, p: float) -> np.ndarray:
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15                     # avoid exact zeros
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Dirichlet draw biased towards the edges:
    concentration ∝ ((i+0.5)*(n-i-0.5))^{-beta_exp} + 0.1
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    return np.random.dirichlet(conc)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best list of non‑negative floats found within the given
    time budget (seconds).  The algorithm follows the four‑stage plan
    described in the module docstring.
    """
    # ------------------------------------------------------------------
    # Setup and timing
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin for finalisation
    eps = 1e-15

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Helper to update global best
    def _update_best(vec: np.ndarray):
        nonlocal best_seq, best_score, best_n
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score - 1e-12:
            best_seq = vec.copy()
            best_score = sc
            best_n = vec.shape[0]
            print(f"[{time.time():.1f}] New best n={best_n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣  Load previous solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            _update_best(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration over several lengths and families
    lengths = [2000, 4000, 8000, 16000, 32000]
    # allow a few larger lengths if enough time remains
    if deadline - time.time() > 150:
        lengths += [64000, 128000]

    edge_power_ps = (0.2, 0.5, 0.8, 1.2, 1.5, 2.0)
    beta_sym_ps = (0.2, 0.5, 0.8, 1.0, 1.5, 2.0)
    dirichlet_exps = (0.5, 0.8, 1.0, 1.5, 2.0)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families ---------------------------------
        candidates = []

        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))

        for p in edge_power_ps:
            candidates.append(_distribution_edge_power(n, p))

        for p in beta_sym_ps:
            candidates.append(_distribution_beta_sym(n, p))

        # ----- random families ----------------------------------------
        for _ in range(3):
            candidates.append(np.random.dirichlet(np.ones(n)))

        for bex in dirichlet_exps:
            candidates.append(_distribution_edge_dirichlet(n, bex))

        # ----- up‑sample current best if we have one ---------------
        if best_seq is not None and best_n != n:
            x_old = np.linspace(0.0, 1.0, best_n)
            x_new = np.linspace(0.0, 1.0, n)
            up = np.interp(x_new, x_old, best_seq)
            candidates.append(up)

        # ----- evaluate raw candidates --------------------------------
        for vec in candidates:
            if time.time() >= deadline:
                break
            _update_best(vec)

        # ----- short mirror‑descent refinements on the top 3 ---------
        # compute scores for ranking
        raw_scores = [_max_convolution_score(v) for v in candidates]
        top_ids = np.argsort(raw_scores)[:3]
        for tid in top_ids:
            if time.time() >= deadline:
                break
            a = candidates[tid].copy()
            a = np.maximum(a, eps)
            a /= a.sum()
            total_iters = 2000
            for it in range(total_iters):
                if time.time() >= deadline:
                    break
                t = it / total_iters
                # temperature grows exponentially, learning rate decays
                alpha = 1.0 * (1e5) ** t
                lr = 0.3 * (0.01) ** t
                conv = _fft_self_convolve(a)
                maxc = conv.max()
                w = np.exp(alpha * (conv - maxc))
                w_sum = w.sum()
                if w_sum == 0 or not np.isfinite(w_sum):
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum
                # gradient of the smooth‑max
                grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
                a = a * np.exp(-lr * grad)
                a = _project_onto_simplex(a, 1.0)
                a = np.maximum(a, eps)
                a /= a.sum()
            _update_best(a)

    # ------------------------------------------------------------------
    # Guard against missing solution
    if best_seq is None:
        best_seq = _distribution_uniform(4000)
        best_seq /= best_seq.sum()
        best_n = best_seq.shape[0]
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform n={best_n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Intensive mirror‑descent on the current best
    a = best_seq.copy()
    n = a.shape[0]
    remaining = deadline - time.time()
    if remaining > 10:
        # estimate possible iterations (≈0.004 s per iteration for moderate n)
        iters = int(min(50000, max(5000, remaining / 0.004)))
        for it in range(iters):
            if time.time() >= deadline:
                break
            t = it / iters
            alpha = 1.0 * (1e6) ** t          # very large final temperature
            lr = 0.5 * (0.001) ** t           # decaying learning rate
            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0 or not np.isfinite(w_sum):
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()
            if it % max(1, iters // 20) == 0:
                cur_sc = _max_convolution_score(a)
                if cur_sc < best_score - 1e-12:
                    best_seq = a.copy()
                    best_score = cur_sc
                    print(f"[{time.time():.1f}] Mirror‑descent it={it} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected Adam‑style sub‑gradient descent on the true max
    a = best_seq.copy()
    m = np.zeros_like(a)   # first moment
    beta1 = 0.9
    sub_it = 0
    while time.time() < deadline:
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** sub_it)

        step = 0.5
        while True:
            cand = a - step * m_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _max_convolution_score(cand)
            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_score - 1e-12:
            best_seq = a.copy()
            best_score = cand_sc
            print(f"[{time.time():.1f}] Adam sub‑grad it={sub_it} new best {best_score:.6f}")

        # small random jitter every few hundred iterations
        if sub_it % 500 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing hill‑climbing polish
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0
    T0 = 0.01                               # initial temperature
    # estimate how many moves we can afford
    remaining = deadline - time.time()
    est_moves = int(remaining / 0.003) if remaining > 0 else 0

    for _ in range(est_moves):
        if time.time() >= deadline:
            break
        hill_it += 1
        T = T0 * (0.99 ** hill_it)

        # pick a source weighted by current mass (large entries more likely)
        source = np.random.choice(n, p=a)
        # pick a different target uniformly
        target = np.random.randint(n - 1)
        if target >= source:
            target += 1

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a, M = a2, new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            print(f"[{time.time():.1f}] Hill‑climb it={hill_it} new best {best_score:.6f}")
        else:
            # probabilistic uphill move
            if T > 1e-9 and random.random() < np.exp(-(new_M - M) / T):
                a, M = a2, new_M

        # occasional global jitter
        if hill_it % 2000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            M = _fft_self_convolve(a).max()

    # ------------------------------------------------------------------
    # Upsample the current best to a larger size and give it a final push
    if time.time() < deadline - 5:
        # try a few up‑sampling factors
        for factor in (2, 3, 4):
            N = int(best_n * factor)
            if N > 200_000:          # keep memory reasonable
                continue
            if time.time() >= deadline - 5:
                break
            # linear interpolation of the shape
            xs_old = np.linspace(0.0, 1.0, best_n)
            xs_new = np.linspace(0.0, 1.0, N)
            up = np.interp(xs_new, xs_old, best_seq)
            up = np.maximum(up, eps)
            up /= up.sum()
            # a short but aggressive mirror‑descent on the larger space
            a_big = up
            iters_big = int(min(15000, (deadline - time.time()) / 0.006))
            for it in range(iters_big):
                if time.time() >= deadline:
                    break
                t = it / max(1, iters_big)
                alpha = 1.0 * (1e6) ** t
                lr = 0.5 * (0.001) ** t
                conv = _fft_self_convolve(a_big)
                maxc = conv.max()
                w = np.exp(alpha * (conv - maxc))
                w_sum = w.sum()
                if w_sum == 0 or not np.isfinite(w_sum):
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum
                grad = 2.0 * _fft_convolve(a_big[::-1], w)[N - 1 : N - 1 + N]
                a_big = a_big * np.exp(-lr * grad)
                a_big = _project_onto_simplex(a_big, 1.0)
                a_big = np.maximum(a_big, eps)
                a_big /= a_big.sum()
            _update_best(a_big)

    # ------------------------------------------------------------------
    final_sc = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_sc:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
