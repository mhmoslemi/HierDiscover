def _lp_improve(a, deadline, max_iter=200, t0=0.5, t_decay=0.99, tol=1e-12):
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _objective(a)
    t = t0
    while time.time() < deadline and max_iter > 0:
        conv = _fft_self_convolve(a)
        M = conv.max()
        # active offsets causing max
        active = np.where(conv >= M - max(tol, 1e-12 * M))[0]
        # limit number of constraints to, say, 10 by selecting most extreme offsets
        if len(active) > 10:
            # pick few offsets with largest conv (maybe top 10)
            idx_sorted = np.argsort(-conv)[:10]
            active = idx_sorted
        # Build LP matrices
        A_rows = []
        for k in active:
            row = np.zeros(n, dtype=np.float64)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            row[i_range] = a[k - i_range]
            A_rows.append(row)
        if not A_rows:
            break
        A_ub = np.vstack(A_rows)
        b_ub = np.full(len(active), M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum b
        bounds = [(0.0, None)] * n
        # solving LP
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            print("LP error:", e)
            break
        if not res.success:
            # perhaps constraints too tight, break
            break
        b_opt = np.maximum(res.x, 0.0)
        sum_b = b_opt.sum()
        if sum_b < 1e-12:
            # trivial solution; stop
            break
        b_opt /= sum_b
        # update a
        a = (1 - t) * a + t * b_opt
        # ensure sum=1, nonnegativity
        a = np.maximum(a, 0.0)
        a /= a.sum()
        # evaluate new score
        sc = _objective(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            # maybe print
            print(f"[{time.time():.1f}] LP improve new best {sc:.6f}")
        # decay step size
        t *= t_decay
        max_iter -= 1
        # optionally break if not improving after many iter
        if time.time() >= deadline - 2.0:
            break
    return best_a, best_sc

"""
A fast mixed‑strategy optimizer for the step‑function problem.

The algorithm works in four stages:

1. **Broad sampling** – generate a small elite set of candidates from many
   analytic families (uniform, arcsine, Beta, Dirichlet, sparse).
2. **Linear‑programming boost** – for a given vector `a` compute the
   maximizing partner `b` from the linear program
        maximise   1·b      (i.e. maximise the ℓ₁‑norm)
        subject to  a * b ≤ M   for the active convolution offsets,
                    b ≥ 0,
   where `M = max(a*a)`.  Blend the pair with a shrinking step size.
3. **Projected sub‑gradient polishing** – Adam‑style sub‑gradient descent
   on the exact max‑convolution, with back‑tracking line search.
4. **Hill‑climbing & pruning** – tiny random mass transfers from the
   dominant pair, occasional pruning of negligible entries and jitter.

All stages respect the time budget (default 1000 s).  The best vector found
is returned as a plain list of non‑negative floats.
"""

import time
import random
import numpy as np

# optional LP support -------------------------------------------------
try:
    from scipy.optimize import linprog
    _HAS_LP = True
except Exception:
    _HAS_LP = False
    linprog = None

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """Objective for a vector already normalised to sum = 1:
       2·n·max(a*a).  (Identical to evaluate_sequence for sum=1.)"""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return sum = 1)
def _dist_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dist_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _dist_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _dist_beta_asym(n, p, q):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _dist_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _dist_powerlaw(n, p):
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _dist_sparse(n, k):
    """k non‑zero entries, random locations, random positive weights."""
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    return sparse

# LP‑based improvement ------------------------------------------------
def _lp_improve(a: np.ndarray, deadline: float,
                max_iter: int = 80, t0: float = 0.5,
                t_decay: float = 0.98) -> tuple[np.ndarray, float]:
    """Iteratively improve `a` by solving the LP described in the
    statement.  Returns the best vector seen and its objective."""
    if not _HAS_LP:
        return a, _objective(a)

    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _objective(best_a)
    t = t0

    while time.time() < deadline and max_iter > 0:
        conv = _fft_self_convolve(a)
        M = conv.max()
        # active indices – those that attain (or are very close to) the max
        tol = max(1e-12, 1e-12 * M)
        active = np.where(conv >= M - tol)[0]
        # keep at most 10 constraints for speed
        if active.size > 10:
            active = np.argsort(-conv)[:10]
        # build A_ub rows
        rows = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            rows[r, i_range] = a[k - i_range]
        if rows.size == 0:
            break
        A_ub = rows
        b_ub = np.full(active.size, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)               # maximise sum(b)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True})
        except Exception as e:
            print("LP failure:", e)
            break
        if not res.success:
            # infeasible – nothing to improve
            break
        b_opt = np.maximum(res.x, 0.0)
        sum_b = b_opt.sum()
        if sum_b < eps:
            break
        b_opt /= sum_b                     # normalise to sum = 1

        # --- line‑search on the mixing parameter t -----------------
        a_old = a.copy()
        t_cur = t
        improved = False
        while t_cur > 1e-5 and time.time() < deadline:
            cand = (1 - t_cur) * a_old + t_cur * b_opt
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < best_sc - 1e-12:
                a = cand
                best_a = a.copy()
                best_sc = sc
                improved = True
                break
            t_cur *= 0.5
        if not improved:
            # no improvement – stop the LP loop
            break
        t *= t_decay
        max_iter -= 1
    return best_a, best_sc

# Projected sub‑gradient polishing ------------------------------------
def _projected_subgradient(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Adam‑style sub‑gradient descent on the exact max‑convolution."""
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_step = 0.4
    it = 0
    best_a = a.copy()
    best_sc = _objective(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # Adam update
        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_step
        # simple back‑tracking
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break
    return best_a, best_sc

# Hill‑climbing & pruning ---------------------------------------------
def _hill_climb_and_prune(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Local search that moves tiny masses away from the dominant pair
    and occasionally prunes very small entries."""
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        # ----- mass transfer from the pair that creates the maximal conv -----
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find the pair (i,j) with maximal product a[i]*a[j]
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            max_delta = min(a[source] * 0.5, 0.03)
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-13:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    new_sc = _objective(a)
                    if new_sc < best_sc - 1e-12:
                        best_sc = new_sc
                        best_a = a.copy()
                        if it % 200 == 0:
                            print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ----- pruning of tiny entries every few thousand steps -----
        if it % 2500 == 0:
            for keep_frac in (0.98, 0.95, 0.92):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                pruned = np.zeros_like(a)
                pruned[idx] = a[idx]
                pruned /= pruned.sum()
                sc = _objective(pruned)
                if sc < best_sc - 1e-12:
                    a = pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ----- occasional jitter ------------------------------------------------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 3.0          # a small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []                                 # (score, vector)

    # ----------------------------------------------------------------
    # 1.  Broad sampling – keep a tiny elite
    lengths = [1800, 2500, 3200, 4000, 5000]   # moderate sizes, fast to evaluate
    for n in lengths:
        if time.time() >= deadline:
            break
        # families
        candidates = [
            _dist_uniform(n),
            _dist_arcsine(n),
            _dist_exponential(n, lam=0.001),
            _dist_exponential(n, lam=0.02),
            _dist_beta_sym(n, 0.2),
            _dist_beta_sym(n, 0.5),
            _dist_beta_sym(n, 1.0),
            _dist_beta_sym(n, 1.5),
            _dist_beta_asym(n, 0.3, 1.0),
            _dist_beta_asym(n, 1.0, 0.3),
            _dist_powerlaw(n, 1.5),
            _dist_powerlaw(n, 2.5),
        ]
        # random Dirichlet draws
        for alpha in (0.2, 0.7, 1.5, 3.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))
        # a few sparse starts
        candidates.append(_dist_sparse(n, max(2, n // 400)))

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")
            # maintain elite (size ≤ 8)
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ----------------------------------------------------------------
    # 2.  Refine elite members using LP → sub‑gradient → hill‑climb
    remaining = len(elite)
    for idx, (sc0, vec0) in enumerate(elite):
        if time.time() >= deadline:
            break
        # give each elite a fair slice of the remaining budget
        now = time.time()
        per_candidate_deadline = now + (deadline - now) / (remaining - idx)
        a = vec0.copy()
        # LP boost
        a, sc = _lp_improve(a, per_candidate_deadline,
                            max_iter=80, t0=0.5, t_decay=0.98)
        # sub‑gradient polishing
        a, sc = _projected_subgradient(a, per_candidate_deadline)
        # final hill‑climb
        a, sc = _hill_climb_and_prune(a, per_candidate_deadline)

        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.time():.1f}] After elite {idx+1}/{len(elite)} new global best {sc:.6f}")

    # ----------------------------------------------------------------
    # 3.  Final safety check (fallback to uniform if nothing found)
    if best_seq is None:
        best_seq = _dist_uniform(3000)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)

    # ----------------------------------------------------------------
    # 4.  Return the best vector (already normalised to sum = 1)
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_sc = _objective(best_seq)
    print(f"Final best score ≈ {final_sc:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
