"""
Propose a non‑negative step function minimizing ``evaluate_sequence``.

Algorithm
---------
1. Broad exploration – generate analytic families (uniform, arcsine,
   beta, power‑law, exponential, Gaussian, cosine‑window, etc.) for
   several lengths (2000…20000).  Keep a tiny elite of the best vectors.

2. Local refinement – for each elite vector run three deterministic
   stages (smooth‑max mirror descent, Adam‑style projected sub‑gradient,
   and a hill‑climber with pruning).  These stages are the same as the
   reference implementation but with a slightly more aggressive schedule.

3. Simulated‑annealing polishing – starting from the best vector found
   so far, repeatedly move a small amount of mass away from the pair
   (i,j) that currently attains the maximal convolution value.  The
   move is accepted if it improves the objective or with a probability
   that decays with a temperature schedule.  Occasionally random swaps
   and jitter are performed to escape plateaus.  This stage is the new
   contribution and is the main source of improvement.

All work stops before the given time budget (default 1000 s) and the
best sequence (as a plain Python list) is returned.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Returns the value of ``evaluate_sequence`` for a vector that has been
    normalised to sum = 1 (the scaling is irrelevant because the objective
    is homogeneous of degree 0).  The factor 2·n is retained.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _generate_candidates_for_length(n):
    """Return a list of candidate probability vectors for a given length."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta
    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta
    for p, q in ((0.2, 1.0), (1.0, 0.2), (0.5, 1.5), (1.5, 0.5)):
        cand.append(_distribution_beta_asym(n, p, q))

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws (different concentration)
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0, 0.05, 0.01):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Sparse random start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Shuffle a few base families to introduce asymmetry
    for base in (cand[0], cand[1], cand[4]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

def _smooth_max_mirror_descent(a, deadline):

    n = a.shape[0]

        (1e3, 1e5, 8000, 4e-5, 1e-6),

    ]

    best_a = a.copy()
    best_sc = _objective(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:

            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 2000 == 0:
                sc = _objective(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a, deadline):
    """Adam‑style projected sub‑gradient descent."""
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    base_step = 0.4
    it = 0

    best_a = a.copy()
    best_sc = _objective(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # tolerance for active set
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum
        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_step
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc

def _annealing_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing (mass‑transfer + swap moves) starting from a0.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_score = _objective(a)

    cur_score = best_score
    conv = _fft_self_convolve(a)
    temperature = best_score  # initialise temperature proportional to scale
    decay = 0.99995  # geometric cooling

    it = 0
    while time.time() < deadline:
        it += 1

        # Identify the active (i,j) pair that yields the maximal convolution entry
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find the specific pair giving the max product (may be multiple)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # ----- propose a move -------------------------------------------------
        move_type = random.random()
        a2 = a.copy()

        if move_type < 0.6:          # mass‑transfer from the active pair
            # transfer a tiny epsilon from both ends towards random locations
            eps_frac = 0.02
            delta = eps_frac * min(a[best_i], a[best_j], 0.05)
            if delta > eps:
                tgt1 = random.randint(0, n - 1)
                while tgt1 == best_i:
                    tgt1 = random.randint(0, n - 1)
                tgt2 = random.randint(0, n - 1)
                while tgt2 == best_j:
                    tgt2 = random.randint(0, n - 1)
                a2[best_i] -= delta
                a2[tgt1] += delta
                a2[best_j] -= delta
                a2[tgt2] += delta

        elif move_type < 0.85:       # swap two random entries
            i, j = random.sample(range(n), 2)
            a2[i], a2[j] = a2[j], a2[i]

        else:                         # pure jitter (global perturbation)
            a2 += np.random.randn(n) * 1e-4
            a2 = np.maximum(a2, eps)

        # project back to simplex
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_score = _objective(a2)

        # acceptance test
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            accept = random.random() < prob

        if accept:
            a = a2
            cur_score = new_score
            conv = _fft_self_convolve(a)

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] Anneal it={it} new best {best_score:.6f}")

        # cooling
        temperature *= decay
        if temperature < 1e-8:
            temperature = 1e-8

        # occasional heavy jitter to escape deep plateaus
        if it % 5000 == 0:
            a = a + np.random.randn(n) * 5e-5
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            cur_score = _objective(a)

    return best_a, best_score

# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 4.0

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # --------------------------------------------------------------
    # 1. Broad exploration (tiny elite)
    lengths = [2000, 3500, 5000, 8000, 12000, 16000, 20000]
    elite = []          # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain elite of size ≤ 8
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2. Incorporate a previous best if it exists (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Fallback (should never happen)
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Deterministic local optimisation from elite (max 5 restarts)
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.6f}")

        # smooth‑max mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # projected sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # hill‑climber & pruning
        a_mid, sc_mid = _hill_climb_and_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after deterministic restart {sc_mid:.6f}")

    # --------------------------------------------------------------
    # 5. Simulated‑annealing polishing (the new stage)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] Starting annealing polishing (len={best_seq.shape[0]})")
        a_final, sc_final = _annealing_optimize(best_seq, deadline)
        if sc_final < best_score - 1e-12:
            best_score = sc_final
            best_seq = a_final.copy()
            best_n = a_final.shape[0]
            print(f"[{time.time():.1f}] New global best after annealing {sc_final:.6f}")

    # --------------------------------------------------------------
    # 6. Final sanitisation
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
