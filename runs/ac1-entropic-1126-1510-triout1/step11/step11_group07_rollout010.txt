"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The approach consists of three layers:

1. **Broad analytic sweep** – generate many candidate probability vectors
   from several parametric families (Beta, edge‑biased power laws,
   Dirichlet draws, mixtures).  Only the best few (by the exact
   2·n·max‑convolution score) are kept.

2. **Intensive refinement** – each kept candidate is improved using a
   schedule of three optimisation primitives:
   * a smooth‑max mirror‑descent (exponentiated‑gradient) step that
     reduces a soft‑max approximation of the convolution peak,
   * a projected Adam‑style sub‑gradient step on the true max‑convolution,
   * a final hill‑climber that moves tiny probability mass from the most
     contributing pair to the smallest entry.

   The three phases are repeated while time permits; the best vector
   found overall is retained.

3. **Optional simulated‑annealing polish** – a light random‑walk that
   occasionally accepts uphill moves, allowing the search to escape shallow
   plateaus.

All arithmetic is performed with NumPy and the exact convolution is
computed via a real FFT (O(n log n)).  The algorithm respects the 1000 s
budget, adapts the number of iterations to the remaining time and prints
progress information for debugging.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Edge‑biased deterministic distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Random Dirichlet draw with edge‑biased concentration:
        concentration ∝ ((i+½)(n‑i‑½))^{‑beta_exp} + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


def _mix_two(v1: np.ndarray, v2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·v1 + (1‑λ)·v2, subsequently projected onto the simplex."""
    mixed = lam * v1 + (1.0 - lam) * v2
    mixed = np.maximum(mixed, 0.0)
    mixed /= mixed.sum()
    return mixed

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float,
                            eps: float = 1e-12):
    """
    One exponentiated‑gradient step on the smooth‑max of the convolution.
    Returns updated vector and the exact current max (for logging).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max on the convolution values
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]
    # KL‑mirror (exponentiated‑gradient) update
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray, vel: np.ndarray,
                       step0: float, beta1: float, beta2: float,
                       it: int, eps: float = 1e-12):
    """
    Projected Adam‑style step on the true max‑convolution.
    Returns the updated vector, the new velocity and the (unscaled) max.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # actively contributing indices (within a tiny tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

    # Adam momentum (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line‑search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Deterministic hill‑climber that transfers a tiny chunk of mass from the
    index contributing most to the current peak towards the global minimum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())                # location of the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # Determine the pair (i,j) that gives the largest product for this k.
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # Choose source (the larger of the two) and a tiny target (global min).
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
        # occasional jitter to break ties
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a


def _simulated_annealing_climb(a: np.ndarray,
                               time_budget: float,
                               eps: float = 1e-12):
    """
    Light stochastic walk that occasionally accepts uphill moves.
    The temperature decays linearly with the remaining time.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(best)
    start = time.time()
    while time.time() - start < time_budget:
        # draw a small move: move mass from a random 'source' to a random 'target'
        source = np.random.randint(n)
        target = np.random.randint(n)
        if source == target:
            continue
        max_delta = min(best[source] * 0.3, 0.05)
        if max_delta <= eps:
            continue
        delta = max_delta * random.random()
        cand = best.copy()
        cand[source] -= delta
        cand[target] += delta
        cand = np.maximum(cand, eps)
        cand /= cand.sum()
        cand_score = _max_convolution_score(cand)

        # linear cooling schedule
        elapsed = time.time() - start
        T = max(1e-4, (time_budget - elapsed) / time_budget)

        # Metropolis acceptance
        if cand_score < best_score or random.random() < math.exp(-(cand_score - best_score) / T):
            best = cand
            best_score = cand_score
    return best

def _refine_candidate(a0: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Given an initial probability vector `a0`, run a sequence of optimisation
    phases within `time_limit` seconds and return the best vector and its
    exact score.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best = a.copy()
    best_score = _max_convolution_score(best)

    # ---- 1. smooth‑max mirror descent ---------------------------------
    mirror_end = time.time() + time_limit * 0.35
    # ramp α from 0.5 to 2e5, lr from 0.4 to 0.001
    alpha_start, alpha_end = 0.5, 2e5
    lr_start, lr_end = 0.4, 0.001
    it = 0
    while time.time() < mirror_end:
        it += 1
        t = (time.time() - (mirror_end - time_limit * 0.35)) / (time_limit * 0.35)
        t = max(0.0, min(1.0, t))
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] mirror it={it} new best {best_score:.6f}")

    # ---- 2. Adam sub‑gradient descent --------------------------------
    sub_end = time.time() + time_limit * 0.5
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.6
    sub_it = 0
    while time.time() < sub_end:
        sub_it += 1
        a, velocity, cur_max = _adam_subgrad_step(a, velocity, step0,
                                                  beta1, beta2, sub_it, eps)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {best_score:.6f}")

        if sub_it % 800 == 0:
            # jitter to avoid stagnation
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ---- 3. Deterministic hill‑climber --------------------------------
    hill_end = time.time() + time_limit * 0.15
    max_hill_iters = 5000
    a = best.copy()
    a = _mass_transfer_hill_climb(a, max_hill_iters, eps)
    cur_score = _max_convolution_score(a)
    if cur_score < best_score - 1e-12:
        best, best_score = a.copy(), cur_score
        print(f"[{time.time():.1f}] hill‑climber improved to {best_score:.6f}")

    # ---- 4. Optional stochastic polish (use remaining time) ---------
    remaining = time_limit - (time.time() - (time.time() - time_limit))
    if remaining > 0.2:
        a = _simulated_annealing_climb(best, remaining, eps)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] annealing improved to {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  The function obeys a hard time budget (default
    1000 s) and returns the best sequence found as a plain Python list.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin
    eps = 1e-12

    # ------------------------------------------------------------------
    # Load a previously discovered construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            best_seq = prev.copy()
            best_score = _max_convolution_score(best_seq)
            print(f"[{time.time():.1f}] loaded previous best score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 0️⃣  Global exploration over several lengths & families
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 16000, 32000, 64000, 128000]
    # Parameter grids (kept modest to respect the time budget)
    p_grid = np.linspace(0.02, 0.50, 13)               # symmetric Beta(p,p)
    asym_grid = np.linspace(0.02, 0.50, 13)            # p for asymmetric Beta(p,0.5) and vice‑versa
    alpha_grid = [0.20, 0.30, 0.40, 0.50, 0.60]        # edge‑power exponent α
    K = 12                                            # keep this many best candidates
    top_candidates = []                               # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- deterministic families ------------------------------------
        # arcsine
        vec = _distribution_arcsine(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        top_candidates.append((sc, vec.copy()))
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] n={n} arcsine score={sc:.6f}")

        # symmetric Beta(p,p)
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f}) score={sc:.6f}")

        # asymmetric Beta (left‑heavy & right‑heavy)
        for p in asym_grid:
            if time.time() >= deadline:
                break
            # left‑heavy (p small, q fixed = 0.5)
            vec = _distribution_beta(n, p, 0.5)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f},q=0.5) score={sc:.6f}")

            # right‑heavy (swap)
            vec = _distribution_beta(n, 0.5, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p=0.5,q={p:.3f}) score={sc:.6f}")

        # edge‑biased deterministic power‑law
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_power(n, alpha)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edgePower(α={alpha:.2f}) score={sc:.6f}")

        # few random Dirichlet draws with edge bias
        for _ in range(3):
            if time.time() >= deadline:
                break
            bex = random.uniform(0.1, 0.9)
            vec = _distribution_edge_dirichlet(n, bex)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Dirichlet(β={bex:.3f}) score={sc:.6f}")

        # mixture of a symmetric Beta and an edge‑power law
        for _ in range(2):
            if time.time() >= deadline:
                break
            beta_vec = _distribution_beta_sym(n, random.choice(p_grid))
            ap = random.choice(alpha_grid)
            power_vec = _distribution_edge_power(n, ap)
            lam = random.random()
            vec = _mix_two(beta_vec, power_vec, lam)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} mixture score={sc:.6f}")

    # keep only the K best globally
    top_candidates.sort(key=lambda tup: tup[0])
    top_candidates = top_candidates[:K]

    # ------------------------------------------------------------------
    # 1️⃣  Refine each promising candidate (time‑sliced)
    # ------------------------------------------------------------------
    # Allocate the remaining budget fairly among the members of top_candidates
    remaining_budget = deadline - time.time()
    if remaining_budget <= 0:
        return best_seq.tolist() if best_seq is not None else _distribution_uniform(1).tolist()

    per_candidate = remaining_budget / max(1, len(top_candidates))
    refined = []
    for idx, (_, vec) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        alloc = min(per_candidate, deadline - time.time() - 0.1)
        if alloc <= 0:
            break
        print(f"[{time.time():.1f}] Refining candidate {idx+1}/{len(top_candidates)} (len={vec.shape[0]}, budget={alloc:.2f}s)")
        cand, cand_score = _refine_candidate(vec, alloc)
        refined.append((cand_score, cand))
        if cand_score < best_score:
            best_seq, best_score = cand.copy(), cand_score
            print(f"[{time.time():.1f}] NEW GLOBAL BEST {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  One final polishing on the overall best (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.1:
        extra = deadline - time.time()
        print(f"[{time.time():.1f}] Final polish extra time={extra:.2f}s")
        best_seq, best_score = _refine_candidate(best_seq, extra)

    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
