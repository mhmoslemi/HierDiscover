def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact score for a vector already normalised to sum = 1.
    Returns 2·n·max(a*a) (the same value produced by `evaluate_sequence`).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float) -> np.ndarray:
    """Deterministic edge‑biased power law: w_i ∝ ((i+½)(n‑i‑½))^{‑α}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha)
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_power_bias(n: int, alpha: float, c_rel: float) -> np.ndarray:
    """
    Edge‑biased power law with a uniform floor.
    c_rel is a factor multiplied by 1/n to obtain an absolute floor.
    """
    idx = np.arange(n, dtype=np.float64)
    base = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha)
    c = c_rel / n
    w = base + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """lam∈[0,1] blends arcsine (lam=0) with uniform (lam=1)."""
    arcs = _distribution_arcsine(n)
    uni = _distribution_uniform(n)
    w = (1.0 - lam) * arcs + lam * uni
    w = np.maximum(w, 1e-15)
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """Random Dirichlet draw with edge‑biased concentration."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec

# Optimisation primitives
# ----------------------------------------------------------------------


def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float,
                           eps: float = 1e-15):
    """
    One exponentiated‑gradient step on the smooth‑max approximation.
    Returns the updated probability vector and the exact max of the
    (unsmoothed) convolution.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights, numerically stable
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # mirror‑descent (KL‑projection) update
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-15):
    """
    One Adam‑style sub‑gradient step on the true max‑convolution.
    Returns the updated vector, the new velocity, and the current max.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices achieving the max, tolerance safe)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-15):
    """
    Tiny mass‑transfer hill‑climber.
    Moves a small amount of mass from the pair contributing most to the
    current convolution peak to the globally smallest entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()

    for it in range(max_iters):
        k = int(conv.argmax())               # index of the current peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) with the largest product a_i*a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            # second‑smallest if source is already the minimum
            target = np.argsort(a)[1]

        # take a tiny amount of probability mass
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional jitter to escape flat regions
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return a

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative sequence (list of floats) that minimises
    `evaluate_sequence` within the supplied time budget.
    """
    # ------------------------------------------------------------------
    # 0️⃣  Random seeds and timing
    # ------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0      # safety margin

    eps = 1e-15

    # ------------------------------------------------------------------
    # 0️⃣  Load a previously discovered construction if present
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            best_seq = prev.copy()
            best_score = _max_convolution_score(best_seq)
            print(f"[{time.time():.1f}] Loaded previous best score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration over several lengths and families
    # ------------------------------------------------------------------
    # Candidate lengths (moderate – larger sizes give diminishing returns)
    lengths = [2000, 3000, 4000, 5000, 8000, 12000]

    # Parameter grids
    p_grid_sym = np.linspace(0.01, 0.50, 45)        # symmetric Beta(p,p)
    p_grid_asym = np.linspace(0.01, 0.50, 20)      # asymmetric Beta(p,0.5) and its swap
    alpha_grid = [0.30, 0.35, 0.40, 0.45, 0.50]    # edge‑power exponent
    c_rel_grid = [0.0, 0.2, 0.5, 1.0, 2.0]         # uniform floor factor (relative to 1/n)
    lam_grid = np.linspace(0.0, 0.5, 7)            # mixture arcsine ↔ uniform

    K = 12                                        # keep top‑K globally
    top_heap = []                                 # min‑heap of (score, vector)

    def _push_candidate(vec):
        nonlocal best_seq, best_score
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] New global best score={sc:.6f} (len={vec.shape[0]})")
        if len(top_heap) < K:
            heapq.heappush(top_heap, (-sc, vec.copy()))
        else:
            # heap stores negative scores for max‑heap behaviour
            if -sc > top_heap[0][0]:
                heapq.heapreplace(top_heap, (-sc, vec.copy()))

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- deterministic families ------------------------------------
        # arcsine
        _push_candidate(_distribution_arcsine(n))

        # uniform (useful as a sanity check)
        _push_candidate(_distribution_uniform(n))

        # symmetric Beta(p,p)
        for p in p_grid_sym:
            if time.time() >= deadline:
                break
            _push_candidate(_distribution_beta_sym(n, p))

        # asymmetric Beta (left‑heavy and right‑heavy)
        q_fixed = 0.5
        for p in p_grid_asym:
            if time.time() >= deadline:
                break
            _push_candidate(_distribution_beta(n, p, q_fixed))
            _push_candidate(_distribution_beta(n, q_fixed, p))

        # edge‑biased power law (no floor)
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            _push_candidate(_distribution_edge_power(n, alpha))

        # edge‑power with uniform floor
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            for c_rel in c_rel_grid:
                if time.time() >= deadline:
                    break
                _push_candidate(_distribution_edge_power_bias(n, alpha, c_rel))

        # mixture arcsine + uniform
        for lam in lam_grid:
            if time.time() >= deadline:
                break
            _push_candidate(_distribution_mixture_arcsine_uniform(n, lam))

        # random Dirichlet draws with edge bias
        for _ in range(6):
            if time.time() >= deadline:
                break
            bex = random.uniform(0.2, 0.8)
            _push_candidate(_distribution_edge_dirichlet(n, bex))

    # Extract the best K candidates (ascending by score)
    top_candidates = [( -score, vec) for (score, vec) in top_heap]
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = [vec for (_, vec) in top_candidates]

    # ------------------------------------------------------------------
    # 2️⃣  Multi‑start optimisation
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution of a medium size
        n0 = 5000
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    # Helper to run the full optimisation pipeline on a given start vector
    def _optimise_start(a_start):
        nonlocal best_seq, best_score
        a = a_start.copy()
        n = a.shape[0]

        # ----- Smooth‑max mirror descent -----
        remaining = deadline - time.time()
        # allocate a modest budget per start (max 0.25 s per 5k entries)
        est_per_iter = 0.004 if n > 8000 else 0.002
        max_iters = max(2000, min(12000, int(remaining / (est_per_iter * len(top_candidates) + 1e-9))))
        alpha_start, alpha_end = 0.5, 3e5
        lr_start, lr_end = 0.4, 0.001

        for it in range(max_iters):
            if time.time() >= deadline:
                break
            t = it / max_iters
            alpha = alpha_start * (alpha_end / alpha_start) ** t
            lr = lr_start * (lr_end / lr_start) ** t
            a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)
            if (it + 1) % max(1, max_iters // 8) == 0:
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score - 1e-12:
                    best_seq, best_score = a.copy(), cur_score
                    print(f"[{time.time():.1f}] Mirror it={it}/{max_iters} improved to {cur_score:.6f}")

        # ----- Projected Adam sub‑gradient descent -----
        velocity = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        step0 = 0.6
        sub_it = 0
        while time.time() < deadline - 0.5:
            sub_it += 1
            a, velocity, cur_max = _adam_subgrad_step(a, velocity, step0,
                                                      beta1, beta2, sub_it, eps)
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] Subgrad it={sub_it} improved to {cur_score:.6f}")

            if sub_it % 800 == 0:
                jitter = np.random.randn(n) * 1e-5
                a = np.maximum(a + jitter, eps)
                a /= a.sum()

        # ----- Hill‑climber polishing -----
        a = _mass_transfer_hill_climb(a, max_iters=4000, eps=eps)
        final_sc = _max_convolution_score(a)
        if final_sc < best_score:
            best_seq, best_score = a.copy(), final_sc
            print(f"[{time.time():.1f}] Hill‑climb improved to {final_sc:.6f}")

        return a

    # Run optimisation for each of the top candidates (except the already‑best one)
    for idx, vec in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # Skip if this candidate is already the global best (within tolerance)
        if np.allclose(vec, best_seq, atol=1e-12):
            continue
        print(f"[{time.time():.1f}] Optimising start #{idx + 1}/{len(top_candidates)} (len={vec.shape[0]})")
        _optimise_start(vec)

    # ------------------------------------------------------------------
    # 3️⃣  Final fine‑tuning of the global best
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print(f"[{time.time():.1f}] Final polishing of global best (len={best_seq.shape[0]})")
        a = best_seq.copy()
        n = a.shape[0]

        # a short extra mirror‑descent run with higher temperature
        extra_iters = 3000
        alpha_start, alpha_end = 0.5, 5e5
        lr_start, lr_end = 0.3, 0.0005
        for it in range(extra_iters):
            if time.time() >= deadline:
                break
            t = it / extra_iters
            alpha = alpha_start * (alpha_end / alpha_start) ** t
            lr = lr_start * (lr_end / lr_start) ** t
            a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)

        # final hill‑climb
        a = _mass_transfer_hill_climb(a, max_iters=6000, eps=eps)
        final_score = _max_convolution_score(a)
        if final_score < best_score:
            best_seq, best_score = a.copy(), final_score
            print(f"[{time.time():.1f}] Final polish improved score to {final_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
