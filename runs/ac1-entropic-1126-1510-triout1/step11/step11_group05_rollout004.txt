        # sort

"""
Improved search for a non‑negative step function that minimises
`evaluate_sequence`.

The algorithm combines several ideas that were not used together before:

1. **Rich analytic families** – uniform, symmetric and asymmetric Beta
   distributions, mixtures with the uniform law, exponential, Gaussian,
   logarithmic and sinusoidal shapes, sparse Dirichlet draws and shuffled
   variants.  All families return vectors already normalised to sum = 1.

2. **Genetic‐type population search** – a small population (≈30) is
   evolved with crossover, Gaussian mutation and simplex projection.
   The population is refreshed repeatedly until the time budget for the
   current length is exhausted.

3. **Upsampling & refinement** – the best vectors found for a moderate
   length are interpolated to larger powers of two and then polished
   with three deterministic local‑optimisers:
   * smooth‑max mirror descent,
   * Adam‑style projected sub‑gradient descent,
   * greedy hill‑climbing with aggressive pruning.

All heavy loops respect a global deadline (``budget_s – 5 s`` safety
margin).  The final sequence is returned as a plain ``list[float]`` and
conforms to the security checks of ``evaluate_sequence``.
"""

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, a, b):
    """Asymmetric Beta(a,b) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_mixture_two_beta(n, a1, b1, a2, b2, w):
    """w·Beta(a1,b1) + (1‑w)·Beta(a2,b2)."""
    beta1 = _distribution_beta_asym(n, a1, b1)
    beta2 = _distribution_beta_asym(n, a2, b2)
    mix = w * beta1 + (1.0 - w) * beta2
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_exponential(n, r):
    """Truncated exponential on {0,…,n‑1} with rate r (0<r<1)."""
    idx = np.arange(n, dtype=np.float64)
    w = (1 - r) * (r ** idx)
    w /= w.sum()
    return w


def _distribution_gaussian(n, sigma):
    """Discrete symmetric Gaussian centred in the middle."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_logarithmic(n):
    """a_i ∝ 1/(i+1) + 1/(n‑i). Symmetric logarithmic decay."""
    i = np.arange(n, dtype=np.float64) + 1.0
    w = 1.0 / i + 1.0 / (n + 1 - i)
    w /= w.sum()
    return w


def _distribution_sine(n):
    """a_i ∝ sin(π (i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))   # arcsine

    # symmetric beta grid
    for p in np.concatenate(
        (np.logspace(-2, -0.2, 10), np.linspace(0.1, 2.0, 15))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # asymmetric beta grid
    for a, b in [(0.3, 1.5), (1.5, 0.3), (0.2, 2.0), (2.0, 0.2)]:
        cand.append(_distribution_beta_asym(n, a, b))

    # mixtures
    for p in (0.05, 0.2, 0.6, 1.0):
        for w in (0.2, 0.5, 0.8):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    for (a1, b1, a2, b2) in [(0.3, 0.3, 2.0, 2.0), (0.5, 1.5, 1.5, 0.5)]:
        for w in (0.3, 0.7):
            cand.append(_distribution_mixture_two_beta(n, a1, b1, a2, b2, w))

    # exponential decays
    for r in (0.9, 0.95, 0.98):
        cand.append(_distribution_exponential(n, r))

    # Gaussian blobs
    for sigma in (n / 6.0, n / 4.0, n / 3.0):
        cand.append(_distribution_gaussian(n, sigma))

    # logarithmic and sine
    cand.append(_distribution_logarithmic(n))
    cand.append(_distribution_sine(n))

    # random Dirichlet draws
    for alpha in (0.05, 0.2, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start
    k = max(2, n // 400)
    cand.append(_distribution_sparse_random(n, k))

    # shuffled versions of a few bases
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Local optimisation ----------------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """Exponentiated‑gradient descent on a smooth max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    schedule = [
        (1.0, 1e3, 3000, 0.15, 0.015),
        (1e3, 1e5, 4000, 0.015, 0.0015),
        (1e5, 1e6, 5000, 0.0015, 0.00015),
        (1e6, 1e8, 3000, 0.00015, 1e-5),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline):
    """Adam‑style projected sub‑gradient with back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a0, deadline):
    """Deterministic hill‑climbing with aggressive pruning."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # move mass away from the most offending pair
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j

        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # aggressive pruning every few thousand steps
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # occasional jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Upsampling ------------------------------------------------------------
def _upsample(a, n_target):
    """Linear interpolation of a normalised vector to a new length."""
    n0 = a.shape[0]
    if n_target == n0:
        return a.copy()
    x_src = np.linspace(0.0, 1.0, n0, endpoint=False)
    x_tgt = np.linspace(0.0, 1.0, n_target, endpoint=False)
    a_tgt = np.interp(x_tgt, x_src, a)
    a_tgt = np.maximum(a_tgt, 0.0)
    s = a_tgt.sum()
    if s == 0:
        a_tgt = _distribution_uniform(n_target)
    else:
        a_tgt /= s
    return a_tgt

# Genetic‑type population search -----------------------------------------
def _run_ga_for_length(n, deadline):
    """Simple GA that respects the given deadline."""
    pop_size = 30
    elite_size = 5
    eps = 1e-12

    # initialise population with random Dirichlet draws
    pop = []
    scores = []
    for _ in range(pop_size):
        vec = np.random.dirichlet(np.full(n, 0.5))
        pop.append(vec)
        scores.append(_max_convolution_score(vec))

    order = np.argsort(scores)
    pop = [pop[i] for i in order]
    scores = [scores[i] for i in order]

    best_vec = pop[0].copy()
    best_sc = scores[0]

    # main GA loop
    while time.time() < deadline:
        # generate offspring from elite
        offspring = []
        for _ in range(pop_size - elite_size):
            i, j = random.sample(range(elite_size), 2)
            parent1, parent2 = pop[i], pop[j]
            w = random.random()
            child = w * parent1 + (1.0 - w) * parent2

            # mutation
            if random.random() < 0.8:
                sigma = 0.02
                child += sigma * np.random.randn(n)
                child = np.maximum(child, eps)

            child_sum = child.sum()
            if child_sum == 0:
                child = _distribution_uniform(n)
            else:
                child /= child_sum

            offspring.append(child)

        # evaluate offspring
        off_scores = [_max_convolution_score(v) for v in offspring]

        # merge and keep best pop_size individuals
        pop.extend(offspring)
        scores.extend(off_scores)
        order = np.argsort(scores)
        pop = [pop[i] for i in order[:pop_size]]
        scores = [scores[i] for i in order[:pop_size]]

        if scores[0] < best_sc - 1e-12:
            best_sc = scores[0]
            best_vec = pop[0].copy()
            print(f"[{time.time():.1f}] GA n={n} new best {best_sc:.6f}")

        # occasional injection of a fresh random individual
        if random.random() < 0.07:
            vec = np.random.dirichlet(np.full(n, 0.5))
            sc = _max_convolution_score(vec)
            if sc < scores[-1]:
                pop[-1] = vec
                scores[-1] = sc

    return best_vec, best_sc

# Main orchestrator ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step function (list of non‑negative floats) that
    minimises ``evaluate_sequence``.  The routine respects a 1000 s
    budget and returns the best sequence it has found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # safety margin of a few seconds
    deadline = start + budget_s - 5.0

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []        # list of (score, vector) tuples, sorted low→high

    # ------------------------------------------------------------------
    # 1. Broad search on moderate lengths (powers of two)
    lengths_small = [1024, 2048, 4096, 8192]

    for n in lengths_small:
        if time.time() >= deadline:
            break

        # ---- deterministic families -------------------------------------------------
        candidates = _generate_candidates_extended(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")

            # keep elite (max size 6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, vec.copy())
                elite.sort(key=lambda x: x[0])

        # ---- genetic search ---------------------------------------------------------
        remaining = deadline - time.time()
        # allocate at most half of the remaining budget for GA on this length
        ga_deadline = time.time() + min(remaining * 0.5, 30.0)
        if ga_deadline - time.time() > 2.0:
            vec_ga, sc_ga = _run_ga_for_length(n, ga_deadline)
            if sc_ga < best_score:
                best_score = sc_ga
                best_seq = vec_ga.copy()
                print(f"[{time.time():.1f}] GA global new best len={n}  score={sc_ga:.6f}")

            if len(elite) < 6:
                elite.append((sc_ga, vec_ga.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc_ga < elite[-1][0]:
                elite[-1] = (sc_ga, vec_ga.copy())
                elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Use any historic solution (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, prev.copy())
                elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. If nothing found (should not happen), fall back to uniform
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Refinement of elite solutions on their native lengths
    restarts = min(4, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.6f}")

        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Upsample elite candidates to larger lengths and polish
    lengths_big = [16384, 32768, 65536]
    for n_big in lengths_big:
        if time.time() >= deadline:
            break
        # allocate a slice of remaining time for this length
        slice_deadline = time.time() + (deadline - time.time()) * 0.4
        for _, vec in elite[:3]:          # only the very best few
            if time.time() >= slice_deadline:
                break
            a_up = _upsample(vec, n_big)
            # modest local polishing (no need to repeat all three phases)
            a_up, sc_up = _smooth_max_mirror_descent(a_up, slice_deadline)
            a_up, sc_up = _projected_subgradient(a_up, slice_deadline)
            a_up, sc_up = _hill_climb_prune(a_up, slice_deadline)

            if sc_up < best_score - 1e-12:
                best_score = sc_up
                best_seq = a_up.copy()
                print(f"[{time.time():.1f}] Upsample len={n_big} new best {sc_up:.6f}")

    # ------------------------------------------------------------------
    # 6. Final sanity – enforce non‑negativity and normalise
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
