def _improved_mass_transfer(a, max_iters, eps=1e-12):
    n = a.shape[0]
    for it in range(max_iters):
        conv = _fft_self_convolve(a)
        M = conv.max()
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find pair giving max product
        # we can vectorize: compute products for all i in range; but loop ok as range up to maybe n.
        # Usually i_end - i_start small near edges? Actually k can be anywhere 0..2n-2. Worst-case range length n.
        # We'll loop.
        best_i = i_start
        best_j = k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end+1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        # sources: best_i, best_j
        # Choose two distinct minimal indices (could be same if many equal)
        mins = np.argsort(a)[:2]  # two smallest
        tgt1, tgt2 = mins[0], mins[1] if mins[1] != best_i and mins[1] != best_j else mins[2]
        # transfer delta from each source proportional to its weight
        delta_i = min(a[best_i] * 0.5, 0.02) * random.random()
        delta_j = min(a[best_j] * 0.5, 0.02) * random.random()
        # ensure small
        if delta_i < eps and delta_j < eps:
            continue
        a2 = a.copy()
        a2[best_i] -= delta_i
        a2[tgt1] += delta_i
        a2[best_j] -= delta_j
        a2[tgt2] += delta_j
        # clip
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
        # occasional jitter
        if (it + 1) % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
    return a

def _random_local_search(a, iters, eps=1e-12):
    n = len(a)
    for it in range(iters):
        i, j = random.sample(range(n), 2)
        # transfer small mass from i to j
        delta = min(a[i] * 0.5, 0.01) * random.random()
        if delta < eps:
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        if _max_convolution_score(a2) < _max_convolution_score(a) - 1e-12:
            a = a2
    return a

"""
Algorithm:
1. Broad parametric search over several families (arcsine, symmetric/asymmetric Beta,
   edge‑biased power law with additive constant, sinusoidal power) for multiple
   lengths. Keep the best candidate.
2. Refine the best candidate with a smooth‑max mirror‑descent (exponentiated‑gradient)
   that approximates the max convolution. Temperature grows to a very large value,
   learning‑rate decays.
3. Apply an Adam‑style projected sub‑gradient descent on the true max‑convolution.
4. Perform an improved mass‑transfer hill‑climber that removes probability mass
   from the pair contributing to the current peak and redistributes it to the
   smallest entries.
5. Finish with a cheap random local search.
All steps respect non‑negativity and the simplex constraint.
"""

import time, random, math, numpy as np

# Helper functions: projection onto simplex, FFT convolution, etc.

def _project_onto_simplex(v, s=1.0):
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a):
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """Assume a already sums to 1. Returns 2*n*max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_beta(n, p, q):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_edge_power(n, alpha, c):
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w

def _distribution_edge_dirichlet(n, beta_exp):
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec

def _distribution_sin_pow(n, beta, c=0.0):
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** beta + c
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= s
    return vals

# --- optimisation primitives -------------------------------------------------
def _smooth_max_mirror_step(a, alpha, lr, eps=1e-12):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of smooth max: 2 * (a * w) convolved with reversed a
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()

def _adam_subgrad_step(a, vel, step0, beta1, beta2, it, eps=1e-12):
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)
    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5
    return a_cand, vel, conv.max()

def _improved_mass_transfer(a, max_iters, eps=1e-12):
    n = a.shape[0]
    for it in range(max_iters):
        conv = _fft_self_convolve(a)
        M = conv.max()
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # locate pair giving max product
        best_i = i_start
        best_j = k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        # choose two smallest distinct indices (avoid sources)
        mins = np.argsort(a)
        # ensure they are not the source indices
        tgt = []
        for idx in mins:
            if idx != best_i and idx != best_j:
                tgt.append(idx)
            if len(tgt) == 2:
                break
        if len(tgt) < 2:
            # fallback to any distinct index
            tgt = [(best_i + 1) % n, (best_i + 2) % n]
        tgt1, tgt2 = tgt[0], tgt[1]
        # transfer small mass from both sources
        delta_i = min(a[best_i] * 0.5, 0.02) * random.random()
        delta_j = min(a[best_j] * 0.5, 0.02) * random.random()
        if delta_i < eps and delta_j < eps:
            continue
        a2 = a.copy()
        a2[best_i] -= delta_i
        a2[tgt1] += delta_i
        a2[best_j] -= delta_j
        a2[tgt2] += delta_j
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
        # jitter occasionally
        if (it + 1) % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
    return a

def _random_local_search(a, iters, eps=1e-12):
    n = len(a)
    best_score = _max_convolution_score(a)
    for it in range(iters):
        i, j = random.sample(range(n), 2)
        delta = min(a[i] * 0.5, 0.01) * random.random()
        if delta < eps:
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_score = _max_convolution_score(a2)
        if new_score < best_score - 1e-12:
            a, best_score = a2, new_score
    return a

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine works under a hard time budget (default
    1000 s) and returns the best sequence found.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    eps = 1e-12

    # Try to load a previously discovered construction
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            best_seq = prev.copy()
            best_score = _max_convolution_score(best_seq)
            print(f"[{time.time():.1f}] loaded previous best score={best_score:.6f}")
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 1️⃣  Broad parametric search
    # -----------------------------------------------------------------
    def consider(vec):
        nonlocal best_seq, best_score
        # make sure of simplex
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq = vec.copy()
            best_score = sc
            print(f"[{time.time():.1f}]   new best {sc:.6f} length={len(vec)}")
        return

    lengths = [2000, 3000, 4000, 5000, 8000, 12000, 20000]
    beta_factors = [0.0, 0.01, 0.05, 0.1]   # added constant as fraction of 1/n
    for n in lengths:
        if time.time() >= deadline:
            break
        # arcsine
        consider(_distribution_arcsine(n))
        # uniform as sanity
        consider(_distribution_uniform(n))
        # symmetric beta
        for p in np.linspace(0.08, 0.50, 9):
            if time.time() >= deadline: break
            consider(_distribution_beta_sym(n, p))
        # asymmetric beta (fix q=0.5, vary p)
        q_fixed = 0.5
        for p in np.linspace(0.08, 0.50, 9):
            if time.time() >= deadline: break
            consider(_distribution_beta(n, p, q_fixed))
            consider(_distribution_beta(n, q_fixed, p))
        # edge‑biased power law with offset
        for alpha in np.arange(0.08, 0.91, 0.08):
            if time.time() >= deadline: break
            for cf in beta_factors:
                c = cf / n
                consider(_distribution_edge_power(n, alpha, c))
        # sinusoidal power
        for beta in [0.5, 1.0, 2.0, 3.0, 4.0]:
            if time.time() >= deadline: break
            consider(_distribution_sin_pow(n, beta))
        # random Dirichlet draws biased to edges
        for _ in range(6):
            if time.time() >= deadline: break
            bex = random.uniform(0.1, 1.0)
            consider(_distribution_edge_dirichlet(n, bex))

    # -----------------------------------------------------------------
    # 2️⃣  Smooth‑max mirror descent (fine‑grained)
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = _distribution_arcsine(8000)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    a = best_seq.copy()
    n = a.shape[0]

    remaining = deadline - time.time()
    # Estimate cheap iteration cost: ~0.001 s per iter for n<=2e4, else a bit more.
    est_per = 0.0015 if n <= 20000 else 0.003
    max_iters = max(0, int(remaining / est_per))
    max_iters = min(max_iters, 30000)   # cap to keep time reasonable

    if max_iters > 0:
        print(f"[{time.time():.1f}] Mirror descent – n={n}, iters={max_iters}")
        # schedule parameters
        alpha_start, alpha_end = 0.3, 5e5
        lr_start, lr_end = 0.4, 0.001
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            t = it / max_iters
            alpha = alpha_start * (alpha_end / alpha_start) ** t
            lr = lr_start * (lr_end / lr_start) ** t
            a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_seq = a.copy()
                best_score = cur_score
                print(f"[{time.time():.1f}] mirror it={it} new best {cur_score:.6f}")

    # -----------------------------------------------------------------
    # 3️⃣  Adam sub‑gradient descent on the true max
    # -----------------------------------------------------------------
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.6
    sub_it = 0
    while time.time() < deadline - 0.5:
        sub_it += 1
        a, velocity, cur_max = _adam_subgrad_step(a, velocity, step0, beta1, beta2, sub_it, eps)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_seq = a.copy()
            best_score = cur_score
            print(f"[{time.time():.1f}] Adam it={sub_it} new best {cur_score:.6f}")

        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # -----------------------------------------------------------------
    # 4️⃣  Improved mass‑transfer hill‑climber
    # -----------------------------------------------------------------
    a = best_seq.copy()
    a = _improved_mass_transfer(a, max_iters=8000, eps=eps)
    final_score = _max_convolution_score(a)
    if final_score < best_score:
        best_seq = a.copy()
        best_score = final_score
        print(f"[{time.time():.1f}] hill‑climb improved to {final_score:.6f}")

    # -----------------------------------------------------------------
    # 5️⃣  Random local search for any remaining slack
    # -----------------------------------------------------------------
    a = best_seq.copy()
    a = _random_local_search(a, iters=12000)
    final_score = _max_convolution_score(a)
    if final_score < best_score:
        best_seq = a.copy()
        best_score = final_score
        print(f"[{time.time():.1f}] random local search final best {final_score:.6f}")

    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()

Search for a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  The routine works under a strict time budget (default
1000 s) and proceeds in four stages:

1️⃣  **Broad parametric exploration** – many analytic families
    (arcsine, symmetric / asymmetric Beta, edge‑biased power‑law with an
    additive constant, sinusoidal powers, random Dirichlet draws) are
    instantiated for several lengths.  The best candidate is retained.

2️⃣  **Smooth‑max mirror descent** – an exponentiated‑gradient step is taken
    on the soft‑max approximation of the max‑convolution.  Temperature grows
    to a very large value while the learning rate decays, turning the
    optimisation from exploration to precise fine‑tuning.

3️⃣  **Projected Adam sub‑gradient descent** on the true max‑convolution,
    followed by a tiny jitter every few hundred steps.

4️⃣  **Improved mass‑transfer hill‑climber** – probability mass is repeatedly
    moved from the pair that contributes most to the current peak towards the
    two smallest entries, with occasional jitter.

5️⃣  **Random local search** – a cheap stochastic mass‑transfer that tries to
    escape any remaining shallow minima.

All updates respect the simplex constraint (non‑negative entries that sum to 1)
by means of Euclidean projection.  The best sequence found before the deadline
is returned as a plain Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    a is assumed to already sum to 1.
    Returns the exact value produced by ``evaluate_sequence`` (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (each returns a probability vector, sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Random Dirichlet draw with concentration concentrated at the edges:
    concentration ∝ ((i+½)(n‑i‑½))^{‑beta_exp} + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


def _distribution_sin_pow(n: int, beta: float, c: float = 0.0) -> np.ndarray:
    """Sinusoidal power law: w_i ∝ sin(π·(i+½)/n)^β + c."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** beta + c
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= s
    return vals

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float,
                            eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector and the (exact) current maximum
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray, vel: np.ndarray,
                       step0: float, beta1: float, beta2: float,
                       it: int, eps: float = 1e-12):
    """
    One Adam‑style projected sub‑gradient step on the true max‑convolution.
    Returns the updated vector, the new velocity and the new maximal
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _improved_mass_transfer(a: np.ndarray, max_iters: int,
                           eps: float = 1e-12) -> np.ndarray:
    """
    Hill‑climber that repeatedly removes a tiny amount of probability mass
    from the *pair* that contributes most to the current peak and splits it
    between the two smallest entries (different from the sources).
    """
    n = a.shape[0]
    for it in range(max_iters):
        conv = _fft_self_convolve(a)
        M = conv.max()
        k = int(conv.argmax())

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # pair with maximal product a_i * a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # two smallest distinct indices not equal to sources
        mins = np.argsort(a)
        targets = []
        for idx in mins:
            if idx != best_i and idx != best_j:
                targets.append(idx)
            if len(targets) == 2:
                break
        if len(targets) < 2:        # extremely unlikely, but protect
            targets = [(best_i + 1) % n, (best_i + 2) % n]

        tgt1, tgt2 = targets[0], targets[1]

        # tiny transfers (never more than 2 % of the source weight)
        delta_i = min(a[best_i] * 0.5, 0.02) * random.random()
        delta_j = min(a[best_j] * 0.5, 0.02) * random.random()
        if delta_i < eps and delta_j < eps:
            continue

        a2 = a.copy()
        a2[best_i] -= delta_i
        a2[tgt1] += delta_i
        a2[best_j] -= delta_j
        a2[tgt2] += delta_j

        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2

        if (it + 1) % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
    return a


def _random_local_search(a: np.ndarray, iters: int,
                         eps: float = 1e-12) -> np.ndarray:
    """Very cheap random mass‑transfer hill‑climber."""
    n = len(a)
    best_score = _max_convolution_score(a)
    for _ in range(iters):
        i, j = random.sample(range(n), 2)
        delta = min(a[i] * 0.5, 0.01) * random.random()
        if delta < eps:
            continue
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_score = _max_convolution_score(a2)
        if new_score < best_score - 1e-12:
            a, best_score = a2, new_score
    return a

        sc = _max_convolution_score(vec)

    lengths = [2000, 3000, 4000, 5000, 8000, 12000, 20000]
    const_factors = [0.0, 0.01, 0.05, 0.1]          # additive constant = factor / n
    for n in lengths:
        if time.time() >= deadline:

        # 1) basic families
        consider(_distribution_arcsine(n))
        consider(_distribution_uniform(n))

        # 2) symmetric Beta(p,p)

            if time.time() >= deadline: break
            consider(_distribution_beta_sym(n, p))

            consider(_distribution_beta(n, p, q_fixed))

        for _ in range(6):
            if time.time() >= deadline: break

        # occasional jitter to shake out tiny plateaus
        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # -----------------------------------------------------------------
    # 4️⃣  Improved mass‑transfer hill‑climber
    # -----------------------------------------------------------------
    a = best_seq.copy()
    a = _improved_mass_transfer(a, max_iters=8000, eps=eps)
    final_score = _max_convolution_score(a)

        best_seq, best_score = a.copy(), final_score

    # 5️⃣  Random local search for any remaining slack

    # Return the best sequence found
    # -----------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
