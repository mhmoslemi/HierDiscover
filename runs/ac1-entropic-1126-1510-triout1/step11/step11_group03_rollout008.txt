def _adam_descent(a, deadline, best_seq, best_score, ...):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # compute softmax weights:
        alpha = alpha_start + (alpha_end - alpha_start) * min(t / max_iter, 1.0)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        # gradient:
        grad = _gradient_from_weights(a, w)
        # compute Adam step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)

def _mass_transfer_greedy(a, deadline, best_seq, best_score):
    n = len(a)
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute gradient based on active max entries
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # compute mask active indices within tolerance 1e-12? Actually difference < 1e-12?
        tol = 1e-12
        active = np.where(conv >= max_conv - tol)[0]
        # compute weight w = uniform over active
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
        grad = _gradient_from_weights(a, w)
        # find indices with largest gradient and smallest
        # gradient is positive; we want to move mass from largest to smallest
        # if many same, random
        grads = grad
        src = np.argmax(grads)
        dst = np.argmin(grads)
        # propose transfer
        delta = a[src] * 0.2  # move up to 20% of source
        if delta < 1e-12:
            # find another source with >0
            candidates = np.where(a > 1e-12)[0]
            if candidates.size == 0:
                break
            src = rng.choice(candidates)
            delta = a[src] * 0.2
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        # symmetrize after move

def _smooth_descent(a, deadline,
                    best_seq, best_score,
                    max_iter=60000,
                    lr0=0.9,
                    alpha_start=0.5,
                    alpha_end=30.0,
                    eval_interval=500):
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter
        # compute softmax weights
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad = _gradient_from_weights(a, w)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # project
        a = _project_onto_simplex(a)
        # symmetrize
        a = (a + a[::-1]) / 2.0
        # optional: evaluate
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.2f} new best {best_score:.6f}")
    return best_seq, best_score

def _mass_transfer_greedy(a, deadline, best_seq, best_score):
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # active indices
        active = np.where(conv >= max_conv - 1e-12)[0]
        if active.size == 0:
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)
        grad = _gradient_from_weights(a, w)
        # find source with highest gradient, sink with lowest
        src = np.argmax(grad)
        dst = np.argmin(grad)
        if a[src] < 1e-12:
            # skip zero source, find another
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size == 0:
                break
            src = rng.choice(src_candidates)
        # propose moving a fraction
        delta = a[src] * 0.2  # move up to 20%
        if delta < 1e-12:
            break
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        # symmetrize
        new_a = (new_a + new_a[::-1]) / 2.0
        # normalize (sum unchanged but after symmetrization may cause rounding error)
        new_a = new_a / new_a.sum()
        cur = _objective(new_a)
        if cur < best_score - 1e-12:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] mass move src={src} dst={dst} Δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score

The algorithm works in three stages:

1. **Seed generation** – a broad pool of analytic distributions
   (uniform, arcsine, beta, power‑law, exponential, sparse,
   two‑block, triangular, Dirichlet) is created for several lengths.
   All seeds are normalised to sum = 1 and their exact objective
   `2·n·max(conv(a,a))` is evaluated.  The best few seeds are kept.

2. **Smooth exponentiated‑gradient descent** – for each selected seed
   we minimise a soft‑max approximation of the maximal self‑convolution:
   `w_i ∝ exp(α·conv_i)` with a temperature α that is increased from a
   low value to a large one (≈150).  The gradient of the smooth objective
   is `∇ = 2·a⋆w` (cross‑correlation) which is obtained via FFT.
   An exponentiated‑gradient update `a ← a·exp(‑η·∇)` is performed,
   followed by Euclidean projection onto the simplex and an explicit
   symmetrisation step `a ← (a + a[::-1])/2`.  The step size η decays
   like `η₀/√t`.  The best iterate over the run is kept.

3. **Gradient‑driven greedy polishing** – using the exact max‑convolution
   we compute a sub‑gradient (average over all active maximal indices)
   and repeatedly move a fraction of mass from the index with the
   largest gradient to the one with the smallest.  Each move is accepted
   only if the true objective improves.  The sequence is kept symmetric
   throughout.

The routine respects a total time budget (default 1000 s); it stops
early if the budget is exhausted and returns the best sequence found.
All numbers are non‑negative and sum to 1, so `evaluate_sequence`
receives a valid step‑function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))  (sum(a) assumed = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - centre) / (centre + 1.0)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float,
                    max_iter: int = 60000,
                    lr0: float = 0.9,
                    alpha_start: float = 0.5,
                    alpha_end: float = 30.0,
                    eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the max‑conv.
    The sequence is kept symmetric after every projection.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # soft‑max weights (stable)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient w.r.t. the smooth objective
        grad = _gradient_from_weights(a, w)

        # exponentiated‑gradient step
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project back onto the simplex
        a = _project_onto_simplex(a)

        # enforce symmetry (mirror around centre)
        a = (a + a[::-1]) / 2.0

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _mass_transfer_greedy(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer walk driven by the sub‑gradient of the exact max‑conv.
    Each step moves a fraction of mass from the index with the largest
    gradient to the one with the smallest, keeps the move only if the true
    objective improves, and re‑symmetrises the vector.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # active set (indices attaining the maximum, up to a tiny tolerance)
        tol = max(1e-14, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]
        if active.size == 0:
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # sub‑gradient of max‑conv (average over active indices)
        grad = _gradient_from_weights(a, w)

        # choose source (largest grad) and destination (smallest grad)
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))

        # avoid moving negligible mass
        if a[src] < 1e-12:
            candidates = np.where(a > 1e-12)[0]
            if candidates.size == 0:
                break
            src = int(rng.choice(candidates))

        delta = a[src] * 0.20               # move up to 20 % of the source mass
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # re‑symmetrise and renormalise
        new_a = (new_a + new_a[::-1]) / 2.0
        new_a = new_a / new_a.sum()

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] mass‑move src={src}→dst={dst} Δ={delta:.3e} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Generates analytic seeds for several lengths,
    runs a longer exponentiated‑gradient descent with symmetry on the
    most promising seeds, and finishes with a gradient‑driven greedy polish.
    Returns the best sequence (as a plain list of floats) found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    lengths = [2000, 4000, 8000, 16000, 32000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates = []   # (score, array)

    # ------------------------------------------------------------------
    # 1. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length = {n} ---")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep a modest elite set across all lengths
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 2. Incorporate a previous solution if it exists

                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep only the very best seeds (at most 7) for the intensive phase
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:7]

    # ------------------------------------------------------------------
    # 4. Intensive smooth gradient descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        per_seed = max(1.0, remaining / len(top_candidates))
        for score, seed_seq in top_candidates:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            n = seed_seq.shape[0]
            # iteration budget scales inversely with n (roughly constant work)
            base_iter = 80000
            max_iter = max(5000, int(base_iter * 2000 / n))
            seq_out, score_out = _smooth_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_iter=max_iter,
                lr0=1.2,
                alpha_start=0.5,
                alpha_end=150.0,
                eval_interval=200,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smooth descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final gradient‑driven greedy mass‑transfer polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _mass_transfer_greedy(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
