def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    for it in range(max_iters):

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # Find the pair (i,j) giving the largest product a_i * a_j for this k
        best_i, best_j = i_start, k - i_start

        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # Choose source (the larger of the two) and a tiny target (global min)
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:

            target = np.argsort(a)[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:

        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:

            conv = _fft_self_convolve(a)

    return a
