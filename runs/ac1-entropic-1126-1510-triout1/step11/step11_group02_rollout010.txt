"""
Hybrid optimiser for the max‑convolution step‑function problem.

The optimiser builds a large set of analytically‑motivated seed vectors
(uniform, arcsine, symmetric/asymmetric Beta families, mixtures, Dirichlet,
random sparse patterns).  Each seed is normalised to a probability vector
(∑a_i = 1).  The best few seeds are refined with two stages of exponentiated‑gradient
descent implemented as Adam on the softmax (log‑sum‑exp) surrogate
(“smooth‑max” phase) and thereafter on the true max‑convolution
(“hard‑max” phase).  Both phases operate directly on unconstrained
variables u with a = softmax(u), so positivity and the simplex constraint are
always satisfied.  The temperature α of the smooth‑max surrogate is raised
geometrically, gradually turning the surrogate into the genuine max.
Finally a cheap greedy mass‑transfer walk (pairwise swaps) is applied.

The routine respects a 1000 s wall‑clock budget and returns the best
coefficient list found.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes sum(a) > 0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic distribution families ----------------------------------------

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid { (i+0.5)/n }.  p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    """Asymmetric Beta(alpha,beta)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_mixture(n: int, p1: float, p2: float, mix: float) -> np.ndarray:
    """Mixture w·Beta(p1,p1)+(1‑w)·Beta(p2,p2).  mix∈[0,1]."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    x = np.clip(x, eps, 1 - eps)
    b1 = np.exp((p1 - 1) * np.log(x) + (p1 - 1) * np.log(1 - x))
    b2 = np.exp((p2 - 1) * np.log(x) + (p2 - 1) * np.log(1 - x))
    vals = mix * b1 + (1.0 - mix) * b2
    if vals.sum() == 0.0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _generate_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """
    Produce a broad collection of seed vectors for a given length.
    All vectors are normalised to sum to 1.
    """
    cand = []

    # simple analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # random symmetric Beta draws
    for _ in range(4):
        p = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric Beta draws
    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # mixture of two symmetric Betas
    for _ in range(6):
        p1 = 10 ** rng.uniform(-2, 1)
        p2 = 10 ** rng.uniform(-2, 1)
        w = rng.random()
        cand.append(_distribution_beta_mixture(n, p1, p2, w))

    # Dirichlet samples (low concentration encourages sparsity)
    for _ in range(5):
        alpha = rng.random() * 0.4  # in (0,0.4)
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    # random sparse patterns
    for _ in range(3):
        cand.append(_distribution_random_sparse(n, rng, density=0.02))

    # completely random probability vectors
    for _ in range(4):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    # clean up (guarantee non‑negativity and normalisation)
    out = []
    for v in cand:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            continue
        v /= s
        out.append(v)
    return out

def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  best_seq: np.ndarray,
                  best_score: float,
                  obj_type: str = "smooth",
                  lr0: float = 0.5,
                  max_iter: int = 200000,
                  eval_interval: int = 250,
                  alpha0: float = 1.0,
                  alpha_max: float = 80.0) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the softmax‑parameterised simplex.
    obj_type = "smooth"  → surrogate f_α (log‑sum‑exp) with temperature α.
    obj_type = "hard"    → true max (active‑set) surrogate.
    Returns the best vector encountered.
    """
    n = a0.shape[0]

    # initialise u such that softmax(u) = a0
    eps = 1e-16
    u = np.log(a0 + eps)

    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_local = a0.copy()
    best_local_score = best_score

    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        # ----- construct a from u (softmax) -----
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / exp_u.sum()

        # ----- convolution of a -----
        conv = _fft_self_convolve(a)

        # ----- build weight vector w -----
        if obj_type == "smooth":
            shift = conv.max()
            w = np.exp(alpha * (conv - shift))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = w / w_sum
        else:  # hard max (active set)
            max_val = conv.max()
            mask = conv >= max_val - 1e-14
            cnt = mask.sum()
            if cnt == 0:
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = mask.astype(np.float64) / cnt

        # ----- gradient wrt a (exact) -----
        grad_a = _gradient_from_weights(a, w)  # shape (n,)

        # ----- chain rule to u (softmax Jacobian) -----
        # ∂L/∂u = a * (grad_a - <a,grad_a>)
        dot = np.dot(grad_a, a)
        grad_u = a * (grad_a - dot)

        # ----- Adam update -----
        lr = lr0 / math.sqrt(t)          # modest decay
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- temperature schedule (smooth phase only) -----
        if obj_type == "smooth" and (t % 300 == 0) and (alpha < alpha_max):
            alpha = min(alpha * 1.25, alpha_max)

        # ----- occasional exact evaluation -----
        if t % eval_interval == 0:
            cur_score = _objective(a)
            if cur_score < best_local_score - 1e-13:
                best_local_score = cur_score
                best_local = a.copy()
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} type={obj_type} α={alpha:.2f} "
                      f"new best {best_local_score:.6f}")

    return best_local, best_local_score

def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk, accepting only improving moves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score
