"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.

Algorithm
---------
1. **Broad exploration** – Scan a set of candidate families (uniform,
   symmetric Beta, mixtures, cosine/tent shapes, random Dirichlet) on a
   hierarchy of lengths (powers of two from 256 up to 65536).  Keep a short
   elite list of the best few vectors (by the surrogate objective
   `2·n·max(a*a)`).

2. **Upsampling** – The elite vectors from a smaller length are up‑sampled
   (by integer repetition) to the next larger length.  This transfers a
   good “shape” to finer resolutions.

3. **Local optimisation** – For each elite vector a three‑stage optimiser
   runs until the time budget expires:
   * *Smooth‑max mirror descent* with a slowly increasing temperature
     (α from 1 to 1e6) and many steps.
   * *Projected sub‑gradient descent* (Adam‑style momentum, back‑tracking
     line‑search) for fine polishing.
   * *Hill‑climbing + pruning* that moves mass away from the currently
     maximal convolution entry and occasionally removes tiny components.

4. **Result** – The best vector found (renormalised to sum = 1) is returned.
   Scaling does not affect `evaluate_sequence`, so any positive scaling
   would be equivalent; we keep the simplex normalisation for convenience.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return vectors with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) discretised on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Use log‑space to stay stable
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_mixture_beta_uniform(n, p, w):
    """Mixture w·Beta(p,p) + (1‑w)·Uniform."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix

def _distribution_cosine(n):
    """a_i ∝ sin(π·i/(n‑1)), i=0…n‑1 (zero at the ends)."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * i / (n - 1))
    w[0] = w[-1] = 0.0
    w += 1e-16   # avoid exact zeros
    w /= w.sum()
    return w

def _distribution_tent(n):
    """Triangular (tent) shape, peak at centre."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    w = 1.0 - np.abs(i - half) / half
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

def _distribution_random_dirichlet(n, alpha):
    """Dirichlet draw with symmetric concentration."""
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec

def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

# Candidate generator for a given length
def _generate_candidates(n):
    """Return a list of normalised candidate vectors for length n."""
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))          # arcsine (U‑shaped)
    cand.append(_distribution_beta_sym(n, 2.0))          # bell‑shaped
    cand.append(_distribution_cosine(n))
    cand.append(_distribution_tent(n))

    # dense grid of symmetric betas
    p_vals = np.concatenate(
        (np.logspace(-2, -0.2, 8), np.linspace(0.1, 2.5, 12))
    )
    for p in p_vals:
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with the uniform distribution
    for p in (0.2, 0.5, 1.0, 1.5):
        for w in (0.3, 0.6, 0.9):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # random Dirichlet draws (several alphas)
    for alpha in (0.05, 0.2, 0.8, 2.0):
        for _ in range(3):
            cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse starters
    k = max(2, n // 300)
    cand.append(_distribution_sparse_random(n, k))

    # shuffled variants of a few bases
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Upsample a vector to a larger length (integer factor)
def _upsample(v, new_n):
    n = v.shape[0]
    if new_n == n:
        return v.copy()
    factor = new_n // n
    if new_n % n != 0:
        # fallback to linear interpolation for non‑integer factor
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, new_n, endpoint=False)
        v_new = np.interp(x_new, x_old, v)
        v_new = np.maximum(v_new, 0.0)
        v_new /= v_new.sum()
        return v_new
    # repeat each entry factor times
    v_new = np.repeat(v, factor)
    v_new = np.maximum(v_new, 0.0)
    v_new /= v_new.sum()
    return v_new

# Smooth‑max mirror descent (exponentiated‑gradient)
def _smooth_max_mirror_descent(a0, deadline):
    """Run mirror descent on a smoothed max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # schedule: (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e3, 4000, 0.12, 0.012),
        (1e3, 1e5, 4000, 0.012, 0.0012),
        (1e5, 1e6, 6000, 0.0012, 0.00012),
    ]

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # soft‑max weights (stable)
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 800 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
    return best_a, best_sc

# Projected sub‑gradient descent with Adam‑style momentum
def _projected_subgradient(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (within tiny tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-14 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-14:
            best_sc = cand_sc
            best_a = a.copy()
        # occasional jitter
        if it % 1500 == 0:
            jitter = np.random.randn(n) * 8e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
        if time.time() + 0.5 > deadline:
            break
    return best_a, best_sc

# Hill‑climbing + pruning
def _hill_climb_prune(a0, deadline):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1
        # move mass away from the pair producing the current max
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j
        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.015) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-14:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
        # aggressive pruning every few thousand steps
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.94, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-14:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
        # occasional jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 3e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence found (plain Python list of floats).
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []   # list of (score, vector) pairs, size ≤ 6

    # ------------------------------------------------------------------
    # 1. Broad exploration on a hierarchy of lengths
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)

            if sc < best_score - 1e-14:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] New global best (len={n}) score={sc:.6f}")

            # maintain elite list of size ≤6
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate any historic solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score - 1e-14:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. If nothing found (unlikely), fallback to uniform
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upsample elite vectors progressively and locally optimise
    #    We iterate over the elite list (max 4 restarts) and for each
    #    continue to the next larger length if time permits.
    max_restarts = min(4, len(elite))

    for restart_idx in range(max_restarts):
        if time.time() >= deadline:
            break
        a0, init_score = elite[restart_idx][1].copy(), elite[restart_idx][0]
        cur_n = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {restart_idx+1}/{max_restarts} "
              f"starting length={cur_n}, score={init_score:.6f}")

        # iterate through the remaining larger lengths
        for target_n in lengths:
            if target_n <= cur_n:
                continue
            if time.time() >= deadline:
                break
            # upsample
            a0 = _upsample(a0, target_n)
            cur_n = target_n
            print(f"[{time.time():.1f}]   Upsampled to {cur_n}")

            # smooth‑max mirror descent
            a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)
            # projected sub‑gradient polishing
            a_mid, sc_mid = _projected_subgradient(a_mid, deadline)
            # hill‑climbing + pruning
            a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

            if sc_mid < best_score - 1e-14:
                best_score = sc_mid
                best_seq = a_mid.copy()
                print(f"[{time.time():.1f}]   New global best after upsample {sc_mid:.6f}")

            a0 = a_mid   # continue from the refined vector

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity, normalise, report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_sc = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_sc:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
