"""
Propose a step–function that minimizes `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – try a few hand‑crafted families (arcsine,
   edge‑power with a very small exponent, random Dirichlet).  Pick the
   best of them as a start.
2. **High‑temperature mirror descent** – minimise a smooth approximation
   of the max‑convolution:
        w_i ∝ exp(α·(c_i−max(c)))   (c = a∗a)
   with the temperature α growing from 1 to 1e7 and the learning‑rate
   decaying from 0.5 to 0.001.  An exponentiated‑gradient update
   (multiplicative) keeps the vector non‑negative; we renormalise to the
   simplex after each step.
3. **Projected Adam‑style sub‑gradient descent** on the true
   max‑convolution.  The active set (indices attaining the current max)
   yields a sub‑gradient; Adam momentum and a back‑tracking line‑search
   accelerate convergence.
4. **Targeted stochastic hill‑climbing** – repeatedly move a tiny
   amount of mass from the coordinate that contributes most to the
   current peak to the smallest coordinate; accept only improvements.
   Small random jitter is added periodically to escape plateaus.
5. Keep the best sequence seen throughout the run and return it as a
   plain Python list.

All heavy work is done with real FFTs (O(n log n)).  The routine never
writes to disk, uses only NumPy and the standard library and respects
the supplied time budget.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex handling (renormalisation) ------------------------------------
def _renorm(a: np.ndarray) -> np.ndarray:
    """Force non‑negativity (tiny epsilon) and renormalise to sum = 1."""
    eps = 1e-15
    a = np.maximum(a, eps)
    s = a.sum()
    if s == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# FFT based convolutions -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Scoring ---------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Return the exact value of `evaluate_sequence` for a (sum = 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Hand‑crafted families (all return a vector whose sum = 1) -------------
def _dist_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dist_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    v = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    v /= v.sum()
    return v


def _dist_edge_power(n: int, p: float) -> np.ndarray:
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)(n‑i‑0.5))^{‑p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15
    w /= w.sum()
    return w


def _dist_dirichlet(n: int) -> np.ndarray:
    return np.random.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Stage 1 – Mirror‑descent on a soft‑max of the convolution ------------
def _mirror_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """High‑temperature exponentiated‑gradient descent."""
    n = a.shape[0]
    # schedule parameters
    alpha_start, alpha_end = 1.0, 1e7
    lr_start, lr_end = 0.5, 0.001

    # estimate how many iterations we can afford (roughly 0.002 s each)
    remaining = max(0.0, deadline - time.time())
    est_iter_time = 0.002
    max_iters = int(remaining / est_iter_time)
    total_iters = max(5000, min(150000, max_iters))

    best_a = a.copy()
    best_sc = _score(a)

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # soft‑max weights (log‑stable)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum

        # gradient of the smooth‑max (mirror‑descent direction)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient step (keeps positivity)
        a = a * np.exp(-lr * grad)
        a = _renorm(a)

        # occasional bookkeeping
        if it % max(1, total_iters // 10) == 0 or it == total_iters - 1:
            cur_sc = _score(a)
            if cur_sc < best_sc - 1e-12:
                best_sc, best_a = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror it={it}/{total_iters}  score={cur_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stage 2 – Adam‑style sub‑gradient descent on the true max‑conv. -------
def _subgradient_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Projected Adam on the piecewise‑linear max‑convolution."""
    n = a.shape[0]
    m = np.zeros_like(a)        # 1st moment
    beta1, beta2 = 0.9, 0.999
    step0 = 0.5
    it = 0
    best_a = a.copy()
    best_sc = _score(a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # sub‑gradient (sum of gradients of active linear pieces)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            cand = a - step * m_hat
            cand = _renorm(cand)
            cand_sc = _score(cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-12:
            best_sc, best_a = cand_sc, a.copy()
            print(f"[{time.time():.1f}] subgrad it={it}  score={cand_sc:.6f}")

        # occasional jitter to shake off plateaus
        if it % 700 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = _renorm(a + jitter)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Stage 3 – Stochastic hill‑climbing (mass transfer) --------------------
def _hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Very local polishing: move a tiny amount of probability mass."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0
    eps = 1e-15

    while time.time() < deadline:
        it += 1
        # index of the current peak
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair contributing the most to this peak
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod, best_i, best_j = prod, i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = _renorm(a2)

        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a, conv, M = a2, _fft_self_convolve(a2), new_M
            best_a, best_sc = a.copy(), 2.0 * n * M
            if it % 100 == 0:
                print(f"[{time.time():.1f}] hill it={it}  score={best_sc:.6f}")

        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = _renorm(a + jitter)
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={it}")

    return best_a, best_sc


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    best_seq = None
    best_sc = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Pick a few candidate lengths (the sweet‑spot found earlier is
    #     around a few thousand elements – large enough to be flexible,
    #     small enough that the linear‐programming‑free optimisation is
    #     cheap).
    lengths = [2500, 3000, 3500, 4000, 4500]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- 1a – try a handful of simple families -----------------
        cand_fams = [
            _dist_arcsine(n),
            _dist_edge_power(n, 0.05),   # very edge‑heavy
            _dist_edge_power(n, 0.10),
            _dist_uniform(n),
            _dist_dirichlet(n),
        ]

        # also keep the previous best if it exists
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size > 0:
                prev = _renorm(prev)
                if prev.size == n:
                    cand_fams.append(prev)
        except Exception:
            pass

        for vec in cand_fams:
            vec = _renorm(vec)
            sc = _score(vec)
            if sc < best_sc:
                best_sc, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} init score={sc:.6f}")

        # ----- 1b – mirror descent on the chosen start -----------------
        a = best_seq if best_n == n else _dist_arcsine(n)
        a = _renorm(a)

        a, sc = _mirror_descent(a, deadline)
        if sc < best_sc:
            best_sc, best_seq, best_n = sc, a.copy(), n
            print(f"[{time.time():.1f}] n={n} after mirror descent best={sc:.6f}")

        # ----- 1c – Adam style sub‑gradient refinement ------------------
        a, sc = _subgradient_descent(a, deadline)
        if sc < best_sc:
            best_sc, best_seq, best_n = sc, a.copy(), n
            print(f"[{time.time():.1f}] n={n} after sub‑grad best={sc:.6f}")

        # ----- 1d – final stochastic hill‑climb -----------------------
        a, sc = _hill_climb(a, deadline)
        if sc < best_sc:
            best_sc, best_seq, best_n = sc, a.copy(), n
            print(f"[{time.time():.1f}] n={n} final best={sc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    if best_seq is None:
        n = 3000
        best_seq = _dist_uniform(n)
        best_sc = _score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform n={n} score={best_sc:.6f}")

    print(f"Overall best score ≈ {best_sc:.6f} (length={best_n})")
    return best_seq.tolist()
