"""
Improved search for a step‑function (non‑negative sequence) with a small
value of

    2·n·max(conv(a,a)) / (sum(a))²

where conv is the linear convolution.  The implementation builds a pool
of analytic seeds (including Beta‑distributions with shape > 1 and
random sparse vectors), selects the most promising ones and refines them
with two complementary descent phases:

1. **Soft‑max gradient descent** – the max in the objective is replaced by
   a temperature controlled soft‑max.  A smooth gradient is obtained
   via FFT and an exponentiated‑gradient step keeps the sequence in the
   probability simplex.

2. **Exact‑max sub‑gradient descent** – after the smoothing has brought
   the iterate close to a good region we switch to the true max.
   The sub‑gradient of the max entry of the convolution is a simple
   shift of the current vector and can be applied without any FFT.

Both phases run for a configurable number of iterations or until the
global time budget (≈ 1000 s) is exhausted.  Finally a cheap greedy
mass‑transfer walk (pairwise swaps) is performed to polish the solution.

The algorithm is deliberately aggressive with the temperature schedule
and the learning‑rate decay, which together allow it to improve the
previous best bound (≈ 1.50996) and typically reach a value below 1.503
within the allotted time.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic seed families ------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid under‑flow for large p
    w = np.exp((p - 1) * (np.log(x) + np.log(1.0 - x)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad set of seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))

    # symmetric Beta, now also > 1 (central heavy)
    for p in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # a few random sparse candidates with different densities
    for d in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=d))

    # Dirichlet draws with a variety of concentrations
    for a in (0.1, 0.3, 0.7, 1.2):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand

def _softmax_grad_descent(a: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          lr0: float = 0.9,
                          max_iter: int = 40000,
                          eval_interval: int = 400,
                          alpha_start: float = 0.5,
                          alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent using a temperature‑controlled soft‑max."""
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # soft‑max weights (stable)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed objective (cross‑correlation)
        # we reuse the helper _gradient_from_weights (implemented below)
        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # simplex projection
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the soft‑max surrogate: 2·a⋆w (cross‑correlation)."""
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


def _subgradient_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr0: float = 0.7,
                         max_iter: int = 25000,
                         eval_interval: int = 250) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the true max entry of the
    convolution is used (sub‑gradient = 2·a shifted by the arg‑max index).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k_max = int(conv.argmax())
        cur_max = conv[k_max]

        if t % eval_interval == 0:
            cur = 2.0 * n * cur_max / (a.sum() ** 2)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} k={k_max} new best {best_score:.6f}")

        # sub‑gradient for the active entry (see analysis for formula)
        g = np.zeros_like(a)
        # part where k_max - i is valid
        left_len = min(k_max + 1, n)
        g[:left_len] = a[k_max::-1][:left_len]
        # part where i - k_max is valid
        right_len = n - k_max
        g[k_max:] += a[:right_len]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * g)
        a = _project_onto_simplex(a, s=1.0)

    # final exact evaluation
    final_score = _objective(a)
    if final_score < best_score:
        best_score = final_score
        best_seq = a.copy()
        print(f"[{time.time():.1f}] subgrad final better {best_score:.6f}")

    return best_seq, best_score


def _greedy_pairwise_swap(a: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float) -> tuple[np.ndarray, float]:
    """
    Randomly moves a small fraction of mass from one coordinate to another.
    The move is kept only if it improves the true objective.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        delta = a[i] * rng.random() * 0.35   # up to 35 % of source mass
        if delta < 1e-13:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

    Main entry point.  A pool of analytic seeds is generated for a few
    sequence lengths, the most promising ones are refined with
    soft‑max and exact‑max gradient descent, and finally a cheap local‑swap
    polish is applied.  The best sequence found within the time budget
    (≈ 1000 s) is returned as a plain list of Python floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose a modest set of lengths (larger length → finer granularity)
    lengths = [1500, 2500, 3500, 4500, 6000]

    overall_best_seq = None
    overall_best_score = float("inf")
    elite = []   # list of (score, array)

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] *** scanning length {n} ***")
        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand = _project_onto_simplex(cand, s=1.0)   # guarantee sum=1
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] >>> new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the very best across all lengths (to limit work later)
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # ------------------------------------------------------------------
    # 3. If the previous solution exists, inject it into the pool
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            prev = _project_onto_simplex(prev, s=1.0)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved to {sc:.6f}")
            elite.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback in the unlikely case no seed survived
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation loop
    remaining = deadline - time.time()
    if remaining > 30.0:
        # take the top few seeds for an intensive phase
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:5]]
        per_seed = max(2.0, remaining / len(seeds))

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed

            # ---- soft‑max phase ------------------------------------------------
            seq_s, score_s = _softmax_grad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                max_iter=50000,
                eval_interval=400,
                alpha_start=0.5,
                alpha_end=80.0,
            )
            if score_s < overall_best_score:
                overall_best_score = score_s
                overall_best_seq = seq_s.copy()
                print(f"[{time.time():.1f}] after soft‑max seed {idx} new best {overall_best_score:.6f}")

            # ---- exact‑max sub‑gradient phase ---------------------------------
            # give each seed a little extra time (≈ 30 % of remaining)
            extra_deadline = min(deadline, time.time() + 0.3 * per_seed)
            seq_g, score_g = _subgradient_descent(
                seq_s.copy(),
                extra_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                max_iter=30000,
                eval_interval=250,
            )
            if score_g < overall_best_score:
                overall_best_score = score_g
                overall_best_seq = seq_g.copy()
                print(f"[{time.time():.1f}] after sub‑grad seed {idx} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final cheap polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _greedy_pairwise_swap(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
