def propose_candidate(seed:int=42, budget_s:int=1000):
    np.random.seed(seed)
    random.seed(seed)
    start=time.time()
    deadline=start+budget_s-2.0

    best_score = float('inf')
    best_seq = None
    elite=[]

    # Choose length schedule (depends on time left). Use powers of 2 up to 65536.
    lengths = [1024,2048,4096,8192,16384,32768,65536]
    for n in lengths:
        if time.time()>=deadline: break
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time()>=deadline: break
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score=sc; best_seq=vec.copy()
                print(f"[{time.time():.1f}] New global best length={n} score={sc:.6f}")
            if len(elite)<6:
                elite.append((sc,vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc<elite[-1][0]:
                    elite[-1]=(sc,vec.copy())
                    elite.sort(key=lambda x: x[0])
    # fallback and incorporate previous solution same as earlier.

    # If we have elapsed less than something or we want more restarts.
    # Use only top few elite as start for local optimization.
    restarts = min(3, len(elite))
    for i in range(restarts):
        if time.time()>=deadline: break
        a0 = elite[i][1].copy()
        print(f"[{time.time():.1f}] Restart {i+1}/{restarts} starting score={elite[i][0]:.6f}")
        a, sc = _smooth_max_mirror_descent(a0, deadline)

def _distribution_beta_asym(n, p, q):
    x = (np.arange(n, dtype=float) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum()==0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return vectors with Σ = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform (already normalised)."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_hann(n):
    """sin² window (Hann). Zero at the ends, peak at the centre."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * i / (n - 1)) ** 2
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangle(n):
    """Symmetric triangle (linear rise then fall)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    w = 1.0 - np.abs(i - mid) / (mid + 1.0)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_random_dirichlet(n, alpha):
    """Draw from Dirichlet(alpha,…,alpha)."""
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_sparse_random(n, k):
    """k random non‑zero entries (uniform on the simplex)."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

# Candidate generator ----------------------------------------------------
def _generate_candidates(n):
    """Return a list of normalised candidate vectors for length n."""
    cand = []

    # Core families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))      # arcsine
    cand.append(_distribution_hann(n))
    cand.append(_distribution_triangle(n))

    # Grid of symmetric Betas
    for p in np.concatenate(
        (np.logspace(-2, -0.1, 14), np.linspace(0.1, 2.0, 20))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Random asymmetric Betas
    for _ in range(6):
        p = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        q = 10 ** np.random.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, p, q))

    # Mixtures with the uniform distribution
    for p in (0.2, 0.5, 1.0, 2.0):
        for w in (0.25, 0.5, 0.75):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # Several Dirichlet draws (different concentration)
    for alpha in (0.05, 0.2, 0.8, 2.0, 5.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # Sparse start – a handful of spikes
    k = max(2, n // 800)
    cand.append(_distribution_sparse_random(n, k))

    # Random shuffles of a few bases (break symmetry)
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Optimisation primitives -----------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """Exponentiated‑gradient descent on a smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e4, 4000, 0.12, 0.012),
        (1e4, 1e5, 3000, 0.012, 0.0012),
        (1e5, 1e6, 3000, 0.0012, 0.00012),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # soft‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed max
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _true_max_subgradient_descent(a0, deadline):
    """Projected sub‑gradient descent on the true max with Adam‑style momentum."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (within relative tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam updates
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line‑search on the true objective
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-13 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-13:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter to avoid stagnation
        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a0, deadline):
    """Simple hill‑climber with aggressive pruning of tiny entries."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- move mass away from the pair that produces the current max ----
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j
        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-13:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # ---- aggressive pruning every few thousand steps ----
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-13:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ---- occasional random jitter ----
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc


def _annealing_refine(a0, deadline, max_iter=20000):
    """Simulated‑annealing style refinement that can escape shallow minima."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    cur_sc = _max_convolution_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    temp_start = 0.12
    temp_end = 1e-5

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        # temperature schedule (geometric)
        t = it / max_iter
        temp = temp_start * (temp_end / temp_start) ** t

        src = np.random.randint(n)
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        max_delta = min(a[src], 0.08)  # move at most ~8 % of src mass
        if max_delta < eps:
            continue
        delta = max_delta * np.random.rand()
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_sc = _max_convolution_score(a_new)

        if new_sc < cur_sc:
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc - 1e-13:
                best_sc = new_sc
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] anneal it={it} new best {best_sc:.6f}")
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_sc - cur_sc) / temp)
            if random.random() < prob:
                a = a_new
                cur_sc = new_sc

        # periodic re‑evaluation of the current best (light weight)
        if it % 2500 == 0:
            cur_sc = _max_convolution_score(a)

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best vector found, normalised to sum = 1.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # a small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []               # (score, vector) list, kept sorted

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths and analytic families
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)

            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")

            # maintain a tiny elite list (≤6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate any historic solution (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never trigger)
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool (few restarts)
    restarts = min(3, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Restart {idx + 1}/{restarts} start score={elite[idx][0]:.6f}")

        # a) smooth‑max mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # b) true‑max sub‑gradient polishing
        a_mid, sc_mid = _true_max_subgradient_descent(a_mid, deadline)

        # c) hill‑climber + pruning
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        # d) simulated annealing refinement
        a_mid, sc_mid = _annealing_refine(a_mid, deadline)

        if sc_mid < best_score - 1e-13:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity, normalise and report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
