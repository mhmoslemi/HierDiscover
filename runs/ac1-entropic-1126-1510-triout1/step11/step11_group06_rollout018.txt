"""
Propose a non‑negative step function minimizing `evaluate_sequence`.

Algorithm
---------
1. **Broad parametric sweep** – try many lengths (2 k … 80 k) and a rich
   collection of deterministic families:
   * uniform, arcsine, arcsine with offset,
   * symmetric/asymmetric Beta,
   * edge‑biased power‑law (p > 0) and interior‑biased power‑law (p > 0),
   * discrete semi‑circle.
   The best candidate (sum = 1) is retained.

2. **Projected Adam‑style sub‑gradient descent** – minimise the true
   objective `2·n·max(a*a)` on the probability simplex.  The sub‑gradient
   is obtained from the active set of convolution entries attaining the
   maximum.  Adam momentum with a slowly decaying step‑size is used,
   and every iterate is projected back onto the simplex.

3. **Stochastic hill‑climbing with simulated annealing** – repeatedly
   transfer a small amount of mass from a large coefficient to a small
   one.  Moves that improve the maximal convolution are always accepted;
   otherwise they are accepted with a temperature‑dependent probability.
   Periodic jitter avoids stagnation.

All stages respect a 1000 s budget; intermediate progress is printed.
The returned list is a probability vector (any scaling works because the
objective is scale‑invariant).  The code uses only NumPy and runs on a
single core.
"""

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT based convolutions -----------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    Score for a probability vector (sum = 1):
    2·n·max(a*a).  This equals the value returned by `evaluate_sequence`
    for the same shape (the scaling cancels out).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_arcsine_offset(n, off):
    """Arcsine with a configurable offset (0 < off < n)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + off) * (n - i - off))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑domain to avoid underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n, p):

    idx = np.arange(n, dtype=np.float64)

    w = np.maximum(w, 1e-15)
    w /= w.sum()
    return w


def _distribution_interior_power(n, p):
    """Interior‑biased power law a_i ∝ ((i+0.5)*(n-i-0.5))^{+p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** p
    w = np.maximum(w, 1e-15)
    w /= w.sum()
    return w

    idx = np.arange(n, dtype=np.float64)
    w = np.sqrt((idx + 0.5) * (n - idx - 0.5))

    lengths = [2000, 4000, 8000, 16000, 32000]          # moderate sizes

                             np.linspace(0.5, 2.0, 20)])
    off_grid = np.linspace(0.2, 0.8, 7)
    beta_grid = np.concatenate([np.linspace(0.05, 0.5, 10),
                                np.linspace(0.5, 2.0, 15)])

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families ---------------------------------
        # Uniform
        vec = _distribution_uniform(n)
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score, best_seq, best_n = sc, vec.copy(), n
            print(f"[{time.time():.1f}] n={n:5d} uniform                score={sc:.6f}")

        # Arcsine (standard)
        vec = _distribution_arcsine(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score, best_seq, best_n = sc, vec.copy(), n
            print(f"[{time.time():.1f}] n={n:5d} arcsine                score={sc:.6f}")

        # Arcsine with offsets
        for off in off_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_arcsine_offset(n, off)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} arcsine_off({off:.2f})   score={sc:.6f}")

        # Edge‑biased power law (p>0 means stronger edges)
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_power(n, p)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} edge_power(p={p:.2f})   score={sc:.6f}")

        # Interior‑biased power law (p>0 emphasises centre)
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_interior_power(n, p)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} interior_power(p={p:.2f}) score={sc:.6f}")

        # Semicircle
        vec = _distribution_semicircle(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score, best_seq, best_n = sc, vec.copy(), n
            print(f"[{time.time():.1f}] n={n:5d} semicircle              score={sc:.6f}")

        # Symmetric Beta(p,p)
        for p in beta_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} beta_sym(p={p:.2f})      score={sc:.6f}")

        # Asymmetric Beta, left‑heavy (p varies, q fixed at 0.5)
        for p in np.linspace(0.05, 0.5, 8):
            if time.time() >= deadline:
                break
            vec = _distribution_beta(n, p, 0.5)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} beta(p={p:.2f},q=0.5)    score={sc:.6f}")

            # right‑heavy (swap p and q)
            vec = _distribution_beta(n, 0.5, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} beta(p=0.5,q={p:.2f})    score={sc:.6f}")

        # Random Dirichlet draws
        for _ in range(4):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.full(n, 1.0))
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:5d} dirichlet(1)            score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Try longer lengths with a few promising families
    if time.time() < deadline:
        larger_lengths = [50000, 80000]
        for n in larger_lengths:
            if time.time() >= deadline:
                break
            # edge power with a handful of exponents
            for p in [0.3, 0.5, 0.7, 0.9, 1.1]:
                if time.time() >= deadline:
                    break
                vec = _distribution_edge_power(n, p)
                sc = _max_convolution_score(vec)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec.copy(), n
                    print(f"[{time.time():.1f}] n={n:5d} edge_power(p={p:.2f})   score={sc:.6f}")

            # interior power (may work for very large n)
            for p in [0.4, 0.6, 0.8]:
                if time.time() >= deadline:
                    break
                vec = _distribution_interior_power(n, p)
                sc = _max_convolution_score(vec)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec.copy(), n
                    print(f"[{time.time():.1f}] n={n:5d} interior_power(p={p:.2f}) score={sc:.6f}")

            # arcsine offsets (few values)
            for off in [0.3, 0.5, 0.7]:
                if time.time() >= deadline:
                    break
                vec = _distribution_arcsine_offset(n, off)
                vec = np.maximum(vec, eps)
                vec /= vec.sum()
                sc = _max_convolution_score(vec)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec.copy(), n
                    print(f"[{time.time():.1f}] n={n:5d} arcsine_off({off:.2f})   score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Load previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), prev.size
                print(f"[{time.time():.1f}] previous best            score={sc:.6f}")
    except Exception:
        pass

    # Fallback – should never trigger
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq = np.maximum(best_seq, eps)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform          score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected Adam‑style sub‑gradient descent on the exact objective
    a = best_seq.copy()
    n = a.shape[0]
    m = np.zeros_like(a)        # first moment
    v = np.zeros_like(a)        # second moment
    beta1, beta2 = 0.9, 0.999
    step_counter = 0
    lr0 = 0.5

    while time.time() < deadline:
        step_counter += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (indices attaining the maximum within tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # sub‑gradient of the max‑convolution
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** step_counter)
        v_hat = v / (1.0 - beta2 ** step_counter)

        # learning‑rate schedule (slow decay)
        lr = lr0 / np.sqrt(step_counter)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = _project_onto_simplex(a_new, 1.0)
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_score = _max_convolution_score(a_new)
        if new_score < best_score - 1e-12:
            best_score, best_seq = new_score, a_new.copy()
            print(f"[{time.time():.1f}] subgrad it={step_counter} new best {new_score:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    # 5️⃣  Stochastic hill‑climbing with simulated annealing
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0
    start_hill = time.time()
    T0 = M * 0.1                 # initial temperature

    while time.time() < deadline:
        hill_it += 1

        # pick a source index (favor large coefficients)
        if random.random() < 0.7:
            # top‑10 % indices (deterministic choice)
            source = np.argsort(-a)[random.randint(0, max(0, n // 10 - 1))]
        else:
            source = np.random.choice(n, p=a)   # proportional to mass

        # target is the smallest entry (different from source)
        target = np.argmin(a)
        if target == source:
            continue

        # propose a modest transfer
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_M = _fft_self_convolve(a2).max()
        improved = new_M < M - 1e-12

        if improved:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 100 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")
        else:
            # simulated‑annealing acceptance
            T = T0 * (deadline - time.time()) / (deadline - start_hill)
            if T > 0 and random.random() < np.exp(-(new_M - M) / T):
                a = a2
                conv = _fft_self_convolve(a)
                M = new_M

        # occasional jitter
        if hill_it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at iter={hill_it}")

    # ------------------------------------------------------------------
    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
```

This implementation explores a wide variety of parameterised distributions, refines the best candidate with a projected Adam‑style sub‑gradient method, and finishes with a stochastic hill‑climbing/annealing phase. All steps respect the 1000 s budget and the returned list complies with `evaluate_sequence`.
