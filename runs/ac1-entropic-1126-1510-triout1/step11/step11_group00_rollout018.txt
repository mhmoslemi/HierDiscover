   Euclidean projection onto the probability simplex.

3. **Targeted max‑convolution balancing** – After the descent the index
   where the self‑convolution attains its maximum is identified.  Mass is
   moved from the pair that contributes most to this index towards a pair
   that contributes least, accepting only improving moves.  This directly
   flattens the worst peak.

4. **Local polishing** – A few rounds of deterministic pairwise swaps
   and a cheap simulated‑annealing walk are applied.

5. **Upscaling** – If time permits the best distribution is interpolated
   to a larger length and refined once more.

All steps respect the supplied time budget (default 1000 s).  The
returned list is plain Python; ``evaluate_sequence`` will handle the
final normalisation, clipping, etc.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    need = len(u) + len(v) - 1
    size = 1 << ((need - 1).bit_length())
    U = np.fft.rfft(u, n=size)
    V = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(U * V, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) assuming sum(a)=1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    """Uniform density."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    # guard against overflow / NaN
    w = np.nan_to_num(w, nan=0.0, posinf=0.0)
    total = w.sum()
    if total == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_beta_general(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) density on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow when p,q<1
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0)
    total = w.sum()
    if total == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Soft‑max mirror descent (smooth approximation of the max)
def _softmax_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_lr: float = 0.8,
    init_alpha: float = 10.0,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the sub‑gradient is obtained from a
    soft‑max over all convolution indices.  The temperature (``alpha``) is
    slowly increased so that the algorithm gradually focuses on the true
    maximum.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    lr0 = init_lr
    alpha = init_alpha
    t = 0
    no_improve = 0
    while time.time() < deadline and t < max_iters:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # soft‑max weights (stable shift by the maximum)
        exp_vals = np.exp(alpha * (conv - max_c))
        sum_exp = exp_vals.sum()
        if sum_exp == 0.0:
            p = np.full_like(conv, 1.0 / conv.size)
        else:
            p = exp_vals / sum_exp

        # gradient of 2·n·Σ p_i·conv_i w.r.t. a
        a_rev = a[::-1]
        tmp = _fft_convolve(p, a_rev)          # length = 3n‑2
        grad = 4.0 * n * tmp[n - 1 : n - 1 + n]  # slice of length n

        # exponentiated‑gradient update
        a = a * np.exp(-lr0 / math.sqrt(t) * grad)

        # Euclidean projection onto the simplex (sum = 1)
        s = a.sum()
        if s == 0.0:
            a = _distribution_uniform(n)
        else:
            a /= s

        # occasional evaluation
        if t % 120 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score, best_seq = cur, a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] softmax descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # schedule – increase focus on the true max
        alpha = min(alpha * 1.001, 200.0)

        # jitter if stagnated
        if no_improve >= 3000:
            jitter = rng.exponential(scale=0.003, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Targeted max‑convolution balancing
def _max_conv_balance(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly locate the index of the maximal self‑convolution and move a
    small amount of mass from the most contributing pair to the least
    contributing one.  Only moves that strictly improve the objective are
    kept.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())          # index where the max is attained

        # contributions a_i * a_{k-i}
        i = np.arange(n, dtype=np.int32)
        j = k - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[i[mask]] * a[j[mask]]

        i_high = int(np.argmax(contrib))           # most harmful pair
        # gradient w.r.t. a_i at the max index is 2*a_{k-i}
        g = np.zeros_like(a)
        g[mask] = a[j[mask]]
        i_low = int(np.argmin(g))                  # best place to receive mass

        if a[i_high] <= 1e-15:
            it += 1
            continue

        delta = a[i_high] * rng.random() * 0.2   # move up to 20 % of the mass
        if delta < 1e-15:
            it += 1
            continue

        new_a = a.copy()
        new_a[i_high] -= delta
        new_a[i_low] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_seq = a.copy()
            best_score = cur
            print(
                f"[{time.time():.1f}] balance k={k} {i_high}->{i_low} δ={delta:.3e} new best {best_score:.6f}"
            )
        it += 1
    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement)
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a small chunk of mass; keep it if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.01,
    decay: float = 0.9997,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for an optimal step‑function within the supplied time budget.
    Returns a plain Python list suitable for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a rich pool of seeds
    lengths = [2000, 4000, 8000]  # moderate sizes – allow many iterations
    p_grid = np.concatenate(
        [
            np.logspace(-2, -1, 9),   # 0.01 .. 0.1
            np.logspace(-1, 0, 9),    # 0.1 .. 1
            np.logspace(0, 0.5, 7),   # 1 .. 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)  # 0,…,1

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # uniform
        cand = _distribution_uniform(n)
        sc = _objective(cand)
        top_candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] new global best (uniform) len={n} score={sc:.6f}")

        # edge‑weighted family
        for p in p_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_edge_weighted(n, p)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best len={n} p={p:.3g} score={sc:.6f}"
                )

        # asymmetric beta families (randomly sampled)
        for _ in range(6):
            if time.time() >= deadline:
                break
            p = rng.uniform(0.1, 3.0)
            q = rng.uniform(0.1, 3.0)
            cand = _distribution_beta_general(n, p, q)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best asymmetric Beta len={n} (p,q)=({p:.2f},{q:.2f}) score={sc:.6f}"
                )

        # mixtures with uniform (helps to smooth extremes)
        for p in p_grid[::3]:  # thin the grid a bit for speed
            base = _distribution_edge_weighted(n, p)
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(base, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] new global best mix len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                    )

        # random Dirichlet
        for _ in range(4):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 2.0)
            cand = rng.dirichlet(np.full(n, alpha))
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}"
                )

        # sparse random seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}"
                )

        # keep only the strongest seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Guard against empty result
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Soft‑max mirror descent on the most promising seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] soft‑max descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _softmax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                init_lr=0.8,
                init_alpha=12.0,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5. Targeted max‑convolution balancing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] max‑convolution balancing")
        overall_best_seq, overall_best_score = _max_conv_balance(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_iters=200_000,
        )

    # ------------------------------------------------------------------
    # 6. Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] pairwise swap refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 7. Final stochastic local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] simulated‑annealing refinement")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    # 8. Optional up‑scaling (if still time)
    if time.time() < deadline - 10.0:
        cur_n = len(overall_best_seq)
        new_n = cur_n * 2
        print(f"[{time.time():.1f}] up‑scaling to length {new_n}")
        x_old = np.linspace(0.0, 1.0, cur_n)
        x_new = np.linspace(0.0, 1.0, new_n)
        up_seq = np.interp(x_new, x_old, overall_best_seq)
        up_seq = up_seq / up_seq.sum()
        # short refinement at the larger scale
        up_deadline = time.time() + 10.0
        up_seq, up_score = _softmax_descent(
            up_seq,
            up_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_lr=0.6,
            init_alpha=20.0,
        )
        if up_score < overall_best_score:
            overall_best_seq, overall_best_score = up_seq, up_score
            print(f"[{time.time():.1f}] after up‑scale new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
