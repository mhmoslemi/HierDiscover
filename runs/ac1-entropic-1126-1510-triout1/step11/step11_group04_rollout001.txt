"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

Algorithm
---------
1. **Broad analytic sweep** – for several lengths (2 k … 50 k) generate
   many deterministic families (uniform, triangular, arcsine, Beta,
   sine‑power, cosine‑window, Gaussian, power‑law, exponential) together
   with random Dirichlet draws.  Keep a tiny elite of the best few vectors.

2. **Length‑independent fine tuning** – take the elite vectors one‑by‑one
   and apply a long‑run *smooth‑max mirror‑descent* (exponentiated‑gradient
   on the log‑sum‑exp approximation of the max‑convolution).  The temperature
   α is increased from 1 up to 1 e9 while the learning‑rate decays, giving a
   global‑exploration phase followed by aggressive local polishing.

3. **Exact sub‑gradient polishing** – a few Adam‑style projected sub‑gradient
   steps on the true max‑convolution are performed with back‑tracking line‑
   search.

4. **Hill‑climb & pruning** – a light random mass‑transfer hill‑climber that
   moves a fraction of the mass from the pair that currently creates the
   maximal convolution term to a random location, together with occasional
   pruning of negligible entries.

All stages respect a global deadline (`budget_s` seconds, default 1000 s) and
the best vector found is returned as a plain Python list.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Evaluate `evaluate_sequence` for a vector already normalised to sum = 1.
    (The scalar part of the original objective is homogeneous, so the sum
    constraint can be imposed without loss of generality.)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """σ is a fraction of the length (0 < σ < 0.5)."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _generate_candidates_for_length(n):
    """
    Produce a list of candidate probability vectors (sum = 1) for a given
    length n.  The list is deliberately short – each candidate will be
    evaluated immediately, so we keep memory usage low.
    """
    cand = []

    # deterministic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # exponential (few λ values)
    for lam in (0.002, 0.01, 0.05):
        cand.append(_distribution_exponential(n, lam))

    # symmetric Beta – dense sweep
    for p in np.linspace(0.2, 4.0, 39):          # 0.2,0.3,…,4.0
        cand.append(_distribution_beta_sym(n, p))

    # asymmetric Beta – a couple of promising shapes
    for p, q in ((0.2, 1.0), (1.0, 0.2), (0.5, 1.5), (1.5, 0.5)):
        cand.append(_distribution_beta_asym(n, p, q))

    # sine‑power
    for pw in np.linspace(0.5, 12.0, 24):        # 0.5,…,12
        cand.append(_distribution_sine_power(n, pw))

    # cosine window
    for coeff in np.linspace(0.2, 0.9, 15):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian sigma (as fraction of length)
    for sigma in np.linspace(0.08, 0.45, 15):
        cand.append(_distribution_gaussian(n, sigma))

    # power‑law (both forward and reversed)
    for p in (0.8, 1.2, 1.5, 2.0, 2.5, 3.0, 3.5):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # random Dirichlet draws (different concentration)
    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0, 5.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Sparse start – a few random spikes
    k = max(2, n // 400)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Shuffle a few base families to introduce asymmetry
    for base in (cand[0], cand[1], cand[2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

def _smooth_max_mirror_descent(a0, deadline):
    """
    Exponentiated‑gradient (mirror descent) on the log‑sum‑exp approximation
    of the max‑convolution.  The temperature α is increased exponentially,
    the learning rate decays accordingly.  The routine returns the best
    vector seen and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _objective(a)

    # schedule: (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e3, 8000, 0.5, 0.04),
        (1e3, 1e6, 12000, 0.04, 0.004),
        (1e6, 1e9, 20000, 0.004, 0.0004),
    ]

    for (a0_temp, a1_temp, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break

            t = it / steps
            alpha = a0_temp * (a1_temp / a0_temp) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)          # shape (2n‑1,)
            maxc = conv.max()
            # smoothed max via log‑sum‑exp (stable)
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # grad of smoothed max = 2 * (a reversed * w) restricted to n positions
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient update
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            # occasional bookkeeping
            if it % 4000 == 0:
                sc = _objective(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline):
    """
    Adam‑style projected sub‑gradient descent on the *exact* max‑convolution.
    Uses a simple back‑tracking line‑search to guarantee improvement.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_step = 0.4
    it = 0

    best_a = a.copy()
    best_sc = _objective(a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (tolerance relative to max)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum
        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_step
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-14 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-14:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        # light jitter every few thousand steps
        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_and_prune(a0, deadline):
    """
    Simple hill‑climber that repeatedly moves a tiny amount of mass away
    from the pair (i,j) that produces the current maximal convolution term.
    Also periodically prunes entries with negligible mass.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # --- identify the active pair (i*,j*) generating the max term ---
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # move a small delta from the larger of the two to a random index
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            max_delta = min(a[source] * 0.5, 0.03)
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-13:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    new_sc = _objective(a)
                    if new_sc < best_sc - 1e-14:
                        best_sc = new_sc
                        best_a = a.copy()
                        if it % 200 == 0:
                            print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # pruning every few thousand iterations
        if it % 2500 == 0:
            for keep_frac in (0.98, 0.95, 0.92):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                pruned = np.zeros_like(a)
                pruned[idx] = a[idx]
                pruned /= pruned.sum()
                sc = _objective(pruned)
                if sc < best_sc - 1e-14:
                    a = pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # occasional jitter to keep the search alive
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list
    of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # keep a small safety margin
    deadline = start + budget_s - 2.0

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None
    elite = []          # list of (score, vector) with size ≤ 8

    # --------------------------------------------------------------
    # 1. Broad exploration over several lengths
    lengths = [2000, 3500, 5000, 8000, 12000, 20000, 35000, 50000]
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score - 1e-14:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain elite of size ≤ 8
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, vec.copy())
                elite.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2. If a previous best vector exists (height_sequence_1), inject it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score - 1e-14:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, prev.copy())
                elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Fallback in the unlikely case that we found nothing
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Local optimisation – run mirror descent / sub‑gradient / hill‑climb
    #    on the elite vectors (up to 5 restarts).  Each phase consumes the
    #    remaining time budget.
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.6f}")

        # ----- smooth‑max mirror descent -----
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # ----- projected sub‑gradient polishing -----
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # ----- final hill‑climber & pruning -----
        a_mid, sc_mid = _hill_climb_and_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-14:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # --------------------------------------------------------------
    # 5. Final sanitisation (ensure strict positivity, normalisation)
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
