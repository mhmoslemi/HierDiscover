def _correlation(v, a):
    n = a.shape[0]
    size = 1 << ((len(v) + n - 1).bit_length())
    V = np.fft.rfft(v, n=size)
    A = np.fft.rfft(a, n=size)
    r = np.fft.irfft(V * np.conj(A), n=size)
    # r is size length, but we need first n entries.
    return r[:n]

    # Adam state:

    v = np.zeros_like(a)

    t=0

    for alpha in alphas:

        # maybe 2000 per alpha
        max_iters = 2000 # may adjust based on time
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            t+=1
            # compute conv
            conv = _fft_self_convolve(a)
            # compute weight (softmax)
            maxc = conv.max()
            shifted = alpha * (conv - maxc)
            # might underflow: set min to -700
            shifted = np.clip(shifted, -700, None)

"""
Hybrid smooth‑max / Adam optimisation for the self‑convolution problem.

The goal is to minimise
    2·n·max_k (a*a)[k]      with a_i ≥ 0, Σ a_i = 1

The algorithm proceeds as follows

1.  Randomly generate a large pool of starting vectors of various
    lengths (5000 … 30000).  The pool contains uniform, arcsine,
    Beta‑, edge‑weighted, Dirichlet‑ and sparse seeds, plus the
    previously discovered construction if it exists.

2.  For each seed a smooth‑max approximation of the max‑convolution
    is minimised with an Adam‑type projected gradient descent.
    The smooth‑max uses a temperature α; we start with a small α
    (≈ 3) and increase it geometrically (≈ ×2) – a continuation
    scheme that refines the approximation while staying in a
    well‑behaved landscape.

    •  Convex‑conjugate gradient:
        grad sm(a) = 2·corr(softmax_α(conv(a)), a)
        where corr is the (non‑circular) correlation.
    •  Adam update on the gradient, followed by Euclidean projection
        onto the probability simplex (Σ a_i = 1, a_i ≥ 0).

    The inner loop stops when the global time budget (≈ 1000 s) is
    exhausted or when a per‑seed budget runs out.  Every 200
    iterations the exact objective (2·n·max(conv)) is evaluated and
    the best vector seen so far is stored.

3.  After the continuous optimisation a short greedy “pairwise mass
    transfer” phase attempts local improvements: a small amount of
    mass is moved from one index to another and the move is kept only
    if it strictly reduces the exact objective.

4.  The best sequence among all seeds is returned as a plain Python
    list; `evaluate_sequence` will handle the final scaling and safety
    checks.

The whole search respects the 1000 s budget (a small safety margin)
and makes heavy use of FFT‑based convolutions, which are O(N log N)
and easily handle vectors of length 30 000 on a single machine.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _correlation(v: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Non‑circular correlation of v (length L) with a (length n).
    Returns the first n entries, i.e.   r[p] = Σ_i v[i+p] a[i] .
    """
    n = a.shape[0]
    L = v.shape[0]
    size = 1 << ((L + n - 1).bit_length())
    V = np.fft.rfft(v, n=size)
    A = np.fft.rfft(a, n=size)
    corr = np.fft.irfft(V * np.conj(A), n=size)
    return corr[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input is assumed non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smoothmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_i = exp(alpha*conv_i) / Σ exp(alpha*conv)
    using a numerically stable shift.
    """
    maxc = conv.max()
    # shift to avoid overflow, clip to a safe exponent range
    shifted = alpha * (conv - maxc)
    shifted = np.clip(shifted, -700, None)   # np.exp(-700) ≈ 5e-305
    exps = np.exp(shifted)
    w = exps / exps.sum()
    return w

def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence (and its exact score) seen during the run.
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.25

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # continuation over the temperature α
    alphas = [3.0, 6.0, 12.0, 24.0, 48.0, 96.0]

    for alpha in alphas:
        # a few hundred iterations per α; stop early if out of time
        max_iters = 1500
        for it in range(max_iters):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            # 1) self‑convolution
            conv = _fft_self_convolve(a)

            # 2) soft‑max weights for the smooth max
            w = _smoothmax_weights(conv, alpha)

            # 3) gradient of the smooth max (2·correlation)
            grad = 2.0 * _correlation(w, a)

            # 4) Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr / (np.sqrt(v_hat) + eps)
            a = a - step * m_hat

            # 5) enforce simplex constraints
            a = _project_onto_simplex(a, s=1.0)

            # 6) occasional exact evaluation
            if it % eval_interval == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] New global best {cur:.6f} "
                        f"n={n} α={alpha:.1f} it={it}"
                    )
        # after finishing one α we keep the current a as start for the next
    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Random small mass transfers; keep only improving moves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # pick a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35  # up to 35 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

def _seed_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _seed_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _seed_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _seed_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _seed_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a


def _seed_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _seed_uniform(prev.shape[0])
            prev = prev / prev.sum()
            _register(prev, "previous")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2.  Main stochastic search loop.
    # ------------------------------------------------------------------
    lengths = [5000, 8000, 12000, 20000, 30000]
    # Expected number of seeds we can afford (very rough)
    MIN_TIME_PER_SEED = 15.0

    while time.time() < deadline - MIN_TIME_PER_SEED:
        # pick a length (biased towards larger n)
        n = int(rng.choice(lengths, p=np.array([1, 2, 3, 3, 1]) / 10))

        # Choose a seed generation strategy at random
        strat = rng.integers(0, 6)
        if strat == 0:
            a0 = _seed_uniform(n)
            src = "uniform"
        elif strat == 1:
            a0 = _seed_arcsine(n)
            src = "arcsine"
        elif strat == 2:
            p = rng.uniform(0.001, 1.5)
            a0 = _seed_beta_sym(n, p)
            src = f"beta(p={p:.3f})"
        elif strat == 3:
            p = rng.uniform(0.001, 1.5)
            a0 = _seed_edge_weighted(n, p)
            src = f"edge(p={p:.3f})"
        elif strat == 4:
            a0 = _seed_dirichlet(n, rng)
            src = "dirichlet"
        else:
            dens = rng.uniform(0.003, 0.03)
            a0 = _seed_sparse(n, rng, density=dens)
            src = f"sparse(d={dens:.3f})"

        # Occasionally mix the seed with the uniform vector – this
        # creates a richer set of starting points.
        if rng.random() < 0.25:
            lam = rng.random()
            a0 = _mix_with_uniform(a0, lam)
            src += f"+mix(lam={lam:.2f})"

        a0 = _project_onto_simplex(a0, s=1.0)

        # register the raw seed (useful for debugging)
        _register(a0, f"{src} seed (n={n})")

        # allocate a slice of the remaining time to this seed
        remaining = deadline - time.time()
        per_seed = min(remaining * 0.4, 70.0)  # at most 70 s per seed, 40 % of what’s left
        sub_deadline = time.time() + per_seed

        # ------------------------------------------------------------------
        # 2a.  Adam + smooth‑max refinement
        # ------------------------------------------------------------------
        a_opt, sc_opt = _adam_optimize(
            a0,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            eval_interval=200,
        )
        # (the optimizer already updates the globals; we just keep the final a_opt)
        # ------------------------------------------------------------------
        # 2b.  Final greedy polishing on the refined vector (short budget)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            polish_deadline = time.time() + min(5.0, deadline - time.time())
            a_opt, sc_opt = _pairwise_swap_local(
                a_opt,
                polish_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_trials=50000,
            )
        # ------------------------------------------------------------------
        # Register the result of this seed
        # ------------------------------------------------------------------
        _register(a_opt, f"{src} after optimisation (n={n})")

    # ------------------------------------------------------------------
    # 3.  Final fall‑back: if for any reason we have no candidate yet,
    #     use a uniform distribution of a moderate size.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n_fallback = 8000
        overall_best_seq = _seed_uniform(n_fallback)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform n={n_fallback} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"*** FINAL BEST: score={overall_best_score:.6f}, length={len(overall_best_seq)} ***"
    )
    # Return a plain Python list – evaluate_sequence will perform its own
    # scaling and safety checks.
    return overall_best_seq.tolist()
