- Checks and normalizes sequence: convert to floats, clip negatives to 0, clip values >1000 to 1000.
- n = len(sequence)
- b_sequence = np.convolve(sequence, sequence) returns length 2n-1 full convolution.
- max_b = max(b_sequence)
- sum_a = np.sum(sequence)
- If sum_a < 0.01 then returns inf.
- Returns 2 * n * max_b / (sum_a ** 2).

def _adam_descent(a: np.ndarray, deadline: float, best_seq: np.ndarray, best_score: float,
                  lr: float = 0.1, tau_start: float = 5.0, tau_max: float = 40.0,
                  eval_interval: int = 200, patience: int = 5000) -> tuple[np.ndarray, float]:

    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 200_000:
        t += 1
        # compute conv and softmax
        conv = _fft_self_convolve(a)
        # get sup objective for record
        cur_obj = 2.0 * n * conv.max()
        # update best
        if cur_obj < best_score - 1e-14:
            best_score = cur_obj
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] adam iter={t} new best {best_score:.6f}")
        else:
            no_improve += 1

        # compute softmax weights
        tau = min(tau_max, tau_start + t * (tau_max - tau_start) / 200_000)  # linearly increase to tau_max
        # stable exponentiate
        shifted = conv - conv.max()
        # clip? nothing
        exp_vals = np.exp(tau * shifted)
        w = exp_vals / exp_vals.sum()

        # gradient
        rev_w = w[::-1]
        conv_rev = _fft_convolve_full(rev_w, a)  # full conv
        grad = 2.0 * conv_rev[n-1:2*n-1]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a = a + step
        # projection onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # jitter if stale
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.01, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

        # early break if time exceed deadline: loop condition handles.

    return best_seq, best_score

"""
Multi‑scale soft‑max Adam descent for the step‑function self‑convolution
problem.

The algorithm works as follows:

1. Build a modest pool of analytic candidates (uniform, arcsine,
   symmetric beta, edge‑weighted, sine‑weighted) for a medium length
   (≈4000).  The best of these seeds (according to the exact objective)
   becomes the starting distribution.

2. Run a smooth gradient descent on the simplex:
   * use a soft‑max (temperature annealed with the progress) to obtain a
     differentiable approximation of the max‑convolution,
   * compute the gradient by a second FFT‑based convolution,
   * update the vector with an Adam‑style rule,
   * periodically project back onto the probability simplex.

3. After a fixed amount of time at the current resolution, up‑sample the
   distribution to a larger number of steps (≈1.8× larger) by linear
   interpolation and continue the descent.  This progressive refinement
   lets the method approach the continuous optimum while keeping each
   iteration inexpensive.

4. When the time budget is almost exhausted, apply two cheap local
   refinements that were already used by the previous state‑of‑the‑art
   code: a deterministic pairwise‑swap walk and a tiny simulated‑annealing
   mass‑transfer walk.

The whole search respects the supplied time budget (default 1000 s) and
returns a plain Python list; ``evaluate_sequence`` will perform the final
normalisation, clipping, etc.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via FFT."""
    need = len(x) + len(y) - 1
    size = 1 << ((need - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _upsample_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, preserving total mass."""
    old_n = a.shape[0]
    old_x = (np.arange(old_n, dtype=np.float64) + 0.5) / old_n
    new_x = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    new_a = np.interp(new_x, old_x, a, left=0.0, right=0.0)
    new_a[new_a < 0] = 0.0
    s = new_a.sum()
    if s == 0.0:
        new_a = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        new_a = new_a / s
    return new_a

def _adam_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.12,
    alpha_start: float = 0.20,
    alpha_end: float = 0.02,
    eval_interval: int = 200,
    patience: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a soft‑max approximation of the max‑convolution.
    The temperature (via ``alpha``) is annealed from ``alpha_start`` to
    ``alpha_end`` while the vector stays on the probability simplex.
    """
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0
    no_improve = 0
    rng = np.random.default_rng()

    max_iters = 250_000
    while time.time() < deadline and t < max_iters:
        t += 1

        conv = _fft_self_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_score - 1e-14:
            best_score = cur_obj
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] adam it={t} new global best {best_score:.6f}")
        else:
            no_improve += 1

        # ----- soft‑max weights approximating the max -----
        maxc = conv.max()
        if maxc <= 0:
            w = np.full(conv.shape, 1.0 / conv.shape[0])
        else:
            # linear annealing of the temperature parameter ``alpha``
            progress = t / max_iters
            alpha = alpha_start * (1.0 - progress) + alpha_end * progress
            inv_temp = 1.0 / (alpha * maxc)          # large number
            scaled = conv * inv_temp
            shifted = scaled - scaled.max()          # for numerical stability
            exp_vals = np.exp(shifted)
            w = exp_vals / exp_vals.sum()

        # ----- gradient of the smoothed objective (see analysis) -----
        rev_w = w[::-1]                               # length 2n‑1
        conv_rev = _fft_convolve_full(rev_w, a)      # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : 2 * n - 1]      # slice of length n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a + step
        a = _project_onto_simplex(a, s=1.0)

        # ----- occasional jitter to escape plateaux -----
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

        # periodic verbose (every eval_interval steps)
        if t % eval_interval == 0:
            # just to keep the loop honest – the objective has already been
            # updated above, so we only print a short trace.
            pass

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Simple deterministic local refinement – move a small mass between
    two positions if it improves the exact objective."""
    n = a.shape[0]
    rng = np.random.default_rng()
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the supplied time budget (default 1000 s) and returns
    a plain Python list.  It combines analytic seeding, a multi‑scale Adam
    descent, and cheap final polishing.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a tiny safety margin

    # --------------------------------------------------------------
    # 1.  Build a modest pool of analytic and random seeds (length ≈4000)
    n0 = 4000
    seed_pool: list[np.ndarray] = []

    # deterministic families
    seed_pool.append(_distribution_uniform(n0))
    seed_pool.append(_distribution_arcsine(n0))

    # parametrised families + uniform mixing
    for p in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for base in (
            _distribution_beta_sym(n0, p),
            _distribution_edge_weighted(n0, p),
            _distribution_sine_weighted(n0, p),
        ):
            for lam in (0.0, 0.25, 0.5, 0.75, 1.0):
                seed_pool.append(_mix_with_uniform(base, lam))

    # random Dirichlet seeds
    for _ in range(6):
        alpha = rng.uniform(0.02, 2.0)
        a = rng.dirichlet(np.full(n0, alpha))
        seed_pool.append(a)

    # sparse random seeds
    for _ in range(6):
        seed_pool.append(_distribution_random_sparse(n0, rng, density=0.008))

    # incorporate a previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            seed_pool.append(prev)
    except Exception:
        pass

    # --------------------------------------------------------------
    # 2.  Evaluate seeds, keep the best few
    top_k = 12
    scored = [( _objective(a), a.copy() ) for a in seed_pool]
    scored.sort(key=lambda x: x[0])
    top_seeds = [a for _, a in scored[:top_k]]

    overall_best_seq = top_seeds[0].copy()
    overall_best_score = _objective(overall_best_seq)
    print(f"[{time.time():.1f}] initial best score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Multi‑scale Adam descent – progressively upsample
    max_n = 20000                      # absolute limit for up‑sampling
    cur_a = overall_best_seq.copy()
    cur_n = cur_a.shape[0]

    while time.time() < deadline:
        # allocate a fraction of the remaining budget to the current resolution
        time_left = deadline - time.time()
        if time_left < 15.0:
            break
        alloc = min(0.3 * time_left, 120.0)   # at most two minutes per scale
        sub_deadline = time.time() + alloc

        print(f"[{time.time():.1f}] Adam descent on length={cur_n} (budget {alloc:.1f}s)")
        cur_a, cur_score = _adam_descent(
            cur_a,
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr=0.12,
            alpha_start=0.20,
            alpha_end=0.02,
            eval_interval=200,
            patience=3000,
        )
        # update the global best if improved
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best_seq = cur_a.copy()
            print(f"[{time.time():.1f}] new global best after descent {overall_best_score:.6f}")

        # decide whether to up‑sample
        next_n = int(cur_n * 1.8)
        if next_n <= cur_n or next_n > max_n:
            break
        # ensure we still have enough time for at least a tiny refinement
        if time.time() + 5.0 >= deadline:
            break
        cur_a = _upsample_distribution(cur_a, next_n)
        cur_n = next_n
        # after up‑sampling the objective may have slightly increased;
        # recompute it for bookkeeping
        cur_score = _objective(cur_a)
        if cur_score < overall_best_score:
            overall_best_score = cur_score
            overall_best_seq = cur_a.copy()
            print(f"[{time.time():.1f}] new global best after upsample {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Final polishing – deterministic swaps then cheap SA
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=200_000,
        )

    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
