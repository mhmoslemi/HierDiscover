    idx = np.arange(n, dtype=np.float64)

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

Key ideas
---------

1. **Rich candidate pool** – In addition to the families used before we
   add monotone power‑law and exponential families (including their
   reversed versions).  All candidates are normalised to sum = 1, so the
   objective to minimise becomes `2·n·max(conv)`.

2. **Linear‑programming polishing** – Starting from the best few
   candidates we run a small number of LP‑iterations.
   For a current vector `f` we compute `M = max(f*f)` and solve

       maximise   sum(g)
       subject to  (f * g)[k] ≤ M   for every shift k
                   g ≥ 0

   (the convolution is linear in `g`).  If the LP yields a vector `g`
   whose total mass > 1 we can scale it back to the simplex and mix it
   with `f`.  This is exactly the iteration described in the original
   paper and quickly drives the bound down.

3. **Light local optimisation** – After the LP steps a short mirror‑
   descent (exponentiated‑gradient) runs to fine‑tune the vector.

The whole search respects the 1000 s deadline, keeps an elite list of
promising vectors and finally returns the best one found.
"""

import time
import random
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ------------------------------------------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_power(n, p, reverse=False):
    """a_i ∝ i**p   (i from 0 to n‑1).  reverse flips the order."""
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        idx = n - 1 - idx
    w = idx ** p
    # keep a zero at the very start when p>0 (i=0 → 0)
    w[0] = 0.0 if p > 0 else 1.0
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_exponential(n, beta, reverse=False):
    """a_i ∝ exp(beta * i)   (or decreasing if reverse=True)."""
    idx = np.arange(n, dtype=np.float64)
    if reverse:
        idx = n - 1 - idx
    w = np.exp(beta * idx)
    total = w.sum()
    if total == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= total
    return w


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

# Local optimisation ----------------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """Exponentiated‑gradient (mirror) descent on a smooth‑max surrogate."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    schedule = [
        (1.0, 1e3, 2000, 0.12, 0.012),
        (1e3, 1e5, 2000, 0.012, 0.0012),
        (1e5, 1e6, 2000, 0.0012, 0.00012),
    ]

    for a0_alpha, a1_alpha, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _hill_climb_prune(a0, deadline):
    """Very light hill‑climb with aggressive pruning of tiny entries."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1
        # move a tiny amount from the index that contributes most to M
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair (i,j) that actually attains the max
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j
        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        if it % 2000 == 0:
            # prune tiny entries
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

    return best_a, best_sc

# Candidate generator ----------------------------------------------------
def _generate_candidates(n):
    """Return a list of normalised candidate vectors for length n."""
    cand = []

    # Classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))          # arcsine
    cand.append(_distribution_beta_sym(n, 1.0))          # triangular beta

    # power‑law families (including reversed)
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        cand.append(_distribution_power(n, p, reverse=False))
        cand.append(_distribution_power(n, p, reverse=True))

    # exponential families
    for beta in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, beta, reverse=False))
        cand.append(_distribution_exponential(n, beta, reverse=True))

    # dense grid of symmetric betas
    for p in np.concatenate(
        (np.logspace(-2, -0.2, 12), np.linspace(0.1, 2.0, 20))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with the uniform distribution
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.2, 0.5, 0.8):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # several Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start
    k = max(2, n // 300)
    cand.append(_distribution_sparse_random(n, k))

    return cand

# Linear‑programming polishing -------------------------------------------
def _lp_polish(f, deadline, max_iters=5, mix_factor=0.5):
    """
    Perform up to `max_iters` LP‑based improvements of vector `f`.
    Returns a (potentially) better vector and its score.
    """
    n = f.shape[0]
    a = f.copy()
    best_sc = _max_convolution_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # current bound
        M = _fft_self_convolve(a).max()
        # build constraint matrix: (2n‑1) × n
        rows = 2 * n - 1
        A = np.zeros((rows, n), dtype=np.float64)
        for k in range(rows):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i = np.arange(i_start, i_end + 1, dtype=int)
            j = k - i
            A[k, j] = a[i]

        # solve LP: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        b_ub = np.full(rows, M, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
        )
        if not res.success:
            # LP failed – abort polishing
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # No improvement possible
            break

        # scale to the simplex and blend
        g = g / sum_g
        a = (1 - mix_factor) * a + mix_factor * g
        a = _project_onto_simplex(a, 1.0)

        sc = _max_convolution_score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            print(f"[{time.time():.1f}] LP iteration {it+1} new best {sc:.6f}")

    return a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []   # (score, vector)

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)

            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")

            # maintain elite list (max 8 entries)
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, vec.copy())
                elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. LP‑based polishing of the elite vectors (restricted to modest n)
    #    – we only polish vectors with n ≤ 1024 to keep the LP cheap.
    for idx, (sc0, vec0) in enumerate(elite[:4]):   # up to four restarts
        if time.time() >= deadline:
            break
        n = vec0.shape[0]
        if n > 1024:
            continue  # skip large sizes for the LP stage
        print(f"[{time.time():.1f}] LP polish start {idx+1}/4 len={n} init score={sc0:.6f}")

        # a couple of LP iterations
        vec, sc = _lp_polish(vec0, deadline, max_iters=5, mix_factor=0.5)

        # short mirror‑descent fine‑tuning
        vec, sc = _smooth_max_mirror_descent(vec, deadline)

        # final tiny hill‑climb
        vec, sc = _hill_climb_prune(vec, deadline)

        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = vec.copy()
            print(f"[{time.time():.1f}] New global best after LP polish {sc:.6f}")

    # ------------------------------------------------------------------
    # 3. If nothing was found (unlikely) fall back to a uniform vector
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final sanitisation
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
