"""
Optimise a non‑negative step‑function (list of floats) for the
`evaluate_sequence` objective.

The optimiser works in three phases:

1. **Broad exploration** – a richer set of analytic families (Beta,
   asymmetric Beta, edge‑biased power‑law, Dirichlet draws, random
   Gauss‑perturbed vectors) are instantiated for several lengths up
   to 32 000.  All candidates are normalised to sum = 1 and the best few
   are remembered.

2. **Projected sub‑gradient descent** – the current best vector is refined
   by an ADAM‑style projected sub‑gradient scheme that directly minimises
   the true max‑convolution (the quantity inside `evaluate_sequence`).
   The step size decays and the velocity is periodically jittered to
   avoid plateaus.

3. **Targeted mass‑transfer hill‑climber** – a fast deterministic
   improvement procedure repeatedly moves a tiny amount of probability mass
   from the pair that contributes most to the current peak toward the
   globally smallest entry.  Afterwards a few random mass‑transfer moves
   are attempted as a final polish.

The routine respects a hard time‑budget (default 1000 s) and returns the
best sequence found as a plain Python list.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1,
    returns the exact value produced by `evaluate_sequence`
    (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Random Dirichlet draw with concentration concentrated at the edges:
    concentration ∝ ((i+½)(n‑i‑½))^{-beta_exp} + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


def _distribution_random_gauss(n: int, scale: float = 0.3) -> np.ndarray:
    """Gaussian draw, clipped to non‑negative and normalised."""
    v = np.abs(np.random.randn(n)) * scale

def _distribution_mix(n: int, mix_ratio: float = 0.5) -> np.ndarray:
    """Mixture of uniform and arcsine: (1‑r)·U + r·A, re‑normalised."""
    u = _distribution_uniform(n)
    a = _distribution_arcsine(n)
    w = (1.0 - mix_ratio) * u + mix_ratio * a
    w /= w.sum()
    return w

def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One ADAM‑style projected sub‑gradient step on the true max‑convolution.
    Returns the updated vector, the new velocity and the current maximal
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # indices attaining the maximal value (within a tight tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # ADAM update (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Targeted hill‑climber that repeatedly moves a tiny amount of probability
    mass from the pair that contributes most to the current peak towards the
    smallest entry.  Stops after `max_iters` or when time runs out.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())                # location of the current peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # Find the pair (i,j) giving the largest product a_i * a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # Move mass from the larger of the two towards a tiny target (global min)
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            # pick second‑smallest if they coincide
            target = np.argsort(a)[1]

        # step size – a few percent of the source mass, capped at 0.02
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-14:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional small jitter to keep progress
        if (it + 1) % 3000 == 0:
            jitter = np.random.randn(n) * 1e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a


def _random_mass_moves(a: np.ndarray,
                       max_iters: int,
                       eps: float = 1e-12):
    """
    Very lightweight stochastic move: repeatedly select a source index
    proportional to its mass and a random target, transfer a small fraction
    of probability, keep the move if it improves the score.
    """
    n = a.shape[0]
    best_score = _max_convolution_score(a)
    for _ in range(max_iters):
        # source weighted by current mass
        source = np.random.choice(n, p=a / a.sum())
        # ensure target differs
        target = np.random.randint(n)
        if target == source:
            continue
        # transfer up to 5 % of source mass
        delta = a[source] * 0.05 * random.random()
        if delta <= eps:
            continue
        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_score = _max_convolution_score(a2)
        if new_score < best_score:
            a = a2
            best_score = new_score
    return a, best_score

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The function respects a hard time‑budget (default
    1000 s) and returns the best sequence found as a plain list.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # keep a small safety margin for the evaluator and prints
    deadline = start + budget_s - 5.0

    eps = 1e-12

    # ------------------------------------------------------------------
    # Load a previously discovered construction if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            best_seq = prev.copy()
            best_score = _max_convolution_score(best_seq)
            print(f"[{time.time():.1f}] loaded previous best score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration over several lengths and families
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]
    # extended parameter grids
    p_grid = np.linspace(0.05, 2.0, 15)        # for symmetric Beta(p,p)
    asym_p_grid = np.linspace(0.05, 2.0, 12)   # for asymmetric Beta(p, q)
    q_fixed = [0.5, 1.0, 2.0]                 # a few different q's
    alpha_grid = [0.20, 0.30, 0.40, 0.50, 0.60]

    # keep the top K candidates globally (K small to stay cheap)
    K = 6
    top_candidates = []  # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families -----------------------------------
        # uniform
        vec = _distribution_uniform(n)
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] n={n} uniform    score={sc:.6f}")
        top_candidates.append((sc, vec.copy()))

        # arcsine
        vec = _distribution_arcsine(n)
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] n={n} arcsine    score={sc:.6f}")
        top_candidates.append((sc, vec.copy()))

        # symmetric Beta(p,p)
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f}) score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

        # asymmetric Beta(p,q) with several q
        for q in q_fixed:
            for p in asym_p_grid:
                if time.time() >= deadline:
                    break
                vec = _distribution_beta(n, p, q)
                sc = _max_convolution_score(vec)
                if sc < best_score:
                    best_seq, best_score = vec.copy(), sc
                    print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f},q={q}) score={sc:.6f}")
                top_candidates.append((sc, vec.copy()))

        # edge‑biased deterministic power‑law
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_power(n, alpha)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edgePower(alpha={alpha:.2f}) score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

        # random Dirichlet draws (edge‑biased)
        for _ in range(5):
            if time.time() >= deadline:
                break
            beta_exp = random.uniform(0.2, 1.0)
            vec = _distribution_edge_dirichlet(n, beta_exp)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edge‑Dirichlet(beta={beta_exp:.3f}) score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

        # random Gaussian draw
        for _ in range(4):
            if time.time() >= deadline:
                break
            vec = _distribution_random_gauss(n, scale=0.5)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} random‑Gauss    score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

        # mixture of uniform and arcsine
        for _ in range(4):
            if time.time() >= deadline:
                break
            r = random.random()
            vec = _distribution_mix(n, mix_ratio=r)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} mix(r={r:.2f}) score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

    # Keep only the K best candidates (including the current best)
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:K]

    # ------------------------------------------------------------------
    # 2️⃣  Projected ADAM‑style sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: a modest uniform vector
        n0 = 8000
        best_seq = _distribution_uniform(n0)
        best_score = _max_convolution_score(best_seq)

    a = best_seq.copy()
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta1 = 0.9
    step0 = 0.8                 # a slightly larger initial step
    sub_it = 0

    # sub‑gradient phase – run until about 80 % of the budget is spent
    sub_deadline = start + 0.78 * budget_s
    while time.time() < sub_deadline:
        sub_it += 1
        a, velocity, cur_max = _adam_subgrad_step(a, velocity, step0,
                                                  beta1, sub_it, eps)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cur_score:.6f}")

        # occasional jitter to escape shallow plateaus
        if sub_it % 500 == 0:
            jitter = np.random.randn(n) * 3e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 3️⃣  Targeted mass‑transfer hill‑climber for final polishing
    # ------------------------------------------------------------------
    a = best_seq.copy()
    hill_iters = 25000                     # aggressive hill‑climb budget
    a = _mass_transfer_hill_climb(a, hill_iters, eps)
    score_after_hill = _max_convolution_score(a)
    if score_after_hill < best_score:
        best_seq, best_score = a.copy(), score_after_hill
        print(f"[{time.time():.1f}] hill‑climber improved score to {score_after_hill:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Light random mass‑move polishing (stochastic local search)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_random_moves = int(1e4 * remaining / 200.0)   # roughly 100 moves per second
        a, rnd_score = _random_mass_moves(best_seq.copy(),
                                          max_random_moves, eps)
        if rnd_score < best_score:
            best_seq, best_score = a.copy(), rnd_score
            print(f"[{time.time():.1f}] random‑move polish improved score to {rnd_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
