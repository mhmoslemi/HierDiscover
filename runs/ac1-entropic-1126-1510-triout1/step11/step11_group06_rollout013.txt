"""
Search for a non‑negative step function that minimizes `evaluate_sequence`.

The method consists of three phases

1️⃣  **Broad exploration** – generate candidates from a rich family of
    parametric distributions (uniform, arcsine, Beta, edge‑power,
    Dirichlet) for several lengths.  For each candidate we also try a
    handful of permutations (original order, reversed, a deterministic
    high‑low interleaving and a few random shuffles) because the ordering
    of the values strongly influences the maximal self‑convolution.

2️⃣  **Continuous refinement** – starting from the best candidate we run
    a smooth‑max mirror‑descent (exponentiated‑gradient on a soft‑max
    approximation of the max‑convolution) followed by an Adam‑style
    sub‑gradient descent on the true max‑convolution.  Both stages keep
    the vector on the probability simplex.

3️⃣  **Discrete polishing** – a cheap hill‑climbing that repeatedly
    moves a small amount of mass from a relatively large coordinate to
    a small one (and also performs occasional random swaps).  Every
    improvement updates the global best.

The whole routine respects the supplied `budget_s` (default 1000 s) and
always returns a plain Python list of floats.
"""

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions -----------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Score for a probability vector (sum == 1):
        2 * n * max(a * a)
    This equals the value returned by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families (all return vectors with sum == 1) ---------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n: int, p: float) -> np.ndarray:
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w = np.maximum(w, 1e-15)
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """Dirichlet draw biased towards the edges."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    return np.random.dirichlet(conc)


def _distribution_sparse_dirichlet(n: int, alpha: float) -> np.ndarray:
    """
    Dirichlet with concentration < 1 (sparse) – each coordinate gets a
    gamma(alpha) sample.
    """
    if alpha <= 0:
        raise ValueError
    g = np.random.gamma(alpha, 1.0, size=n)
    # Avoid exact zeros
    g = np.maximum(g, 1e-15)
    g /= g.sum()
    return g


# ----------------------------------------------------------------------
# Permutation helpers ----------------------------------------------------
def _interleaved_desc(a: np.ndarray) -> np.ndarray:
    """Place the largest value, then the smallest, then the second largest,
    then the second smallest, …  (input must be sorted descending)."""
    n = a.shape[0]
    out = np.empty_like(a)
    lo, hi = 0, n - 1
    pos = 0
    while lo <= hi:
        out[pos] = a[lo]
        pos += 1
        lo += 1
        if lo <= hi:
            out[pos] = a[hi]
            pos += 1
            hi -= 1
    return out
