def _penalty_descent(a0, deadline, max_iters=20000):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best = a.copy()
    best_sc = _max_convolution_score(a)
    it = 0
    target = None

        it += 1
        conv = _fft_self_convolve(a)
        M = conv.max()
        # reduce target gradually
        if target is None:
            target = M * 0.99
        else:
            # gradually lower target over iterations maybe
            target = max(target * 0.9999, M * 0.99)  # ensure still <= M
        excess = np.maximum(conv - target, 0.0)
        if excess.sum() == 0:
            # target already feasible, reduce further

            continue

        grad = 4.0 * _fft_convolve(excess, a[::-1])[n-1:n-1+n]

   lengths (powers of two up to 32768).  Keep a small elite list of the
   best candidates.

2. **Heavy local optimisation** – for each elite vector run
   * a longer smooth‑max mirror‑descent (exponential‑gradient) with a
     three‑stage temperature schedule (α ramps from 1 to 1e 8);
   * a projected sub‑gradient descent with Adam‑style momentum;
   * a *penalty descent* that gradually lowers the allowed maximum
     convolution value by penalising all entries that exceed a moving
     target; this pushes the whole convolution profile towards a flat
     shape;
   * a final hill‑climbing / support‑pruning phase.

   All optimisation loops are *time‑aware*: they stop as soon as the
   global deadline (budget_s – 2 s safety margin) is reached.

3. **Result normalisation** – enforce non‑negativity, renormalise to
   unit sum and return a plain Python list.

The algorithm works entirely on the simplex (`sum a = 1`), which makes
the scaling performed by `evaluate_sequence` irrelevant.  Extensive
printing is used to monitor progress and intermediate best scores.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow/underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform (both already normalised)."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    return np.random.dirichlet(np.full(n, alpha))


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a


def _distribution_sin_power(n, p):
    """Half‑sine raised to power p (zero at the edges, peak in the middle)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * x / n) ** p
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_cosine_heavy(n, p):
    """|cos| raised to power p – heavy at the ends, low in the centre."""
    x = np.arange(n, dtype=np.float64) + 0.5
    w = np.abs(np.cos(np.pi * x / n)) ** p
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Local optimisation ----------------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """
    Exponentiated‑gradient (mirror) descent on a smooth‑max surrogate.
    The temperature α is raised from 1 to 1e 8, while the learning rate
    decays.  Very many steps are taken (≈ 25 000) unless the deadline is
    reached.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # schedule: (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e3, 6000, 0.20, 0.020),
        (1e3, 1e5, 6000, 0.020, 0.0020),
        (1e5, 1e8, 8000, 0.0020, 0.00020),
    ]

    for a0_alpha, a1_alpha, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 1000 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline, max_iters=50000):
    """
    Sub‑gradient descent with Adam‑style momentum and back‑tracking line
    search.  Runs as long as time permits (capped at max_iters).
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search on the true objective
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _penalty_descent(a0, deadline, max_iters=20000):
    """
    Gradually lower the allowed maximum self‑convolution value.
    Uses a simple squared‑penalty on all entries that exceed a moving
    target and performs projected gradient steps.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # start with a target a few percent below the current maximum
    conv = _fft_self_convolve(a)
    M = conv.max()
    target = M * 0.99

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        M = conv.max()
        # shrink the target slowly, never above the current max
        target = max(target * 0.9999, M * 0.99)

        excess = np.maximum(conv - target, 0.0)
        if excess.sum() == 0:
            # nothing exceeds – lower target more aggressively
            target = M * (1 - 0.02 * (it + 1) / max_iters)
            continue

        # grad = 4 * (excess * a_rev) restricted to length n
        grad = 4.0 * _fft_convolve(excess, a[::-1])[n - 1 : n - 1 + n]

        # simple step size based on gradient magnitude
        step = 0.7 / (np.linalg.norm(grad) + 1e-12)

        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, 1.0)
        a_candidate = np.maximum(a_candidate, eps)
        a_candidate /= a_candidate.sum()

        sc = _max_convolution_score(a_candidate)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_candidate.copy()
            print(f"[{time.time():.1f}] penalty it={it} new best {sc:.6f}")

        a = a_candidate

    return best_a, best_sc


def _hill_climb_prune(a0, deadline):
    """Simple hill‑climbing moves plus aggressive pruning of tiny entries."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # move a little mass away from the pair that generates the current max
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j

        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # aggressive pruning every few thousand steps
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Candidate generator ----------------------------------------------------
def _generate_candidates(n):
    """Yield a list of normalised vectors for a given length n."""
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))   # arcsine

    # grid of symmetric betas (richer than the original script)
    for p in np.concatenate(
        (np.logspace(-4, -0.2, 10), np.linspace(0.1, 2.0, 20))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with uniform
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.2, 0.5, 0.8):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # sine / cosine based families
    for p in (1.0, 2.0, 4.0):
        cand.append(_distribution_sin_power(n, p))
        cand.append(_distribution_cosine_heavy(n, p))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start
    k = max(2, n // 400)
    cand.append(_distribution_sparse_random(n, k))

    # random shuffles of a few base vectors to introduce asymmetry
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []               # list of (score, vector)

    # ------------------------------------------------------------------
    # 1. Broad exploration – several lengths (powers of two)
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)

            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")

            # maintain a short elite list (size ≤6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate any historic solution (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in case nothing was generated (should never happen)
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy local optimisation from the elite pool
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.6f}")

        # smooth‑max mirror descent (longer than original)
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # projected sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # penalty descent to flatten the whole convolution profile
        a_mid, sc_mid = _penalty_descent(a_mid, deadline)

        # final hill‑climber + pruning
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity, normalise and report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
