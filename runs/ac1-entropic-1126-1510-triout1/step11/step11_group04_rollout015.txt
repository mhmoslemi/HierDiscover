def _smooth_max_mirror_descent(a, deadline):
    n = a.shape[0]
    eps=1e-12
    best_a = a.copy()
    best_sc = _objective(best_a)
    # schedule: list of (alpha0, alpha1, steps, lr0, lr1)
    schedule = [
        (1.0, 1e3, 15000, 0.5, 0.05),
        (1e3, 1e6, 20000, 0.05, 0.004),
        (1e6, 1e8, 25000, 0.004, 0.0003)
    ]
    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it/steps
            alpha = alpha0 * (alpha1/alpha0)**t
            lr = lr0 * (lr1/lr0)**t
            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0/w.size)
            else:
                w /= w_sum
            grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()
            if it % 5000 == 0:
                sc = _objective(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")
    return best_a, best_sc

        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)  # distribute weight evenly for stability
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        vel = beta1 * vel + (1 - beta1) * grad
        v_hat = vel / (1 - beta1**it)
        step = base_step
        # simple backtrack
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5
        a = cand

            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")
        # occasional jitter to escape plateaus

            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
    return best_a, best_sc

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

Algorithm (≈1000 s budget)
--------------------------------
1. **Wide‑range seeding** – generate many analytic families (Beta,
   Gaussian, power‑law, sine‑power, exponential, etc.) for several
   lengths.  Keep a tiny elite (≤ 8) of the best candidates (according
   to the *scale‑invariant* objective 2·n·max(conv)).

2. **Heavily‑tuned local search** – for each elite vector run three
   phases:
     * *Smooth‑max mirror descent* – exponentiated‑gradient descent on a
       log‑sum‑exp “soft‑max’’ of the convolution.  Three temperature
       stages are used (α from 1 → 10⁸) with 15 k → 25 k iterations per
       stage and a geometrically decaying learning‑rate.
     * *Projected (Adam‑style) sub‑gradient polishing* – true
       sub‑gradient steps on the exact max‑convolution with back‑track
       line‑search, up to 20 k iterations.
     * *Hill‑climbing & pruning* – small random mass transfers from the
       pair that dominates the maximal convolution term, occasional
       pruning of negligible entries and jitter to escape plateaus.
   All phases respect the simplex constraint (non‑negative, sum = 1).

3. **Random restarts** – while time permits, draw random Dirichlet
   vectors (various concentrations and lengths) and run a short
   optimisation (≈ 5 k mirror‑descent steps).  If an improvement is
   found it replaces the current elite.

4. **Return the best vector** – projected onto the feasible region and
   converted to a plain Python list.

The optimisation works directly on the scale‑invariant objective
`2·n·max(conv)`.  Because `evaluate_sequence` is homogeneous of degree 0,
the returned list can be any non‑negative vector with the same shape;
the evaluator will automatically normalise it.
"""

# Linear‑algebra utilities ------------------------------------------------

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Scale‑invariant objective used during optimisation.
    The vector a is assumed to be normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())   # equals evaluate_sequence for sum=1

# Analytic families – each returns a vector with sum = 1 -----------------

def _dist_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dist_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _dist_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _dist_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _dist_beta(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on the discrete grid x_i = (i+0.5)/n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _dist_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _dist_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _dist_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _dist_powerlaw(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law a_i ∝ (i+1)^{‑p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _generate_candidates_for_length(n: int) -> list[np.ndarray]:
    """
    Produce a reasonably diverse set of initial vectors of length n,
    all normalised to sum = 1.
    """
    cand = []

    # core analytic families
    cand.append(_dist_uniform(n))
    cand.append(_dist_arcsine(n))
    cand.append(_dist_triangular(n))
    cand.append(_dist_exponential(n, lam=0.003))
    cand.append(_dist_exponential(n, lam=0.02))

    # symmetric Beta – dense sweep on p
    for p in np.linspace(0.1, 3.0, 30):
        cand.append(_dist_beta(n, p, p))

    # asymmetric Beta – a few interesting pairs
    for p, q in [(0.3, 1.2), (1.2, 0.3), (0.5, 1.5), (1.5, 0.5),
                 (0.2, 0.8), (0.8, 0.2)]:
        cand.append(_dist_beta(n, p, q))

    # sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_dist_sine_power(n, pw))

    # cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_dist_cosine_window(n, coeff))

    # Gaussian – several σ
    for sigma in (0.10, 0.15, 0.20, 0.25, 0.30, 0.35):
        cand.append(_dist_gaussian(n, sigma))

    # power‑law and its reverse
    for p in (1.1, 1.3, 1.5, 1.8, 2.0, 2.5):
        pl = _dist_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Dirichlet draws
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # Sparse start – few non‑zeros
    k = max(2, n // 400)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Random shuffles of a few base vectors to add asymmetry
    for base in (cand[0], cand[1], cand[4]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

def _smooth_max_mirror_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror descent on a smoothed version of the
    max‑convolution.  Three temperature stages, geometric learning‑rate
    decay; continues until the deadline is reached.
    """
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _objective(best_a)

    # (alpha0, alpha1, steps, lr0, lr1)
    schedule = [
        (1.0, 1e3, 15000, 5e-1, 5e-2),
        (1e3, 1e6, 20000, 5e-2, 4e-3),
        (1e6, 1e8, 25000, 4e-3, 3e-4),
    ]

    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # soft‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed max
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient update (multiplicative mirror descent)
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)      # avoid exact zeros
            a /= a.sum()

            if it % 5000 == 0:
                sc = _objective(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a: np.ndarray, deadline: float, max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the *exact* max‑convolution.
    Uses a simple back‑tracking line‑search.  Stops after the deadline
    or after `max_iters` iterations.
    """
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    base_step = 0.5

    best_a = a.copy()
    best_sc = _objective(best_a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size   # distribute equally over the active set

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum update
        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_step
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        if it % 5000 == 0:
            # tiny jitter to shake possible plateaus
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    return best_a, best_sc

        k = int(conv.argmax())

    Returns the best sequence as a plain Python list.
    """

    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # a small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # 1. Broad exploration – keep a tiny elite (≤ 8)
    lengths = [3000, 4000, 5000, 8000, 12000]

            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2. Incorporate previously found sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Fallback – should never trigger
    if best_seq is None:
        best_n = 4000
        best_seq = _dist_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Random Dirichlet restarts while time permits (short runs)
    while time.time() < deadline - 10.0:        # keep 10 s for the deep phase
        n = random.choice(lengths)
        alpha = 10 ** random.uniform(-1, 1)     # concentration between 0.1 and 10
        vec = np.random.dirichlet(np.full(n, alpha))
        vec = np.maximum(vec, eps)
        vec /= vec.sum()

        # a cheap mirror‑descent run (5 k iters) to filter promising starts
        a_mid, sc_mid = _smooth_max_mirror_descent(vec, deadline - 5.0)
        if sc_mid < best_score - 1e-6:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = n
            print(f"[{time.time():.1f}] Random restart improved to {sc_mid:.6f}")
            elite.append((sc_mid, a_mid.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > 8:
                elite = elite[:8]

    # --------------------------------------------------------------
    # 5. Deep local optimisation from elite candidates (max 5 restarts)
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={a0.shape[0]}) start score {elite[idx][0]:.6f}")

        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        a_mid, sc_mid = _hill_climb_and_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # --------------------------------------------------------------
    # 6. Final sanitisation
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
