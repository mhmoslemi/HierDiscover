"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

Algorithm
---------
1. Choose a reasonable length `n` (powers of two are FFT‑friendly).  
2. Build a rich set of analytic candidates:
   * uniform distribution
   * symmetric / asymmetric Beta(p,q) families
   * mixtures of a Beta with the uniform distribution
   * a few random Dirichlet draws
   * the historic solution `height_sequence_1` (resampled if needed)
   The best few candidates are kept.
3. Starting from the current best candidate run a **simulated‑annealing**
   local search (mass transfer between two coordinates) that accepts
   improving moves and occasional uphill moves.  This quickly explores
   the neighbourhood of the best point.
4. Polish the final candidate with a short **smooth‑max mirror‑descent**
   run (few thousand steps).  The mirror‑descent operates on the
   smooth‑max surrogate of the max‑convolution and always stays on the
   probability simplex.
5. Return the best sequence found (already normalised to sum = 1).

All heavy loops check the remaining time and stop early if the 1000 s
budget is almost exhausted.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities (used by the mirror‑descent)
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT based convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Returns the quantity 2·n·max(a*a) for a vector a that already satisfies
    sum(a)=1.  This matches `evaluate_sequence` on a normalised vector.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families
def _distribution_uniform(n):
    """Uniform density on the discrete grid."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta(p, q, n):
    """
    Asymmetric Beta(p,q) on the grid x_i = (i+0.5)/n.
    Returns a normalised vector (sum = 1).  Works for any p,q>0.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    # under‑/overflow protection
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture(beta_vec, uniform_vec, w):
    """Linear mixture w·beta + (1‑w)·uniform, re‑normalised."""
    mix = w * beta_vec + (1.0 - w) * uniform_vec
    mix = np.maximum(mix, 0.0)
    mix_sum = mix.sum()
    if mix_sum == 0.0:
        mix = uniform_vec.copy()
    else:
        mix /= mix_sum
    return mix


def _distribution_random_dirichlet(n, alpha):
    """Random draw from Dirichlet(alpha,…,alpha) and normalise."""
    vec = np.random.dirichlet(np.full(n, alpha))
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        vec = _distribution_uniform(n)
    else:
        vec /= s
    return vec

# Simulated‑annealing local optimiser
def _simulated_annealing(start_vec, deadline, max_iters=25000):
    """
    Simple SA that moves a random amount of mass from one coordinate to
    another.  The move is accepted if it improves the objective or with
    probability exp( -Δ / T ) where the temperature T decays exponentially.
    Returns the best vector seen.
    """
    a = start_vec.copy()
    n = a.shape[0]
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score

    # temperature schedule
    T0 = 1e-2
    T_end = 1e-6
    decay = (T_end / T0) ** (1.0 / max_iters)

    # maximum absolute mass we are willing to transfer in one step
    max_step = 0.04

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # update temperature
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = random.sample(range(n), 2)
        delta = min(a[i] * 0.5, max_step) * random.random()
        if delta <= 0.0:
            continue
        a_cand = a.copy()
        a_cand[i] -= delta
        a_cand[j] += delta
        # (sum unchanged, non‑negativity already guaranteed)
        cand_score = _max_convolution_score(a_cand)

        accept = False
        if cand_score < cur_score - 1e-14:
            accept = True
        else:
            # uphill move with probability
            if T > 0.0 and random.random() < math.exp(-(cand_score - cur_score) / T):
                accept = True

        if accept:
            a = a_cand
            cur_score = cand_score
            if cand_score < best_score - 1e-14:
                best_score = cand_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA it={it} new best {best_score:.6f}")

        # occasional random jitter to escape flat plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = _distribution_uniform(n)
            else:
                a /= a_sum
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA jitter it={it} new best {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Light mirror‑descent (smooth‑max) used for final polishing
def _smooth_max_mirror_descent(a0, deadline, steps_per_phase=1500):
    """
    Exponentiated‑gradient mirror descent on the smooth max surrogate.
    A very compact schedule (three phases) is used – sufficient for a
    final polish after SA.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(a)

    schedule = [
        (1.0, 1e3, steps_per_phase, 0.12, 0.012),
        (1e3, 1e5, steps_per_phase, 0.012, 0.0012),
        (1e5, 1e6, steps_per_phase, 0.0012, 0.00012),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smoothed max (see original code)
            grad = 2.0 * np.fft.irfft(
                np.fft.rfft(a[::-1]) * np.fft.rfft(w), n=_next_pow2(2 * n - 1)
            )[:n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 300 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] MD it={it} new best {sc:.6f}")

    return best_a, best_sc

# Helper: resample a given vector to a new length (linear interpolation)
def _resample_vector(v, new_len):
    v = np.asarray(v, dtype=np.float64)
    if v.ndim != 1 or v.size == 0:
        raise ValueError
    if v.shape[0] == new_len:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, v.shape[0])
    x_new = np.linspace(0.0, 1.0, new_len)
    v_new = np.interp(x_new, x_old, v)
    v_new = np.maximum(v_new, 0.0)
    s = v_new.sum()
    if s == 0.0:
        v_new = _distribution_uniform(new_len)
    else:
        v_new /= s
    return v_new
