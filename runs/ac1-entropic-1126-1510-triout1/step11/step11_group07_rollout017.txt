def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Exact 2·n·max(a*a) for a already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space for stability
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α}+c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w


def _distribution_dirichlet_edge(n: int, beta_exp: float) -> np.ndarray:
    """Dirichlet draw with edge‑biased concentration."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float, eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective  f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector and the exact current max.
    """
    conv = _fft_self_convolve(a)
    m = conv.max()
    w = np.exp(alpha * (conv - m))          # soft‑max weights (numerically stable)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    a_new = a * np.exp(-lr * grad)          # KL‑mirror update
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One Adam‑style projected sub‑gradient step on the true max‑convolution.
    Returns (new vector, new velocity, max‑conv value before projection).
    """
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices where the max is essentially attained)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

    # Adam bias‑corrected momentum
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search
    step = step0
    while True:
        cand = a - step * v_hat
        cand = _project_onto_simplex(cand, 1.0)
        cand = np.maximum(cand, eps)
        cand /= cand.sum()
        cand_score = _max_convolution_score(cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return cand, vel, cur_max


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Deterministic hill‑climber: repeatedly move a tiny amount of mass from the
    pair (i,j) that contributes most to the current peak towards the globally
    smallest entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())                 # location of the current peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair with the largest product on this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a


def _random_mass_transfer(a: np.ndarray,
                          rng: np.random.Generator,
                          max_iters: int,
                          max_frac: float = 0.04,
                          eps: float = 1e-12):
    """
    Stochastic hill‑climber – repeatedly pick a source index with probability
    proportional to its weight and a random target, transfer a tiny random
    fraction of the source and keep the change if the max‑convolution improves.
    """
    n = a.shape[0]
    cur_score = _max_convolution_score(a)
    for _ in range(max_iters):
        source = rng.choice(n, p=a)               # favour larger entries
        target = rng.integers(0, n)
        while target == source:
            target = rng.integers(0, n)

        delta = min(a[source] * max_frac, a[source] * rng.random() * max_frac)
        if delta < eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_score = _max_convolution_score(a2)
        if new_score < cur_score - 1e-12:
            a = a2
            cur_score = new_score
    return a

def _refine_candidate(a0: np.ndarray,
                     remaining: float,
                     rng: np.random.Generator,
                     eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Run mirror‑descent → Adam sub‑gradient → deterministic & stochastic hill‑climb
    on a single starting vector a0, using at most `remaining` seconds.
    Returns (best_vector, best_score).
    """
    start = time.time()
    a = a0.copy()
    n = a.shape[0]

    # ---------- mirror descent ----------
    # allocate half of the remaining budget to this phase (rough heuristic)
    phase_time = remaining * 0.4
    iters_md = 0
    while time.time() - start < phase_time and iters_md < 30000:
        # schedule: alpha grows, lr shrinks
        t = iters_md / 30000
        alpha = 0.3 * (5e5 / 0.3) ** t
        lr = 0.25 * (0.001 / 0.25) ** t
        a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)
        iters_md += 1
    # -----------------------------------------------------------------
    # ---------- Adam sub‑gradient ----------
    phase_time = remaining * 0.3
    vel = np.zeros_like(a)
    it = 0
    while time.time() - start < phase_time:
        it += 1
        a, vel, cur_max = _adam_subgrad_step(a, vel,
                                             step0=0.6,
                                             beta1=0.9,
                                             beta2=0.999,
                                             it=it,
                                             eps=eps)
    # -----------------------------------------------------------------
    # ---------- deterministic hill‑climber ----------
    phase_time = remaining * 0.2
    max_iters = int(phase_time / 0.003) if phase_time > 0 else 0
    a = _mass_transfer_hill_climb(a, max_iters, eps)

    # -----------------------------------------------------------------
    # ---------- stochastic hill‑climber ----------
    phase_time = remaining * 0.1
    max_iters = int(phase_time / 0.0005) if phase_time > 0 else 0
    a = _random_mass_transfer(a, rng, max_iters, max_frac=0.02, eps=eps)

    best_score = _max_convolution_score(a)
    return a, best_score

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The function respects a hard time budget (default 1000 s) and returns
    the best sequence found as a plain list.
    """
    random.seed(seed)
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 1.0          # small safety margin
    eps = 1e-12

    # ------------------------------------------------------------------
    # Try to import a previously known construction
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            best_seq = prev.copy()
            best_score = _max_convolution_score(best_seq)
            print(f"[{time.time():.1f}] loaded previous best score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration over several lengths and families
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000]       # modest sizes – fast evaluation
    p_grid = np.linspace(0.02, 0.48, 12)      # beta shape parameters
    q_grid = np.linspace(0.52, 0.98, 5)       # asymmetric partner
    alpha_grid = [0.30, 0.45, 0.60, 0.80]    # edge‑power exponents
    K = 12                                   # keep the top‑K candidates

    top_cand = []  # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        # Uniform – a cheap baseline
        vec = _distribution_uniform(n)
        sc = _max_convolution_score(vec)
        top_cand.append((sc, vec.copy()))
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] n={n} uniform   score={sc:.6f}")

        # Symmetric and asymmetric Beta families
        for p in p_grid:
            if time.time() >= deadline:
                break
            # symmetric
            vec = _distribution_beta(n, p, p)
            sc = _max_convolution_score(vec)
            top_cand.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f},q={p:.3f}) score={sc:.6f}")

            # asymmetric (p small, q large) and its mirror
            for q in q_grid:
                vec = _distribution_beta(n, p, q)
                sc = _max_convolution_score(vec)
                top_cand.append((sc, vec.copy()))
                if sc < best_score:
                    best_seq, best_score = vec.copy(), sc
                    print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f},q={q:.3f}) score={sc:.6f}")

                vec = _distribution_beta(n, q, p)  # mirror
                sc = _max_convolution_score(vec)
                top_cand.append((sc, vec.copy()))
                if sc < best_score:
                    best_seq, best_score = vec.copy(), sc
                    print(f"[{time.time():.1f}] n={n} Beta(p={q:.3f},q={p:.3f}) score={sc:.6f}")

        # Edge‑biased deterministic power‑law
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_power(n, alpha)
            sc = _max_convolution_score(vec)
            top_cand.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edgePower(alpha={alpha:.2f}) score={sc:.6f}")

        # Random Dirichlet draws with edge‑bias
        for _ in range(3):
            if time.time() >= deadline:
                break
            bex = random.uniform(0.1, 0.9)
            vec = _distribution_dirichlet_edge(n, bex)
            sc = _max_convolution_score(vec)
            top_cand.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edge‑Dirichlet(beta={bex:.3f}) score={sc:.6f}")

    # keep only the K best candidates (including the current best)
    top_cand.sort(key=lambda x: x[0])
    top_cand = top_cand[:K]

    # ------------------------------------------------------------------
    # 2️⃣  Refine each promising candidate (time‑aware)
    # ------------------------------------------------------------------
    for idx, (_, cand) in enumerate(top_cand):
        if time.time() >= deadline - 2.0:      # keep a tiny margin
            break
        remaining = deadline - time.time()
        a_opt, sc_opt = _refine_candidate(cand, remaining, rng, eps)
        if sc_opt < best_score - 1e-12:
            best_seq, best_score = a_opt.copy(), sc_opt
            print(f"[{time.time():.1f}] refined candidate {idx} new best {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final polishing with a short extra stochastic climb (if time left)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        extra_iters = int((deadline - time.time()) / 0.0006)
        best_seq = _random_mass_transfer(best_seq, rng, extra_iters, max_frac=0.015, eps=eps)
        best_score = _max_convolution_score(best_seq)

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
