"""
Propose a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  The search consists of three phases:

1. **Broad parametric sweep** – many lengths (2 k … 128 k) are tried and
   a rich set of families (uniform, arcsine, symmetric/asymmetric
   Beta, deterministic edge‑power, mixtures with uniform/arcsine,
   Dirichlet draws) are sampled.  The best normalised vector (sum = 1)
   is retained.

2. **Soft‑max mirror‑descent** – the current best vector is refined by
   minimising a smooth approximation of the max‑convolution
   (soft‑max with a temperature that grows from 1 to 1e6).  Exponentiated
  ‑gradient updates are projected onto the simplex.

3. **Projected sub‑gradient (Adam‑style) + targeted hill‑climbing** –
   the true max‑convolution is used to obtain a sub‑gradient; an
   Adam‑momentum step with back‑tracking line‑search is applied, and a
   final stochastic mass‑transfer polish removes remaining plateaus.

All work respects the supplied time budget (≈1000 s).  The returned
sequence is a plain Python list and already has sum = 1, which is
compatible with `evaluate_sequence`.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT‑based convolutions ------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _score_normalised(a: np.ndarray) -> float:
    """Score for a normalised vector (Σa = 1): 2·n·max(a*a)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families (all return a vector with Σ = 1) -----------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n: int, p: float) -> np.ndarray:
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15  # avoid exact zeros
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """Dirichlet draw biased towards the edges."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    return np.random.dirichlet(conc)

    `evaluate_sequence`.  Returns a plain Python list."""

    random.seed(seed)

    start = time.time()

    eps = 1e-15

    best_score = float("inf")
    best_n = None

    lengths = [2000, 4000, 8000, 16000, 32000, 64000, 128000]

    beta_sym_ps = np.concatenate([
        np.linspace(0.001, 0.010, 10),
        np.linspace(0.010, 0.050, 15),
        np.linspace(0.050, 0.200, 20),
        np.linspace(0.200, 0.800, 15),
        np.linspace(0.800, 1.200, 10),
    ])

    beta_asym_ps = np.linspace(0.001, 0.150, 20)

    edge_power_ps_fine = np.linspace(0.05, 1.20, 30)
    edge_power_ps_coarse = np.array([0.05, 0.10, 0.20, 0.40, 0.60, 0.80, 1.00, 1.20])

    mix_ws = np.array([0.0, 0.05, 0.10, 0.20, 0.30, 0.40, 0.50])

    for n in lengths:

        sc = _score_normalised(vec)

        vec = _distribution_arcsine(n)

        sc = _score_normalised(vec)

                best_score, best_seq, best_n = sc, base.copy(), n

                vec = np.maximum(vec, eps)
                vec /= vec.sum()
                sc = _score_normalised(vec)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec.copy(), n
                    print(f"[{time.time():.1f}] n={n} edge_power(p={p:.3f})+mixU({mix:.2f}) score={sc:.6f}")

            # mixtures with arcsine
            arcs = _distribution_arcsine(n)
            for mix in mix_ws:
                if mix == 0.0:
                    continue
                vec = (1 - mix) * base + mix * arcs
                vec = np.maximum(vec, eps)
                vec /= vec.sum()
                sc = _score_normalised(vec)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec.copy(), n
                    print(f"[{time.time():.1f}] n={n} edge_power(p={p:.3f})+mixA({mix:.2f}) score={sc:.6f}")

        # Symmetric Beta
        for p in beta_sym_ps:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _score_normalised(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} beta_sym(p={p:.4f}) score={sc:.6f}")

        # Asymmetric Beta (left‑heavy / right‑heavy)
        for p in beta_asym_ps:
            if time.time() >= deadline:
                break
            # left‑heavy
            vec = _distribution_beta(n, p, 0.5)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _score_normalised(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} beta(p={p:.4f},q=0.5) score={sc:.6f}")

            # right‑heavy (swap p,q)
            vec = _distribution_beta(n, 0.5, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _score_normalised(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} beta(p=0.5,q={p:.4f}) score={sc:.6f}")

        # Random Dirichlet draws (concentration 1)
        for _ in range(3):
            if time.time() >= deadline:
                break
            vec = np.random.dirichlet(np.full(n, 1.0))
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _score_normalised(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} dirichlet(1)       score={sc:.6f}")

        # Edge‑biased Dirichlet draws
        for bex in edge_dirichlet_exps:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_dirichlet(n, bex)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _score_normalised(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} edge_dirichlet(b={bex}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  incorporate a previously stored solution (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _score_normalised(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best            score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – should never happen
        n = 4000
        best_seq = _distribution_uniform(n)
        best_score = _score_normalised(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform          score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Soft‑max mirror‑descent refinement on the best candidate
    a = best_seq.copy()
    n = a.shape[0]

    # rough run‑time estimate for one iteration (soft‑max + two FFTs)
    t0 = time.time()
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    wtmp = np.exp(conv - maxc)
    _ = _fft_convolve(a[::-1], wtmp / wtmp.sum())
    per_iter = max(time.time() - t0, 1e-6)

    remaining = deadline - time.time()
    # allow at most 80 % of the remaining budget for this stage
    max_iters = int(min(200000, max(5000, (remaining * 0.8) / per_iter)))
    if max_iters < 1:
        max_iters = 0

    alpha_start, alpha_end = 1.0, 1e6
    lr_start, lr_end = 0.5, 0.001

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        t = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights (stable)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of soft‑max (mirror‑descent direction)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it & max(1, max_iters // 10)) == 0 or it == max_iters - 1:
            cur_sc = _score_normalised(a)
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror it={it}/{max_iters} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected Adam‑style sub‑gradient descent on true max‑conv.
    a = best_seq.copy()
    m = np.zeros_like(a)  # first moment
    beta1, beta2 = 0.9, 0.999
    step0 = 0.5
    sub_it = 0

    while time.time() < deadline:
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # indices attaining the maximum (tolerance for floating error)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # build sub‑gradient (average of active linear pieces)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** sub_it)

        # back‑tracking line search
        step = step0
        while True:
            a_cand = a - step * m_hat
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()
            cand_sc = _score_normalised(a_cand)

            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_cand
        if cand_sc < best_score - 1e-12:
            best_score, best_seq = cand_sc, a.copy()
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cand_sc:.6f}")

        # occasional perturbation to escape stagnation
        if sub_it % 750 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 5️⃣  Targeted stochastic hill‑climbing (mass‑transfer) polish
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0

    while time.time() < deadline:
        hill_it += 1

        # locate the index k where the peak occurs
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair (i,j) on this diagonal with the largest product
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]  # second smallest

        # propose a modest transfer (up to 2 % of source mass)
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 100 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")

        # occasional jitter
        if hill_it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_it}")

    # ------------------------------------------------------------------
    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
