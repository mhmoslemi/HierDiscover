"""
A fast heuristic for the “minimum max‐convolution” problem.

The algorithm consists of three stages:

1. **Broad sampling** – many analytic families (uniform, Beta(p,p),
   mixtures with the uniform law, Dirichlet draws, sparse starts) are
   generated for a few powers‑of‑two lengths.  The best few candidates
   (the *elite* set) are kept.

2. **Active–set linear programming** – for each elite vector we repeatedly
   solve a tiny LP that maximises the ℓ₁‑mass of a new vector `g` while
   keeping the active convolution constraints `(a∗g)[k] ≤ M` (where
   `M` is the current maximum of `a∗a`).  After solving we rescale `g`
   to unit sum and mix it with `a`.  This mimics the classical LP‑step
   described in the problem statement and typically drives the objective
   down quickly.

3. **Projected sub‑gradient polishing** – a final short sub‑gradient
   descent (with Adam‑style momentum) cleans up any remaining slack.

All heavy loops respect a global deadline (the provided time budget minus a
small safety margin).  The returned list is a plain Python list of
non‑negative floats whose sum is positive, which guarantees a finite
value from `evaluate_sequence`.
"""

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow/underflow
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

def _generate_candidates(n):
    """Return a list of normalised vectors for the given length."""
    cand = []

    # very basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))               # arcsine

    # dense grid of symmetric Betas – extra small p values added
    p_grid = np.concatenate((
        np.logspace(-4, -0.5, 16),                # 1e‑4 … 0.316
        np.logspace(-0.5, -0.2, 12),              # 0.316 … 0.63
        np.linspace(0.1, 2.0, 24)                 # 0.1 … 2
    ))
    for p in p_grid:
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures of Beta(p,p) with uniform – finer w‑grid
    for p in (0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 0.9):
        for w in (0.1, 0.25, 0.5, 0.75, 0.9):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # random Dirichlet draws – a few concentration levels
    for alpha in (0.01, 0.05, 0.2, 0.8, 2.0):
        for _ in range(3):
            cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start – very few non‑zeros
    k = max(2, n // 500)
    cand.append(_distribution_sparse_random(n, k))

    # add a shuffled copy of a few bases to break symmetry
    for base in (cand[0], cand[2], cand[-3]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Linear‑programming improvement ----------------------------------------
def _build_constraint_matrix(a, active_idx):
    """Build A_ub where each row corresponds to (a * g)[k] ≤ M."""
    n = a.shape[0]
    K = len(active_idx)
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i                      # positions of g that multiply a[i]
        A[row, j] = a[i]
    return A


def _lp_maximize_mass(a, M, active_idx):
    """
    Solve  max 1ᵀg   subject to g ≥ 0 and (a∗g)[k] ≤ M   for k∈active_idx.
    Returns the optimal g (may be None on failure).
    """
    n = a.shape[0]
    A_ub = _build_constraint_matrix(a, active_idx)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  → minimise -sum(g)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:          # pragma: no cover
        return None
    if not res.success:
        return None
    return res.x


def _lp_iterative_optimize(a0, deadline, mix_frac=0.5):
    """
    Repeatedly improve a vector by the active‑set LP described in the
    problem statement.  Stops when time runs out or no improvement is
