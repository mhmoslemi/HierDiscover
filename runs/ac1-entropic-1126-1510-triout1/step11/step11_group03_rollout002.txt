"""
Hybrid optimizer for the step‑function attack.

The algorithm builds a diverse pool of analytic and random seeds,
keeps the best few, and then refines them with an Adam optimizer
working on a soft‑max approximation of the “max’’ in the objective.
A temperature (α) is increased from a modest start to a large value,
so the smooth objective converges to the true one.
After the Adam phase a cheap greedy mass‑transfer walk is applied.

Key ideas compared to the previous code:
* Adam (adaptive moments) instead of a simple decreasing step size.
* Soft‑max temperature schedule up to α≈80 for a tighter max approximation.
* A richer seed pool (extra Dirichlet draws, varied block fractions, etc.).
* Simpler projection: logits → softmax → simplex (no explicit Euclidean proj).
* Time is split between seed search, Adam refinement and final polishing.
"""

import time

# ------------------- Utility functions ---------------------------------

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic and random seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(6):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10, 0.20, 0.30):
        cand.append(_distribution_two_blocks(n, block_frac))

    # Dirichlet draws with varying concentration
    for _ in range(8):
        alpha = 10 ** rng.uniform(-3, 0)  # from very sparse to near‑uniform
        cand.append(_distribution_dirichlet(n, rng, alpha=alpha))

    cand.append(_distribution_random_sparse(n, rng, density=0.02))

    return cand


# ------------------- Adam optimisation --------------------------------

def _adam_optimize(a0: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float,
                   max_iter: int = 30000,
                   eval_interval: int = 400,
                   alpha_start: float = 0.5,
                   alpha_end: float = 80.0,
                   lr: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    Internally works with logits (log‑probabilities) to keep iterates in the simplex.
    """
    n = a0.shape[0]
    # initialise logits (logits = log a)
    eps = 1e-30
    x = np.log(np.maximum(a0, eps))

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8

    for t in range(1, max_iter + 1):
        # time guard
        if time.time() >= deadline:
            break

        # current point on simplex
        a = np.exp(x)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # compute convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter
        # stable soft‑max
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient of smoothed objective w.r.t. a
        # O_s = 2·n·logsumexp(alpha·conv)/alpha
        # grad_a = 2·n·_gradient_from_weights(a, w)
        grad_a = 2.0 * n * _gradient_from_weights(a, w)

        # convert to gradient w.r.t. logits x using chain rule:
        # ∂O/∂x_i = ∂O/∂a_i * ∂a_i/∂x_i
        # a_i = softmax(x)_i → ∂a_i/∂x_j = a_i (δ_ij - a_j)
        # Hence grad_x = a * (grad_a - <a, grad_a>)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        x -= (lr / (np.sqrt(v_hat) + epsilon)) * m_hat

        # occasional exact evaluation of objective
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ------------------- Greedy local swap -------------------------------

def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple greedy mass‑transfer walk that accepts improvements.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick source with non‑negligible mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(n)
        if j == i:
            continue
        # move a random fraction (up to 30 %) of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score
