# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow/underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform (both already normalised)."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix

def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec

def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

def _distribution_triangular(n):
    """Symmetric triangular distribution, peak in the middle."""
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        tri = np.concatenate([half, half[::-1]])
    else:
        tri = np.concatenate([half, np.array([half[-1] + 1]), half[::-1]])
    tri /= tri.sum()
    return tri

def _distribution_semicircle(n):
    """Discretised Wigner semicircle (density ∝ sqrt(1 - (2x-1)^2))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sqrt(np.maximum(0.0, 1.0 - (2 * x - 1) ** 2))
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_sine_power(n, power):
    """a_i ∝ sin(π·(i+0.5)/n) ** power, symmetric."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(np.pi * x) ** power
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

    for power in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_sine_power(n, power))

    # random Dirichlet draws (various concentrations)
    for alpha in (0.02, 0.05, 0.2, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start (few non‑zeros)
    k = max(2, n // 500)
    cand.append(_distribution_sparse_random(n, k))

    # random shuffles of a few bases to introduce asymmetry
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

# Smooth‑max mirror descent ---------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """
    Exponentiated‑gradient (mirror) descent on a log‑sum‑exp surrogate.
    α is annealed from 1 to 1e6.  Returns best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # schedule: (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e3, 3000, 0.14, 0.014),
        (1e3, 1e5, 3000, 0.014, 0.0014),
        (1e5, 1e6, 5000, 0.0014, 0.00014),
    ]

    for a0_alpha, a1_alpha, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # soft‑max weights (numerically stable)
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed max (same formula as sub‑gradient)
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # multiplicative update stays on the simplex automatically
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.8f}")

    return best_a, best_sc

# Projected sub‑gradient descent (improved) -----------------------------
def _projected_subgradient_descent(a0, deadline):
    """
    Classic sub‑gradient descent with back‑tracking line‑search that
    guarantees improvement over the *current* objective (not the global
    best).  Uses plain additive updates and projection onto the simplex.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(a)
    cur_sc = best_sc

    # parameters for the line‑search
    step0 = 0.7
    min_step = 1e-9
    max_iter = 10 ** 7  # safety upper bound
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = _fft_self_convolve(a)
        cur_sc = conv.max()
        # active constraints (within tiny tolerance)
        tol = 1e-12 * (abs(cur_sc) + 1.0)
        active = (conv >= cur_sc - tol).astype(np.float64)

        # sub‑gradient of the true max
        grad = 2.0 * _fft_convolve(a[::-1], active)[n - 1 : n - 1 + n]

        # back‑tracking line‑search aiming for a decrease over cur_sc
        step = step0
        while True:
            a_candidate = a - step * grad
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < cur_sc - 1e-14 or step < min_step:
                break
            step *= 0.5

        a = a_candidate
        cur_sc = cand_sc
        if cur_sc < best_sc - 1e-14:
            best_sc = cur_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_sc:.8f}")

        # occasional jitter to escape tiny plateaus
        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        # stop early if we are very close to the deadline
        if time.time() + 0.4 > deadline:
            break

    return best_a, best_sc

# Hill‑climber + aggressive pruning --------------------------------------
def _hill_climb_prune(a0, deadline):
    """Simple hill‑climber with random mass transfers and periodic pruning."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- move mass away from a pair that attains the current max ----
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair (i,j) with maximal product among those that give k
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j
        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-14:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.8f}")

        # ---- aggressive pruning (keep a large fraction of the biggest entries) ----
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-14:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.8f}")

        # ---- occasional random jitter ----
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []   # list of (score, vector)

    # ------------------------------------------------------------------
    # 1. Broad exploration over a few lengths
    lengths = [512, 1024, 2048, 4096, 8192]
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates(n)

        # add up‑scaled versions of the globally best solution (if any)
        if best_seq is not None:
            factor = n // best_seq_len if (best_seq_len := len(best_seq)) and n % best_seq_len == 0 else None
            if factor:
                upscaled = np.repeat(best_seq, factor)
                upscaled = np.maximum(upscaled, eps)
                upscaled /= upscaled.sum()
                candidates.append(upscaled)

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.8f}")

            # maintain a compact elite list (size ≤ 8)
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Pull in any historic solution (height_sequence_1) if present
    try:
        prev = np.asarray(globals().get('height_sequence_1'), dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.8f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Ensure we have a fallback if everything else failed
    if best_seq is None:
        n0 = 2048
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.8f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(6, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.8f}")

        # a) smooth‑max mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # b) projected sub‑gradient descent (improved)
        a_mid, sc_mid = _projected_subgradient_descent(a_mid, deadline)

        # c) final hill‑climber + pruning
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-14:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.8f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity, normalise, report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.8f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
