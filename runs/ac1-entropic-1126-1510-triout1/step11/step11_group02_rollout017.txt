   t = 0; alpha = alpha0
   while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w /= w_sum
        grad = _gradient_from_weights(a, w)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth Adam it={t} α={alpha:.2f} new best {best_score:.6f}")
   return best_seq, best_score

"""
propose_candidate – a fast hybrid optimiser for the min‑max convolution problem.

The objective to minimise is

    2·n·max(conv(a,a)) / sum(a)^2

for a non‑negative step‑function a (a list of floats).  The objective is
scale‑invariant, so we restrict ourselves to probability vectors
(sum‑to‑1).  The code combines several complementary ideas:

1.  A rich set of analytic initialisations (uniform, arcsine, beta,
    Dirichlet, random sparse and a *square‑root‑uniform* construction
    obtained via an FFT de‑convolution).

2.  A smooth‑max descent with an Adam optimiser.  The smooth‑max
    approximates the true max, a temperature α starts small and is
    increased geometrically, so the optimiser first sees a smooth
    landscape and then gradually focuses on the true maximum.

3.  A classic multimax sub‑gradient descent (exponentiated‑gradient) that
    directly optimises the non‑smooth max after the smooth phase.

4.  An optional linear‑programming “mix‑step”.  For a fixed a the
    programme

        maximise   Σ b_j
        subject to   a * b ≤ M   (elementwise)
                    b ≥ 0

    (where M is the current maximal convolution value) is a linear
    program in the variables b.  Solving it yields a direction that
    strictly reduces the maximal convolution; the new point is taken as
    a convex combination a←(1−t)·a+t·b̂.

5.  A cheap greedy pairwise‑swap local optimiser that moves a small
    amount of mass from one position to another only when it improves
    the exact objective.

The optimiser runs for up to the allotted budget (default 1000 s) and
returns the best sequence discovered as a plain Python list.
All helper functions are top‑level, there are no closures, and the code
uses only NumPy, SciPy and the supplied `evaluate_sequence` function.
"""

import time
import math
import numpy as np
from typing import List, Tuple
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k(a,a)  w.r.t. a.
    Implements 2·a ⋆ w   (⋆ = cross‑correlation) via FFT.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (assumes sum>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families – all return probability vectors (sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_sqrt_uniform(n: int) -> np.ndarray:
    """
    Approximate a probability vector whose self‑convolution is close to uniform.
    We take the square‑root of the FFT of a uniform distribution, inverse‑FFT,
    clip negatives and renormalise.
    """
    m = 1 << ((2 * n - 1) - 1).bit_length()
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
    P = np.fft.rfft(target, n=m)
    A = np.sqrt(P)                      # complex square‑root
    a_full = np.fft.irfft(A, n=m)
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _smooth_max_adam(a: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     lr0: float = 0.5,
                     beta1: float = 0.9,
                     beta2: float = 0.999,
                     eps: float = 1e-8,
                     alpha0: float = 0.5,
                     alpha_max: float = 80.0,
                     max_iter: int = 40000,
                     eval_interval: int = 250) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate
        f_α(a) = (1/α)·log Σ_k exp(α·conv_k(a)).
    The temperature α grows geometrically; after every `eval_interval`
    iterations the exact objective is evaluated and the global best is
    updated if an improvement is found.
    """
    n = a.shape[0]
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0
    alpha = alpha0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step                     # additive update

        # keep a on the simplex (non‑negative, sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        # increase temperature gradually
        if t % 500 == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.2, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth‑Adam it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.7,
                                 max_iter: int = 30000,
                                 eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Classic exponentiated‑gradient descent where the gradient is the
    average over the *active* (maximal) convolution entries.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)      # multiplicative (mirror) update
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk: move a small amount from i to j if it improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _lp_mix_and_update(a: np.ndarray,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float,
                       mix_fraction: float = 0.2) -> Tuple[np.ndarray, float]:
    """
    Solve the linear programme

        maximise   Σ b_j
        s.t.       a * b ≤ M      (elementwise)
                   b ≥ 0

    where M = max(conv(a,a)).  The optimiser returns a new direction b̂,
    which is mixed with the current a.
    """
    n = a.shape[0]

    # LP becomes costly for very large n – only attempt for n ≤ 2000
    if n > 2000 or time.time() >= deadline - 1.0:
        return a, best_score

    conv = _fft_self_convolve(a)
    M = conv.max()

    # Build the (2n‑1) × n constraint matrix for a * b ≤ M.
    # Entry (i+j, j) = a_i   (i from 0..n‑1, j from 0..n‑1)
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)

    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n)                     # maximise Σ b_j  → minimise -Σ b_j
    bounds = [(0, None)] * n

    # Limit the LP solve time to the remaining budget (highs supports time_limit)
    remaining = max(0.0, deadline - time.time())
    lp_opts = {"presolve": True, "time_limit": remaining}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method="highs", options=lp_opts)

    if not res.success or res.x is None:
        return a, best_score

    b = res.x
    sb = b.sum()
    if sb == 0.0:
        return a, best_score

    g = b / sb                         # normalise to a probability vector
    new_a = (1.0 - mix_fraction) * a + mix_fraction * g
    new_a = _project_onto_simplex(new_a, s=1.0)

    cur = _objective(new_a)
    if cur < best_score - 1e-13:
        best_score = cur
        best_seq = new_a.copy()
        a = new_a
        print(f"[{time.time():.1f}] LP‑mix improved to {best_score:.6f}")

    return a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects a 1000 s budget and returns
    the best sequence (as a plain list of floats) found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a few lengths – larger n gives more freedom, but each
    #    iteration costs O(n log n).  We start with a moderate size
    #    and later try a larger one if time permits.
    lengths = [2000, 4000, 8000]

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Gather a diverse set of seed vectors for every length.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic analytic families
        candlist: List[np.ndarray] = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_sqrt_uniform(n),
        ]

        # symmetric beta family
        for p in (0.05, 0.2, 0.5, 0.8, 1.0, 2.0):
            candlist.append(_distribution_beta_sym(n, p))

        # Dirichlet draws
        for _ in range(3):
            alpha = 0.1 + 0.9 * rng.random()
            candlist.append(_distribution_dirichlet(n, rng, alpha=alpha))

        # random sparse patterns
        candlist.append(_distribution_random_sparse(n, rng, density=0.02))

        # evaluate each candidate, keep the best few
        for cand in candlist:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the most promising seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # ------------------------------------------------------------------
    # 3. Incorporate the previously submitted solution if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback if nothing was produced yet.
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main refinement loop – run the various phases on the most
    #    promising seeds.  The number of seeds is limited to keep the
    #    total runtime well under the budget.
    seeds = [overall_best_seq.copy()]
    seeds += [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:3]
              if not np.may_share_memory(seq, overall_best_seq)]

    for seed_seq in seeds:
        if time.time() >= deadline:
            break
        a = seed_seq.copy()

        # 5a. smooth‑max descent with Adam
        a, overall_best_score = _smooth_max_adam(
            a,
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            alpha0=0.5,
            alpha_max=100.0,
            max_iter=35000,
            eval_interval=300,
        )
        overall_best_seq = a.copy()

        # 5b. classic multimax sub‑gradient descent
        a, overall_best_score = _subgradient_multimax_descent(
            a,
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.6,
            max_iter=30000,
            eval_interval=250,
        )
        overall_best_seq = a.copy()

        # 5c. optional LP‑mix step (cheap for moderate n)
        a, overall_best_score = _lp_mix_and_update(
            a,
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            mix_fraction=0.2,
        )
        overall_best_seq = a.copy()

        # 5d. final greedy local polishing
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
