"""
Propose a step‑function (non‑negative vector) that minimizes
`evaluate_sequence`.  The algorithm consists of three stages:

1️⃣  **Broad exploration** – generate candidates from several analytic
    families (arcsine, symmetric Beta(p,p), asymmetric Beta(p,½),
    edge‑biased power‑law) for a few lengths (8k–32k).  The best K
    candidates (by the true max‑convolution score) are retained.

2️⃣  **Refinement** – each retained candidate is refined separately:
    * a long mirror‑descent on a smooth‑max approximation of the
      max‑convolution,
    * a projected Adam‑style sub‑gradient descent on the exact
      max‑convolution,
    * all updates are symmetrised (a_i = a_{n‑1‑i}) and projected onto the
      probability simplex.
    The schedules for temperature and learning‑rate are aggressive
    (temperature grows from ≈0.5 to 10⁵, lr decays from 0.5 to 10⁻³).

3️⃣  **Final polishing** – a tiny mass‑transfer hill‑climber performs
    many local moves that shift a small amount of probability mass from
    the pair responsible for the current peak towards the smallest entry.

The whole procedure respects a 1000 s wall‑time limit and returns the
best vector found (as a plain list of floats).  The vector is normalised
to sum = 1; `evaluate_sequence` is scale‑invariant, so this is sufficient.
"""

# Linear‑algebra / FFT utilities

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1,
    returns the exact value produced by `evaluate_sequence`
    (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry v[i] = v[n-1-i] (preserves the sum)."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Analytic candidate families  (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)  # stability
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Random Dirichlet draw with edge‑biased concentration.
    Concentration at index i is ((i+½)(n‑i‑½))^{-beta_exp} + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec

# Optimisation primitives (all keep the vector normalised and symmetric)
# ----------------------------------------------------------------------


def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float,
                            eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective  f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector and the current max(conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    # exponentiated‑gradient update on the simplex (KL‑mirror)
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    a_new = _symmetrize(a_new)
    a_new = _project_onto_simplex(a_new, 1.0)   # re‑project after symmetry
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    Projected Adam‑style sub‑gradient step on the exact max‑convolution.
    Returns the updated vector, the new velocity and the new maximal
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices achieving the max, within tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam moment update (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        a_cand = _symmetrize(a_cand)
        a_cand = _project_onto_simplex(a_cand, 1.0)

        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    new_max = _fft_self_convolve(a_cand).max()
    return a_cand, vel, new_max


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Tiny hill‑climber that repeatedly moves a small amount of probability
    mass from the index that contributes most to the current peak towards
    the smallest entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())                # location of the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # Find the pair (i,j) giving the largest product a_i * a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # Choose source (larger of the two) and global target (minimum entry)
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]   # second‑smallest

        # Move a tiny amount
        max_delta = min(a[source] * 0.4, 0.03)   # allow a bit larger moves
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        a2 = _symmetrize(a2)
        a2 = _project_onto_simplex(a2, 1.0)

        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional jitter to avoid getting stuck
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            a = _symmetrize(a)
            a = _project_onto_simplex(a, 1.0)
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:

    `evaluate_sequence`.  The function respects a hard time budget

    """

    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    eps = 1e-12

    # ------------------------------------------------------------------
    # 0️⃣  Load a previously discovered construction if present
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]

            prev /= prev.sum()
            best_seq = prev.copy()

    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration over several lengths and families
    # ------------------------------------------------------------------
    lengths = [8000, 16000, 32000]   # moderate sizes keep evaluation cheap

    # asymmetric Beta – keep q fixed at 0.5
    asym_q = 0.5
    # edge‑biased power‑law exponents – extend beyond 0.5
    alpha_grid = np.linspace(0.5, 1.2, num=25)

    K = 7                     # keep the best K globally
    top_candidates = []        # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families -----------------------------------
        # arcsine (Beta(½,½) analogue)
        vec = _distribution_arcsine(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] n={n} arcsine score={sc:.6f}")
        top_candidates.append((sc, vec.copy()))

        # symmetric Beta(p,p)
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.4f}) score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

        # asymmetric Beta(p,½) and Beta(½,p)
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_beta(n, p, asym_q)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.4f},q={asym_q}) score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

            vec = _distribution_beta(n, asym_q, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={asym_q},q={p:.4f}) score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

        # edge‑biased deterministic power‑law
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_power(n, alpha)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edgePower(alpha={alpha:.3f}) score={sc:.6f}")
            top_candidates.append((sc, vec.copy()))

        for _ in range(5):

            vec = _distribution_edge_dirichlet(n, bex)

            sc = _max_convolution_score(vec)

    def _refine_candidate(a_init: np.ndarray) -> np.ndarray:
        a = a_init.copy()
        n = a.shape[0]

        # Schedule: temperature grows exponentially, lr decays exponentially

        vel = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        step0 = 0.6

            a, vel, cur_max = _adam_subgrad_step(a, vel, step0,

            if cur_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] Adam it={sub_it} new best {cur_score:.6f}")

            if sub_it % 700 == 0:
                # Inject a tiny jitter to escape shallow plateaus
                jitter = np.random.randn(n) * 1e-5
                a = np.maximum(a + jitter, eps)
                a /= a.sum()
                a = _symmetrize(a)
                a = _project_onto_simplex(a, 1.0)

        # ---------- final tiny hill‑climber ----------
        a = _mass_transfer_hill_climb(a, max_iters=6000, eps=eps)
        final_score = _max_convolution_score(a)
        if final_score < best_score:
            best_seq, best_score = a.copy(), final_score
            print(f"[{time.time():.1f}] Hill‑climb final improved score {final_score:.6f}")

        return a

    # Run refinement on each of the top candidates (in order of current quality)
    for rank, (sc, vec) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Refining candidate #{rank+1} (score={sc:.6f})")
        _refine_candidate(vec)   # updates global best_seq / best_score internally

    # ------------------------------------------------------------------
    # 3️⃣  Tiny final polishing on the globally best sequence (if time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        a = best_seq.copy()
        a = _mass_transfer_hill_climb(a, max_iters=8000, eps=eps)
        final_score = _max_convolution_score(a)
        if final_score < best_score:
            best_seq, best_score = a.copy(), final_score
            print(f"[{time.time():.1f}] Global hill‑climb improved score to {final_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    # Convert to Python list of floats
    return best_seq.tolist()
