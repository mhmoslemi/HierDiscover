"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

Algorithm overview
------------------

1. **Broad exploration** – generate many analytic families (uniform, Beta,
   mixtures, exponential, Dirichlet draws, sparse seeds) for a range of
   lengths (powers of two up to 65536).  The best few candidates are kept
   as an *elite* set.

2. **Multi‑scale refinement** – each elite vector is repeatedly refined at
   its current resolution and, when time permits, up‑sampled to the next
   resolution and refined again.  Refinement consists of three stages:

   * *Smooth‑max mirror descent* – an exponentiated‑gradient method on a
     log‑sum‑exp surrogate of the max‑convolution.  The temperature α is
     ramped from 1 to 1e⁸ while the learning rate decays, giving many
     exploratory and many fine‑tuning steps.

   * *Projected sub‑gradient descent* – true sub‑gradient descent on the
     exact max‑convolution (active set of binding shifts) with Adam‑style
     momentum and a back‑tracking line‑search.

   * *Hill‑climbing & pruning* – cheap stochastic moves that transfer a
     tiny amount of mass away from the pair responsible for the current
     maximum, together with aggressive pruning of negligible entries.

3. **Time‑aware loop** – every heavy loop checks the global deadline
   (1000 s minus a small safety margin) and aborts early if required.

The code works entirely on the simplex (vectors sum to 1); the scaling
performed by `evaluate_sequence` is therefore irrelevant.  The final
vector is returned as a plain Python list of floats.
"""

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow/underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """Convex mixture w·Beta(p,p) + (1‑w)·Uniform."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_exponential(n, rate):
    """Monotonically decreasing exponential shape, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_sparse_random(n, k):
    """k random non‑zero entries, normalised."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a


def _upsample(a, new_n):
    """Linear interpolation from a (len m) to length new_n (new_n > m)."""
    m = a.shape[0]
    if new_n == m:
        return a.copy()
    x_old = (np.arange(m, dtype=np.float64) + 0.5) / m
    x_new = (np.arange(new_n, dtype=np.float64) + 0.5) / new_n
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    b /= b.sum()
    return b

# Local optimisation ----------------------------------------------------
def _smooth_max_mirror_descent(a0, deadline):
    """
    Mirror (exponentiated‑gradient) descent on the log‑sum‑exp surrogate.
    The temperature α is increased from 1 to 1e8 while the learning rate
    decays, providing many exploratory steps followed by fine‑tuning.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # schedule: (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e3, 4000, 0.12, 0.012),
        (1e3, 1e5, 5000, 0.012, 0.0012),
        (1e5, 1e8, 6000, 0.0012, 0.00012),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # soft‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smooth max wrt a
            # derivative of (a*a)_k w.r.t. a_j is a_{k-j} + a_{j-k}
            # the compact form is 2 * (a reversed) * w (convolution)
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient update
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _subgradient_descent(a0, deadline):
    """
    Projected sub‑gradient descent on the true max‑convolution objective,
    using Adam‑style momentum and an adaptive back‑tracking line search.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (allow a tiny tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # simple back‑tracking line search
        step = step0
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _max_convolution_score(cand)

            if cand_sc < best_sc - 1e-14 or step < 1e-9:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-14:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        # occasional random jitter to escape plateaus
        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.4 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a0, deadline):
    """
    Light‑weight hill‑climbing with aggressive pruning of tiny entries.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # move mass away from the pair that creates the current maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # locate the pair (i,j) with maximal product among those indices
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j

        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-14:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # aggressive pruning every few thousand steps
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-14:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned

        # random jitter occasionally
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc


def _refine_candidate(a, deadline):
    """
    Run the three‑stage refinement pipeline on a single candidate.
    Returns the best vector found together with its score.
    """
    a_mid, sc_mid = _smooth_max_mirror_descent(a, deadline)
    a_mid, sc_mid = _subgradient_descent(a_mid, deadline)
    a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)
    return a_mid, sc_mid

def _generate_candidates(n):
    """Return a list of normalised vectors (sum = 1) for the given length."""
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))   # arcsine

    # dense grid of symmetric Beta(p,p)
    for p in np.concatenate(
        (np.logspace(-3, -0.2, 14), np.linspace(0.1, 2.0, 22))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with the uniform distribution
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.1, 0.3, 0.5, 0.7, 0.9):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # simple exponential families (left‑skewed)
    for rate in (5.0, 10.0, 20.0, 40.0):
        cand.append(_distribution_exponential(n, rate))
        cand.append(_distribution_exponential(n, rate)[::-1])   # mirrored

    # several Dirichlet draws
    for alpha in (0.02, 0.05, 0.2, 0.8, 2.0):
        for _ in range(3):
            cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start (few non‑zeros)
    k = max(2, n // 500)
    cand.append(_distribution_sparse_random(n, k))

    # shuffled versions to create asymmetry
    base = cand[0]   # uniform
    sh = base.copy()
    np.random.shuffle(sh)
    cand.append(sh)

    return cand
