"""
Improved search for a non‑negative step‑function minimising
`evaluate_sequence`.  The algorithm explores a rich family of analytic
distributions, refines the most promising seeds with a soft‑max
gradient descent (full gradient obtained by FFT), then up‑samples the
best candidate and continues the refinement.  Finally a cheap greedy
pairwise mass‑transfer polish is applied.

The core idea of the gradient step:
    * For a normalized vector a (Σa = 1) compute the full self‑convolution
      c = a * a (length 2n‑1).
    * Form a smooth approximation of the max by a soft‑max with a
      temperature β: w_k = exp(β(c_k – max(c))) / Σ exp(·).
    * The gradient of Σ w_k c_k w.r.t. a is 2·(a ⊛ w_rev) evaluated at
      indices (2n‑2‑i), where w_rev is w reversed.  This is computed by
      FFT, yielding an O(n log n) update.
    * An exponentiated‑gradient (multiplicative) update keeps a on the
      simplex.  β is increased slowly so that the optimisation first
      smooths the landscape and later focuses on the true maximum.

The whole routine respects a 1000 s budget, prints progress, and
returns a plain Python list ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Beta(p,p) density on a regular grid, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture base·(1‑lam) + uniform·lam."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Gradient computation utilities ----------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution vector (numerically stable)."""
    cmax = conv.max()
    exp_vals = np.exp(beta * (conv - cmax))
    return exp_vals / exp_vals.sum()


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Full gradient of Σ_k w_k (a*a)_k w.r.t. a.
    Implements  g_i = 2· Σ_j w_{i+j} a_j   using FFT.
    """
    n = a.shape[0]
    w_rev = w[::-1]                     # length 2n‑1
    size = 1 << ((n + (2 * n - 1) - 1).bit_length())  # >= 3n‑2
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_full = np.fft.irfft(A * W, n=size)
    idx = (2 * n - 2) - np.arange(n)    # positions of interest
    grad = 2.0 * conv_full[idx]
    return grad


def _full_gradient_descent(
    a: np.ndarray,
    deadline: float,
    global_best_seq: np.ndarray,
    global_best_score: float,
    eta0: float = 0.8,
    beta0: float = 5.0,
    beta_factor: float = 1.02,
    eval_interval: int = 150,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth approximation of the max‑convolution.
    Uses the full gradient obtained from the soft‑max weights.
    """
    n = a.shape[0]
    t = 0
    beta = beta0
    best_local_seq = a.copy()
    best_local_score = _objective(a)

    while time.time() < deadline and t < max_iter:
        t += 1

        # convolution and soft‑max weights
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = _softmax_weights(conv, beta)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = 2.0 * n * maxc
            if cur < global_best_score - 1e-14:
                global_best_score = cur
                global_best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] GD it={t} new GLOBAL best {global_best_score:.6f}"
                )
            if cur < best_local_score - 1e-14:
                best_local_score = cur
                best_local_seq = a.copy()

        # gradient
        grad = _gradient_from_weights(a, w)

        # exponentiated‑gradient update
        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # raise temperature slowly
        beta = min(beta * beta_factor, 1e6)

    # final exact check
    final_score = _objective(a)
    if final_score < global_best_score - 1e-14:
        global_best_score = final_score
        global_best_seq = a.copy()
        print(f"[{time.time():.1f}] GD final improvement to {global_best_score:.6f}")

    if final_score < best_local_score - 1e-14:
        best_local_score = final_score
        best_local_seq = a.copy()

    return best_local_seq, best_local_score, global_best_seq, global_best_score

# Greedy pairwise mass transfer -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly relocate a fraction of mass; keep the move only when it improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.4  # up to 40 % of source mass
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g}  new best {best_score:.6f}"
            )
    return best_seq, best_score

        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            candidate_pool.append(prev)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"  previous best improved score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3️⃣  Keep only the most promising seeds (≈30).
    # ------------------------------------------------------------------
    # evaluate once more to guard against any stray NaNs
    cand_scores = [(_objective(c), c) for c in candidate_pool]
    cand_scores.sort(key=lambda x: x[0])
    top_seeds = [seq for _, seq in cand_scores[:30]]
    print(f"[{time.time():.1f}] selected {len(top_seeds)} top seeds for refinement")

    # ------------------------------------------------------------------
    # 4️⃣  Full‑gradient descent on the top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0 and top_seeds:
        per_seed = remaining / max(1, len(top_seeds))
        print(f"[{time.time():.1f}] entering gradient ascent, {per_seed:.1f}s per seed")
        for idx, seed in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt, overall_best_seq, overall_best_score = _full_gradient_descent(
                seed.copy(),
                sub_deadline,
                overall_best_seq,
                overall_best_score,
                eta0=0.9,
                beta0=5.0,
                beta_factor=1.015,
                eval_interval=100,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after GD seed {idx} new GLOBAL best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5️⃣  Upscale the current best and continue refinement (higher resolution).
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline:
        for factor in (2, 4):
            if time.time() >= deadline:
                break
            n = overall_best_seq.shape[0]
            N = n * factor
            # linear interpolation onto a finer grid
            x_old = (np.arange(n, dtype=np.float64) + 0.5) / n
            x_new = (np.arange(N, dtype=np.float64) + 0.5) / N
            a_up = np.interp(x_new, x_old, overall_best_seq)
            a_up = np.maximum(a_up, 0.0)
            a_up /= a_up.sum()
            sub_deadline = time.time() + (deadline - time.time()) * 0.4
            print(f"[{time.time():.1f}] upscaling to {N} points, starting GD")
            a_opt, sc_opt, overall_best_seq, overall_best_score = _full_gradient_descent(
                a_up,
                sub_deadline,
                overall_best_seq,
                overall_best_score,
                eta0=0.8,
                beta0=10.0,
                beta_factor=1.02,
                eval_interval=150,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after up‑scaled GD new GLOBAL best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 6️⃣  Final greedy local polishing.
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline:
        print(f"[{time.time():.1f}] final pairwise‑swap polishing")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
            max_trials=None,
        )

    # ------------------------------------------------------------------
    # 7️⃣  Return the result.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # Fallback – uniform on a modest size
        overall_best_seq = _distribution_uniform(5000)
        overall_best_score = _objective(overall_best_seq)
        print(
            f"[{time.time():.1f}] fallback uniform, score={overall_best_score:.6f}"
        )

    print(
        f"[{time.time():.1f}] FINAL best score {overall_best_score:.6f} (len={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
