def smooth_descent(a, deadline):
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)
    # parameters
    total_iter = 200000
    start_alpha = 1.0
    end_alpha = 1e12
    start_lr = 1.0
    end_lr = 1e-4
    eps = 1e-12
    for it in range(total_iter):
        if time.time() >= deadline:
            break
        t = it / total_iter
        alpha = start_alpha * (end_alpha / start_alpha) ** t
        lr = start_lr * (end_lr / start_lr) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        w = w / w_sum

        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1 : n-1 + n]

        a = a * np.exp(-lr * grad)
        # projection onto simplex
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        if it % 5000 == 0:
            cur_score = _objective(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.2f}] iter {it} new best {cur_score:.6f}")
    return best, best_score

def lp_refine(a, deadline):
    import cvxpy as cp, scipy.linalg as la
    n = a.shape[0]
    c = np.concatenate([a, np.zeros(n-1)])
    r = np.concatenate([a[:1], np.zeros(n-1)])
    A = la.toeplitz(c, r)  # shape (2n-1, n)
    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [A @ g <= t,
                   cp.sum(g) == 1,
                   g >= 0]
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9)
    if g.value is None:
        return a, _objective(a)
    g_val = np.maximum(g.value, 0)
    g_val = g_val / g_val.sum()
    # compute convex combination maybe: new_a = (a + g_val)/2?

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.

Algorithm
---------
* Choose a few promising lengths (≈ 3000) and initialise each with a set of
  analytic families (arcsine, symmetric Beta, uniform, random Dirichlet).
* For every start vector run a **large‑scale exponentiated‑gradient mirror
  descent** on the smooth‑max of the self‑convolution.  The temperature
  α is raised from 1 to 1e12 while the learning‑rate is decayed – this
  yields a very fine approximation of the true max without ever evaluating
  the full sub‑gradient.
* After the smoothed phase a short **Adam‑style projected sub‑gradient**
  polishing is performed on the exact max‑convolution.
* When time permits a single **LP‑refinement** is carried out:
  minimise the maximal convolution of the current sequence with a new
  sequence g (linear programme).  The resulting g is mixed with the
  current best to obtain a final improvement.
* The best sequence found over all lengths / restarts is returned.
"""

import time
import random
import math
import numpy as np
import scipy.linalg

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Returns the value of evaluate_sequence for a vector already
    normalised to sum = 1 (the scaling of the original problem is
    homogeneous, so fixing the sum is safe).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Mirror‑descent on the smooth max of the convolution
# ----------------------------------------------------------------------
def _smooth_max_mirror_descent(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient mirror descent on the smooth‑max of the
    self‑convolution.  The temperature α is increased up to 1e12,
    the learning‑rate is decreased to 1e‑4.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    # schedule parameters (chosen to fill the available time)
    total_iters = 200_000          # large enough for good convergence
    alpha_start = 1.0
    alpha_end = 1e12
    lr_start = 1.0
    lr_end = 1e-4

    best_a = a.copy()
    best_sc = _objective(a)

    for it in range(total_iters):
        if time.time() >= deadline:
            break

        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient update (mirror descent)
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)       # avoid numerical zero
        a /= a.sum()

        # occasional bookkeeping
        if it % 5_000 == 0 or it == total_iters - 1:
            sc = _objective(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.2f}] MD it={it} new best {sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing (Adam‑style)
# ----------------------------------------------------------------------
def _projected_subgradient(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    True sub‑gradient descent with Adam momentum on the exact
    max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    epsilon_adam = 1e-8
    base_lr = 0.4

    best_a = a.copy()
    best_sc = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set where the max is attained (with small tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr / (np.sqrt(v_hat) + epsilon_adam)
        cand = a - step * m_hat
        cand = np.maximum(cand, eps)
        cand = _project_onto_simplex(cand, 1.0)

        cand_sc = _objective(cand)
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = cand.copy()
            print(f"[{time.time():.2f}] Subgrad it={it} new best {cand_sc:.6f}")

        a = cand

        # occasional random jitter to escape tiny plateaux
        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = _project_onto_simplex(a, 1.0)

        # safety break
        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Linear‑programming refinement (optional – uses cvxpy if available)
# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Solve
        minimise   t
        subject to A @ g ≤ t,
                   sum(g) = 1,
                   g ≥ 0
    where A encodes the convolution with the current a.
    Returns a convex combination (0.5,0.5) of a and the LP solution.
    """
    try:
        import cvxpy as cp
    except Exception:
        # cvxpy not present – skip refinement
        return a, _objective(a)

    n = a.shape[0]

    # Build convolution matrix A (size (2n‑1) × n)
    # First column: a then zeros, first row: a[0] then zeros
    c = np.concatenate([a, np.zeros(n - 1)])
    r = np.concatenate([a[:1], np.zeros(n - 1)])
    A = scipy.linalg.toeplitz(c, r)  # shape (2n‑1, n)

    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        A @ g <= t,
        cp.sum(g) == 1,
        g >= 0,
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    # warm‑start + modest iteration limit keeps this fast
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, abstol=1e-9, reltol=1e-9)

    if g.value is None:
        return a, _objective(a)

    g_val = np.maximum(g.value, 0.0)
    g_val /= g_val.sum()

    # Mix the current best with the LP solution (tuned weight 0.5)
    new_a = 0.5 * a + 0.5 * g_val
    new_a = _project_onto_simplex(new_a, 1.0)

    new_sc = _objective(new_a)
    return new_a, new_sc

    """

    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # small safety margin

    best_seq = None
    best_score = float("inf")

    # 1. Choose a handful of lengths – a compromise between quality and
    #    speed of the LP step.
    # ------------------------------------------------------------------

        # incorporate the globally‑available previous solution if it matches the size
        try:

            if prev.ndim == 1 and prev.size == n:

        except Exception:
            pass

        # ----------------------------------------------------------------

        # ----------------------------------------------------------------
        for vec in starters:
            vec = np.maximum(vec, 0.0)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.2f}] init n={n} new global best {sc:.6f}")

        # ----------------------------------------------------------------

        # ----------------------------------------------------------------

        # ----------------------------------------------------------------
        if time.time() < deadline - 1.0:
            a_lp, sc_lp = _lp_refine(a_sub, deadline)
            if sc_lp < best_score:
                best_score = sc_lp
                best_seq = a_lp.copy()
                print(f"[{time.time():.2f}] after LP n={n} new best {sc_lp:.6f}")
