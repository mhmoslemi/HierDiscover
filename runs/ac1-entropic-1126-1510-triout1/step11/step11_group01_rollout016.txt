"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  The search combines a very rich set of analytic
distributions, a deterministic optimisation based on a *full* sub‑gradient
(gradient taken from all active convolution entries), and a short
simulated‑annealing polishing phase.  All work is performed inside the
1000 s time budget, keeping the code pure‑Python/NumPy (optional SciPy
windows are used when available).

The objective used by ``evaluate_sequence`` is
    2·n·max(conv(a,a)) / (sum(a)**2)
and is homogeneous – we may normalise every candidate to have L1‑norm 1
and simply minimise 2·n·max(conv).  The optimisation therefore only the
shape of the distribution matters.

The algorithm proceeds in four stages:

1. **Candidate generation** – a large pool of analytic families (uniform,
   arcsine, symmetric Beta, edge‑weighted, Gaussian, sine‑bell, many
   window functions from SciPy) and random Dirichlet / sparse seeds.
2. **Keep the best few** – after normalising to sum 1, keep the ≈40
   smallest objective values.
3. **Exact sub‑gradient mirror descent** – for each elite seed run a
   mirror‑descent where the gradient is computed from *all* indices that
   attain the maximal convolution value (the true sub‑gradient of the
   max‑operator).  An exponentiated‑gradient update keeps the iterate on
   the simplex.  A decaying stepsize η_t = η₀ / sqrt(t) is used.
4. **Simulated‑annealing polishing** – a lightweight SA moves a random
   fraction of the mass from one position to another; moves are kept
   when they improve the true objective or with a Metropolis probability
   that slowly cools to zero.

The routine respects the 2‑second safety margin required by the judge

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = s}.
    """
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # next power‑of‑two ≥ 2·n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by the optimiser (same as evaluate_sequence for
    a vector with Σa = 1):
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * float(b.max())

# Analytic families ----------------------------------------------------
def _dist_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dist_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w /= w.sum()
    return w


def _dist_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _dist_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _dist_gaussian(n: int, sigma: float) -> np.ndarray:
    """
    Truncated Gaussian (discrete) centred at the middle.
    sigma is given relative to n (σ = sigma * n).
    """
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    w /= w.sum()
    return w


def _dist_sinebell(n: int, power: float) -> np.ndarray:
    """w_i = sin(π (i+½) / n)^power, normalised."""
    x = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * x / n) ** power
    w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mix lam·uniform + (1‑lam)·base."""
    n = base.shape[0]
    return lam * np.full(n, 1.0 / n) + (1.0 - lam) * base

def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Standard Dirichlet with random concentration."""
    alpha = rng.uniform(0.01, 2.0)
    a = rng.dirichlet(np.full(n, alpha))
    return a


def _random_sparse(n: int, rng: np.random.Generator, density: float = 0.01) -> np.ndarray:
    """Sparse random vector (controlled density)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

def _subgradient_descent_full(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    eta0: float = 0.6,
    eval_every: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the *exact* sub‑gradient of the max‑convolution.
    The gradient is the sum of all shifted copies of ``a`` that correspond
    to active (max‑attaining) convolution indices.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        M = conv.max()
        # indices where the maximum is attained (allow a tiny tolerance)
        max_idx = np.where(conv >= M - 1e-15)[0]

        # build gradient: grad[i] = 2 * Σ_{k∈max_idx} a[i‑k]  (if index inside)
        grad = np.zeros_like(a)
        for k in max_idx:
            # contribution of shift k
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            # i runs from start..end, we add a[i‑k] = a[i‑k+?] – just use slice trick
            src = a[start - k + n - 1 : end - k + n - 1]  # out‑of‑range slice yields empty, careful!
            # Simpler: use vectorised shift via np.take with mode='clip'
            # but a loop over k is cheap because max_idx is usually small (<5)
            for i in range(start, end + 1):
                grad[i] += 2.0 * a[i - k]

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # exponentiated‑gradient update
        a = _project_onto_simplex(a)         # back to the simplex

        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t:5d} new best {best_score:.6f}"
                )
    return best_seq, best_score

def _simulated_annealing(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    start_temp: float = 0.03,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Very light‑weight annealing that repeatedly moves a random fraction
    of mass from one index to another.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _objective(cur)
    T = start_temp
    trials = 0
    while time.time() < deadline:
        trials += 1
        # pick source with non‑zero mass
        src = np.random.choice(np.flatnonzero(cur))
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # move a random fraction (up to 30 %) of the source mass
        delta = cur[src] * np.random.rand() * 0.30
        if delta < 1e-15:
            continue
        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = _objective(new)
        if new_score < cur_score - 1e-14:
            cur, cur_score = new, new_score
            if new_score < best_score - 1e-14:
                best_score, best_seq = new_score, new.copy()
                print(
                    f"[{time.time():.1f}] SA accept improve i={src}->{dst} δ={delta:.3g} new best {best_score:.6f}"
                )
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                cur, cur_score = new, new_score

        T *= decay
        if T < 1e-8:
            T = start_temp

    print(f"[{time.time():.1f}] SA finished after {trials} trials, best={best_score:.6f}")
    return best_seq, best_score

    if remaining > 30.0 and elite:
        per_seed = remaining / len(elite)
        print(f"[{time.time():.1f}] launching sub‑gradient phase on {len(elite)} seeds")
        for i, (_, seed_vec) in enumerate(elite):
            if time.time() >= deadline:
                break
            sub_dead = time.time() + per_seed
            opt_seq, opt_score = _subgradient_descent_full(
                seed_vec.copy(),
                sub_dead,
                best_seq=best_seq,
                best_score=best_score,
                eta0=0.8,
                eval_every=100,
            )
            if opt_score < best_score - 1e-12:
                best_score, best_seq = opt_score, opt_seq.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {i} new global best {best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3.  Final simulated‑annealing polishing (few seconds).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealing polish")
        best_seq, best_score = _simulated_annealing(
            best_seq.copy(),
            deadline,
            best_seq=best_seq,
            best_score=best_score,
            start_temp=0.02,
            decay=0.9998,
        )

    # ------------------------------------------------------------------
    print(
        f"Finished. Best objective = {best_score:.6f} (length = {len(best_seq)})"
    )
    return best_seq.tolist()
