"""
Algorithm for minimizing `evaluate_sequence`.

* Stage 0 – quick scans over a few lengths (2‑k…8 k) and a rich set of
  parametric families (uniform, arcsine, Beta(p,p), edge‑biased, exponential).
  The Beta family is tuned with a cheap golden‑section search on the shape
  parameter.

* Stage 1 – intensive projected sub‑gradient descent on the true
  max‑convolution.  The active set of the current peak yields a sub‑gradient.
  A back‑tracking line‑search with a simple decay schedule is run until the
  time budget is exhausted.

* Stage 2 – stochastic annealing moves (transfer a small amount of mass from
  a large entry to a small entry) that are accepted with a Metropolis rule.
  This polishing step can escape shallow plateaus.

The whole routine respects the supplied `budget_s` (default 1000 s) and
returns a plain list of non‑negative floats.  All intermediate vectors are
kept normalised to sum = 1, which does not affect the value returned by
`evaluate_sequence`.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x):
    """smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _score_normalised(a):
    """Score for a vector already normalised to sum = 1:
    2·n·max(a*a). This equals evaluate_sequence for the same shape."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# Parametric families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0)
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    w /= w.sum()
    return w


def _distribution_edge_power(n, p):
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15
    w /= w.sum()
    return w


def _distribution_center_power(n, p):
    """Center‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** p
    w += 1e-15
    w /= w.sum()
    return w


def _distribution_exponential(n, lam):
    """Exponential decay from the centre."""
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * np.abs(idx - centre))
    w += 1e-15
    w /= w.sum()
    return w


def _distribution_dirichlet(n, conc):
    """Draw from a Dirichlet with constant concentration."""
    return np.random.dirichlet(np.full(n, conc))


def _distribution_edge_dirichlet(n, beta_exp):
    """Dirichlet biased towards the edges."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    return np.random.dirichlet(conc)

# Helper: Golden‑section search for a single‑parameter family
def _golden_section_optimize(func, a, b, max_iter=15, tol=1e-12):
    """Return x in [a,b] that minimises func(x) using golden‑section search."""
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
        if b - a < tol:
            break
    return (a + b) / 2.0

    `evaluate_sequence`).

    random.seed(seed)

    eps = 1e-15

    best_seq = None
    best_score = float("inf")
    best_n = None

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)

def _golden_section_optimize(func, a, b, max_iter=12, tol=1e-12):
    """
    Minimise `func` on [a,b] using golden‑section search.
    Returns an (approximately) optimal x.
    """
    phi = (np.sqrt(5) - 1) / 2          # 0.618...
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
        if b - a < tol:
            break
    return (a + b) / 2.0

        improved = False
        while step > 1e-8:
            a_cand = a - step * grad
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()
            cand_score = _score_normalised(a_cand)
            if cand_score < best_score - 1e-12:
                a = a_cand
                best_seq = a.copy()
                best_score = cand_score
                improved = True
                print(f"[{time.time():.1f}] subgrad step={step:.4g} new best {cand_score:.6f}")
                break
            step *= 0.5

        if not improved:
            # occasional tiny jitter to shake a possible plateau
            if random.random() < 0.03:
                jitter = np.random.randn(n) * 1e-6
                a = np.maximum(a + jitter, eps)
                a /= a.sum()

    # ------------------------------------------------------------------
    # 2️⃣  Stochastic annealing polish
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    total_iters = max(1, int((deadline - time.time()) / 0.0006))
    T0, Tf = 1e-3, 1e-8

    for it in range(1, total_iters + 1):
        # pick source (large) and target (small)
        src = np.argmax(a)
        tgt = np.argmin(a)
        if src == tgt:
            src, tgt = random.sample(range(n), 2)

        max_delta = min(a[src] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_score = _score_normalised(a2)

        # temperature schedule (log‑linear)
        t_frac = it / total_iters
        T = T0 * (Tf / T0) ** t_frac

        if new_score < cur_score or random.random() < math.exp(-(new_score - cur_score) / T):
            a = a2
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] anneal it={it} new best {cur_score:.6f}")

        # occasional jitter to keep the chain lively
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            cur_score = _score_normalised(a)

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
