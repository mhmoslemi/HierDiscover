    n = len(a0)
    eps = 1e-14
    # log parameters:
    z = np.log(np.maximum(a0, eps))
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    # scheduling alpha
    alpha0 = 1.0
    alpha_max = 1e6
    best_a = a0.copy()
    best_sc = _max_convolution_score(best_a)
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute softmax
        z_max = np.max(z)
        a = np.exp(z - z_max)
        a = a / a.sum()
        # compute conv and softmax weights
        conv = _fft_self_convolve(a)

   a Catalan‑type distribution (the discrete analogue of (1‑z)^‑½) and a
   power‑law family whose mass is concentrated near the ends.  All candidates
   are normalised to sum = 1.

2. **Fast evaluation** – the objective
      score = 2·n·max(a*a)       (a already sums to 1)
   is computed with FFT‐based self‑convolution in O(n log n).

3. **Local optimisation** – each elite candidate is refined with a
   full‑gradient Adam optimiser applied to a *soft‑max* surrogate of the
   max‑operator:
      L_α(a) = (1/α)·log∑_k exp(α·(a*a)[k]).
   The temperature α is increased from 1 to 1e6, turning the smooth surrogate
   into the true max.  The optimisation works in log‑space (softmax) so the
   simplex constraints are satisfied automatically.

4. **Final polishing** – a short hill‑climber that swaps tiny amounts of mass
   away from the indices contributing most to the current maximum.

The whole search is time‑aware: every heavy loop checks the deadline
(`budget_s‑5 s` safety margin).  The best sequence found before the deadline
is returned.
"""

import time
import random

# ----------------------------------------------------------------------
# Simplex utilities (not needed for log‑softmax optimisation but kept for safety)
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection onto {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Score 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return vectors with sum = 1) --------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_catalan(n):
    """Discrete Catalan coefficients c_i = binom(2i,i)/4**i, normalised."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def _distribution_powerlaw(n, alpha):
    """Symmetric power‑law: weight ∝ ((i+1)*(n‑i))⁻ᵅ  (i = 0…n‑1)."""
    i = np.arange(n, dtype=np.float64) + 1.0               # 1 … n
    prod = i * (n - i + 1.0)                               # (i)*(n‑i+1)
    w = prod ** (-alpha)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_mixture_powerlaw_uniform(n, alpha, w):
    pl = _distribution_powerlaw(n, alpha)
    uni = _distribution_uniform(n)
    mix = w * pl + (1.0 - w) * uni
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate ----------------------------
def _adam_optimize_from_a(a0, deadline, max_iter=20000, lr=0.2):
    """
    Optimises a0 (already normalised to sum = 1) using Adam on the
    smooth‑max surrogate L_α(a).  Returns (best_a, best_score).
    """
    eps = 1e-14
    n = a0.shape[0]
    # initialise log‑softmax parameters
    z = np.log(np.maximum(a0, eps))
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    best_a = a0.copy()
    best_sc = _max_convolution_score(best_a)

    alpha_start, alpha_end = 1.0, 1e6

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # ---------- softmax ----------
        z_max = np.max(z)
        a = np.exp(z - z_max)
        a = a / a.sum()

        # ---------- convolution ----------
        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # temperature schedule (exponential)
        t = it / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** t

        # ---------- smooth‑max weights ----------
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient w.r.t. a ----------
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ---------- chain rule for log‑softmax ----------
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        z = z - step

        # ---------- occasional true evaluation ----------
        if (it & 511) == 0:            # every 512 iterations
            a_test = np.exp(z - np.max(z))
            a_test = a_test / a_test.sum()
            sc = _max_convolution_score(a_test)
            if sc < best_sc:
                best_sc = sc
                best_a = a_test.copy()
                print(f"[{time.time():.1f}] Adam it={it} new best {sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple hill‑climber / support pruner (keeps the fast version from before)
def _hill_climb_prune(a0, deadline):
    """Transfer tiny mass away from the currently most overloaded pair."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # locate the pair (i,j) that actually yields the maximal product
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j

        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # occasional aggressive pruning of tiny entries
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # random jitter to escape plateaus
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Candidate generator ----------------------------------------------------
def _generate_candidates(n):
    """Return a list of normalised candidate vectors for length n."""
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))          # arcsine

    # grid of symmetric betas
    for p in np.concatenate(
        (np.logspace(-2, -0.2, 12), np.linspace(0.1, 2.0, 20))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with uniform
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.2, 0.5, 0.8):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # random Dirichlet draws
    for alpha in (0.05, 0.2, 0.8, 2.0):
        for _ in range(2):
            cand.append(_distribution_random_dirichlet(n, alpha))

    # Catalan‑type (discrete sqrt‑kernel) distribution
    cand.append(_distribution_catalan(n))

    # power‑law families
    for a in (0.2, 0.5, 1.0):
        cand.append(_distribution_powerlaw(n, a))
        for w in (0.3, 0.6):
            cand.append(_distribution_mixture_powerlaw_uniform(n, a, w))

    # a very sparse start (few non‑zeros)
    k = max(2, n // 400)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # shuffled versions of a few bases to introduce asymmetry
    for base in (cand[0], cand[2], cand[-3]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    # keep a safety gap (~5 s) for Python bookkeeping / output
    deadline = start + budget_s - 5.0

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []                     # list of (score, vector)

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths
    lengths = [1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)

            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")

            # maintain a short elite list (up to 6 elements)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate the historic solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the extremely unlikely case we have nothing
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from a small elite pool
    # Sort elite by score, keep the best few for intensive optimisation
    elite.sort(key=lambda x: x[0])
    restarts = min(4, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.6f}")

        # Adam optimisation with smooth‑max surrogate
        a_opt, sc_opt = _adam_optimize_from_a(a0, deadline, max_iter=20000, lr=0.2)

        # Final quick hill‑climber / pruning
        a_opt, sc_opt = _hill_climb_prune(a_opt, deadline)

        if sc_opt < best_score - 1e-12:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity, normalise and output
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
