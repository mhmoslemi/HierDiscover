def gradient_max_conv(a):
    conv = _fft_self_convolve(a)
    k = conv.argmax()
    n = len(a)
    g = np.zeros_like(a)
    # first term
    start = max(0, k - (n-1))
    end = min(n-1, k)
    # for i from start to end inclusive, conv term includes a_i * a_{k - i}
    # gradient for a_i includes a_{k - i}
    # gradient for a_{k - i} includes a_i
    # So we can iterate i:
    for i in range(start, end+1):
        j = k - i
        if i == j:
            g[i] += 2 * a[i]  # double because both terms equal
        else:
            g[i] += a[j]
            g[j] += a[i]
    return g

"""
Hybrid multi‑start optimiser for the self‑convolution max‑ratio problem.

The optimiser works in three stages:

1. **Seed generation** – a wide family of analytic and random
   distributions (uniform, arcsine, Beta, Dirichlet, exponential,
   power‑law, sparse, two‑block, sine) is created for several
   lengths.  All seeds are normalised to sum = 1 and their exact
   objective `2·n·max(conv(a,a))` (the denominator is 1 after normalising)
   is evaluated.

2. **Two‑phase descent** – the most promising seeds are refined.
   *Phase A* uses the same exponentiated‑gradient / soft‑max trick as the
   previous state‑of‑the‑art (a temperature‑controlled soft‑max over the
   convolution entries).  *Phase B* switches to a **hard‑max**
   sub‑gradient (gradient of the single maximal convolution entry) and
   continues with an exponentiated‑gradient update.  Both phases work
   on the simplex via a simple renormalisation after each step.

   Learning‑rate decays as `ηₜ = lr0/√t`.  The temperature α is ramped
   from a small start value up to a large final value (≈ 80) to
   concentrate the gradient on the true max in the later stage.

3. **Greedy polishing** – a cheap random‑mass‑transfer walk (pairwise
   swaps) is performed until the time budget expires.  The walk moves a
   random fraction (up to 30 %) of mass from one index to another and
   keeps the change only if the true objective improves.

The whole routine respects the provided time budget (`budget_s` seconds)
and returns the best sequence found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a))/sum(a)^2.
    The vector a is assumed to be non‑negative.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _gradient_max_index(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the single maximal convolution entry `conv[k]`.
    Returns a vector of size len(a) with non‑negative entries.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i

    eq_mask = (i == j)
    # diagonal terms contribute 2*a[i]
    if np.any(eq_mask):
        g[i[eq_mask]] += 2.0 * a[i[eq_mask]]

    # off‑diagonal terms contribute a[j] to i and a[i] to j
    if np.any(~eq_mask):
        ii = i[~eq_mask]
        jj = j[~eq_mask]
        g[ii] += a[jj]
        g[jj] += a[ii]

    return g


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * x / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_sine(n))
    cand.append(_distribution_two_blocks(n, block_frac=0.07))

    # symmetric beta on a fine grid
    for p in np.linspace(0.05, 1.0, 12):
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power law
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential
    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sparse random draws
    for dens in (0.005, 0.01, 0.02):
        cand.append(_distribution_random_sparse(n, rng, density=dens))

    # Dirichlet draws with various concentrations
    for a in (0.1, 0.3, 0.5, 1.0):
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    return cand


# ----------------------------------------------------------------------
def _softmax_grad_descent(a: np.ndarray, deadline: float,
                          best_seq: np.ndarray, best_score: float,
                          lr0: float = 0.9,
                          max_iter: int = 60000,
                          eval_interval: int = 500,
                          alpha_start: float = 0.5,
                          alpha_end: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the
    maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear schedule for temperature α
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # soft‑max weights (stable)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project back onto the simplex (just renormalise)
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] soft‑max it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _max_grad_descent(a: np.ndarray, deadline: float,
                      best_seq: np.ndarray, best_score: float,
                      lr0: float = 0.5,
                      max_iter: int = 80000,
                      eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent using the true maximal entry of the convolution.
    The update is exponentiated‑gradient with a decaying step size.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())

        grad = _gradient_max_index(a, k)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hard‑max it={t} idx={k} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Cheap random mass‑transfer walk.  Randomly moves a fraction of mass
    from one index to another and keeps the change only if the true
    objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        # move up to 30 % of source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point used by the checker.

    The routine respects the global time budget, generates a diverse
    pool of analytic and random seeds for several lengths, refines the
    most promising ones with a two‑phase gradient descent (soft‑max
    followed by hard‑max), and finally performs a greedy local polishing.
    The best sequence (as a plain list of floats) is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # 1. Choose a set of lengths.  Longer lengths give a tighter
    #    theoretical lower bound but increase per‑iteration cost.
    lengths = [4000, 8000, 12000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    elite = []   # (score, seq) tuples across all lengths

    # --------------------------------------------------------------
    # 2. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length = {n} ---")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the best few across all lengths so far
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # --------------------------------------------------------------
    # 3. Incorporate the previously found sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Fallback (should never be needed)
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Main optimisation – two‑phase descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        # take the very best elite seeds
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:5]]
        # split remaining time roughly evenly between the two phases
        per_seed = max(1.0, remaining / (len(seeds) * 2.0))

        for seed_seq in seeds:
            if time.time() >= deadline:
                break

            # -------- Phase A: soft‑max descent --------------------
            sub_deadline_a = time.time() + per_seed
            seq_a, score_a = _softmax_grad_descent(
                seed_seq.copy(),
                sub_deadline_a,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                max_iter=80000,
                eval_interval=400,
                alpha_start=0.4,
                alpha_end=80.0,
            )
            if score_a < overall_best_score:
                overall_best_score = score_a
                overall_best_seq = seq_a.copy()
                print(f"[{time.time():.1f}] after soft‑max new best {overall_best_score:.6f}")

            # -------- Phase B: hard‑max sub‑gradient descent -------
            if time.time() >= deadline:
                break
            sub_deadline_b = time.time() + per_seed
            seq_b, score_b = _max_grad_descent(
                seq_a.copy(),
                sub_deadline_b,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.5,
                max_iter=80000,
                eval_interval=400,
            )
            if score_b < overall_best_score:
                overall_best_score = score_b
                overall_best_seq = seq_b.copy()
                print(f"[{time.time():.1f}] after hard‑max new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final polishing with cheap local swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # --------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
```

This implementation builds on the previous state‑of‑the‑art method but introduces:

* a richer seed pool (including sine‑shaped and finer Beta grids);
* a two‑stage optimisation (soft‑max then exact hard‑max sub‑gradient);
* more aggressive temperature (`α_end = 80`) and longer iteration budgets;
* adaptive per‑seed time allocation;
* the same fast FFT‑based convolution machinery and a simplex projection step.

All helper functions are top‑level, there are no closures or lambdas, and the
routine respects the 1000‑second budget while striving for a lower bound
than the previous best (≈ 1.509964).
