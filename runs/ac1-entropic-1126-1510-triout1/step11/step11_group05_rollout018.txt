"""
Optimisation via alternating linear‑programming refinement.

The idea follows the scheme from the description:
1. start from a non‑negative step function `f` (∑f = 1);
2. compute M = max_k (f∗f)[k];
3. solve the linear programme  
       maximise   1ᵀg
       subject to  f∗g ≤ M ,  g ≥ 0
   (the convolution constraints are linear in `g`);
   the optimum `g` is the most massive vector that does not raise the
   current bound `M`;
4. blend the current iterate with the normalised solution of the LP  
      f ← (1‑t)·f + t·(g/∑g)   (t≈0.3);
   this never increases `M` and usually makes it smaller;
5. repeat steps 2‑4 a few times, then polish the best candidate with
   the smooth‑max mirror‑descent / sub‑gradient routine from the
   previous state‑of‑the‑art code.

The algorithm explores several analytic families (uniform, symmetric
and asymmetric Beta, mixtures, Dirichlet draws) and a few random
sparse starts.  It is time‑aware – all heavy loops abort when the
remaining budget is below a safety margin, guaranteeing a return
before the 1000 s deadline.
"""

# Simplex projection
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return vectors with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n: int, p: float, wmix: float) -> np.ndarray:
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = wmix * beta + (1.0 - wmix) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n: int, alpha: float) -> np.ndarray:
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_sparse_random(n: int, k: int) -> np.ndarray:
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

def _smooth_max_mirror_descent(a0: np.ndarray, deadline: float):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    schedule = [
        (1.0, 1e3, 3000, 0.15, 0.015),
        (1e3, 1e5, 3000, 0.015, 0.0015),
        (1e5, 1e6, 4000, 0.0015, 0.00015),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0: np.ndarray, deadline: float):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a0: np.ndarray, deadline: float):
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # move mass away from the pair that generates the current maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j

        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Linear‑programming refinement step
def _lp_max_sum(f: np.ndarray, M: float):
    """Solve   maximize 1ᵀg    subject to   f∗g ≤ M , g ≥ 0.
       Returns the optimiser `g` (may be None on failure)."""
    import scipy.optimize

    n = f.shape[0]
    rows = 2 * n - 1

    # build the constraint matrix A where (A g)_k = (f∗g)[k]
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n)                     # minimise –∑g  <=> maximise ∑g
    b = np.full(rows, M, dtype=np.float64)
    bounds = [(0, None)] * n

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine mixes analytic families, an LP‑based alternating refinement,
    and the smooth‑max mirror‑descent / sub‑gradient polishing from the
    previous state‑of‑the‑art implementation.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []                                      # (score, vector)

    # --------------------------------------------------------
    # 1. generate a pool of analytic initial vectors (length = 2048)
    n = 2048
    candidates = []

    candidates.append(_distribution_uniform(n))
    candidates.append(_distribution_beta_sym(n, 0.5))            # arcsine

    # symmetric betas on a dense grid
    for p in np.concatenate((np.logspace(-2, -0.2, 12),
                             np.linspace(0.1, 2.0, 20))):
        candidates.append(_distribution_beta_sym(n, float(p)))

    # asymmetric betas
    for p in (0.3, 0.6, 1.0):
        for q in (0.5, 1.0, 1.5):
            candidates.append(_distribution_beta_asym(n, float(p), float(q)))

    # mixtures with the uniform distribution
    for p in (0.05, 0.2, 0.5, 1.0):
        for w in (0.2, 0.5, 0.8):
            candidates.append(_distribution_mixture_beta_uniform(n, float(p), w))

    # random Dirichlet draws
    for alpha in (0.05, 0.2, 0.8, 2.0):
        candidates.append(_distribution_random_dirichlet(n, float(alpha)))

    # sparse start
    k = max(2, n // 400)
    candidates.append(_distribution_sparse_random(n, k))

    # previous solution if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    # --------------------------------------------------------
    # 2. loop over initial candidates
    for idx, vec in enumerate(candidates):
        if time.time() >= deadline:
            break
        a = np.maximum(vec, eps)
        a /= a.sum()
        conv_max = _fft_self_convolve(a).max()
        score = 2 * n * conv_max
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] init {idx}  score={score:.6f}")

        # keep a short elite list (size ≤ 6)
        if len(elite) < 6:
            elite.append((score, a.copy()))
            elite.sort(key=lambda x: x[0])
        elif score < elite[-1][0]:
            elite[-1] = (score, a.copy())
            elite.sort(key=lambda x: x[0])

        # ----------------------------------------------------
        # 2a. LP‑based alternating refinement
        M = conv_max
        for it in range(25):          # limited number of LP updates
            if time.time() >= deadline:
                break
            g = _lp_max_sum(a, M)
            if g is None:
                break
            s = g.sum()
            if s <= 1.0 + 1e-12:       # no improvement possible
                break
            g_norm = g / s
            t = 0.3                    # blending factor
            a = (1.0 - t) * a + t * g_norm
            a = np.maximum(a, eps)
            a /= a.sum()
            new_M = _fft_self_convolve(a).max()
            new_score = 2 * n * new_M
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] LP‑refine it={it} new best {new_score:.6f}")

            if new_M >= M - 1e-12:   # stop if bound stagnates
                break
            M = new_M

        # ----------------------------------------------------
        # 2b. Fine‑tune the result with the smooth‑max / sub‑gradient pipeline
        a_mid, sc_mid = _smooth_max_mirror_descent(a, deadline)
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] Post‑opt new best {sc_mid:.6f}")

    # --------------------------------------------------------
    # 3. final sanity – enforce non‑negativity, normalise and report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
