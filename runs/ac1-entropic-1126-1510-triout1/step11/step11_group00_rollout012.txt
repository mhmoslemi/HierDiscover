            w = np.ones_like(w) / len(w)  # fallback
        else:
            w = w / w_sum
        # compute grad as weighted sum contributions:
        grad = np.zeros_like(a)
        ind_grid = np.arange(n)
        for idx, weight in zip(idx_top, w):
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]] * weight
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] weighted exp it={t} new best {best_score:.6f}")

            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve=0
    return best_seq, best_score

"""
Improved search for a non‑negative step function that minimizes
``evaluate_sequence`` (the L∞‑norm of the self‑convolution).

Main ideas
----------
1. **Diverse analytic seeds** – families (uniform, arcsine, symmetric / asymmetric
   Beta, edge‑weighted, sine‑weighted) are sampled for several lengths together
   with a few random Dirichlet / sparse vectors.  The best seeds are kept.

2. **Weighted exponential‑gradient descent** – instead of using only the
   single maximal convolution index we take the top *k* indices (k≈0.02·2n)
   and form a softmax‑weighted sub‑gradient.  The update is multiplicative,
   i.e.  a←a·exp(−η·∇)  and the vector is re‑normalised to the simplex.
   The temperature β of the softmax grows slowly, a small jitter is added
   after long stagnation.

3. **Local polishing** – greedy pairwise mass moves and a cheap simulated‑annealing
   walk finish the optimisation.

The whole procedure respects the supplied time budget (default 1000 s) and
returns a plain Python list; ``evaluate_sequence`` will perform the final
checks (clipping, scaling, …).
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    rho = rho_arr[-1] if rho_arr.size else -1
    theta = cssv[rho] / (rho + 1) if rho >= 0 else 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())         # next power‑of‑2
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric beta (Beta(p,q))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ sin(π (i+½)/n)^p , vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector (density = fraction of non‑zeros)."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture lam·U + (1‑lam)·base (U = uniform)."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Weighted exponential‑gradient descent (active set) --------------------
def _weighted_exp_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.7,
    eval_interval: int = 120,
    top_frac: float = 0.02,
    max_top: int = 30,
    beta_start: float = 3.0,
    beta_end: float = 15.0,
    jitter_scale: float = 0.001,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponential) gradient descent.
    The sub‑gradient is a softmax‑weighted sum of the gradients of the
    top‑k convolution entries.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    ind_grid = np.arange(n, dtype=np.int32)

    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # ---- choose the top‑k indices ---------------------------------
        k = max(2, int(top_frac * conv.size))
        k = min(k, max_top)
        idx_top = np.argpartition(conv, -k)[-k:]          # unsorted
        # sort descending to obtain a monotone temperature schedule
        order = np.argsort(-conv[idx_top])
        idx_top = idx_top[order]
        scores = conv[idx_top]

        # ---- temperature (softmax) ------------------------------------
        beta = beta_start + (beta_end - beta_start) * (t / 200_000.0)
        beta = min(beta, beta_end)
        weights = np.exp(beta * (scores - max_c))
        wsum = weights.sum()
        if wsum == 0.0:
            weights = np.full_like(weights, 1.0 / len(weights))
        else:
            weights /= wsum

        # ---- weighted sub‑gradient ------------------------------------
        grad = np.zeros_like(a)
        for idx, w in zip(idx_top, weights):
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]] * w

        # ---- multiplicative update ------------------------------------
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # ---- renormalise to the simplex --------------------------------
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # ---- occasional evaluation ------------------------------------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(
                    f"[{time.time():.1f}] weighted‑exp it={t} new global best {best_score:.6f}"
                )
            else:
                no_improve += 1

        # ---- jitter after stagnation ----------------------------------
        if no_improve >= patience:
            jitter = rng.exponential(scale=jitter_scale, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0

    return best_seq, best_score

# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 0.008,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    )
    lam_grid = np.linspace(0.0, 1.0, 9)                # 0 … 1

    for n in lengths:

            break

        # deterministic families (no mixing)

            cand = cand / cand.sum()
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families (symmetric beta, edge, sine) and mix with uniform
