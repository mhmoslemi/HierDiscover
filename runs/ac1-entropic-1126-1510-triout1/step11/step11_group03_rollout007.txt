def _subgradient_descent(a_initial, deadline, best_seq, best_score,
                         max_iter=20000, eval_interval=200, eta0=1.0):
    a = a_initial.copy()
    n = a.shape[0]
    for t in range(1, max_iter+1):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        mx = conv.max()
        # get active indices (within tolerance)
        tol = 1e-12
        active = np.where(conv >= mx - tol)[0]
        # weight uniform
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0/active.size
        # compute grad
        grad = _gradient_from_weights(a, w)
        eta = eta0 / math.sqrt(t)
        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        # renorm
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / s
        # occasional evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"{time.time():.1f} subgrad it={t} new best {best_score:.6f}")
    return best_seq, best_score

"""Improved search for a low‑value step‑function.

The algorithm explores a variety of analytic seeds, then refines the best
candidates with a two‑stage gradient descent:

1. **Exact‑max sub‑gradient descent** – the gradient of the current maximal
   convolution entry is used (or a random active entry when several share the
   maximum).  An exponentiated‑gradient (mirror descent on the simplex) with
   momentum updates the coefficients, keeping them non‑negative and normalised.

2. **Soft‑max refinement** – a temperature‑controlled soft‑max replaces the
   hard max, allowing a smoother gradient that is annealed to a large

Finally a cheap greedy “mass‑transfer” walk tries random δ‑moves that improve
the true (non‑smoothed) objective.  The best sequence found before the
budget expires is returned.

The code respects the overall time budget (default 1000 s) and works for any
reasonable sequence length."""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1, sum(w)=1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sinusoid(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - center) / (center + 1.0)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _subgradient_descent(a_initial: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_iter: int = 20000,
                         eval_interval: int = 200,
                         eta0: float = 1.0,
                         momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max of the self‑convolution.
    Uses exponentiated‑gradient updates with momentum and projects back onto
    the probability simplex.
    """
    a = a_initial.copy()
    n = a.shape[0]
    rng = np.random.default_rng()
    mom = np.zeros_like(a)

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        mx = conv.max()
        # active indices (within tiny tolerance)
        active = np.where(conv >= mx - 1e-12)[0]
        if active.size == 0:
            break
        # pick a random active index to form the sub‑gradient
        k = rng.choice(active)
        w = np.zeros_like(conv)
        w[k] = 1.0

        grad = _gradient_from_weights(a, w)
        # momentum on the gradient
        mom = momentum * mom + (1.0 - momentum) * grad

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * mom)

        # simplex projection (renormalisation)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _smooth_grad_descent(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         lr0: float = 0.9,
                         max_iter: int = 60000,
                         eval_interval: int = 500,
                         alpha_start: float = 0.5,
                         alpha_end: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the gradient is taken w.r.t.
    a soft‑max approximation of the maximal convolution entry.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear schedule for the soft‑max temperature
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # soft‑max weights (stable via subtraction of max)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # normalise onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer walk; moves a random fraction of mass from one index
    to another and keeps the move only if the true objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # random fraction up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point used by the checker.
    Generates a diverse set of seeds, refines the most promising with
    sub‑gradient descent followed by a soft‑max fine‑tune, and finishes
    with a greedy polishing walk.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to explore (larger → finer, but also slower)
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]

    overall_best_seq = None
    overall_best_score = float("inf")
    candidates = []   # list of (score, array)

    # ------------------------------------------------------------------
    # 2. Incorporate previous solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            else:
                prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length = {n} ---")
        seed_arrays = []

        # deterministic families
        seed_arrays.append(_distribution_uniform(n))
        seed_arrays.append(_distribution_arcsine(n))
        for p in (0.2, 0.5, 0.8, 1.0):
            seed_arrays.append(_distribution_beta_sym(n, p))
        for lam in (0.002, 0.01, 0.05):
            seed_arrays.append(_distribution_exponential_increasing(n, lam))
        seed_arrays.append(_distribution_two_blocks(n, block_frac=0.07))
        seed_arrays.append(_distribution_random_sparse(n, rng, density=0.02))
        for p in (0.5, 1.0, 2.0):
            seed_arrays.append(_distribution_power_increasing(n, p))
        seed_arrays.append(_distribution_sinusoid(n))
        seed_arrays.append(_distribution_triangular(n))

        # a few Dirichlet draws
        for _ in range(4):
            alpha = 10 ** rng.uniform(-2, 1)
            seed_arrays.append(_distribution_dirichlet(n, rng, alpha))

        # evaluate and keep elite
        for a in seed_arrays:
            if time.time() >= deadline:
                break
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                a = _distribution_uniform(n)
            a = a / a.sum()
            sc = _objective(a)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            candidates.append((sc, a.copy()))

        # keep only the top few across all lengths so far
        candidates.sort(key=lambda x: x[0])
        candidates = candidates[:20]

    # ------------------------------------------------------------------
    # 4. Main optimisation on the elite seeds
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 25.0:
            break
        # pick a handful of current elite seeds
        elite = [seq for _, seq in sorted(candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / len(elite)
        for seed_seq in elite:
            if time.time() >= deadline:
                break
            # --- sub‑gradient phase ------------------------------------
            sub_deadline = time.time() + per_seed * 0.6
            seq_opt, score_opt = _subgradient_descent(
                seed_seq.copy(),
                sub_deadline,
                overall_best_seq,
                overall_best_score,
                max_iter=25000,
                eval_interval=250,
                eta0=2.0,
                momentum=0.85,
            )
            if score_opt < overall_best_score:
                overall_best_score = score_opt
                overall_best_seq = seq_opt.copy()
                print(f"[{time.time():.1f}] after subgrad new best {overall_best_score:.6f}")

            # --- soft‑max refinement ------------------------------------
            if time.time() < deadline - 5.0:
                ref_deadline = time.time() + per_seed * 0.35
                seq_ref, score_ref = _smooth_grad_descent(
                    seq_opt.copy(),
                    ref_deadline,
                    overall_best_seq,
                    overall_best_score,
                    lr0=0.8,
                    max_iter=50000,
                    eval_interval=500,
                    alpha_start=0.5,
                    alpha_end=120.0,
                )
                if score_ref < overall_best_score:
                    overall_best_score = score_ref
                    overall_best_seq = seq_ref.copy()
                    print(f"[{time.time():.1f}] after refinement new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing with cheap local swaps
    if time.time() < deadline - 0.5:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        print(f"[{time.time():.1f}] final polishing best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
