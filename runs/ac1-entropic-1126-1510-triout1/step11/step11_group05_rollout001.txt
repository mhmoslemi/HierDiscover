Advanced search for a non‑negative step function that minimizes
`evaluate_sequence`.

Main ideas
~~~~~~~~~~
* **Richer candidate pool** – in addition to the Uniform and symmetric Beta
  families we also consider symmetrised asymmetric Beta distributions,
  mixtures of several Betas, Dirichlet draws with very small concentration
  and sparse random starts.
* **Extended smooth‑max mirror descent** – the temperature α now grows up to
  1e8 and the schedule is longer, giving a tighter approximation of the
  true max.

* **More restarts & elite handling** – we keep the 10 best candidates and
  restart the whole pipeline from each of them.
* **Time aware** – every heavy loop checks the deadline (1000 s minus a small
  safety margin) and aborts early if necessary.

The objective for a normalised vector a (sum = 1) is
    score = 2·n·max(a*a)
which is exactly the quantity returned by `evaluate_sequence` after its
internal normalisation.  All optimisation is performed on the simplex, so

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) discretised on a uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid overflow/underflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym_sym(n, p, q):
    """Symmetrised asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    # symmetrise
    w = (w + w[::-1]) / 2.0
    w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform (both already normalised)."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_mixture_two_betas(n, p1, p2, w):
    """w·Beta(p1,p1) + (1‑w)·Beta(p2,p2)."""
    b1 = _distribution_beta_sym(n, p1)
    b2 = _distribution_beta_sym(n, p2)
    mix = w * b1 + (1.0 - w) * b2
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):
    vec = np.random.dirichlet(np.full(n, alpha))
    return vec


def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a

def _smooth_max_mirror_descent(a0, deadline):
    """
    Exponentiated‑gradient (mirror) descent on a smooth‑max surrogate.
    The temperature α is increased up to 1e8, giving a very tight
    approximation of the true max.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # schedule: (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e4, 4000, 0.18, 0.018),
        (1e4, 1e6, 4000, 0.018, 0.0018),
        (1e6, 1e8, 5000, 0.0018, 0.00018),
    ]

    for a0_alpha, a1_alpha, steps, lr0, lr1 in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.8f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline):
    """
    True sub‑gradient descent with Adam‑style momentum and a back‑tracking
    line‑search, run until the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.7
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        it_f = float(it)
        v_hat = velocity / (1.0 - beta1 ** it_f)

        # back‑tracking line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

                break
            step *= 0.5

            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.8f}")

            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _simulated_annealing(a0, deadline, max_iter=8000):
    """
    Simple SA on the simplex: repeatedly move a random amount of mass
    from one index to another, accepting uphill moves with probability
    exp(-Δ/T). Temperature decays geometrically.
    """
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    cur_sc = _max_convolution_score(a)
    best_a = a.copy()
    best_sc = cur_sc

    T0 = 0.05 * cur_sc   # a fraction of the current objective
    T_end = 1e-7
    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        # geometric temperature schedule
        T = T0 * (T_end / T0) ** (it / max_iter)

        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1  # ensure i != j

        # propose a transfer delta (up to 30 % of the source mass)
        max_delta = min(a[i] * 0.3, 0.02)
        if max_delta <= eps:
            continue
        delta = random.random() * max_delta

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # numerical safety
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_sc = _max_convolution_score(a_new)
        delta_sc = new_sc - cur_sc

        accept = False
        if delta_sc < 0:
            accept = True
        else:
            prob = math.exp(-delta_sc / (T + 1e-18))
            if random.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_sc = new_sc
            if cur_sc < best_sc - 1e-14:
                best_sc = cur_sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] SA it={it} new best {best_sc:.8f}")

    return best_a, best_sc

        for i in range(i_start, i_end + 1):

def _generate_candidates(n):
    """Return a list of candidate vectors (already normalised to sum = 1)."""
    cand = []

    # Uniform and arcsine
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))

    # Symmetric Beta grid – log‑space for very small p
    for p in np.concatenate(
        (np.logspace(-4, -0.2, 14), np.linspace(0.05, 2.5, 30))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # Symmetrised asymmetric Beta
    for p, q in [(0.02, 1.5), (0.02, 5.0), (0.1, 4.0), (0.3, 3.0)]:
        cand.append(_distribution_beta_asym_sym(n, float(p), float(q)))

    # Mixtures Beta + Uniform
    for p in (0.02, 0.05, 0.1, 0.2, 0.4, 0.8):
        for w in (0.1, 0.3, 0.5, 0.7, 0.9):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # Mixtures Two Betas
    for (p1, p2) in [(0.02, 0.5), (0.02, 1.0), (0.05, 0.8), (0.1, 0.4)]:
        for w in (0.3, 0.5, 0.7):
            cand.append(_distribution_mixture_two_betas(n, p1, p2, w))

    # Dirichlet draws with a range of concentration parameters
    for alpha in (0.001, 0.005, 0.02, 0.1, 0.5):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # Sparse random start
    k = max(2, n // 400)
    cand.append(_distribution_sparse_random(n, k))

    # Random shuffles of a few bases to introduce asymmetry
    for base in (cand[0], cand[2], cand[-3]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand
