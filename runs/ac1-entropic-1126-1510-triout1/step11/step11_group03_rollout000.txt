assistantfinal**Solution Overview**

The goal is to produce a non‑negative step‑function `a = (a₀,…,a_{n‑1})` that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0,
\]

where `a*a` is the full linear convolution.  
Because `F(λa)=F(a)` we may safely keep the sum fixed to **1** and only
optimise the shape of the vector.

The program follows three stages

1. **Exploration of analytic families** – for several lengths we create a
   diverse set of seeds (uniform, arcsine, Beta, power‑law, exponential,
   two‑block, sparse, Dirichlet …).  
   All seeds are normalised to sum = 1 and the exact objective is
   evaluated. The best 20 seeds (across all lengths) are kept.

2. **Soft‑max mirror descent** – each elite seed undergoes a long
   exponentiated‑gradient descent.  
   The non‑smooth `max` is replaced by a temperature‑controlled soft‑max  

   \[
   w_k\propto\exp(\alpha\,c_k),\qquad c=a*a,
   \]

   with the temperature `α` linearly increased from `0.5` to `60`.  
   The gradient of the smoothed objective is  
   `∇ = 2·a ⋆ w` (cross‑correlation) which is obtained by FFT.
   After the gradient step we project back onto the probability simplex.

   The descent is run for **up to 120 000 iterations** on the modest
   lengths (2 k…8 k).  Progress is reported and the best ever sequence is
   stored.

3. **Multi‑scale up‑sampling & polishing** – the best sequence obtained
   at the medium scale is interpolated to larger lengths (16 k and 32 k)
   and refined briefly (20 k – 30 k iterations each).  Finally a cheap
   greedy mass‑transfer walk (randomly moving a small fraction of mass
   from one index to another) tries to squeeze a last improvement.

All operations use only `numpy` and the standard library, respect the
global time budget (`budget_s` seconds) and never produce negative or
NaN entries.  The returned list can be fed directly to `evaluate_sequence`.

"""
Hybrid multi‑scale soft‑max mirror descent for the step‑function attack.

1.   Generates a large pool of analytic seeds (uniform, arcsine, beta,
     power‑law, exponential, two‑block, sparse, Dirichlet) for several
     lengths (2k … 8k).  Keeps the 20 best after exact evaluation.

2.   Refines each elite seed with exponentiated‑gradient descent on a
     soft‑max approximation of the max‑convolution.  The temperature α
     is increased from 0.5 to 60, learning rate η = η₀/√t, and after each
     step the vector is projected onto the simplex.  Up to 120 k
     iterations per seed are performed (time‑limited).

3.   Upsamples the best medium‑scale solution to 16 k and 32 k, runs a
     short fine‑tuning descent, and finishes with a greedy mass‑transfer
     local search.

The best vector found (as plain list of floats) is returned.
"""

import time
import math
import numpy as np

# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based self‑convolution (full length 2·n‑1)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (sum is forced to 1)
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    # sum is 1 by construction, but keep safety check
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Gradient of the soft‑max smoothed objective.
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Returns ∇ = 2·a⋆w   (cross‑correlation) where
    a has length n, w has length 2·n‑1 and sum(w)=1.
    Implemented via real FFT.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad

# Analytic families (all return a probability vector)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals

def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a wide variety of analytic seeds for a fixed length."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
    ]

    # symmetric beta family
    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    # asymmetric beta
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    # power‑law
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential
    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    # two‑block (mass at both ends)
    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # few Dirichlet draws
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))
    # sparse random
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand

def _softmax_mirror_descent(a: np.ndarray,
                            deadline: float,
                            best_seq: np.ndarray,
                            best_score: float,
                            max_iter: int = 120000,
                            lr0: float = 0.9,
                            alpha_start: float = 0.5,
                            alpha_end: float = 60.0,
                            eval_every: int = 400) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using a soft‑max approximation of the
    max‑convolution.  The temperature α grows linearly from α_start to
    α_end.  After each update the iterate is projected onto the simplex.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        # current full convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        # stable soft‑max weights
        exp_vals = np.exp(alpha * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # gradient = 2·a⋆w
        grad = _gradient_from_weights(a, w)

        # learning rate – classic 1/√t schedule
        eta = lr0 / math.sqrt(t)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad)

        # project onto simplex (sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        # occasional exact evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] descent it={t} α={alpha:.2f} "
                      f"new best {best_score:.6f}")

    return best_seq, best_score

def _greedy_swap_local(a: np.ndarray,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from i→j, keep move if it improves."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick a source with non‑negligible mass
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue
        # move up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} "
                  f"new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample_vector(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(a) to new_n, then re‑normalise."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0.0)
    new_a /= new_a.sum()
    return new_a

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Main entry point called by the checker.  Returns a list of non‑negative
    floats (the heights of the step function) that (hopefully) improves the
    current best bound.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1.  Generate analytic seeds for several moderate lengths
    base_lengths = [2000, 4000, 8000]          # sizes where we afford many iterations
    elite = []                                # (score, vector)
    overall_best_seq = None
    overall_best_score = float("inf")

    for n in base_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning seeds for length {n} ---")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand = _project_onto_simplex(cand, s=1.0)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} "
                      f"score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only the top 20 across all lengths visited so far
        elite.sort(key=lambda x: x[0])
        elite = elite[:20]

    # ------------------------------------------------------------------
    # 2.  Insert a previously known solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = _project_onto_simplex(prev, s=1.0)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous solution improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:20]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Soft‑max mirror descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and elite:
        per_seed_time = remaining / len(elite)
        for score, seed_vec in elite:
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed_time
            # allocate iterations proportionally to length (longer ⇒ fewer iters)
            base_iters = 120000
            max_iter = int(base_iters * (8000 / seed_vec.shape[0]))
            max_iter = max(20000, min(max_iter, 120000))

            best_seq, best_score = _softmax_mirror_descent(
                seed_vec.copy(),
                deadline=sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                max_iter=max_iter,
                lr0=0.9,
                alpha_start=0.5,
                alpha_end=60.0,
                eval_every=400,
            )
            if best_score < overall_best_score:
                overall_best_score = best_score
                overall_best_seq = best_seq.copy()
                print(f"[{time.time():.1f}] after descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Multi‑scale up‑sampling and short refinements
    upscale_lengths = [16000, 32000]
    for n_big in upscale_lengths:
        if time.time() >= deadline:
            break
        if overall_best_seq is None:
            break
        up_seq = _upsample_vector(overall_best_seq, n_big)
        # short fine‑tuning (10 k – 20 k iterations)
        sub_deadline = time.time() + min(40.0, deadline - time.time())
        max_iter = 20000
        up_seq, up_score = _softmax_mirror_descent(
            up_seq.copy(),
            deadline=sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_iter=max_iter,
            lr0=0.7,
            alpha_start=0.5,
            alpha_end=50.0,
            eval_every=300,
        )
        if up_score < overall_best_score:
            overall_best_score = up_score
            overall_best_seq = up_seq.copy()
            print(f"[{time.time():.1f}] up‑scaled length {n_big} new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final greedy polishing
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _greedy_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # ------------------------------------------------------------------
    # 6.  Return result
    if overall_best_seq is None:
        # Fallback: uniform distribution of a modest size
        overall_best_seq = _distribution_uniform(2000)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
