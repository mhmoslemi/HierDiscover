    current = a.copy()
    cur_sc = _objective(current)
    best = current.copy()
    best_sc = cur_sc
    temp = init_temp

        # propose move
        i = random.randrange(len(current))
        j = random.randrange(len(current))
        if i == j: continue
        delta = (current[i] * random.uniform(0.1, 0.5)) # choose fraction of source
        if delta <= 1e-12: continue
        if current[i] <= delta: continue
        cand = current.copy()

        cand[j] += delta

        cand_sc = _objective(cand)
        if cand_sc < cur_sc or random.random() < math.exp((cur_sc - cand_sc) / temp):
            current = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = current.copy()

"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The search proceeds in three phases:

1. **Broad sampling** – generate many analytic families (uniform,
   beta, power‑law, exponential, Gaussian, sine‑power, cosine‑window,
   arcsine, triangular) together with a few Dirichlet draws for several
   lengths (2000 … 20000).  Keep a small elite of the best candidates.

2. **Intensive local optimisation** – for each elite vector run a
   long mirror‑descent on the smoothed max‑convolution, followed by true
   sub‑gradient descent (Adam style) and finally a simulated‑annealing
   hill‑climber.  All updates are projected onto the probability simplex
   and keep the vector normalised to sum = 1.

3. **Return the best vector** – after the deadline the best sequence is
   returned as a plain Python list.  The code respects the 1000 s budget
   and uses only NumPy (FFT based convolutions) for speed.
"""

import time
import random
import math
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolution utilities ---------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Objective (fast surrogate – 2·n·max(conv) , sum is 1) -----------------

def _objective(a: np.ndarray) -> float:
    """Fast surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (return vectors with sum = 1) -----------------------

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Candidate generator ---------------------------------------------------

def _generate_candidates_for_length(n: int) -> List[np.ndarray]:
    """Return a list of candidate vectors (sum = 1) for a given length."""
    cand = []

    # families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # symmetric Beta – sweep fine grid
    for p in np.logspace(-2, 0.7, num=25):          # 0.01 … 5
        cand.append(_distribution_beta_sym(n, float(p)))

    # asymmetric Beta – few hand‑picked pairs
    for p, q in ((0.2, 1.0), (1.0, 0.2), (0.5, 1.5), (1.5, 0.5), (0.3, 0.9), (0.9, 0.3)):
        cand.append(_distribution_beta_asym(n, float(p), float(q)))

    # sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0, 16.0):
        cand.append(_distribution_sine_power(n, pw))

    # cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45, 0.55):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0, 4.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Dirichlet draws – a variety of concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            vec = np.random.dirichlet(np.full(n, alpha))
            cand.append(vec)

    # Sparse random starts (few non‑zeros)
    k = max(2, n // 250)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Shuffle a few base families to inject asymmetry
    for base in (cand[0], cand[1], cand[4]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand


# ----------------------------------------------------------------------
# Mirror‑descent on smoothed max ----------------------------------------

def _smooth_max_mirror_descent(a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """Exponentiated‑gradient mirror descent on a log‑sum‑exp smoothed max."""
    eps = 1e-12
    n = a.shape[0]

    # temperature schedule – go from very smooth to almost exact
    schedule = [
        (1.0, 1e4, 12000, 1e-2, 2e-4),
        (1e4, 1e6, 12000, 2e-4, 4e-6),
        (1e6, 1e8, 12000, 4e-6, 8e-8),
    ]

    best_a = a.copy()
    best_sc = _objective(best_a)

    for (a0, a1, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0 * (a1 / a0) ** t          # temperature
            lr = lr0 * (lr1 / lr0) ** t          # learning‑rate

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # log‑sum‑exp weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smoothed max
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            # exponentiated‑gradient update
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            # occasional true‑objective check
            if it % 3000 == 0:
                sc = _objective(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] MD it={it} new best {sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient -------------------------------------

def _projected_subgradient(a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """True sub‑gradient descent on the exact max, with Adam momentum."""
    eps = 1e-12
    n = a.shape[0]
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.5
    it = 0

    best_a = a.copy()
    best_sc = _objective(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # indices attaining the max (within tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update on the gradient
        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_lr / (1.0 + it * 1e-4)   # slowly decaying step

        cand = a - step * v_hat
        cand = _project_onto_simplex(cand, 1.0)
        cand = np.maximum(cand, eps)
        cand /= cand.sum()
        cand_sc = _objective(cand)

        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = cand.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        a = cand

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 2e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber ---------------------------------------

def _simulated_annealing(a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """Light SA that moves tiny masses between coordinates."""
    eps = 1e-14
    n = a.shape[0]
    cur = a.copy()
    cur_sc = _objective(cur)
    best = cur.copy()
    best_sc = cur_sc

    temp = 1.0
    cooling = 0.9995
    min_temp = 1e-6

    while time.time() < deadline:
        # pick source & target
        i = random.randrange(n)
        j = random.randrange(n)
        if i == j:
            continue
        # move a random fraction of source to target
        max_move = cur[i] * 0.5
        if max_move < eps:
            continue
        delta = max_move * random.random()
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.maximum(cand, eps)
        cand /= cand.sum()
        cand_sc = _objective(cand)

        accept = False
        if cand_sc < cur_sc:
            accept = True
        else:
            prob = math.exp(-(cand_sc - cur_sc) / max(temp, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_sc = cand_sc
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = cur.copy()
                print(f"[{time.time():.1f}] SA new best {best_sc:.6f}")

        # temperature update
        temp = max(temp * cooling, min_temp)

        # prune tiny entries every 5000 accepted moves
        if random.random() < 0.001:
            thresh = 1e-12
            small = cur < thresh
            if np.any(small):
                cur[small] = 0.0
                cur /= cur.sum()
                cur = np.maximum(cur, eps)
                cur /= cur.sum()

    return best, best_sc


# ----------------------------------------------------------------------
# Main proposal function ------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # --------------------------------------------------------------
    # 1. Broad exploration (retain a tiny elite)
    lengths = [2000, 3500, 5000, 8000, 12000, 16000, 20000]
    elite: List[Tuple[float, np.ndarray]] = []   # (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            # global update
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain elite (size ≤ 10)
            if len(elite) < 10:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2. Inject previous best if available
    try:

        if prev.ndim == 1 and prev.size > 0:

            if len(elite) < 10:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 3. Fallback (should never be needed)

        best_seq = _distribution_uniform(best_n)

        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

            break
        a0 = elite[idx][1].copy()

        # Mirror‑descent

        # Sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # Simulated‑annealing hill‑climber

        if sc_mid < best_score - 1e-12:

            best_n = a_mid.shape[0]

    # --------------------------------------------------------------

    return best_seq.tolist()
