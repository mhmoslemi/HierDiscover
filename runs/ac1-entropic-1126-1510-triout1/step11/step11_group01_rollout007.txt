def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a non‑negative vector with Σa = 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families -----------------------------------------------------
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """Weight ∝ (i·j)^{‑p} with i,j the distances to the two edges."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangle(n: int) -> np.ndarray:
    """Symmetric triangular window (linear rise then fall)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = np.minimum(i, n + 1 - i)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    """Cosine window w_i = (1‑cos(2π·i/(n‑1)))/2."""
    i = np.arange(n, dtype=np.float64)
    w = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) * 0.5
    w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred at the middle, sigma as a fraction of n."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Sub‑gradient / mirror descent -----------------------------------------
# ----------------------------------------------------------------------


def _subgradient_descent_active(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    eval_interval: int = 80,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact L∞‑convolution objective.
    Gradient is taken from *all* entries attaining the maximal value (within
    a tiny tolerance).  Exponentiated‑gradient updates keep the variables
    non‑negative; a Euclidean simplex projection restores Σa = 1.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # active set: indices whose value is within 1e‑12 of the maximum.
        # The tolerance is absolute because the scale of conv is ~1/n.
        active = np.where(conv >= maxc - 1e-12)[0]
        # accumulate sub‑gradient from the active shifts
        grad = np.zeros_like(a)
        for idx in active:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]
        # decaying step‑size
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # exponentiated‑gradient step
        # project back onto the simplex Σa = 1
        a = _project_onto_simplex(a, s=1.0)

        # occasional exact evaluation (cheap because we already have conv)
        if t % eval_interval == 0:
            cur = 2.0 * n * maxc               # same as _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad iter={t} new global best={best_score:.6f}"
                )
        # a safety cap – it is cheap to stop after a large number of steps
        if t >= 400_000:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy polishing ------------------------------------------------------
# ----------------------------------------------------------------------


def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a modest amount of mass from one position to another.
    The move is kept only if it improves the exact objective.
    """
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random fraction of the source mass (up to 40 %)
        delta = a[i] * np.random.rand() * 0.4
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method mixes analytic seeding, an accurate mirror‑descent on the active
    set of maximal convolution entries, and an aggressive final greedy polish.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # a tiny safety margin – we never want to overrun the 1000 s limit
    deadline = start + budget_s - 2.0

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Build a rich pool of analytic seeds.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 16000, 32000]
    # p‑grid for the beta / edge families (denser near 0 where optimum lives)
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.5, 14, base=10),
            np.logspace(-0.5, 0.0, 14, base=10),
            np.logspace(0.0, 0.7, 10, base=10),
        ]
    )
    lambda_grid = np.linspace(0.0, 1.0, 7)          # mixing with uniform
    sigma_grid = np.linspace(0.05, 0.30, 7)        # Gaussian width (as fraction of n)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] enumerating families for length {n}")

        # ----- deterministic families -------------------------------------------------
        families = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangle(n),
            _distribution_cosine(n),
        ]
        for base in families:
            sc = _objective(base)
            top_candidates.append((sc, base.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = base.copy()
                print(f"[{time.time():.1f}] new global best (det) len={n} score={sc:.6f}")

        # ----- parametrised families -------------------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)

            for base in (beta, edge):
                for lam in lambda_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best (param) len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # ----- Gaussian windows ------------------------------------------------------
        for sigma_frac in sigma_grid:
            if time.time() >= deadline:
                break
            cand = _distribution_gaussian(n, sigma_frac)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(
                    f"[{time.time():.1f}] new global best (gauss) len={n} σ={sigma_frac:.3f} score={sc:.6f}"
                )

        # ----- random Dirichlet seeds ------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            # α drawn log‑uniformly between 0.01 and 2.0
            log_alpha = rng.uniform(np.log10(0.01), np.log10(2.0))
            alpha = 10 ** log_alpha
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] new global best (rand‑dir) len={n} α={alpha:.4f} score={sc:.6f}"
                )

        # ----- sparse random seeds ---------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] new global best (sparse) len={n} score={sc:.6f}"
                )

        # retain only a modest number of the very best seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]   # keep 40 to give enough diversity

    # ------------------------------------------------------------------
    # 2.  If a former construction exists, inject it as a seed.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.size)
            else:
                prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen but ensures we have a candidate.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Intensive sub‑gradient descent on the most promising seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        # take the 6 best seeds (more would fragment the time budget)
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] launching sub‑gradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_active(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                eval_interval=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Final greedy polishing (mass‑transfer moves).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy polishing")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=None,                 # run until time expires
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after polishing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Report and return.
    # ------------------------------------------------------------------
    print(
        f"Final best objective = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # ``evaluate_sequence`` will perform its own clipping / normalisation.
    return overall_best_seq.tolist()
