"""
Improved search for a non‑negative step‑function (a list of floats) that
minimises the supplied `evaluate_sequence` score.

Algorithm
---------
1. **Broad parametric sweep** – generate many candidates using a wide
   collection of parametric families (uniform, arcsine, symmetric /
   asymmetric Beta, edge‑power, Gaussian and triangular shapes,
   Dirichlet draws with several concentration parameters).  All candidates
   are normalised to sum = 1 and scored with a fast FFT based proxy
   (`_max_convolution_score`).  The best candidate becomes the starting
   point.

2. **Soft‑max mirror‑descent (SM‑MD)** – minimise a smooth
   log‑sum‑exp approximation of the max‑convolution.
   The iterate lives on the simplex (exponentiated‑gradient update
   followed by Euclidean projection).  The temperature (`α`) is raised
   from 1 to 1e6, the learning‑rate decays from 0.8 to 0.005, and the
   procedure runs as long as the time budget permits.

3. **Gradient‑guided mass‑transfer hill‑climbing** – repeatedly move a
   small amount of probability mass from the index with the largest
   sub‑gradient (i.e. the entry that contributes most to the current
   peak) to a low‑gradient index.  The step size is adapted
   automatically; every successful move updates the best solution.

4. **Simulated‑annealing polish** – random pairwise mass transfers are
   proposed; improvements are always kept, while occasional worsening
   moves are accepted with probability `exp(-Δ/Τ)` where `Τ` follows a
   geometric cooling schedule.  This stage helps to escape shallow
   plateaus.

All stages respect the overall wall‑clock budget (`budget_s`).  The code
uses only NumPy and the supplied `evaluate_sequence`.  It runs on a
single core and returns the best list found.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions -------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    Fast proxy for `evaluate_sequence`.  The vector is assumed to be
    normalised to sum = 1, therefore the returned value equals the
    value that `evaluate_sequence` would give (scaling cancels out).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families (all return a vector with sum = 1) -----------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n, p):
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15            # avoid exact zeros
    w /= w.sum()
    return w


def _distribution_gaussian(n, sigma_rel):
    """Discrete Gaussian (mean centred) with σ = sigma_rel·n."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = max(1e-9, sigma_rel * n)
    w = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    w += 1e-15
    w /= w.sum()
    return w


def _distribution_triangle(n):
    """Symmetric triangular distribution (peak in the centre)."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    w = np.maximum(w, 1e-15)
    w /= w.sum()
    return w


def _distribution_dirichlet(n, concentration):
    """Draw from a Dirichlet with constant concentration."""
    return np.random.dirichlet(np.full(n, concentration))

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function that minimises `evaluate_sequence`.

    The function returns a plain Python list of non‑negative floats.
    """
    # ------------------------------------------------------------------
    # Initialise -------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a safety margin
    eps = 1e-15

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration -------------------------------------------
    # a mix of lengths (including a few small ones) and families
    lengths = [1024, 1500, 2000, 2500, 3000, 3500, 4000,
               5000, 6000, 8000, 12000, 20000]

    # grids for Beta parameters

        np.linspace(0.02, 0.20, 12),
        np.linspace(0.20, 0.80, 20),
        np.linspace(0.80, 1.50, 15)
    ])
    q_grid = np.concatenate([
        np.linspace(0.02, 0.20, 12),
        np.linspace(0.20, 0.80, 20),
        np.linspace(0.80, 1.50, 15)
    ])

    edge_ps = (0.3, 0.5, 0.7, 0.9, 1.2)
    sigma_rels = (0.07, 0.10, 0.13, 0.16)            # for Gaussian
    conc_params = (0.2, 0.5, 1.0, 2.0, 5.0)        # Dirichlet concentration

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families -----------------------------------
        families = [
            ("uniform",    _distribution_uniform(n)),
            ("arcsine",    _distribution_arcsine(n)),
            ("triangle",   _distribution_triangle(n)),
        ]

        # Edge‑power family
        for p in edge_ps:
            families.append((f"edge_pow(p={p:.2f})",
                             _distribution_edge_power(n, p)))

        # Gaussian family
        for sr in sigma_rels:
            families.append((f"gauss(σ={sr:.2f})",
                             _distribution_gaussian(n, sr)))

        # evaluate deterministic candidates
        for name, vec in families:
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} {name:15s} score={sc:.6f}")

        # ----- symmetric Beta -----------------------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} beta_sym(p={p:.3f}) score={sc:.6f}")

        # ----- asymmetric Beta (grid over p,q) -------------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            for q in q_grid:
                vec = _distribution_beta(n, p, q)
                vec = np.maximum(vec, eps); vec /= vec.sum()
                sc = _max_convolution_score(vec)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec.copy(), n
                    print(f"[{time.time():.1f}] n={n} beta(p={p:.3f},q={q:.3f}) score={sc:.6f}")

        # ----- Dirichlet draws (various concentrations) ---------------
        for conc in conc_params:
            if time.time() >= deadline:
                break
            for _ in range(3):
                vec = _distribution_dirichlet(n, conc)
                vec = np.maximum(vec, eps); vec /= vec.sum()
                sc = _max_convolution_score(vec)
                if sc < best_score:
                    best_score, best_seq, best_n = sc, vec.copy(), n
                    print(f"[{time.time():.1f}] n={n} dirichlet(c={conc:.1f}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Incorporate a previously known solution (if available) -------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), prev.size
                print(f"[{time.time():.1f}] previous best    score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard – should never happen, but keep a sensible fallback
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Soft‑max mirror descent (SM‑MD) -------------------------------
    a = best_seq.copy()                     # always kept on the simplex
    n = a.shape[0]

    remaining = deadline - time.time()
    # we will spend roughly half the remaining time on this phase
    md_time = remaining * 0.45
    it_per_sec = 1.0 / 0.0015               # empirical ≈ 1500 it/s for n≈4000
    max_iters = max(5_000, int(md_time * it_per_sec))

    alpha_start, alpha_end = 1.0, 1e6
    lr_start, lr_end = 0.8, 0.005

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        t = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights (stable)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth objective
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        # bookkeeping
        if it % max(1, max_iters // 20) == 0 or it == max_iters - 1:
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] SM‑MD it={it}/{max_iters} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Gradient‑guided mass‑transfer hill‑climbing --------------------
    a = best_seq.copy()
    n = a.shape[0]
    cur_sc = best_score
    tol = 1e-12

    while time.time() < deadline:
        conv = _fft_self_convolve(a)
        M = conv.max()
        # active set (indices attaining the current maximum)
        active = np.flatnonzero(conv >= M - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0 / max(1, active.size)

        # sub‑gradient of the true max‑convolution
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # pick source (largest grad) and target (smallest grad)
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            # fall back to a random alternative
            tgt = random.randint(0, n - 1)
            if tgt == src:
                continue

        # propose a transfer; magnitude limited to a fraction of source mass
        max_delta = min(a[src] * 0.5, 0.02)  # at most 2 % absolute or half of src
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_sc = _max_convolution_score(a2)
        if new_sc < cur_sc - 1e-14:
            a = a2
            cur_sc = new_sc
            best_score, best_seq = new_sc, a2.copy()
            print(f"[{time.time():.1f}] mass‑transfer improved score={new_sc:.6f}")
        # occasional random jitter to avoid stagnation
        if random.random() < 0.001:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            cur_sc = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing polish ------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    cur_sc = best_score

    # geometric cooling schedule
    T0 = max(1e-3, (cur_sc * 0.02))   # start relatively warm
    T_end = 1e-8
    total_sa_iters = int((deadline - time.time()) / 0.0008)  # very cheap moves
    if total_sa_iters < 1:
        total_sa_iters = 1

    for it in range(total_sa_iters):
        if time.time() >= deadline:
            break
        T = T0 * (T_end / T0) ** (it / total_sa_iters)

        # pick a random ordered pair (src→tgt) with bias towards large grad
        src = random.randint(0, n - 1)
        tgt = random.randint(0, n - 2)
        if tgt >= src:
            tgt += 1  # ensure src != tgt

        # magnitude of transfer: a fraction of source mass
        max_delta = min(a[src] * 0.4, 0.03)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        new_sc = _max_convolution_score(a2)
        Δ = new_sc - cur_sc
        if Δ < -1e-14 or random.random() < math.exp(-Δ / max(T, 1e-12)):
            a = a2
            cur_sc = new_sc
            if new_sc < best_score - 1e-14:
                best_score, best_seq = new_sc, a2.copy()
                print(f"[{time.time():.1f}] SA it={it}/{total_sa_iters} new best {new_sc:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final sanity check (optional)
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
