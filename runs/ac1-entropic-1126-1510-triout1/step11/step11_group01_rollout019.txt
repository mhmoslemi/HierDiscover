"""
A more aggressive search for the step‑function that minimises
evaluate_sequence().  The algorithm expands the family pool (Gaussian,
binomial‑smoothed, many signal‑processing windows), uses an exact
sub‑gradient (the shift attaining the maximal autocorrelation), performs
exponentiated‑gradient updates with a decaying step‑size, and finishes
with a cheap simulated‑annealing pairwise‑mass‐transfer polish.

The code respects the 1000 s budget, keeps only the best few seeds,
and tries to beat the previous bound 1.509950.  It works with only
standard libraries (numpy) and optionally scipy.signal if available.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    # next power of two ≥ 2n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective_exact(a: np.ndarray) -> float:
    """Exact evaluate_sequence objective for a (no explicit clipping)."""
    n = len(a)
    if n == 0:
        return float("inf")
    b = _fft_self_convolve(a)
    maxb = b.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    return 2.0 * n * maxb / (sum_a ** 2)

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian‑like shape; sigma is a fraction of the length."""
    sigma = max(sigma_frac * n, 1e-12)
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _distribution_binomial_smooth(n: int, steps: int) -> np.ndarray:
    """Repeated convolution with [0.5,0.5]; mimics a binomial smoothing."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    kernel = np.array([0.5, 0.5], dtype=np.float64)
    for _ in range(steps):
        a = np.convolve(a, kernel, mode="same")
        a = np.clip(a, 0, None)
        a /= a.sum()
    return a


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with the given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_dirichlet(
    n: int, rng: np.random.Generator, alpha: float

# Exact sub‑gradient descent (single active shift)
def _subgradient_descent_exact(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    eval_interval: int = 100,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent using the exact gradient of the active (max) convolution term.
    """
    n = len(a)
    a = a / a.sum()
    t = 0
    while time.time() < deadline:
        t += 1

        # convolution and index of the active shift
        conv = _fft_self_convolve(a)
        idx_max = int(conv.argmax())
        shift = idx_max - (n - 1)                     # shift ∈ [-(n‑1), n‑1]

        # build sub‑gradient of max(conv)
        grad = np.zeros_like(a)
        if shift >= 0:
            # overlap region length = n‑shift
            grad[: n - shift] += a[shift:]
            grad[shift:] += a[: n - shift]
        else:
            s = -shift
            grad[s:] += a[: n - s]
            grad[: n - s] += a[s:]

        # exponentiated‑gradient update
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective_exact(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

        # safety cap
        if t >= 300000:
            break
    return best_seq, best_score

# Simulated‑annealing pairwise mass transfer
def _annealing_pairwise(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Randomly moves a fraction of mass between two positions, keeping
    the move if it improves or with a probability that decays with temperature.
    """
    n = a.shape[0]
    T = best_score * 0.1 if best_score > 0 else 0.1
    decay = 0.995
    it = 0
    while time.time() < deadline:
        it += 1
        # pick two distinct indices, i is source
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # propose a delta; explore a few candidates
        delta = a[i] * rng.random() * 0.5          # up to 50 % of source mass
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective_exact(new_a)
        dE = cur - best_score
        accept = False
        if dE < -1e-14:
            accept = True
        else:
            prob = math.exp(-dE / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = new_a
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] anneal i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
                )
        # temperature schedule
        T *= decay
        if it % 5000 == 0 and T < 1e-12:
            T = best_score * 0.1
    return best_seq, best_score
