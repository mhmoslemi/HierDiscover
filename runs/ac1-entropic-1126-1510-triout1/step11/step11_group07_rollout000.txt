"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The algorithm proceeds in four phases:

1. **Broad sweep** – try many analytic families (arcsine, Beta,
   edge‑biased power‑law, exponential decay, sinusoid, Dirichlet draws)
   for a set of lengths ranging from a few thousand up to ~300 k.
   The best few (by the exact objective) are kept.

2. **Mirror‑descent on a smooth‑max** – starting from each of the
   kept candidates we run exponentiated‑gradient (KL‑mirror) steps on
   the soft‑max approximation of the max‑convolution.  The temperature
   and learning‑rate are annealed so that the optimisation moves from
   exploration to a fine‑grained search.

3. **Projected Adam on the true max‑convolution** – a few Adam‑style
   sub‑gradient steps (with back‑tracking line‑search) are applied to
   the exact max‑convolution objective.

4. **Fine‑grained hill‑climber** – tiny mass moves from the index that
   contributes most to the current peak towards the globally smallest
   entry, repeatedly, with occasional jitter to escape plateaus.

All operations are pure NumPy, use FFT‑based convolutions (O(n log n)),
and respect the global time‑budget (default 1000 s).  The best vector
found is returned as a plain Python list.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Scoring (exact objective when sum(a)=1)
# ----------------------------------------------------------------------


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a probability vector a (sum=1) returns the exact value produced by
    `evaluate_sequence`: 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """Random Dirichlet draw with edge‑bias."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


def _distribution_sin(n: int) -> np.ndarray:
    """Sinusoidal shape, positive everywhere."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_exp_decay(n: int, decay: float) -> np.ndarray:
    """Exponential decay from the centre."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-decay * np.abs(i - centre))
    vals /= vals.sum()
    return vals

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float,
                           eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective  f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector and the true max of conv.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient wrt a (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    # exponentiated‑gradient update
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One Adam‑style projected sub‑gradient step on the true max‑convolution.
    Returns updated a, updated velocity, and the current max value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices achieving the max, tolerance‑wise)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Tiny mass‑transfer hill‑climber.  Repeatedly moves a small amount of
    probability mass from the entry that participates most in the peak
    towards the globally smallest entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())                # location of the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) giving the largest product a_i*a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            # fallback to second‑smallest
            target = np.argsort(a)[1]

        # move a small random fraction, but never more than 2 % of the source
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional jitter to shake the system
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The function respects a hard time budget (default 1000 s) and returns the
    best sequence found as a plain Python list.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 1.0   # safety margin

    eps = 1e-12

    # ------------------------------------------------------------------
    # Load a previously discovered construction if available
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    try:
        # `height_sequence_1` may be defined in the global scope by the runner
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            best_seq = prev.copy()
            best_score = _max_convolution_score(best_seq)
            print(f"[{time.time():.1f}] Loaded previous best: {best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration over several lengths and families
    # ------------------------------------------------------------------
    lengths = [4000, 8000, 20000, 40000, 80000, 160000, 320000]
    p_grid = np.linspace(0.10, 0.50, 17)          # symmetric Beta(p,p)
    asym_q = 0.5
    alpha_grid = [0.15, 0.20, 0.25, 0.33, 0.40, 0.45, 0.50]  # edge‑power
    decay_grid = [0.001, 0.003, 0.006, 0.010, 0.020]        # exponential decay
    K = 12                      # keep this many candidates
    top_candidates = []          # (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- deterministic families -----------------------------------
        # arcsine
        vec = _distribution_arcsine(n)
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        top_candidates.append((sc, vec.copy()))
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] n={n} arcsine               score={sc:.6f}")

        # uniform (useful baseline)
        vec = _distribution_uniform(n)
        sc = _max_convolution_score(vec)
        top_candidates.append((sc, vec.copy()))
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] n={n} uniform               score={sc:.6f}")

        # symmetric Beta(p,p)
        for p in p_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f})       score={sc:.6f}")

        # asymmetric Beta(p,q) with q fixed
        for p in p_grid:
            if time.time() >= deadline:
                break
            # left‑heavy
            vec = _distribution_beta(n, p, asym_q)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f},q={asym_q})  score={sc:.6f}")
            # right‑heavy (swap)
            vec = _distribution_beta(n, asym_q, p)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} Beta(p={asym_q},q={p:.3f})  score={sc:.6f}")

        # edge‑biased deterministic power‑law
        for alpha in alpha_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_power(n, alpha)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edgePower(alpha={alpha:.2f})  score={sc:.6f}")

        # exponential decay from centre
        for decay in decay_grid:
            if time.time() >= deadline:
                break
            vec = _distribution_exp_decay(n, decay)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} expDecay(decay={decay:.3f})   score={sc:.6f}")

        # sinusoid
        vec = _distribution_sin(n)
        vec = np.maximum(vec, eps); vec /= vec.sum()
        sc = _max_convolution_score(vec)
        top_candidates.append((sc, vec.copy()))
        if sc < best_score:
            best_seq, best_score = vec.copy(), sc
            print(f"[{time.time():.1f}] n={n} sinusoid               score={sc:.6f}")

        # random edge‑biased Dirichlet draws
        for _ in range(3):
            if time.time() >= deadline:
                break
            beta_exp = random.uniform(0.2, 0.8)
            vec = _distribution_edge_dirichlet(n, beta_exp)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            top_candidates.append((sc, vec.copy()))
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edge‑Dirichlet(beta={beta_exp:.3f}) score={sc:.6f}")

    # keep only the K best candidates overall
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:K]

    if best_seq is None:
        # fallback: uniform of moderate size
        n0 = 20000
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max mirror descent (run on a few top candidates)
    # ------------------------------------------------------------------
    mirror_iters_per_candidate = 8000

    for idx, (init_score, init_vec) in enumerate(top_candidates[:3]):
        if time.time() >= deadline:
            break
        a = init_vec.copy()
        n = a.shape[0]

        # schedule based on remaining time
        remaining = deadline - time.time()
        est_one_it = 0.004 if n > 80000 else 0.002
        max_possible = int(remaining / est_one_it)
        iters = max(2000, min(mirror_iters_per_candidate, max_possible))

        print(f"[{time.time():.1f}] Mirror‑descent #{idx+1} n={n} iters={iters}")

        alpha_start, alpha_end = 0.5, 2e5
        lr_start, lr_end = 0.3, 0.001

        for it in range(iters):
            if time.time() >= deadline:
                break
            t = it / iters
            alpha = alpha_start * (alpha_end / alpha_start) ** t
            lr = lr_start * (lr_end / lr_start) ** t
            a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)

            # periodic exact evaluation
            if (it + 1) % max(1, iters // 10) == 0 or it == iters - 1:
                cur_score = 2.0 * n * cur_max
                if cur_score < best_score - 1e-12:
                    best_seq, best_score = a.copy(), cur_score
                    print(f"[{time.time():.1f}] Mirror #{idx+1} it={it} new best {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected Adam‑style sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta1 = 0.9
    step0 = 0.7
    sub_iter = 0

    while time.time() < deadline - 0.8:
        sub_iter += 1
        a, velocity, cur_max = _adam_subgrad_step(a, velocity,
                                                  step0, beta1, sub_iter, eps)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] Adam subgrad it={sub_iter} new best {cur_score:.6f}")

        # occasional random jitter
        if sub_iter % 700 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 4️⃣  Tiny mass‑transfer hill‑climber for final polishing
    # ------------------------------------------------------------------
    a = best_seq.copy()
    max_hill_iters = 6000
    a = _mass_transfer_hill_climb(a, max_hill_iters, eps)
    final_score = _max_convolution_score(a)
    if final_score < best_score:
        best_seq, best_score = a.copy(), final_score
        print(f"[{time.time():.1f}] Hill‑climber improved score to {final_score:.6f}")

    # ------------------------------------------------------------------
    # Return best sequence
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
