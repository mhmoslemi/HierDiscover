"""
Optimisation of a non‑negative step function for `evaluate_sequence`.

The algorithm proceeds in four stages within a hard time budget:

1. **Broad exploration** – generate many analytic families (arcsine,
   symmetric Beta, edge‑biased power‑law, cosine) for several lengths.
   The K best candidates (by the exact max‑convolution score) are kept.

2. **Smooth‑max mirror descent** – an exponentiated‑gradient (KL‑mirror)
   optimisation on the soft‑max approximation of the max‑convolution.
   Temperature α is increased geometrically while the learning‑rate is
   decreased, driving the solution toward the true max objective.

3. **Projected Adam‑style sub‑gradient descent** – direct minimisation
   of the exact max‑convolution using an Adam momentum scheme and a
   back‑tracking line‑search.

4. **Tiny mass‑transfer hill‑climber** – a final local search that moves
   a small amount of probability mass from the most offending pair
   towards the smallest entry, polishing the result.

All steps respect the simplex constraint (non‑negative, sum = 1) and
use FFT‑based convolution for speed.  The routine returns the best
sequence (as a plain Python list) found before the deadline.
"""

# ----------------------------------------------------------------------
# Linear‑algebra utilities

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact score returned by evaluate_sequence for a probability vector
    (i.e. sum(a)=1).  This is 2·n·max(a*a) where * is linear convolution.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w /= w.sum()
    return w

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p) on the grid { (i+0.5)/n }."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Use log‑space for stability when p<1
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_cosine(n: int) -> np.ndarray:
    """w_i ∝ sin(π (i+0.5)/n) – a smooth bell shape."""
    i = np.arange(n, dtype=np.float64)
    w = np.sin(np.pi * (i + 0.5) / n)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

# Optimisation primitives

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float, eps: float = 1e-12):
    """
    One KL‑mirror (exponentiated‑gradient) step on the smooth‑max
    approximation of the max‑convolution.
    Returns the updated probability vector and the exact max value of the
    (unsmoothed) convolution (for bookkeeping).
    """
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights (numerically stable)
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max, using the fact that
    # d/da (conv_k) = 2 * (a reversed convolve e_k)
    grad = 2.0 * _fft_convolve(a[::-1], w)[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    # exponentiated‑gradient update (mirror w.r.t. KL)
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One Adam‑style projected sub‑gradient step on the **exact** max‑convolution.
    Returns (new_a, new_velocity, new_max).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices attaining the maximum, with a tiny tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum update (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # back‑tracking line search on the feasible simplex
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    new_max = _fft_self_convolve(a_cand).max()
    return a_cand, vel, new_max


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Tiny, focused hill‑climber: repeatedly move a small chunk of mass from
    the entry that participates most in the current peak towards the
    globally smallest entry.  The move is only kept if it strictly lowers
    the convolution maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        # location of the current peak
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find pair (i,j) with maximal product a_i * a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        # choose source (the larger of the two) and target (global minimum)
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            # pick second smallest if clash
            target = np.argsort(a)[1]
        # magnitude of transfer – a modest fraction of source mass
        max_delta = min(a[source] * 0.5, 0.03)   # up to 3 % of source
        delta = max_delta * random.random()
        if delta <= eps:
            continue
        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
        # occasional jitter to avoid stagnation
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a
