def propose_candidate(seed=42, budget_s=1000):

    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 1.0 # safety

    lengths = [2000, 4000, 8000, 16000, 20000] maybe up to 25000.

    overall_best_seq = None

    overall_best_score = float('inf')

    # Use scanning across candidate families for each length but no need for all lengths; maybe only the smallest for initial.

    # We'll use arcsine as default for each length; optionally random Dirichlet if wanted.

    # We'll keep best_seq = None

    seq_cur = None

    for idx, n in enumerate(lengths):

        if time.time() >= deadline:
            break

        if seq_cur is None:
            # compute seed candidates for this length
            cand_list = [
                _distribution_arcsine(n),
                _distribution_uniform(n),
                _distribution_beta_sym(n, p=0.2),
                _distribution_beta_sym(n, p=0.5),
                _distribution_beta_sym(n, p=1.0),
                _distribution_beta_asym(n, alpha=0.2, beta=5.0),
                _distribution_two_blocks(n, block_frac=0.05),
                _distribution_random_sparse(n, rng, density=0.01),
                _distribution_dirichlet(n, rng, alpha=0.3)
            ]
            # evaluate quickly
            best_local_score = float('inf')
            best_local_seq = None
            for cand in cand_list:
                seq = np.maximum(cand, 0.0)
                seq_sum = seq.sum()
                if seq_sum == 0:
                    continue
                seq = seq / seq_sum
                sc = _objective(seq)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local_seq = seq
                    print(f"[{time.time():.1f}] length {n}, seed candidate score {sc:.6f}")
            seq_cur = best_local_seq.copy()
            best_score_cur = best_local_score
        else:
            # upsample previous best sequence to new length
            old_n = seq_cur.shape[0]
            x_old = np.linspace(0, 1, old_n, endpoint=False) + 0.5/old_n
            x_new = np.linspace(0, 1, n, endpoint=False) + 0.5/n
            seq_cur = np.interp(x_new, x_old, seq_cur)
            # re-normalize
            seq_cur = np.maximum(seq_cur, 0.0)
            seq_cur /= seq_cur.sum()
            best_score_cur = _objective(seq_cur)
            print(f"[{time.time():.1f}] upsampled to length {n}, init score {best_score_cur:.6f}")

        # allocate time slice for this length
        remaining_time = deadline - time.time()
        if remaining_time <= 0:
            break
        # allocate a portion for this length, let's give 40% of remaining for largest length but easier: allocate proportion = 1/ (len(lengths) - idx)
        slice_factor = 1.0 / (len(lengths) - idx)
        sub_deadline = time.time() + remaining_time * slice_factor

        # Stage 1: smooth max descent
        seq_cur, best_score_cur = _smooth_max_descent(seq_cur.copy(),
                                                      sub_deadline,
                                                      best_seq=seq_cur,
                                                      best_score=best_score_cur,
                                                      eta0=0.8,
                                                      alpha0=1.0,
                                                      alpha_max=80.0,
                                                      max_iter=20000,
                                                      eval_interval=200)
        # Stage 2: Adam multimax descent
        seq_cur, best_score_cur = _adam_multimax_descent(seq_cur.copy(),
                                                         sub_deadline,
                                                         best_seq=overall_best_seq if overall_best_seq is not None else seq_cur,
                                                         best_score=overall_best_score,
                                                         lr=0.1,
                                                         max_iter=30000,
                                                         eval_interval=200)

        # Stage 3: local swap polishing
        seq_cur, best_score_cur = _pairwise_swap_local(seq_cur.copy(),
                                                       sub_deadline,
                                                       best_seq=overall_best_seq if overall_best_seq is not None else seq_cur,
                                                       best_score=overall_best_score)

        # Update overall best
        if best_score_cur < overall_best_score:
            overall_best_score = best_score_cur
            overall_best_seq = seq_cur.copy()
            print(f"[{time.time():.1f}] Updated global best score {overall_best_score:.6f} at length {n}")

    # If still None, fallback uniform of some moderate length
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f}, length={len(overall_best_seq)}")
    # Transform to list of floats for return
    return overall_best_seq.tolist()

def _adam_multimax_descent(a: np.ndarray,
                          deadline: float,
                          best_seq: np.ndarray,
                          best_score: float,
                          lr: float = 0.1,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8,
                          max_iter: int = 20000,
                          eval_interval: int = 200) -> Tuple[np.ndarray, float]:

    n = a.shape[0]
    log_a = np.log(a + eps)   # to avoid log(0)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    while t < max_iter and time.time() < deadline:
        t+=1
        # compute conv and active mask
        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-12
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)   # shape n

        # gradient w.r.t log a
        grad_log = a * grad   # elementwise

        # Adam updates
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # compute new a
        a = np.exp(log_a)
        # projection onto simplex
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum

        # occasionally evaluate objective
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new best {best_score:.6f}")

    return best_seq, best_score

"""
Hybrid optimiser for the max‑convolution step‑function problem.

The algorithm proceeds in three main phases:
1. **Seeding** – a small collection of analytically‑motivated probability
   vectors (arcsine, Beta, two‑block spikes, random Dirichlet, etc.) is
   generated for a modest length (≈2000).  The best seed is kept.
2. **Hierarchical refinement** – the current best vector is repeatedly
   up‑sampled to larger lengths (4000 → 8000 → 16000 → 20000) and
   optimised at each scale.
   * A smooth‑max (log‑sum‑exp) surrogate is minimised using an
     exponentiated‑gradient update with a projection onto the simplex.
   * Afterwards an Adam‑style optimiser is run on the true multimax
     sub‑gradient (average over the active set attaining the maximal
     convolution value).  Adam works on the log‑domain to keep all
     entries positive.
3. **Local polishing** – a cheap greedy pairwise‑swap walk finishes the
   search.

All steps respect the overall time budget (default 1000 s) and keep the
sequence non‑negative with unit sum (the objective is scale‑invariant,
so we can fix the sum).  The routine returns a plain ``list[float]``
ready to be fed into ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient 2·a⋆w where w (length 2·n‑1) sums to 1.
    Implemented via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (scale invariant)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families of probability vectors (all sum to 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_max_descent(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        eta0: float = 0.7,
                        alpha0: float = 1.0,
                        alpha_max: float = 50.0,
                        max_iter: int = 20000,
                        eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth approximation
        f_α(a) = (1/α)·log Σ_i exp(α·conv_i(a)),
    using exponentiated‑gradient updates and projection onto the simplex.
    The temperature α is increased geometrically up to ``alpha_max``.
    """
    n = a.shape[0]
    t = 0
    alpha = alpha0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        grad = _gradient_from_weights(a, w)

        eta = eta0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)          # exponentiated‑gradient step
        a = _project_onto_simplex(a)          # keep on simplex

        if t % 300 == 0 and alpha < alpha_max:   # raise temperature slowly
            alpha = min(alpha * 1.25, alpha_max)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t:,} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _adam_multimax_descent(a: np.ndarray,
                           deadline: float,
                           best_seq: np.ndarray,
                           best_score: float,
                           lr: float = 0.1,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-8,
                           max_iter: int = 30000,
                           eval_interval: int = 200) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser on the true multimax objective.
    Works in the log‑domain (x = log a) so that positivity is automatic.
    At each step the sub‑gradient is the average over all indices attaining the
    maximal convolution value (the classic “multimax” sub‑gradient).
    """
    n = a.shape[0]
    log_a = np.log(a + eps)          # initialise log‑variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)          # ∂/∂a
        grad_log = a * grad                         # chain rule for log‑domain

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.exp(log_a)
        a = _project_onto_simplex(a)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t:,} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Greedy pairwise mass‑transfer walk, accepting only improving moves."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

        if overall_best_seq is None:

                _distribution_random_sparse(n, rng, density=0.01),

    return overall_best_seq.tolist()
