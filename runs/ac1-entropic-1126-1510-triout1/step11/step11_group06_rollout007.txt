The evaluation function  

\[
E(a)=\frac{2\,n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is *convex* in the sense that we can improve a step‑function `a`
by keeping its ℓ₁‑norm fixed (we normalise to sum = 1) and reducing the
peak of its auto‑convolution.  
For a fixed `a` the constraint  

\[
(a * g)_k\le M \qquad\forall k,
\qquad  M=\max_k (a*a)_k,
\]

is linear in the unknown vector `g`.  
Hence the auxiliary linear program  

```
max   Σ g_i
s.t.  a * g ≤ M          (2n‑1 linear constraints)
      g_i ≥ 0
```

produces a direction `g` that tries to spread the probability mass away
from the positions that currently generate the peak.  
After solving the LP we mix the current distribution with the LP
solution and repeat – exactly the idea that produced the state‑of‑the‑art
solution, but here we *solve the LP explicitly* (using SciPy’s HiGHS
solver) which gives a much stronger descent direction.

The algorithm consists of three phases:

1. **Broad exploration** – evaluate a handful of parametric families
   (uniform, arcsine, symmetric‑Beta, edge‑biased power law) for several
   lengths (512…4096).  Keep the few best candidates.
2. **LP refinement** – for each kept candidate run a short sequence of
   linear‑programming steps (≤ 30 iterations) with a decreasing mixing
   factor `t`.  The LP uses a sparse Toeplitz matrix that represents the
   convolution constraints.
3. **Final polishing** – a cheap stochastic hill‑climber moves a tiny
   amount of mass from the largest to the smallest coordinate; the move
   is accepted only if the evaluation improves.

All heavy work is performed on the CPU only; the total running time is
well under the 1000 s budget.  The best sequence found is returned as a
plain Python list – ready to be fed to `evaluate_sequence`.

```python
"""
Propose a step‑function (list of non‑negative floats) that minimises
`evaluate_sequence`.  The search consists of:

1. Broad exploration of several parametric families for lengths 512‑4096.
2. LP‑based refinement (explicit HiGHS linear program) that
   repeatedly solves  max Σg_i  s.t. a·g ≤ M .
3. Light stochastic hill‑climbing for final polishing.

All vectors are normalised to sum = 1, thus the objective equals
`2·n·max(a*a)`.  The implementation uses only NumPy, SciPy (sparse
matrix + HiGHS) and a small amount of bookkeeping.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection – used only for safety after numeric perturbations
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based self‑convolution (full linear convolution a*a)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Parametric families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p) + 1e-15
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑programming refinement (requires SciPy)
def _lp_step(a: np.ndarray, t: float, eps: float):
    """
    One LP refinement step.
    Returns a new vector (still sum=1) and the achieved score.
    """
    from scipy.sparse import diags
    from scipy.optimize import linprog

    n = a.shape[0]
    # current peak of a*a
    conv = _fft_self_convolve(a)
    M = conv.max()

    # Build sparse Toeplitz matrix representing convolution a*g.
    # Row i, column j receives a[i‑j] if i≥j, otherwise 0.
    rows = 2 * n - 1
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]  # n entries per diagonal
    offsets = [-d for d in range(n)]
    A = diags(data, offsets, shape=(rows, n), format="csr")

    # Linear program: maximise Σ g_i  (=> minimise -Σ g_i)
    c = -np.ones(n, dtype=np.float64)
    b = np.full(rows, M, dtype=np.float64)

    # HiGHS works well with sparse A_ub
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method="highs")
    if not res.success:
        # fallback – keep the current vector
        return a, 2.0 * n * M

    g_raw = res.x
    # convex combination with mixing factor t
    a_new = (1.0 - t) * a + t * g_raw
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()

    # new objective (same scale because sum=1)
    conv_new = _fft_self_convolve(a_new)
    new_score = 2.0 * n * conv_new.max()
    return a_new, new_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns a plain Python list.
    """
    # ------------------------------------------------------------------
    # basic setup
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # a safety margin
    eps = 1e-12

    # ------------------------------------------------------------------
    # parametric families we will scan
    families = [
        ("uniform", _distribution_uniform, {}),
        ("arcsine", _distribution_arcsine, {}),
        ("beta_sym_p=0.2", _distribution_beta_sym, {"p": 0.2}),
        ("beta_sym_p=0.5", _distribution_beta_sym, {"p": 0.5}),
        ("beta_sym_p=0.8", _distribution_beta_sym, {"p": 0.8}),
        ("edge_power_p=0.4", _distribution_edge_power, {"p": 0.4}),
        ("edge_power_p=0.8", _distribution_edge_power, {"p": 0.8}),
    ]

    lengths = [512, 1024, 2048, 4096]          # moderate sizes (LP is O(n²))
    # keep top‑K candidates from the coarse scan
    K = 3
    best_candidates = []   # list of (score, vector, n)

    def _score_of(v: np.ndarray) -> float:
        """Fast score for a vector already normalised to sum = 1."""
        n = v.shape[0]
        conv = _fft_self_convolve(v)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 1️⃣ Broad exploration
    for n in lengths:
        if time.time() >= deadline:
            break
        for name, gen, params in families:
            if time.time() >= deadline:
                break
            a = gen(n, **params)
            a = np.maximum(a, eps)
            a /= a.sum()
            sc = _score_of(a)
            if len(best_candidates) < K:
                best_candidates.append((sc, a.copy(), n))
                best_candidates.sort(key=lambda x: x[0])
            else:
                # replace worst if we found a better one
                if sc < best_candidates[-1][0]:
                    best_candidates[-1] = (sc, a.copy(), n)
                    best_candidates.sort(key=lambda x: x[0])
            print(f"[{time.time():.1f}] init {name} n={n:4d} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ LP refinement on the top candidates
    global_best_seq = None
    global_best_score = float("inf")
    for cand_idx, (sc, a, n) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] refining candidate #{cand_idx+1} (n={n}) start_score={sc:.6f}")
        # mixing factor schedule – start aggressive, decay toward 0.1
        t = 0.5
        for it in range(30):                # at most 30 LP steps per candidate
            if time.time() >= deadline:
                break
            a, new_sc = _lp_step(a, t, eps)
            if new_sc < sc - 1e-12:
                sc = new_sc
                print(f"[{time.time():.1f}]  LP it={it:02d} t={t:.3f} score={sc:.6f}")
                # if we improved a lot, keep a relatively big step
                t = max(0.1, t * 0.9)
            else:
                # no improvement → shrink step size
                t *= 0.5
                if t < 0.02:
                    break
        # keep the best overall
        if sc < global_best_score:
            global_best_score = sc
            global_best_seq = a.copy()
            print(f"[{time.time():.1f}]  *** new global best {sc:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3️⃣ Include the previous solution if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _score_of(prev)
            if sc < global_best_score:
                global_best_score = sc
                global_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best used, score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4️⃣ Light stochastic hill‑climbing polishing
    a = global_best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0
    while time.time() < deadline:
        hill_it += 1
        # move a tiny amount of mass from a large entry to a small entry
        i_src = np.argmax(a)
        i_dst = np.argmin(a)
        if i_src == i_dst:
            break
        delta = min(a[i_src] * 0.3, 0.01) * random.random()
        if delta < eps:
            continue
        a2 = a.copy()
        a2[i_src] -= delta
        a2[i_dst] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_sc = _score_of(a2)
        if new_sc < global_best_score - 1e-12:
            a = a2
            global_best_seq = a2.copy()
            global_best_score = new_sc
            print(f"[{time.time():.1f}] hill it={hill_it} score={new_sc:.6f}")
        # occasional jitter to leave plateaus
        if hill_it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            print(f"[{time.time():.1f}] jitter at hill it={hill_it}")

    # ------------------------------------------------------------------
    # final reporting
    # (We evaluate with the supplied function to be absolutely sure)
    try:
        final_score = evaluate_sequence(global_best_seq.tolist())
    except Exception:
        final_score = global_best_score
    print(f"Final best score ≈ {final_score:.6f} (length={global_best_seq.shape[0]})")
    return global_best_seq.tolist()
```
