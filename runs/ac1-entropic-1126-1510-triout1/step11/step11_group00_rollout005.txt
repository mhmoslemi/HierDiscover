"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

The algorithm uses a *continuation* approach on a smooth approximation of the
max‑convolution objective.  Starting from a few diverse seeds we optimise the
exact objective by minimising the soft‑max of the self‑convolution using an
Adam‑style exponentiated‑gradient (θ‑parametrisation with a = softmax(θ)).
A temperature γ is slowly increased so that the smooth objective morphs into
the true max; this guides the optimiser through a well‑behaved landscape and
avoids getting stuck in poor local minima.

After the main optimisation a cheap pairwise‑swap hill‑climb and a tiny
simulated‑annealing walk are applied for final polishing.
The whole routine respects the supplied time budget (default 1000 s) and
returns a plain Python list; ``evaluate_sequence`` will handle the final
normalisation and clipping.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


def _softmax_weights(b: np.ndarray, gamma: float) -> np.ndarray:
    """Soft‑max over the entries of b with temperature γ (larger → closer to max)."""
    # stable softmax
    c = b.max()
    exps = np.exp(gamma * (b - c))
    return exps / exps.sum()


def _correlate_valid(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (valid mode):
    (a ⋆ w)[i] = Σ_j a_j * w_{i+j},  i = 0,…,len(a)‑1.
    Implemented via FFT, O(L log L) where L≈3·n.
    """
    n = a.shape[0]
    m = w.shape[0]               # m = 2·n‑1
    size = 1 << ((n + m - 1).bit_length())   # next power of two ≥ 3·n‑2
    Fa = np.fft.rfft(a, n=size)
    Fw = np.fft.rfft(w, n=size)
    corr_full = np.fft.irfft(np.conj(Fw) * Fa, n=size)
    return corr_full[:n]         # exactly n entries (valid part)

def _smooth_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr: float = 0.05,
    gamma_start: float = 8.0,
    gamma_max: float = 500.0,
    gamma_factor: float = 1.001,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    jitter_scale: float = 1e-5,
    stagnation_patience: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Optimises a probability vector a (sum = 1, a_i≥0) by minimising a
    smooth approximation of max(conv(a,a)).  The parametrisation is
    a_i = exp(θ_i) / Σ exp(θ).

    Returns the best sequence seen and its objective value.
    """
    n = a0.shape[0]
    # avoid exact zeros for the log‑parametrisation
    eps0 = 1e-12
    a = np.maximum(a0, eps0)
    a = a / a.sum()
    theta = np.log(a)                     # initialise θ

    m = np.zeros_like(theta)              # Adam first moment
    v = np.zeros_like(theta)              # Adam second moment
    it = 0
    no_improve = 0
    gamma = gamma_start
    rng = np.random.default_rng()

    while time.time() < deadline and it < 300_000:
        it += 1

        # ----- re‑compute a from θ (softmax, guaranteed sum = 1) -----
        max_th = theta.max()
        exp_th = np.exp(theta - max_th)
        a = exp_th / exp_th.sum()

        # ----- exact objective (for bookkeeping) -----
        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] smooth descent it={it} new best {best_score:.6f}")
        else:
            no_improve += 1

        # ----- smooth max via soft‑max -----
        b = _fft_self_convolve(a)                     # full self‑convolution (len 2n‑1)
        w = _softmax_weights(b, gamma)                # probability weights, sum‑to‑1

        # ----- gradient wrt a (cross‑correlation) -----
        grad_a = 2.0 * _correlate_valid(a, w)          # shape (n,)

        # ----- chain rule to θ (exponentiated‑gradient) -----
        avg = np.dot(a, grad_a)                       # 〈a,∇_a L〉
        grad_theta = a * (grad_a - avg)               # ∂L/∂θ

        # ----- Adam update on θ -----
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        theta -= step

        # ----- temperature schedule -----
        gamma = min(gamma * gamma_factor, gamma_max)

        # ----- occasional jitter to escape plateaus -----
        if no_improve >= stagnation_patience:
            theta += rng.normal(scale=jitter_scale, size=theta.shape)
            no_improve = 0

    return best_seq, best_score

def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Deterministic hill‑climb: move a random fraction of mass from i→j."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}")
    return best_seq, best_score


def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing‑style mass transfer."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget (default 1000 s) and returns a
    plain Python list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0       # small safety margin

    # ------------------------------------------------------------------
    # 0.  Determine a base length – reuse previous construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim != 1 or prev.size == 0:
            raise ValueError
        n = prev.size
        base_seq = prev / prev.sum()
        print(f"[{time.time():.1f}] loaded previous height_sequence_1 (n={n})")
    except Exception:
        n = 3000
        base_seq = np.full(n, 1.0 / n, dtype=np.float64)   # uniform fallback
        print(f"[{time.time():.1f}] no previous sequence – using uniform n={n}")

    # ------------------------------------------------------------------
    # 1. Build an initial pool of diverse seeds
    seeds: list[np.ndarray] = [base_seq]

    # Uniform and arcsine families
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))
    i = np.arange(n, dtype=np.float64)
    arcs = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcs /= arcs.sum()
    seeds.append(arcs)

    # Symmetric Beta with a few exponents
    for p in (0.2, 0.5, 1.0, 2.0):
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
        w /= w.sum()
        seeds.append(w)

    # Random Dirichlet – a handful of stochastic starts
    for _ in range(5):
        a = rng.dirichlet(np.full(n, rng.uniform(0.01, 2.0)))
        seeds.append(a)

    # Sparse random starts
    for _ in range(5):
        k = max(2, int(n * 0.008))
        idx = rng.choice(n, size=k, replace=False)
        vals = np.zeros(n, dtype=np.float64)
        vals[idx] = rng.random(k)
        vals /= vals.sum()
        seeds.append(vals)

    # ------------------------------------------------------------------
    # 2. Run the continuation‑gradient optimiser on the most promising seeds
    overall_best_seq = None
    overall_best_score = float("inf")
    # quick evaluation of all seeds, keep only the few best
    seed_scores = [(float(_objective(s)), s) for s in seeds]
    seed_scores.sort(key=lambda x: x[0])
    top_seeds = [s for _, s in seed_scores[:8]]   # keep eight promising starts

    remaining = deadline - time.time()
    per_seed = max(5.0, remaining / max(1, len(top_seeds)))   # at least a few seconds each
    print(f"[{time.time():.1f}] starting gradient descent on {len(top_seeds)} seeds")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        sub_dead = time.time() + per_seed
        best_seq, best_score = _smooth_gradient_descent(
            seed_vec.copy(),
            sub_dead,
            best_seq=overall_best_seq if overall_best_seq is not None else seed_vec,
            best_score=overall_best_score,
            lr=0.05,
            gamma_start=8.0,
            gamma_max=500.0,
            gamma_factor=1.001,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            jitter_scale=1e-5,
            stagnation_patience=5000,
        )
        if best_score < overall_best_score:
            overall_best_score, overall_best_seq = best_score, best_seq.copy()
            print(f"[{time.time():.1f}] after seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local deterministic refinement (pairwise swaps)
    if overall_best_seq is None:
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = _objective(overall_best_seq)

    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 4. Final cheap SA‑style local search
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final simulated‑annealing walk")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=0.008,
            decay=0.9997,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
