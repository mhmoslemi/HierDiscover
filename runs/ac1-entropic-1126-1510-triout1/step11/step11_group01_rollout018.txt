"""
A fast, multi‑phase optimiser for the “step‑function” objective.

Phases
------
1.  Broad exploration – a rich family of analytic shapes (uniform, triangular,
    cosine, exponential, β‑distributions, edge‑weighted, Chebyshev/Kaiser
    windows, Dirichlet, sparse, and an up‑sampled copy of the previous best)
    on several lengths (powers of two up to 32768).  The 40 best candidates are
    retained.

2.  Mirror‑descent – exponentiated‑gradient updates on the exact max‑convolution
    objective.  The update uses the gradients coming from the *top_k* largest
    convolution entries (k=10) and a decaying learning‑rate.  The 10 best seeds
    are refined for a roughly equal share of the remaining time.

3.  Annealed polishing – random pairwise mass transfers are performed
    repeatedly; moves that improve the objective are kept, otherwise they are
    accepted with a temperature‑dependent probability.  This helps escape any
    remaining shallow local minima.

All vectors are kept on the simplex (non‑negative, Σ=1) via Euclidean projection.
The routine respects the 1000 s budget (leaving a 2 s safety margin) and returns
a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # smallest power‑of‑2 ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1.
    The input is assumed non‑negative.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_triangular(n: int) -> np.ndarray:
    """Symmetric triangle, peak in the centre."""
    x = np.arange(n, dtype=np.float64)
    peak = (n - 1) / 2.0
    w = 1.0 - np.abs(x - peak) / peak
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_cosine(n: int) -> np.ndarray:
    """Cosine window (non‑negative, symmetric)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 0.5 * (1 + np.cos(2 * np.pi * i / n))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_exponential(n: int, decay: float = 0.005) -> np.ndarray:
    """Monotonically decreasing exponential, mirrored to stay symmetric."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-decay * i)
    w = (w + w[::-1]) / 2.0
    w /= w.sum()
    return w


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    a = rng.dirichlet(np.full(n, alpha))
    a /= a.sum()
    return a


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex blend of a base distribution with the uniform one."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

def _subgradient_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 1.0,
    top_k: int = 10,
    eval_interval: int = 60,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient / mirror descent on the exact max‑convolution objective.
    Only the `top_k` largest convolution entries contribute to the gradient.
    """
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        # indices of the largest entries (including the centre)
        idxs = np.argpartition(-conv, range(top_k))[:top_k]

        grad = np.zeros_like(a)
        for idx in idxs:
            ind = idx - np.arange(n)
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)            # decaying stepsize
        a = a * np.exp(-eta * grad)        # exponentiated‑gradient step
        a = _project_onto_simplex(a)        # back onto the simplex

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

        if t >= 300000:                     # hard safety cap
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
def _annealing_refine(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style local refinement (pairwise mass transfers)."""
    n = a.shape[0]
    cur_score = best_score
    # start temperature – a modest fraction of the current gap to the theoretical lower bound (1)
    T = max(0.001, (cur_score - 1.0) * 0.05)

    while time.time() < deadline:
        # pick a source index proportional to its current mass (more mass → more chance to move)
        src = rng.choice(n, p=a)
        if a[src] < 1e-14:
            continue
        # pick a distinct destination uniformly
        dst = rng.integers(n - 1)
        if dst >= src:
            dst += 1

        # move a random fraction of the source mass (up to 30 %)
        delta = a[src] * rng.random() * 0.30
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        new_score = _objective(new_a)
        improvement = new_score - cur_score

        if improvement < -1e-14:                     # strict improvement
            a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = new_a.copy()
                print(f"[{time.time():.1f}] anneal improved to {best_score:.6f}")
        else:
            # accept with probability exp(-Δ/T)
            if rng.random() < math.exp(-improvement / max(T, 1e-12)):
                a = new_a
                cur_score = new_score

        # geometric cooling
        T *= 0.9999
        if T < 1e-8:
            T = 1e-8
    return best_seq, best_score

    )
    lambda_grid = np.linspace(0.0, 1.0, 6)                # mixing with uniform
    atten_grid = np.linspace(30, 150, 7)                # Chebyshev dB
    kaiser_grid = np.linspace(0.0, 14.0, 8)              # Kaiser β

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # deterministic families -------------------------------------------------
        families = [
            _distribution_uniform(n),
            _distribution_triangular(n),
            _distribution_cosine(n),
            _distribution_exponential(n, decay=0.002),
            _distribution_arcsine(n),
        ]
        for cand in families:
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # β‑ and edge‑weighted families (parameter sweep) ----------------------
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            for base in (beta, edge):
                for lam in lambda_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # optional SciPy windows ------------------------------------------------
        if _HAS_SCI:
            for atten in atten_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_chebyshev(n, atten)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best Cheb len={n} att={atten:.0f} score={sc:.6f}")

            for beta_par in kaiser_grid:
                if time.time() >= deadline:
                    break
                cand = _distribution_kaiser(n, beta_par)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new best Kaiser len={n} beta={beta_par:.2f} score={sc:.6f}")

        # random Dirichlet seeds ------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            cand = _distribution_dirichlet(n, rng, alpha)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best Dirichlet len={n} α={alpha:.3g} score={sc:.6f}")

        # sparse random seeds ---------------------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            cand = _distribution_random_sparse(n, rng, density=0.015)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] new best sparse len={n} score={sc:.6f}")

        # --- up‑sample previous best if we have one ----------------------------
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1:
                # linearly interpolate onto the new grid
                old_n = prev.shape[0]
                x_old = np.linspace(0, 1, old_n, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                up = np.interp(x_new, x_old, prev)
                up = np.maximum(up, 0.0)
                if up.sum() == 0.0:
                    up = _distribution_uniform(n)
                else:
                    up /= up.sum()
                sc = _objective(up)
                top_candidates.append((sc, up.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = up.copy()
                    print(f"[{time.time():.1f}] new best upsampled prev len={n} score={sc:.6f}")
        except Exception:
            pass

        # keep only the very best seeds (size 40) to limit memory
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:40]

    # ------------------------------------------------------------------
    # 2.  Ensure we have a non‑empty starting point.
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 4096
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Mirror‑descent on a few top seeds.
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:10]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting subgradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_multi(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=1.0,
                top_k=10,
                eval_interval=60,
            )
            if sc_opt < overall_best_score - 1e-12:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 4.  Final annealed polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final annealed polishing")
        a_opt, sc_opt = _annealing_refine(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            rng=rng,
        )
        if sc_opt < overall_best_score - 1e-12:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after annealing new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score (2·n·max(conv)) = {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – ``evaluate_sequence`` will perform the final checks.
    return overall_best_seq.tolist()
