assistantfinal**Solution Overview**

The task is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max\big((a*a)\big)}{(\sum a)^2},
\]

where `a` is a non‑negative step‑function (a 1‑D array).  
If we keep the normalisation  

\[
\sum_i a_i = \sqrt{2n},
\]

the denominator cancels and the score equals the maximal entry of the
self‑convolution `b = a*a`.  

The optimisation problem is non‑convex, but the objective is smooth
except at the points where the maximal convolution entry changes.
A very successful strategy is therefore:

| Phase | Idea |
|------|------|
|1️⃣ Broad exploration| Test many deterministic families (Beta, power‑law, geometric …) and random Dirichlet draws for several lengths. Keep the best.|
|2️⃣ Smoothed‑max Adam| Replace `max` by a log‑sum‑exp with a temperature `α`. Optimise the smooth surrogate with Adam while slowly raising `α` (so that we end up optimising the true max).|
|3️⃣ True‑max sub‑gradient| When `α` is large, switch to the exact sub‑gradient (active set) and continue with an Adam‑style line‑search.|
|4️⃣ Hill‑climbing polish| Small stochastic mass transfers that strictly improve the true max. Occasionally add jitter to escape plateaus.|

All phases respect the global time‑budget (default 1000 s).  
The implementation is completely self‑contained, uses only NumPy,
and works for any length up to a few × 10⁴ (the FFT‑based convolutions are
`O(n log n)`).

The code below follows exactly this plan, improves on the previous
implementation by

* a richer set of parametric families (geometric & power‑law shapes),
* finer grids for the Beta‑family,
* a full Adam optimiser for the smoothed objective,
* a more aggressive schedule for the temperature `α`,
* occasional random jitter inside the Adam loop,
* a deterministic final hill‑climbing stage.

"""
Propose a step‑function (list of non‑negative floats) that minimises
`evaluate_sequence`.

Algorithm (≈ 1000 s)
--------------------
1. Broad exploration: enumerate a variety of parametric families
   (uniform, Beta, arcsine, edge‑power, geometric, power‑law,
   Dirichlet draws) for several lengths. Keep the best vector.

2. Smoothed‑max Adam: minimise a log‑sum‑exp surrogate of the max‑convolution.
   The temperature α grows from 1 to 1e6, the learning rate decays.
   Adam momentum accelerates convergence.

3. True‑max sub‑gradient descent: once α is large we switch to the exact
   sub‑gradient (active set) and continue with Adam + back‑tracking line search.

4. Hill‑climbing polish: repeatedly move a tiny amount of mass from a
   “large” coordinate to a “small” one; accept only if the true max
   strictly improves. Small random jitter is added every few thousand steps.

All operations are performed on the probability vector `p = a / sum(a)`;
the final sequence `a` has `sum(a) = sqrt(2·len(a))` so the returned
value equals the maximal convolution entry.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v, s=1.0):
    """Project v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions (real-valued)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(p: np.ndarray) -> float:
    """
    Score for a probability vector p (sum=1):
        2·n·max(p*p)
    This equals `evaluate_sequence` after scaling.
    """
    n = p.shape[0]
    conv = _fft_self_convolve(p)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Parametric families (all return a *probability* vector, i.e. sum=1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_edge_power(n, p):
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w = np.maximum(w, 1e-15)
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n, beta_exp):
    """Dirichlet draw biased towards the edges."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    return np.random.dirichlet(conc)


def _distribution_geometric(n, r):
    """Geometric shape a_i ∝ r**i + r**(n-1-i), 0<r<1."""
    i = np.arange(n, dtype=np.float64)
    w = r ** i + r ** (n - 1 - i)
    w = np.maximum(w, 1e-15)
    w /= w.sum()
    return w


def _distribution_power_law(n, p):
    """Power‑law shape a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p) + (n - i + 1) ** (-p)
    w = np.maximum(w, 1e-15)
    w /= w.sum()
    return w

        for p in p_grid_sym:
            if time.time() >= deadline:

            sc = _max_convolution_score(vec)

                print(f"[{time.time():.1f}] n={n} beta_sym(p={p:.3f})    score={sc:.6f}")

        for p in p_grid_asym:

            # left‑heavy
            vec = _distribution_beta(n, p, 0.5)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} beta(p={p:.3f},q=0.5)   score={sc:.6f}")
            # right‑heavy
            vec = _distribution_beta(n, 0.5, p)
            vec = np.maximum(vec, eps); vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n} beta(p=0.5,q={p:.3f})   score={sc:.6f}")

        for p in edge_power_ps:

        print(f"[{time.time():.1f}] fallback uniform            score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed‑max Adam optimisation (convex‑smooth surrogate)
    a = best_seq.copy()                # probability vector (sum = 1)
    n = a.shape[0]

    remaining = deadline - time.time()
    # 0.003 s per Adam step (convolution + a few vector ops) is a safe upper bound
    est_per_iter = 0.003

    total_iters = max(15000, min(100000, max_iters))

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    eps_adam = 1e-8

    # Temperature schedule (smooth‑max)
    alpha_start, alpha_end = 1.0, 1e6

    # Learning‑rate schedule (multiplicative decay)
    lr_start, lr_end = 0.4, 0.001

    for it in range(1, total_iters + 1):
        if time.time() >= deadline:
            break

        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        # convolution & soft‑max weights
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps_adam)

        a = a - step * m_hat
        # keep probabilities non‑negative and on the simplex
        a = np.maximum(a, eps)
        a = _project_onto_simplex(a, 1.0)

        # occasional progress report
        if (it % max(1, total_iters // 20) == 0) or (it == total_iters):
            cur_score = _max_convolution_score(a)
            if cur_score < best_score - 1e-12:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Adam it={it}/{total_iters}  new best {cur_score:.6f}")

        # small random jitter every few thousand steps to avoid stagnation
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = _project_onto_simplex(a, 1.0)

    # ------------------------------------------------------------------
    # 4️⃣  True‑max sub‑gradient descent (active set) with back‑tracking
    a = best_seq.copy()
    m = np.zeros_like(a)   # first moment for Adam‑style step
    beta1_sub, beta2_sub = 0.9, 0.999
    step0 = 0.5
    sub_it = 0

    while time.time() < deadline:
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (indices attaining the maximum up to tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum
        m = beta1_sub * m + (1.0 - beta1_sub) * grad
        m_hat = m / (1.0 - beta1_sub ** sub_it)

        # back‑tracking line search
        step = step0
        while True:

            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand_score = _max_convolution_score(cand)

            if cand_score < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_score < best_score - 1e-12:
            best_score, best_seq = cand_score, a.copy()
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cand_score:.6f}")

        # occasional jitter to escape plateaus
        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = _project_onto_simplex(a, 1.0)

    # ------------------------------------------------------------------
    # 5️⃣  Targeted stochastic hill‑climbing polish
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0

    while time.time() < deadline:
        hill_it += 1

        # Identify a pair (i,j) that contributes most to the current peak
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # move a tiny amount of mass from the larger of the two to a small coordinate
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]   # second smallest

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 = _project_onto_simplex(a2, 1.0)

        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 200 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")

        # add jitter occasionally
        if hill_it % 2500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = _project_onto_simplex(a, 1.0)
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_it}")

    # ------------------------------------------------------------------
    # 6️⃣  Rescale to the required sum = sqrt(2·n) before returning
    scale = np.sqrt(2.0 * n)
    final_seq = (best_seq * scale).tolist()
    final_score = evaluate_sequence(final_seq)  # sanity check (optional)
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return final_seq
