"""
propose_candidate
=================

A time‑bounded search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm consists of three phases:

1. **Massive analytic sweep** – for several lengths (up to 2·10⁵) we
   instantiate many one‑parameter families (Beta‑densities, edge‑biased
   power laws, and mixtures with the uniform distribution).  Every
   candidate is evaluated with an FFT‑based exact convolution and the
   best few are kept.

2. **Fine‑grained mirror descent** – the current best candidate is
   refined by an exponentiated‑gradient (mirror‑descent) step on the
   smooth‑max of the convolution.  The temperature α is increased to
   10⁶ and the learning‑rate decays geometrically, while the number of
   iterations is chosen from the remaining time budget.

3. **Projected Adam sub‑gradient + tiny hill‑climber** – after the
   smooth‑max phase we switch to true sub‑gradients of the max‑convolution
   (using the active set), perform a few Adam‑style steps, and finish
   with a mass‑transfer hill‑climber that nudges probability mass from
   the worst contributing pair toward the smallest entry.

All helper routines are pure NumPy; no I/O is performed.  The function
returns the best sequence (as a plain list of floats) found before the
deadline.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact value of `evaluate_sequence` for a vector that already satisfies
    sum(a)=1 (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density, shifted for stability
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """Deterministic edge‑biased power‑law:
           w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= s
    return w

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float,
                            eps: float = 1e-12):
    """
    One mirror‑descent step on the smooth‑max objective
         f_α(a) = (1/α)·log Σ exp(α·conv_k).

    Returns the updated probability vector and the exact current max.
    """
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights (stable)
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1: len(a) - 1 + len(a)]
    # EG update on the simplex (KL mirror)
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, m


def _adam_subgrad_step(a: np.ndarray, vel: np.ndarray,
                       step0: float, beta1: float, beta2: float,
                       it: int, eps: float = 1e-12):
    """
    One Adam‑style projected sub‑gradient step on the true max‑convolution.
    Returns the updated vector, the new velocity and the current maximal
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices achieving the max, up to a tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

    # Adam update (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # line‑search (simple back‑tracking)
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray, max_iters: int,
                             eps: float = 1e-12):
    """
    Very light hill‑climber: repeatedly move a tiny amount of probability
    mass from the index that participates in the current peak toward the
    globally smallest entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        # location of the peak (might be a range, pick first)
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) attaining the largest product a_i a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional random jitter to escape plateaus
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

def _evaluate_and_update(vec: np.ndarray, best_seq, best_score):
    """Compute the score of `vec` (already normalised to sum=1) and
    return possibly improved (best_seq, best_score)."""
    sc = _max_convolution_score(vec)
    if sc < best_score:
        return vec.copy(), sc
    return best_seq, best_score
