"""
Search for a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Broad family sweep** – for several lengths (≈2000–5000) we evaluate a
   collection of parametric families (uniform, arcsine, symmetric / asymmetric
   Beta, deterministic edge‑power) and all convex mixes of these families
   with the uniform distribution.  The best vector (sum = 1) is kept.

2. **Smooth‑max projected gradient descent** – we minimise the
   soft‑max surrogate of the max‑convolution:
       fα(a)=α⁻¹·log∑exp(α·(a*a)).
   The gradient is obtained by a convolution with the soft‑max weights.
   An additive update followed by Euclidean projection onto the simplex
   (simplex projection) is performed; the temperature α and the step size
   are increased / decreased slowly.  This stage quickly drives the solution
   close to a local optimum.

3. **True sub‑gradient descent** – using the exact active set of the
   current maximum of the convolution we take an Adam‑style step and
   back‑track the step size until the objective improves.

4. **Stochastic hill‑climbing (simulated‑annealing style)** – random
   transfers of a small amount of probability mass from a “source’’ index
   to a “target’’ index are proposed.  Moves that improve the true
   max‑convolution are always accepted; uphill moves are accepted with a
   temperature that cools linearly to zero.  This final polishing can
   escape shallow plateaus.

All stages respect the global time budget (default 1000 s).  The final
vector is returned as a plain Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT utilities ---------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_self_convolve(a):
    """Linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a, b):
    """Linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """
    Score for a vector normalised to sum = 1:
        2·n·max(a*a)
    This equals the value returned by ``evaluate_sequence`` for the same
    shape (the scaling cancels out).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Parametric families (all return a vector with sum = 1) -----------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    if p == 1.0:
        w = np.ones_like(x)
    else:
        logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
        logw -= np.max(logw)
        w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w).all():
        w = np.full_like(x, 1.0 / n)
    else:
        w /= w_sum
    return w

def _distribution_beta(n, p, q):
    """Asymmetric Beta(p,q)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w).all():
        w = np.full_like(x, 1.0 / n)
    else:
        w /= w_sum
    return w

def _distribution_edge_power(n, p):
    """Deterministic edge‑biased shape a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    w += 1e-15            # avoid exact zeros
    w /= w.sum()
    return w

    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 1.5      # safety margin for finalisation
    eps = 1e-15

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad sweep over lengths and simple families
    lengths = list(range(2000, 5001, 200))          # 2000,2200,…,5000
    # symmetric beta shape parameters (wide range)
    p_grid_sym = np.concatenate([
        np.linspace(0.01, 0.10, 10),
        np.linspace(0.10, 0.50, 15),
        np.linspace(0.50, 1.00, 10),
    ])
    # asymmetric beta – left‑heavy (q fixed = 0.5)
    p_grid_asym = np.linspace(0.01, 0.30, 8)
    edge_power_ps = np.linspace(0.2, 1.2, 11)      # deterministic edge bias
    mix_lambdas = np.linspace(0.0, 1.0, 11)        # mixture weight with uniform

    for n in lengths:
        if time.time() >= deadline:
            break

        uniform = _distribution_uniform(n)
        # ---- evaluate raw families ------------------------------------------------
        families = []

        # uniform (baseline)
        families.append(("uniform", uniform))

        # arcsine
        arcs = _distribution_arcsine(n)
        families.append(("arcsine", arcs))

        # symmetric beta
        for p in p_grid_sym:
            if time.time() >= deadline:
                break
            families.append((f"beta_sym(p={p:.3f})", _distribution_beta_sym(n, p)))

        # asymmetric beta (left‑heavy)
        for p in p_grid_asym:
            if time.time() >= deadline:
                break
            families.append((f"beta(p={p:.3f},q=0.5)",
                             _distribution_beta(n, p, 0.5)))
            families.append((f"beta(p=0.5,q={p:.3f})",
                             _distribution_beta(n, 0.5, p)))

        # deterministic edge‑power
        for p in edge_power_ps:
            if time.time() >= deadline:
                break
            families.append((f"edge_power(p={p:.2f})",
                             _distribution_edge_power(n, p)))

        # ------------------------------------------------------------------
        # evaluate each family and all mixtures with uniform
        for name, vec in families:
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score, best_seq, best_n = sc, vec.copy(), n
                print(f"[{time.time():.1f}] n={n:4d} {name:30s} score={sc:.6f}")

            # try mixing with uniform
            for lam in mix_lambdas:
                if lam == 1.0:
                    continue          # already evaluated as pure
                mixed = lam * vec + (1.0 - lam) * uniform
                mixed = np.maximum(mixed, eps)
                mixed /= mixed.sum()
                sc2 = _max_convolution_score(mixed)
                if sc2 < best_score:
                    best_score, best_seq, best_n = sc2, mixed.copy(), n
                    print(f"[{time.time():.1f}] n={n:4d} {name:30s} mix λ={lam:.2f} score={sc2:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate any previously found sequence (if the variable exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq, best_n = sc, prev.copy(), len(prev)
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Guard against not having found anything
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Smooth‑max projected gradient descent (additive version)
    a = best_seq.copy()
    n = a.shape[0]

    # Determine how many iterations we can afford (roughly 0.003 s each)
    remaining = deadline - time.time()
    est_per_iter = 0.004
    max_iters = max(0, int(remaining / est_per_iter))
    # we bound it to a reasonable interval
    max_iters = max(1000, min(50000, max_iters))

    t0 = time.time()
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # progress in [0,1]
        prog = it / max_iters
        # temperature for soft‑max
        alpha = 1.0 * (1e5) ** prog          # grows to 1e5
        # additive step size (decreases)
        lr = 0.5 * (0.001 / 0.5) ** prog

        conv = _fft_self_convolve(a)
        mx = conv.max()
        # soft‑max weights (stable)
        w = np.exp(alpha * (conv - mx))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the soft‑max surrogate
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # additive projected gradient step
        a = a - lr * grad
        a = np.maximum(a, eps)
        a = project_onto_simplex(a, 1.0)

        # bookkeeping
        if (it % max(1, max_iters // 10) == 0) or (it == max_iters - 1):
            cur_sc = _max_convolution_score(a)
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] GD it={it}/{max_iters} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient descent on the true max‑convolution
    a = best_seq.copy()
    m = np.zeros_like(a)          # first‑moment estimate
    beta1 = 0.9
    it_sub = 0

    while time.time() < deadline:
        it_sub += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (tolerance for floating‑point noise)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # sub‑gradient (sum over active constraints)
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active) if active.size > 0 else 0.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** it_sub)

        # simple back‑tracking line search
        step = 0.5
        while True:
            cand = a - step * m_hat
            cand = np.maximum(cand, eps)
            cand = project_onto_simplex(cand, 1.0)
            cand_sc = _max_convolution_score(cand)

            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_score - 1e-12:
            best_score, best_seq = cand_sc, a.copy()
            print(f"[{time.time():.1f}] subgrad it={it_sub} new best {cand_sc:.6f}")

        # occasional random jitter to avoid stagnation
        if it_sub % 500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = project_onto_simplex(a, 1.0)

    # ------------------------------------------------------------------
    # 5. Stochastic hill‑climbing (simulated‑annealing style) ----------
    a = best_seq.copy()
    cur_sc = _max_convolution_score(a)
    hill_it = 0
    # temperature schedule for uphill moves
    T_start = cur_sc * 0.02
    T_end = 1e-9

    while time.time() < deadline:
        hill_it += 1
        # linear cooling
        prog = hill_it / max(1, (deadline - start) / 0.1)   # approximate number of steps
        T = T_start * (1.0 - prog) + T_end * prog

        # propose a random mass transfer
        src = np.random.choice(n, p=a)          # source weighted by its mass
        tgt = np.random.choice(n - 1)
        if tgt >= src:
            tgt += 1

        max_delta = min(a[src] * 0.5, 0.02)      # keep changes modest
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.maximum(cand, eps)
        cand /= cand.sum()
        cand_sc = _max_convolution_score(cand)

        accept = False
        if cand_sc < cur_sc - 1e-12:
            accept = True
        else:
            # uphill move – accept with probability exp(-(Δ)/T)
            prob = np.exp(-(cand_sc - cur_sc) / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_sc = cand_sc
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] hill it={hill_it} new best {cur_sc:.6f}")

        # tiny jitter every few thousand steps to keep exploration alive
        if hill_it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a = project_onto_simplex(a, 1.0)
            cur_sc = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # 6. Return the best sequence found
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
