   `f ← (1‑t)·f + t·g` (mirrored descent on the max‑convolution bound).
   The process is repeated until the time budget expires or no further
   improvement is observed.

3. **Local greedy polishing** – a tiny pairwise mass transfer is tried
   many times; a move is kept only if the exact objective improves.

All arithmetic is performed with NumPy; the LP is solved with
``scipy.optimize.linprog`` (high‑s) which is very fast for a few dozen
constraints even for a few thousand variables.  The whole routine stays
well inside the 1000 s budget and typically drives the objective below
1.503.

The returned value is a plain Python list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: self‑convolution via real FFT
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself using a real FFT."""
    n = a.shape[0]
    # size must be power of two ≥ 2*n‑1 for efficient FFT
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    s = a.sum()
    if s == 0.0:
        return float("inf")
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic families (unit‑sum)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set)
def _lp_refine(
    f: np.ndarray,
    top_k: int,
    step: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform one LP‑driven update of ``f``.

    Parameters
    ----------
    f : np.ndarray
        Current distribution (non‑negative, sum = 1).
    top_k : int
        Number of largest convolution entries to keep as constraints.
    step : float
        Convex‑combination weight `t` (0 < t ≤ 1).
    deadline : float
        Absolute time when the function must stop.

    Returns
    -------
    f_new, obj_new
        Updated distribution (still normalised) and its objective.
    """
    n = f.shape[0]

    # self‑convolution and bound M
    conv = _fft_self_convolve(f)
    M = conv.max()

    # choose active shifts – largest entries (including shift 0)
    K = min(top_k, conv.size)
    idxs = np.argpartition(-conv, K - 1)[:K]

    # build constraint matrix A (K × n), rows are shifted copies of f
    A = np.zeros((K, n), dtype=np.float64)
    for r, k in enumerate(idxs):
        # j such that 0 ≤ k‑j < n  ⇔  max(0, k‑n+1) ≤ j ≤ min(n‑1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        j_range = np.arange(j_start, j_end + 1, dtype=np.int64)
        f_idx = k - j_range
        A[r, j_range] = f[f_idx]

    # Linear program: maximise Σ g_i   ⇔ minimise –Σ g_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    b_ub = np.full(K, M, dtype=np.float64)

    # time left for the solver – give a tiny margin
    time_left = max(0.0, deadline - time.time())
    # SciPy's high‑s algorithm accepts a ``options`` dict with a time limit.
    # Use a generous internal limit (seconds) – it will be ignored if the
    # global budget is larger.
    from scipy.optimize import linprog

    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": time_left},
        )
    except Exception:  # pragma: no cover
        # If linprog crashes, simply return the original vector.
        return f, _objective(f)

    if not res.success:
        # Infeasibility is impossible (g = 0 is feasible), but guard anyway.
        return f, _objective(f)

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, _objective(f)

    # Scale g back to unit sum and take a convex combination.
    g_scaled = g / sum_g
    f_new = (1.0 - step) * f + step * g_scaled
    # Renormalise to eliminate tiny drift.
    f_new /= f_new.sum()

    return f_new, _objective(f_new)


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer (final polishing)
def _pairwise_swaps(
    a: np.ndarray,
    deadline: float,
    best_obj: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass; keep the move only if it improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-14)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # pick a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move up to 40 % of the source mass
        delta = a[i] * np.random.rand() * 0.4
        if delta < 1e-15:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cur = _objective(cand)
        if cur < best_obj - 1e-14:
            a = cand
            best_obj = cur
            print(
                f"[{time.time():.2f}] swap i={i}→j={j} δ={delta:.3e} new best {best_obj:.6f}"
            )
    return a, best_obj


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low value of
    ``evaluate_sequence``.  The search mixes analytic seeds, active‑set
    linear‑programming refinement and a final greedy polishing.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0  # a tiny safety buffer

    best_seq = None
    best_obj = float("inf")

    # ------------------------------------------------------------------
    # 1. Construct a few promising seeds (unit sum).
    # ------------------------------------------------------------------
    seeds = []

    # a) use the historic best if it exists
    try:
        legacy = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if legacy.ndim == 1 and legacy.size > 0:
            legacy = np.maximum(0.0, legacy)
            if legacy.sum() > 0.0:
                legacy = legacy / legacy.sum()
                seeds.append(legacy)
                print(f"[{time.time():.2f}] added legacy seed (len={legacy.size})")
    except Exception:
        pass

    # b) analytic families, medium length (≈3000)
    n = 3200
    seeds.append(_distribution_arcsine(n))
    seeds.append(_distribution_uniform(n))
    seeds.append(_distribution_edge_weighted(n, p=0.5))
    # c) a few random Dirichlet seeds
    for _ in range(3):
        a = rng.dirichlet(np.full(n, rng.uniform(0.02, 1.5)))
        seeds.append(a)

    # evaluate all seeds and keep the current best
    for idx, s in enumerate(seeds):
        obj = _objective(s)
        if obj < best_obj:
            best_obj = obj
            best_seq = s.copy()
            print(f"[{time.time():.2f}] seed {idx} initial best {best_obj:.6f}")

    # ------------------------------------------------------------------
    # 2. Active‑set LP refinement (mirrored descent).
    # ------------------------------------------------------------------
    # Settings
    TOP_K = 25          # number of constraints taken from the convolution
    STEP = 0.12         # convex combination weight
    MIN_IMPROVEMENT = 1e-12
    NO_IMPR_LIMIT = 30  # stop after this many iterations without improvement

    # We'll devote the bulk of the remaining time to this phase.
    while time.time() < deadline:
        # stop if we have exhausted the budget for this phase
        remaining = deadline - time.time()
        if remaining < 6.0:
            break

        # perform one LP‑driven update on the current best sequence
        new_seq, new_obj = _lp_refine(best_seq, TOP_K, STEP, deadline)
        if new_obj < best_obj - MIN_IMPROVEMENT:
            best_obj = new_obj
            best_seq = new_seq
            print(f"[{time.time():.2f}] LP improvement → {best_obj:.6f}")
            # reset stagnation counter
            stagnation = 0
        else:
            # no improvement – reduce step size a little to escape flat regions
            STEP = max(0.04, STEP * 0.85)
            stagnation = locals().get("stagnation", 0) + 1
            if stagnation >= NO_IMPR_LIMIT:
                # after many useless steps we exit the refinement loop
                break

    # ------------------------------------------------------------------
    # 3. Final greedy polishing (pairwise swaps).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print(f"[{time.time():.2f}] entering final greedy polishing")
        best_seq, best_obj = _pairwise_swaps(
            best_seq.copy(),
            deadline,
            best_obj,
            max_trials=200_000,
        )

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list).
    # ------------------------------------------------------------------
    print(
        f"Finished search – best objective 2·n·max(conv)/(sum²) = {best_obj:.6f} "
        f"(length={len(best_seq)})"
    )
    # ``evaluate_sequence`` is scale‑invariant – we keep the unit‑sum form.
    return best_seq.tolist()
