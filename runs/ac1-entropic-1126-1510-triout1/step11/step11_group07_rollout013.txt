def random_hill_climb(a, max_iters, eps):
    for iter in range(max_iters):
        # compute conv and max
        conv = _fft_self_convolve(a)
        M = conv.max()
        # identify indices set that contributes to max
        # we find one index k where conv is max.
        k = conv.argmax()
        # compute best pair i,j with max a_i * a_j in that sum
        i_start = max(0, k-(a.shape[0]-1))
        i_end = min(a.shape[0]-1, k)
        # find pair that maximizes product
        # vectorized: compute products for feasible i
        idx = np.arange(i_start, i_end+1)
        j = k - idx
        prod = a[idx] * a[j]
        best_idx = np.argmax(prod)
        i = idx[best_idx]; j = j[best_idx]
        # move small delta from larger of these to smallest entry
        source = i if a[i] >= a[j] else j
        target = np.argmin(a)
        # ensure source != target
        if source == target:
            # find second smallest
            sorted_idx = np.argsort(a)
            target = sorted_idx[1] if sorted_idx[0] == source else sorted_idx[0]
        delta = min(a[source] * 0.3, 0.01) * random.random()
        if delta < eps:
            continue
        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        # project onto simplex implicitly as we conserve mass sum (should be 1)
        # but ensure nonneg and normalization
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-13:
            a = a2
        # occasional random jitter
        if (iter+1) % 1000 == 0:
            jitter = np.random.randn(a.shape[0]) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
    return a

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of floats) that minimizes the
provided ``evaluate_sequence``.  The algorithm works in three stages
under a hard time budget (default 1000 s).

1. **Broad analytic exploration** – for several lengths a collection of
   parametrised families (arcsine, symmetric/asymmetric Beta, edge‑biased
   power‑law with optional uniform baseline, Dirichlet draws) is sampled.
   The best few candidates are kept.

2. **Refinement** – starting from the current best candidate a mirror‑descent
   step on a smooth‑max approximation of the max‑convolution is performed,
   followed by an Adam‑style projected sub‑gradient descent on the true
   max‑convolution.  Both stages respect the simplex constraint
   (non‑negative entries that sum to one).  The refinement is repeated from
   a few jittered restarts.

3. **Final polishing** – a cheap mass‑transfer hill‑climber that moves a
   tiny amount of probability mass from the most harmful pair towards the
   smallest entry is applied.

All heavy lifting uses NumPy FFTs, no file or network I/O, and the code
continually checks the deadline so that a valid sequence is always returned
before the timeout.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# FFT utilities (real transforms)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Scoring (objective with sum(a)=1)
# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` for a probability vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power_baseline(
    n: int, alpha: float, c: float = 0.0
) -> np.ndarray:
    """
    Edge‑biased power‑law with an optional uniform baseline:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Random Dirichlet draw with edge‑biased concentration:
    concentration_i ∝ ((i+½)(n‑i‑½))^{‑beta_exp} + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _smooth_max_mirror_step(
    a: np.ndarray, alpha: float, lr: float, eps: float = 1e-12
) -> tuple[np.ndarray, float]:
    """
    One exponentiated‑gradient step on the smooth‑max of the convolution.
    Returns the updated vector and the exact maximal convolution entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    # exponentiated‑gradient (KL‑mirror) update
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(
    a: np.ndarray,
    vel: np.ndarray,
    step0: float,
    beta1: float,
    beta2: float,
    it: int,
    eps: float = 1e-12,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    One Adam‑style projected sub‑gradient step on the true max‑convolution.
    Returns (new_a, new_velocity, new_maximum_of_convolution).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (indices attaining the maximum, with a tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum update (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search (max 6 halvings)
    step = step0
    for _ in range(6):
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(
    a: np.ndarray, max_iters: int, eps: float = 1e-12
) -> np.ndarray:
    """
    Tiny deterministic hill‑climber: move a small amount of mass from the
    entry most responsible for the current peak towards the global minimum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) giving the largest product for this k
        idx = np.arange(i_start, i_end + 1)
        jdx = k - idx
        prod = a[idx] * a[jdx]
        best = np.argmax(prod)
        i, j = idx[best], jdx[best]

        source = i if a[i] >= a[j] else j
        target = np.argmin(a)
        if target == source:
            # pick a different tiny target
            order = np.argsort(a)
            target = order[1] if order[0] == source else order[0]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional jitter to avoid stagnation
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a
