def _smooth_max_update(a, beta, lr):
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    w = np.exp(beta*(conv - max_val))
    w_sum = w.sum()
    w = w / w_sum
    # Gradient.
    # Compute convolution of w and a (full length).
    grad_full = np.convolve(w, a)  # using FFT for speed
    grad = grad_full[len(a)-1 : len(a)-1 + len(a)]   # slice length n
    # update via exponentiated gradient
    a_new = a * np.exp(-lr * grad)  # ensures positivity
    a_new = a_new / a_new.sum()
    return a_new

def _smooth_descent(a, deadline, best_seq, best_score, beta0=10.0, beta_inc=0.001, lr0=0.2, eval_interval=100):
    t=0
    while time.time()<deadline:
        t+=1
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # Evaluate objective
        if t % eval_interval == 0:
            cur = 2.0 * len(a) * max_c
            if cur < best_score:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth iter {t} new best {best_score:.6f}")

        # compute smoothed subgradient
        beta = beta0 + beta_inc*t
        # To avoid overflow, compute exp(beta*(conv - max_c)), values in [0,1]
        exp_weight = np.exp(beta * (conv - max_c))
        w = exp_weight / exp_weight.sum()

        # compute gradient = conv(w, a) slice
        grad_full = np.convolve(w, a)  # using np.convolve (maybe slower). Use FFT for performance: we'll create custom function _fft_convolve(w, a)
        # use FFT for speed
        grad_full = _fft_convolve_general(w, a)
        n = a.shape[0]
        grad = grad_full[n-1:n-1+n]

        # Update using exponentiated gradient
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # Reproject onto simplex
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0/len(a))
        else:
            a /= a_sum
    return best_seq, best_score

def _fft_convolve_general(x, y):
    n = len(x); m = len(y)
    size = 1 << ((n + m - 2).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X*Y, n=size)
    return conv[:n+m-1]

"""
Hybrid search for a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------

1. **Rich initial pool** – generate many analytic families
   (uniform, arcsine, symmetric beta, edge‑weighted, sine‑weighted) on a
   range of lengths and a few random Dirichlet / sparse vectors.
   Keep the ~30 best seeds (according to the exact objective
   ``2·n·max(conv(a,a))`` where ``sum(a)=1``).

2. **Broad‑active mirror descent** – on each top seed run exponentiated‑
   gradient descent, but the active set now contains *all* convolution
   indices whose value lies within a small fraction (default 2 %) of the
   current maximum.  The active set size is capped (default 30) and the
   step size decays as 1/√t.  A jitter is injected after long
   stagnation.

3. **Smooth‑max descent** – starting from the best vector obtained
   above, minimise a smooth approximation of the maximum
   ``max(conv)`` using the log‑sum‑exp (temperature β slowly increased).
   The (sub)gradient of the smooth max is computed via a second FFT‑based
   convolution.  Exponentiated‑gradient updates with projection onto the
   probability simplex are used.

4. **Local polishing** – cheap pairwise mass moves and a simulated‑annealing
   walk (as in the reference code) finish the optimisation.

All steps respect the supplied time budget (default 1000 s).  The
function returns a plain Python list; ``evaluate_sequence`` will apply
the final normalisation, clipping, etc.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (real FFT). Returns length 2·n‑1."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


def _fft_convolve_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    n, m = x.shape[0], y.shape[0]
    need = n + m - 1
    size = 1 << ((need - 1).bit_length())
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_random_dirichlet(
    n: int, rng: np.random.Generator, alpha: float = 1.0
) -> np.ndarray:
    a = rng.dirichlet(np.full(n, alpha))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Broad‑active exponentiated‑gradient descent ---------------------------
def _subgradient_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.8,
    eval_interval: int = 150,
    active_frac: float = 0.02,
    max_active: int = 30,
    patience: int = 2500,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent where the sub‑gradient is accumulated
    from all convolution indices whose value lies within ``active_frac``
    of the current maximum.  ``max_active`` caps the size of that set.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 500_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        thresh = max_c * (1.0 - active_frac)
        act = np.where(conv >= thresh)[0]
        if act.shape[0] > max_active:
            top_idx = np.argsort(conv[act])[-max_active:]
            act = act[top_idx]

        # gradient accumulation (factor 2 for symmetry)
        grad = np.zeros_like(a)
        ind_grid = np.arange(n, dtype=np.int32)
        for idx in act:
            ind = idx - ind_grid
            mask = (ind >= 0) & (ind < n)
            grad[mask] += 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] subgrad_mult it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.006, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a /= a.sum()
            no_improve = 0
    return best_seq, best_score

# Smooth‑max descent (log‑sum‑exp) ------------------------------------
def _smooth_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    beta0: float = 12.0,
    beta_inc: float = 0.002,
    lr0: float = 0.25,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Optimise a smooth approximation of max(conv) using a temperature
    parameter β that is linearly increased.  The gradient of the soft‑max
    is computed via a second FFT‑based convolution.  Updates are
    exponentiated‑gradient followed by projection onto the simplex.
    """
    n = a.shape[0]
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline and t < 300_000:
        t += 1

        conv = _fft_self_convolve(a)
        max_c = conv.max()
        if t % eval_interval == 0:
            cur = 2.0 * n * max_c
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth_descent it={t} new best {best_score:.6f}")

        beta = beta0 + beta_inc * t
        # soft‑max weights (stable)
        exp_w = np.exp(beta * (conv - max_c))
        w = exp_w / exp_w.sum()

        # gradient = 2 * (w * a) central slice
        grad_full = _fft_convolve_general(w, a)
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional tiny jitter to escape plateaus
        if t % 5000 == 0:
            jitter = rng.normal(scale=1e-4, size=n)
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
    return best_seq, best_score

# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.33
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 7e-3,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.28
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:6]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting broad‑active descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_multi(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.85,
                eval_interval=120,
                active_frac=0.02,
                max_active=30,
                patience=2500,
            )
            if sc_opt < overall_best_score:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after descent seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Smooth‑max descent (log‑sum‑exp) on the current best
    if time.time() < deadline:
        print(f"[{time.time():.1f}] starting smooth‑max descent")
        overall_best_seq, overall_best_score = _smooth_descent_multi(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            beta0=12.0,
            beta_inc=0.003,
            lr0=0.30,
            eval_interval=150,
        )

    # ------------------------------------------------------------------
    # 6.  Greedy pairwise refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] greedy pairwise refinement")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_trials=250_000,
        )

    # ------------------------------------------------------------------
    # 7.   Final stochastic local search (simulated annealing style)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final random‑local search")
        overall_best_seq, overall_best_score = _random_local_search(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            init_temp=8e-3,
            decay=0.9996,
        )

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will handle clipping, etc.
    return overall_best_seq.tolist()
