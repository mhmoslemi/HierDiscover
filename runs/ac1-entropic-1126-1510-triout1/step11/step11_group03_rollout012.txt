def improve_via_lp(a, n_iters, t, deadline):

   best = a.copy()
   best_score = objective(a)

   for i in range(n_iters):
       if time.time() >= deadline:
           break
       M = max_convolution(a)
       # Build constraint matrix A (size (2n-1, n))
       A = np.zeros((2*n-1, n))
       for k in range(2*n-1):
           start = max(0, k-(n-1))
           end = min(n-1, k)
           # set A[k, start:end+1] based on f
           idx = np.arange(start, end+1)
           A[k, idx] = a[k - idx]
       # Solve LP
       c = -np.ones(n)
       bounds = [(0, None)] * n
       res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
       if not res.success:
           break
       b = res.x
       # Rescale b to have same sum normalization sqrt(2n)
       Sb = b.sum()
       if Sb == 0:
           b = np.full(n, 1.0/n)
           Sb = 1.0
       scale = np.sqrt(2*n) / Sb
       g = b * scale

       # Mix
       a = (1 - t) * a + t * g

       # enforce small floor epsilon to keep positivity
       eps = 1e-8
       a = np.maximum(a, eps)
       a = a / a.sum()   # Normalize to sum = 1

       cur = _objective(a)
       if cur < best_score:
           best_score = cur
           best = a.copy()
           print(f"[{time.time():.1f}] LP iter {i} new best {best_score:.6f}")

"""
Hybrid LP + soft‑max descent for the step‑function attack.

*   Start from the best known sequence (global variable *height_sequence_1*),
    or fall back to a small family of analytic seeds.
*   The sequence is kept on the simplex with the fixed  ℓ₁‑norm
    √(2·n)  (this makes the objective equal to the maximal
    self‑convolution coefficient, i.e. 2·n·max/∑a² = max(conv)).
*   One LP iteration:
        maximise   Σ bⱼ
        s.t.       (a  *  b)ₖ ≤ M   for all k,
                    bⱼ ≥ 0 .
    The optimal b concentrates mass where a is small.
    After solving we renormalise b to the same ℓ₁‑norm and mix
    a ← (1‑t)·a + t·b.
*   A long exponentiated‑gradient descent with a temperature‑controlled
    soft‑max replaces the “max’’ in the objective, allowing a smooth
    sub‑gradient to be computed via FFT.
*   Finally a cheap greedy pairwise‑mass‑transfer walk refines the result.

The whole procedure stops before the supplied time‑budget expires and
returns the best sequence found (as a plain list of floats) ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional SciPy – if unavailable we fall back to the original
# gradient‑descent only implementation

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)²."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2·n‑1, Σw = 1).
    Implemented by FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2·n‑1, n)) such that
    (A @ b)  ==  f * b   (full linear convolution).
    """
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=f.dtype)
    # each column j is f placed from row j to row j+n‑1
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
def _lp_step(a: np.ndarray, t_mix: float) -> np.ndarray:
    """
    Perform one LP update: maximise Σ bⱼ under (a⋆b) ≤ M,
    then mix the result with the current a using step size t_mix.
    The ℓ₁‑norm of a (and therefore of every iterate) is kept fixed.
    """
    n = a.shape[0]
    target_sum = a.sum()                     # √(2·n) in our convention
    M = _fft_self_convolve(a).max()

    # ----- linear programme ------------------------------------------------
    A = _build_conv_matrix(a)                # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)                          # maximise Σb ⇔ minimise –Σb
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:                      # safety fallback
        return a

    b = res.x
    sb = b.sum()
    if sb <= 0.0:
        # degenerate LP – fall back to a copy of a
        g = a.copy()
    else:
        # rescale to retain the fixed ℓ₁‑norm
        scale = target_sum / sb
        g = b * scale

    # ----- mixing -----------------------------------------------------------
    a_new = (1.0 - t_mix) * a + t_mix * g
    # keep strict positivity (helps the LP in the next round)
    eps = 1e-12
    a_new = np.maximum(a_new, eps)
    a_new = a_new * (target_sum / a_new.sum())
    return a_new


# ----------------------------------------------------------------------
def _lp_descent(start_a: np.ndarray, deadline: float,
                max_iters: int = 20, t_mix: float = 0.25) -> tuple[np.ndarray, float]:
    """Repeated LP steps followed by a short exponential‑gradient phase."""
    a = start_a.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        a = _lp_step(a, t_mix)
        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] LP it={it+1} new best {best_score:.6f}")

    # ----- smooth soft‑max gradient descent (few thousand iter) -------------
    a, best_score = _smooth_descent(
        a, deadline,
        best_seq=best_seq,
        best_score=best_score,
        max_iter=40000,
        eval_interval=800,
        alpha_start=0.5,
        alpha_end=30.0,
        lr0=0.9,
    )
    return a, best_score


# ----------------------------------------------------------------------
def _smooth_descent(a: np.ndarray, deadline: float,
                    best_seq: np.ndarray, best_score: float,
                    max_iter: int = 40000,
                    eval_interval: int = 500,
                    alpha_start: float = 0.5,
                    alpha_end: float = 30.0,
                    lr0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the max‑
    convolution.  The sequence stays on the simplex with ℓ₁‑norm = √(2·n).
    """
    n = a.shape[0]
    target_sum = a.sum()
    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a)
        max_conv = conv.max()
        # linear schedule for the temperature
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        grad = _gradient_from_weights(a, w)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # keep on the simplex
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full(n, target_sum / n)
        else:
            a = a * (target_sum / a.sum())
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} new best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Greedy local walk: transfer a random fraction of mass from one index
    to another, keeping the move only if the true objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src_idx = np.where(a > 1e-12)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)
        j = rng.integers(n)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.30  # up to 30 % of source mass
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score

        arcs = arcs / arcs.sum() * target_sum
        seed_list.append(arcs)

        # c) random Dirichlet draws (α ≈ 0.5)
        for _ in range(2):
            dir_raw = rng.gamma(0.5, size=n)
            dir_raw = dir_raw / dir_raw.sum() * target_sum
            seed_list.append(dir_raw)

        # d) previous best if it matches the current length
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size == n:
                prev = np.maximum(prev, 0.0)
                if prev.sum() > 0.0:
                    prev = prev / prev.sum() * target_sum
                    seed_list.append(prev)
        except Exception:
            pass

        # --------------------------------------------------------------
        # 2️⃣  Evaluate the pool and keep a few most promising candidates
        evaluated = [( _objective(s), s) for s in seed_list]
        evaluated.sort(key=lambda x: x[0])
        top_seeds = [seq for _, seq in evaluated[:3]]

        # --------------------------------------------------------------
        # 3️⃣  Run the LP‑+‑gradient descent pipeline on each top seed
        for idx, s in enumerate(top_seeds):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}]   seed {idx+1}/{len(top_seeds)} (initial score {_objective(s):.6f})")
            # ensure strict positivity – the LP needs every column to have a
            # non‑zero coefficient somewhere
            eps = 1e-9
            s = np.maximum(s, eps)
            s = s * (target_sum / s.sum())

            # ? use the LP based descent when SciPy is present
            if _SCIPY_AVAILABLE:
                a, score = _lp_descent(s, deadline,
                                       max_iters=18,   # a few LP rounds
                                       t_mix=0.22)
            else:
                # fallback – the original soft‑max+swap routine
                a, score = _smooth_descent(
                    s, deadline,
                    best_seq=s,
                    best_score=_objective(s),
                    max_iter=50000,
                    eval_interval=500,
                    alpha_start=0.5,
                    alpha_end=30.0,
                    lr0=0.9,
                )
                a, score = _pairwise_swap_local(a, deadline, a, score)

            if score < overall_best_score - 1e-12:
                overall_best_score = score
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}]   *** new global best {overall_best_score:.6f} (len={n}) ***")

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing – cheap mass‑transfer walk
    if overall_best_seq is not None and time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            overall_best_seq,
            overall_best_score,
        )

    # ------------------------------------------------------------------
    if overall_best_seq is None:
        # extremely unlikely – return a uniform fallback
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n)
        overall_best_score = _objective(overall_best_seq)

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
