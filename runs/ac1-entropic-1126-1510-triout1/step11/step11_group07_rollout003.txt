"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The algorithm consists of three phases:

1️⃣  **Global parametric sweep** – explore a dense grid of Beta(p,q) and
    edge‑biased power‑law families on a moderate size (n≈8000).  The
    best few shapes are kept.

2️⃣  **Intensive refinement** – each retained shape is up‑sampled to a
    larger size (20 k – 80 k) and optimised with
        • a smooth‑max mirror‑descent (exponentiated‑gradient) phase,
        • a projected Adam‑style sub‑gradient phase on the exact max‑convolution,
        • a tiny mass‑transfer hill‑climber.
    All steps respect the simplex constraint and enforce a tiny
    positivity floor.

3️⃣  **Global random hill‑climbing** – starting from the overall best
    vector, small random mass transfers are tried; improvements are
    kept and occasional uphill moves are allowed with a tiny probability
    (simulated‑annealing style).

The routine keeps a hard time budget (default 1000 s) and returns the
best sequence found before the deadline.  The implementation uses only
NumPy and the helper functions defined below; no file or network IO is
performed.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1,
    returns the exact value produced by `evaluate_sequence`
    (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α}+c
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float, eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective  f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector and the *exact* maximal convolution.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One projected Adam‑style sub‑gradient step on the true max‑convolution.
    Returns the updated vector, the new velocity and the *exact* maximal
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Very cheap hill‑climber: repeatedly move a tiny amount of probability
    mass from the index that most contributes to the current maximal
    convolution towards the globally smallest entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())                # location of the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # Find the pair (i,j) attaining the largest product a_i*a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]  # second‑smallest

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
        # occasional jitter to avoid stagnation
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

def _upscale(vec: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a larger grid."""
    n_old = len(vec)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    vec_new = np.interp(x_new, x_old, vec, left=vec[0], right=vec[-1])
    vec_new = np.maximum(vec_new, 1e-12)
    vec_new /= vec_new.sum()
    return vec_new
