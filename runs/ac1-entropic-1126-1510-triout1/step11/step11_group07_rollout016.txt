"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Analytic seeds** – for a few lengths (512 … 4096) we start from simple
   families (arcsine, symmetric Beta(p,p) and an edge‑biased power law).

2. **Linear‑programming descent** – for a fixed length we repeatedly solve
   a small LP:

        maximise   Σ g_j
        subject to f * g ≤ M   (convolution constraints)
                   g_j ≥ 0

   where ``M`` is the current maximal self‑convolution of ``f`` (the value
   returned by ``evaluate_sequence`` when ``Σ f = √(2·n)``).  The LP gives a
   vector ``g₀`` with the largest possible total mass while respecting the
   bound ``M``.  After scaling ``g₀`` back to the required ℓ₁‑norm
   ``√(2·n)`` we move a fraction ``t`` towards it:
   ``f ← (1−t)·f + t·g``.  Because the bound never increases, the maximal
   convolution value strictly decreases and the procedure converges.

3. **Tiny mass‑transfer hill‑climber** – a final local search that moves a
   little probability mass from the most offending pair to the smallest
   entry, repeatedly improving the max‑convolution.

All steps respect the simplex constraint (non‑negative entries, fixed
ℓ₁‑norm).  The implementation uses NumPy, SciPy’s sparse matrices and the
high‑performance “highs” LP solver.  The total run‑time is kept well below
the 1000 s budget; the best sequence found (any length) is returned as a
plain list of floats.
"""

import time
import random
import math
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# FFT based self‑convolution (real, O(n log n))
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Analytic families (return probability vectors, sum = 1)
# ----------------------------------------------------------------------


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid { (j+0.5)/n }."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

# Simple utilities
# ----------------------------------------------------------------------


def _max_conv_value(a: np.ndarray) -> float:
    """Maximum entry of a*a (self‑convolution).  No normalisation."""
    conv = _fft_self_convolve(a)
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the matrix A (size (2n‑1) × n) such that A @ g = f * g.
    Row k, column i contains f[i] iff 0 ≤ k−i < n,
    otherwise 0.
    """
    n = f.shape[0]
    # integer index vectors (int32 to save memory)
    idx = np.arange(n, dtype=np.int32)
    rows = (idx[:, None] + idx).ravel()          # shape (n*n,)
    cols = np.repeat(idx, n)                     # shape (n*n,)
    data = np.repeat(f.astype(np.float64), n)    # shape (n*n,)
    m = 2 * n - 1
    return sp.csr_matrix((data, (rows, cols)), shape=(m, n))


def _lp_optimize(f: np.ndarray,
                 max_outer: int,
                 t_start: float,
                 deadline: float) -> np.ndarray:
    """
    Perform the LP based descent described in the docstring.
    ``f`` must already satisfy Σ f = √(2·n) and be strictly positive.
    Returns the improved vector (still Σ = √(2·n)).
    """
    n = f.shape[0]
    target_sum = math.sqrt(2.0 * n)
    t = t_start
    for it in range(max_outer):
        if time.time() >= deadline:
            break

        M = _max_conv_value(f)                     # current maximal conv value
        A = _build_conv_matrix(f)                  # (2n‑1) × n, sparse
        b = np.full(A.shape[0], M, dtype=np.float64)

        # maximise Σ g  <=> minimise -Σ g
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # solve LP
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b, bounds=bounds,
            method='highs', options={'presolve': True}
        )
        if not res.success:
            # LP failure – abort the descent for this length
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            break

        # rescale to keep the ℓ₁‑norm = √(2·n)
        scale = target_sum / sum_g0
        g = g0 * scale

        # convex combination
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 1e-12)               # keep strict positivity
        f_new = f_new * (target_sum / f_new.sum())    # re‑normalise

        M_new = _max_conv_value(f_new)

        if M_new < M - 1e-12:       # genuine improvement
            f = f_new
            # slowly increase step size up to a modest ceiling
            t = min(t * 1.2, 0.4)
        else:
            # shrink step size; if it becomes too tiny we stop
            t *= 0.5
            if t < 1e-4:
                break
    return f


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int = 5000,
                             eps: float = 1e-12) -> np.ndarray:
    """Tiny local optimiser that nudges probability mass away from the
    most responsible pair towards the smallest entry."""
    n = a.shape[0]
    for it in range(max_iters):
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())               # index of the current peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # best pair (i,j) that realises the peak at k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = int(np.argmin(a))
        if target == source:
            # pick the second smallest if they coincide
            target = int(np.argsort(a)[1])

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _max_conv_value(a2)
        if new_M < conv.max() - 1e-12:
            a = a2
        # occasional jitter to break ties
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
    return a

def propose_candidate(seed: int = 1234, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  The function respects a hard time budget (default
    1000 s) and returns the best sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a modest safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0️⃣  Load a previously discovered construction if it exists
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size:
            # normalise to a proper probability vector (scale does not matter)
            prev = np.maximum(prev, 1e-12)
            prev = prev / prev.sum()
            best_seq = prev.copy()
            best_score = evaluate_sequence(best_seq.tolist())
            print(f"[{time.time():.1f}] loaded previous best score={best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 1️⃣  Parameters for the exploration
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]               # modest sizes – fast LP solves
    beta_ps = [0.15, 0.25, 0.35, 0.45]              # p for symmetric Beta(p,p)
    edge_alphas = [0.30, 0.35, 0.40]                # α for edge‑biased power law

    # ------------------------------------------------------------------
    # 2️⃣  Main search loop
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- helper: create a probability vector and scale it -------------
        def _prepare(vec: np.ndarray) -> np.ndarray:
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()
            # scale to Σ = √(2·n) – this makes the evaluation equal to the max‑conv.
            return vec * math.sqrt(2.0 * n)

        # ----- try a few analytic seeds ------------------------------------
        seed_fns = []

        # arcsine
        seed_fns.append(lambda n=n: _distribution_arcsine(n))

        # a couple of symmetric Betas
        for p in beta_ps:
            seed_fns.append(lambda n=n, p=p: _distribution_beta_sym(n, p))

        # edge‑biased power‑law
        for a in edge_alphas:
            seed_fns.append(lambda n=n, a=a: _distribution_edge_power(n, a))

        for init_fn in seed_fns:
            if time.time() >= deadline:
                break

            a = _prepare(init_fn())
            # ----------------------------------------------------------------
            # LP descent phase
            # ----------------------------------------------------------------
            a = _lp_optimize(a,
                             max_outer=25,
                             t_start=0.3,
                             deadline=deadline)

            # ----------------------------------------------------------------
            # Final hill‑climber polish
            # ----------------------------------------------------------------
            a = _mass_transfer_hill_climb(a, max_iters=3000)

            # Convert back to a plain probability vector (sum = 1) for scoring
            a_norm = a / a.sum()
            try:
                score = evaluate_sequence(a_norm.tolist())
            except Exception:
                score = float("inf")

            if score < best_score:
                best_score = score
                best_seq = a_norm.copy()
                print(f"[{time.time():.1f}] n={n} init={init_fn.__name__} "
                      f"new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  If nothing succeeded, fall back to a simple uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} "
          f"(length={best_seq.shape[0]})")
    return best_seq.tolist()
