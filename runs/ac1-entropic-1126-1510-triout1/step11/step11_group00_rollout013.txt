"""
Non‑negative step‑function optimisation.

The algorithm proceeds in three phases:

1. **Diverse seeding** – Analytic families (uniform, arcsine,
   symmetric Beta, edge‑weighted, sine‑weighted, power‑law) are sampled
   for several lengths together with a handful of random Dirichlet /
   sparse vectors.  The ~30 best seeds are retained.

2. **Smooth gradient descent** – Starting from each retained seed a
   “soft‑max” of the self‑convolution is built:
   w_k ∝ exp(β·(conv_k‑max(conv)).  The gradient of the weighted
   convolution is computed efficiently by a single FFT‑based
   cross‑correlation.  An additive step with a decreasing learning rate
   (η_t ∝ 1/√t) is taken and the iterate is projected back onto the
   probability simplex.  β is increased from 0 to a target value so that
   the early phase smooths the whole convolution while the later phase
   concentrates on the peak(s).  Stagnation triggers a tiny random
   jitter.

3. **Local polishing** – a fast pairwise‑swap walk and a cheap
   simulated‑annealing style mass‑transfer walk finish the search.

All work respects the supplied time budget (default 1000 s).  The
returned list is a plain Python list; ``evaluate_sequence`` will take
care of the final normalisation, clipping, etc.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _gradient_via_fft(w: np.ndarray, a_rev: np.ndarray, n: int) -> np.ndarray:
    """
    Compute 2·(w * a_rev) restricted to indices n‑1 … 2n‑2.
    This yields the gradient of the weighted self‑convolution.
    """
    need = w.shape[0] + a_rev.shape[0] - 1          # = 3n‑2
    size = 1 << ((need - 1).bit_length())
    W = np.fft.rfft(w, n=size)
    A = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_law(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = i ** (-p)
    w /= w.sum()
    return w


def _distribution_random_sparse(n: int,
                                rng: np.random.Generator,
                                density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max of the convolution) -------------
def _smooth_gradient_descent(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             lr0: float = 0.6,
                             beta_target: float = 8.0,
                             beta_ramp: int = 50000,
                             eval_interval: int = 200,
                             patience: int = 3000) -> tuple[np.ndarray, float]:
    """
    Additive (projected) gradient descent where the gradient is obtained
    from a weighted convolution w_k ∝ exp(β·(conv_k‑max(conv))).

    β is increased linearly from 0 to ``beta_target`` during the first
    ``beta_ramp`` iterations.  The step size decays as 1/√t.  Stagnation
    triggers a small random jitter.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1

        # 1) current self‑convolution and its maximum
        conv = _fft_self_convolve(a)
        max_c = conv.max()

        # 2) weight vector – soft‑max around the peak(s)
        if t < beta_ramp:
            beta = beta_target * t / beta_ramp
        else:
            beta = beta_target
        w = np.exp(beta * (conv - max_c))
        # prune negligible entries (helps the FFT a little)
        w[w < 1e-12] = 0.0

        # 3) gradient from the weighted convolution
        grad = _gradient_via_fft(w, a[::-1], n)

        # 4) additive update + simplex projection
        eta = lr0 / math.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        # 5) occasional evaluation
        if t % eval_interval == 0:
            cur = 2.0 * n * max_c
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth_descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # 6) jitter on long stagnation
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.004, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap local refinement ----------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Move a small chunk of mass from i to j if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Simple simulated‑annealing style local search -------------------------
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         init_temp: float = 8e-3,
                         decay: float = 0.9997) -> tuple[np.ndarray, float]:
    """Cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score

        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list; ``evaluate_sequence`` will handle clipping etc.
    return overall_best_seq.tolist()
