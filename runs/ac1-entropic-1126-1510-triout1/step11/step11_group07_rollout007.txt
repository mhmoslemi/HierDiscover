assistantfinal**Solution Overview**

The goal is to minimise  

```
score = 2·n·max( a * a )          ( a is the normalised step‑function )
```  

because `evaluate_sequence` removes the overall scale of the vector.  
Hence we have to find a probability vector `a` whose *self‑convolution* has the
smallest possible infinity‑norm.

The algorithm consists of four stages:

1. **Broad analytic search** – enumerate many families of probability vectors
   (`arcsine`, symmetric `Beta(p,p)`, edge‑biased power laws, mixtures with the
   uniform distribution, …) for several lengths (`n`).  The best few candidates
   are remembered.

2. **Exponential‑gradient (mirror) descent** – starting from the current best
   vector we minimise a smooth‑max approximation of the maximal convolution.
   The temperature is increased geometrically while the learning rate is
   decreased, turning the optimisation from a global search into a precise
   local refinement.

3. **Projected Adam sub‑gradient descent** – a few Adam‑style steps on the
   *true* max‑convolution are applied to escape shallow plateaus.

4. **Tiny mass‑transfer hill‑climber** – repeatedly moves a tiny amount of
   probability mass from the pair that contributes most to the current peak
   towards the smallest entry.  This polish often yields the last tiny
   improvement.

All operations are pure NumPy, use real FFTs for fast convolutions,
and respect a hard time budget (default 1000 s).  The best vector found is
returned as a plain Python list.

```python
"""
Propose a step‑function (non‑negative list of floats) that minimises
`evaluate_sequence`.  The search runs under a fixed time budget (~1000 s)
and combines:

1. Exhaustive exploration of analytic families (arcsine, Beta, edge‑biased
   power‑law) together with mixtures with the uniform distribution.
2. Smooth‑max mirror descent on the best candidate.
3. Projected Adam sub‑gradient descent.
4. A tiny mass‑transfer hill‑climber for final polishing.

All computations use FFT‑based convolutions and work on normalised
probability vectors (sum = 1), because the evaluation function is scale‑
invariant.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    a is assumed to be a probability vector (Σa = 1).
    Returns the exact value produced by ``evaluate_sequence``,
    i.e. 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, Σ = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑domain for stability
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Edge‑biased deterministic distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{‑α}+c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w

# Optimisation primitives

def _smooth_max_mirror_step(a: np.ndarray,
                            alpha: float,
                            lr: float,
                            eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror) step on the smooth‑max objective
        f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector and the current max convolution.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]
    # exponentiated‑gradient update on the simplex (KL mirror)
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One projected Adam‑style step on the *true* max‑convolution.
    Returns the updated vector, the new velocity and the current maximal
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

    # Adam momentum (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Move a tiny amount of probability mass from the pair that contributes most
    to the current peak towards the globally smallest entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())               # location of the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) that gives the largest product a_i*a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional jitter to avoid stagnation
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

                print(f"[{time.time():.1f}] n={n} Beta(p={p:.3f}) score={sc:.6f}")

            for lam in [0.75, 0.85, 0.95]:
                seq = lam * vec + (1.0 - lam) * uni
                sc = _max_convolution_score(seq)
                if sc < best_score:
                    best_seq, best_score = seq.copy(), sc
                    print(f"[{time.time():.1f}] n={n} mix Beta/uni λ={lam:.2f} p={p:.3f} score={sc:.6f}")

        # edge‑biased power‑law and mixtures
        for alpha in alphas:
            if time.time() >= deadline:
                break
            vec = _distribution_edge_power(n, alpha)
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"[{time.time():.1f}] n={n} edgePower(α={alpha:.2f}) score={sc:.6f}")

            for lam in [0.80, 0.90]:
                seq = lam * vec + (1.0 - lam) * uni
                sc = _max_convolution_score(seq)
                if sc < best_score:
                    best_seq, best_score = seq.copy(), sc
                    print(f"[{time.time():.1f}] n={n} mix edge/uni λ={lam:.2f} α={alpha:.2f} score={sc:.6f}")

    # fallback in the unlikely case that nothing was generated
    if best_seq is None:
        best_seq = _distribution_arcsine(20000)
        best_score = _max_convolution_score(best_seq)

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max mirror descent (refinement)
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # estimate how many cheap iterations we can afford
    remaining = deadline - time.time()
    est_it_time = 0.004 if n > 20000 else 0.002
    max_iters = max(2000, min(150000, int(remaining / est_it_time)))
    print(f"[{time.time():.1f}] Mirror‑descent phase: n={n} iters={max_iters}")

    # temperature and learning‑rate schedules
    alpha_start, alpha_end = 0.5, 2e5
    lr_start, lr_end = 0.4, 0.001

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        t = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)

        # occasional exact check
        if (it + 1) % max(1, max_iters // 10) == 0 or it == max_iters - 1:
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] mirror it={it}/{max_iters} new best {cur_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Projected Adam‑style sub‑gradient descent
    # --------------------------------------------------------------
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.6
    sub_it = 0
    while time.time() < deadline - 0.5:
        sub_it += 1
        a, velocity, cur_max = _adam_subgrad_step(a, velocity,
                                                  step0, beta1, beta2,
                                                  sub_it, eps)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] Adam subgrad it={sub_it} new best {cur_score:.6f}")

        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # --------------------------------------------------------------
    # 4️⃣  Tiny mass‑transfer hill‑climber (final polishing)
    # --------------------------------------------------------------
    a = best_seq.copy()
    a = _mass_transfer_hill_climb(a, max_iters=5000, eps=eps)
    final_score = _max_convolution_score(a)
    if final_score < best_score:
        best_seq, best_score = a.copy(), final_score
        print(f"[{time.time():.1f}] final hill‑climber improved score to {final_score:.6f}")

    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
```
