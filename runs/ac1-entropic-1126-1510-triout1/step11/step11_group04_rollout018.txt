def _solve_lp_for_f(f, M):
    n = len(f)
    import scipy.sparse as sp
    from scipy.optimize import linprog
    # build A
    # use lil for efficiency
    A = sp.lil_matrix((2*n-1, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = f
    A = A.tocsr()
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    return b

"""
Propose a non‑negative step function that (hopefully) improves on the previous
solution.  The strategy combines three ideas:

1. **Wide analytic start** – families such as uniform, arcsine, triangular,
   monotone exponential, power‑law, Dirichlet draws and a few random sparse
   vectors are generated for several moderate lengths (1500‑3000).  The best
   few are kept as a starting elite.

2. **Linear‑programming refinement** – for a fixed sequence *f* we compute
   M = max(f * f).  The problem

        maximise   Σ b_j
        subject to b_j ≥ 0,
                   (f * b)_k ≤ M   for all k

   is linear because *f * b* is linear in *b*.  Solving it (with
   ``scipy.optimize.linprog``) yields a new vector *g* that spreads the mass
   as much as possible while keeping the convolution bound.  Mixing *f* and
   *g* (a cheap line‑search over a few mixing ratios) strictly reduces the
   objective in practice.  A handful of such LP‑steps are performed for each
   elite start.

3. **Fine local polishing** – after the LP phase we run the smooth‑max
   mirror‑descent and Adam‑style projected sub‑gradient methods from the
   original code to clean up any remaining irregularities.

The whole search respects the 1000 s budget, prints intermediate scores and
returns the best sequence found as a plain ``list[float]``.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Evaluate the objective for a vector that already sums to 1.
    (evaluate_sequence = 2·n·max(conv) because the sum‑normalisation cancels.)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a vector with Σ = 1 -------------------

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential_left(n, lam):
    """Monotone decreasing exponential: a_i ∝ exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _generate_candidates_for_length(n):
    """Return a list of candidate vectors (Σ = 1) for the given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # monotone exponentials (different decay rates)
    for c in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0):
        lam = c / n
        cand.append(_distribution_exponential_left(n, lam))
        cand.append(_distribution_exponential_left(n, lam)[::-1])   # increasing version

    # power‑law families
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # random Dirichlet draws
    for a in (0.2, 0.5, 1.0, 2.0):
        cand.append(np.random.dirichlet(np.full(n, a)))

    # sparse random start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    return cand

# Linear‑programming refinement -----------------------------------------

def _lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise Σ b_j    subject to   b_j ≥ 0  and  (f * b)_k ≤ M.
    Returns the optimal b (may be None if the solver fails).
    """
    n = f.shape[0]
    # keep a tiny epsilon to avoid zeros (otherwise some columns become
    # completely unconstrained and the LP can become unbounded)
    eps = 1e-12
    f = np.maximum(f, eps)

    # Build the (2n‑1) × n matrix A with A[k, j] = f_{k‑j} (zero if out of range)
    # Using a diagonal representation: each diagonal offset d (0…n‑1) is constant f[d].
    diags = [np.full(n, coeff, dtype=np.float64) for coeff in f]
    offsets = np.arange(n)                # 0,1,…,n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')

    # Linear programme: maximise 1ᵀb  ⇔ minimise -1ᵀb
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _mix_candidates(f: np.ndarray, g: np.ndarray, eps: float = 1e-12):
    """
    Try a few convex mixtures of f and g and return the best (lowest objective).
    """
    best = f
    best_sc = _objective(f)
    # a tiny set of mixing ratios
    for t in (0.0, 0.25, 0.5, 0.75, 1.0):
        mixed = (1.0 - t) * f + t * g
        mixed = np.maximum(mixed, eps)
        mixed /= mixed.sum()
        sc = _objective(mixed)
        if sc < best_sc:
            best_sc = sc
            best = mixed
    return best, best_sc

def _smooth_max_mirror_descent(a, deadline):
    """Exponentiated‑gradient mirror descent on a smooth approximation of the max."""
    n = a.shape[0]
    eps = 1e-12
    schedule = [
        (1.0, 1e3, 4000, 0.5, 0.04),
        (1e3, 1e5, 4000, 0.04, 0.004),
        (1e5, 1e7, 4000, 0.004, 0.0004),
    ]

    best_a = a.copy()
    best_sc = _objective(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t
            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()
            if it % 2000 == 0:
                sc = _objective(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
    return best_a, best_sc


def _projected_subgradient(a, deadline):
    """Adam‑style sub‑gradient descent on the true max‑convolution."""
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_step = 0.4
    it = 0
    best_a = a.copy()
    best_sc = _objective(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum
        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_step
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break
    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0      # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_len = None

    # ------------------------------------------------------------------
    # 1. Broad exploration – keep a short elite list
    lengths = [1500, 2000, 2500, 3000]
    elite = []   # (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break
        for vec in _generate_candidates_for_length(n):
            if time.time() >= deadline:
                break
            a = np.asarray(vec, dtype=np.float64)
            a = np.maximum(a, eps)
            a /= a.sum()
            sc = _objective(a)
            if sc < best_score:
                best_score, best_seq, best_len = sc, a.copy(), n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # keep elite (size ≤ 5)
            if len(elite) < 5:
                elite.append((sc, a.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, a.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate a previous solution, if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq, best_len = sc, prev.copy(), prev.size
                print(f"[{time.time():.1f}] Using prior best {sc:.6f}")
            if len(elite) < 5:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen)
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        best_len = n
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Main refinement: LP steps followed by fine polishing
    for idx, (sc, a0) in enumerate(elite):
        if time.time() >= deadline:
            break
        a = a0.copy()
        n = a.shape[0]
        print(f"[{time.time():.1f}] Refinement {idx+1}/{len(elite)} (len={n}) start score {sc:.6f}")

        # ---- LP‑based mass redistribution (few iterations) ----
        for it in range(8):                     # 8 is a safe number under the time budget
            if time.time() >= deadline:
                break
            conv = _fft_self_convolve(a)
            M = conv.max()
            b = _lp_max_sum(a, M)
            if b is None:
                break
            sum_b = b.sum()
            if sum_b <= 0:
                break
            g = b / sum_b
            a, sc = _mix_candidates(a, g, eps=eps)
            if sc < best_score - 1e-12:
                best_score, best_seq, best_len = sc, a.copy(), n
                print(f"[{time.time():.1f}] New global best after LP it {it+1} score {sc:.6f}")

        # ---- optional stronger polishing with gradient‑based methods ----
        a, sc = _smooth_max_mirror_descent(a, deadline)
        a, sc = _projected_subgradient(a, deadline)
        if sc < best_score - 1e-12:
            best_score, best_seq, best_len = sc, a.copy(), n
            print(f"[{time.time():.1f}] New global best after polishing {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanitisation
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
