    n = a.shape[0]

    best_sc = _objective(a)
    t = 0.6

        conv = _fft_self_convolve(a)

        col = np.concatenate([a, np.zeros(n-1)])

"""
Improved step‑function search
================================

This implementation builds on the previous “analytic families → mirror descent →
sub‑gradient → hill‑climb” pipeline but adds a *linear‑programming refinement*
stage inspired by the classic alternating LP method described in the statement.

For a fixed length `n` we keep the vector normalized to unit sum, because
`evaluate_sequence` is homogeneous of degree 0.  The objective for a
probability vector `a` is

    Obj(a) = 2·n·max_k (a*a)[k]                (1)

where `*` is the linear convolution.

Given a current distribution `f`, let `M = max (f*f)`.  The LP

    maximise   1ᵀ b
    subject to  f * b ≤ M      (element‑wise)
                b ≥ 0

finds another distribution `b` whose self‑convolution never exceeds `M`,
while trying to put as much total mass as possible into `b`.  Mixing
`f` and the normalised `b` strictly reduces the peak `M` (unless `b = f`),
and repeating the step drives the objective down.

The algorithm works as follows:

1. **Broad exploration** – generate a small elite of candidates from several
   analytic families (uniform, beta, exponential, …) for a few moderate
   lengths (≈ 2500–4500).  The best few candidates are kept.

2. **LP refinement** – for each elite vector we run a limited number of LP
   iterations (≈ 15–20).  After each LP solve we mix the current vector with
   the LP solution using a decaying mixing factor `t`.  This quickly lowers
   the convolution peak.

3. **Polishing** – the refined vectors are fed through the smoother
   exponentiated‑gradient mirror‑descent and a projected sub‑gradient
   routine (the same utilities used previously).  Finally a light
   hill‑climber with pruning is applied.

All stages respect the global time‑budget (default 1000 s).  The final
vector is returned as a plain Python list of non‑negative floats.

The code requires NumPy and SciPy (both are available in the execution
environment).
"""

# ----------------------------------------------------------------------
# Linear‑algebra / convolution utilities --------------------------------

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum x = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """
    Return the value of ``evaluate_sequence`` for a vector that has already
    been normalised to sum = 1 (the scaling is irrelevant because the
    objective is homogeneous of degree 0).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Candidate generator ----------------------------------------------------
def _generate_candidates_for_length(n):
    """Return a list of probability vectors for a given length."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta
    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta
    for p, q in ((0.2, 1.0), (1.0, 0.2), (0.5, 1.5), (1.5, 0.5)):
        cand.append(_distribution_beta_asym(n, p, q))

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws (different concentration)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Sparse random start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Shuffle a few base families to introduce asymmetry
    for base in (cand[0], cand[1], cand[4]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

def _smooth_max_mirror_descent(a, deadline):
    """Exponentiated‑gradient mirror descent on a smoothed max."""
    n = a.shape[0]
    eps = 1e-12

    # Temperature / learning‑rate schedule (more aggressive)
    schedule = [
        (1.0, 1e3, 12000, 0.6, 0.08),
        (1e3, 1e5, 12000, 0.08, 0.008),
        (1e5, 1e7, 12000, 0.008, 0.0008),
    ]

    best_a = a.copy()
    best_sc = _objective(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 2000 == 0:
                sc = _objective(a)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a, deadline):
    """Adam‑style sub‑gradient descent with back‑tracking."""
    n = a.shape[0]
    eps = 1e-12
    vel = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_step = 0.4
    it = 0

    best_a = a.copy()
    best_sc = _objective(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        vel = beta1 * vel + (1.0 - beta1) * grad
        v_hat = vel / (1.0 - beta1 ** it)

        step = base_step
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-13 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-13:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        if it % 3000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_and_prune(a, deadline):
    """Small hill‑climber with mass transfers from the dominant pair."""
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1

        # mass transfer from the pair that creates the maximal convolution term
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            max_delta = min(a[source] * 0.5, 0.03)
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-13:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    new_sc = _objective(a)
                    if new_sc < best_sc - 1e-13:
                        best_sc = new_sc
                        best_a = a.copy()
                        if it % 200 == 0:
                            print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # pruning of very small entries every few thousand steps
        if it % 2500 == 0:
            for keep_frac in (0.98, 0.95, 0.92):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                pruned = np.zeros_like(a)
                pruned[idx] = a[idx]
                pruned /= pruned.sum()
                sc = _objective(pruned)
                if sc < best_sc - 1e-13:
                    a = pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# LP‑based refinement ----------------------------------------------------
def _lp_refine(a, deadline, max_iters=30):
    """
    Alternating linear‑programming refinement.
    Starting from probability vector `a`, repeatedly:
        1. compute M = max(a*a)
        2. solve maximise 1ᵀ b   s.t. a * b ≤ M , b ≥ 0
        3. mix a ← (1‑t)·a + t·(b / sum(b))   with a decaying t
    Returns the best vector seen during the run.
    """
    a = a.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = _objective(a)

    t = 0.6                     # initial mixing factor
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        M = conv.max()

        # Build the full linear constraint matrix A (shape (2n‑1, n))
        # Each column j is the vector a shifted down by j positions.
        col = np.concatenate([a, np.zeros(n - 1)])
        row = np.concatenate([a[:1], np.zeros(n - 1)])
        A = toeplitz(col, row)           # dense (2n‑1) × n matrix

        # Solve the LP: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n)                    # objective coefficients
        ub = np.full(2 * n - 1, M * 1.0000001)  # tiny slack for numerical safety
        res = opt.linprog(c, A_ub=A, b_ub=ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True})

        if not res.success:
            print(f"[{time.time():.1f}] LP failed (iter {it}): {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b

        # convex combination – this never increases the peak M
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        sc = _objective(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP‑refine it={it} new best {sc:.6f}")

        t *= 0.9           # slowly shrink the step size

    return best_a, best_sc

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    # reproducibility
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1️⃣  Broad exploration – keep a tiny elite of promising vectors
    lengths = [2500, 3500, 4500]               # moderate sizes – LP‑friendly
    elite = []                                 # list of (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _objective(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain elite (size ≤ 6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, vec.copy())
                elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2️⃣  Incorporate a previously discovered best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            elif sc < elite[-1][0]:
                elite[-1] = (sc, prev.copy())
                elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  LP‑based refinement + polishing for each elite vector
    for idx, (sc, vec) in enumerate(elite):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Elite {idx+1}/{len(elite)} (len={len(vec)}) start score {sc:.6f}")

        # ---- LP refinement -------------------------------------------------
        vec_lp, sc_lp = _lp_refine(vec, deadline, max_iters=20)

        # ---- smooth‑max mirror descent (further global search) ------------
        vec_md, sc_md = _smooth_max_mirror_descent(vec_lp, deadline)

        # ---- projected sub‑gradient polishing -------------------------------
        vec_sg, sc_sg = _projected_subgradient(vec_md, deadline)

        # ---- final hill‑climber & pruning ----------------------------------
        vec_final, sc_final = _hill_climb_and_prune(vec_sg, deadline)

        if sc_final < best_score - 1e-13:
            best_score = sc_final
            best_seq = vec_final.copy()
            best_n = len(best_seq)
            print(f"[{time.time():.1f}] New global best after elite {sc_final:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final sanitisation
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()

    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
