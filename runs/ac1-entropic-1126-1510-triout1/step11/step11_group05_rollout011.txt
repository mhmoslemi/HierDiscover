def _lp_step(f, M, deadline):
    n = len(f)
    # Build constraints A_ub * g <= M
    # Use scipy.sparse to avoid dense memory; fill data arrays.
    rows = []
    cols = []
    data = []
    for i in range(n):
        # rows from i to i+n-1
        r = np.arange(i, i + n, dtype=np.int32)
        rows.append(r)
        cols.append(np.full(n, i, dtype=np.int32))
        data.append(f)  # vector f of length n
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    # b vector: M * ones
    b = np.full(2*n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)] * n

    # solve
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    except Exception as e:
        print(f"LP solver error: {e}")
        return None, False

    if res.success:
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            return None, False
        g_norm = g / sum_g
        return g_norm, True
    else:
        # Infeasibility may arise due to strict constraints; perhaps relax M slightly
        # try with M*(1+1e-9)
        M_relax = M * 1.000001
        b_relax = np.full(2*n - 1, M_relax)
        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_relax, bounds=bounds, method='highs')
        except Exception:
            return None, False
        if res.success:
            g = res.x; g_norm = g / g.sum(); return g_norm, True
    return None, False

def _lp_step(f, M, deadline):
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    idx = 0
    for i in range(n):
        rows[idx: idx+n] = np.arange(i, i+n)
        cols[idx: idx+n] = i
        data[idx: idx+n] = f
        idx += n
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n -1, n))
    b = np.full(2*n -1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64) # maximize sum(g)
    # Solve
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                    bounds=[(0, None)]*n,
                                    method='highs')
    except Exception as e:
        print("LP error:", e)
        return None, False
    if not res.success:
        # degree of infeasibility may be due to numeric rounding; try tiny relax
        eps_rel = 1e-9
        b_rel = np.full_like(b, M*(1+eps_rel))
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_rel,
                                    bounds=[(0, None)]*n, method='highs')
        if not res.success:
            return None, False
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None, False
    g_norm = g / sum_g
    return g_norm, True

"""
Propose a non‑negative step function that minimises `evaluate_sequence`.

The algorithm combines three complementary search phases:

1. **Broad exploration** – candidates from analytic families
   (uniform, Beta, mixtures, Dirichlet, sparse) for several powers‑of‑two
   lengths (up to 16384).  The best few are kept as “elite”.

2. **Local refinement** – for each elite start we run:
   * a smooth‑max mirror‑descent (exponentiated‑gradient) stage,
   * a long projected sub‑gradient stage,

   * a final hill‑climber / pruning stage.

3. **Time‑aware orchestration** – every heavy loop checks the overall
   deadline (1000 s – safety margin) and aborts early if needed.

All vectors are kept on the simplex (sum = 1); the objective on the
simplex is `score = 2·n·max(a*a)`.  The returned Python list contains

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a * b using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_mixture_beta_uniform(n, p, w):
    """w·Beta(p,p) + (1‑w)·Uniform, already normalised."""
    beta = _distribution_beta_sym(n, p)
    uniform = _distribution_uniform(n)
    mix = w * beta + (1.0 - w) * uniform
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()
    return mix


def _distribution_random_dirichlet(n, alpha):

def _distribution_sparse_random(n, k):
    """k random non‑zero entries."""

    idx = np.random.choice(n, k, replace=False)

def _smooth_max_mirror_descent(a0, deadline):
    """Mirror descent on a smooth‑max surrogate for the max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 1e3, 2500, 0.12, 0.012),
        (1e3, 1e5, 2500, 0.012, 0.0012),
        (1e5, 1e6, 3000, 0.0012, 0.00012),
    ]

    for (a0_alpha, a1_alpha, steps, lr0, lr1) in schedule:
        for it in range(steps):
            if time.time() >= deadline:
                return best_a, best_sc
            t = it / steps
            alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 400 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] mirror it={it} new best {sc:.6f}")

    return best_a, best_sc

def _projected_subgradient(a0, deadline):
    """True sub‑gradient descent with momentum and simple back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    step0 = 0.8
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)

            if cand_sc < best_sc - 1e-12 or step < 1e-9:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {cand_sc:.6f}")

        if it % 1500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.4 > deadline:
            break

    return best_a, best_sc

    """
    Solve
        max  Σ g_i
        s.t. conv(f,g) ≤ M  (for every shift)
             g_i ≥ 0
    and return the normalised g (sum = 1).  Returns (g,True) on success.
    """
    n = f.shape[0]

    # Build sparse matrix A where A[k,i] = f_{k-i}  (0 if out of bounds)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        cols[pos:pos + n] = i
        data[pos:pos + n] = f
        pos += n
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # minimise -Σg  == maximise Σg
    bounds = [(0, None)] * n

    # Linprog may be a bit expensive – enforce a soft time guard
    remaining = max(0.0, deadline - time.time())
    try:
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b, bounds=bounds, method="highs",
            options={"presolve": True, "time_limit": remaining},
        )
    except Exception as e:
        print("LP solve exception:", e)
        return None, False

    if not res.success:
        # Try a tiny relaxation of M to recover feasibility
        b_rel = np.full_like(b, M * (1.0 + 1e-9))
        try:
            res = scipy.optimize.linprog(
                c,
                A_ub=A,
                b_ub=b_rel,
                bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": remaining},
            )
        except Exception:
            return None, False
        if not res.success:
            return None, False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None, False
    g_norm = g / sum_g
    return g_norm, True


def _lp_refine(a0, deadline, max_iters=4):
    """
    Repeated LP refinement interleaved with a convex combination step.
    Each iteration solves the LP described above, then mixes the solution
    with the current iterate (mixing weight decreasing with iteration).
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        M = _fft_self_convolve(a).max()
        g, ok = _lp_step(a, M, deadline)
        if not ok or g is None:
            break

        # mixing weight – larger at early iterations, shrinking later
        t = 0.5 / it
        a = (1 - t) * a + t * g
        a = _project_onto_simplex(a, 1.0)

        sc = _max_convolution_score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP it={it} new best {sc:.6f}")

    return best_a, best_sc

# Hill‑climber + aggressive support pruning -----------------------------
def _hill_climb_prune(a0, deadline):
    """Simple hill‑climbing with occasional aggressive pruning."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # move mass away from the pair that generates the current max
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        src_i, src_j = i_start, k - i_start
        max_prod = a[src_i] * a[src_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                src_i, src_j = i, j
        src = src_i if a[src_i] >= a[src_j] else src_j
        trg = random.randint(0, n - 1)
        if trg != src:
            delta = min(a[src] * 0.5, 0.02) * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[src] -= delta
                a2[trg] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 250 == 0:
                        print(f"[{time.time():.1f}] hill it={it} new best {best_sc:.6f}")

        # aggressive pruning every few thousand steps
        if it % 2000 == 0:
            for keep_frac in (0.98, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        if it % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

# Candidate generator ----------------------------------------------------
def _generate_candidates(n):
    """Return a list of normalised candidate vectors for a given length."""
    cand = []

    # basic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_beta_sym(n, 0.5))   # arcsine

    # dense grid of symmetric betas
    for p in np.concatenate(
        (np.logspace(-2, -0.2, 12), np.linspace(0.1, 2.0, 20))
    ):
        cand.append(_distribution_beta_sym(n, float(p)))

    # mixtures with the uniform distribution
    for p in (0.05, 0.1, 0.2, 0.4, 0.6, 1.0):
        for w in (0.2, 0.5, 0.8):
            cand.append(_distribution_mixture_beta_uniform(n, p, w))

    # several Dirichlet draws (different concentration)
    for alpha in (0.05, 0.2, 0.8, 2.0):
        cand.append(_distribution_random_dirichlet(n, alpha))

    # sparse start (few non‑zeros)
    k = max(2, n // 400)
    cand.append(_distribution_sparse_random(n, k))

    # random shuffles of a few bases to introduce asymmetry
    for base in (cand[0], cand[2], cand[-2]):
        sh = base.copy()
        np.random.shuffle(sh)
        cand.append(sh)

    return cand

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising `evaluate_sequence`.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []               # (score, vector)

    # ------------------------------------------------------------------
    # 1. Broad exploration over several lengths
    lengths = [1024, 2048, 4096, 8192, 16384]   # powers of two – FFT friendly
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)

            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Global new best len={n}  score={sc:.6f}")

            # maintain a short elite list (size ≤6)
            if len(elite) < 6:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate any historic solution (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous solution score={sc:.6f}")
            if len(elite) < 6:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be hit)
    if best_seq is None:
        n0 = 4096
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(4, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.6f}")

        # smooth‑max mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # projected sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # LP‑based refinement (only for modest lengths to keep memory low)
        if a_mid.shape[0] <= 2048:
            a_mid, sc_mid = _lp_refine(a_mid, deadline)

        # final hill‑climber + pruning
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – enforce non‑negativity, normalise and report
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
