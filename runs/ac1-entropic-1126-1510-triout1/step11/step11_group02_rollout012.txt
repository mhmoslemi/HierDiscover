    cands = []
    # uniform
    cands.append(_distribution_uniform(n))

    # arcsine
    cands.append(_distribution_arcsine(n))

    # beta symmetric param grid

"""
Hybrid optimiser for the “max‑convolution” task with extended search.

Main ideas
----------
* Generate a diverse pool of seeds using many analytic families (Beta,
  arcsine, sine‑power, Dirichlet, random sparse, etc.) for several
  lengths.  All seeds are normalised to sum = 1.
* Keep the best few seeds (by the exact objective) and improve each of
  them with three deterministic phases:
    – Smooth‑max descent (log‑sum‑exp surrogate) with an increasing
      temperature and an exponentiated‑gradient update.
    – Classical multimax sub‑gradient descent (active‑set max).
    – Aggressive pairwise‑swap local refinement (always keep improving
      moves).
* When time permits, repeat the whole pipeline on fresh random seeds.
* The whole routine respects the 1000 s budget, reports progress and
  finally returns the best sequence found (as a plain ``list[float]``).

The optimiser works for any length up to ≈30 000, uses only NumPy and
stays well within the time limit.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  Σ_k w_k (a*a)_k  with respect to a.
    Implements 2·a⋆w  via FFT cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.shape[0] - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective  2·n·max(conv(a,a)) / (sum(a)^2)  (sum>0)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_sine_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p    (peaked in the middle for p>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Collect a wide variety of seed vectors for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta (grid)
    for p in (0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0,
              1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # random symmetric beta
    for _ in range(5):
        p = 10 ** rng.uniform(-3, 1)   # 0.001 … 10
        cand.append(_distribution_beta_sym(n, p))

    # random asymmetric beta
    for _ in range(5):
        a = 10 ** rng.uniform(-3, 2)
        b = 10 ** rng.uniform(-3, 2)
        cand.append(_distribution_beta_asym(n, a, b))

    # Dirichlet with low concentration
    for _ in range(5):
        a = rng.uniform(0.01, 0.5)
        cand.append(_distribution_dirichlet(n, rng, alpha=a))

    # increasing power families
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    # exponential families
    for lam in (0.001, 0.003, 0.005):
        cand.append(_distribution_exponential_increasing(n, lam))

    # sine‑power shapes
    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_sine_power(n, p))

    # a couple of sparse patterns
    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # completely random probability vectors
    for _ in range(4):
        v = rng.random(n)
        v = np.maximum(v, 0.0)
        v /= v.sum()
        cand.append(v)

    return cand


# ----------------------------------------------------------------------
def _smooth_max_descent(a: np.ndarray,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        eta0: float = 0.7,
                        alpha0: float = 1.0,
                        alpha_max: float = 80.0,
                        max_iter: int = 200000,
                        eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth max
    f_α(a) = (1/α)·log Σ exp(α·conv_k).
    The temperature α is raised geometrically up to α_max.
    """
    n = a.shape[0]
    t = 0
    alpha = alpha0
    while t < max_iter and time.time() < deadline:
        t += 1

        # full self‑convolution
        conv = _fft_self_convolve(a)

        # soft‑max weights (stable)
        shift = conv.max()
        w = np.exp(alpha * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w /= w_sum

        # sub‑gradient (averaged over all indices)
        grad = _gradient_from_weights(a, w)

        # step‑size η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(t)

        # multiplicative (exponentiated) update
        a = a * np.exp(-eta * grad)

        # project back onto the simplex (keeps sum = 1)
        a = _project_onto_simplex(a, s=1.0)

        # slowly increase temperature
        if t % 400 == 0 and alpha < alpha_max:
            alpha = min(alpha * 1.25, alpha_max)

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                 deadline: float,
                                 best_seq: np.ndarray,
                                 best_score: float,
                                 lr0: float = 0.5,
                                 max_iter: int = 200000,
                                 eval_interval: int = 500) -> Tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent where the weight vector is uniform over
    the active set (those indices attaining the current maximal convolution).
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a, s=1.0)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] multimax it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """
    Simple greedy mass‑transfer walk.
    Pick a source index i with non‑zero mass, move a random fraction
    (up to 30 %) of its weight to a randomly chosen destination j.
    Keep the move only if the exact objective improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src_idx = np.where(a > 1e-12)[0]
        if src_idx.size == 0:
            break
        i = rng.choice(src_idx)

        # pick a destination distinct from i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # random fraction of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs up to ``budget_s`` seconds and returns the best sequence
    (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0      # safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to explore.  Small to medium sizes give cheap iterations,
    #    yet we also try a few larger values for diversity.
    lengths = [800, 1600, 2400, 3200, 4000, 6000, 8000, 12000]

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    top_candidates: List[Tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 2. Scan analytic families for every length (very cheap).
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] generating candidates for length {n}")

        cand_list = _initial_candidates(n, rng)

        # occasionally include the previously found solution (if any)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
            if prev.ndim == 1 and prev.size == n:
                cand_list.append(prev.copy())
        except Exception:
            pass

        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s                       # normalise to sum = 1
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only the most promising across all lengths seen so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:15]

    # ------------------------------------------------------------------
    # 3. If we somehow have no candidate, fall back to a uniform vector.
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop.
    #    We repeatedly pick the current best few seeds, give each a time slice,
    #    and run the three deterministic phases.
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:  # not enough time for another full round
            break

        # take the top 5 distinct seeds
        seeds = []
        for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]:
            if seeds and np.may_share_memory(seq, seeds[-1]):
                continue
            seeds.append(seq.copy())
        if overall_best_seq is not None and not any(np.may_share_memory(overall_best_seq, s) for s in seeds):
            seeds = [overall_best_seq.copy()] + seeds

        per_seed_deadline = time.time() + remaining / max(1, len(seeds))

        for seed_seq in seeds:
            if time.time() >= deadline:
                break
            # ---------- smooth‑max descent ----------
            a, sc = _smooth_max_descent(
                seed_seq.copy(),
                per_seed_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                eta0=0.8,
                alpha0=0.5,
                alpha_max=120.0,
                max_iter=200000,
                eval_interval=800,
            )
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] after smooth descent new best {overall_best_score:.6f}")

            # ---------- multimax sub‑gradient descent ----------
            a, sc = _subgradient_multimax_descent(
                a.copy(),
                per_seed_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                max_iter=200000,
                eval_interval=800,
            )
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] after multimax descent new best {overall_best_score:.6f}")

            # ---------- aggressive local swaps ----------
            a, sc = _pairwise_swap_local(
                a.copy(),
                per_seed_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
            )
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] after local swaps new best {overall_best_score:.6f}")

        # keep only the best few for the next outer round

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
