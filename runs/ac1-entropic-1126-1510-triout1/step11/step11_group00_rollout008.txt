def _smooth_gradient_descent(a, deadline, best_seq, best_score,
                             lr0=0.5, beta0=2.0, beta_mul=1.01, max_iter=500000,
                             eval_interval=200, patience=5000):
    t=0; no_improve=0; rng = np.random.default_rng()
    n = a.shape[0]
    # Precompute size for FFT for gradient: conv size = 3n-2 -> pow2 size
    grad_fft_size = 1 << ((3*n - 2 - 1).bit_length())
    while time.time() < deadline and t < max_iter:
        t+=1
        # compute self-convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # compute softmax weights
        # Use stable exponentials
        # Beta schedule: increase slowly
        beta = beta0 * (beta_mul ** (t//1000)) # or linear increase
        # compute w
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # compute gradient using w
        w_rev = w[::-1]
        # FFT conv a * w_rev
        A = np.fft.rfft(a, n=grad_fft_size)
        Wr = np.fft.rfft(w_rev, n=grad_fft_size)

        conv_rev = conv_rev_full[:3*n-2]  # ensure length
        grad_base = conv_rev[2*n-2 - np.arange(n)]
        grad = 2.0 * grad_base
        # update a via exponentiated gradient
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # renormalize
        a_sum = a.sum()
        if a_sum <= 0:
            a = np.full(n, 1.0/n)

            a = a / a_sum
        # optionally reduce jitter

def _smooth_gradient_descent_multi(a, deadline, best_seq, best_score,
                                  lr0=0.6, beta0=1.0, beta_mul=1.01,
                                  eval_interval=200, patience=3000):
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    # Determine FFT size for convolution length 3n-2
    conv_len = 3*n - 2
    fft_size = 1 << ((conv_len - 1).bit_length())
    # Pre-allocate maybe not needed.
    while time.time() < deadline:
        t += 1
        # self convolution
        conv = _fft_self_convolve(a)
        max_c = conv.max()
        # compute beta schedule
        beta = beta0 * (beta_mul ** (t // 500))   # increase every 500 iterations
        # compute softmax weights
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        w_rev = w[::-1]
        A = np.fft.rfft(a, n=fft_size)
        Wr = np.fft.rfft(w_rev, n=fft_size)
        conv_rev_full = np.fft.irfft(A * Wr, n=fft_size)
        conv_rev = conv_rev_full[:conv_len]
        grad = 2.0 * conv_rev[2*n-2 - np.arange(n)]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)
        # renorm
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The search proceeds in four stages:

1. **Diverse seeds** – a rich pool of analytic families (uniform, arcsine,
   symmetric Beta, edge‑weighted, sine‑weighted, exponential‑edge) and random
   Dirichlet / sparse vectors is sampled for several lengths.  The best ~20
   seeds are retained.

2. **Smooth gradient descent** – for each seed we run a mirror‑descent style
   optimisation on the exact objective.  The objective
   ``2·n·max(conv(a,a))`` (with ``sum(a)=1``) is smoothed by a soft‑max
   approximation ``log Σ exp(β·conv)``; the gradient of the smoothed loss is
   computed analytically via two FFT‑based convolutions.  The learning rate
   decays as 1/√t and β is increased slowly, so the method first moves
   quickly on a smooth landscape and later concentrates on the true maximum.

3. **Greedy pairwise swaps** – a deterministic local refinement that transfers
   a small amount of mass between two positions whenever the objective improves.

4. **Final simulated‑annealing walk** – a cheap stochastic mass‑transfer walk
   with a Metropolis acceptance rule to escape shallow local minima.

All steps respect the supplied time budget (default 1000 s).  The returned list
contains non‑negative heights; ``evaluate_sequence`` will handle normalisation,
clipping, etc.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2n‑1)."""
    n = a.shape[0]
    need = 2 * n - 1
    size = 1 << ((need - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:need]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector normalised to sum = 1.
    The function is homogeneous of degree 0, therefore we keep the sum
    normalised to 1 during optimisation.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_weighted(n: int, p: float) -> np.ndarray:
    """Weight proportional to sin(π (i+½)/n)^p, vanishing at the ends."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.sin(theta) ** p
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential_edge(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay away from the centre."""
    i = np.arange(n, dtype=np.float64)
    dist = np.exp(-lam * np.minimum(i, n - 1 - i))
    if dist.sum() == 0.0:
        dist = np.full_like(dist, 1.0 / n)
    else:
        dist /= dist.sum()
    return dist


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth gradient descent (soft‑max smoothing) -------------------------
def _smooth_gradient_descent_multi(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    beta0: float = 1.0,
    beta_mul: float = 1.01,
    eval_interval: int = 200,
    patience: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed loss
        L(a) = log Σ_k exp(β·conv_k(a))
    where conv = a*a (full convolution).  The gradient of L is computed
    analytically via FFTs.  The update is multiplicative (exponentiated‑gradient)
    and the point is projected back onto the simplex by renormalisation.
    """
    n = a.shape[0]
    t = 0
    no_improve = 0
    rng = np.random.default_rng()
    # FFT size for the cross‑correlation a * w_rev (length 3n‑2)
    conv_len = 3 * n - 2
    fft_size = 1 << ((conv_len - 1).bit_length())

    while time.time() < deadline:
        t += 1

        # --- self‑convolution -------------------------------------------------
        conv = _fft_self_convolve(a)          # length 2n‑1
        max_c = conv.max()

        # --- soft‑max weights ------------------------------------------------
        beta = beta0 * (beta_mul ** (t // 500))   # increase β slowly
        # shift by max to avoid overflow
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()            # shape (2n‑1,)

        # --- gradient of the smoothed loss ----------------------------------
        w_rev = w[::-1]                          # reverse for cross‑correlation
        A = np.fft.rfft(a, n=fft_size)
        Wr = np.fft.rfft(w_rev, n=fft_size)
        conv_rev_full = np.fft.irfft(A * Wr, n=fft_size)
        conv_rev = conv_rev_full[:conv_len]      # length 3n‑2
        # extract the needed slice: grad_i = 2 * Σ_j w_{i+j} a_j
        grad_base = conv_rev[2 * n - 2 - np.arange(n)]
        grad = 2.0 * grad_base

        # --- multiplicative update (exponentiated gradient) -----------------
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # --- projection onto the simplex (renormalise) -----------------------
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # --- occasional exact evaluation -------------------------------------
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                no_improve = 0
                print(f"[{time.time():.1f}] smooth descent it={t} new best {best_score:.6f}")
            else:
                no_improve += 1

        # --- jitter when stuck ------------------------------------------------
        if no_improve >= patience:
            jitter = rng.exponential(scale=0.005, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            no_improve = 0

    return best_seq, best_score


# ----------------------------------------------------------------------
# Pairwise swap (deterministic local refinement) ------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small mass; keep the move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)

        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * rng.random() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3e} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing style local search ---------------------------------
def _random_local_search(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    init_temp: float = 1e-2,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """Very cheap SA that repeatedly transfers a random amount of mass."""
    n = a.shape[0]
    rng = np.random.default_rng()
    T = init_temp
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * rng.random() * 0.3
        if delta < 1e-14:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] SA improve {i}->{j} Δ={delta:.3e} best {best_score:.6f}")
        else:
            # metropolis acceptance
            if rng.random() < math.exp(-(cur - best_score) / max(T, 1e-12)):
                a = new_a
        T *= decay
    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns a plain Python list; the caller will apply the remaining checks.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0           # safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    candidate_pool: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Build a wide pool of analytic / random seeds
    lengths = [8000, 12000, 20000]
    # p‑grid spans a wide range of shapes (U‑shaped to mild)
    p_grid = np.concatenate(
        [
            np.logspace(-3, -1, 9),      # 0.001 … 0.1
            np.logspace(-1, 0, 9),       # 0.1 … 1
            np.logspace(0, 0.5, 7),      # 1 … 3.16
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 7)      # mixing with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # deterministic families
        for cand in (_distribution_uniform(n), _distribution_arcsine(n)):
            sc = _objective(cand)
            candidate_pool.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(f"[{time.time():.1f}] new global best len={n} score={sc:.6f}")

        # parametrised families
        for p in p_grid:
            if time.time() >= deadline:
                break
            beta = _distribution_beta_sym(n, p)
            edge = _distribution_edge_weighted(n, p)
            sine = _distribution_sine_weighted(n, p)
            expo = _distribution_exponential_edge(n, p)

            for base in (beta, edge, sine, expo):
                for lam in lam_grid:
                    if time.time() >= deadline:
                        break
                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    candidate_pool.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score, overall_best_seq = sc, cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.2f} score={sc:.6f}"
                        )

        # random Dirichlet seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            candidate_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # sparse random seeds
        for _ in range(6):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.008)
            sc = _objective(a)
            candidate_pool.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the strongest seeds overall (to limit memory)
        candidate_pool.sort(key=lambda x: x[0])
        candidate_pool = candidate_pool[:40]

    # ------------------------------------------------------------------
    # 2. Incorporate a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")

    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if overall_best_seq is None:

        overall_best_seq = _distribution_uniform(n)

        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth gradient descent on the most promising seeds

    return overall_best_seq.tolist()
