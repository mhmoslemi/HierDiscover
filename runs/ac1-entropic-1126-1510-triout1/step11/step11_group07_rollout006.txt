"""
Hybrid search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm works in four phases:

1️⃣  **Broad exploration** – enumerate several analytic families
    (arcsine, symmetric/asymmetric Beta, edge‑biased power‑law,
    Dirichlet draws) for a few lengths.  The best few candidates are
    kept.

2️⃣  **Soft‑max mirror descent** – starting from each retained
    candidate we run many exponentiated‑gradient steps on the
    smooth‑max approximation of the max‐convolution.  The temperature
    (α) grows to ≈10⁶ while the learning‑rate decays, giving a smooth
    transition from exploration to high‑precision optimisation.

3️⃣  **Projected Adam sub‑gradient descent** – we refine the result
    with an Adam‑style update on the true max‑convolution (active‑set
    gradient).  A simple back‑tracking line‑search guarantees descent.

4️⃣  **Polishing** – a mass‑transfer hill‑climber and a series of
    random convex combinations of the top solutions finish the search.

The whole routine is time‑aware (default 1000 s) and always returns the
best sequence found before the deadline.  All vectors are kept on the
probability simplex (sum = 1) so the evaluation formula reduces to
`2·n·max(conv)`.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    a is assumed to be a probability vector (sum = 1).
    Returns the exact value produced by `evaluate_sequence`,
    i.e. 2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (each returns a probability vector, sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)                # stability
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """Deterministic edge‑biased distribution:
    w_i ∝ ((i+½)(n‑i‑½))^{‑α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Random Dirichlet draw whose concentration is larger at the edges.
    The concentration parameters are ((i+½)(n‑i‑½))^{‑beta_exp}+0.1.
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float,
                            eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective f_α(a) = (1/α)·log Σ exp(α·conv_k).

    Returns the updated probability vector and the (exact) max of the
    current convolution.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    # exponentiated‑gradient update (KL‑mirror)
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    Projected Adam step on the true max‑convolution.
    Returns (new_a, new_velocity, new_max_conv).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set (within tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # simple back‑tracking line‑search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        cur_score = _max_convolution_score(a)
        if cand_score <= cur_score or step < 1e-9:
            break
        step *= 0.5

    new_max = _fft_self_convolve(a_cand).max()
    return a_cand, vel, new_max


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Tiny mass‑transfer hill‑climber.
    Repeatedly moves a little probability mass from the index that
    contributes most to the current peak towards a globally minimal entry.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            target = np.argsort(a)[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M

        # occasional jitter to avoid stagnation
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

            if cur_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cur_score:.6f}")

            # occasional jitter
            if sub_it % 700 == 0:
                jitter = np.random.randn(n) * 1e-5
                a = np.maximum(a + jitter, eps)
                a /= a.sum()

        # ------------------------------------------------------------------
        # 4️⃣  Tiny hill‑climber polish
        # ------------------------------------------------------------------
        a = best_seq.copy()
        a = _mass_transfer_hill_climb(a, max_hill_iters=6000, eps=eps)
        final_score = _max_convolution_score(a)
        if final_score < best_score:
            best_seq, best_score = a.copy(), final_score
            print(f"[{time.time():.1f}] hill‑climber improved score to {final_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Random convex combinations of the top solutions (global polishing)
    # ------------------------------------------------------------------
    if len(top_candidates) >= 2:
        comb_iters = 300
        vectors = [vec for _, vec in top_candidates]
        for _ in range(comb_iters):
            if time.time() >= deadline:
                break
            # pick 2 or 3 vectors at random
            k = random.choice([2, 3])
            chosen = random.sample(vectors, k)
            weights = np.random.dirichlet(np.ones(k))
            combo = sum(w * v for w, v in zip(weights, chosen))
            combo = np.maximum(combo, eps)
            combo /= combo.sum()
            sc = _max_convolution_score(combo)
            if sc < best_score - 1e-12:
                best_seq, best_score = combo.copy(), sc
                print(f"[{time.time():.1f}] random combo improved best to {sc:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
