def solve_optimal_g(f, M):
    n = len(f)
    # Build A_ub as sparse matrix
    # Data: repeat f_i n times each.
    data = np.repeat(f, n).astype(np.float64)  # length n*n
    # Row indices: for each i (col i), rows i...i+n-1.
    # Create row_offset = np.arange(n)[None,:] + np.arange(n)[:,None] -> shape (n,n). flatten.
    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    col_idx = np.repeat(np.arange(n), n)
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    # bounds: each variable >=0, unbounded above
    # RHS vector:
    b = np.full(2*n-1, M, dtype=np.float64)
    # Solve
    # Use high-s sparsity; Choose method='highs'.
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return uniform
        return np.full(n, 1.0/n)
    g0 = res.x
    # If sum(g0) is very small, fallback uniform
    if g0.sum() < 1e-12:
        return np.full(n, 1.0/n)
    g = g0 / g0.sum()
    return g

def solve_g_lp(f, M):
    # returns normalized vector g
    n = int(f.shape[0])
    # Build CSR.
    # Using data = np.repeat(f, n), row_idx = (np.arange(n)[:,None] + np.arange(n)[None,:]).ravel()
    # col_idx = np.repeat(np.arange(n), n)

    # The row indices may exceed 2n-2. Since maximum i + (n-1) = 2n-2.
    # So shape is (2*n-1, n).

    row_idx = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    col_idx = np.repeat(np.arange(n), n)
    data = np.repeat(f, n)
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    # Use linprog:
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    if not res.success:
        # fallback uniform
        return np.full(n, 1.0/n)
    g = res.x
    sum_g = g.sum()
    if sum_g == 0:
        return np.full(n, 1.0/n)
    return g / sum_g

"""
Propose a (near‑optimal) non‑negative step function.

Algorithm
---------

1. **Initial guess** – use the arcsine (Beta(½,½)) density on a modest grid
   (`n≈3000`).  This distribution is known to be close to optimal.

2. **Linear‑programming refinement** – for a fixed vector `f` we solve
   the LP

        maximise   Σ g_i
        subject to  f * g ≤ M   (component‑wise)
                    g_i ≥ 0

   where `M = max(f*f)`.  The constraints are linear because the convolution
   in the first argument is a Toeplitz matrix built from `f`.  The LP pushes
   probability mass to positions where `f` is smallest, thereby lowering the
   peak of the self‑convolution when the two are mixed.  We perform a few
   iterations of the standard exchange step `f←(1‑t)f + t·g` with a
   decreasing mixing coefficient `t`.

3. **Local mass‑transfer hill climbing** – after the LP phase we repeatedly
   move a tiny amount of probability mass from the pair that currently
   contributes most to the maximum convolution value towards the globally
   smallest entry.  The move is accepted only if it strictly reduces the
   peak.  This simple greedy procedure often yields the last few
   improvements.

4. **Time‑budget handling** – the whole search respects the supplied budget
   (default 1000 s).  The best sequence seen at any moment is returned.

The objective evaluated by `evaluate_sequence` is
    2·n·max(f*f)   (the sum of the vector is irrelevant because of scaling
    invariance),  so we keep all vectors normalised to sum = 1 and work
    directly with the exact quantity `2·n·max(conv)`.
"""

import time
import random
import math
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Exact value returned by `evaluate_sequence` for a normalised vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Analytic families (only arcsine is needed for the core search)
# ----------------------------------------------------------------------


def _distribution_arcsine(n: int) -> np.ndarray:
    """Discrete arcsine law: a_i ∝ 1/√((i+½)(n‑i‑½))."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _solve_g_lp(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ g_i  s.t.  f * g ≤ M  (component‑wise),  g≥0.
    Returns a *normalised* feasible vector g (Σ g = 1).  Falls back to uniform.
    """
    n = f.shape[0]
    # Build the Toeplitz‑style constraint matrix A (size (2n‑1) × n)
    # A[k,i] = f_i   when 0 ≤ k‑i < n   else 0
    i = np.arange(n, dtype=np.int32)
    row_idx = (i[:, None] + np.arange(n, dtype=np.int32)[None, :]).ravel()
    col_idx = np.repeat(i, n)
    data = np.repeat(f, n)
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

    # RHS: every convolution entry must be ≤ M
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise Σ g_i  ==>  minimise −Σ g_i
    c = -np.ones(n, dtype=np.float64)

    # Bounds: g_i ≥ 0
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if res.success:
            g0 = res.x
            s = g0.sum()
            if s > 0:
                return g0 / s
    except Exception:
        pass
    # fallback – uniform distribution
    return np.full(n, 1.0 / n, dtype=np.float64)


def _lp_refine(a: np.ndarray, schedule: list[float]) -> np.ndarray:
    """Iteratively apply the LP exchange step using the supplied mixing schedule."""
    for t in schedule:
        conv = _fft_self_convolve(a)
        M = conv.max()
        g = _solve_g_lp(a, M)
        a = (1.0 - t) * a + t * g
        # keep numerical safety
        a = np.maximum(a, 1e-12)
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Mass‑transfer hill‑climber
# ----------------------------------------------------------------------


def _mass_transfer_hill_climb(a: np.ndarray, max_iters: int,
                             deadline: float, eps: float = 1e-12) -> np.ndarray:
    """
    Greedy hill‑climber that repeatedly moves a tiny amount of probability mass
    from the pair (i,j) contributing most to the current peak towards the
    globally smallest entry.
    """
    n = a.shape[0]
    while max_iters > 0 and time.time() < deadline:
        conv = _fft_self_convolve(a)
        M = conv.max()
        k = int(conv.argmax())                      # location of the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # vectorised search for (i,j) with maximal a_i * a_j for this k
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        prods = a[i_range] * a[j_range]
        best_local = int(prods.argmax())
        best_i = i_range[best_local]
        best_j = j_range[best_local]

        # source = the larger of the two, target = global minimum
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = int(np.argmin(a))
        if target == source:
            # pick the second smallest entry if they coincide
            sorted_idx = np.argsort(a)
            target = int(sorted_idx[1])

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            max_iters -= 1
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
        max_iters -= 1
    return a
