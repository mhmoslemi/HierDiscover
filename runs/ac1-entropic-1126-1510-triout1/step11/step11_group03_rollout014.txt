"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the ordinary linear convolution (np.convolve).

The algorithm is a lightweight “soft‑max” gradient descent
followed by a simple stochastic local‑swap polishing.
It runs inside the given time budget (default 1000 s) and returns the
best sequence found.

Key ideas
---------

1. **Analytic seeds** – a diverse family of analytic PDFs
   (uniform, arcsine, Beta, power‑law, exponential, two‑block,
   Hann / Hamming windows, Dirichlet, sparse) are instantiated for a
   single problem size (n≈8000).  All seeds are normalised to sum = 1
   and their exact objective is evaluated.

2. **Smooth exponential‑gradient descent** – the nondifferentiable
   `max` is replaced by a temperature‑controlled soft‑max
   `w_k ∝ exp(α·conv_k)`.  The gradient w.r.t. the weights is the
   cross‑correlation `2·a⋆w`.  An exponentiated‑gradient (mirror‑descent)
   step

        a ← a·exp(‑η·grad)

   keeps the iterates non‑negative; a Euclidean projection onto the
   simplex restores the normalisation.  The temperature `α` is raised
   from a small start (0.1) to a large end (≈200) during the run, and
   the learning‑rate follows `η = η0 / sqrt(t)`.

3. **Greedy mass‑transfer polish** – after the continuous descent a cheap
   random‑mass‑transfer walk (move a fraction of the mass from one index
   to another) is performed, keeping a move only if the *true* (non‑smoothed)
   objective improves.

The whole routine respects the time limit, prints occasional progress
messages and finally returns the best sequence as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/(sum(a)**2).  Assumes sum>0."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a given weight vector w (length 2·n‑1, sum(w)=1).
    Implemented by FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = _next_pow2(n + len(w) - 1)
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    vals /= vals.sum()
    return vals


def _distribution_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a broad pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_hann(n))
    cand.append(_distribution_hamming(n))

    for p in (0.05, 0.1, 0.2, 0.4, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(4):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    return cand


# ----------------------------------------------------------------------
def _smooth_grad_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float,
                         lr0: float = 1.0,
                         max_iter: int = 200000,
                         eval_interval: int = 400,
                         alpha_start: float = 0.1,
                         alpha_end: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the max.
    The schedule for the temperature α is exponential, which concentrates
    the soft‑max more quickly than the original linear schedule.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # exponential temperature schedule
        progress = t / max_iter
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        # soft‑max weights (stable)
        exp_vals = np.exp(alpha * (conv - max_conv))
        w = exp_vals / exp_vals.sum()

        # gradient of the smoothed objective
        grad = _gradient_from_weights(a, w)

        # exponentiated‑gradient step
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # keep on the simplex
        s = a.sum()
        if s == 0.0:
            a = _distribution_uniform(n)
        else:
            a = a / s

        # occasional exact evaluation
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] iter={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Simple greedy mass‑transfer walk.  Randomly move a fraction of mass from
    one index to another; keep the modification only if the true objective
    improves.
    """
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick source proportional to its mass – this concentrates moves on the larger entries
        src = rng.choice(n, p=a / a.sum())
        j = rng.integers(n)
        if j == src:
            continue

        delta = a[src] * rng.random() * 0.30          # up to 30 % of the source mass
        if delta < 1e-15:
            continue

        new_a = a.copy()
        new_a[src] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {src}→{j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Entry point required by the judge.
    It respects the total time budget ``budget_s`` seconds,
    builds a handful of analytic seeds (length ≈ 8000),
    runs a long soft‑max gradient descent on the most promising ones
    and finally performs greedy local polishing.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Problem size – a single medium‑size problem gives a good trade‑off
    n = 8000

    # ------------------------------------------------------------------
    # 2. Generate analytic seeds and keep an elite set
    candidates = []
    overall_best_seq = None
    overall_best_score = float("inf")

    print(f"[{time.time():.1f}] generating analytic seeds for n={n}")
    for cand in _initial_candidates(n, rng):
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = _objective(cand)
        candidates.append((sc, cand.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = cand.copy()
            print(f"[{time.time():.1f}] !!! new global best (seed) score={sc:.6f}")

    # keep only the top few seeds
    candidates.sort(key=lambda x: x[0])
    elite = candidates[:8]    # keep 8 best seeds

    # ------------------------------------------------------------------
    # 3. Main optimisation – extended smooth gradient descent
    remaining = deadline - time.time()
    if remaining > 30.0:
        per_seed = remaining / len(elite)
        for idx, (sc, seed_seq) in enumerate(elite):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            print(f"[{time.time():.1f}] gradient descent on seed {idx+1}/{len(elite)} (score={sc:.6f})")
            seq_out, score_out = _smooth_grad_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=1.0,
                max_iter=200000,
                eval_interval=400,
                alpha_start=0.1,
                alpha_end=200.0,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after descent new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final polishing – cheap local swaps
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Convert to plain Python list (as required by the judge)
    return overall_best_seq.tolist()
