Finally a cheap greedy “mass‑transfer” walk (random δ‑transfer from
one index to another) is applied for polishing.  The whole search
terminates well before the allotted 1000 s deadline, returning the
best sequence found.

All helper functions are defined at top level; no closures or lambdas
are used.  The code respects the 2‑CPU limit and uses only NumPy,
SciPy and the provided ``evaluate_sequence`` routine.
"""

import time
import math
import random
from typing import List, Tuple

import numpy as np
import scipy.sparse
import scipy.optimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)² (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)


def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w for a weight vector w (length 2n‑1).
    Implemented using FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                     # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (sum = 1)
# ----------------------------------------------------------------------


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_power_decreasing(n: int, p: float) -> np.ndarray:
    # just reverse the increasing version
    return _distribution_power_increasing(n, p)[::-1]


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_exponential_decreasing(n: int, lam: float = 0.005) -> np.ndarray:
    return _distribution_exponential_increasing(n, lam)[::-1]


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> List[np.ndarray]:
    """Wide pool of analytic seeds for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))
        cand.append(_distribution_power_decreasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))
        cand.append(_distribution_exponential_decreasing(n, lam))

    for block_frac in (0.03, 0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.008))

    return cand


# ----------------------------------------------------------------------
# Smooth exponentiated‑gradient descent (soft‑max approximation)
# ----------------------------------------------------------------------


def _smooth_exponentiated_gradient(a: np.ndarray,
                                   deadline: float,
                                   best_seq: np.ndarray,
                                   best_score: float,
                                   lr0: float = 0.9,
                                   max_iter: int = 80000,
                                   eval_interval: int = 500,
                                   alpha_start: float = 0.5,
                                   alpha_end: float = 30.0) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a soft‑max approximation of the max‑convolution.
    The iterate is always projected onto the probability simplex.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        max_conv = conv.max()

        # linear temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * t / max_iter

        exp_vals = np.exp(alpha * (conv - max_conv))          # stability trick
        w = exp_vals / exp_vals.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)               # element‑wise exponentiated step
        a = _project_onto_simplex(a, s=1.0)        # back to simplex

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smooth it={t} α={alpha:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass‑transfer walk (polishing)
# ----------------------------------------------------------------------


def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> Tuple[np.ndarray, float]:
    """Randomly moves a fraction of mass from one index to another, keeping improvements."""
    n = a.shape[0]
    rng = np.random.default_rng()
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = rng.choice(src)
        j = rng.integers(n)
        if j == i:
            continue

        delta = a[i] * rng.random() * 0.30   # up to 30 % of source mass
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (the “LP‑step” from the statement)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """Return a sparse matrix A where A @ g = conv(f,g) (full linear convolution)."""
    n = f.shape[0]
    # rows for each constraint k = i + j
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.concatenate([np.full(n, j, dtype=int) for j in range(n)])
    data = np.tile(f, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _lp_refine(f: np.ndarray,
               max_iters: int = 3,
               mix_rate: float = 0.2,
               deadline: float = 0.0) -> np.ndarray:
    """
    Perform a few LP‑based refinement steps as described in the problem statement.
    ``mix_rate`` is the step size for the convex combination f←(1‑t)f + t·g.
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current bound
        M = _fft_self_convolve(f).max()

        # Linprog: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)

        bounds = [(0.0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = scipy.optimize.linprog(c,
                                     A_ub=A,
                                     b_ub=b_ub,
                                     bounds=bounds,
                                     method='highs',
                                     options={'presolve': True, 'time_limit': max(30, int(deadline - time.time()))})

        if not res.success:
            # LP failed – abort the refinement
            print(f"[{time.time():.1f}] LP iteration {it} failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0.0:
            break

        # Scale g back to a probability distribution (sum = 1)
        g = g / g.sum()

        # convex combination
        f = (1.0 - mix_rate) * f + mix_rate * g
        f = _project_onto_simplex(f, s=1.0)   # numerical safety

        cur = _objective(f)
        print(f"[{time.time():.1f}] LP refinement it={it} score={cur:.6f}")

    return f


# ----------------------------------------------------------------------
# Main entry point required by the checker
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000) -> List[float]:
    """
    Search for a step‑function (list of non‑negative heights) with the smallest
    possible value of ``evaluate_sequence``.
    The routine respects the total time budget ``budget_s`` (seconds) and
    returns the best sequence it discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths to explore
    # (shorter lengths are cheaper for the LP‑refinement)
    lengths = [2000, 3000, 4000, 6000, 8000]

    overall_best_seq = None
    overall_best_score = float("inf")
    elite = []   # (score, vector) across all lengths

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] --- scanning length = {n} ---")
        for cand in _initial_candidates_extended(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand = _project_onto_simplex(cand, s=1.0)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] !!! new global best len={n} score={sc:.6f}")
            elite.append((sc, cand.copy()))

        # keep only top‑12 elite across all lengths seen so far
        elite.sort(key=lambda x: x[0])
        elite = elite[:12]

    # ------------------------------------------------------------------
    # 3. If a previous solution exists (height_sequence_1) incorporate it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0.0:
                prev = _distribution_uniform(prev.shape[0])
            prev = _project_onto_simplex(prev, s=1.0)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            elite.append((sc, prev.copy()))
            elite.sort(key=lambda x: x[0])
            elite = elite[:12]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback if nothing was generated (should never happen)
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Intensive smooth‑gradient descent on the elite seeds
    remaining = deadline - time.time()
    if remaining > 30.0:
        # run on at most 5 elite seeds, divide time equally
        seeds = [seq for _, seq in sorted(elite, key=lambda x: x[0])[:5]]
        per_seed = max(5.0, remaining / len(seeds))
        for idx, seed_seq in enumerate(seeds, 1):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            print(f"[{time.time():.1f}] smooth descent on seed {idx}/{len(seeds)} (len={len(seed_seq)})")
            seq_out, score_out = _smooth_exponentiated_gradient(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.9,
                max_iter=90000,
                eval_interval=600,
                alpha_start=0.5,
                alpha_end=30.0,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after smooth descent new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. LP‑based polishing (only if the length is modest)
    if time.time() < deadline and overall_best_seq.shape[0] <= 4000:
        print(f"[{time.time():.1f}] starting LP‑refinement (len={overall_best_seq.shape[0]})")
        overall_best_seq = _lp_refine(
            overall_best_seq.copy(),
            max_iters=4,
            mix_rate=0.15,
            deadline=deadline - 5.0,
        )
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] after LP refinement score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final cheap greedy polishing
    if time.time() < deadline:
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
