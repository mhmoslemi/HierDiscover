"""
Hybrid optimiser for the step‑function evaluation problem.

The routine builds a diverse pool of initial probability vectors
(normalised to sum=1), picks the best by the cheap proxy
2·n·max(a*a) and then refines it using three stages:

1. Continuation / smoothed‑max minimisation with a projected FISTA
   optimiser (many inner iterations and a very small final smoothing
   parameter μ≈1e‑9).  This stage usually drives the bulk of the gain.

2. Projected sub‑gradient descent on the true max‑convolution objective
   (the same technique used in the reference solution, but with a
   longer schedule).

3. Simple stochastic hill‑climbing on the simplex: repeatedly move a
   small amount of mass from a randomly chosen index i to another j
   and keep the change only if the raw objective 2·n·max(a*a) improves.
   The step size ε is adapted automatically.

All three stages respect the simplex constraints (a_i ≥ 0, Σa_i = 1)
so the final vector can be fed directly to ``evaluate_sequence`` which
rescales it automatically.

The whole search respects the global time‑budget (default 1000 s) and
leaves a safety margin before returning the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf for numerical stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # prevent overflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _sin_distribution(n: int, p: float = 1.0) -> np.ndarray:
    """Sine‑shaped distribution: (sin(π·x))**p on n bins, Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.sin(math.pi * x) ** p
    total = vals.sum()
    if total == 0.0:
        vals[:] = 1.0 / n
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector (Σa=1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights, Σp=1

    # gradient = 2·(a_rev * p)  – take the central n entries
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smoothed objective value
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation + projected FISTA to minimise the smoothed max‑convolution.
    The smoothing parameter μ is reduced geometrically down to 1e‑9.
    """
    a = a0.copy()
    n = a.shape[0]

    # initial smoothing – half of current max‑conv (but not smaller than 1e‑7)
    mu = max(_max_self_convolution(a) * 0.5, 1e-7)
    mu_target = 1e-9          # final smoothing (practically exact max)
    mu_factor = 0.5           # halve μ each outer iteration

    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # ----- inner FISTA for current μ ---------------------------------
        # Lipschitz estimate – safe over‑estimate  (grad of smooth max ≤ 2/μ)
        L = max(1.0, 2.0 / mu)

        y = a.copy()
        t = 1.0

        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break

        # allocate a generous number of cheap iterations, limited only by time left
        # empirically a single iteration costs ~0.0007 s for n≈8000
        iters = int(min(20000, max(1, remaining / 0.0008)))
        if iters <= 0:
            break

        for it in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # back‑tracking line search (max two doublings of L)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-14:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional assessment of the true (non‑smoothed) objective
            if (it & 31) == 0:
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()
        # -----------------------------------------------------------------
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        # warm‑restart from the best sequence at current μ
        a = best_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses a diminishing step‑size 0.5/√t and a simple momentum term.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        # exact sub‑gradient of max_k (a*a)_k
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-13, cur_max * 1e-13)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # step‑size schedule (larger than in the reference code)
        step = 0.5 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional tiny jitter to escape flat plateaux
        if it % 400 == 0 and np.random.rand() < 0.09:
            a += np.random.normal(scale=1e-6, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-13:
            best_val = cur_raw
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _local_hill_climb(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Very simple stochastic hill‑climbing on the simplex.
    Moves a tiny amount of mass from i to j and keeps the move only if
    the raw objective improves.  The step‑size ε is automatically
    reduced when the acceptance rate drops.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    rng = np.random.default_rng()
    start = time.time()
    epsilon = 0.02            # initial relative move size
    it = 0
    accept = 0

    while time.time() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount we can take from i
        max_delta = a[i]
        if max_delta <= 0.0:
            continue
        # relative move, capped to keep enough mass
        delta = rng.random() * min(epsilon * max_delta, max_delta)
        old_i, old_j = a[i], a[j]
        a[i] = old_i - delta
        a[j] = old_j + delta

        cur_val = _raw_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            accept += 1
        else:
            # revert
            a[i] = old_i
            a[j] = old_j

        it += 1
        # adapt ε every few thousand proposals
        if it % 2000 == 0:
            acc_rate = accept / it
            if acc_rate < 0.01:
                epsilon = max(epsilon * 0.5, 5e-4)
            elif acc_rate > 0.1:
                epsilon = min(epsilon * 1.5, 0.4)

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the competition harness.
    Returns a list of non‑negative step heights (the optimiser runs for
    ``budget_s`` seconds and tries to beat the target raw score).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of seed vectors (all normalised to Σ=1)
    n = 8000                                   # length for the final answer
    candidates = []

    # a) warm‑start from the globally supplied construction, if any
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base = base / base.sum()
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # b) simple deterministic baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))                # arcsine

    # c) a collection of symmetric Beta shapes
    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n, a, a))

    # d) a handful of random Beta draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(_beta_distribution(n, alpha, beta))

    # e) a few random Dirichlet draws
    for _ in range(6):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec /= vec.sum()
        candidates.append(vec)

    # f) sine‑shaped families (different exponents)
    for p in (0.5, 1.0, 2.0, 3.0):
        candidates.append(_sin_distribution(n, p))

    # ------------------------------------------------------------------
    # 2. Pick the best raw‑score seed
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    print(f"Initial best raw score (proxy): {best_raw:.6f}")

    # ------------------------------------------------------------------
    # 3. Smoothed‑gradient refinement (continuation + FISTA)
    remaining = deadline - time.time()
    if remaining > 12.0:                         # ensure a decent budget
        fista_budget = remaining * 0.55
        best_seq, best_raw = _fista_refine(best_seq, fista_budget)

    # ------------------------------------------------------------------
    # 4. Projected sub‑gradient descent (polish)
    remaining = deadline - time.time()
    if remaining > 6.0:
        subgrad_budget = remaining * 0.60
        best_seq, best_raw = _subgradient_polish(best_seq, subgrad_budget)

    # ------------------------------------------------------------------
    # 5. Final stochastic hill‑climbing polish
    remaining = deadline - time.time()
    if remaining > 2.0:
        hill_budget = remaining * 0.90
        best_seq, best_raw = _local_hill_climb(best_seq, hill_budget)

    # ------------------------------------------------------------------
    # 6. Return the sequence (the harness will call evaluate_sequence)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
