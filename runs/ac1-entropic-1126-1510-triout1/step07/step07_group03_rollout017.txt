def _lbfgs_obj_and_grad(x, alpha, n):
    a = _softmax(x)
    b = _fft_self_convolve(a)
    maxb = b.max()
    w_raw = np.exp(alpha * (b - maxb))
    w_sum = w_raw.sum()

    w = w_raw / w_sum
    # smoothed max
    s = maxb + np.log(w_sum) / alpha
    obj = 2.0 * n * s
    # grad_a = 2 * conv_rev(w) (size n)
    conv_rev = _fft_convolve(a[::-1], w)

    # gradient of obj wrt a
    grad_a = 2.0 * n * grad_a
    # gradient wrt x
    # softmax Jacobian: grad_x = a * (grad_a - np.sum(a * grad_a))
    dot = np.sum(a * grad_a)
    grad_x = a * (grad_a - dot)
    return obj, grad_x

"""
Optimise a non‑negative step‑function to minimise
``evaluate_sequence`` (2·n·max(conv(a,a)) / sum(a)²).

The algorithm works in three phases:

1. **Broad initialisation** – a collection of analytic families
   (uniform, arcsine, beta, exponential, alternating, block‑spike,
   sparse random) is generated for several lengths.  The ten best
   vectors (lowest exact objective) are kept as seeds.

2. **Refinement** – each seed is refined by a continuation scheme that
   minimises a smooth log‑sum‑exp approximation of the max‑convolution
   using L‑BFGS‑B (softmax re‑parameterisation enforces the simplex
   constraints).  The smoothing parameter α is increased
   progressively (1 → 5 → 20 → 100 → 500 → 2000 → 5000).  After each
   α‑stage the exact objective is evaluated and the best vector is
   retained.

3. **Fine‑tuning** – a greedy active‑set swap, followed by a short
   simulated‑annealing walk, is applied to the currently best vector.

All stages respect the global time budget (default 1000 s).  The final
vector is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int) -> list[np.ndarray]:
    """Generate a broad set of plausible start vectors for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    for lam in (0.001, 0.002, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    for stride in (2, 3, 4):
        cand.append(_distribution_alternating(n, stride))

    cand.append(_distribution_two_blocks(n, block_frac=0.05))
    cand.append(_distribution_two_blocks(n, block_frac=0.10))

    # sparse random start (≈0.5 % non‑zeros)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    return cand

def _smooth_adam_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth approximation of max‑convolution."""
    n = a.shape[0]

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    schedule = [
        (1.0, 2000),
        (5.0, 2000),
        (20.0, 2000),
        (100.0, 2000),
        (500.0, 2000),
        (2000.0, 2000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smooth α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


def _exact_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the genuine max‑convolution objective."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        if best_score <= 1.503:
            break
        if time.time() + 0.5 >= deadline:
            break

    return best_seq, best_score


def _greedy_swap_opt(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from high‑gradient to low‑gradient indices while improving."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]
        j = int(np.argmin(grad))
        if i == j:
            break

        delta = min(a[i] * 0.5, 0.005)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")
        else:
            delta = delta * 0.25
            if delta < 1e-7:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] tiny swap improved best {best_score:.6f}")

    return best_seq, best_score


def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis‑type random walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    T = 0.04
    iter_cnt = 0

    while time.time() < deadline and T > 1e-4:
        iter_cnt += 1

        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter_cnt} new best {best_score:.6f}")

        if iter_cnt % 2500 == 0:
            T *= 0.85

    return best_seq, best_score

# L‑BFGS refinement ----------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    e = np.exp(x - np.max(x))
    return e / e.sum()


def _lbfgs_obj_and_grad(x: np.ndarray, alpha: float, n: int):
    """
    Returns (objective, gradient) for the smoothed max‑convolution
    (log‑sum‑exp) when the vector a is parameterised as softmax(x).
    """
    a = _softmax(x)

    # convolution and log‑sum‑exp of the convolution values
    b = _fft_self_convolve(a)
    maxb = b.max()
    w_raw = np.exp(alpha * (b - maxb))          # stability: maxb is subtracted
    w_sum = w_raw.sum()
    w = w_raw / w_sum

    smooth_max = maxb + np.log(w_sum) / alpha   # (1/α)·log Σ exp(α·b)
    obj = 2.0 * n * smooth_max

    # gradient wrt a  (∇ₐ Σ w_k·b_k = 2·conv_rev(w))
    conv_rev = _fft_convolve(a[::-1], w)
    grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]   # size n
    grad_a = 2.0 * n * grad_a                 # chain rule for the outer factor 2·n

    # chain rule through the softmax
    dot = np.sum(a * grad_a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def _lbfgs_refine(a0: np.ndarray, deadline: float,
                  best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """
    Refine a candidate vector with a continuation of L‑BFGS‑B optimisations
    on increasingly sharp smooth approximations of the max‑convolution.
    """
    n = a0.shape[0]
    # initialise the log‑itinerary
    x = np.log(np.maximum(a0, 1e-12))

    # schedule of smoothing parameters
    alphas = (1.0, 5.0, 20.0, 100.0, 500.0, 2000.0, 5000.0)

    for alpha in alphas:
        if time.time() >= deadline:
            break
        # a modest iteration budget; each iteration costs two FFTs
        max_iter = 250
        res = minimize(_lbfgs_obj_and_grad, x,
                       args=(alpha, n), method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
        x = res.x
        a = _softmax(x)
        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] lbfgs α={alpha:.0f} new best {best_score:.6f}")

    return best_seq, best_score

            prev = prev / prev.sum()
            sc = _objective(prev)
            top_seeds.append((sc, prev))
            print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Optimise each seed (in order of increasing score) until time runs out
    overall_best_seq = None
    overall_best_score = float("inf")

    for seed_idx, (seed_score, seed_vec) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        n = seed_vec.shape[0]
        print(f"[{time.time():.1f}] Optimising seed #{seed_idx+1}/{len(top_seeds)} length={n} init_score={seed_score:.6f}")

        a = seed_vec.copy()
        best_seq = a.copy()
        best_score = seed_score

        # 3a – smoothed Adam descent (lightweight)
        a, best_score = _smooth_adam_descent(a, deadline, best_seq, best_score)

        # 3b – exact sub‑gradient refinement
        a, best_score = _exact_subgradient_descent(a, deadline, best_seq, best_score)

        # 3c – L‑BFGS refinement on the smooth log‑sum‑exp surrogate
        a, best_score = _lbfgs_refine(a, deadline, best_seq, best_score)

        # 3d – greedy active‑set swaps
        a, best_score = _greedy_swap_opt(a, deadline, best_seq, best_score)

        # 3e – simulated‑annealing fine‑tuning
        a, best_score = _simulated_annealing(a, deadline, best_seq, best_score)

        if best_score < overall_best_score - 1e-12:
            overall_best_score = best_score
            overall_best_seq = best_seq.copy()
            print(f"[{time.time():.1f}] *** NEW GLOBAL BEST {overall_best_score:.6f} (len={len(overall_best_seq)})")

        # occasional random perturbation + short L‑BFGS run to escape local minima
        if time.time() < deadline and overall_best_seq is not None:
            pert = overall_best_seq + 0.05 * np.random.randn(*overall_best_seq.shape)
            pert = np.maximum(pert, 0.0)
            pert = _project_onto_simplex(pert, 1.0)
            pert, _ = _lbfgs_refine(pert, deadline, pert, _objective(pert))
            pert_score = _objective(pert)
            if pert_score < overall_best_score - 1e-12:
                overall_best_score = pert_score
                overall_best_seq = pert.copy()
                print(f"[{time.time():.1f}] *** GLOBAL BEST IMPROVED BY PERTURB {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Fallback – should never happen, but safe default.
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
