    best_raw = _raw_score(a)

    mu_target = 1e-6

        # compute if time

            # compute objective + gradient for smoothed max

            max_logit = np.max(logits)
            exps = np.exp(logits - max_logit)
            a_local = exps / exps.sum()
            conv = _self_convolution_full(a_local)
            M = conv.max()

            scaled = (conv - M) / mu
            exp_scaled = np.exp(scaled)

"""
Optimisation of a non‑negative step‑function to minimise
    2·n·max(a*a)  (= evaluate_sequence when Σa = 1).

Algorithm
---------
1. Build a small diverse pool of probability vectors (uniform, arcsine,
   several Beta shapes, random Dirichlet draws, and the global warm‑start
   `height_sequence_1` if present).  Choose the one with the smallest
   raw objective as the seed.

2. Continuation on the smoothed max‑convolution:
   minimise   F_μ(a) = μ·log∑_k exp((a*a)_k / μ)
   using a projected L‑BFGS optimiser on the soft‑max (logits)
   parametrisation of the simplex.  Starting with μ = ½·max(a*a) the
   smoothing is halved repeatedly until μ ≤ 1e‑6.

3. A short projected sub‑gradient descent cleans the final few digits.

The code respects the 1000 s budget, prints progress, and returns a
list of heights (the harness rescales it automatically).  No closures
or lambdas are used – all helpers are top‑level functions.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)
    M = conv.max()
    scaled = (conv - M) / mu               # ≤ 0
    exp_scaled = np.exp(scaled)            # safe: max 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                     # soft‑max weights

    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _lbfgs_obj_and_grad(logits: np.ndarray, mu: float):
    """
    Objective and gradient for L‑BFGS optimisation on the logits.
    The underlying variable a = softmax(logits) lives on the simplex.
    """
    # soft‑max, numerically stable
    m = np.max(logits)
    exp_shift = np.exp(logits - m)
    a = exp_shift / exp_shift.sum()

    conv = _self_convolution_full(a)
    M = conv.max()
    scaled = (conv - M) / mu               # ≤ 0
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    f_val = M + mu * np.log(Z)

    p = exp_scaled / Z
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad_a = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # gradient w.r.t. logits (softmax Jacobian)
    dot = np.dot(a, grad_a)
    grad_logits = a * (grad_a - dot)
    return f_val, grad_logits.astype(np.float64)


def _lbfgs_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation optimiser based on L‑BFGS on the logits.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_raw = _raw_score(a)

    start = time.time()
    mu = _max_self_convolution(a) * 0.5
    mu_target = 1e-6
    mu_factor = 0.5

    while True:
        # time check
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        # L‑BFGS on current μ
        init_logit = np.log(a + 1e-15)   # avoid log(0)
        res = minimize(_lbfgs_obj_and_grad,
                       init_logit,
                       args=(mu,),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300, 'gtol': 1e-12, 'disp': False})

        # recover a from optimiser output
        x_opt = res.x
        m = np.max(x_opt)
        exp_shift = np.exp(x_opt - m)
        a = exp_shift / exp_shift.sum()

        cur_raw = _raw_score(a)
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_seq = a.copy()
            print(f"  L‑BFGS: μ={mu:.2e}, new best raw={best_raw:.6f}")

        # decrease smoothing
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        # warm‑restart from the best so far for the next μ
        a = best_seq.copy()

    return best_seq, best_raw


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=len(a))
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
            print(f"  Sub‑grad iteration {it}: raw={best_val:.6f}")

    return best_seq, best_val
