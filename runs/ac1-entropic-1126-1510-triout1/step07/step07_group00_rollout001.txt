def _sgd_subgradient_descent(a0, deadline, max_iters=500000, eta0=0.5):
    a = a0.copy()

            break
        grad, cur_max, _ = _subgrad(a)

def _random_local_search(a0, deadline, max_iters=200000, eps_init=0.01, eps_decay=0.9999):
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    eps = eps_init
    n = a.size
    for i in range(max_iters):
        if time.time() >= deadline:
            break
        i_src = np.random.randint(0, n)
        i_dst = np.random.randint(0, n)
        if i_src == i_dst:
            continue
        # amount to move
        transfer = min(eps, a[i_src])
        if transfer <= 0:
            continue
        a_new = a.copy()
        a_new[i_src] -= transfer
        a_new[i_dst] += transfer
        # no need to project - stays sum =1, may be non-negative
        cur_score = _max_convolution_score(a_new)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a_new.copy()
            a = a_new
        else:
            # maybe accept with small prob to escape
            if np.random.rand() < 0.001:
                a = a_new
        eps *= eps_decay
    return best, best_score
```

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1) Generate a diverse set of candidate sequences using analytic families
   (arcsine, symmetric beta, triangular, exponential, power‑law) for a
   range of lengths (≈ 5 k … 30 k).  Also add a few random Dirichlet draws.
   All candidates are ℓ₁‑normalised (sum = 1) and scored with the cheap
   surrogate  S(a)=2·n·max(a*a)  (exact when sum=1).

2) Keep the best few seeds (default 10) and improve each of them with a
   suite of local optimisation tools, all respecting the simplex
   constraint:
   * Adam‑style adaptive moment descent.
   * Exponentiated‑gradient (mirror descent w.r.t. KL).
   * Simple projected sub‑gradient descent.
   * Random “mass‑transfer” hill‑climbing (pairwise exchange of a small
     ε amount of probability mass).

   Each optimiser runs until the global deadline (≈ 1000 s) is reached.

3) Starting from the best sequence found so far, perform a Frank‑Wolfe–
   like pairwise mass‑transfer refinement: move probability mass from the
   component with the largest sub‑gradient (most responsible for the
   current peak) to the component with the smallest sub‑gradient, with a
   line‑search (full, half, quarter transfer).  Occasionally add a tiny
   jitter to escape plateaus.

4) Return the best sequence (list of Python floats).  The whole routine
   respects the time budget, prints progress, and never raises an
   exception – the returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate score = 2·n·max(a*a)  (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even‑indexed active positions
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 150_000,
                         eta0: float = 0.6):
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _sgd_subgradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 200_000,
                             eta0: float = 0.5):
    """Plain projected sub‑gradient descent (step ∝ 1/√t)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_local_search(a0: np.ndarray, deadline: float,
                         max_iters: int = 150_000,
                         eps_init: float = 0.02,
                         eps_decay: float = 0.9995):
    """
    Random pairwise mass‑transfer hill‑climbing.
    At each step move a small ε from a random source to a random target.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    eps = eps_init
    n = a.size
    rng = np.random.default_rng()
    for i in range(max_iters):
        if time.time() >= deadline:
            break
        i_src = rng.integers(0, n)
        i_dst = rng.integers(0, n)
        if i_src == i_dst:
            continue
        transfer = min(eps, a[i_src])
        if transfer <= 0:
            continue
        a_new = a.copy()
        a_new[i_src] -= transfer
        a_new[i_dst] += transfer
        cur_score = _max_convolution_score(a_new)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a_new.copy()
            a = a_new
        else:
            # occasional acceptance of non‑improving move to escape flat areas
            if rng.random() < 0.001:
                a = a_new
        eps *= eps_decay
    return best_a, best_score

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # p can be <1, >1 – use log‑space for stability
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 0.5
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(alpha,…,alpha)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()
