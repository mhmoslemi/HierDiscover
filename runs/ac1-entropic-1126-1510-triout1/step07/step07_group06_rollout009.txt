def _beta_start(n, alpha):
    # symmetric Beta(alpha, alpha) distribution over i+0.5 normalized on [0,1]
    x = (np.arange(n, dtype=float) + 0.5) / n
    # compute density = x^{alpha-1} (1-x)^{alpha-1}
    # protect against zero: add epsilon for log.
    eps = 1e-12
    vals = np.maximum(x * (1-x), eps) ** (alpha - 1)
    # However for alpha < 1, exponent is negative, leads to heavy edges.
    # If alpha=1, uniform.
    # Normalize
    s = vals.sum()
    if s>0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals

"""
Propose a step function (non‑negative heights) with a low
evaluate_sequence() score.

Algorithm
---------
* Progressive lengths: 2000 → 4000 → 8000 → 16000 → 32000.
  The best shape found for a length is interpolated and used to warm‑start
  the next length.

* For each length several initial vectors are tried:
    – warm‑start (interpolated previous best)
    – uniform (simplex centre)
    – deterministic arcsine shape (β(0.5,0.5))
    – a few β(α,α) shapes with α∈{0.15,0.25,0.35,0.45}
    – a few random Dirichlet draws.

* Each start is optimised with an Adam optimiser on the smooth
  approximation  f_λ(a)= (1/λ)·log Σ exp(λ·(a*a)_k ).
  λ is increased slowly (λ←λ·λ_factor).  The optimiser works in the
  log‑softmax (log‑it) space, guaranteeing the simplex constraints.

* The best result for the current length is kept.  After the final length
  a projected sub‑gradient descent (on the true max) is performed for a
  remaining time budget, with occasional random perturbations.

* The final vector is returned as a plain Python list; evaluate_sequence()
  will apply the required scaling/clamping.

The implementation is fully self‑contained, uses only numpy/math and
respect the 1000 s time limit.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _uniform_start(n: int) -> np.ndarray:
    """Uniform probability vector."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (β(0.5,0.5))."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _beta_start(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape on the simplex."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    # density proportional to x^{α-1} (1-x)^{α-1}
    vals = np.maximum(x * (1.0 - x), eps) ** (alpha - 1.0)
    # cap extreme values to keep numerics stable
    vals = np.minimum(vals, 1e6)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0.0:
        g = np.ones_like(g)
    return g / g.sum()


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 3000,
                            lr0: float = 0.12,
                            lam_start: float = 0.5,
                            lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed log‑sum‑exp of the self‑convolution.
    Returns the best probability vector (Σa=1) found and its raw (unsmoothed)
    objective 2·n·max_k (a*a)_k .
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_raw = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- simplex vector ----------
        a = _softmax(x)                     # Σa = 1, a ≥ 0

        # ---------- raw objective ----------
        conv = _fft_self_convolve_full(a)   # length 2·n‑1
        cur_max = conv.max()
        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_raw:.6f} λ={lam:.3f}")

        # ---------- smooth max ----------
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()               # (2n‑1,) probabilities

        # ---------- gradient w.r.t. a ----------
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1:n - 1 + n]

        # ---------- chain rule through soft‑max ----------
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- increase λ ----------
        lam *= lam_factor

        # ---------- jitter to escape plateaus ----------
        if it % 500 == 0:
            x += np.random.randn(*x.shape) * 0.008

    if best_a is None:
        best_a = _uniform_start(n)
        best_raw = 2.0 * n * _fft_self_convolve_full(best_a).max()
    return best_a, best_raw

def _optimise_length_with_initialisations(n: int,
                                          deadline: float,
                                          warm_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n using a diverse set of initial vectors.
    Returns the best probability vector and its raw score.
    """
    eps = 1e-12
    init_logits = []

    # Warm‑start (interpolated) ------------------------------------------------
    if warm_seq is not None:
        cand = _interpolate_to_length(warm_seq, n)
        init_logits.append(np.log(cand + eps))

    # Uniform start ------------------------------------------------------------
    init_logits.append(np.zeros(n, dtype=np.float64))

    # Deterministic arcsine ------------------------------------------------------
    init_logits.append(np.log(_arcsine_start(n) + eps))

    # A few Beta(α,α) shapes ----------------------------------------------------
    for alpha in (0.15, 0.25, 0.35, 0.45):
        init_logits.append(np.log(_beta_start(n, alpha) + eps))

    # Random Dirichlet draws ----------------------------------------------------
    for _ in range(2):
        a = _random_dirichlet_start(n)
        init_logits.append(np.log(a + eps))

    best_local_seq = None
    best_local_raw = float("inf")

    # --------------------------------------------------------------
    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        # adapt iteration budget to remaining time (rough estimate)
        remain = deadline - time.time()
        # limit iterations to at most 3000 per start, but fewer if time is short
        max_iters = min(3000, int(remain * 2.0))   # 2 iters per second approx
        if max_iters < 200:
            # not enough time left for a meaningful run
            break
        print(f"[{time.time():.1f}] length={n} start {idx+1}/{len(init_logits)} (max_iters={max_iters})")
        seq, raw = _adam_optimize_smoothed(
            x0, n, deadline,
            max_iters=max_iters,
            lr0=0.12,
            lam_start=0.5,
            lam_factor=1.001
        )
        if raw < best_local_raw:
            best_local_raw = raw
            best_local_seq = seq.copy()
            print(f"  *** new best for n={n}: raw={best_local_raw:.6f}")

    if best_local_seq is None:
        best_local_seq = _uniform_start(n)
        best_local_raw = 2.0 * n * _fft_self_convolve_full(best_local_seq).max()
    return best_local_seq, best_local_raw


# ----------------------------------------------------------------------
def _post_refine(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the true max objective.
    Simple random perturbations are occasionally applied.
    """
    n = a.shape[0]
    best_raw = 2.0 * n * _fft_self_convolve_full(a).max()
    best = a.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        # active set (indices within relative tolerance)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # step size schedule (larger at the beginning)
        step = 0.25 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)

        # tiny occasional noise to jump out of flat regions
        if it % 600 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 3e-7
            a = _project_onto_simplex(a)

        raw = 2.0 * n * conv.max()
        if raw < best_raw - 1e-12:
            best_raw = raw
            best = a.copy()
            print(f"[{time.time():.1f}] post‑refine it={it} raw={best_raw:.6f}")

        # safety break if improvement stalls for a very long time
        if it % 50000 == 0 and time.time() > deadline - 5.0:
            # not enough time left for many more iterations
            break
    return best

    # ------------------------------------------------------------------
    # Length schedule – larger n gives more flexibility but costs more time.
    lengths = [2000, 4000, 8000, 16000, 32000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        seq, raw = _optimise_length_with_initialisations(
            n, deadline, warm_seq=best_seq
        )
        print(f"  finished length {n}: best raw={raw:.6f}")
        if raw < best_raw:
            best_raw = raw
            best_seq = seq.copy()
        # warm‑start for the next length
        prev = best_seq

    # ------------------------------------------------------------------
    # Final refinement using true max (projected sub‑gradient)
    if best_seq is not None:
        best_seq = _post_refine(best_seq, deadline)

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
