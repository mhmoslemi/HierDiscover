import cvxpy as cp

def solve_lp_max_sum(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # create constraints
    constraints = []
    for k in range(2*n-1):
        # compute sum_{i} f[i] * g[k-i]
        i_min = max(0, k - (n-1))
        i_max = min(k, n-1)
        # need sum_{i=i_min}^{i_max} f[i] * g[k-i] <= M
        expr = cp.sum(cp.multiply(f[i_min:i_max+1], g[k-i_max:k-i_min+1]))
        constraints.append(expr <= M)
    # objective
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=10000)
    except Exception:
        prob.solve(solver=cp.SCS, warm_start=True, verbose=False, max_iters=10000)
    if g.value is None:
        return None
    return np.array(g.value)

"""
Multi‑scale annealed Adam + projected sub‑gradient optimizer for the
self‑convolution L∞ problem.

The algorithm proceeds through a series of lengths (starting at a few
thousand entries).  For each length we generate several random starts,
run an Adam optimiser on a smooth log‑sum‑exp surrogate of the
infinite‑norm of the self‑convolution, and finish the run with a short
projected sub‑gradient descent that works on the exact max‑convolution.
The temperature of the soft‑max is increased aggressively so that the
surrogate converges to the true objective as the optimisation proceeds.
A final long sub‑gradient refinement is performed on the best global
solution before returning it.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    n = v.shape[0]
    # Fast projection based on sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # Active set of indices attaining (or almost attaining) the maximum.
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    # Gradient of a*a wrt a is 2 * (a_rev convolve w)
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

    init_logits: np.ndarray,

    max_iters: int = 8000,
    tau_start: float = 0.5,
    tau_end: float = 200.0,
    lr_start: float = 0.1,
    lr_end: float = 0.005,

    Adam optimisation on the soft‑max surrogate of the max‑convolution.

    Parameters
    ----------
    init_logits : np.ndarray
        Initial ``log``‑probabilities (any real numbers).

    lr_start / lr_end : float

    Returns
    -------
    best_a : np.ndarray
        Best probability vector seen (Σa = 1).
    best_raw : float
        Corresponding raw score 2·n·max(a*a).
    """
    x = init_logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_raw = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- soft‑max to enforce simplex -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

"""
Multi‑scale annealed Adam + projected sub‑gradient optimiser for the
self‑convolution L∞ problem.

The routine builds a non‑negative vector a (Σa = 1) that minimises

    2·n·max_k (a*a)_k ,

which after the evaluation wrapper equals the reported score.
The algorithm runs through several lengths, using many random restarts.
For each length it first runs Adam on a smooth log‑sum‑exp surrogate of
the max‑convolution, then refines the result with a projected sub‑gradient
descent that works on the exact max.  A final long refinement is performed
on the best global solution before returning the list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a * a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – small tolerance that scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    # gradient of a*a w.r.t. a is 2·(a_rev convolve w)
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals
