def _hillclimb(a, conv, max_iter, deadline):
    n = a.shape[0]
    M = conv.max()
    best_seq = a.copy()
    best_M = M
    it = 0
    no_improve = 0
    eps = 1e-15
    while time.time() < deadline and it < max_iter:
        it += 1
        # locate current max index
        k = int(conv.argmax())
        # Determine i range
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find pair (i,j) with largest product among active sum
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        source = best_i if a[best_i] >= a[best_j] else best_j
        # choose target not equal source
        target = np.random.randint(n)
        if target == source:
            continue
        # choose delta as a fraction of source
        max_delta = min(a[source] * 0.5, 0.05)  # larger possible movement
        delta = max_delta * np.random.rand()
        if delta < eps:
            continue
        # apply transfer
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < best_M - 1e-14:
            best_M = new_M
            best_seq = a.copy()
            no_improve = 0
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill it={it} M={best_M:.8f}")
        else:
            # revert
            # need to revert the move: compute reverse transfer (delta) from target to source
            # but easier: revert by swapping source and target and applying same delta
            _apply_transfer(a, conv, target, source, delta)  # reverse
            no_improve += 1
        # jitter occasionally
        if it % 2000 == 0 and no_improve > 500:
            jitter = np.random.randn(n) * 1e-5
            a[:] = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv[:] = _fft_self_convolve(a)
            best_M = conv.max()
            best_seq = a.copy()
            no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at {it}")
    return best_seq, best_M

def _optimize(a, deadline):
    # a initial vector with sum=1
    n = a.shape[0]
    # combo of ADAM
    a = a.copy()
    best_seq = a.copy()
    best_score = _max_convolution_score(a)
    # ADAM schedule
    schedule = [(0.5, 2000), (1.0, 2000), (5.0, 3000), (20.0, 4000), (100.0, 4000), (500.0, 4000), (2000.0, 2000)]
    # hyper
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.15
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq
            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w /= w.sum()
            grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            if t % 500 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam step {t} new best {cur:.6f}")
    # Subgradient refinement
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3
    it = 0
    while time.time() < deadline - 5.0:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)
        w = np.zeros_like(b)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        velocity = beta * velocity + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad it {it} new best {cur:.6f}")
    # Hill climb
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    it_no_improve = 0
    while time.time() < deadline - 1.0:
        it += 1
        # similar as previous
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta < 1e-15:
            continue
        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            it_no_improve = 0
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill it={it} new best score {best_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1
        if it % 2000 == 0 and it_no_improve > 500:
            jitter = np.random.randn(n) * 5e-6
            a += jitter

            best_score = 2.0 * n * M
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter at hill iter {it}")
    return best_seq

def _apply_transfer(a, conv, i, j, delta):
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part
    conv[i:i+n] -= 2.0 * delta * a_old
    conv[j:j+n] += 2.0 * delta * a_old
    # quadratic
    conv[2*i] += delta * delta
    conv[2*j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta
    # update a
    a[i] -= delta
    a[j] += delta

"""
A robust search for a low‑value step‑function.

The algorithm works in three phases:

1. **Broad scan** – a few analytic families (uniform, arcsine, beta,
   exponential, cosine‑windows, Dirichlet random draws, sparse starts)
   are generated for several lengths.  The best candidate (lowest
   ``2·n·max(conv)``) is retained.

2. **Heavy optimisation** – the chosen vector is refined by
   * Adam on a soft‑max approximation of the max‑convolution,
   * an exact sub‑gradient descent on the true max,
   * a targeted hill‑climber that updates the full self‑convolution
     incrementally (correct quadratic term!).  The whole refinement
     respects the 1000 s time budget.

3. **Return** – the final vector is normalised to sum = 1 and converted
   to a plain Python list.

All updates keep the vector on the probability simplex, use only
numpy/fft (fast O(n log n) convolutions) and stay well within the
time limit.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer `delta` from index i to index j, updating both `a`
    and its full self‑convolution `conv` in O(n) time.
    The quadratic term at i+j is also updated (previous version
    missed this).
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()          # keep the *old* values for the linear term
    # linear influence on all shifts
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic term
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta
    # actual weights
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _adam_optimize(a, deadline):
    """
    Adam optimiser on the smooth soft‑max approximation of the max‑convolution.
    Returns the improved vector and its (fast) score.
    """
    n = a.shape[0]
    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.13
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best = a.copy()
    best_score = _max_convolution_score(a)

    # gradually tighten the soft‑max (large α ⇒ closer to true max)
    schedule = [
        (0.5, 2000), (1.0, 2000), (5.0, 4000),
        (20.0, 6000), (100.0, 6000), (500.0, 6000),
        (2000.0, 4000), (5000.0, 2000)
    ]

    eps = 1e-15
    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best, best_score
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w /= w.sum()

            grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if t % 500 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-12:
                    best_score = cur
                    best = a.copy()
                    print(f"[{time.time():.1f}] Adam step {t} new best {cur:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
def _subgradient_refine(a, deadline):
    """
    Exact max‑sub‑gradient descent (similar to reference implementation).
    Returns an improved vector and its score.
    """
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    best = a.copy()
    best_score = _max_convolution_score(a)

    eps = 1e-15
    while time.time() < deadline - 0.5:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()
            print(f"[{time.time():.1f}] Subgrad it {it} new best {cur:.6f}")
    return best, best_score


# ----------------------------------------------------------------------
def _hillclimb(a, deadline):
    """
    Targeted hill‑climber that works directly on the self‑convolution.
    Returns the best vector found and its score.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    best = a.copy()
    best_score = 2.0 * n * M

    eps = 1e-15
    it = 0
    it_no_improve = 0
    while time.time() < deadline - 0.2:
        it += 1
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) with the largest product a_i*a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue

        # propose a modest transfer
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta < eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best = a.copy()
            best_score = 2.0 * n * M
            it_no_improve = 0
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill it {it} new best {best_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        # occasional jitter to escape plateaux
        if it % 2000 == 0 and it_no_improve > 500:
            jitter = np.random.randn(n) * 5e-6
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            best = a.copy()
            best_score = 2.0 * n * M
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at hill iter {it}")

    return best, best_score


# ----------------------------------------------------------------------
def _optimize(a, deadline):
    """
    Run the three refinement phases (Adam → sub‑gradient → hill‑climb)
    on a single start vector `a`.  Returns the best vector discovered.
    """
    # 1) Adam (soft‑max)
    a, _ = _adam_optimize(a, deadline)
    if time.time() >= deadline:
        return a

    # 2) Exact sub‑gradient descent
    a, _ = _subgradient_refine(a, deadline)
    if time.time() >= deadline:
        return a

    # 3) Hill‑climbing with exact convolution updates
    a, _ = _hillclimb(a, deadline)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects the 1000 s budget, tries several lengths,
    seeds from analytic families, then performs a heavy optimisation
    (Adam + sub‑gradient + hill‑climb) on the best candidate.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # tiny safety margin
    eps = 1e-15

    # ------------------------------------------------------------------
    # 1. Broad scan over a handful of lengths and analytic families
    lengths = [1500, 2500, 4000, 6000, 8000, 12000]
    best_seq = None
    best_score = float('inf')
    best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.001),
            _distribution_exponential(n, lam=0.01),
        ]

        # beta‑shaped families
        for p in (0.3, 0.5, 0.8, 1.2, 2.0):
            candidates.append(_distribution_beta_sym(n, p))

        # sine‑power windows
        for pw in (0.5, 1.0, 2.0, 4.0, 6.0):
            candidates.append(_distribution_sine_power(n, pw))

        # cosine windows (Hann‑like)
        for c in (0.5, 0.46, 0.42):
            candidates.append(_distribution_cosine_window(n, c))

        # few Dirichlet draws (different concentration)
        for conc in (0.3, 0.7, 1.5):
            x = np.random.gamma(conc, 1.0, size=n)
            candidates.append(x / x.sum())

        # random sparse start
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 250)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")
                if sc <= 1.503:          # early stop if we already beat the target
                    break

    # ------------------------------------------------------------------
    # 2. Include any previously discovered global best (if available)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.shape[0]
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # fallback – should never happen
    if best_seq is None:
        best_seq = _distribution_uniform(4000)
        best_n = 4000
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform n={best_n} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Heavy optimisation on the best candidate found so far
    print(f"[{time.time():.1f}] start heavy optimisation (n={best_n})")
    best_seq = _optimize(best_seq, deadline)

    # final normalisation and reporting
    best_seq = best_seq / best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] final best score ≈ {final_score:.6f} (length={best_seq.size})")
    return best_seq.tolist()
