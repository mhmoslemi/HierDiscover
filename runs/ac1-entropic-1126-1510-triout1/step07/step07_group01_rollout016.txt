"""
Propose a step‑function (non‑negative vector that sums to 1) that minimises
the evaluation function:

    2 * n * max(conv(a, a))

The algorithm consists of three phases:

1. **Rich candidate generation** – a wide collection of analytic families
   (arcsine, beta (symmetric & asymmetric), power‑law, exponential,
   triangular, sine, cosine, uniform) and several random Dirichlet draws.
   For each length we sweep a few shape parameters, normalise the vector
   and keep the cheap score 2·n·max(a*a).

2. **Refinement** – the best few candidates are optimised with a smooth
   surrogate of the max‑operator.  We use a soft‑max (log‑sum‑exp) of the
   convolution values, obtain its gradient via FFT‐based convolutions and
   run an Adam optimiser.  The temperature (β) of the soft‑max is increased
   gradually, turning the surrogate into the true max.

3. **Final polishing** – a long plain sub‑gradient descent (the same as in
   the previous state‑of‑the‑art) followed by tiny random mass‑transfer
   hill‑climbing.

All updates keep the vector on the probability simplex, so the scale‑invariant
objective is unaffected.  The routine respects a 1000 s budget and always
returns a feasible sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a) (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
#   Analytic families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(x, p - 1) * np.power(1.0 - x, p - 1)
    w[np.isnan(w)] = 0.0
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) – possibly skewed."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(x, p - 1) * np.power(1.0 - x, q - 1)
    w[np.isnan(w)] = 0.0
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _generalized_arcsine_distribution(n: int, alpha: float, beta: float, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+α)*(n‑1‑i+β))^{-γ}."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + alpha) * (n - 1 - i + beta)
    denom = np.maximum(denom, 1e-12)
    vals = denom ** (-gamma)
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= s
    return vals


def _power_law_distribution(n: int, p: float, offset: float = 0.5) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+offset)^{-p}."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + offset) ** (-p)
    s = vals.sum()
    if s == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= s
    return vals


def _random_dirichlet(n: int) -> np.ndarray:
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _random_dirichlet_sorted(n: int) -> np.ndarray:
    """Sorted Dirichlet (monotonically decreasing)."""
    x = _random_dirichlet(n)
    return np.sort(x)[::-1]


# ----------------------------------------------------------------------
#   Gradient of the smooth (log‑sum‑exp) surrogate of max(conv)
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, beta: float):
    """
    Returns (grad, smooth_value) where:
        smooth_value = (1/beta) * log Σ exp(beta * (a*a))
    and grad = ∂smooth_value/∂a .
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)            # length 2n‑1
    # soft‑max weights (numerically stable)
    b_max = b.max()
    w = np.exp(beta * (b - b_max))
    w_sum = w.sum()
    w /= w_sum                           # now Σ w = 1

    # gradient of Σ w_i * b_i  (no factor 1/β because it cancels)
    L = b.size                           # = 2n‑1
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[L - 1 - np.arange(n)]   # shape (n,)

    # smooth objective value (optional, not used for step size)
    smooth_val = (np.log(w_sum) + b_max) / beta
    return grad, smooth_val


# ----------------------------------------------------------------------
#   Optimisers
# ----------------------------------------------------------------------
def _adam_subgrad(a0: np.ndarray, deadline: float,
                  max_iters: int = 500_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Classic Adam on the true max‑convolution (sub‑gradient)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # sub‑gradient of max(conv)
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best = cur, a.copy()
    return best, best_score


def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 300_000,
                         init_beta: float = 1.0,
                         max_beta: float = 30.0,
                         beta_inc_iters: int = 30_000,
                         step0: float = 0.2,
                         beta1: float = 0.9, beta2: float = 0.999):
    """
    Adam optimiser on the smooth surrogate.
    The temperature β is increased every `beta_inc_iters` steps.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    beta = init_beta
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        if it % beta_inc_iters == 0 and beta < max_beta:
            beta = min(beta * 1.5, max_beta)

        grad, _ = _smooth_gradient(a, beta)   # gradient of smooth loss
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - step * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best = cur, a.copy()
    return best, best_score


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient before diagonal correction
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for positions where k = 2*i
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector) that
    minimises `evaluate_sequence`.  The function respects the given time
    budget and returns a Python list ready for evaluation.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine a base length (use previously discovered length if possible)
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (balanced small/medium/large)
    lengths = set()
    for f in (0.9, 1.0, 1.1, 1.2, 1.4, 1.6):
        n = int(base_n * f)
        if 500 <= n <= 8000:
            lengths.add(n)
    # add a few random lengths
    for _ in range(6):
        n = np.random.randint(800, min(8000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Helper to store a candidate
    candidates = []   # (array, cheap_score, description)

    def _store(arr: np.ndarray, descr: str):
        if arr.ndim != 1:
            return
        arr = np.maximum(arr, 0.0)
        if arr.sum() == 0:
            return
        arr = arr / arr.sum()
        score = _max_convolution_score(arr)
        candidates.append((arr, score, descr))
        print(f"[{time.time():.1f}] n={len(arr):4d} {descr:30s} score={score:.6f}")

    # ------------------------------------------------------------------
    # 1. Broad sampling of analytic families & random draws
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families
        _store(_arcsine_distribution(n), "arcsine")
        _store(_triangular_distribution(n), "triangular")
        _store(_uniform_distribution(n), "uniform")
        _store(_sine_distribution(n), "sine")
        _store(_cosine_distribution(n), "cosine")
        for lam in (0.001, 0.003, 0.005, 0.01):
            _store(_exponential_distribution(n, lam), f"exponential(lam={lam:.3f})")

        # symmetric beta family (already used in the previous code)
        beta_ps = [0.2, 0.4, 0.5, 0.6, 0.8, 1.0, 1.25, 1.5, 2.0, 3.0]
        for p in beta_ps:
            _store(_beta_symmetric_distribution(n, p), f"beta_sym(p={p:.2f})")

        # asymmetric beta (skewed)
        asym_vals = [0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]
        for p in asym_vals:
            for q in asym_vals:
                _store(_beta_asymmetric_distribution(n, p, q),
                       f"beta_asym(p={p:.2f},q={q:.2f})")
                # limit number of combos per length
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

        # generalized arcsine family
        alphas = [0.0, 0.5, 1.0, 2.0]
        betas = [0.0, 0.5, 1.0, 2.0]
        gammas = [0.4, 0.5, 0.6, 0.8, 1.0]
        for a0 in alphas:
            for b0 in betas:
                for g0 in gammas:
                    _store(_generalized_arcsine_distribution(n, a0, b0, g0),
                           f"gen_arc(a={a0:.1f},b={b0:.1f},g={g0:.1f})")
                    if time.time() >= deadline:
                        break
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break

        # power‑law decreasing
        for p_exp in [0.5, 0.8, 1.0, 1.3, 1.6, 2.0]:
            _store(_power_law_distribution(n, p_exp), f"powerlaw(p={p_exp:.1f})")

        # random Dirichlet draws (unsorted & sorted)
        for _ in range(3):
            _store(_random_dirichlet(n), "rand_dirichlet")
            _store(_random_dirichlet_sorted(n), "rand_dirichlet_sorted")
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # 2. Include previously discovered best sequence, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            _store(prev, "previous_best")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the top‑k seeds for heavy refinement
    if not candidates:
        # As a last resort, fall back to a simple uniform vector.
        best_seq = np.full(base_n, 1.0 / base_n)
        return best_seq.tolist()

    candidates.sort(key=lambda x: x[1])
    top_k = 15
    top_candidates = candidates[:top_k]
    print(f"=== Selected {len(top_candidates)} seeds for refinement "
          f"(best cheap score={top_candidates[0][1]:.6f}) ===")

    # ------------------------------------------------------------------
    # 4. Refinement of the seeds – smooth Adam followed by classic Adam
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, _, descr) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Refining seed {idx+1}/{len(top_candidates)}: {descr}")

        # smooth Adam (temperature schedule)
        seq, sc = _smooth_adam_descent(
            seed_seq,
            deadline,
            max_iters=300_000,
            init_beta=1.0,
            max_beta=30.0,
            beta_inc_iters=25_000,
            step0=0.2)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"    >>> smooth‑Adam improvement: {sc:.6f}")

        # classic Adam on the true max‑objective (sub‑gradient inside)
        seq, sc = _adam_subgrad(seq, deadline, max_iters=250_000, step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"    >>> Adam‑subgrad improvement: {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient descent on the current best vector
    subgrad_deadline = deadline - 30.0   # reserve ~30 s for final hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _adam_subgrad(best_seq, subgrad_deadline, max_iters=800_000, step0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] long sub‑gradient descent improvement: {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained hill‑climbing (tiny random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.03  # start with up to 3 % of total mass per move

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 12_000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if iteration % 5_000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"=== Final cheap score: {best_score:.6f} (length={len(best_seq)}) ===")
    return best_seq.tolist()
