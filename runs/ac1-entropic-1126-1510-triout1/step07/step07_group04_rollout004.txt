"""
Propose a low‑score step function.

The search proceeds in three stages:

1️⃣  Random + grid exploration on modest lengths (2k‑8k).  Candidates are
    uniform, symmetric Beta, random Dirichlet draws and simple mixtures.
    The best shape (normalised to Σ=1) is kept.

2️⃣  Smooth mirror‑descent refinement on the current best length.
    Instead of a hard “active‑set” sub‑gradient we use a softmax
    (log‑sum‑exp) smoothing of the max‑convolution:
        w_i ∝ exp((a*a)_i / μ) .
    This yields a dense sub‑gradient that drives the iterates faster
    through plateaus.  An exponentiated‑gradient (KL‑mirror) update
    keeps the iterate on the simplex.  The temperature μ is annealed.

3️⃣  Up‑sample the refined shape to larger lengths (12k‑20k) and continue
    the same smooth mirror‑descent.  More degrees of freedom lower the
    limiting value 2·max(f∗f) and give a better score.

The whole routine respects the 1000 s budget, prints progress and
finally returns the best sequence found (as a plain Python list)."""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable version
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)       # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

def _self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_fft(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy score for a vector already normalised to Σ=1.
    Equals the true evaluation value (scale‑invariant)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

def _smooth_subgradient_step(a: np.ndarray, mu: float):
    """
    Returns a dense sub‑gradient of  f(a)=max_k (a*a)_k
    using a soft‑max smoothing with temperature mu.
    Also returns the current max entry (useful for logging).
    """
    n = a.shape[0]

    # ---- self‑convolution ----
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

    # ---- smooth weights (soft‑max) ----
    max_b = conv.max()
    # stability: subtract max before exponentiation
    exp_arg = (conv - max_b) / mu
    # clip to avoid overflow in exp
    np.clip(exp_arg, -700, 700, out=exp_arg)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ---- gradient = 2 * (rev(a) * w) ----
    a_rev = a[::-1]
    size2 = 1 << ((len(w) + n - 1 - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=size2)
    W = np.fft.rfft(w, n=size2)
    conv_rev_full = np.fft.irfft(A_rev * W, n=size2)[: len(w) + n - 1]
    # central n entries correspond to indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_rev_full[n - 1 : n - 1 + n]

    return grad, max_b


# ----------------------------------------------------------------------
def _mirror_descent_smooth(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    eta0: float = 0.6,
    mu0: float = 0.02,
    mu_decay: float = 0.9999,
) -> tuple[np.ndarray, float]:
    """
    Smooth mirror‑descent (exponentiated‑gradient) on the simplex.
    Returns the best iterate and its raw score discovered before `deadline`.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)

    it = 0
    mu = mu0

    while time.time() < deadline:
        it += 1

        grad, cur_max = _smooth_subgradient_step(a, mu)

        eta = eta0 / math.sqrt(it)

        # multiplicative (KL) update; clamp exponent to avoid overflow
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)

        # project back onto the simplex
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny jitter to escape flat regions
        if (it % 1000 == 0) and (rng.random() < 0.05):
            a += rng.normal(scale=1e-6, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(
                f"[{time.time():.1f}] smooth MD len={n} it={it} "
                f"score={best_val:.6f} max_self={cur_max:.6e}"
            )

        # anneal temperature
        mu = max(mu * mu_decay, 1e-7)

    return best_a, best_val

    best_seq = None
    best_score = np.inf

    def _maybe_update(seq: np.ndarray):
        """Update global best if the candidate improves the raw score."""
        nonlocal best_seq, best_score
        val = _raw_score(seq)
        if val < best_score - 1e-12:
            best_seq = seq.copy()
            best_score = val
            print(f"[{time.time():.1f}] new best raw score={best_score:.6f} len={len(seq)}")

    # ------------------------------------------------------------------
    # Phase 1 – coarse random / grid exploration on modest lengths
    lengths_phase1 = [2000, 4000, 8000]
    for n in lengths_phase1:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # start from current global best (interpolated) if we already have one
        if best_seq is not None:
            cand = _interpolate_to_length(best_seq, n)
            _maybe_update(cand)

        # deterministic baselines
        _maybe_update(np.full(n, 1.0 / n, dtype=np.float64))                # uniform
        _maybe_update(_beta_distribution(n, 0.5, 0.5))                     # arcsine

        # symmetric Beta grid (coarse)
        alphas = np.concatenate((
            np.linspace(0.12, 0.6, 9, endpoint=False),
            np.linspace(0.6, 2.0, 12),
            np.linspace(2.0, 5.0, 7)
        ))
        for a_alpha in alphas:
            if time.time() >= deadline:
                break
            _maybe_update(_beta_distribution(n, a_alpha, a_alpha))

        # random Dirichlet draws
        for _ in range(200):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-2, 1)
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            _maybe_update(vec)

        # mixture of two Betas (simple but expressive)
        for _ in range(150):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1 - lam) * p2
            vec = _project_onto_simplex(vec)
            _maybe_update(vec)

    # ------------------------------------------------------------------
    # If nothing was discovered (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)
        best_score = _raw_score(best_seq)

    # ------------------------------------------------------------------
    # Phase 2 – smooth mirror‑descent refinement on the current best length
    if time.time() < deadline:
        print(f"[{time.time():.1f}] start smooth mirror‑descent on len={len(best_seq)}")
        refined, scr = _mirror_descent_smooth(best_seq.copy(), deadline, rng,
                                              eta0=0.6, mu0=0.02, mu_decay=0.9999)
        _maybe_update(refined)
        best_seq = refined

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to larger lengths and continue refinement
    larger_lengths = [12000, 16000, 20000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)   # bullet‑proof
        a, scr = _mirror_descent_smooth(a, deadline, rng,
                                        eta0=0.55, mu0=0.018, mu_decay=0.9999)
        _maybe_update(a)
        best_seq = a
        if best_score <= 1.503:          # early exit if target already met
            print("[early exit] target bound reached")
            break

    # ------------------------------------------------------------------
    # Final exact evaluation (the evaluator also clips & rescales)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} "
          f"(len={len(best_seq)})")
    return best_seq.tolist()
