"""
Improved step‑function search for the “max‑convolution” score.

The algorithm works in three stages:

1. **Warm‑start** – if a previous solution (height_sequence_1) exists it is
   normalised and used as the initial shape.
2. **Length cascade** – we start with a modest length (4000) and repeatedly
   double the size while interpolating the current best shape onto the new
   grid.  On each length we run a long Adam optimisation on a smooth
   approximation of the max‑convolution (log‑sum‑exp) with an exponential
   temperature schedule, followed by a short exact sub‑gradient refinement.
3. **Final polishing** – a longer projected sub‑gradient descent finishes the
   search.

All optimisation stays on the probability simplex (Σaᵢ = 1, aᵢ ≥ 0) so that the
raw value 2·n·max(a∗a) equals the value returned by ``evaluate_sequence`` after
the final normalisation.

The code respects the global time budget (default 1000 s) and prints progress.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 15000,
                            tau0: float = 0.5,
                            tau_max: float = 300.0,
                            lr0: float = 0.15) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(a*a).

    Returns the best normalised sequence (a) seen and its *raw* score
    2·n·max(a*a).  The temperature τ grows exponentially from tau0 to tau_max.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_raw = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # soft‑max to obtain a simplex point
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # exact max convolution (for reporting & best‑track)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_raw:
            best_raw = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_raw:.6f}")

        # temperature schedule – exponential increase
        tau = tau0 * (tau_max / tau0) ** ((it - 1) / max_iters)

        # smooth‑max weights (stable log‑sum‑exp)
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # gradient of smooth max w.r.t a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            x += np.random.randn(*x.shape) * 0.005

    # fallback – uniform distribution
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_raw


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0,
                           max_iters: int = 1_200_000) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact max‑convolution.
    Stops when the time deadline is reached.
    Returns refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(a)
    it = start_iter

    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient of max_k (a*a)_k
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(a_rev, w)[n - 1: n - 1 + n]

        # diminishing step size
        step = 0.15 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)

        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_raw:.6f}")

        # occasional tiny random jitter
        if it % 1000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 4e-7
            a = _project_onto_simplex(a)

    return best_a, best_raw

        # ------------------------------------------------------------------
        # run optimisation from each start, keep the local best
        local_best_seq = None
        local_best_raw = float("inf")

        for idx, logits in enumerate(init_logits):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] start {idx+1}/{len(init_logits)} for n={n}")

            # 1) Adam on the smooth surrogate
            seq, raw_score = _adam_optimize_smoothed(
                logits, n, deadline,
                max_iters=18000,
                tau0=0.5, tau_max=300.0, lr0=0.15
            )

            # 2) short exact sub‑gradient polish
            seq, raw_score = _refine_by_subgradient(seq, deadline)

            if raw_score < local_best_raw:
                local_best_raw = raw_score
                local_best_seq = seq.copy()
                print(f"   *** new local best for n={n}: raw={local_best_raw:.6f}")

        if local_best_seq is not None and local_best_raw < best_raw:
            best_raw = local_best_raw
            best_seq = local_best_seq.copy()
            best_len = n
            print(f">>> Updated global best: raw={best_raw:.6f} (length={best_len})")

    # ------------------------------------------------------------------
    # Final long‑run sub‑gradient refinement on the global best
    if best_seq is not None:
        print("\nFinal long‑run refinement")
        best_seq, best_raw = _refine_by_subgradient(best_seq, deadline)
    else:
        # fallback – uniform distribution of moderate size
        n = 5000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_seq)

    # ------------------------------------------------------------------
    # tiny pruning: drop values that are negligible (helps numerical stability)
    eps = 1e-14
    best_seq[best_seq < eps] = 0.0
    s = best_seq.sum()
    if s > 0.0:
        best_seq /= s
    else:
        best_seq[:] = 1.0 / best_seq.size

    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
