```python
"""
Multi-start Adam‑based smooth‑max optimisation for the step‑function
evaluation problem.

- Build a diverse pool of probability vectors (uniform, arcsine,
  Beta‑families, random Dirichlet draws, sparse uniform spikes).
- Keep the best few seeds w.r.t. the cheap proxy 2·n·max(a*a).
- Refine each seed with a continuation scheme:
  minimise the smooth approximation
        fµ(a) = µ·log Σ exp((a*a)_k / µ)
  using the Adam optimiser and Euclidean projection onto the simplex.
  µ is reduced geometrically until a tiny value (~1e‑8) – an accurate
  proxy for the true max‑convolution.
- Finish with a short projected sub‑gradient descent to polish the
  last digits.
- Return the best sequence (the harness rescales it automatically).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _real_fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    return _real_fft_convolve(a, a)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Convenient proxy for the evaluation (sum a = 1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)               # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)                # all ≤ 1, safe
    Z = exp_scaled.sum()
    p = exp_scaled / Z                         # soft‑max weights

    # gradient = 2 * (a_rev * p)   – central n entries are needed
    a_rev = a[::-1]
    tmp = _real_fft_convolve(a_rev, p)         # length 3·n‑2
    grad = 2.0 * tmp[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑pdf to avoid overflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # clip extreme values
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _random_sparse_vector(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k uniformly random positions each receiving weight 1/k."""
    vec = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    vec[idx] = 1.0 / k
    return vec


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_raw = _raw_score(a)

def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:

"""
Multi‑start Adam‑based smooth‑max optimisation for the step‑function problem.

- Build a varied pool of probability vectors (uniform, arcsine, several Beta
  families, random Dirichlet draws, sparse spike patterns and the globally
  supplied warm‑start if available).
- Keep the few best seeds w.r.t. the cheap proxy 2·n·max(a*a).
- Refine each of them with a continuation scheme that minimises the smooth
  surrogate  f_μ(a)= μ·log Σ exp((a*a)_k/μ) using the Adam optimiser and an
  Euclidean projection onto the simplex.  μ is reduced geometrically until a
  tiny value (~1e‑8) – a very accurate proxy for the true max‑convolution.
- From the best refined seed run a short projected sub‑gradient descent to
  polish the last digits.
- Return the final sequence (the harness rescales it automatically).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _real_fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using an FFT of power‑of‑2 length."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1)."""
    return _real_fft_convolve(a, a)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector (Σa=1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)               # length 2·n‑1
    M = conv.max()
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)                # safe because all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                         # soft‑max weights, Σp=1

    a_rev = a[::-1]
    tmp = _real_fft_convolve(a_rev, p)         # length 3·n‑2
    grad = 2.0 * tmp[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)  # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _random_sparse_vector(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform weight 1/k on k random positions."""
    vec = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    vec[idx] = 1.0 / k
    return vec
