"""
Optimisation of a step‑function (non‑negative sequence) for the
evaluate_sequence() objective.

The algorithm proceeds in three phases:

1. **Broad random search** – generate a large pool of analytically‑defined
   probability vectors (uniform, Beta‑distributions – symmetric and
   asymmetric, Dirichlet draws, power‑law, exponential, and simple mixtures)
   and keep the best seed w.r.t. the cheap proxy
   ``raw_score = 2·n·max(a∗a)``.

2. **Continuation + projected FISTA** – smooth the max‑convolution by a
   log‑sum‑exp (parameter μ) and minimise the smooth surrogate on the
   simplex using an accelerated proximal gradient method.  μ is reduced
   geometrically (continuation) until it reaches 1e‑8, which approximates
   the true max.

3. **Polishing** – a short projected sub‑gradient descent is applied,
   followed by a fast random‑permutation hill‑climb (re‑ordering the
   entries can lower the convolution peak).  The best sequence found
   before the budget expires is returned; ``evaluate_sequence`` will
   automatically rescale it.

All helper functions are top‑level and use only NumPy.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log of beta normalisation constant
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p, normalised to Σ=1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** p
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals[:] = 1.0 / n
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(lam * (i - (n-1)/2) / n), normalised."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    vals = np.exp(lam * i / n)
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals[:] = 1.0 / n
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution via log‑sum‑exp and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)                # length 2*n-1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)                     # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                              # soft‑max weights, Σp=1

    # gradient: 2 * (a_rev * p)  (central n entries)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smoothed objective value (M + μ·log Z)
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Refine a probability vector with continuation + projected FISTA.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    mu = max(_max_self_convolution(a) * 0.5, 1e-7)
    mu_target = 1e-8          # essentially the true max
    mu_factor = 0.6
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # ---- inner FISTA for current μ ---------------------------------
        L = max(1.0, 2.0 / mu)          # simple Lipschitz estimate
        y = a.copy()
        t = 1.0

        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        # each iteration costs ~1 ms; be generous with the bound
        iters = int(min(4000, remaining / 0.0012))
        if iters <= 0:
            break

        for _ in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # back‑tracking line search (max two doublings)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional true‑objective check
            if (_raw_score(a) < best_val):
                best_val = _raw_score(a)
                best_seq = a.copy()
        # ----------------------------------------------------------------
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        a = best_seq.copy()          # warm‑restart

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish the solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=len(a))
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray, rng: np.random.Generator, time_budget: float) -> tuple:
    """
    Very cheap random‑pair hill‑climb (move a tiny amount of mass between
    two coordinates).  Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    n = len(a)
    it = 0
    while time.time() - start < time_budget:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 1e-12:
            continue
        # transfer a small relative fraction of a[i] to a[j]
        delta = a[i] * rng.uniform(0.001, 0.01)
        a[i] -= delta
        a[j] += delta
        cur_val = _raw_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point – searches for a non‑negative sequence that minimises
    ``evaluate_sequence``.  The method combines a large random seed pool,
    a continuation‑FISTA refinement, a sub‑gradient polish and a fast
    permutation hill‑climb.  The best sequence found before the time limit
    is returned (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1️⃣  Build a diverse pool of probability vectors (all Σ=1)
    n = 8000                                   # final length
    candidates = []

    # a) warm‑start from the provided global if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        else:
            base /= base.sum()
        if base.shape[0] != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, base.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            base = np.interp(x_new, x_old, base)
            base = np.maximum(base, 0.0)
            base /= base.sum()
        candidates.append(base)
    except Exception:
        pass

    # b) deterministic baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))                # arcsine

    # c) symmetric Beta shapes (log‑spaced)
    for a in np.geomspace(0.05, 10.0, num=12):
        candidates.append(_beta_distribution(n, a, a))

    # d) random asymmetric Beta draws
    for _ in range(200):
        alpha = 10 ** rng.uniform(-2, 1)   # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(_beta_distribution(n, alpha, beta))

    # e) Dirichlet draws with varying concentration
    for _ in range(80):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec_sum = vec.sum()
        if vec_sum == 0.0:
            vec[:] = 1.0 / n
        else:
            vec /= vec_sum
        candidates.append(vec)

    # f) power‑law families
    for p in np.linspace(-2.0, 2.0, num=15):
        candidates.append(_power_law_distribution(n, p))

    # g) exponential families
    for lam in np.linspace(-5.0, 5.0, num=15):
        candidates.append(_exponential_distribution(n, lam))

    # h) simple mixtures of two Betas
    for _ in range(60):
        a1 = 10 ** rng.uniform(-2, 1)
        b1 = 10 ** rng.uniform(-2, 1)
        a2 = 10 ** rng.uniform(-2, 1)
        b2 = 10 ** rng.uniform(-2, 1)
        beta1 = _beta_distribution(n, a1, b1)
        beta2 = _beta_distribution(n, a2, b2)
        w = rng.random()
        seq = w * beta1 + (1.0 - w) * beta2
        seq /= seq.sum()
        candidates.append(seq)

    # --------------------------------------------------------------
    # 2️⃣  Pick the best seed according to the cheap proxy
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    print(f"Initial best raw score: {best_raw:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Smoothed‑gradient refinement (continuation + FISTA)
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate the bulk of the remaining budget to FISTA
        refine_budget = remaining * 0.7
        best_seq, best_raw = _fista_refine(best_seq, refine_budget)

    # --------------------------------------------------------------
    # 4️⃣  Short sub‑gradient polishing
    remaining = deadline - time.time()
    if remaining > 5.0:
        polish_budget = remaining * 0.5
        best_seq, best_raw = _subgradient_polish(best_seq, polish_budget)

    # --------------------------------------------------------------
    # 5️⃣  Random‑permutation hill‑climb (fast)
    remaining = deadline - time.time()
    if remaining > 3.0:
        # try a handful of random re‑orderings
        best_perm_seq = best_seq
        best_perm_val = best_raw
        num_perm = int(min(200, remaining / 0.015))   # roughly 15 ms per trial
        for _ in range(num_perm):
            perm = rng.permutation(n)
            seqp = best_seq[perm]
            val = _raw_score(seqp)
            if val < best_perm_val:
                best_perm_val = val
                best_perm_seq = seqp.copy()
        best_seq = best_perm_seq
        best_raw = best_perm_val
        print(f"After permutation hill‑climb: raw score {best_raw:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Final tiny coordinate‑wise hill‑climb
    remaining = deadline - time.time()
    if remaining > 1.0:
        best_seq, best_raw = _hill_climb(best_seq, rng, remaining)

    # --------------------------------------------------------------
    # 7️⃣  Return the sequence (evaluate_sequence will rescale it)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
