"""
Advanced convex optimisation for step‑function minimisation.

We search for a non‑negative vector a on the simplex that minimises
    F(a) = 2·n·max_k (a*a)_k
where (a*a) denotes the linear convolution.
The core is a continuation method that minimises a smooth log‑sum‑exp
approximation of the max with Adam, starting from a diverse set of
initial guesses (uniform, Dirichlet, arcsine‑type power‑law shapes and a
warm‑start from the best sequence seen so far).  After the smoothed
optimisation we apply an exponentiated‑gradient sub‑gradient refinement
that directly targets the true max.

The routine respects a hard time‑budget (default 1000 s) and returns the
best sequence discovered.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    # Fast path for already feasible vectors
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b via FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    total = new_vals.sum()
    if total > 0.0:
        new_vals /= total
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1, alpha_high: float = 4.0) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    g_sum = g.sum()
    if g_sum == 0.0:
        g = np.ones_like(g)
        g_sum = g.shape[0]
    return g / g_sum

def _arcsine_start(n: int) -> np.ndarray:
    """Classical arcsine‑like shape (p = 0.5 power‑law)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total > 0.0:
        vals /= total
    else:
        vals[:] = 1.0 / n
    return vals

def _powerlaw_start(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ ((i+½)(n‑½‑i))^{‑p}."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    base = (i + 0.5) * (n - 0.5 - i) + eps
    vals = base ** (-p)
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total > 0.0:
        vals /= total
    else:
        vals[:] = 1.0 / n
    return vals

def _adam_optimize_smoothed(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smooth approximation of the max.
    Returns the best probability vector found and its raw (non‑smoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector on simplex
        a = _softmax(x)

        # raw objective (exact max)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.3f}")

        # smooth max via log‑sum‑exp
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()      # shape (2n‑1,)

        # gradient wrt a (restricted to first n entries)
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through soft‑max (logits)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase λ
        lam = lam * lam_factor

        # occasional jitter
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score

def _optimise_length(

    deadline: float,

    Returns the best normalised vector and its raw score.
    """
    init_logits = []

    # Warm‑start (interpolated)

    best_local_seq = None
    best_local_score = float("inf")

    # Determine remaining time per start (simple heuristic)
    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, start {idx+1}/{len(init_logits)}")
        # allocate up to 4000 iterations; shrink if time low
        remaining = deadline - time.time()
        # adapt max_iters proportionally (e.g., 4000 * remaining / (budget_left+small))
        # For simplicity, keep a fixed cap but break inside loop if deadline passes.
        seq, score = _adam_optimize_smoothed(
            x0,
            n,
            deadline,
            max_iters=5000,
            lr0=0.18,
            lam_start=0.5,
            lam_factor=1.01,
        )
        if score < best_local_score:
            best_local_score = score
            best_local_seq = seq.copy()

    if best_local_seq is None:
        best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_local_score = 2.0 * n * _fft_self_convolve_full(best_local_seq).max()

    return best_local_seq, best_local_score

def _refine_sequence_expg(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient refinement targeting the true max.
    Operates until the deadline and returns the best vector seen.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    conv = _fft_self_convolve_full(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        # active set (tolerance relative)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        # gradient of the max (subgradient)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Exponentiated‑gradient update
        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        # renormalise to simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= sum_a

        # occasional jitter to escape tiny plateaus
        if it % 800 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] refine it={it} raw={best_score:.6f}")

    return best_a, best_score

"""
Advanced convex optimisation for step‑function minimisation.

We search for a non‑negative vector a on the simplex that minimises
    F(a) = 2·n·max_k (a*a)_k
where (a*a) denotes the linear convolution.
The core is a continuation method that minimises a smooth log‑sum‑exp
approximation of the max with Adam, starting from a diverse set of
initial guesses (uniform, Dirichlet, arcsine‑type power‑law shapes and a
warm‑start from the best sequence seen so far).  After the smoothed
optimisation we apply an exponentiated‑gradient sub‑gradient refinement
that directly targets the true max.

The routine respects a hard time‑budget (default 1000 s) and returns the
best sequence discovered.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b via FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    total = new_vals.sum()
    if total > 0.0:
        new_vals /= total
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.0) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    s = g.sum()
    if s == 0.0:
        g = np.ones_like(g)
        s = g.shape[0]
    return g / s


def _arcsine_start(n: int) -> np.ndarray:
    """Classic arcsine‑like shape (p = 0.5 power‑law)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _powerlaw_start(n: int, p: float) -> np.ndarray:
    """Power‑law shape a_i ∝ ((i+½)(n‑½‑i))^{‑p}."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    base = (i + 0.5) * (n - 0.5 - i) + eps
    vals = base ** (-p)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth approximation of the max.
    Returns the best probability vector found and its exact (non‑smoothed)
    raw score 2·n·max_k (a*a)_k.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                     # Σa = 1, a ≥ 0
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.3f}")

        # smooth max (log‑sum‑exp)
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()

        # gradient w.r.t. a (restricted to n entries)
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through soft‑max (logits)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase λ
        lam = lam * lam_factor

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()
    return best_a, best_score

def _optimise_length(

def _refine_sequence_expg(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient polishing targeting the exact max.
    Runs until the deadline; returns the best sequence and its raw score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_full(a).max()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()

        # active set (tolerance to capture ties)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        # sub‑gradient of the max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # exponentiated‑gradient update
        eta = 0.5 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional tiny jitter
        if it % 800 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] refine it={it} raw={best_score:.6f}")

    return best_a, best_score
