def propose_candidate(...):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    best_seq = uniform at length 2000 maybe, best_score = rawscore.

    # 1. systematic search on symmetric Beta single component
    for n in [4000,8000,12000]:
        alphas = np.logspace(-3,0, num=150) # 0.001 to 1
        for α in alphas:
            if time.time()>deadline: break
            vec = _beta_distribution(n, α, α)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec, sc
                best_len=n
                print(f"SingleBeta n={n} α={α:.5f} score={sc:.6f}")

    # 2 random mixtures
    for n in [4000,8000,12000]:
        for _ in range(2000):
            if time.time()>deadline: break
            # Choose 2 components
            α1 = 10**rng.uniform(-3,0)
            α2 = 10**rng.uniform(-3,0)
            w = rng.random()
            p1 = _beta_distribution(n, α1, α1)
            p2 = _beta_distribution(n, α2, α2)
            vec = w * p1 + (1-w) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec, sc
                best_len=n
                print(f"Mix2 n={n} α1={α1:.5g} α2={α2:.5g} w={w:.3f} score={sc:.6f}")

    # 3 Mirror descent refinement on best_seq length
    best_seq = best_seq.copy()
    best_seq,_ = _refine_mirror_descent(best_seq, deadline, rng, eta0=0.35)

    # 4 Up-sample to larger lengths 20000, 25000
    for n in [20000,25000,30000]:
        if time.time()>deadline: break
        a = _interpolate_to_length(best_seq, n)
        a,_ = _refine_mirror_descent(a, deadline, rng, eta0=0.35)
        sc = _raw_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            best_len = n
            print(f"Upsample n={n} new best score={sc:.6f}")

"""
Propose a low‑score step function.

Algorithm
---------
1.  Load a warm‑start (if the global `height_sequence_1` exists); otherwise start
    from a uniform vector.
2.  Perform a *fast parametric search*:
    * exhaustive sweep of symmetric Beta(α,α) for 0.001 ≤ α ≤ 1 on several lengths,
      keeping the best.
    * random mixtures of two symmetric Betas (weights + α‑parameters drawn
      from wide log‑uniform ranges) – each candidate is projected onto the
      simplex.
3.  Refine the current best vector with a *smooth mirror‑descent*:
    we minimise the smoothed objective  
        φγ(a)= (1/γ)·log ∑ₖ exp(γ·(a*a)ₖ) .
    Its gradient is a weighted sum of the sub‑gradients of the individual
    convolution entries, which we compute by a single FFT‑based convolution.
    The update is exponentiated (multiplicative) and the iterate is projected
    back onto the simplex.  The smoothing parameter γ is increased slowly,
    the stepsize η decays as 1/√t, and occasional jitter helps escape plateaux.
4.  Upsample the refined sequence to larger lengths (12 k, 16 k, 20 k, 30 k)
    and run the same refinement again – larger n reduces the theoretical lower
    bound (≈ 1) and often yields a smaller score.
5.  Return the best sequence (as a plain Python list).  The supplied
    `evaluate_sequence` rescales it, so we do not need to normalise the final
    output.

The whole routine respects the 1000 s budget; all heavy work is done with
FFT‑based convolutions (O(n log n)) and pure NumPy, no external solvers.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) (mid‑point rule) of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable version
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # stable normalisation (log‑sum‑exp)
    shift = np.max(logpdf)
    logpdf -= shift
    pdf = np.exp(logpdf)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing simplex‑vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The value equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

def _smooth_subgradient(a: np.ndarray, gamma: float):
    """
    Gradient of the smooth approximation
        φγ(a) = (1/γ)·log Σₖ exp(γ·(a*a)ₖ).
    Returns (grad, current_max_of_convolution).
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    bmax = b.max()
    # softmax weights (numerically stable)
    w = np.exp(gamma * (b - bmax))
    w_sum = w.sum()
    if w_sum == 0.0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, p)[n - 1 : n - 1 + n]
    return grad, bmax


def _refine_smooth_mirror_descent(a: np.ndarray,
                                  deadline: float,

    """
    Smoothed mirror‑descent (exponentiated‑gradient) on the simplex.
    Returns the best vector seen and its raw score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        gamma = gamma0 + gamma_inc * it
        grad, cur_max = _smooth_subgradient(a, gamma)
        eta = eta0 / math.sqrt(it)

        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)   # avoid overflow

    return best_a, best_val

            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   new best single‑Beta α={α:.5f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random mixtures of two symmetric Betas
    for n in [4000, 8000, 12000]:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] random 2‑Beta mixtures length {n}")
        for _ in range(2500):
            if time.time() >= deadline:
                break
            α1 = 10 ** rng.uniform(-3, 0)   # 0.001 … 1
            α2 = 10 ** rng.uniform(-3, 0)
            w = rng.random()
            p1 = _beta_distribution(n, α1, α1)
            p2 = _beta_distribution(n, α2, α2)
            vec = w * p1 + (1.0 - w) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   new best mix2 α1={α1:.4g} α2={α2:.4g} w={w:.3f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – smooth mirror‑descent refinement on current best length
    if time.time() < deadline:
        print(f"[{time.time():.1f}] starting smooth refinement (len={best_len})")
        best_seq, best_score = _refine_smooth_mirror_descent(best_seq,
                                                             deadline,
                                                             rng,
                                                             eta0=0.38,
                                                             gamma0=5.0,
                                                             gamma_inc=0.001)

    # ------------------------------------------------------------------
    # Phase 4 – up‑sampling to larger lengths and re‑refining
    larger_lengths = [16000, 20000, 25000, 30000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)
        a, scr = _refine_smooth_mirror_descent(a, deadline, rng,
                                                eta0=0.35,
                                                gamma0=6.0,
                                                gamma_inc=0.0015)
        if scr + 1e-12 < best_score:
            best_seq, best_score, best_len = a.copy(), scr, n
            print(f"   new global best after up‑sample score={best_score:.6f} (len={best_len})")
        # early exit if we have already crossed the target bound
        if best_score <= 1.503:
            print("[early exit] target bound reached.")
            break

    # ------------------------------------------------------------------
    # Final exact evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
