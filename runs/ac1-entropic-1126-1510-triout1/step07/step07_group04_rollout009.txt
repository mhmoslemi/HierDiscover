def _refine_exponentiated_gradient(a, deadline, rng, eta0=0.5, random_active_prob=0.5):
    n = len(a)
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute gradient: with prob random_active_prob use random active subgradient else uniform active
        if rng.random() < random_active_prob:
            grad, cur_max = _max_subgradient_random(a, rng)
        else:
            grad, cur_max = _max_subgradient_step(a)
        eta = eta0 / math.sqrt(it)
        exp = -eta * grad
        exp = np.clip(exp, -50, 50)  # avoid overflow
        a = a * np.exp(exp)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0/n
        else:
            a /= s
        # occasional jitter
        if it % 300 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=1e-6, size=n)
            a = _project_onto_simplex(a)
        cur_val = _raw_score(a)

            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] EG len={n} it={it} score={best_val:.6f}")

"""
Propose a low‑score step function.

Algorithm Overview
------------------
1. Warm‑start from the global variable `height_sequence_1` if it exists.
2. Perform a quick sweep of symmetric Beta(α,α) distributions on several
   lengths (4000‑20000).  The sweep is dense enough to locate a good
   shape (α controls the U‑shape of the distribution).
3. Refine the current best candidate with an exponentiated‑gradient
   (mirror‑descent) method on the simplex.  The sub‑gradient is taken
   either as the average over all active constraints (as before) or,
   with probability `random_active_prob`, from a *single* active index.
   The latter provides a larger, more aggressive step and helps escape
   plateaus.
4. Upsample the refined vector to larger lengths and repeat the refinement.
   Each up‑sampled phase gets a fresh time budget.
5. Finally evaluate the best vector with the provided `evaluate_sequence`
   and return it.

All operations respect the security checks of `evaluate_sequence`
(non‑negative, no NaNs, values ≤ 1000).  The vector is always
normalised to sum to 1, which makes the objective scale‑invariant.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable version
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # clip to avoid overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy score for a vector that already satisfies Σa = 1.
    Equals the true objective because the evaluator is scale‑invariant."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Sub‑gradient of f(a)=max_k (a*a)_k using the average over all active indices."""
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _max_subgradient_random(a: np.ndarray, rng: np.random.Generator):
    """Sub‑gradient taking a *single* active index (chosen uniformly)."""
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    if active.size:
        idx = rng.choice(active)
    else:
        idx = int(np.argmax(b))
    w = np.zeros_like(b)
    w[idx] = 1.0
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_exponentiated_gradient(a: np.ndarray,
                                   deadline: float,
                                   rng: np.random.Generator,
                                   eta0: float = 0.6,
                                   random_active_prob: float = 0.5):
    """
    Mirror‑descent (exponentiated‑gradient) refinement on the simplex.
    Sub‑gradient is chosen either as the average over all active constraints
    or from a single active constraint (controlled by `random_active_prob`).
    Returns the best vector seen and its raw score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        # Choose which sub‑gradient to use
        if rng.random() < random_active_prob:
            grad, cur_max = _max_subgradient_random(a, rng)
        else:
            grad, cur_max = _max_subgradient_step(a)

        eta = eta0 / math.sqrt(it)

        # Exponentiated update (multiplicative)
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)   # avoid overflow / underflow
        a = a * np.exp(exponent)

        # Re‑project onto the simplex
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # Tiny jitter occasionally to escape flat regions
        if it % 400 == 0 and rng.random() < 0.07:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] EG len={n} it={it} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _beta_sweep_lengths(lengths, alphas, current_best, current_score, deadline):
    """Exhaustive sweep of symmetric Beta(α,α) on the supplied lengths."""
    for n in lengths:
        if time.time() >= deadline:
            break
        for a_alpha in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a_alpha, a_alpha)
            sc = _raw_score(vec)
            if sc < current_score:
                current_best = vec.copy()
                current_score = sc
                print(f"[{time.time():.1f}] beta n={n} α={a_alpha:.3f} score={sc:.6f}")
    return current_best, current_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with a low ``evaluate_sequence`` score.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from the globally supplied solution if available.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if base_seq.sum() == 0.0:
            raise ValueError
        base_seq = base_seq / base_seq.sum()
    except Exception:
        # fallback: uniform distribution of moderate size
        base_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)

    best_seq = base_seq.copy()
    best_score = _raw_score(best_seq)
    print(f"[{time.time():.1f}] initial best raw score = {best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 0 – coarse sweep of symmetric Beta families on several lengths.
    sweep_lengths = [4000, 8000, 12000, 20000]
    # a dense grid of α – enough to locate the optimum roughly.
    alphas_grid = np.concatenate((
        np.linspace(0.08, 0.30, 12, endpoint=False),
        np.linspace(0.30, 0.70, 21),
        np.linspace(0.70, 0.95, 13)
    ))
    best_seq, best_score = _beta_sweep_lengths(
        sweep_lengths, alphas_grid, best_seq, best_score, deadline)

    # ------------------------------------------------------------------
    # Phase 1 – refinement on the current best length (moderate time).
    if time.time() < deadline:
        a = best_seq.copy()
        # allocate roughly 30 % of the remaining budget to this phase
        phase_deadline = time.time() + (deadline - time.time()) * 0.30
        a, scr = _refine_exponentiated_gradient(
            a, phase_deadline, rng,
            eta0=0.9, random_active_prob=0.6)
        if scr < best_score:
            best_seq, best_score = a.copy(), scr
            print(f"[{time.time():.1f}] after medium‑scale refinement: score={scr:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to larger lengths and refine intensively.
    larger_lengths = [16000, 24000, 30000, 36000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)   # safety
        # allocate a slice of the remaining time to this length
        remaining = deadline - time.time()
        # give each up‑sampled stage at most 25 % of what is left
        phase_deadline = time.time() + min(remaining * 0.25, 120.0)
        a, scr = _refine_exponentiated_gradient(
            a, phase_deadline, rng,
            eta0=0.7, random_active_prob=0.6)
        if scr + 1e-12 < best_score:
            best_seq, best_score = a.copy(), scr
            print(f"[{time.time():.1f}] new best after up‑sample: score={scr:.6f} (len={n})")
        # early stop if we have already crossed the target bound
        if best_score <= 1.503:
            print(f"[{time.time():.1f}] target bound reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final evaluation with the exact evaluator (handles clipping, etc.).
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
