"""
Propose a step‑function sequence with a very low ``evaluate_sequence`` score.

Core idea
---------
*  Build a one‑parameter family a_i(γ) that is analytically motivated:
   a_i ∝ ((i+½)*(n‑½‑i))^{-γ}.
   γ=0 gives the uniform distribution (score = 2);
   γ≈½ yields the discrete analogue of the arcsine law and makes the
   self‑convolution almost flat, i.e. close to the theoretical lower bound ≈ 1.
*  For each candidate length n we perform a cheap grid search on γ,
   followed by a few refinement steps (local grid + a short projected
   sub‑gradient descent) to drive the maximal convolution value down.
*  The search is completely on the probability simplex (Σa_i=1) so the
   scale‑invariant objective 2·n·max(conv)/(Σa_i)² reduces to 2·n·max(conv).
*  The algorithm stays well within the 1000 s budget: every score evaluation
   uses a single FFT (O(n log n)), and we only evaluate a few hundred γ‑candidates.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0) -> tuple[np.ndarray, float]:
    """
    Short projected sub‑gradient descent on the exact max‑convolution.
    Returns the refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)
    it = start_iter
    # we cap the total number of iterations to keep the run fast
    max_iters = 3000
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # occasional tiny jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-7
            a = _project_onto_simplex(a)
    return best_a, best_score


# ----------------------------------------------------------------------
def _gen_seq_gamma(n: int, gamma: float) -> np.ndarray:
    """
    Generate a symmetric non‑negative sequence of length n depending on a single
    parameter gamma.  The formula is
        a_i ∝ ((i+½)*(n‑½‑i))^{-γ}.
    The sequence is normalised to sum to 1.
    """
    pos = np.arange(n, dtype=np.float64) + 0.5          # i+½
    prod = pos * (n - pos)                              # (i+½)*(n‑½‑i)
    # prod > 0 for all i, so the power is safe
    weights = np.power(prod, -gamma, where=prod > 0)
    # avoid NaNs when gamma=0
    if gamma == 0.0:
        weights[:] = 1.0
    seq = weights / weights.sum()
    # numerical safety: ensure non‑negativity
    seq = np.maximum(seq, 0.0)
    return seq

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point invoked by the harness.  Returns a list of non‑negative
    heights that (hopefully) yields a low ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Candidate lengths – larger n gives a lower theoretical lower bound.
    # We stop early if we run out of time.
    lengths = [2000, 4000, 8000, 16000, 32000]

    best_seq = None
    best_score = float("inf")
    best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ----------------------------------------------------------------
        # 1) coarse grid search on γ
        gamma_grid = np.linspace(0.0, 1.5, 31)          # 31 points
        local_best_seq = None
        local_best_raw = float("inf")
        local_best_gamma = None

        for gamma in gamma_grid:
            seq = _gen_seq_gamma(n, gamma)
            maxb = _fft_self_convolve_max(seq)
            raw = 2.0 * n * maxb                     # raw score = 2·n·max(conv)
            if raw < local_best_raw:

    return final_list
