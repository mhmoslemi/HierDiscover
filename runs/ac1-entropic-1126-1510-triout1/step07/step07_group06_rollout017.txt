def beta_shape(n, alpha, beta):
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = (idx ** (alpha - 1)) * ((n - idx) ** (beta - 1))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a

"""
search for a low‑score step function.

We combine three complementary ideas:

1. **Parametric seeding** – a family of Beta‑shaped probability vectors
   (a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1}) that includes the uniform and the
   classic “arcsine’’ shape (α=β=½).  Random draws of (α,β) give many diverse
   candidates without expensive optimisation.

2. **Smooth‑max Adam** – each promising candidate is further optimised by
   minimising the log‑sum‑exp surrogate of ``max_k (a*a)_k`` (the same
   method that yielded the previous best solution).  The Adam optimiser
   works on the logit parametrisation and respects the simplex constraint
   via a soft‑max.

3. **Stochastic hill‑climbing** – starting from the best vector obtained
   above, we perform many cheap, two‑coordinate moves (transfer a small
   ε from index i to j) projected back onto the simplex.  The move size
   σ decays during the run, allowing the algorithm to escape shallow
   plateaux and to fine‑tune the ℓ∞‑convolution bound.

The routine respects a user‑provided time budget (default 1000 s) and
returns a plain list suitable for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of the convolution peak.
    Returns the best probability vector (Σ=1) and its *raw* score
    (2·n·max_k (a*a)_k) which equals the value returned by ``evaluate_sequence``
    when the sum of the vector is 1.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                     # Σa = 1, a ≥ 0
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} Adam it={it} raw={best_score:.6f} λ={lam:.3f}")

        # smooth max via log‑sum‑exp
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()                 # (2n‑1,) softmax weights

        # gradient w.r.t. a : 2·(a_rev * p) restricted to n entries
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = lam * lam_factor

        # occasional jitter for robustness
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()
    return best_a, best_score


# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Return a normalised Beta‑shaped vector of length n:
        a_i ∝ (i+½)^{α‑1}·(n‑i‑½)^{β‑1},   i = 0,…,n‑1.
    The result lies on the probability simplex.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    # work in log‑space to avoid overflow/underflow
    log_a = (alpha - 1.0) * np.log(i) + (beta - 1.0) * np.log(n - i)
    a = np.exp(log_a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a[:] = 1.0 / n
    return a


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """2·n·max_k (a*a)_k with Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _grid_search_length(
    n: int,
    deadline: float,
    n_samples: int = 200,

    """

    for a in candidates:

        raw = _raw_score(a)
        if raw < best_score:
            best_score = raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} sample={k+1:04d} raw={raw:.6f} α={alpha:.3f} β={beta:.3f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def _hill_climb_refine(
    a: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing that perturbs two coordinates at a
    time and projects back onto the simplex.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_raw = _raw_score(cur)
    sigma = 0.02                # initial move size
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # maximal transferable mass from i
        max_move = min(cur[i], sigma)
        if max_move <= 0.0:
            continue
        eps = np.random.rand() * max_move
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand_raw = _raw_score(cand)

        # accept if improvement; a very small probability to accept worsening
        if cand_raw < cur_raw or np.random.rand() < 0.001:
            cur = cand
            cur_raw = cand_raw
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill‑climb it={it} raw={cur_raw:.6f} σ={sigma:.5f}")

        # decay sigma slowly
        if it % 2000 == 0:
            sigma = max(sigma * 0.9, 1e-6)

    return cur, cur_raw

        final_seq, final_score = _adam_optimize_smoothed(
            x0,
            best_seq.shape[0],
            deadline,
            max_iters=1200,
            lr0=0.08,
            lam_start=0.6,
            lam_factor=1.002,
        )
        if final_score < best_score:
            best_score = final_score
            best_seq = final_seq.copy()
            print(f"Final Adam polish raw={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Evaluate with the black‑box and return.
    if best_seq is None:
        # As a last resort fall back to a uniform vector of length 1000.
        best_seq = np.full(1000, 1.0 / 1000, dtype=np.float64)
        print("Fallback to uniform sequence.")

    final_list = best_seq.tolist()
    final_eval = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_eval:.6f} (length={len(final_list)})")
    return final_list
