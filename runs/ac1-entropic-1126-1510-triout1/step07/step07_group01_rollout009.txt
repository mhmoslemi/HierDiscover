```python
"""
Propose a step‑function (probability vector) that minimises
``evaluate_sequence``.  The optimizer works in three phases:

1.  Explore several analytic families (arcsine, beta, triangular,
    exponential, sine, cosine, uniform) and random Dirichlet draws for
    a few different lengths.  Every candidate is projected onto the
    simplex and scored by the cheap proxy ``2·n·max(a*a)``.

2.  Refine the most promising seeds with three gradient‑based methods:
    exponentiated‑gradient (mirror‑descent), Adam and a *smooth*
    sub‑gradient descent that uses a soft‑max (log‑sum‑exp) smoothing
    of the max‑operator.  This smoothing gradually sharpens, allowing
    the optimizer to escape shallow local minima.

3.  A final hill‑climbing stage performs tiny random mass transfers
    (and occasional jitter) to polish the solution.

All operations keep the vector on the probability simplex, so the
objective is scale‑invariant and the returned list can be fed directly
to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # from https://doi.org/10.1007/978-3-540-85990-9_9
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap proxy for ``evaluate_sequence`` when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 80_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # exact sub‑gradient of the max (as a sum of active constraints)
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)
        w = np.zeros_like(b)
        w[active] = 1.0
        grad_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * grad_rev[len(a)-1:len(a)-1+len(a)]
        # diagonal correction (same as in the reference implementation)
        even = (active % 2 == 0)
        diag_idx = active[even] // 2
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 80_000,
                  step0: float = 0.25,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style adaptive gradient on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # sub‑gradient (same as above)
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)
        w = np.zeros_like(b)
        w[active] = 1.0
        grad_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * grad_rev[len(a)-1:len(a)-1+len(a)]
        even = (active % 2 == 0)
        diag_idx = active[even] // 2
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_subgradient_descent(a0: np.ndarray, deadline: float,
                                max_iters: int = 300_000,
                                eta0: float = 0.6,
                                beta_start: float = 1.0,
                                beta_end: float = 250.0) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the log‑sum‑exp smoothing of ``max``.
    ``beta`` is increased linearly from ``beta_start`` to ``beta_end``.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:

        b = _fft_self_convolve(a)                     # length 2n‑1
        # smooth weights (soft‑max over convolution values)
        cur_max = b.max()
        # linear schedule for beta
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)
        # to keep numbers stable subtract cur_max before exponentiation
        exp_vals = np.exp(beta * (b - cur_max))
        p = exp_vals / exp_vals.sum()

        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:

def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n

    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.003) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:

    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0

"""
Propose a step‑function (probability vector) that minimises
``evaluate_sequence``.  The optimizer works in three phases:

1.  Explore several analytic families (arcsine, beta, triangular,
    exponential, sine, cosine, uniform) and random Dirichlet draws for
    a few different lengths.  Every candidate is projected onto the
    simplex and scored by the cheap proxy ``2·n·max(a*a)`` (exactly the
    value returned by ``evaluate_sequence`` when the vector sums to 1).

2.  Refine the most promising seeds with three gradient‑based methods:
    exponentiated‑gradient (mirror‑descent), Adam and a *smooth*
    sub‑gradient descent that uses a soft‑max (log‑sum‑exp) smoothing
    of the max‑operator.  The smoothing parameter β is increased
    gradually, allowing the optimiser to escape shallow local minima
    before converging to a true sub‑gradient step.

3.  A final hill‑climbing stage performs tiny random mass transfers
    (and occasional jitter) to polish the solution.

All operations keep the vector on the probability simplex, so the
objective is scale‑invariant and the returned list can be fed directly
to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap proxy for ``evaluate_sequence`` when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 80_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)
        w = np.zeros_like(b)
        w[active] = 1.0
        grad_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * grad_rev[len(a) - 1: len(a) - 1 + len(a)]
        even = (active % 2 == 0)
        diag_idx = active[even] // 2
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 80_000,
                  step0: float = 0.25,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style adaptive gradient on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)
        w = np.zeros_like(b)
        w[active] = 1.0
        grad_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * grad_rev[len(a) - 1: len(a) - 1 + len(a)]
        even = (active % 2 == 0)
        diag_idx = active[even] // 2
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score

def _smooth_subgradient_descent(a0: np.ndarray, deadline: float,
                                max_iters: int = 300_000,
                                eta0: float = 0.6,
                                beta_start: float = 1.0,
                                beta_end: float = 250.0) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a log‑sum‑exp smoothing of the max‑operator.
    β grows linearly from ``beta_start`` to ``beta_end``.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        b = _fft_self_convolve(a)                     # length 2n‑1
        cur_max = b.max()

        exp_vals = np.exp(beta * (b - cur_max))
        p = exp_vals / exp_vals.sum()
        conv_rev = _fft_convolve(a[::-1], p)
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]      # no diagonal correction needed

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad

        if time.time() >= deadline:
            break

        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=30_000,
                                        eta0=0.9)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} exp‑grad improved to {sc:.6f}")

        seq, sc = _adam_descent(seed_seq, deadline,
                                max_iters=30_000,
                                step0=0.25)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Heavy smooth‑subgradient descent on the current best
    smooth_deadline = deadline - 30.0   # reserve ~30 s for hill‑climbing
    if time.time() < smooth_deadline:
        seq, sc = _smooth_subgradient_descent(best_seq, smooth_deadline,
                                              max_iters=400_000,
                                              eta0=0.5,
                                              beta_start=1.0,
                                              beta_end=300.0)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq, sc
            print(f"[{time.time():.1f}] smooth sub‑grad finished with best={sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained hill‑climbing (tiny random transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.03    # up‑to‑3 % of total mass per move, decays geometrically

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 12_000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        if iteration % 5_000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
