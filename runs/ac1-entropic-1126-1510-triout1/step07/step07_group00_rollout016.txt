    deadline = start + budget_s - 0.5

    else:
        base_n = 3500

    # Candidate generation
    candidates = []
    for n in lengths:
        # define families
        # uniform: [1/n]*n
        a = np.full(n, 1.0/n)
        candidates.append((a, _max_convolution_score(a)))
        # power law
        for p in [0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 2.0]:
            idx = np.arange(n)+1
            a = idx**(-p)
            a /= a.sum()
            candidates.append((a, _max_convolution_score(a)))
        # exponential
        for lam in [0.001, 0.003, 0.005, 0.01, 0.02]:
            idx = np.arange(n)
            a = np.exp(-lam*idx)
            a /= a.sum()
            candidates.append((a,_max_convolution_score(a)))
        # beta symmetric
        for beta in [0.1,0.2,0.3,0.5,0.8,1.0,1.5,2.0]:
            a = _beta_symmetric_distribution(n,beta)
            candidates.append((a,_max_convolution_score(a)))
        # random Dirichlet
        for alpha in [0.1,0.2,0.5,1.0,2.0]:
            a = _random_dirichlet_alpha(n,alpha)
            candidates.append((a,_max_convolution_score(a)))

        if time.time() >= deadline:
            break

    # Sort and keep top K seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(3, len(candidates))
    top_seeds = [c[0] for c in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)

    # allocate time for refinement: remaining time / top_k
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_seq.tolist()
    per_seed_time = remaining / top_k

    for seed_idx, a0 in enumerate(top_seeds):
        seed_deadline = time.time() + per_seed_time
        # run Adam descent for longer
        a_opt, sc = _adam_descent(a0, seed_deadline, max_iters=300000, step0=0.2)
        if sc < best_score:
            best_score, best_seq = sc, a_opt.copy()

            print(f"seed {seed_idx+1} ExpGrad improved to {best_score:.6f}")

        # local swap refinement
        a_local = best_seq.copy()
        cur_score = best_score
        while time.time() < seed_deadline:
            # pick random i,j
            i, j = np.random.randint(0, a_local.size, size=2)
            if i == j:
                continue
            # propose move epsilon = random fraction of min(a[i], 1 - a[j])
            max_eps = min(a_local[i], 1.0 - a_local[j])
            if max_eps <= 0:

            a_new[j] += eps

            new_score = _max_convolution_score(a_new)
            if new_score < cur_score - 1e-12:

        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:

        # But we can also binary search for epsilon leading to max reduction.

        a[i_src] -= max_transfer
        a[i_dst] += max_transfer
        new_score = _max_convolution_score(a)

            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"FW improved to {best_score:.6f}")

            # revert

            a[i_dst] -= max_transfer

"""
Propose a low‑score non‑negative step function.

The algorithm consists of three stages:

1️⃣ **Broad seeding** – a collection of analytic families (uniform,
   power‑law, exponential, symmetric beta, triangular) and a few
   Dirichlet draws is sampled for several lengths (around the length of
   the previously best sequence).  All candidates are ℓ₁‑normalised and
   ranked by the cheap surrogate `2·n·max(conv(a,a))`.

2️⃣ **Intensive continuous optimisation** – the best seeds are polished
   with two complementary mirror‑descent schemes (Adam on the Euclidean
   simplex and exponentiated‑gradient on the KL‑simplex).  They run for
   as many iterations as the remaining time permits.

3️⃣ **Local stochastic refinement** – starting from the globally best
   sequence we perform a fast simulated‑annealing / hill‑climbing loop:
   random transfers of mass between coordinates and random Dirichlet
   re‑allocations on small subsets.  Any improvement is kept, and the
   best ever seen sequence is returned.

All steps respect the supplied time budget and never return `None`,
so the caller always receives a list that can be fed directly to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate: 2·n·max(a*a) assuming sum(a)=1 (exact for evaluate_sequence)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (sum(a)=1).

    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before diagonal correction
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (i = j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(alpha,…,alpha)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric beta on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Single‑sided exponential decay, centred at the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    """Simple symmetric triangle (peak in the centre)."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}   (non‑symmetric, decreasing)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals


def _uniform_distribution(n: int) -> np.ndarray:
    """Exactly uniform vector."""
    return np.full(n, 1.0 / n)

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300000,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style adaptive optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _stochastic_refinement(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap simulated‑annealing / hill‑climbing loop.
    Randomly moves mass between coordinates and occasionally reshuffles
    a small subset using a Dirichlet draw.
    """
    n = a.size
    cur = a.copy()
    cur_score = _max_convolution_score(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    while time.time() < deadline:
        # Choose move type
        if np.random.rand() < 0.65:
            # simple transfer i -> j
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            max_eps = min(cur[i], 1.0 - cur[j])
            if max_eps <= 1e-15:
                continue
            eps = max_eps * np.random.rand() * 0.3  # up‑to 30 % of feasible transfer
            new_seq = cur.copy()
            new_seq[i] -= eps
            new_seq[j] += eps
        else:
            # redistribute mass on a random small subset
            s = np.random.randint(2, 5)          # size of the subset
            idx = np.random.choice(n, size=s, replace=False)
            total = cur[idx].sum()
            if total < 1e-15:
                continue
            # Dirichlet with concentration 0.5 favours sparsity
            alphas = np.full(s, 0.5)
            w = np.random.gamma(alphas, 1.0)
            w = total * w / w.sum()
            new_seq = cur.copy()
            new_seq[idx] = w

        new_score = _max_convolution_score(new_seq)

        # deterministic acceptance of improvements
        if new_score < cur_score:
            cur, cur_score = new_seq, new_score
            if new_score < best_score:
                best, best_score = new_seq.copy(), new_score
                print(f"[{time.time():.1f}] stochastic improved to {best_score:.6f}")
        else:
            # occasional uphill moves (simulated‑annealing style)
            # temperature schedule: linear decay
            elapsed = time.time() - start
            frac = elapsed / (deadline - start + 1e-9)
            T = 1e-3 * (1.0 - frac) + 1e-6
            if np.random.rand() < np.exp(-(new_score - cur_score) / T):
                cur, cur_score = new_seq, new_score

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of floats) that minimises
    ``evaluate_sequence``.  The function respects the supplied time budget
    (≈ 1000 s) and always returns a feasible sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 0. Determine a base length (use previous solution if available)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # 1. Choose a handful of lengths to explore (not too many!)
    lengths = set()
    for scale in (0.8, 1.0, 1.2, 1.4, 1.6, 2.0):
        n = int(base_n * scale)
        if 500 <= n <= 8000:
            lengths.add(n)
    for _ in range(3):
        n = np.random.randint(500, min(8000, base_n * 2) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 2. Broad sampling of analytic families + random Dirichlet draws
    candidates = []   # list of (vector, score)

    beta_ps = [0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    power_ps = [0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 2.0]
    explams = [0.001, 0.003, 0.005, 0.01, 0.02]
    dirichlet_alphas = [0.1, 0.3, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- deterministic families -----------------------------------
        for gen, name in (
            (_uniform_distribution, "uniform"),
            (_triangular_distribution, "triangular"),
            (_exponential_distribution, "exponential"),
        ):
            a = gen(n)
            candidates.append((a, _max_convolution_score(a)))
            print(f"[{time.time():.1f}] n={n:<5d} {name:12s}")

        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        for p in power_ps:
            if time.time() >= deadline:
                break
            a = _power_law_distribution(n, p)
            candidates.append((a, _max_convolution_score(a)))

        for lam in explams:
            if time.time() >= deadline:
                break
            a = _exponential_distribution(n, lam=lam)
            candidates.append((a, _max_convolution_score(a)))

        # ---- random Dirichlet draws -----------------------------------
        for alpha in dirichlet_alphas:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 3. Bring the previously best sequence into the pool (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Keep a few promising seeds for deep optimisation
    candidates.sort(key=lambda x: x[1])
    top_k = min(3, len(candidates))
    top_seeds = [c[0] for c in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"Generated {len(candidates)} candidates, top {top_k} kept.")
    print(f"Current best seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Allocate remaining wall‑time among the seeds
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_seq.tolist()
    per_seed = remaining / top_k

    for idx, seed_vec in enumerate(top_seeds):
        seed_deadline = time.time() + per_seed
        print(f"--- Optimising seed {idx+1}/{top_k} (size={seed_vec.size})")
        # Adam descent
        a_adam, sc_adam = _adam_descent(seed_vec, seed_deadline,
                                         max_iters=300000, step0=0.2)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, a_adam.copy()
            print(f"[{time.time():.1f}] Adam improved to {best_score:.6f}")

        # Exponentiated‑gradient descent
        a_exp, sc_exp = _exp_gradient_descent(a_adam, seed_deadline,
                                              max_iters=150000, eta0=0.8)
        if sc_exp < best_score - 1e-14:
            best_score, best_seq = sc_exp, a_exp.copy()
            print(f"[{time.time():.1f}] Exp‑Grad improved to {best_score:.6f}")

        # Local stochastic refinement (fast hill‑climbing)
        a_loc, sc_loc = _stochastic_refinement(best_seq, seed_deadline)
        if sc_loc < best_score - 1e-14:
            best_score, best_seq = sc_loc, a_loc.copy()
            print(f"[{time.time():.1f}] Stochastic refinement improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final pairwise Frank‑Wolfe style mass‑transfer refinement
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 50000:
        it_fw += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # biggest contributor to the current max
        i_dst = int(np.argmin(grad))   # best sink
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try full transfer first
        a_try = a.copy()
        a_try[i_src] -= max_transfer
        a_try[i_dst] += max_transfer
        sc_try = _max_convolution_score(a_try)

        improved = False
        if sc_try < cur_score - 1e-12:
            a, cur_score = a_try, sc_try
            improved = True
        else:
            # try a half transfer as a fallback
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            sc_half = _max_convolution_score(a_half)
            if sc_half < cur_score - 1e-12:
                a, cur_score = a_half, sc_half
                improved = True

        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={it_fw} new best {best_score:.6f}")

        # occasional tiny jitter – helps to escape microscopic plateaus
        if it_fw % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Ensure a proper simplex vector – tiny numerical errors are projected away
    best_seq = _project_onto_simplex(best_seq, s=1.0)
    return best_seq.tolist()
