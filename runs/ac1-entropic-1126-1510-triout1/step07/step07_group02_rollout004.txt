def hill_climb(a, conv, M, deadline):
    n = len(a)
    eps = 1e-14
    best_seq = a.copy()
    best_M = M
    it = 0
    while time.time() < deadline:
        it += 1
        # find current max location
        k = int(conv.argmax())
        # find pair (i,j) with highest product in that window
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        best_prod = -1.0
        best_i = i_start
        best_j = k - i_start
        # we can also sample a subset for speed. But n is moderate.
        for i in range(i_start, i_end+1):
            j = k - i
            prod = a[i] * a[j]
            if prod > best_prod:
                best_prod = prod
                best_i, best_j = i, j
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n-1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.03)  # allow up to 3% of mass
        delta = max_delta * random.random()
        if delta <= eps:
            continue
        # snapshot
        a_old = a.copy()
        conv_old = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < best_M - 1e-12:
            best_M = new_M
            best_seq = a.copy()
            # maybe log
            if it % 5000 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {2*n*best_M:.6f}")
        else:
            # revert
            a = a_old
            conv = conv_old
        # occasional jitter if stuck
        if it % 3000 == 0 and random.random() < 0.05:

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta(n, a, b):
    """Beta(a,b) on {0,…,n‑1}, normalised to sum = 1 (a,b>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log to stay stable for extreme a,b
    logw = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(logw - np.max(logw))    # shift for numeric stability
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w


def _distribution_beta_sym(n, p):
    return _distribution_beta(n, p, p)


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j,
    updating both ``a`` and its full self‑convolution array ``conv``
    in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part of the convolution (2*delta*old_a)
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

        w = np.zeros_like(b)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad iter={it} new best {cur:.6f}")

        # guard against spending the last fraction of a second without progress
        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
