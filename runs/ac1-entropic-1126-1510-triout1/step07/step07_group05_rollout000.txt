def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    # cell‑centre coordinates in [0,1)
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5000,
                            tau0: float = 0.5,
                            tau_max: float = 200.0,
                            lr0: float = 0.18) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max surrogate of max(a*a).

    Parameters
    ----------
    x0 : np.ndarray
        Initial logits (any real numbers).
    n  : int
        Length of the sequence.
    deadline : float
        Unix time at which the optimisation must stop.
    max_iters : int
        Upper bound on the number of Adam iterations.
    tau0, tau_max : float
        Exponential temperature schedule: τ_i = τ0 * (τ_max/τ0)^{(i‑1)/(max_iters‑1)}.
    lr0 : float
        Base learning rate (Adam’s “effective” step).

    Returns
    -------
    best_a   : np.ndarray
        Normalised sequence (sum = 1) achieving the lowest *raw* score.
    best_raw : float
        Corresponding raw value 2·n·max(a*a) (the denominator of the
        evaluation function does not depend on scaling).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_raw = float("inf")
    it = 0

    # pre‑compute the exponential factor for τ schedule
    if max_iters > 1:
        tau_factor = (tau_max / tau0) ** (1.0 / (max_iters - 1))
    else:
        tau_factor = 1.0

    tau = tau0

    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- a = softmax(x)  (ensures Σa = 1, a ≥ 0) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- exact convolution & raw score (for bookkeeping) -----
        conv = _fft_self_convolve_full(a)          # length 2·n‑1
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_raw:
            best_raw = raw_score
            best_a = a.copy()
            # occasional progress output
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_raw:.6f}")

        # ----- temperature (exponential schedule) -----
        if it > 1:
            tau *= tau_factor
            if tau > tau_max:
                tau = tau_max

        # ----- smooth‑max weights (log‑sum‑exp) -----
        max_conv = cur_max
        # use a stable softmax on (τ * (conv - max_conv))
        exp_w = np.exp(tau * (conv - max_conv))
        w = exp_w / exp_w.sum()

        # gradient of smooth‑max w.r.t. a (via convolution with reversed a)
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # ----- chain rule through the soft‑max parametrisation -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter to escape plateaus (every ~800 iterations)
        if it % 800 == 0:
            x += np.random.randn(*x.shape) * 0.02

    # fallback – should never happen
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_raw


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           max_iters: int = 2_000_000,
                           step0: float = 0.09) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact max‑convolution.

    Parameters
    ----------
    a          : np.ndarray, sum(a)=1, a≥0
    deadline   : float, time constraint
    max_iters  : int, upper bound on iterations (may stop earlier)
    step0      : float, initial step‑size factor (scaled by 1/√t)

    Returns
    -------
    a_refined : np.ndarray (normalised)
    best_raw  : float   2·n·max(a*a)
    """
    n = a.shape[0]
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- sub‑gradient of the exact max -----
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        # active set tolerance (relative)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        # gradient w.r.t. a (same as in Adam but with exact active‑set weights)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # diminishing step size
        step = step0 / math.sqrt(it)

        a = a - step * grad
        a = _project_onto_simplex(a)

        # occasional tiny random perturbations (helps escape shallow plateaux)
        if it % 1000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 4e-7
            a = _project_onto_simplex(a)

        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_raw:.6f}")

    return best_a, best_raw

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the competition harness.

    Returns a list of non‑negative heights that (hopefully) yields a low
    evaluate_sequence score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Try to warm‑start from a previously discovered sequence if present.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # Length schedule – start small (cheap) and grow.
    lengths = [500, 1000, 2000, 4000, 8000, 12000]

    best_seq = None
    best_raw = float("inf")
    best_n = None

    # Early baseline if a previous solution is available.
    if prev is not None:
        raw = 2.0 * len(prev) * _fft_self_convolve_max(prev)
        print(f"Baseline (len={len(prev)}) raw={raw:.6f}")
        best_seq = prev.copy()
        best_raw = raw
        best_n = len(prev)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # ----------- build initial logits --------------------------------
        init_logits = []

        # warm‑start from global best, interpolated to new length
        if best_seq is not None:
            cand = _interpolate_to_length(best_seq, n)
            init_logits.append(np.log(cand + 1e-12))

        # deterministic shapes
        half_sine = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
        half_sine /= half_sine.sum()
        init_logits.append(np.log(half_sine + 1e-12))

        hann = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n, dtype=float) / (n - 1)))
        hann /= hann.sum()
        init_logits.append(np.log(hann + 1e-12))

        # uniform start
        init_logits.append(np.zeros(n, dtype=np.float64))

        # one random Dirichlet draw (moderate concentration)
        alpha = np.random.rand() * 3.0 + 0.5   # α ∈ (0.5, 3.5)
        d = np.random.gamma(alpha, 1.0, size=n)
        d /= d.sum()
        init_logits.append(np.log(d + 1e-12))

        local_best_seq = None
        local_best_raw = float("inf")

        # ----------- run optimisation from each start --------------------
        for start_id, logits in enumerate(init_logits, 1):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] start {start_id}/{len(init_logits)} (n={n})")

            # 1) Adam on smooth max‑convolution
            seq, raw_score = _adam_optimize_smoothed(
                logits,
                n,
                deadline,
                max_iters=5000,          # enough for a quick continuation
                tau0=0.5,
                tau_max=200.0,
                lr0=0.18
            )

            # 2) Short exact sub‑gradient refinement (few thousand steps)
            seq, raw_score = _refine_by_subgradient(
                seq,
                deadline,
                max_iters=2000,          # keep it short – we will revisit later
                step0=0.09
            )

            if raw_score < local_best_raw:
                local_best_raw = raw_score
                local_best_seq = seq.copy()
                print(f"   *** new local best for n={n}: raw={local_best_raw:.6f}")

        # Update global best if this length improved things
        if local_best_seq is not None and local_best_raw < best_raw:
            best_raw = local_best_raw
            best_seq = local_best_seq.copy()
            best_n = n
            print(f">>> Updated global best: raw={best_raw:.6f} (len={best_n})")

    # ------------------------------------------------------------------
    # Final long‑run sub‑gradient refinement on the best sequence discovered.
    if best_seq is not None:
        print("\n=== Final refinement on global best ===")
        a = best_seq.copy()
        a, final_raw = _refine_by_subgradient(a, deadline, max_iters=5_000_000)
        best_seq = a.copy()
        best_raw = final_raw
    else:
        # Fallback – simple uniform distribution of moderate size
        n = 5000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_seq)

    # ------------------------------------------------------------------
    # Convert to list and evaluate via the supplied black‑box.
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
