"""
Propose a step‑function (non‑negative vector) that minimises

    score = 2 * n * max_k (a * a)_k   (sum(a) is forced to 1)

The implementation combines several ideas that improve on the reference
solution:

* **Multiple length schedule** – start with a cheap short vector, repeatedly
  interpolate the current best to a longer length and continue the search.
  Longer vectors allow a finer shape and usually a lower bound.

* **Rich set of initialisations** – besides the usual uniform/Dirichlet
  draws we also include simple analytic shapes (inverse‑polynomial) that
  concentrate mass near the edges.  All starters are expressed as logits
  (log‑softmax parameters) for the Adam optimiser.

* **Adam optimisation on a smooth max** – the true max of the convolution
  is approximated by a temperature‑controlled log‑sum‑exp; the temperature
  grows linearly so the optimiser gradually focuses on the true maximum.

* **Projected sub‑gradient refinement** – after Adam we switch to the exact
  nonsmooth objective using the classic sub‑gradient step.

* **Local hill‑climbing (mass transfer)** – a cheap stochastic move that
  shifts a small amount of probability mass from one index to another.
  Accepted moves strictly improve the raw score; occasional random
  perturbations help escape shallow plateaus.

* **Time‑aware loops** – every phase checks the global deadline (the
  harness gives us 1000 s) and aborts early if necessary, guaranteeing a
  finite return.

All helper functions are defined at module level (no closures) and only
standard libraries (numpy, math, time) are used.  The final list is
returned as plain Python floats; the harness will call `evaluate_sequence`
on it.

import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _adam_optimize_smoothed(
    logits: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 6000,
    tau0: float = 0.5,
    tau_max: float = 100.0,
    lr0: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam on the soft‑max parametrisation.
    Returns the best normalised vector a (sum = 1) and the associated raw
    score 2·n·max(a*a).
    """
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_raw = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        # ----- soft‑max (ensures Σa = 1) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution -----
        conv = _fft_self_convolve_full(a)          # length 2n‑1
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_raw:
            best_raw = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_raw:.6f}")

        # ----- temperature schedule -----
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # ----- smooth‑max weight (stable log‑sum‑exp) -----
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # ----- gradient w.r.t a (exact for the smooth max) -----
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ----- chain rule through the soft‑max -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to avoid stagnation
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback – uniform distribution (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_raw

def _max_subgradient_step(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_by_subgradient(
    a: np.ndarray,
    deadline: float,
    start_iter: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for the exact max‑convolution.
    Returns a refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(a)

def _hill_climb_optimize(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search that repeatedly transfers a tiny amount
    of mass from one index to another.  Every accepted move strictly improves
    the raw score; the schedule for the transferred mass slowly decays.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_raw = 2.0 * n * _fft_self_convolve_max(cur_a)
    best_a = cur_a.copy()
    best_raw = cur_raw
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # pick donor and recipient
        i = np.random.randint(n)
        if cur_a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # mass to move – decays with iterations
        frac = 0.2 * (1.0 - it / max_iters) + 0.001
        eps = frac * cur_a[i]
        if eps <= 0.0:
            continue

        a_new = cur_a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # keep numerical sanity (no negatives)
        if a_new[i] < -1e-12:
            continue

        # renormalise periodically to fight floating‑point drift
        if it % 5000 == 0:
            a_new = _project_onto_simplex(a_new)

        new_raw = 2.0 * n * _fft_self_convolve_max(a_new)
        if new_raw < cur_raw:
            cur_a = a_new
            cur_raw = new_raw
            if new_raw < best_raw:
                best_raw = new_raw
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] hill‑climb it={it}  raw={best_raw:.6f}")
        # occasional random jitter to escape tiny plateaus
        if it % 10000 == 0 and np.random.rand() < 0.05:
            cur_a += np.random.randn(n) * 1e-6
            cur_a = _project_onto_simplex(cur_a)

    # final clean‑up projection
    best_a = _project_onto_simplex(best_a)
    best_raw = 2.0 * n * _fft_self_convolve_max(best_a)
    return best_a, best_raw

def _make_shape_initial_logits(n: int, power: float) -> np.ndarray:
    """
    Analytic “inverse‑polynomial” shape: a_i ∝ ((i+0.5)*(n-0.5-i))**power .
    Power is negative for edge‑heavy distributions.
    Returns logits = log(a) (with a tiny epsilon for safety).
    """
    idx = np.arange(n, dtype=float) + 0.5
    base = idx * (n - idx)
    # avoid zero division when power < 0
    eps = 1e-12
    a = (base + eps) ** power
    a = np.maximum(a, eps)
    a /= a.sum()
    return np.log(a)

            start_logits.append(np.log(d + 1e-12))

        # a couple of very spiky Dirichlet draws
        for _ in range(2):
            alpha = np.random.rand() * 0.09 + 0.001       # α ∈ (0.001,0.091)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            start_logits.append(np.log(d + 1e-12))

        # ---- run the optimiser from each start -----------------
        local_best_seq = None
        local_best_raw = float("inf")

        # number of Adam iterations – scale down for very large n
        max_adam_iters = 8000 if n <= 8000 else 4000

        for sid, logits in enumerate(start_logits, 1):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] start {sid}/{len(start_logits)} for n={n}")

            # 1) Adam on the smooth max
            seq, raw = _adam_optimize_smoothed(
                logits,
                n,
                deadline,
                max_iters=max_adam_iters,
                tau0=0.5,
                tau_max=200.0,
                lr0=0.10,
            )

            # 2) Exact sub‑gradient refinement
            seq, raw = _refine_by_subgradient(seq, deadline)

            # 3) Optional short hill‑climb (cheap, a few hundred iterations)
            # only for the cheapest n to keep time budget
            if n <= 8000:
                seq, raw = _hill_climb_optimize(seq, deadline, max_iters=20000)

            if raw < local_best_raw:
                local_best_raw = raw
                local_best_seq = seq.copy()
                print(f"   *** new local best for n={n}: raw={local_best_raw:.6f}")

        if local_best_seq is not None and local_best_raw < best_raw:
            best_raw = local_best_raw
            best_seq = local_best_seq.copy()
            best_n = n
            print(f">>> Updated global best: raw={best_raw:.6f} (length={best_n})")

    # --------------------------------------------------------------
    # Final intensive refinement on the current global best
    if best_seq is not None:
        print("\n=== Final intensive refinement ===")
        a = best_seq.copy()
        a, best_raw = _refine_by_subgradient(a, deadline)
        a, best_raw = _hill_climb_optimize(a, deadline, max_iters=200000)
        best_seq = a.copy()
    else:
        # Fallback: uniform distribution of moderate size
        n = 5000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_seq)

    # --------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # the harness supplies this
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
