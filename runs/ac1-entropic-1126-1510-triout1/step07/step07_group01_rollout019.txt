def beta_distribution(n, p,q):

    w[np.isnan(w)] = 0
    sumw = w.sum()
    if sumw==0:
        return np.full(n, 1.0/n)

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Broad sampling** – generate many candidate probability vectors
   (arcsine, triangular, exponential, sine, cosine, uniform,
   symmetric & asymmetric Beta distributions, random Dirichlet draws,
   and random mixtures of two families).  All candidates are normalised
   to the simplex and scored by the cheap surrogate
   `2·n·max(a*a)` (identical to ``evaluate_sequence`` for sum=1).

2. **Selection** – keep the top‑k cheapest candidates.

3. **Heavy optimisation** – each seed is refined by several
   optimisation passes that stay on the simplex:
   * exponentiated‑gradient descent,
   * Adam,
   * plain sub‑gradient descent,
   * a *smoothed* gradient descent that replaces the max‑operator by a
     soft‑max with a continuation parameter ``γ`` (increasing slowly).

4. **Fine‑grained hill‑climbing** – tiny random mass transfers and
   occasional jitter finish the search.

All steps respect the global time budget (default 1000 s) and the
objective is scale‑invariant, so the returned vector is already a valid
input for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import sparse

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of `max(a*a)` on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8):
    """Mirror‑descent (exponentiated‑gradient) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s != 0 else np.full_like(a, 1.0 / a.size)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 400_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    w = np.where(np.isfinite(w), w, 0.0)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, gamma: float):
    """Weighted gradient of the soft‑max of a*a (γ controls smoothness)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    b_max = b.max()
    # soft‑max weights, stabilised by subtracting the maximum
    w = np.exp(gamma * (b - b_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # 2 * conv_rev(a, w)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    diag_w = w[2 * np.arange(n)]
    grad -= a * diag_w
    return grad, b_max


def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.7,
                             gamma0: float = 5.0,
                             gamma_factor: float = 1.001):
    """Continuation method: start with smooth max (γ≈5) and increase γ."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    gamma = gamma0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _ = _smooth_gradient(a, gamma)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()
        gamma *= gamma_factor
    return best_a, best_score

                a2 = _exponential_distribution(n, lam=lam)

            lam_mix = np.random.rand()
            a = lam_mix * a1 + (1.0 - lam_mix) * a2
            a = a / a.sum()
            candidates.append((a, _max_convolution_score(a)))

    # ------------------------------------------------------------------
    # 2. Include previously best sequence, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 20
    top_seeds = [seq for seq, _ in candidates[:top_k]]
    best_seq = top_seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"[{time.time():.1f}] initial cheap best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy optimisation on each seed (time‑sharing)
    remaining_time = deadline - time.time()
    per_seed_budget = remaining_time * 0.8 / max(len(top_seeds), 1)

    for idx, seed_seq in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        seed_deadline = time.time() + per_seed_budget

        # exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, seed_deadline,
                                        max_iters=60_000,
                                        eta0=0.9)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} exp‑grad → {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, seed_deadline,
                                max_iters=60_000,
                                step0=0.25)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} Adam → {sc:.6f}")

        # Smoothed‑max descent (continuation)
        seq, sc = _smooth_gradient_descent(seed_seq, seed_deadline,
                                            max_iters=120_000,
                                            eta0=0.7,
                                            gamma0=5.0,
                                            gamma_factor=1.001)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} smoothed → {sc:.6f}")

        # Plain sub‑gradient descent as a final tidy‑up
        seq, sc = _subgradient_descent(seed_seq, seed_deadline,
                                       max_iters=200_000,
                                       eta0=0.5)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1} subgrad → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Fine‑grained hill climbing (tiny random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.03   # start with up to 3 % of total mass

    while time.time() < deadline:
        it += 1
        # geometric decay of the transferred amount
        transfer = max_transfer * (0.5 ** (it / 12_000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 5_000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
