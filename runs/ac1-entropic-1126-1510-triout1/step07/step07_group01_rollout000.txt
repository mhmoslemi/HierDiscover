"""
Improved step‑function optimizer for ``evaluate_sequence``.

The search proceeds in three stages:

1️⃣ **Massive cheap sampling** – a large family of analytically
    parametrised probability vectors (arcsine, triangular, exponential,
    sine, cosine, uniform, symmetric and asymmetric beta, power‑law) is
    generated for a handful of lengths around the previously known best
    size.  For each candidate we compute the cheap score
    ``2·n·max(conv)`` (the exact objective for a unit‑sum vector).  The
    best 30 seeds are kept.

2️⃣ **Polyak‑subgradient descent** – starting from each retained seed we
    run a projected sub‑gradient method that uses a *Polyak step size*
    ``(f(x)‑L)/‖g‖²`` with a known lower bound ``L=1.5``.  This yields a
    much more aggressive descent than the plain diminishing step‑size
    used before.  The best refined vector is kept.

3️⃣ **Fine hill‑climbing** – a final short phase performs tiny random
    mass transfers and occasional jitter to polish the best solution.

All vectors are kept on the probability simplex, therefore the returned
list can be fed directly to ``evaluate_sequence`` (the objective is
scale‑invariant).  The routine respects the 1000 s time budget and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of ``max(a*a)`` on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)  # length 2n-1
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # weight vector of active constraints
    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of a (reversed) with w gives the gradient before diagonal fix
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction: when k = 2*i the term a_i*a_i appears once,
    # but we have counted it twice in the convolution.
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex with concentration alpha."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full_like(x, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _beta_distribution(n: int, p: float, q: float) -> np.ndarray:
    """Symmetric or asymmetric beta density on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = (x ** (p - 1)) * ((1.0 - x) ** (q - 1))
    if w.sum() == 0 or np.isnan(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w.sum()
    return w


def _powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    den = (i + 0.5) * (n - i - 0.5)
    vals = den ** (-p)
    if vals.sum() == 0 or np.isnan(vals).any():
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals = vals / vals.sum()
    return vals

def _polyak_subgrad(a0: np.ndarray,
                    deadline: float,
                    lower_bound: float = 1.5,
                    alpha: float = 0.95,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with Polyak step size.
    Returns the best vector found and its cheap score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max, _ = _subgrad(a)
        score = 2.0 * n * cur_max          # cheap score
        norm2 = np.dot(grad, grad)
        if norm2 == 0.0:
            break
        # Polyak step; ensure positivity
        step = alpha * (score - lower_bound) / (norm2 + 1e-15)
        if step <= 0:
            # fallback to a diminishing step
            step = alpha / np.sqrt(it)

        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        new_score = _max_convolution_score(a)
        if new_score < best_score - 1e-15:
            best_score, best_a = new_score, a.copy()

    return best_a, best_score

            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.003), "exponential"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (lambda m: np.full(m, 1.0 / m), "uniform")
        ]
        for gen, name in families:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} {name:10s} score={score:.6f}")

        # Power‑law family
        for p in power_ps:
            a = _powerlaw_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} power(p={p:4.2f}) score={score:.6f}")

        # Symmetric beta (p = q)
        for p in beta_ps:
            a = _beta_distribution(n, p, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} beta_sym(p={p:4.2f}) score={score:.6f}")

        # Asymmetric beta
        for p in beta_ps:
            for q in beta_qs:
                a = _beta_distribution(n, p, q)
                score = _max_convolution_score(a)
                candidates.append((a, score))
                print(f"[{time.time():.1f}] n={n:4d} beta(p={p:4.2f},q={q:4.2f}) score={score:.6f}")

        # Exponential with varied decay
        for lam in exp_lams:
            a = _exponential_distribution(n, lam)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} exponential(lam={lam:.4f}) score={score:.6f}")

        # Random Dirichlet draws with varying concentration
        for _ in range(3):
            alpha = np.exp(np.random.uniform(np.log(0.02), np.log(5.0)))
            a = _random_dirichlet(n, alpha=alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} Dirichlet(α={alpha:.3f}) score={score:.6f}")

    # --------------------------------------------------------------
    # Include previously discovered best sequence, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best (cheap) score={score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 30
    top_candidates = candidates[:top_k]
    print(f"--- Selected {len(top_candidates)} seeds for refinement "
          f"(best cheap score={top_candidates[0][1]:.6f})")

    # --------------------------------------------------------------
    # 2️⃣ Polyak sub‑gradient refinement (aggressive descent)
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # Allocate roughly half the remaining budget for refinement,
    # leaving ~30 s for the final hill‑climb.
    time_left = deadline - time.time()
    refinement_deadline = time.time() + max(5.0, time_left - 30.0)

    # Run Polyak descent on the top few seeds (up to 5)
    for idx, (seed_seq, cheap) in enumerate(top_candidates[:5]):
        if time.time() >= refinement_deadline:
            break
        seq, sc = _polyak_subgrad(seed_seq, refinement_deadline,
                                  lower_bound=1.5, alpha=0.95,
                                  max_iters=400_000)
        if sc < best_score - 1e-12:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} "
                  f"Polyak improved to {sc:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Fine hill‑climbing (tiny mass transfers)
    # Leave ~30 s for this phase.
    hill_deadline = deadline
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.04   # start with up to 4 % of total mass

    while time.time() < hill_deadline:
        iteration += 1
        # geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 12000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} "
                      f"new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if iteration % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 4e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # Return a probability vector; ``evaluate_sequence`` will handle scaling.
    return best_seq.tolist()
