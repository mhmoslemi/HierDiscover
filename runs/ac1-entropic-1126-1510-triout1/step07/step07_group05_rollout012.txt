"""
Propose a step‑function (non‑negative list) with a low value of
evaluate_sequence().

Algorithm
---------
* Work on the probability simplex (sum = 1) and use an exponential
  parametrisation a = softmax(x) so that constraints are always satisfied.
* A first phase runs Adam on a smooth approximation of the max‑convolution
  (log‑sum‑exp) with a slowly increasing temperature τ (τ → ∞ gives the
  true max).  The schedule is more aggressive than the reference
  implementation (τ up to 500, longer inner loops).
* A second phase minimizes the L2 distance between the full self‑convolution
  and the perfectly flat target (1/(2 n‑1) on each entry).  The gradient of
  this quadratic loss is available in closed form via FFTs and a simple
  projected gradient descent is applied.
* Finally a classic exact sub‑gradient descent on the true ∞‑norm of the
  convolution (the objective of evaluate_sequence) cleans up remaining
  violations.
* The optimisation is repeated for several initialisations (uniform,
  Dirichlet draws of various concentrations and a warm‑start from the best
  sequence found so far).  A cascade of lengths (4000 → … → 50000) is tried
  while respecting the global time budget (≈ 1000 s).  The best sequence
  discovered is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 12000,
                            tau0: float = 0.1,
                            tau_max: float = 500.0,
                            lr0: float = 0.18) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max surrogate of the ∞‑norm
    (log‑sum‑exp with a temperature τ that grows to τ_max).
    Returns the best probability vector a (Σa=1) seen and its raw score
    2·n·max(a∗a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_raw = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- soft‑max (probability simplex) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # ----- convolution & current raw score -----
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw = 2.0 * n * cur_max
        if raw < best_raw:
            best_raw = raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_raw:.6f}")

        # ----- temperature (smooth‑max) schedule -----
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # smooth‑max weights on the convolution values
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # gradient of the smooth‑max w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_raw


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a∗a)_k.
    Returns (grad, current_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact ∞‑norm.
    Returns the refined (a, raw_score) pair.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(a)
    it = start_iter
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional tiny jitter
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_raw:.6f}")
    return best_a, best_raw


# ----------------------------------------------------------------------
def _refine_by_l2(a: np.ndarray,
                  target: np.ndarray,
                  deadline: float,
                  max_iters: int = 3000,
                  step0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the L2 loss ‖a∗a – target‖₂².
    Aims to flatten the convolution; after the loop the exact
    max‑convolution is evaluated to obtain the raw score.
    """
    n = a.shape[0]
    step = step0
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve_full(a)
        diff = conv - target
        grad = 2.0 * _fft_convolve(a[::-1], diff)[n - 1 : n - 1 + n]
        a = a - step * grad
        a = _project_onto_simplex(a)
        # simple decay of the step size
        step = step0 / (1.0 + 0.001 * it)

        cur_raw = 2.0 * n * _fft_self_convolve_max(a)
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] L2‑ref it={it}  raw={best_raw:.6f}")
    return best_a, best_raw


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point required by the challenge harness.
    Returns a list of non‑negative heights that (hopefully) gives a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # tiny safety margin

    # ------------------------------------------------------------------
    # try to reuse a previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0.0:
            prev = prev / prev.sum()
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # lengths to explore – we increase them gradually; larger n give a
    # better theoretical lower bound but also cost more time.
    lengths = [4000, 8000, 15000, 25000, 50000]

    best_seq = None
    best_raw = float("inf")
    best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        # target flat distribution for the L2‑refinement phase
        target_const = 1.0 / (2 * n - 1)
        target = np.full(2 * n - 1, target_const, dtype=np.float64)

        # ------------------------------------------------------------------
        # build a collection of start points (logits, i.e. pre‑softmax)
        init_logits = []

        # warm‑start from the current best (interpolated)
        if best_seq is not None:
            cand = _interpolate_to_length(best_seq, n)
            init_logits.append(np.log(cand + 1e-12))

        # uniform start
        init_logits.append(np.zeros(n, dtype=np.float64))

        # a few Dirichlet draws with moderate concentration
        for _ in range(2):
            alpha = np.random.rand() * 3.0 + 0.5         # α ∈ (0.5,3.5)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_logits.append(np.log(d + 1e-12))

        # a few very spiky Dirichlet draws (tiny concentration)
        for _ in range(2):
            alpha = np.random.rand() * 0.08 + 0.001    # α ∈ (0.001,0.081)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_logits.append(np.log(d + 1e-12))

        local_best_seq = None
        local_best_raw = float("inf")

        # ------------------------------------------------------------------
        for idx, x0 in enumerate(init_logits):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] start {idx+1}/{len(init_logits)} for n={n}")

            # 1) Adam optimiser on the smooth‑max surrogate
            seq, raw = _adam_optimize_smoothed(
                x0, n, deadline,
                max_iters=12000,
                tau0=0.1,
                tau_max=500.0,
                lr0=0.18
            )

            # 2) L2‑flattening phase (push the convolution towards a constant)
            seq, raw = _refine_by_l2(seq, target, deadline,
                                     max_iters=3000,
                                     step0=0.12)

            # 3) final exact max‑subgradient polishing
            seq, raw = _refine_by_subgradient(seq, deadline)

            if raw < local_best_raw:
                local_best_raw = raw
                local_best_seq = seq.copy()
                print(f"   *** new local best for n={n}: raw={local_best_raw:.6f}")

        # ------------------------------------------------------------------
        if local_best_seq is not None and local_best_raw < best_raw:
            best_raw = local_best_raw
            best_seq = local_best_seq.copy()
            best_n = n
            print(f">>> Updated global best: raw={best_raw:.6f} (length={best_n})")

    # ------------------------------------------------------------------
    # final long‑run polishing on the current global best
    if best_seq is not None:
        print("\nFinal extensive sub‑gradient refinement")
        best_seq, best_raw = _refine_by_subgradient(best_seq, deadline)
        # One more L2‑flattening pass (cheap, can improve the tail)
        target_const = 1.0 / (2 * best_seq.shape[0] - 1)
        target = np.full(2 * best_seq.shape[0] - 1, target_const)
        best_seq, best_raw = _refine_by_l2(best_seq, target, deadline,
                                           max_iters=5000,
                                           step0=0.08)
    else:
        # fallback – uniform distribution of moderate size
        n = 5000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
