assistantfinal**Solution Overview**

The objective that `evaluate_sequence` finally returns is  

\[
\text{score}=2\;n\;\frac{\max (a*a)}{(\sum a)^2},
\]

and the scaling of the vector does **not** matter – we can safely normalise all candidates to a unit‑sum.  
The problem is therefore to minimise  

\[
F(a)=2n\;\max\limits_k (a*a)_k\qquad\text{with}\qquad a_i\ge 0,\;\sum_i a_i = 1 .
\]

The code below searches this space in three stages

1. **Broad random / parametric exploration** – families that are cheap to generate (uniform,
   symmetric / asymmetric Beta, a triangular shape, a sine‑power shape, Dirichlet draws,
   mixtures of two Beta’s, etc.).  All candidates are projected onto the simplex and their
   *proxy* score `2·n·max_self_convolution` is evaluated by an FFT‑based convolution.
   The best candidate found so far is kept.

2. **Mirror‑descent (exponentiated‑gradient) refinement** – the sub‑gradient of  
   `max_k (a*a)_k` is obtained analytically (see `_max_subgradient_step`).  
   A multiplicative update `a ← a·exp(‑η·g)` keeps the vector non‑negative, and a
   re‑normalisation projects it back onto the simplex.  The step‑size decays like
   `η_t = η₀/√t`.  This optimisation is run until the time budget is exhausted.

3. **Upsampling & second‑level refinement** – the best vector is linearly interpolated
   to larger lengths (12 k – 30 k), refined again with mirror‑descent and kept if it improves
   the score.

All work is bounded by the supplied time budget (default 1000 s).  The routine prints
progress information and finally returns the best sequence as a plain Python list.
The provided `evaluate_sequence` is called only once at the very end, so the returned
score is the exact one.

"""
Propose a low‑score step function.

The algorithm works in three phases:
1️⃣  Coarse random / parametric search over several families (uniform,
    symmetric & asymmetric Beta, triangular, sine‑power, Dirichlet,
    mixtures of two Betas).  Every candidate is projected onto the simplex
    and evaluated by a fast FFT proxy `2·n·max_self_convolution`.
2️⃣  Mirror‑descent (exponentiated‑gradient) refinement using the exact
    sub‑gradient of `max_k (a*a)_k`.  The multiplicative update keeps
    non‑negativity, a final ℓ₁‑projection restores the simplex.
3️⃣  Upsample the current best to larger lengths (12 k‑30 k) and repeat the
    refinement.  The best vector is finally evaluated with the official
    `evaluate_sequence` and returned.

The whole search respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable version
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)       # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _triangular_distribution(n: int, peak: float = 0.5, width: float = 0.6) -> np.ndarray:
    """
    Piecewise‑linear “tent” distribution.
    `peak` is the centre (0‑1) and `width` the support width (0‑1].
    """
    if width <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / (n - 1)
    left = peak - width / 2.0
    right = peak + width / 2.0
    # height = max(0, 1‑|x‑peak|/(width/2))
    vals = np.maximum(0.0, 1.0 - np.abs(x - peak) * 2.0 / width)
    total = vals.sum()
    if total == 0.0:
        vals[:] = 1.0 / n
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _sine_power_distribution(n: int, power: float = 2.0) -> np.ndarray:
    """
    Symmetric distribution proportional to sin(π·x)^power,  x∈(0,1).
    For power→0 the shape approaches uniform, for large power it becomes
    a sharp peak in the centre.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.sin(np.pi * x) ** power
    total = vals.sum()
    if total == 0.0:
        vals[:] = 1.0 / n
    else:
        vals /= total
    return vals

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The value equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of f(a)=max_k (a*a)_k.
    Returns (grad, cur_max).  Grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    # tolerance for the active set
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_mirror_descent(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) refinement on the simplex.
    Returns the best sequence and its raw score found before `deadline`.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        eta = eta0 / math.sqrt(it)

        # multiplicative update; clamp exponent to avoid overflow/underflow
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.06:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror‑descent len={n} it={it} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluate_sequence score.
    The routine respects the given time budget (≈1000 s) and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Warm start from a previously discovered solution if available
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if base_seq.sum() > 0.0:
            base_seq = base_seq / base_seq.sum()
        else:
            raise ValueError
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # Global best tracking
    if base_seq is not None:
        best_seq = base_seq.copy()
    else:
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)
    best_score = _raw_score(best_seq)
    best_len = len(best_seq)
    print(f"[{time.time():.1f}] initial best raw score = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 1 – broad exploration on moderate lengths
    exploration_lengths = [2000, 4000, 8000, 12000]
    for n in exploration_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploration length {n}")

        # start from current best, interpolated to the new length
        a = _interpolate_to_length(best_seq, n)
        sc = _raw_score(a)
        if sc < best_score:
            best_seq, best_score, best_len = a, sc, n
            print(f"   interpolated improvement: score={best_score:.6f}")

        # uniform baseline
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_seq, best_score, best_len = uniform.copy(), sc, n
            print(f"   uniform new best: score={best_score:.6f}")

        # ---- systematic Beta grids (symmetric & asymmetric) ----
        # symmetric Beta (α = β) – log‑spaced grid
        alphas_sym = np.logspace(np.log10(0.12), np.log10(5.0), num=20)
        for a_val in alphas_sym:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a_val, a_val)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   sym‑Beta a={a_val:.4f} score={best_score:.6f}")

        # asymmetric Beta (α,β) – coarse grid
        alphas_asym = np.logspace(np.log10(0.12), np.log10(5.0), num=12)
        for a_val in alphas_asym:
            for b_val in alphas_asym:
                if time.time() >= deadline:
                    break
                vec = _beta_distribution(n, a_val, b_val)
                sc = _raw_score(vec)
                if sc < best_score:
                    best_seq, best_score, best_len = vec.copy(), sc, n
                    print(f"   asym‑Beta α={a_val:.3f} β={b_val:.3f} score={best_score:.6f}")

        # ---- other parametric families ----
        # triangular
        for peak in np.linspace(0.3, 0.7, 5):
            for width in np.linspace(0.2, 0.8, 5):
                if time.time() >= deadline:
                    break
                vec = _triangular_distribution(n, peak, width)
                sc = _raw_score(vec)
                if sc < best_score:
                    best_seq, best_score, best_len = vec.copy(), sc, n
                    print(f"   triangular p={peak:.2f} w={width:.2f} score={best_score:.6f}")

        # sine‑power
        for pwr in np.linspace(0.5, 4.0, 7):
            if time.time() >= deadline:
                break
            vec = _sine_power_distribution(n, power=pwr)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   sine‑power p={pwr:.2f} score={best_score:.6f}")

        # ---- stochastic families ----
        # Dirichlet draws
        for _ in range(200):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-2, 1)
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   Dirichlet conc={conc:.3f} score={best_score:.6f}")

        # Mixture of two Betas
        for _ in range(150):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mix‑2 λ={lam:.3f} score={best_score:.6f}")

    print(f"[{time.time():.1f}] best after coarse search = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent refinement on the current best length
    if time.time() < deadline:
        a_refined, val = _refine_mirror_descent(best_seq.copy(), deadline, rng, eta0=0.5)
        if val < best_score:
            best_seq, best_score = a_refined.copy(), val
            print(f"[{time.time():.1f}] after refinement: score={best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to larger lengths and refine again
    larger_lengths = [18000, 24000, 30000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)
        a, scr = _refine_mirror_descent(a, deadline, rng, eta0=0.4)
        if scr + 1e-12 < best_score:
            best_seq, best_score = a.copy(), scr
            print(f"   new best after up‑sample: score={best_score:.6f} (len={n})")
        # Early exit if we already beat the target bound
        if best_score <= 1.503:
            print("Target bound reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final exact evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
