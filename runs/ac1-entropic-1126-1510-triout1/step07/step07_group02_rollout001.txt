def refine_sequence(a, deadline):
    # a is numpy array sum=1.
    n = a.shape[0]
    best = a.copy()
    best_score = _max_convolution_score(best)
    # set parameters
    # Step 1: Adam (smooth max)
    schedule = [(0.5, 2000), (1.0, 3000), (2.0, 5000), (5.0, 5000), (10.0, 5000), (20.0, 5000)]
    # adjust step counts based on remaining time maybe.
    # after each schedule chunk, check time.

    for alpha, steps in schedule:
        if time.time() >= deadline:
            break
        # arrays for Adam
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1=0.9; beta2=0.999; eps=1e-8
        lr = 0.1
        t=0
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t+=1
            # convolution
            b = _fft_self_convolve(a)
            maxb = b.max()
            # compute softmax
            w = np.exp(alpha * (b - maxb))
            w /= w.sum()
            # gradient
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n-1:n-1+n]
            # Adam update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m/(1-beta1**t)
            v_hat = v/(1-beta2**t)
            a = a - lr*m_hat/(np.sqrt(v_hat)+eps)
            # projection
            a = _project_onto_simplex(a, 1.0)
            # ensure positivity small epsilon
            eps_val = 1e-12
            a = np.maximum(a, eps_val)
            a /= a.sum()
            # tracking best
            if t % 200 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-12:
                    best_score = cur
                    best = a.copy()
                    # print progress

    # Step 2: Subgradient descent (like previous)
    vel = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)
        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n-1:n-1+n]
        vel = beta * vel + (1 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * vel
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()
            # print

    # Step 3: Hill climbing with simulated annealing
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    # temperature schedule
    T0 = max(0.01, M*0.01)  # adapt
    while time.time() < deadline:
        it += 1
        # Determine source index: index used in max.
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find pair with max product
        best_i = i_start
        best_j = k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end+1):
            j = k - i
            prod = a[i]*a[j]
            if prod > max_prod:
                max_prod = prod
                best_i = i
                best_j = j
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n-1)
        if target == source:
            continue
        # delta scaling based on source magnitude and iteration
        max_delta = min(a[source] * 0.5, 0.02)
        # gradually reduce max_delta as time passes
        # we can compute progress fraction
        progress = (time.time() - start) / (deadline - start)
        max_delta *= (1 - progress) + 0.05  # keep small
        delta = max_delta * random.random()
        if delta <= 1e-12:
            continue
        # apply transfer; incremental
        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        # temperature
        # linear cooling
        remaining = deadline - time.time()
        total_rem = deadline - start
        T = T0 * (remaining / total_rem)
        accept = False
        if new_M < M - 1e-14:
            accept = True
        else:
            # probability
            prob = math.exp(-(new_M - M) / (T + 1e-12))
            if random.random() < prob:
                accept = True
        if accept:
            M = new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                # print improvement
        else:
            # revert
            a = a_snap
            conv = conv_snap
        # occasional jitter
        if it % 5000 == 0:
            jitter_scale = 1e-5
            a += np.random.randn(n) * jitter_scale
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    # return best

"""
Propose a (potentially) better step‑function sequence.

Algorithm
---------

1. **Candidate generation** – For several lengths we create a diverse
   set of analytic families (uniform, arcsine, triangular, exponential,
   beta, sine‑power, cosine‑window) and a few random sparse vectors.
   All vectors are projected onto the probability simplex (sum = 1).

2. **Selection** – The candidates are scored with the cheap proxy
   `2·n·max(a*a)` (exact for sum‑1 vectors) and the best few are kept.

3. **Refinement** – Each kept candidate is refined within its own
   time‑budget using three phases:

   * *Adam on a smooth approximation* – a soft‑max with a growing
     temperature ``α`` yields a smooth surrogate of the max‑convolution.
     Projected Adam steps are taken until the allocated time expires.

   * *Projected sub‑gradient* – the true max‑sub‑gradient is computed
     (active set of binding convolution entries) and a classic
     diminishing‑step sub‑gradient method with momentum is applied.

   * *Simulated‑annealing hill‑climber* – a mass transfer ``δ`` from a
     source index (one of the indices that dominate the current
     convolution peak) to a random target index is attempted.
     The convolution is updated *incrementally* (O(n)) and the move is
     accepted either if it improves the peak or with a temperature‑dependent
     probability.  Periodic jitter helps escape plateaus.

4. **Best overall** – The best refined sequence from all candidates is
   returned as a plain Python list.

All operations respect the 1000 s deadline; if the time runs out the
function returns the best solution found so far (never ``inf``).
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Exact objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # update convolution: remove contribution of old a[i], add contribution of new a[i]
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _analytic_candidates(n):
    """Return a list of analytic probability vectors of length n."""
    eps = 1e-12
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    cand.append(vals)

    # triangular
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    tri = np.concatenate([left, right])
    if tri.size != n:
        tri = np.pad(tri, (0, n - tri.size), constant_values=0)
    tri /= tri.sum()
    cand.append(tri)

    # exponential (two lambdas)
    for lam in (0.003, 0.01):
        idx = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        vals = np.exp(-lam * np.abs(idx - centre))
        vals /= vals.sum()
        cand.append(vals)

    # symmetric beta
    for p in (0.2, 0.5, 1.0, 2.0, 3.0):
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
        if w.sum() == 0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w.sum()
        cand.append(w)

    # sine‑power windows
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
        idx = np.arange(n, dtype=np.float64)
        vals = np.sin(np.pi * (idx + 0.5) / n) ** pw
        vals = np.maximum(vals, 0)
        vals /= vals.sum()
        cand.append(vals)

    # cosine windows (Hann‑like)
    for c in (0.5, 0.46, 0.42):
        if n == 1:
            vals = np.array([1.0])
        else:
            i = np.arange(n, dtype=np.float64)
            vals = c - c * np.cos(2.0 * np.pi * i / (n - 1))
        vals = np.maximum(vals, 0)
        vals /= vals.sum()
        cand.append(vals)

    # a random sparse start
    sparse = np.zeros(n, dtype=np.float64)
    k = max(2, n // 200)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # ensure strict positivity
    cand = [np.maximum(v, eps) / np.sum(np.maximum(v, eps)) for v in cand]
    return cand


# ----------------------------------------------------------------------
def _random_sparse_candidate(n, eps=1e-12):
    """Generate a random sparse vector of length n (sum = 1)."""
    k = max(2, n // 250)                     # a few hundred non‑zeros at most
    idx = np.random.choice(n, k, replace=False)
    vals = np.random.rand(k) + eps
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _refine_sequence(a, start_time, deadline):
    """
    Refine a probability vector a (sum = 1) using Adam → sub‑gradient → simulated‑annealing.
    Returns the best vector found and its objective value (2·n·max).
    """
    n = a.shape[0]
    eps = 1e-12
    best = a.copy()
    best_score = _max_convolution_score(best)

    # --------------------------------------------------------------
    # Phase 1 – Adam on a soft‑max surrogate of the max‑convolution
    schedule = [
        (0.5, 2000), (1.0, 2000), (2.0, 3000),
        (5.0, 3000), (20.0, 3000), (100.0, 2000),
        (500.0, 2000), (2000.0, 1000)
    ]
    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.13
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1
            # full self‑convolution
            b = _fft_self_convolve(a)
            maxb = b.max()
            # smooth max via soft‑max
            w = np.exp(alpha * (b - maxb))
            w /= w.sum()
            # gradient of the smooth max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # simplex projection + tiny positivity guard
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if t % 200 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-12:
                    best_score = cur
                    best = a.copy()
                    print(f"[{time.time():.1f}] Adam {t} best {cur:.6f}")

            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 2 – exact max‑sub‑gradient descent with momentum
    vel = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        vel = beta * vel + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * vel
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best = a.copy()
            print(f"[{time.time():.1f}] Subgrad iter={it} best {cur:.6f}")

        # occasional top‑level checkpoint
        if it % 5000 == 0 and time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Phase 3 – hill‑climbing with simulated annealing (incremental updates)
    a = best.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    # initial temperature proportional to current peak
    T0 = max(M * 0.02, 1e-7)

    while time.time() < deadline:
        it += 1
        # locate the shift index where current max occurs
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) inside this shift with the largest product a[i]*a[j]
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # delta size – start relatively large and shrink with elapsed time
        progress = (time.time() - start_time) / (deadline - start_time)
        max_delta = min(a[source] * 0.5, 0.02) * (1.0 - progress + 0.05)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        # keep snapshots for possible revert
        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        # simulated‑annealing acceptance
        remaining = deadline - time.time()
        total_rem = deadline - start_time
        T = T0 * (remaining / total_rem)
        accept = False
        if new_M < M - 1e-14:
            accept = True
        else:
            prob = math.exp(-(new_M - M) / (T + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            M = new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time():.1f}] hill_iter={it} new best {cur_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap

        # jitter if we have been stagnating for long
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    final_score = 2.0 * n * conv.max()
    print(f"[{time.time():.1f}] Finished length={n} final ≈ {final_score:.6f}")
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The search combines analytic seeds, random
    sparse starts, a multi‑phase refinement and respects the 1000 s
    time limit.
    """
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0    # a small safety margin

    # --------------------------------------------------------------
    # 1. Generate a pool of seeds over several lengths
    lengths = [800, 1200, 2000, 3000, 5000, 8000, 12000]
    pool = []
    best_overall_score = float('inf')
    best_overall_seq = None

    for n in lengths:
        if time.time() >= deadline:
            break
        # Analytic families
        for vec in _analytic_candidates(n):
            if time.time() >= deadline:
                break
            sc = _max_convolution_score(vec)
            pool.append((sc, vec.copy()))
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall_seq = vec.copy()
                print(f"[{time.time():.1f}] init n={n} analytic best {sc:.6f}")

        # Random sparse vectors (3 per length)
        for _ in range(3):
            if time.time() >= deadline:
                break
            vec = _random_sparse_candidate(n)
            sc = _max_convolution_score(vec)
            pool.append((sc, vec.copy()))
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall_seq = vec.copy()
                print(f"[{time.time():.1f}] init n={n} random best {sc:.6f}")

    # --------------------------------------------------------------
    # 2. Include the previously known best, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            pool.append((sc, prev.copy()))
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall_seq = prev.copy()
                print(f"[{time.time():.1f}] imported previous best {sc:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Keep only the top K seeds (K = 8) to spend refinement time on
    pool.sort(key=lambda x: x[0])
    top_k = 8
    seeds = [vec for _, vec in pool[:top_k]]
    if not seeds:
        # fallback – uniform distribution of a moderate size
        n = 3000
        seq = np.full(n, 1.0 / n, dtype=np.float64)
        seeds = [seq]

    # --------------------------------------------------------------
    # 4. Refine each seed in turn, sharing the remaining time budget
    while time.time() < deadline and seeds:
        # allocate a slice of the remaining budget to the next seed
        remaining = deadline - time.time()
        if remaining <= 2.0:
            break
        # give at most 30 s per seed, but also ensure we have enough for the rest
        per_seed = min(30.0, remaining / len(seeds))
        seq = seeds.pop(0)
        sub_deadline = time.time() + per_seed
        refined, refined_score = _refine_sequence(seq, start, sub_deadline)
        if refined_score < best_overall_score:
            best_overall_score = refined_score
            best_overall_seq = refined.copy()
            print(f"[{time.time():.1f}] New global best {refined_score:.6f}")

    # --------------------------------------------------------------
    # 5. Return the best sequence found (as plain Python list)
    if best_overall_seq is None:
        # extremely unlikely – return a tiny uniform vector
        n = 1000
        best_overall_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # The evaluator clips negatives and rescales out‑of‑bounds values,
    # but we keep everything in a safe range.
    best_overall_seq = np.maximum(best_overall_seq, 0.0)
    best_overall_seq /= best_overall_seq.sum()   # ensure sum = 1 (any scaling works)
    return best_overall_seq.tolist()
