    contributes most to the current maximal convolution entry to a
    random target.  Moves that lower the maximal entry are kept; occasional
    jitter is added to escape plateaux.

All work is stopped well before the 1000 s budget; the best sequence
found is returned as a plain list of ``float``.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using real FFT."""
    size = 1 << ((a.size + b.size - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    if p <= 0:
        p = 1e-6
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma_ratio=0.3):
    """Discrete Gaussian centred at (n‑1)/2."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    sigma = sigma_ratio * n
    vals = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_left_exponential(n, lam):
    """Mass decays from the left end."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    # linear part (old a)
    a_old = a.copy()
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (finite vector) that minimises
    ``evaluate_sequence``.  The method combines a broad analytic sweep,
    L2‑flattening, Adam sub‑gradient descent and an O(n) hill‑climber.
    The function returns the best sequence found before the time budget
    expires.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad sampling over several lengths and families
    lengths = [1500, 2000, 3000, 4000, 6000, 8000, 12000]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families
        candidates = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.003),
            _distribution_exponential(n, lam=0.01),
        ]

        # beta family (symmetric)
        for p in np.logspace(-0.5, 1.0, num=8):          # ~0.3 … 10
            candidates.append(_distribution_beta_sym(n, p))

        # sine‑power windows
        for pw in np.linspace(0.2, 10.0, num=9):
            candidates.append(_distribution_sine_power(n, pw))

        # cosine windows
        for c in np.linspace(0.3, 0.8, num=6):
            candidates.append(_distribution_cosine_window(n, c))

        # Gaussian shapes
        for sr in np.linspace(0.1, 0.5, num=5):
            candidates.append(_distribution_gaussian(n, sigma_ratio=sr))

        # left‑hand exponential
        for lam in np.logspace(-4, -1, num=6):
            candidates.append(_distribution_left_exponential(n, lam))

        # random sparse vector
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 250)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        # a few pure Dirichlet draws (alpha = 0.5 gives moderate sparsity)
        for _ in range(3):
            rand = np.random.gamma(0.5, 1.0, size=n)
            rand /= rand.sum()
            candidates.append(rand)

        # evaluate all candidates
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Fold in any previously discovered sequence (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # Fallback (should never happen)
    if best_seq is None:
        n = 3000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. L2‑flattening (drive convolution towards the constant 1/(2n‑1))
    a = best_seq.copy()
    n = a.shape[0]
    target_val = 1.0 / (2 * n - 1)
    lr = 0.25                     # initial learning rate for L2 step
    it = 0
    while time.time() < deadline:
        # simple schedule: decay learning rate slowly
        it += 1
        if it % 400 == 0:
            lr *= 0.85
        conv = _fft_self_convolve(a)
        diff = conv - target_val
        grad = 2.0 * _fft_convolve(a[::-1], diff)[n - 1:n - 1 + n]
        a = a - lr * grad
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()
        if it % 200 == 0:
            cur = _max_convolution_score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] L2 iter={it} new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 4. Adam sub‑gradient descent on the true max‑convolution
    a = best_seq.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr_adam = 0.12
    t = 0
    while time.time() < deadline:
        t += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)
        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a = a - lr_adam * m_hat / (np.sqrt(v_hat) + adam_eps)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        if t % 250 == 0:
            cur = _max_convolution_score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

        # early exit check
        if time.time() + 10.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # 5. Targeted hill‑climbing using O(n) incremental updates
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    it_no_improve = 0
    max_jitter = 1e-5

    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())                          # index of current maximal value
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) that contributes most to this shift
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue

        # propose a modest transfer
        max_delta = min(a[source] * 0.5, 0.025)
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snapshot = a.copy()
        conv_snapshot = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-13:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            best_n = n
            it_no_improve = 0
            if it % 100 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {best_score:.6f}")
        else:
            # reject move
            a = a_snapshot
            conv = conv_snapshot
            it_no_improve += 1

        # occasional jitter to escape stagnation
        if it % 2000 == 0 and it_no_improve > 500:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * best_n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_seq.tolist()
