```
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999
    eps=1e-12
    best_a = a.copy()
    best_score = raw_score(a)
    lam = lam_start
    for it in range(1, max_iters+1):
        if time.time() >= deadline: break
        # maybe symmetrize
        if symmetrize:
            a = _symmetrize(a)
        # compute conv
        conv = _fft_self_convolve_full(a)
        # evaluate raw:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute softmax p
        tmp = lam * conv
        max_tmp = tmp.max()
        exp_tmp = np.exp(tmp - max_tmp)
        p = exp_tmp / exp_tmp.sum()
        # gradient
        grad = 2.0 * _fft_convolve(a[::-1], p)[n-1: n-1 + n]  # if symmetrized, ok
        # Adam update on a (not on logits)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 * (np.sqrt(1 - beta2**it) / (1 - beta1**it))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex:
        a = _project_onto_simplex(a)
        # increase lambda

        # occasional random jitter
        if it % 1000 == 0:
            a += np.random.randn(*a.shape) * 0.001
            a = _project_onto_simplex(a)
    return best_a, best_score
```

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{(\sum_i a_i)^2},
\]

where `a` is a non‑negative step‑function (the sum is normalised to 1).  
`a*a` denotes the linear convolution of `a` with itself.  
A low score is achieved when the self‑convolution is as flat as possible.

The programme below implements a **two‑phase optimisation**:

1. **Smoothed‑max Adam optimisation** – the non‑smooth `max` is replaced by a
   log‑sum‑exp with a temperature λ.  λ is increased gradually up to a large
   value (≈ 200) so the optimiser eventually works on the true maximum.
   The variables are the heights themselves (not logits); after each Adam step
   the vector is projected back onto the probability simplex.  
   Symmetry (`a[i]=a[n‑1‑i]`) is enforced after every update – symmetric
   sequences are known to give flatter autocorrelations.

2. **Projected sub‑gradient refinement** – starting from the best vector found
   by Adam, a classic sub‑gradient method on the true `max` is run until the
   time budget expires.  The step size follows a \(c/\sqrt{t}\) schedule and
   the iterate is repeatedly projected onto the simplex.

The optimiser is run for several lengths (`n = 4000, 8000, 12000, 20000`),
always warm‑starting the next length by linear interpolation of the current
best sequence.  Multiple initialisations (uniform, arcsine‑shaped,
random Dirichlet draws and the warm‑start) are tried; the best result over all
lengths is returned.

The code respects the 1000 s budget, prints progress and finally evaluates the
sequence with the supplied `evaluate_sequence` function.

```python
"""
Optimise a non‑negative step function to minimise the
evaluate_sequence() objective.

The algorithm:
  * Optimises a probability vector a (Σa=1, a_i≥0) directly with Adam,
    using a smoothed max (log‑sum‑exp) whose temperature λ is increased
    up to a large value (≈200).  After every update the vector is projected
    onto the simplex and symmetrised (a_i = a_{n‑1‑i}).

  * The best candidate for a given length n is refined with a second Adam
    run that keeps λ fixed at the maximal value and uses a smaller learning
    rate.

  * Finally a projected sub‑gradient descent on the true max is performed
    while time permits.

  * Several lengths are tried (4000 → 20000) and warm‑started by linear
    interpolation.  Multiple start vectors (uniform, arcsine, random Dirichlet,
    warm‑start) are used.  The best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i} (preserves the sum)."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (edges heavier, centre lighter)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _adam_optimize_coeffs(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 5000,
                          lr0: float = 0.08,
                          lam_start: float = 0.5,
                          lam_factor: float = 1.005,
                          lam_max: float = 200.0,
                          symmetrize: bool = True) -> tuple[np.ndarray, float]:
    """
    Adam optimisation directly on the coefficient vector a (∑a=1, a≥0).

    A smoothed max (log‑sum‑exp) with temperature λ is minimised; λ is
    increased each iteration up to lam_max.  After every update the vector
    is projected onto the simplex and optionally symmetrised.
    """
    a = np.array(a0, dtype=np.float64)
    a = _project_onto_simplex(a)               # ensure feasibility
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    lam = lam_start
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_full(a).max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        if symmetrize:
            a = _symmetrize(a)

        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.3f}")

        # ----- smooth max -----
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()                     # shape (2n‑1,)

        # ----- gradient of the smooth objective w.r.t. a -----
        # because a is symmetric we could use a directly,
        # but the generic formula works for any a.
        grad = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # ----- Adam update on a (followed by simplex projection) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_onto_simplex(a)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_max)

        # occasional jitter to shake out of plateaus
        if it % 1000 == 0:
            a += np.random.randn(*a.shape) * 0.001
            a = _project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length(n: int,
                    deadline: float,
                    prev_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n.
    Several initial vectors are tried; each undergoes a two‑stage Adam.
    The best result (lowest raw score) is returned.
    """
    init_vecs = []

    # warm‑start (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        init_vecs.append(cand)

    # uniform
    init_vecs.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine shape
    init_vecs.append(_arcsine_start(n))

    # three random Dirichlet draws
    for _ in range(3):
        init_vecs.append(_random_dirichlet_start(n))

    best_seq = None
    best_score = float("inf")

    for idx, a0 in enumerate(init_vecs):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n} start {idx + 1}/{len(init_vecs)}")

        # ---- Stage 1 : increase λ gradually ----
        a_stage1, sc1 = _adam_optimize_coeffs(
            a0,
            n,
            deadline,
            max_iters=5000,
            lr0=0.08,
            lam_start=0.5,
            lam_factor=1.005,
            lam_max=200.0,
            symmetrize=True,
        )
        # ---- Stage 2 : keep λ fixed at max, smaller steps ----
        a_stage2, sc2 = _adam_optimize_coeffs(
            a_stage1,
            n,
            deadline,
            max_iters=3000,
            lr0=0.04,
            lam_start=200.0,
            lam_factor=1.0,
            lam_max=200.0,
            symmetrize=True,
        )

        # keep the best of the two stages
        if sc2 < best_score:
            best_score = sc2
            best_seq = a_stage2.copy()
        elif sc1 < best_score:
            best_score = sc1
            best_seq = a_stage1.copy()

    # fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_seq).max()

    return best_seq, best_score
```
