"""
Propose a non‑negative step‑function (probability vector) that minimises

        evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)²)

The routine follows three phases:

1. **Broad sampling** – many analytic families (arcsine, beta, power‑law,
   triangular, exponential, sine, cosine, uniform) and random Dirichlet draws
   are generated for several lengths around the previously best size.
   Mixed distributions (arcsine + uniform) are also examined.
   The cheap objective 2·n·max(conv) (exact when the sum is 1) is used for
   fast ranking.

2. **Smooth annealed gradient descent** – for the top few seeds we minimise
   a smooth approximation of the max, namely the log‑sum‑exp of the
   convolution values.  The gradient is obtained analytically via FFT‑based
   convolution; a schedule for the temperature β slowly drives the smooth
   objective towards the true max.  The iterate is projected onto the simplex
   after each step (KL‑mirror‑descent would also work, but Euclidean projection
   is simpler and fast).

3. **Fine‑grained hill‑climb** – tiny random mass transfers and occasional
   jitter finish the optimisation.  The whole process respects the 1000 s
   budget and always returns a valid list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap exact score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # w ∝ x^{p-1} (1-x)^{p-1}
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _generalized_arcsine_distribution(n: int, p: float) -> np.ndarray:
    """(i+0.5)^{-p}·(n-i-0.5)^{-p}  –  p=0.5 gives the classical arcsine."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / ((i + 0.5) * (n - i - 0.5)) ** p
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
def _mix_two_distributions(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mix α·a + (1‑α)·b, returned normalised to sum = 1."""
    mixed = alpha * a + (1.0 - alpha) * b
    s = mixed.sum()
    if s == 0:
        return np.full_like(mixed, 1.0 / mixed.size)
    return mixed / s


# ----------------------------------------------------------------------
def _smooth_gradient_descent(a0: np.ndarray, deadline: float,
                             max_iters: int = 200_000,
                             eta0: float = 0.7,
                             beta_start: float = 0.5,
                             beta_end: float = 30.0):
    """
    Annealed gradient descent on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector seen and its cheap score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # temperature schedule (linear in iteration index)
        beta = beta_start + (beta_end - beta_start) * (it - 1) / (max_iters - 1)

        # self‑convolution and soft‑max weights
        b = _fft_self_convolve(a)
        cur_max = b.max()
        # shift by cur_max for numerical stability
        w = np.exp(beta * (b - cur_max))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth objective: 2·(a_rev * w)  (no diagonal fix)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best_a = cur_score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8):
    """Exponentiated‑gradient (mirror descent with KL) on the exact max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # exact sub‑gradient (the same routine as _subgrad below)
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.2,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam optimiser on the exact non‑smooth max objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(conv(a,a)) on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (each diagonal term counted twice in the above sum)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights minimising evaluate_sequence.
    The routine respects the 1000 s budget and always returns a valid list.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # a small safety margin

    # ------------------------------------------------------------------
    # 1) decide the base length (use a previously found construction if present)
    try:
        base_n = len(height_sequence_1)       # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # Lengths to explore (around base_n, plus a few random ones)
    lengths = set()
    for f in (0.9, 0.95, 1.0, 1.05, 1.1):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(3):
        n = np.random.randint(500, min(8000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 2) broad sampling of analytic families and random draws
    candidates = []   # list of (array, cheap_score)

    # fine grid of shape parameters around the known optimum p≈0.5
    p_grid = np.concatenate([
        np.linspace(0.10, 0.40, 7),          # coarse low‑p
        np.linspace(0.45, 0.55, 11),         # dense around 0.5
        np.linspace(0.60, 1.20, 13)          # higher p
    ])

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----------------- deterministic families -----------------
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_exponential_distribution, "exponential"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_uniform_distribution, "uniform")
        ]
        for gen, name in fams:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} {name:10s} score={score:.6f}")

        # ----------------- beta‑symmetric with many p’s -----------------
        for p in p_grid:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} beta(p={p:4.2f}) score={score:.6f}")

        # ----------------- generalized arcsine (product power law) -----
        for p in p_grid:
            a = _generalized_arcsine_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} gen‑arc(p={p:4.2f}) score={score:.6f}")

        # ----------------- mixtures of arcsine & uniform -------------
        arcs = _arcsine_distribution(n)
        unif = _uniform_distribution(n)
        for alpha in np.linspace(0.0, 1.0, 11):
            a = _mix_two_distributions(arcs, unif, alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} mixArcsUnif(a={alpha:4.2f}) score={score:.6f}")

        # ----------------- random Dirichlet draws --------------------
        for _ in range(3):
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} random Dirichlet score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) add the previously best sequence (if it exists)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4) keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 12
    seeds = [c[0] for c in candidates[:top_k]]
    best_seq = seeds[0].copy()
    best_score = _max_convolution_score(best_seq)
    print(f"Selected {len(seeds)} seeds; best cheap score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) heavy refinement (smooth annealed GD + Adam)
    for idx, seed_seq in enumerate(seeds):
        if time.time() >= deadline:
            break

        # smooth annealed gradient descent
        seq, sc = _smooth_gradient_descent(seed_seq, deadline,
                                            max_iters=120_000,
                                            eta0=0.7,
                                            beta_start=0.5,
                                            beta_end=30.0)
        if sc < best_score - 1e-14:
            best_seq, best_score = seq.copy(), sc
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} smooth GD improved to {sc:.6f}")

        # a quick Adam polishing (helps when we are already close)
        seq2, sc2 = _adam_descent(best_seq, deadline,
                                  max_iters=60_000,
                                  step0=0.15)
        if sc2 < best_score - 1e-14:
            best_seq, best_score = seq2.copy(), sc2
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} Adam polished to {sc2:.6f}")

    # ------------------------------------------------------------------
    # 6) final hill‑climbing with tiny random transfers
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.025                # start with up to 2.5 % of total mass

    while time.time() < deadline:
        it += 1
        # geometrically decaying transfer amount
        transfer = max_transfer * (0.55 ** (it / 8000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_seq = a.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to hop over flat plateaus
        if it % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 4e-5
            a = _project_onto_simplex(best_seq + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # Return the probability vector (scale‑invariant for evaluate_sequence)
    return best_seq.tolist()
