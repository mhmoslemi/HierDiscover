"""
Propose a low‑score non‑negative step function.

Algorithm
---------
* Choose a problem size n (the length of the step‑function).  If a
  previous solution exists we keep its length, otherwise n≈4000.
* Build a modest population (≈30) of random Dirichlet vectors with a
  variety of concentration parameters and, if available, the previous
  best solution.
* Run a simple genetic algorithm: tournament selection, convex‑
  combination crossover, Gaussian mutation, and projection onto the
  probability simplex.  The fitness is the surrogate objective
  ``2·n·max(a*a)`` (exact when ∑a=1).  The GA quickly explores many
  diverse shapes.
* Occasionally polish promising offspring with a few steps of
  exponentiated‑gradient descent.
* After the GA finishes (or the time budget is almost exhausted) run
  a longer mirror‑descent phase on the current best candidate using
  Adam and a KL‑gradient method.
* Finish with a Frank‑Wolfe style mass‑transfer refinement and a very
  cheap hill‑climbing loop.
* Return the best sequence found (all entries ≥0 and summed to 1).

The whole routine respects a 1000 s budget, prints progress, and
produces a list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(a*a) on the simplex (∑a=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)      # (a_rev * w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]   # central n entries

    # diagonal correction for even active indices
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a point on the simplex from Dirichlet(alpha,…,alpha)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200000,
                         eta0: float = 0.5):
    """Exponentiated‑gradient (KL‑mirror) descent on the simplex."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score, best = cur, a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200000,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style adaptive gradient descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score, best = cur, a.copy()
    return best, best_score

        pass

    # fill the rest with Dirichlet draws of varying concentration
    alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    while len(population) < popsize:
        a = _random_dirichlet_alpha(n, np.random.choice(alphas))
        population.append(a)

    # ------------------------------------------------------------------
    # 3. Evaluate the initial population
    scores = np.array([_max_convolution_score(a) for a in population])
    best_idx = int(scores.argmin())
    best_seq = population[best_idx].copy()
    best_score = float(scores[best_idx])
    print(f"[{time.time():.1f}] initial best score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 4. Simple genetic algorithm
    def pick_parent():
        """Tournament selection (size 2)."""
        i, j = np.random.randint(popsize, size=2)
        return population[i] if scores[i] < scores[j] else population[j]

    ga_iter = 0
    while time.time() < deadline - 5.0:
        ga_iter += 1

        # crossover
        p1 = pick_parent()
        p2 = pick_parent()
        lam = np.random.rand()
        child = lam * p1 + (1.0 - lam) * p2

        # mutation (small Gaussian jitter)
        if np.random.rand() < 0.8:
            sigma = 0.02 / np.sqrt(n)          # scale with problem size
            child += np.random.randn(n) * sigma

        child = _project_onto_simplex(child, s=1.0)
        child_score = _max_convolution_score(child)

        # replace the worst individual if improvement
        worst = int(scores.argmax())
        if child_score < scores[worst]:
            population[worst] = child
            scores[worst] = child_score
            if child_score < best_score:
                best_score, best_seq = child_score, child.copy()
                print(f"[{time.time():.1f}] GA iter {ga_iter} new best {best_score:.6f}")

        # Occasionally polish a promising child with a few exp‑grad steps
        if child_score < best_score * 1.05 and time.time() < deadline - 1.0:
            refined, refined_score = _exp_gradient_descent(child, deadline,
                                                           max_iters=3000,
                                                           eta0=0.8)
            if refined_score < best_score - 1e-14:
                best_score, best_seq = refined_score, refined.copy()
                # replace the same slot we just overwrote (or the worst)
                population[worst] = refined
                scores[worst] = refined_score
                print(f"[{time.time():.1f}] GA‑polish iter {ga_iter} improved to {best_score:.6f}")

        # periodic status report
        if ga_iter % 500 == 0:
            print(f"[{time.time():.1f}] GA iter {ga_iter}, best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Intensive mirror‑descent refinements on the current best
    remaining = deadline - time.time()
    if remaining > 10.0:
        # split remaining time roughly half/half
        split = time.time() + remaining / 2.0

        best_seq, best_score = _adam_descent(best_seq, split,
                                             max_iters=500000,
                                             step0=0.2)
        print(f"[{time.time():.1f}] after Adam best {best_score:.6f}")

        best_seq, best_score = _exp_gradient_descent(best_seq, deadline,
                                                     max_iters=500000,
                                                     eta0=0.8)
        print(f"[{time.time():.1f}] after ExpGrad best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Frank‑Wolfe style mass transfer with a tiny line‑search
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline - 5.0 and fw_iter < 60000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # most responsible for the max
        i_dst = int(np.argmin(grad))   # best place to receive mass
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # evaluate a few trial transfer amounts
        candidates = [max_transfer, max_transfer * 0.5, max_transfer * 0.25, max_transfer * 0.75]
        best_local = cur_score
        best_vec = a
        for d in candidates:
            tmp = a.copy()
            tmp[i_src] -= d
            tmp[i_dst] += d
            sc = _max_convolution_score(tmp)
            if sc < best_local:
                best_local, best_vec = sc, tmp.copy()
        if best_local < cur_score - 1e-13:
            a = best_vec
            cur_score = best_local
            if cur_score < best_score - 1e-13:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] FW iter {fw_iter} new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Very cheap hill‑climbing moves (random pair swaps) if time permits
    while time.time() < deadline - 0.5:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_transfer = min(best_seq[i], 1.0 - best_seq[j])
        if max_transfer < 1e-12:
            continue
        delta = max_transfer * np.random.rand()
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand.copy()
            print(f"[{time.time():.1f}] random move improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
