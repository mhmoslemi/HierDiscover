Pseudo:

```
def optimize_distribution(a, deadline):
    best_seq = a.copy()
    best_score = objective(a)
    # schedule for smoothing
    schedule = [(1, 2000), (5, 2000), (20, 2000), (100, 2000), (500, 2000), (2000, 2000), (5000, 2000), (10000, 2000)]
    # Adam hyperparameters
    beta1=0.9; beta2=0.999; eps=1e-8; lr=0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t=0
    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t+=1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            w /= w_sum
            conv_rev = _fft_convolve(a[::-1], w)   # gives length len(a)+len(w)-1
            grad = 2 * conv_rev[len(a)-1:len(a)-1+len(a)]    # size n
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad*grad)
            m_hat = m/(1-beta1**t)
            v_hat = v/(1-beta2**t)
            a = a - lr * m_hat/(np.sqrt(v_hat)+eps)
            a = _project_onto_simplex(a, 1.0)
            # evaluate occasionally
            if t % 100 == 0:
                cur = objective(a)
                if cur < best_score:
                    best_score = cur; best_seq = a.copy()
    # after smoothed steps, do subgrad as earlier
    # ...
    return best_seq, best_score
```

"""
A multi‑stage stochastic optimizer for the max‑self‑convolution problem.

The algorithm follows these steps:

1. **Broad initialisation** – generate a large portfolio of analytic
   families (uniform, arcsine, beta, exponential, block spikes,
   alternating patterns) together with a few random Dirichlet draws.

2. **Quick Adam‑run** – each candidate is briefly refined with a
   smoothed‑max (log‑sum‑exp) Adam optimiser.  This yields a coarse
   ranking of lengths and shapes while staying fast.

3. **Selection** – the three best (lowest) candidates from the quick
   phase are taken as seeds for a deeper optimisation.

4. **Deep optimisation** – for each seed we run a long Adam schedule,
   an exact sub‑gradient descent, a greedy active‑set swap routine and
   finally a short simulated‑annealing fine‑tuner.

5. **Final output** – the best vector found (non‑negative, sum = 1)
   is returned as a plain Python list.  All steps respect the global
   time budget (default 1000 s).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective for a vector normalized to sum = 1:
       2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _initial_candidates_expanded(n: int) -> list[np.ndarray]:
    """Generate a richer set of plausible start vectors for length n."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # symmetric beta – dense sweep on p (log‑spaced)
    for p in np.logspace(-2, 0.7, num=12):  # approx 0.01 → 5
        cand.append(_distribution_beta_sym(n, float(p)))

    # exponentials – wide lam range
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential(n, lam))

    # alternating spikes – several strides
    for stride in (2, 3, 4, 5, 6, 8, 10):
        cand.append(_distribution_alternating(n, stride))

    # two‑block spikes – varied fractions
    for bf in (0.03, 0.05, 0.07, 0.10, 0.15, 0.20):
        cand.append(_distribution_two_blocks(n, block_frac=bf))

    # random sparse Dirichlet draws (different concentration parameters)
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0):
        rnd = np.random.gamma(shape=alpha, scale=1.0, size=n)
        if rnd.sum() == 0:
            continue
        rnd /= rnd.sum()
        cand.append(rnd)

    return cand


# ----------------------------------------------------------------------
# Adam optimiser on the log‑sum‑exp smooth max -------------------------
def _adam_optimize(a: np.ndarray,
                   deadline: float,
                   schedule: list[tuple[float, int]],
                   lr: float = 0.12,
                   eval_interval: int = 150) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best vector (sum = 1) and its objective value.
    """
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    best_seq = a.copy()
    best_score = _objective(a)

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights (log‑sum‑exp)
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full_like(w, 1.0 / w.size)

            # gradient of the smooth max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % eval_interval == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _exact_subgradient_descent(a: np.ndarray,
                               deadline: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the true max‑convolution."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        # early exit if already under target
        if best_score <= 1.503:
            break

        if time.time() + 0.5 >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _greedy_swap_opt(a: np.ndarray,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Greedy transfers from high‑gradient to low‑gradient coordinates."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]
        j = int(np.argmin(grad))

        if i == j:
            break

        delta = min(a[i] * 0.5, 0.005)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")
        else:
            # try a smaller transfer
            delta *= 0.25
            if delta < 1e-7:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] tiny swap improved best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Lightweight Metropolis walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    T = 0.04
    iter_cnt = 0

    while time.time() < deadline and T > 1e-4:
        iter_cnt += 1

        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter_cnt} new best {best_score:.6f}")

        if iter_cnt % 2500 == 0:
            T *= 0.85

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to explore (bigger lengths give more freedom)
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 8000, 10000]

    overall_best_seq = None
    overall_best_score = float("inf")
    best_per_length = {}

    # ------------------------------------------------------------------
    # 2. QUICK SCAN: generate a modest number of candidates per length
    #    and run a brief Adam optimisation to obtain a ranking.
    quick_schedule = [
        (1.0, 200), (5.0, 200), (20.0, 200),
        (100.0, 200), (500.0, 200), (2000.0, 200)
    ]

    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _initial_candidates_expanded(n)

        # add a few extra random Dirichlet draws
        for _ in range(3):
            alpha = np.random.rand() * 2 + 0.05
            rnd = np.random.gamma(shape=alpha, scale=1.0, size=n)
            if rnd.sum() == 0:
                continue
            rnd /= rnd.sum()
            candidates.append(rnd)

        for cand in candidates:
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            cand_opt, cand_score = _adam_optimize(cand, deadline,
                                                  schedule=quick_schedule,
                                                  lr=0.15,
                                                  eval_interval=100)
            # keep global best
            if cand_score < overall_best_score:
                overall_best_score = cand_score
                overall_best_seq = cand_opt.copy()
                print(f"[{time.time():.1f}] quick n={n} score={cand_score:.6f}")

            # remember the best for this length
            if n not in best_per_length or cand_score < best_per_length[n][1]:
                best_per_length[n] = (cand_opt, cand_score)

    # ------------------------------------------------------------------
    # 3. Pick the most promising lengths (up to three)
    top_lengths = sorted(best_per_length.items(),
                         key=lambda kv: kv[1][1])[:3]

    # ------------------------------------------------------------------
    # 4. DEEP REFINEMENT on the selected seeds
    deep_schedule = [
        (1.0, 4000), (5.0, 4000), (20.0, 4000),
        (100.0, 4000), (500.0, 4000), (2000.0, 4000),
        (5000.0, 4000), (10000.0, 4000)
    ]

    for n, (seed_vec, seed_score) in top_lengths:
        if time.time() >= deadline:
            break
        a = seed_vec.copy()
        # long Adam pass
        a, overall_best_score = _adam_optimize(a, deadline,
                                                schedule=deep_schedule,
                                                lr=0.12,
                                                eval_interval=300)
        # exact sub‑gradient descent
        a, overall_best_score = _exact_subgradient_descent(a, deadline,
                                                            best_seq=a,
                                                            best_score=overall_best_score)
        # greedy swaps
        a, overall_best_score = _greedy_swap_opt(a, deadline,
                                                  best_seq=a,
                                                  best_score=overall_best_score)
        # simulated annealing fine‑tune
        a, overall_best_score = _simulated_annealing(a, deadline,
                                                      best_seq=a,
                                                      best_score=overall_best_score)

        if overall_best_score < overall_best_score - 1e-12:
            overall_best_seq = a.copy()
            print(f"[{time.time():.1f}] new overall best after deep n={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final brief annealing if time remains
    if overall_best_seq is not None and time.time() < deadline:
        a, overall_best_score = _simulated_annealing(overall_best_seq.copy(),
                                                     deadline,
                                                     best_seq=overall_best_seq,
                                                     best_score=overall_best_score)
        if overall_best_score < overall_best_score - 1e-12:
            overall_best_seq = a.copy()

    # ------------------------------------------------------------------
    final_len = len(overall_best_seq) if overall_best_seq is not None else 0
    print(f"Final best score: {overall_best_score:.6f} (length={final_len})")
    return (overall_best_seq.tolist() if overall_best_seq is not None else [])
