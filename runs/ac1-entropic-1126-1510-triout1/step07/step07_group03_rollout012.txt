propose_candidate
-----------------
Search for a non‑negative step‑function (probability vector) that

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm is a hybrid of:

objective is scale‑invariant, therefore any normalisation works.
The LP step follows the classical update described in the problem

# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# FFT‑based convolution utilities
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _objective_sum_one(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) when sum(a)=1."""
    n = a.shape[0]

# Linear‑programming refinement (active‑set method)
def _lp_refine(f: np.ndarray, deadline: float,
               max_iter: int = 20) -> tuple[np.ndarray, float]:
    """
    Given a non‑negative vector f with sum = 1, repeatedly solve

        maximise   sum(b)
        s.t.       f * b  ≤  M    (component‑wise)
                   b ≥ 0

    where M = max(conv(f,f)).  The optimiser b0 is scaled to sum = 1,
    convex–combined with f and the best iterate is returned.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_obj = _objective_sum_one(f)
    t = 0.35  # initial convex – step size

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current bound M
        M = _fft_self_convolve(f).max()

        # Toeplitz matrix for the linear map b ↦ f * b (full convolution)
        col = np.r_[f, np.zeros(n - 1)]
        row = np.r_[f[0], np.zeros(n - 1)]
        A = scipy.linalg.toeplitz(col, row)   # shape (2n‑1, n)

        # linprog: maximise sum(b)  ⇒  minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0.0, None)] * n
        res = scipy.optimize.linprog(c,
                                    A_ub=A,
                                    b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                                    bounds=bounds,
                                    method='highs')
        if not res.success:
            # LP failed – abort further LP steps
            break

        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 <= 0.0:
            break

        # scale to keep the simplex constraint
        g = b0 / sum_b0

        # convex combination
        f_new = (1.0 - t) * f + t * g
        obj_new = _objective_sum_one(f_new)

        if obj_new + 1e-13 < best_obj:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            # slowly shrink step size as we get closer
            t = max(0.05, t * 0.92)
            print(f"[{time.time():.1f}] LP iter={it} new best {best_obj:.6f}")
        else:
            # no improvement → damp the move
            t = max(0.01, t * 0.5)
            if t < 0.02:
                break

        # safety check – abort if we have already reached the target
        if best_obj <= 1.503 + 1e-6:
            break

    return best_f, best_obj

# Short exact sub‑gradient descent (polishing)
def _exact_subgradient_descent_short(a: np.ndarray,
                                     deadline: float,
                                     max_steps: int = 1500) -> tuple[np.ndarray, float]:
    """A few sub‑gradient steps (sum = 1) to polish the LP output."""
    n = a.shape[0]
    best_a = a.copy()
    best_obj = _objective_sum_one(a)
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0

    while time.time() < deadline and it < max_steps:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (conv >= cur_max - eps).astype(float)

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _fft_convolve(a[::-1], active)[n - 1:n - 1 + n]

        # momentum + diminishing stepsize
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        obj = _objective_sum_one(a)
        if obj + 1e-13 < best_obj:
            best_obj = obj
            best_a = a.copy()
            # occasional logging
            if it % 200 == 0:
                print(f"[{time.time():.1f}] sub‑grad step={it} best {best_obj:.6f}")

        # early exit if already very good
        if best_obj <= 1.503:
            break

    return best_a, best_obj

    """Search for a step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # 1 s safety margin

    # ------------------------------------------------------------------
    # 1. Lengths to explore (compact enough for the LP, large enough to be competitive)
    lengths = [1200, 1800, 2400, 3000, 3600]

    overall_best_seq = None
    overall_best_obj = float("inf")

    # ------------------------------------------------------------------
    # 2. Iterate over lengths and candidate families
    for n in lengths:
        if time.time() >= deadline:
            break

        # generate a diverse set of seeds for this length
        candidates = _initial_candidates_extended(n)

        # optionally include the previous global best (if defined)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.ndim == 1:
                prev_n = prev.shape[0]
                # either truncate or stretch to current n
                if prev_n >= n:
                    prev = prev[:n]
                else:
                    prev = np.pad(prev, (0, n - prev_n), constant_values=0.0)
                prev = _project_onto_simplex(prev, 1.0)
                candidates.append(prev)
        except Exception:
            pass

        for cand in candidates:
            if time.time() >= deadline:
                break

            # normalise to the simplex (sum = 1)
            a = _project_onto_simplex(cand, 1.0)

            # early pruning – compute a cheap bound
            cur_obj = _objective_sum_one(a)
            if cur_obj < overall_best_obj:
                overall_best_obj = cur_obj
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] init n={n} obj={cur_obj:.6f}")

            # ------------------------------------------------------------------
            # 2a. LP refinement (the main driver)
            a_lp, obj_lp = _lp_refine(a, deadline, max_iter=25)
            if obj_lp + 1e-13 < overall_best_obj:
                overall_best_obj = obj_lp
                overall_best_seq = a_lp.copy()
                print(f"[{time.time():.1f}] after LP n={n} obj={obj_lp:.6f}")

            # ------------------------------------------------------------------
            # 2b. Short sub‑gradient polish (fast, cheap)
            a_sg, obj_sg = _exact_subgradient_descent_short(
                a_lp, deadline, max_steps=1200
            )
            if obj_sg + 1e-13 < overall_best_obj:
                overall_best_obj = obj_sg
                overall_best_seq = a_sg.copy()
                print(f"[{time.time():.1f}] after sub‑grad n={n} obj={obj_sg:.6f}")

            # If we have already beaten the target we can stop early
            if overall_best_obj <= 1.503:
                break

        # optional: a final polishing on the best vector of this length
        if overall_best_seq is not None and time.time() < deadline:
            a_final, obj_final = _lp_refine(overall_best_seq, deadline, max_iter=15)
            if obj_final + 1e-13 < overall_best_obj:
                overall_best_obj = obj_final
                overall_best_seq = a_final.copy()
                print(f"[{time.time():.1f}] final LP polish n={n} obj={obj_final:.6f}")

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case nothing was found
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_obj = _objective_sum_one(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform obj={overall_best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4. Return a plain list; normalise to sum = 1 for clean output
    best_normalised = overall_best_seq / overall_best_seq.sum()
    final_score = evaluate_sequence(best_normalised.tolist())
    print(f"[{time.time():.1f}] FINAL best objective = {final_score:.6f} (length={best_normalised.size})")
    return best_normalised.tolist()
