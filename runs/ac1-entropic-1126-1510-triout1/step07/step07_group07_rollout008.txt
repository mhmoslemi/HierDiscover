            a_new = _project_onto_simplex(a - step * grad)

                step *= 0.5

"""
Hybrid optimizer for the step‑function evaluation problem.

The objective for a non‑negative sequence a with Σa = 1 is

    F(a) = 2·n·max_k (a * a)_k .

Because the evaluation is scale‑invariant we may restrict ourselves to the
probability simplex.  The code builds a diverse pool of seed vectors,
picks the most promising ones, and then refines them with a continuation
method on the smooth approximation

    f_μ(a) = μ·log ∑_k exp( (a*a)_k / μ ),

whose gradient is computed analytically via FFTs.  For each smoothing
parameter μ we perform many projected gradient steps with a stepsize
chosen from the Lipschitz constant L≈2/μ (so step = μ/2).  μ is reduced
geometrically until it reaches ~1e‑8, after which a short projected
sub‑gradient descent (exact max‑convolution) is used to polish the final
solution.  Random perturbations are injected when time permits.

All operations are on the probability simplex (Euclidean projection) so
the scaling is handled automatically by `evaluate_sequence`.

The algorithm respects the 1000 s budget, prints intermediate scores and
returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Convolutions using real FFTs
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b via FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _self_convolution_full(a).max()

def _raw_score(a: np.ndarray) -> float:
    """Proxy objective 2·n·max(a*a) for a probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.

    Returns (f, grad) where f ≈ max(a*a) and grad is ∂f/∂a.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights, Σp = 1

    # gradient: 2 * (a_rev * p)  (central n entries)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smooth value
    f_val = M + mu * np.log(Z)
    return f_val, grad

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to sum = 1."""
    # avoid division by zero when α or β are tiny
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf for numerical stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)          # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# ----------------------------------------------------------------------
def _gradient_descent_continuation(a0: np.ndarray,
                                   time_limit: float,
                                   mu_target: float = 1e-8,
                                   mu_factor: float = 0.6,
                                   inner_iters: int = 5000) -> tuple:
    """
    Continuation on the smooth log‑sum‑exp approximation.
    Returns (best_a, best_raw_score).
    """
    a = a0.copy()
    best_a = a.copy()
    best_raw = _raw_score(a)

    mu = max(_max_self_convolution(a) * 0.5, 1e-7)   # initial smoothing
    start = time.time()

    while True:
        # check time
        if time.time() - start > time_limit:
            break
        # stepsize from Lipschitz constant L ≈ 2/μ  => step = μ/2
        step = mu * 0.5

        # decide how many inner steps we can afford
        remaining = time_limit - (time.time() - start)
        # empirical cost ≈ 0.0006 s per 1000 iterations for n≈16000
        est_iter = max(1, int(min(inner_iters, remaining / 0.0006)))
        for _ in range(est_iter):
            f_val, grad = _smooth_obj_and_grad(a, mu)
            a = _project_onto_simplex(a - step * grad)

            # keep track of best raw score
            cur_raw = _raw_score(a)
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_a = a.copy()

            # early exit if time is gone
            if time.time() - start > time_limit:
                break

        # reduce smoothing
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-14:
            break

        # warm‑restart from the best point found at this level
        a = best_a.copy()

    return best_a, best_raw

# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray,
                        time_limit: float) -> tuple:
    """
    Short projected sub‑gradient descent on the exact max‑convolution.
    Returns (best_a, best_raw_score).
    """
    a = a0.copy()
    best_a = a.copy()
    best_raw = _raw_score(a)

    start = time.time()
    it = 0
    # simple momentum to smooth steps
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_limit:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        # active set (indices attaining (or very close to) the max)
        tol = max(1e-15, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]

        # classic step‑size schedule
        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional jitter to escape flat regions
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=len(a))
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_a = a.copy()

    return best_a, best_raw

# ----------------------------------------------------------------------
def _random_power_law(n: int, exponent: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a power‑law shaped probability vector (i+1)^ex, normalised."""
    vals = np.arange(1, n + 1, dtype=np.float64) ** exponent
    s = vals.sum()
    if s == 0.0:
        vals[:] = 1.0 / n
    else:
        vals /= s
    return vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the competition harness.
    Returns a list of non‑negative heights (the optimizer's best solution).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a convenient length (power of two for FFT speed)
    n = 16384

    # ------------------------------------------------------------------
    # 2. Assemble a diverse pool of seed vectors (all Σ=1)
    seeds = []

    # a) possible global warm‑start
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        base /= base.sum()
        seeds.append(base)
    except Exception:
        pass

    # b) simple deterministic baselines
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    seeds.append(_beta_distribution(n, 0.5, 0.5))                # arcsine

    # c) symmetric Beta families (log‑spaced alphas)
    for alpha in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        seeds.append(_beta_distribution(n, alpha, alpha))

    # d) a few random asymmetric Beta draws
    for _ in range(6):
        a = 10 ** rng.uniform(-2, 1)
        b = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_distribution(n, a, b))

    # e) random Dirichlet draws (different concentrations)
    for _ in range(8):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec /= vec.sum()
        seeds.append(vec)

    # f) power‑law families (both increasing and decreasing)
    for exp in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        seeds.append(_random_power_law(n, exp, rng))

    # ------------------------------------------------------------------
    # 3. Quick screening on the raw proxy (2·n·max(a*a))
    best_seq = None
    best_raw = float("inf")
    for idx, seq in enumerate(seeds):
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    print(f"[Init] Best raw score after screening: {best_raw:.6f}")

    # ------------------------------------------------------------------
    # 4. Main continuation + gradient descent (allocate roughly 65 % of time)
    remaining = deadline - time.time()
    if remaining > 15.0:
        # use the current best as start; we may also try a few extra random restarts
        candidates = [best_seq]
        # add a couple of random Dirichlet restarts for diversity
        for _ in range(3):
            conc = 10 ** rng.uniform(-2, 1)
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            candidates.append(vec)

        global_best_seq = best_seq
        global_best_raw = best_raw

        # split remaining time among the candidates (fair share)
        for cand in candidates:
            cand_time = (deadline - time.time()) * 0.3   # each gets up to 30 % of what's left
            if cand_time < 5.0:
                break
            seq, raw = _gradient_descent_continuation(cand,
                                                      time_limit=cand_time,
                                                      mu_target=1e-8,
                                                      mu_factor=0.55,
                                                      inner_iters=8000)
            if raw < global_best_raw:
                global_best_raw = raw
                global_best_seq = seq.copy()
                print(f"[Grad] New best raw score: {global_best_raw:.6f}")

        best_seq = global_best_seq
        best_raw = global_best_raw

    # ------------------------------------------------------------------
    # 5. Final polishing with exact sub‑gradient descent (≈ 20 % of budget)
    remaining = deadline - time.time()
    if remaining > 5.0:
        seq, raw = _subgradient_polish(best_seq,
                                       time_limit=remaining * 0.9)
        if raw < best_raw:
            best_raw = raw
            best_seq = seq.copy()
            print(f"[Polish] Improved raw score: {best_raw:.6f}")

    # ------------------------------------------------------------------
    # 6. Optional stochastic refinement (tiny random jitters) if time permits
    remaining = deadline - time.time()
    if remaining > 2.0:
        iters = 0
        while time.time() < deadline - 0.5 and iters < 30:
            sigma = 1e-4 * 0.5 ** iters
            pert = best_seq + rng.normal(scale=sigma, size=n)
            pert = _project_onto_simplex(pert)
            # a very short gradient sweep with μ extremely small
            seq, raw = _gradient_descent_continuation(pert,
                                                      time_limit=remaining * 0.2,
                                                      mu_target=1e-9,
                                                      mu_factor=0.5,
                                                      inner_iters=2000)
            if raw < best_raw:
                best_raw = raw
                best_seq = seq.copy()
                print(f"[Jitter] Improved raw score: {best_raw:.6f}")
            iters += 1
            remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 7. Return the final sequence (evaluate_sequence will renormalise automatically)
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"[Done] Final evaluated score: {final_val:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
