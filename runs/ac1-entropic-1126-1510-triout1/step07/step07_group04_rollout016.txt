    # compute conv = a*a
    conv = _self_convolution_full(a)  # shape 2n-1

def _refine_softmax(a, deadline, rng, eta0=0.5, p0=1.0, p_factor=1.01):
    best_a=a.copy()
    best_val = _raw_score(a)
    t=0
    p=p0
    while time.time()<deadline:
        t +=1
        grad, curmax = _grad_softmax(a, p)
        eta = eta0 / math.sqrt(t)
        # exponentiated update
        a = a * np.exp(-eta * grad)
        # renorm
        s = a.sum()
        if s==0:
            a[:] = 1.0/len(a)
        else:
            a/=s
        # maybe occasional jitter
        if t%1000 == 0 and rng.random()<0.1:
            a += rng.normal(scale=1e-6, size=len(a))
            a = _project_onto_simplex(a)

        # update best
        val = _raw_score(a)
        if val+1e-12<best_val:
            best_val=val
            best_a = a.copy()
            print(f"[{time.time():.1f}] softmax{len(a)} it={t} p={p:.2f} score={best_val:.6f}")

        # increase p gradually, but keep bounded
        p = min(p * p_factor, 200.0)  # p up to 200 or so stop increasing
    return best_a, best_val

def _grad_softmax(a, p):
    n = len(a)
    conv = _self_convolution_full(a) # length 2n-1
    # compute softmax weights stable
    maxc = conv.max()
    exp_vals = np.exp(p * (conv - maxc))

    a_rev = a[::-1]

"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises the objective used by
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Warm‑start** – load the previously discovered best sequence (if any)
   and normalise it to Σa = 1.

2. **Soft‑max mirror‑descent** – a smooth approximation of the max‑convolution
   is minimised on the probability simplex.  At iteration *t* we compute

       w_k ∝ exp(p·(a∗a)_k) ,   p → ∞  (annealing)

   and use the weighted gradient
       ∇ = 2·(a_rev ∗ w)[n−1:n−1+n] .
   The update is exponentiated (mirror‑descent) with a decaying step size
   η_t = η₀/√t, together with a momentum term and occasional Gaussian jitter.
   The iterate is always re‑normalised onto the simplex.

   The sub‑routine is called on several lengths (2 k–8 k) and the best
   intermediate sequence is kept.

3. **Upsampling** – the current best is linearly interpolated to larger
   lengths (12 k–20 k) and refined again.  Since the objective is
   scale‑invariant, the final sequence is returned with Σa = 1.

The code never exceeds the supplied time budget (default 1000 s).  The
search is deliberately stochastic: a few random Dirichlet starts and a few
deterministic “sin‑shape’’ starts are tried at each length.  The best raw
score (`2·n·max(a∗a)`) is printed during the run, and the final exact
score is obtained with the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of `old_seq` onto a new length `new_n`."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Score for a vector that already satisfies Σa = 1 (scale invariant)."""
    n = a.shape[0]
    return 2.0 * n * _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, p: float) -> tuple[np.ndarray, float]:
    """
    Weighted gradient of  f(a)=∑_k w_k (a*a)_k  where
        w_k ∝ exp(p·(a*a)_k) , Σ w_k = 1.
    Returns (grad, cur_max), where cur_max is the true max of a*a.
    """
    n = a.shape[0]
    conv = _self_convolution_full(a)                # length 2n‑1
    cur_max = conv.max()

    # soft‑max weights (numerically stable)
    exp_arg = p * (conv - cur_max)                 # shift by max to avoid overflow
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= w_sum

    # weighted gradient: 2·(a_rev ∗ w) restricted to the central n entries
    a_rev = a[::-1]
    size = 1 << ((len(a_rev) + len(w) - 1 - 1).bit_length())
    A_rev = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    grad_full = np.fft.irfft(A_rev * W, n=size)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_softmax(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    eta0: float = 0.5,
    p0: float = 0.5,
    p_factor: float = 1.02,
    max_iter: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a soft‑max approximation of the
    L∞‑norm of the self‑convolution.  Returns the best iterate seen.

    Parameters
    ----------
    a            – initial probability vector (Σa = 1, a_i ≥ 0).
    deadline     – wall‑clock time after which the loop must stop.
    rng          – numpy Generator for jitter.
    eta0         – initial step‑size (decays as eta0/√t).
    p0           – starting temperature for the soft‑max.
    p_factor     – multiplicative increase of p each iteration.
    max_iter     – safety cap on the number of iterations.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)

    # momentum buffer (exponential moving average of gradients)
    momentum = np.zeros_like(a)

    it = 0
    p = p0
    while time.time() < deadline and it < max_iter:
        it += 1

        grad, cur_max = _softmax_gradient(a, p)

        # momentum update
        beta = 0.9
        momentum = beta * momentum + (1.0 - beta) * grad

        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient step (clipped exponent for stability)
        exponent = -eta * momentum
        exponent = np.clip(exponent, -50.0, 50.0)
        a = a * np.exp(exponent)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny Gaussian jitter to escape plateaus
        if it % 2000 == 0 and rng.random() < 0.08:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        # evaluate raw (exact) score
        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(
                f"[{time.time():.1f}] softmax len={n} it={it} p={p:.2f} score={best_val:.6f}"
            )

        # anneal the soft‑max temperature
        p = min(p * p_factor, 200.0)

    return best_a, best_val


# ----------------------------------------------------------------------
def _sin_shape(n: int, power: float = 1.0) -> np.ndarray:
    """Positive sinusoidal shape: (sin(π·x))^power, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.sin(np.pi * x) ** power
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        vals[:] = 1.0 / n
    else:
        vals /= total
    return vals

    best_score = _raw_score(best_seq)
    print(f"[{time.time():.1f}] initial raw score = {best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 1 – explore several moderate lengths.
    lengths = [2000, 4000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # start from the current best, interpolated to the new size
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)
        cur_val = _raw_score(a)
        if cur_val < best_score:
            best_seq, best_score = a.copy(), cur_val
            print(f"   interpolated best score={best_score:.6f}")

        # deterministic simple families
        families = {
            "uniform": np.full(n, 1.0 / n),
            "sin^1": _sin_shape(n, power=1.0),
            "sin^2": _sin_shape(n, power=2.0),
        }
        for name, vec in families.items():
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   {name} score={best_score:.6f}")

        # random Dirichlet seeds
        for _ in range(30):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-2, 1)               # concentration in [0.01,10]
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            # refine this random start a little (few hundred iters)
            vec, _ = _refine_softmax(vec, deadline, rng, eta0=0.3, max_iter=2000)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   dirichlet conc={conc:.3f} score={best_score:.6f}")

        # a handful of random Beta mixtures (already normalised)
        for _ in range(20):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            lam = rng.random()
            p1 = _sin_shape(n, power=1.0) * 0  # placeholder for speed‑up (will be replaced)
            # Instead of recomputing the sin shape every time we directly use Beta PDFs:
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            p1 = np.exp(
                (a1 - 1) * np.log(x) + (b1 - 1) * np.log(1 - x)
            )
            p2 = np.exp(
                (a2 - 1) * np.log(x) + (b2 - 1) * np.log(1 - x)
            )
            p1 = np.maximum(p1, 0.0)
            p2 = np.maximum(p2, 0.0)
            p1 /= p1.sum()
            p2 /= p2.sum()
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            vec, _ = _refine_softmax(vec, deadline, rng, eta0=0.3, max_iter=3000)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   beta‑mix score={best_score:.6f}")

        # after a batch of random seeds, run a medium‑length refinement on the current best
        if time.time() < deadline:
            a_refined, scr = _refine_softmax(best_seq, deadline, rng, eta0=0.5)
            if scr < best_score:
                best_seq, best_score = a_refined.copy(), scr
                print(f"   refined on len={n} score={best_score:.6f}")

    print(f"[{time.time():.1f}] best raw score after coarse search = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final refinement on the current best length with a longer budget
    if time.time() < deadline:
        a_refined, best_score = _refine_softmax(
            best_seq, deadline, rng, eta0=0.6, max_iter=100_000
        )
        best_seq = a_refined.copy()
        print(f"[{time.time():.1f}] after long refinement: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to larger lengths and polish again.
    larger_lengths = [12000, 16000, 20000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)
        a, scr = _refine_softmax(a, deadline, rng, eta0=0.5, max_iter=80_000)
        if scr + 1e-12 < best_score:
            best_seq, best_score = a.copy(), scr
            print(f"   new best after up‑sample: score={best_score:.6f}")

        if best_score <= 1.503:
            print("Target bound reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final evaluation with the official evaluator (exact arithmetic, clipping, etc.)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
