assistantfinal**Solution Overview**

The evaluator is scale‑invariant, therefore we may restrict ourselves to a *probability* vector  
`a ≥ 0 ,  Σ a = 1`.  
The score that the evaluator returns is then  

```
score(a) = 2·n·max_k (a*a)_k                (1)
```

where `(a*a)` is the linear self‑convolution of `a`.  
We minimise the `L∞`‑norm of the convolution on the simplex.

The algorithm consists of three stages

1. **Broad random exploration** – generate many candidate shapes (uniform,
   arcsine, Beta‑families, Dirichlet draws, mixtures, exponential decay,
   piece‑wise constant).  
   All candidates are projected onto the simplex and scored with a fast
   FFT‑based proxy (exact for Σa=1).

2. **Smooth mirror‑descent refinement** – instead of the plain
   sub‑gradient of the max‑operator we use a *soft‑max* (log‑sum‑exp)
   smoothing:
   ```
   w_k ∝ exp( b_k / μ ) ,   b = a*a
   grad  = 2·conv_rev(a , w)
   ```
   With a temperature `μ` proportional to the current maximum the gradient
   is a weighted sum of all active constraints, which helps to escape flat
   plateaus.  
   The update is multiplicative (exponentiated‑gradient) and each iterate is
   projected back onto the simplex.  Step‑size `η_t = η₀/√t` and occasional
   jitter are used for robustness.

3. **Upsampling & final polishing** – the best sequence found so far is
   interpolated to larger lengths (up to 20 000 points) and refined again.
   Larger discretisations reduce the discretisation error of the continuous
   problem without changing the asymptotic score.

The whole search respects the 1000 s budget, constantly checks the remaining
time and stops early if the target bound `1.503` is reached.

```python
"""
Propose a low‑score step function.

The algorithm works in three phases:

1️⃣  Random exploration on several lengths (2000‑8000).  Candidates are
    drawn from a variety of families (uniform, arcsine, symmetric Beta,
    Dirichlet, mixtures, exponential decay, piece‑wise constant) and are
    projected onto the probability simplex.

2️⃣  For every promising candidate a *smooth mirror‑descent* (exponentiated‑
    gradient) is performed.  The gradient is obtained from a soft‑max
    weighting of the self‑convolution entries, i.e.

        w_k ∝ exp(b_k / μ)   with μ ≈ 0.3·max(b)

    This approximates the sub‑gradient of the L∞‑norm while keeping the
    direction useful even when the active set is stationary.

3️⃣  The current best shape is up‑sampled to larger sizes (12 k‑20 k) and
    refined again.  Finally the exact evaluator is called and the best
    sequence is returned.

All operations use FFT‑based convolutions (O(n log n)).  The routine stops
as soon as the time budget (≈1000 s) is exhausted or a score ≤ 1.503 is
found.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    # log‑pdf, safe from overflow
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The value equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, mu_rel: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Gradient of the smooth approximation
        g_μ(a) = μ·log Σ_k exp( (a*a)_k / μ )
    Returns (grad, current_maximum_of_convolution).
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    # temperature proportional to the current maximum (never too small)
    mu = max(cur_max * mu_rel, 1e-12)

    # soft‑max weights, numerically stable
    #   w_k ∝ exp( (b_k - cur_max) / μ )
    w = np.exp((b - cur_max) / mu)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= w_sum

    # grad = 2 * conv_rev(a, w)
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _smooth_mirror_descent(a: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.5,
                           mu_rel: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent with a smooth max‑gradient.
    Returns the best sequence found before `deadline` together with its raw score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _smooth_gradient(a, mu_rel)

        # step‑size schedule η_t = η₀ / sqrt(t)
        eta = eta0 / math.sqrt(it)

        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)          # avoid overflow/underflow
        a = a * np.exp(exponent)

        # re‑project onto the simplex
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny random jitter to jump out of tiny plateaus
        if it % 500 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] smooth‑MD len={n} it={it} score={best_val:.6f}")

        # early exit if target already achieved
        if best_val <= 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _exponential_seq(n: int, lam: float) -> np.ndarray:
    """Truncated exponential distribution, normalised to Σ=1."""
    x = np.arange(n, dtype=np.float64)
    pdf = np.exp(-lam * x)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= pdf_sum
    return pdf


def _piecewise_constant_seq(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """
    Piece‑wise constant shape with `m` random blocks.
    First a random partition of the index set is created, then a random
    height (Dirichlet) is assigned to each block.
    """
    # random block lengths (at least 1 element per block)
    lengths = rng.dirichlet(np.ones(m)) * n
    lens = np.maximum(1, np.rint(lengths).astype(int))
    # adjust to exact length n
    diff = n - lens.sum()
    if diff > 0:
        lens[:diff] += 1
    elif diff < 0:
        lens[: -diff] -= 1
    # random block heights
    heights = rng.gamma(1.0, 1.0, size=m)
    heights /= heights.sum()
    seq = np.repeat(heights, lens)[:n]
    return seq

        a, best_score = _smooth_mirror_descent(a, deadline, rng,
                                               eta0=0.5, mu_rel=0.30)
        best_seq = a.copy()
        best_len = len(a)
        print(f"[{time.time():.1f}] after intensive refinement: score={best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to larger lengths (reducing discretisation error)
    larger_lengths = [12000, 16000, 20000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)   # safety
        a, scr = _smooth_mirror_descent(a, deadline, rng,
                                        eta0=0.35, mu_rel=0.30)
        if scr + 1e-12 < best_score:
            best_seq, best_score, best_len = a.copy(), scr, n
            print(f"   new best after up‑sample: score={best_score:.6f} (len={best_len})")
        if best_score <= 1.503:
            print("Target bound achieved – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final exact evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
```
