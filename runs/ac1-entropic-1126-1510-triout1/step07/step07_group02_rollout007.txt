def smooth_descent(a, steps, start_alpha, end_alpha, start_lr, end_lr):
    for t in range(steps):
        alpha = start_alpha * (end_alpha/start_alpha) ** (t/steps)  # exponential schedule
        lr = start_lr * (end_lr/start_lr) ** (t/steps)
        # compute conv, w, grad
        conv = self_convolve(a)
        maxb = conv.max()
        # compute softmax w with temperature 1/alpha? Actually green: w ∝ exp(alpha*(conv - maxb))
        w = np.exp(alpha * (conv - maxb))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # grad via convolution
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[len(a)-1 : len(a)-1+len(a)]
        # mirror descent
        a = a * np.exp(-lr * grad)
        # project onto simplex
        a = _project_onto_simplex(a, 1.0)
        # optional small epsilon for numerical stability
        a[a < 1e-12] = 0.0
        a = a / a.sum()
    return a

"""Propose a step‐function vector with small max‐self‑convolution.

The routine combines several ideas:

1.  A large collection of analytic families (uniform, arcsine, triangular,
    exponential, symmetric Beta, sine‑power, Gaussian) and random Dirichlet
    draws are generated for a few candidate lengths.
2.  Each candidate is evaluated cheaply (FFT self‑convolution) and the best
    one is kept.
3.  A smooth‑max (soft‑max) mirror‑descent is performed with an annealing
    schedule that drives the temperature to a very high value while
    simultaneously shrinking the learning rate.  This refines the shape
    on the simplex.
4.  A classic sub‑gradient descent with momentum polishes the solution.
5.  Finally a lightweight hill‑climber works directly on the convolution
    using O(n) incremental updates (mass transfer between two indices).

The whole search respects a 1000 s wall‑clock budget.  The function returns
the best sequence (as a Python list of floats) found within that time.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2 with relative width sigma."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

    # ------------------------------------------------------------------
    # 3. Fallback (should never trigger)
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq = best_seq / best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Smooth‑max mirror descent with annealing.
    a = best_seq.copy()
    n = a.shape[0]

    # (alpha_start, alpha_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 50.0, 2000, 0.20, 0.020),
        (50.0, 500.0, 3000, 0.020, 0.002),
        (500.0, 5000.0, 3000, 0.002, 0.0002),
        (5000.0, 50000.0, 2000, 0.0002, 0.00002),
    ]

    for alpha0, alpha1, steps, lr0, lr1 in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            # geometric interpolation of alpha and lr
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # mirror‑descent step (exponentiated gradient)
            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if it % 500 == 0 or it == steps - 1:
                cur_sc = _max_convolution_score(a)
                if cur_sc < best_score - 1e-12:
                    best_score = cur_sc
                    best_seq = a.copy()
                    best_n = n
                    print(f"[{time.time():.1f}] mirror descent iter={it} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Sub‑gradient refinement (momentum, decreasing step size).
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps_rel)

        w = np.zeros_like(conv)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = 0.15 / np.sqrt(it)            # tuned baseline step
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur_sc = _max_convolution_score(a)
        if cur_sc < best_score - 1e-12:
            best_score = cur_sc
            best_seq = a.copy()
            best_n = n
            print(f"[{time.time():.1f}] subgrad iter={it} new best {cur_sc:.6f}")

        # early exit guard
        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 6. Final hill‑climbing directly on the convolution (O(n) updates).
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    while time.time() < deadline:
        it += 1
        # Locate a maximal convolution entry and the pair (i,j) that generates it.
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {best_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap

        if it % 5000 == 0:
            # small jitter to possibly escape plateaus
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
