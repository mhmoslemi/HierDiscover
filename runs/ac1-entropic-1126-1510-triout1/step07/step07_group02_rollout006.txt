"""
Propose a step–function (a non‑negative vector on the simplex) that
minimises ``evaluate_sequence``.

The search consists of four stages:

1. **Broad scan** – several lengths (up to 20 000) and many analytic
   families (beta, sine‑power, exponential, cosine‑window, arcsine,
   uniform, triangular, random sparse).  The best vector among all
   candidates is kept.

2. **Adam descent on a smoothed max** – the maximum of the
   self‑convolution is approximated by a soft‑max with a gradually
   increasing temperature (parameter ``alpha``).  The gradient of this
   smooth objective is computed via FFT, an Adam step is taken and the
   vector is projected back onto the simplex.

3. **Exact sub‑gradient descent** – once the soft‑max has become sharp,
   we switch to the true sub‑gradient (indicator of the active
   convolution entries) and continue with a classical diminishing‑step
   scheme.

4. **Greedy hill‑climber** – tiny mass transfers are performed
   directly on the convolution array using an O(n) incremental update.
   Moves that lower the maximal convolution value are kept; otherwise
   they are reverted.  Occasionally a random jitter is injected to
   escape plateaux.

All time‑consuming loops respect a global deadline (``budget_s`` seconds
minus a small safety margin).  The routine returns the best sequence
found as a plain Python list of ``float``.  The sequence is already
normalised to sum = 1, which makes the returned objective equal to
``2·n·max(a*a)`` – exactly the quantity evaluated by
``evaluate_sequence`` when the sum of the vector is 1.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_general(n, a, b):
    """General Beta(a,b) – a>0, b>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()                     # linear part uses the *old* a
    # linear updates: b_new = b_old - 2*delta*a_old[i] + 2*delta*a_old[j]
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms (i,i) and (j,j)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

    max_jitter = 1e-5

    while time.time() < deadline:
        it += 1
        # locate the shift k where the current maximum occurs
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)

        # find the pair (i,j) inside this shift that gives the biggest product
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # choose source (the larger of the two) and a random target
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue

        # propose a modest transfer from source → target
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        # snapshot before the move (O(n) copies)
        a_snap = a.copy()
        conv_snap = conv.copy()

        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            it_no_improve = 0
            if it % 100 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {best_score:.6f}")
        else:
            # revert
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        # occasional random jitter to escape flat regions
        if it % 2000 == 0 and it_no_improve > 500:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
