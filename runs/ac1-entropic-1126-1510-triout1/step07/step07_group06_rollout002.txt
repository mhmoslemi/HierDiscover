"""
propose_candidate
=================

Generate a non‑negative step‑function (a list of heights) that minimizes the
black‑box ``evaluate_sequence`` score.

The algorithm mixes three ideas:

1. **Beta‑shaped parametric family** – discretised Beta(α,α) densities
   (α≈0.5 gives the classical arcsine shape).  A cheap one‑dimensional search
   over α yields a very good starting point for any length *n*.

2. **Smooth‑max Adam optimisation** – the log‑sum‑exp surrogate for the
   ``max`` in the objective is minimised with Adam.  The optimiser is warm‑started
   from the best β‑shape found and runs a few thousand iterations.

3. **Projected sub‑gradient polishing** – after the Adam phase a classic
   sub‑gradient method (active‑set gradient, Euclidean projection onto the
   simplex) refines the solution to the true non‑smooth objective.

The procedure is repeated for several lengths (2000 … 20000).  Each longer
length is warm‑started by interpolating the best sequence found so far.
The whole search respects a 1000‑second wall‑clock budget; if the time runs
out the best sequence discovered up to that moment is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (edges heavier, centre lighter)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _beta_start(n: int, alpha: float) -> np.ndarray:
    """
    Discretised Beta(α,α) density on the mid‑points of the n bins.
    Returns a probability vector (sum = 1, all entries ≥ 0).
    """
    # mid‑point of each bin in (0,1)
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    a = np.maximum(x, eps)
    b = np.maximum(1.0 - x, eps)
    vals = (a * b) ** (alpha - 1.0)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _raw_score(prob: np.ndarray) -> float:
    """
    Return the exact (non‑smoothed) objective for a probability vector:
    2·n·max_k (prob*prob)_k .
    """
    n = prob.shape[0]
    conv = _fft_self_convolve_full(prob)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of the self‑convolution.
    Returns the best probability vector seen and its raw (unsmoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                      # Σa = 1, a ≥ 0
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={it} raw={best_score:.6f} λ={lam:.3f}")

        # smooth max (log‑sum‑exp)
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()

        # gradient wrt a
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = lam * lam_factor

        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_a)

    return best_a, best_score
