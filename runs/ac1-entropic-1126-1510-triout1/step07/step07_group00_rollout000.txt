def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate score = 2·n·max(a*a) for vectors normalised to sum=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction when the active index equals 2*i
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _grad_smooth(a: np.ndarray, p: int) -> np.ndarray:
    """
    Gradient of f_p(a) = (∑(a*a)^p)^{1/p}  (p ≥ 2).
    The gradient is
        g = 2 * conv_rev(a, (a*a)^{p-1}) / f_p^{p-1}
    where conv_rev denotes convolution of a reversed with the weight vector.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                     # length 2n‑1
    # safeguard against under‑flow/overflow
    b_pow = np.power(b, p - 1, out=np.empty_like(b), where=b > 0)
    f_pow = np.sum(np.power(b, p, out=np.empty_like(b), where=b > 0))
    # f = (f_pow)^{1/p}
    f = f_pow ** (1.0 / p) if f_pow > 0 else 0.0
    if f == 0.0:
        return np.zeros_like(a)

    # convolution of reversed a with weight b^{p-1}
    conv_rev = _fft_convolve(a[::-1], b_pow)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n] / (f ** (p - 1))
    return grad


# ----------------------------------------------------------------------
def _adam_smooth(
    a0: np.ndarray,
    deadline: float,
    p: int,
    max_iters: int = 20000,
    step0: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
) -> tuple[np.ndarray, float]:
    """
    Adam descent on the smooth Lp‑objective for a fixed p.
    Returns the best iterate (according to the surrogate max‑conv score)
    and its surrogate score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _grad_smooth(a, p)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _continuation_descent(
    seed: np.ndarray,
    deadline: float,
    p_schedule: list[int],
    iters_per_p: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Run Adam on a sequence of increasing p values.
    Each stage starts from the best iterate of the previous stage.
    """
    a = seed.copy()
    best_score = _max_convolution_score(a)
    best_a = a.copy()

    for p in p_schedule:
        if time.time() >= deadline:
            break
        # allow a modest number of iterations for each p
        a, sc = _adam_smooth(a, deadline, p, max_iters=iters_per_p)
        if sc < best_score - 1e-14:
            best_score, best_a = sc, a.copy()
    return best_a, best_score

# Analytic families -------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _raised_cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # cosine centred at the middle, zeroed where negative
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n)
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _cauchy_distribution(n: int, scale: float = None) -> np.ndarray:
    if scale is None:
        scale = n / 10.0
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(alpha,…,alpha)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Respects the supplied time budget (≈ 1000 s) and returns the best sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Determine a sensible base length
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    # Choose a handful of lengths to explore (balance quality vs speed)
    lengths = set()
    for scale in (0.75, 1.0, 1.25, 1.5, 2.0):
        n = int(base_n * scale)
        if 500 <= n <= 20000:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(20000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 2. Generate a broad set of candidate sequences
    candidates = []   # (array, surrogate_score)

    beta_ps = (
        list(np.arange(0.02, 0.21, 0.02))
        + list(np.arange(0.25, 0.71, 0.05))
        + list(np.arange(0.80, 1.51, 0.10))
        + [2.0]
    )
    dirichlet_alphas = [0.2, 0.5, 1.0, 2.0, 5.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # Analytic families ------------------------------------------------
        for gen, name in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exponential"),
            (_raised_cosine_distribution, "raised_cosine"),
            (_cauchy_distribution, "cauchy"),
        ):
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:<5d} {name:14s} score={score:.6f}")

        # Symmetric beta family (dense grid)
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Random Dirichlet draws
        for alpha in dirichlet_alphas:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))

    # ------------------------------------------------------------------
    # 3. Seed with the previous best sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            # Normalise to sum=1 for the surrogate
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Keep the top few seeds for the heavy optimisation phase
    candidates.sort(key=lambda x: x[1])
    top_k = min(8, len(candidates))
    top_seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")
    print(f"Current best surrogate score = {top_seeds[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 5. Refinement via smooth continuation (Adam on Lp‑norms)
    best_seq = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]

    # schedule of p‑values (starting smooth → more sharp)
    p_schedule = [4, 6, 8, 12, 16, 20]
    iters_per_stage = 2500

    for idx, (seed_seq, seed_score) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Refinement start for seed {idx+1}/{top_k} (len={seed_seq.size})")
        seq, sc = _continuation_descent(seed_seq, deadline, p_schedule, iters_per_stage)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] New best after continuation: {best_score:.6f}")

        # a quick Adam run directly on the max‑norm to polish
        seq2, sc2 = _adam_smooth(seq, deadline, p=2, max_iters=5000, step0=0.12)
        if sc2 < best_score - 1e-14:
            best_score, best_seq = sc2, seq2.copy()
            print(f"[{time.time():.1f}] Polished with p=2 Adam: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final Frank‑Wolfe pairwise mass‑transfer polish (max‑conv objective)
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 60000:
        it_fw += 1

        grad, cur_max, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # mass source (most harmful)
        i_dst = int(np.argmin(grad))   # mass sink (most beneficial)

        if i_src == i_dst:
            break

        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try full transfer first
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-12:
            a = a_full
            cur_score = score_full
            improved = True
        else:
            # fallback: half transfer
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-12:
                a = a_half
                cur_score = score_half
                improved = True

        if improved:
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] FW iter={it_fw} new best {best_score:.6f}")
            continue

        # occasional tiny jitter to escape stagnation
        if it_fw % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # The returned list will be fed to evaluate_sequence, which will apply its own clamping.
    return best_seq.tolist()
