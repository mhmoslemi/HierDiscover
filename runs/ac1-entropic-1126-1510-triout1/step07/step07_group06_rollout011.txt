def hill_climb(a, n, best_score, deadline):
    cur = a.copy()
    cur_score = best_score
    iters = 0
    while time.time()<deadline and iters < 20000:
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand()*0.001) * cur[i] # some fraction
        if cur[i] < 1e-12: continue
        delta = min(delta, cur[i])
        # propose moving delta from i to j
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand) # ensure positivity sum 1
        # evaluate raw score
        conv = _fft_self_convolve_full(cand)
        cand_score = 2.0 * n * conv.max()
        if cand_score < cur_score - 1e-12:
            cur = cand
            cur_score = cand_score
        iters += 1
    return cur, cur_score

The program searches for a non‑negative step function (a probability vector) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,\;n=\text{len}(a).
\]

Because the ratio is invariant under scaling we may restrict ourselves to
\(\sum a_i = 1\) and simply minimise  

\[
F(a)=2n\;\max_k (a*a)_k .
\]

The optimisation proceeds in three phases  

1. **Smoothed Adam optimisation** – we minimise a differentiable
   log‑sum‑exp approximation of the max.  The probability vector is
   parametrised by logits and projected on the simplex after every
   iteration.  Several diverse initialisations (uniform, arcsine,
   power‑law shapes, random Dirichlet draws, warm‑starts) are tried.

2. **Local refinement** – a cheap stochastic hill‑climbing step moves a
   tiny amount of mass from one index to another and keeps the change
   only if the raw score improves.  Afterwards a projected sub‑gradient
   method (the same idea used in the reference solution) performs a
   few thousand iterations with an aggressive step‑size schedule.

3. **Mix‑and‑match** – the best sequence is linearly mixed with a few
   simple baselines (uniform, arcsine) on a fine grid; the mixed
   sequence is re‑projected onto the simplex and the best one is kept.

All work respects the 1000 s wall‑time budget.  The code is completely
self‑contained, uses only `numpy` and the supplied `evaluate_sequence`,
and prints intermediate scores so the progress can be monitored.

```python
"""Step‑function optimiser for the evaluate_sequence challenge.

The optimiser works in three stages:
  1. Smoothed Adam optimisation of the log‑sum‑exp of the self‑convolution.
  2. Stochastic hill‑climbing + projected sub‑gradient polishing.
  3. Linear mixing with simple baselines (uniform, arcsine) to gain a final
     fraction of a percent improvement.

All sequences are kept on the probability simplex, therefore the final
score is exactly 2·n·max(conv) (the denominator of evaluate_sequence is 1).
"""

import time
import math

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    # mid‑points of each bin as support points
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (heavy edges, light centre)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _powerlaw_start(n: int, p: float = 0.5) -> np.ndarray:
    """Family a_i ∝ ((i+½)(n‑½‑i))^{-p}  (p=0.5 gives the arcsine)."""
    i = np.arange(n, dtype=float) + 0.5
    denom = i * (n + 0.5 - i)
    with np.errstate(divide='ignore'):
        vals = denom ** (-p)
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals

#  Smoothed Adam optimiser

def _adam_optimize_smoothed(x0: np.ndarray, n: int, deadline: float,
                            max_iters: int = 4000,
                            lr0: float = 0.12,
                            lam_start: float = 0.5,
                            lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of a probability vector,
    minimising the log‑sum‑exp of the self‑convolution.
    Returns the best probability vector encountered and its raw score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start
    last_impr_it = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector, enforced to lie on the simplex
        a = _softmax(x)

        # enforce exact symmetry – the optimum is symmetric
        a = (a + a[::-1]) * 0.5

        # raw objective
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_impr_it = it
            print(f"[{time.time():.1f}] n={n} iter={it} raw={best_score:.6f} λ={lam:.3f}")

        # smooth max (log‑sum‑exp)
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()          # shape (2n‑1,)

        # gradient w.r.t a of the smoothed objective
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1: n - 1 + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # anneal λ
        lam = lam * lam_factor

        # jitter if we have been stagnant for a while
        if it % 500 == 0 and (it - last_impr_it) > 2000:
            x += np.random.randn(*x.shape) * 0.02
            last_impr_it = it

    # fallback – uniform distribution (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score

def _hill_climb(a: np.ndarray, n: int, cur_score: float,
                deadline: float, max_steps: int = 5000) -> tuple[np.ndarray, float]:
    """Very cheap stochastic hill‑climbing on the simplex."""
    best = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and it < max_steps:
        i, j = np.random.choice(n, 2, replace=False)
        # move a tiny fraction of mass from i to j
        delta = best[i] * (np.random.rand() * 0.005)
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand)   # safeguard
        conv = _fft_self_convolve_full(cand)
        sc = 2.0 * n * conv.max()
        if sc < best_sc - 1e-12:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc


def _subgradient_refine(a: np.ndarray, n: int, cur_score: float,
                        deadline: float, max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑objective."""
    best = a.copy()
    best_sc = cur_score
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve_full(best)
        cur_max = conv.max()
        # active set (indices that are (almost) maximal)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(best[::-1], w)[n - 1: n - 1 + n]

        # Polyak‑style diminishing step
        step = 0.2 / math.sqrt(it)
        best = best - step * grad
        best = _project_onto_simplex(best)

        sc = 2.0 * n * cur_max
        if sc < best_sc - 1e-12:
            best_sc = sc
            # occasional tiny perturbation to leave plateaux
            if it % 800 == 0 and np.random.rand() < 0.06:
                best += np.random.randn(n) * 5e-7
                best = _project_onto_simplex(best)
        # silent – we only print when a genuine improvement occurs
    return best, best_sc


def _mix_with_baselines(a: np.ndarray, n: int, cur_score: float,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Linear mixing of a with uniform / arcsine baselines on a fine grid."""
    baseline = {
        "uniform": np.full(n, 1.0 / n, dtype=np.float64),
        "arcsine": _arcsine_start(n),
    }
    best = a.copy()
    best_sc = cur_score
    for name, base in baseline.items():
        for alpha in np.linspace(0.0, 1.0, 21):
            if time.time() >= deadline:
                return best, best_sc
            mix = (1.0 - alpha) * a + alpha * base
            mix = _project_onto_simplex(mix)
            conv = _fft_self_convolve_full(mix)
            sc = 2.0 * n * conv.max()
            if sc < best_sc - 1e-12:
                best = mix
                best_sc = sc
    return best, best_sc

def _optimise_length(n: int, deadline: float,
                     prev_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n.
    Several diverse initialisations are tried, the best raw score is returned.
    """
    init_logits = []

    # warm‑start (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        init_logits.append(np.log(cand + 1e-12))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # few random Dirichlet draws
    for _ in range(3):
        a = _random_dirichlet_start(n)
        init_logits.append(np.log(a + 1e-12))

    # deterministic arcsine shape
    init_logits.append(np.log(_arcsine_start(n) + 1e-12))

    # power‑law family (different exponents)
    for p in (0.4, 0.6, 0.8, 1.0):
        init_logits.append(np.log(_powerlaw_start(n, p) + 1e-12))

    best_local_seq = None
    best_local_score = float("inf")

    # empirically fewer iterations for very large n
    max_iters = 4000 if n <= 8000 else 3000

    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length={n} start={idx+1}/{len(init_logits)}")
        seq, score = _adam_optimize_smoothed(
            x0, n, deadline,
            max_iters=max_iters,
            lr0=0.12,
            lam_start=0.5,
            lam_factor=1.001,
        )
        if score < best_local_score:
            best_local_score = score
            best_local_seq = seq.copy()

    # safety fallback
    if best_local_seq is None:
        best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_local_score = 2.0 * n * _fft_self_convolve_full(best_local_seq).max()

    # --------------------------------------------------------------
    # Local refinement (hill‑climb → sub‑gradient → mixing)
    # --------------------------------------------------------------
    cur_seq = best_local_seq
    cur_score = best_local_score

    # hill‑climbing
    cur_seq, cur_score = _hill_climb(cur_seq, n, cur_score, deadline,
                                     max_steps=6000)

    # sub‑gradient polishing
    cur_seq, cur_score = _subgradient_refine(cur_seq, n, cur_score,
                                             deadline, max_iters=5000)

    # final mixing with simple baselines
    cur_seq, cur_score = _mix_with_baselines(cur_seq, n, cur_score,
                                             deadline)

    return cur_seq, cur_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Returns a list of non‑negative heights (a step function) that
    minimises `evaluate_sequence`.  The routine respects the supplied time
    budget (default 1000 s) and prints progress information.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for final evaluation
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if present
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # Length schedule – larger n gives more freedom but costs more time.
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 8000, 12000, 20000]

    best_seq = None
    best_score = float("inf")
    best_n = None

    if prev is not None:
        raw = 2.0 * len(prev) * _fft_self_convolve_full(prev).max()
        print(f"Baseline from height_sequence_1: raw score = {raw:.6f}")
        if raw < best_score:
            best_score = raw
            best_seq = prev.copy()
            best_n = len(prev)

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        seq, score = _optimise_length(n, deadline, prev_seq=best_seq)
        print(f"   finished length {n}: raw best = {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_n = n
        # warm‑start for the next size
        prev = best_seq

    # ------------------------------------------------------------------
    # Final safety check – if nothing was found fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_seq).max()
        best_n = n

    # ------------------------------------------------------------------
    # Final evaluation through the black‑box
    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
```
