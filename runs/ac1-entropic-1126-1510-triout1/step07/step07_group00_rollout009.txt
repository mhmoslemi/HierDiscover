"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Initial pool** – generate a diverse set of candidates:
   * analytic families (arcsine, triangular, exponential, symmetric beta,
     sinusoidal power, piece‑wise linear ramps)
   * random Dirichlet draws with a range of concentration parameters.
   All candidates are ℓ₁‑normalised (∑a=1).

2. **Smooth‑max gradient descent** – minimise a smoothed version of
   ``max(a*a)`` on the simplex.
   The smoothed objective is
        S_λ(a) = (1/λ)·log(∑_k exp(λ·c_k(a)))   ,  c_k = (a∗a)_k .
   Its gradient is a convex combination of the exact subgradients
   corresponding to each lag.  We perform an exponentiated‑gradient
   (KL‑mirror) update, re‑normalise to the simplex and increase λ
   geometrically (λ←1.5·λ) so that the optimisation approaches the
   true max‑objective.

3. **Fine‑grained Frank‑Wolfe transfer** – after the smooth descent we
   polish the current best vector by repeatedly moving probability mass
   from the index with the largest gradient entry to the index with the
   smallest gradient entry, with an aggressive line‑search (full and
   half transfer).  Small Gaussian jitter is injected occasionally to
   escape plateaus.

The routine respects the global time budget (≈ 1000 s) and always returns
a list of non‑negative floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        # already feasible – return a copy to avoid side‑effects
        return v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, len(v) + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of a and b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _convolution_vector(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper returning the full convolution a∗a."""
    return _fft_self_convolve(a)


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact surrogate used in the original code:
        2·n·max(a∗a)   (valid when Σa = 1)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_max(a: np.ndarray, lam: float) -> (float, np.ndarray):
    """
    Compute the smoothed max S_λ(a) and its gradient w.r.t. a.
    Returns (S_λ, grad), where grad is the exact gradient of S_λ.
    """
    n = a.shape[0]
    # full convolution
    b = _fft_self_convolve(a)                     # length 2n‑1
    # numerically stable soft‑max
    max_b = b.max()
    exp_terms = np.exp(lam * (b - max_b))        # shift by max to avoid overflow
    Z = exp_terms.sum()
    w = exp_terms / Z                             # weights (sum to 1)

    # The gradient of each b_k w.r.t a is the same as the sub‑gradient
    # used in the exact case (see original _subgrad implementation).
    # We first compute the convolution of reversed a with the weight vector w.
    conv_rev = _fft_convolve(a[::-1], w)          # length n + (2n‑1) - 1 = 3n‑2
    # Relevant slice corresponds to indices n‑1 … n‑1+n‑1
    grad = 2.0 * conv_rev[n - 1: n - 1 + n]

    # diagonal correction: for even lags (k even) a_i appears twice in b_k,
    # once as a_i·a_i when k = 2·i.  The previous sub‑gradient handled this
    # by subtracting a_i for each such active index.  With a weighted sum we
    # subtract the weighted contribution.
    even_mask = (np.flatnonzero(np.arange(2 * n - 1) % 2 == 0))
    if even_mask.size:
        # even lags = 0,2,4,… .  For each even lag we need the weight w[k].
        # The diagonal contribution of position i is w[2*i] * a_i.
        # Accumulate over all i.
        diag_idx = even_mask // 2          # i = k/2
        diag_contrib = w[even_mask] * a[diag_idx]
        grad[diag_idx] -= diag_contrib
    # smooth max value
    smooth_val = (max_b + np.log(Z) / lam)
    return smooth_val, grad


# ----------------------------------------------------------------------
def _exp_gradient_descent_smooth(a0: np.ndarray,
                                 deadline: float,
                                 init_lam: float = 1.0,
                                 max_lam: float = 64.0,
                                 lam_factor: float = 1.5,
                                 inner_iters: int = 4000) -> (np.ndarray, float):
    """
    Entropic (exponentiated‑gradient) descent on the smoothed max.
    ``lam`` is gradually increased, so that the objective converges to the
    true max.  The routine stops when time runs out or lam exceeds max_lam.
    Returns the best vector seen and its surrogate score (2·n·max).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    lam = init_lam
    while time.time() < deadline and lam <= max_lam:
        # inner loop for a given λ
        for _ in range(inner_iters):
            if time.time() >= deadline:
                break
            _, grad = _smooth_max(a, lam)
            # Exponentiated‑gradient step
            eta = 1.0 / (lam * np.sqrt(grad.var() + 1e-12))
            a = a * np.exp(-eta * grad)
            # re‑normalise to the simplex
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / a.size)
            else:
                a = a / s

        # evaluate plain surrogate (real max) and keep the best
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()

        lam *= lam_factor

    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  max_iters: int = 200000,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> (np.ndarray, float):
    """
    Adam‑style adaptive gradient descent on the exact max‑objective.
    The gradient used is the exact sub‑gradient of max(a∗a).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a∗a) on the simplex Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # indicator vector for active lag positions
    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the un‑corrected gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even lags (k = 2·i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a point on the simplex from Dirichlet(α,…,α)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sinusoidal_power_distribution(n: int, p: float = 2.0) -> np.ndarray:
    """
    a_i ∝ (sin(π·(i+0.5)/n))^p .
    The shape is smooth, symmetric and peaks in the centre.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n) ** p
    return vals / vals.sum()
