"""
Robust search for a non‑negative step function that minimises
evaluate_sequence (i.e. 2·n·max(conv)/sum²).  

The algorithm consists of three phases:

0.  Initialise with a large collection of analytic families (uniform,
    arcsine, beta‑sym, sine‑power, cosine‑window, Gaussian, power‑law,
    exponential, random sparse).  The best candidate (by the cheap
    analytical objective 2·n·max(conv) on the simplex) is kept.

1.  Smooth soft‑max Adam / exponentiated‑gradient descent.
    The max‑convolution is approximated by a soft‑max with a temperature
    parameter α that is increased gradually.  Updates are performed with
    a multiplicative (mirror‑descent) rule which respects the simplex
    constraints without the need for an explicit projection step.

2.  Sub‑gradient refinement (vanilla projected sub‑gradient with a
    decreasing step size).  This drives the solution towards a true
    max‑optimal point.

3.  Deterministic hill‑climbing on the exact convolution.  The gradient
    of the active constraints is computed; mass is moved from the index
    with the largest gradient (the worst offender) to the index with the
    smallest gradient (the most forgiving).  An optimal transfer size
    is obtained by a tiny line‑search, guaranteeing monotone decrease of
    the objective.

All phases respect the 1000 s time budget.  The final sequence is
returned as a plain Python list of floats.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective_on_simplex(a):
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _softmax_weights(b, alpha):
    """Soft‑max weights over the convolution entries (sum to 1)."""
    maxb = b.max()
    w = np.exp(alpha * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
def _gradient_from_weights(a, w):
    """Gradient of  Σ_i w_i (a*a)_i  w.r.t. a   (length‑n vector)."""
    n = a.shape[0]
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a positive amount `delta` from index i to index j,
    updating a (in‑place) and its full self‑convolution `conv`
    (also in‑place) in O(n) time.
    """
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part: (a_i - delta) * a_k  and (a_j + delta) * a_k for all k
    conv[i:i + len(a)] -= 2.0 * delta * a_old
    conv[j:j + len(a)] += 2.0 * delta * a_old
    # quadratic self‑terms on the diagonal positions
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

# Analytic families – all return vectors with sum = 1 ------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma=None):
    """Symmetric Gaussian window, normalised to sum = 1."""
    if sigma is None:
        sigma = n / 6.0  # 3 σ covers the whole support approx.
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p=1.0):
    """Symmetric power‑law: a_i ∝ (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(n, dtype=np.float64)
    left = (i + 1) ** (-p)
    right = (n - i) ** (-p)
    vals = left + right
    vals /= vals.sum()
    return vals

            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill it={it}  new best={new_score:.6f}")
            stagn = 0
        else:
            stagn += 1

        # occasional jitter if we have been stuck for a while
        if stagn > 800 and it % 200 == 0:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            stagn = 0
            print(f"[{time.time():.1f}] jitter applied at hill iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
```
