def _explore_refine(a0, time_budget):
    a = a0.copy()
    n = a.shape[0]
    mu = max(_max_self_convolution(a) * 0.5, 1e-8)
    mu_target = 1e-10
    mu_factor = 0.6
    best_seq = a.copy()
    best_val = _raw_score(a)
    start = time.time()
    while True:
        # compute Lipschitz = max(1, 2.0/mu)
        L = max(1.0, 2.0/mu)
        y = a.copy()
        t = 1.0
        # estimate iteration count based on remaining time
        remaining = time_budget - (time.time() - start)
        if remaining <= 0:
            break
        avg_iter_time = 0.001  # rough
        max_iters = int(min(20000, remaining / avg_iter_time))
        if max_iters <= 0:
            break
        for _ in range(max_iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)
            # backtracking: try step L
            for _ in range(4):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                # sufficient descent condition (Armijo)
                if f_new <= f_val - (np.linalg.norm(grad)**2)/(2*L) + 1e-12:
                    break
                L *= 2.0
            a_prev = a
            a = a_new
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new
            if (_ % 64) == 0:
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target:
            break
        # warm restart to best_seq
        a = best_seq.copy()
    return best_seq, best_val
```

"""
An improved optimiser for the step‑function evaluation problem.

Main ideas
-----------
* Use a larger problem size (n = 12000) – this reduces the
  discretisation error of the continuous optimal density.
* Generate a diverse pool of starting distributions (uniform,
  arcsine, many random Beta / Dirichlet draws and the supplied
  `height_sequence_1`).
* Optimise each promising start with a **continuation**
  smoothed‑max objective using a projected FISTA scheme.
  – Smaller smoothing parameter (mu) → tighter approximation of the
    true max‑convolution.
  – More inner gradient steps per continuation level.
  – Faster decay of mu (factor 0.5) and a stricter final target
    (mu_target = 1e‑9).
* Keep the best sequence seen across all starts and finish with a
  short projected sub‑gradient polish.
* All operations are O(n log n) via FFT, and the total run‑time is
  split dynamically according to the remaining budget.

The returned list is a non‑negative probability vector (∑ a_i = 1);
`evaluate_sequence` rescales it automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of a Beta(α,β) density on n bins, ∑ = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf of the beta distribution (avoid overflow)
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)  # safe cap before exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective: 2·n·max(a*a) for a probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution using the log‑sum‑exp trick.
    Returns (value, gradient) for the current smoothing parameter mu.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights

    # gradient: 2 * (a_rev * p) – take the central n entries
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smoothed objective value
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation + projected FISTA.
    More aggressive than the baseline: larger inner loops,
    tighter final smoothing (mu_target = 1e‑9) and a faster
    decay factor (0.5).
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise smoothing parameter
    mu = max(_max_self_convolution(a) * 0.5, 1e-8)
    mu_target = 1e-9            # tighter approximation of the true max
    mu_factor = 0.5             # halve mu each outer stage
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # ---------- inner FISTA loop ----------
        L = max(1.0, 2.0 / mu)          # safe Lipschitz estimate
        y = a.copy()
        t = 1.0

        # time available for this μ‑stage
        remain = time_budget - (time.time() - start)
        if remain <= 0.0:
            break
        # empirical time for one iteration ~ 0.001 s for n≈8000;
        # we use a generous cap (up to 1 s per stage)
        iters = int(min(12000, remain / 0.001))
        if iters <= 0:
            break

        for _ in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # back‑tracking line‑search (max two doublings of L)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                # standard sufficient‑decrease condition
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # Nesterov extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional check of the true (non‑smoothed) objective
            if (_ & 31) == 0:                # every 32 iterations
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()
        # ---------- end of inner loop ----------

        # move to a smaller smoothing parameter
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        # warm‑restart from the best point found for the current μ
        a = best_seq.copy()

    return best_seq, best_val

def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent (as in the baseline).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        # exact sub‑gradient of max_k (a*a)_k
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # tiny jitter now and then
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()

    return best_seq, best_val
