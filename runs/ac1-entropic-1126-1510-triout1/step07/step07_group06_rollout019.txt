"""
Propose a step‐function (non‑negative heights) that minimizes
evaluate_sequence().  The algorithm has three phases:

1. **Exploratory search on a modest size (n≈400‑800).**  
   We generate many candidate shapes:
   – uniform, arcsine, Dirichlet draws,
   – power‑law, exponential decay and Beta(p,q) families,
   – random mixtures of the above.
   Each candidate is refined by Adam on the smooth log‑sum‑exp
   approximation of the max‑convolution, keeping the best vector.

2. **Scaling & further optimisation on the target size (≈5000).**  
   The best small‑n vector is interpolated to a larger length,
   refined again by Adam and finally by a sub‑gradient method that
   works directly on the true max‑convolution.

3. **Optional LP‑boost (fast linear‑programming step).**  
   For a very small size (n≈200) we perform a few iterations of the
   “LP‑improvement’’ described in the reference solution: maximise the
   ℓ1‑norm of a partner vector under a convolution bound and mix it in.
   This often reduces the max‑convolution before the up‑scaling step.

All operations respect the simplex constraint Σ a_i = 1 and a_i ≥ 0.
The function stops before the time budget runs out and returns the
best sequence found as a plain Python list.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Draw a Dirichlet vector (sum = 1)."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Symmetric arcsine‑like shape."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _power_law_start(n: int, alpha: float) -> np.ndarray:
    """Decreasing power‑law a_i ∝ (i+½)^‑α."""
    i = np.arange(n, dtype=float) + 0.5
    vals = i ** (-alpha)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _exp_decay_start(n: int, beta: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-β·i)."""
    i = np.arange(n, dtype=float) + 0.5
    vals = np.exp(-beta * i)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _beta_start(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) density sampled on the mid‑points of the bins."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    vals = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals

def _adam_optimize_smoothed(x0: np.ndarray, n: int, deadline: float,
                            max_iters: int = 4000,
                            lr0: float = 0.12,
                            lam_start: float = 0.5,
                            lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smooth approximation of
    max_k (a*a)_k.  ``x`` are logits, a = softmax(x) is the probability
    vector (Σa = 1, a ≥ 0). Returns the best probability vector and its
    raw (non‑smoothed) score 2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                     # Σa = 1, a ≥ 0
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.3f}")

        # smooth max (log‑sum‑exp)
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()          # shape (2n‑1,)

        # gradient w.r.t. a : ∇_a f_λ = 2 * (a_rev * p)
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # back‑prop through the soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = lam * lam_factor

        # occasional jitter to hop out of plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:   # fallback – should never happen
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score

def _lp_improvement(f: np.ndarray, t: float, max_lp_iter: int,
                    deadline: float) -> np.ndarray:
    """
    Perform a few LP‑based improvement steps (reference iteration).
    ``f`` is a probability vector (sum = 1).  The step solves:
        maximise Σ b_j  subject to  b_j ≥ 0  and  (f*b)_k ≤ M
    where M = max_k (f*f)_k.  The resulting b is mixed with f using
    parameter t ∈ (0,1].
    """
    n = f.shape[0]
    # build the convolution constraint matrix A (size (2n‑1) × n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    # repeat a few times
    for _ in range(max_lp_iter):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve_full(f)
        M = conv.max()
        # linear programme: minimise -Σ b_j  (i.e. maximise Σ b_j)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs')
        if not res.success:
            break
        b = res.x
        if b.sum() == 0:
            break
        b /= b.sum()
        # convex combination
        f_new = (1 - t) * f + t * b
        # early exit if no improvement
        new_conv = _fft_self_convolve_full(f_new)
        if new_conv.max() >= M - 1e-12:
            break
        f = f_new
    return f
