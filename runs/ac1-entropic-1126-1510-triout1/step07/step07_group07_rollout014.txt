def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """Generate a step–function sequence minimizing evaluate_sequence.

    The algorithm builds several probability‑vector seeds (uniform,
    arcsine, Beta, Dirichlet and the supplied `height_sequence_1`),
    then refines the best one with a soft‑max smoothed max‑convolution
    objective using Adam on the log‑softmax parametrisation.  The
    smoothing parameter μ is halved every few hundred iterations (a
    continuation scheme).  After the main phase a short projected
    sub‑gradient polish is performed.  The returned list is a
    non‑negative vector; `evaluate_sequence` will scale it automatically.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 8000
    eps = 1e-12

    # ------------------------------------------------------------------
    # Build seed candidates (probability vectors)
    seeds = []
    # 1) global warm‑start if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore
        if base.ndim != 1:
            raise ValueError
        base = np.maximum(eps, base)               # avoid exact zeros
        base /= base.sum()
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        seeds.append(base)
    except Exception:
        pass

    # 2) deterministic simple distributions
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))   # uniform
    seeds.append(_beta_distribution(n, 0.5, 0.5))         # arcsine

    # 3) symmetric Beta shapes
    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        seeds.append(_beta_distribution(n, a, a))

    # 4) random Beta draws
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_distribution(n, alpha, beta))

    # 5) random Dirichlet draws
    for _ in range(5):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec = np.maximum(eps, vec)
        vec /= vec.sum()
        seeds.append(vec)

    # ------------------------------------------------------------------
    best_raw = float("inf")
    best_a = None

    # ------------------------------------------------------------------
    # Helper to evaluate raw score quickly
    def _record(a_vec):
        nonlocal best_raw, best_a
        raw = _raw_score(a_vec)
        if raw < best_raw:
            best_raw = raw
            best_a = a_vec.copy()
            print(f"New global best raw score: {best_raw:.6f}")

    # ------------------------------------------------------------------
    # Process each seed in turn; give each a fixed budget (roughly 80% of remaining)
    for seed_idx, a0 in enumerate(seeds):
        if time.time() >= deadline:
            break

        # Ensure strictly positive
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()

        # Record baseline
        _record(a0)

        # ---- Adam optimisation with continuation ----
        # initialise parameters
        mu = max(_max_self_convolution(a0) * 0.5, 1e-9)
        mu_min = 1e-12
        mu_factor = 0.8          # halve every 500 iters
        reduce_every = 500

        u = np.log(a0 + eps)     # unrolled parameters
        m = np.zeros_like(u)
        v = np.zeros_like(u)
        beta1 = 0.9
        beta2 = 0.999
        beta1_pow = 1.0
        beta2_pow = 1.0
        lr = 0.5                 # base learning rate for Adam
        eps_adam = 1e-8

        it = 0
        while time.time() < deadline:
            it += 1

            # softmax to get probability vector
            max_u = np.max(u)
            exp_u = np.exp(u - max_u)
            a = exp_u / exp_u.sum()

            # smoothed objective, gradient w.r.t. a, and max conv value
            f_val, grad_a, M = _smooth_obj_and_grad(a, mu)

            # gradient w.r.t. u (softmax trick)
            grad_u = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_u
            v = beta2 * v + (1 - beta2) * (grad_u * grad_u)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # periodic bookkeeping
            if (it % 200) == 0:

                mu = max(mu * mu_factor, mu_min)

                # record raw score (exact max)
                raw = 2.0 * n * M
                if raw < best_raw:
                    best_raw = raw
                    best_a = a.copy()
                    print(f"Iter {it} seed {seed_idx}: raw {raw:.6f} (mu={mu:.2e})")

            if mu <= mu_min and (it % reduce_every) == 0 and time.time() > deadline - 5.0:
                # We are at final smoothing, keep a few more steps then break
                pass

        # (already stored globally)

    # ------------------------------------------------------------------
    # Final polishing with sub‑gradient descent (short, uses remaining time)
    remaining = deadline - time.time()
    if remaining > 2.0 and best_a is not None:

    final_score = evaluate_sequence(best_a.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_a)})")

"""
Hybrid Adam‑soft‑max optimisation for the step‑function minimisation.

The algorithm:
1. Build a small library of probability‑vector seeds (uniform, arcsine,
   a handful of Beta shapes, random Dirichlet draws and the optional
   supplied `height_sequence_1`).
2. Pick the best seed w.r.t. the exact objective 2·n·max(a∗a).
3. Refine it with a continuation scheme on the smooth approximation
   F_μ(a)=μ·log Σ_k exp((a∗a)_k/μ).  The parameters are optimised using
   Adam on the log‑softmax parametrisation (u↦softmax(u)).  μ is
   multiplied by 0.8 every 500 iterations until it reaches 1e‑12.
4. Finish with a short projected sub‑gradient polish (the routine from
   the previous state‑of‑the‑art) to clean up the last few digits.
5. Return the probability vector; `evaluate_sequence` rescales it
   automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow / underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # cap the log‑pdf
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()

# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Exact objective for a normalised probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad, max_conv) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # values ≤ 1, safe
    Z = exp_scaled.sum()
    p = exp_scaled / Z                         # soft‑max weights, Σp = 1

    # gradient: 2 * (a_rev * p)   (central n entries)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    f_val = M + mu * np.log(Z)
    return f_val, grad, M

# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish a solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    momentum_coeff = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = momentum_coeff * momentum + (1.0 - momentum_coeff) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val

        vec = np.maximum(eps, vec)
        vec /= vec.sum()
        seeds.append(vec)

    # ------------------------------------------------------------------

        raw = _raw_score(a_vec)

    for idx, a_seed in enumerate(seeds):

        v = np.zeros_like(u)

    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        seeds.append(_beta_distribution(n, a, a))

    # d) a few random Beta draws
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta  = 10 ** rng.uniform(-2, 1)
        seeds.append(_beta_distribution(n, alpha, beta))

    # e) a few random Dirichlet draws
    for _ in range(5):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec = np.maximum(eps, vec)
        vec /= vec.sum()
        seeds.append(vec)

    # --------------------------------------------------------------
    # 2. Sort seeds by their exact raw score (cheapest first)
    seeds = sorted(seeds, key=_raw_score)

    # --------------------------------------------------------------
    # 3. Helper to keep the best solution seen so far
    best_raw = float("inf")
    best_a = None

    def _update_global(a_vec: np.ndarray):
        nonlocal best_raw, best_a
        raw = _raw_score(a_vec)
        if raw < best_raw:
            best_raw = raw
            best_a = a_vec.copy()
            print(f"Global best raw score updated to {best_raw:.6f}")

    # --------------------------------------------------------------
    # 4. Optimise each seed (the first one will usually consume the whole
    #    budget – it is the best seed after sorting)
    for seed_idx, a_seed in enumerate(seeds):
        if time.time() >= deadline:
            break

        a0 = np.maximum(a_seed, eps)
        a0 /= a0.sum()
        _update_global(a0)

        # ----- Adam optimisation with continuation (smooth max) -----
        mu = max(_max_self_convolution(a0) * 0.5, 1e-9)
        mu_min = 1e-12
        mu_factor = 0.8                 # multiply every `reduce_every` iters
        reduce_every = 200

        # log‑softmax parameters
        u = np.log(a0 + eps)
        m = np.zeros_like(u)
        v = np.zeros_like(u)
        beta1, beta2 = 0.9, 0.999
        beta1_pow, beta2_pow = 1.0, 1.0
        lr = 0.5
        eps_adam = 1e-8

        it = 0
        while time.time() < deadline:
            it += 1

            # softmax → probability vector
            max_u = np.max(u)
            exp_u = np.exp(u - max_u)
            a = exp_u / exp_u.sum()

            # smoothed objective, its gradient w.r.t. a, and exact max conv
            f_val, grad_a, M = _smooth_obj_and_grad(a, mu)

            # gradient w.r.t. u (softmax trick)
            grad_u = a * (grad_a - np.dot(grad_a, a))

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_u
            v = beta2 * v + (1 - beta2) * (grad_u * grad_u)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            u -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # bookkeeping
            if (it % 200) == 0:
                mu = max(mu * mu_factor, mu_min)
                raw_exact = 2.0 * n * M
                if raw_exact < best_raw:
                    best_raw = raw_exact
                    best_a = a.copy()
                    print(f"Seed {seed_idx} iter {it}: raw {best_raw:.6f} (mu={mu:.2e})")

            if (it % reduce_every) == 0 and time.time() >= deadline:
                break

    # --------------------------------------------------------------
    # 5. Final short sub‑gradient polishing if time permits
    remaining = deadline - time.time()
    if remaining > 2.0 and best_a is not None:
        print(f"Starting sub‑gradient polish, time left = {remaining:.1f}s")
        best_a, best_raw = _subgradient_polish(best_a, remaining)

    # --------------------------------------------------------------
    # 6. Return the sequence (evaluate_sequence rescales it automatically)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_a)})")
    return best_a.tolist()
