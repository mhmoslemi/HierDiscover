def _beta_start(n, alpha, beta):
    x = (np.arange(1, n+1) / n)
    cdf = betainc(alpha, beta, x)   # shape (n,)
    # prefix diff for mass in each bin:
    a = np.empty(n)
    a[0] = cdf[0]
    a[1:] = cdf[1:] - cdf[:-1]
    # Ensure nonnegative
    a = np.maximum(a, 0)
    s = a.sum()
    if s>0:
        a /= s
    else:
        a[:] = 1.0/n
    return a

def _arcsine_start_exact(n):
    # Use CDF of arcsine distribution F(x) = (2/π)*arcsin(sqrt(x))
    idx = np.arange(1, n+1) / n
    cdf = (2/np.pi) * np.arcsin(np.sqrt(idx))
    a = np.empty(n)
    a[0] = cdf[0]
    a[1:] = cdf[1:] - cdf[:-1]
    return a

```
def _gaussian_start(n, sigma):
    i = (np.arange(n) + 0.5) / n
    vals = np.exp(-0.5 * ((i - 0.5) / sigma)**2)
    vals = np.maximum(vals,0)
    s = vals.sum()
    if s>0:
        vals /= s
    else:
        vals[:] = 1.0/n
    return vals
```

def _mixture_start(base, mix, gamma):

        cand_scores.append(raw)

    # Sort and keep top K (say 8)
    top_k = 8
    sorted_idx = np.argsort(cand_scores)
    top_idxs = sorted_idx[:top_k]

    best_seq = None
    best_score = float('inf')
    eps = 1e-12

    for idx in top_idxs:
        if time.time() >= deadline:
            break
        a0 = candidates[idx]
        logits = np.log(a0 + eps)
        # Run Adam
        seq, score = _adam_optimize_smoothed(
            logits, n, deadline,
            max_iters=8000,
            lr0=0.15,
            lam_start=0.3,
            lam_factor=1.0015
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    # Also maybe evaluate the raw best candidate (without Adam) in case all Adam runs didn't improve further
    # Already best_seq may be None if no time; fallback uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)

    # Final subgradient refinement
    a = best_seq.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
        # Step size schedule: 0.05 / sqrt(it)
        step = 0.05 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional jitter
        if it % 1000 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} raw={best_score:.6f}")

    # Final evaluation
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list

"""
Propose a step function (non‑negative heights) that minimises
evaluate_sequence.  The method builds several analytically‑motivated
starting shapes (exact arcsine, mixtures with uniform, symmetric Beta,
Gaussian), evaluates their raw scores, refines the most promising
candidates with a smooth‑max Adam optimiser and finishes with a
projected sub‑gradient polishing phase.  The search adapts to the
available time budget and aims for a score below 1.503.
"""
import time
import math
import numpy as np

# optional SciPy for Beta distributions
try:
    from scipy.special import betainc
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _arcsine_start_exact(n: int) -> np.ndarray:
    """
    Discrete arcsine distribution obtained by integrating the continuous
    arcsine density over each bin.
    """
    # CDF of the arcsine law: F(x) = (2/π) * arcsin(√x)
    x = (np.arange(1, n + 1, dtype=float) / n)
    cdf = (2.0 / np.pi) * np.arcsin(np.sqrt(x))
    a = np.empty(n, dtype=float)
    a[0] = cdf[0]
    a[1:] = cdf[1:] - cdf[:-1]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a[:] = 1.0 / n
    return a


def _beta_start(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta distribution on [0,1] discretised onto n bins.
    Requires SciPy; if SciPy is unavailable a uniform vector is returned.
    """
    if not _SCIPY_AVAILABLE:
        return np.full(n, 1.0 / n, dtype=float)

    x = (np.arange(1, n + 1, dtype=float) / n)
    cdf = betainc(alpha, beta, x)  # regularised incomplete beta
    a = np.empty(n, dtype=float)
    a[0] = cdf[0]
    a[1:] = cdf[1:] - cdf[:-1]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a[:] = 1.0 / n
    return a


def _gaussian_start(n: int, sigma: float) -> np.ndarray:
    """Discrete truncated Gaussian (centered at 0.5) on n bins."""
    i = (np.arange(n, dtype=float) + 0.5) / n
    vals = np.exp(-0.5 * ((i - 0.5) / sigma) ** 2)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _mixture_start(base: np.ndarray, mix: np.ndarray, gamma: float) -> np.ndarray:
    """
    Convex combination (1‑γ)·base + γ·mix, re‑normalised to sum to one.
    """
    a = (1.0 - gamma) * base + gamma * mix
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a[:] = 1.0 / len(a)
    return a


def _random_dirichlet_start(n: int,
                            alpha_low: float = 0.1,
                            alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 6000,
                            lr0: float = 0.15,
                            lam_start: float = 0.3,
                            lam_factor: float = 1.0015) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logit parametrisation,
    minimising the smoothed log‑sum‑exp of the self‑convolution.
    Returns the best probability vector found and its raw (non‑smoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector (simplex)
        a = _softmax(x)                     # Σa = 1, a ≥ 0

        # raw objective
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.3f}")

        # smoothed max via log‑sum‑exp
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()          # shape (2n‑1,)

        # gradient w.r.t. a : ∇_a f_λ = 2 * (a_rev * p) (restricted)
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase smoothing parameter
        lam = lam * lam_factor

        # occasional jitter to perturb plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:                     # should never happen
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()
    return best_a, best_score
