def _smooth_adam_opt(z0, deadline, max_iters=200_000, name=''):
    z = z0.copy()
    n = z0.shape[0]
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    step0 = 0.2
    t = 0
    best_a = None
    best_score = float('inf')
    # schedule: list of (alpha, iters_fraction)
    schedule = [(5, 0.2), (15, 0.3), (30, 0.3), (60, 0.2)]
    for alpha, frac in schedule:
        iters_stage = int(max_iters * frac)
        for _ in range(iters_stage):
            if time.time() >= deadline:
                break
            t += 1
            # compute current a
            a = np.exp(z - np.max(z))
            a /= a.sum()
            # compute conv and cheap score
            b = _fft_self_convolve(a)
            cur_score = 2 * n * b.max()
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
                # logging
                print(f"[{time.time():.1f}] {name} iter {t} score {cur_score:.6f}")
            # compute smooth gradient
            expo = alpha * b
            # stable softmax for w
            max_expo = np.max(expo)
            w = np.exp(expo - max_expo)
            w_sum = w.sum()
            w = w / w_sum
            # gradient w.r.t a
            a_rev = a[::-1]
            conv_full = _fft_convolve(w, a_rev)
            grad_a = 2.0 * conv_full[n-1 : n-1 + n]
            # gradient w.r.t z
            dot = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = step0 / (np.sqrt(v_hat) + 1e-8)
            z -= step * m_hat
    if best_a is None:
        best_a = np.exp(z - np.max(z))
        best_a /= best_a.sum()
        best_score = 2 * n * _fft_self_convolve(best_a).max()
    return best_a, best_score

"""
Improved step‑function optimizer.

Main ideas
----------
1. **Broad seeding** – generate a diverse set of analytic distributions
   (U‑shaped beta, arcsine, triangular, sine, cosine, exponential, uniform)
   and a few random Dirichlet draws.  All are normalised to the simplex.

2. **Smoothed Adam on the simplex** – minimise a smooth approximation of the
   true objective  (log‑sum‑exp of the self‑convolution) using the
   soft‑max parametrisation  a = softmax(z).  The schedule
   α = 5 → 15 → 30 → 60 gradually tightens the approximation to the exact
   max.  Adam’s adaptive steps together with the simplex‑preserving
   parametrisation give a faster escape from the flat regions that
   sub‑gradient descent struggles with.

3. **Final hill‑climbing** – tiny random mass transfers (and occasional jitter)
   polish the best sequence found by the smoothed optimisation.

The code respects the 1000 s budget, works with any previously discovered
sequence available as ``height_sequence_1`` and always returns a list of
non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / e.sum()


# ----------------------------------------------------------------------
# FFT‑based convolutions
# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Cheap objective (exact when sum(a)=1)
# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (no lambdas as required)
# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution_003(n: int) -> np.ndarray:
    """Exponential with λ=0.003 centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.003 * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_opt(z0: np.ndarray,
                     deadline: float,
                     max_iters: int = 200_000,
                     name: str = "") -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *smooth* approximation
        Fα(a) = (1/α)·log∑_k exp(α·(a*a)_k)
    where a = softmax(z).  The parameter α follows a schedule
    [5,15,30,60] to tighten the approximation.
    Returns (best_a, best_score) where best_score is the exact cheap score.
    """
    z = z0.copy()
    n = z0.shape[0]

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    step0 = 0.2

    t = 0
    best_a = None
    best_score = float("inf")

    schedule = [(5, 0.2), (15, 0.3), (30, 0.3), (60, 0.2)]

    for alpha, frac in schedule:
        iters_stage = int(max_iters * frac)
        for _ in range(iters_stage):
            if time.time() >= deadline:
                break
            t += 1

            # ----- a = softmax(z) -----
            a = _softmax(z)

            # ----- exact cheap score (for bookkeeping) -----
            b = _fft_self_convolve(a)
            cur_score = 2.0 * n * b.max()
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.time():.1f}] {name} iter {t:6d} α={alpha:<2d} score={cur_score:.6f}")

            # ----- smooth weights w = softmax(α·b) -----
            expo = alpha * b
            max_expo = np.max(expo)
            w = np.exp(expo - max_expo)
            w /= w.sum()                     # now sum(w)=1

            # ----- gradient wrt a (exact) -----
            a_rev = a[::-1]
            conv_full = _fft_convolve(w, a_rev)          # length 3n‑2
            grad_a = 2.0 * conv_full[n - 1 : n - 1 + n]   # shape (n,)

            # ----- chain rule: ∂F/∂z = a * (grad_a - <a,grad_a>) -----
            dot = np.dot(a, grad_a)
            grad_z = a * (grad_a - dot)

            # ----- Adam update on z -----
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = step0 / (np.sqrt(v_hat) + 1e-8)
            z -= step * m_hat

    # fallback in the unlikely case that no improvement was recorded
    if best_a is None:
        a = _softmax(z)
        best_a = a
        best_score = 2.0 * n * _fft_self_convolve(a).max()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine:
        * builds several analytic / random seeds,
        * refines each seed with the smoothed Adam optimiser,
        * finishes with a tiny hill‑climbing phase.
    Returns the best sequence found (a list of floats, already normalised).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # -------------------------------------------------------------
    # Choose a working length (use previous construction if present)
    # -------------------------------------------------------------
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 4000
    n = max(500, min(8000, base_n))

    # -------------------------------------------------------------
    # 1. Broad sampling of analytic families & random draws
    # -------------------------------------------------------------
    seeds = []   # (array, cheap_score, name)

    def _store(arr: np.ndarray, name: str) -> None:
        arr = np.maximum(arr, 0.0)
        if arr.sum() == 0:
            return
        arr = arr / arr.sum()
        cheap = _max_convolution_score(arr)
        seeds.append((arr, cheap, name))
        print(f"[{time.time():.1f}] seed {name:12s} n={n:4d} cheap={cheap:.6f}")

    _store(_arcsine_distribution(n), "arcsine")
    _store(_triangular_distribution(n), "triangular")
    _store(_exponential_distribution_003(n), "exponential")
    _store(_sine_distribution(n), "sine")
    _store(_cosine_distribution(n), "cosine")
    _store(_uniform_distribution(n), "uniform")

    beta_ps = [0.02, 0.05, 0.08, 0.12, 0.20, 0.30, 0.50, 1.0, 2.0, 3.0]
    for p in beta_ps:
        _store(_beta_symmetric_distribution(n, p), f"beta_{p:.2f}")

    for _ in range(6):
        _store(_random_dirichlet(n), "rand_dir")

    # also keep the previously discovered best, if any
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        _store(prev, "prev")
    except Exception:
        pass

    # -------------------------------------------------------------
    # 2. Choose a few promising seeds
    # -------------------------------------------------------------
    seeds.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = seeds[:top_k]
    print(f"Selected {len(top_seeds)} seeds for refinement, best cheap={top_seeds[0][1]:.6f}")

    # -------------------------------------------------------------
    # 3. Smoothed Adam optimisation (one seed at a time)
    # -------------------------------------------------------------
    time_for_opt = max(0.0, deadline - time.time() - 30.0)   # reserve ~30 s for final hill‑climb
    if time_for_opt <= 0:
        time_for_opt = 0.0

    best_seq = None
    best_score = float("inf")

    # allocate roughly equal time to each seed
    per_seed_time = time_for_opt / max(1, len(top_seeds))

    for arr, cheap, name in top_seeds:
        if time.time() >= deadline - 30.0:
            break

        sub_deadline = time.time() + per_seed_time
        # initialise logits from the seed (log‑softmax)
        z0 = np.log(arr + 1e-12)

        # crude estimate of iterations (≈0.001 s per iteration for n≈4000)
        est_iters = int(per_seed_time / 0.0015)
        est_iters = max(est_iters, 5_000)          # ensure a minimal effort

        seq, sc = _smooth_adam_opt(z0, sub_deadline,
                                   max_iters=est_iters,
                                   name=name)
        if sc < best_score - 1e-14:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] seed {name} improves global best to {sc:.6f}")

    if best_seq is None:                         # fallback (should never happen)
        best_seq = top_seeds[0][0].copy()
        best_score = top_seeds[0][1]

    # -------------------------------------------------------------
    # 4. Fine‑grained hill‑climbing
    # -------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    max_transfer = 0.02   # start with up to 2 % of total mass per move
    it = 0

    while time.time() < deadline:
        it += 1
        # transfer amount shrinks geometrically
        transfer = max_transfer * (0.5 ** (it / 8_000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to jump off flat plateaus
        if it % 5_000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # return a probability vector – the evaluator will rescale it internally
    return best_seq.tolist()
