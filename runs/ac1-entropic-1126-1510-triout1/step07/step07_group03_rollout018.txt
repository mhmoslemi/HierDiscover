"""
Propose a non‑negative step‑function (probability vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The method is a hybrid of two ideas:

1. **Linear‑programming “flattening” step** –  starting from a
   smooth analytic distribution (arcsine), we solve

        max   Σ g_j                (total mass of g)
        s.t.  conv(f,g)_k ≤ M      for all k = 0…2n‑2
              g ≥ 0

   where f is the current distribution and M = max(conv(f,f)).
   The LP pushes mass into positions where f is small without
   increasing the current worst‑case convolution value.
   The resulting g is mixed into f with a decreasing step size.
   Repeating this a few dozen times already drives the objective
   well below the uniform value (≈2).

2. **Fine‑tuning with smoothed Adam descent** – after the LP phase a
   few hundred Adam steps on a log‑sum‑exp smooth approximation of
   the max‑convolution are performed (the code is the same as in the
   previous solution).  The simplex constraint (non‑negative,
   unit sum) is enforced after each update.

All work respects a global deadline (default 1000 s).  The best vector
found is returned as a plain Python list of floats.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize
import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (only the one we really need, others kept for
# possible random‑restart seeds)
def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_random_sparse(n: int, density: float = 0.005) -> np.ndarray:
    """Random sparse vector (density ≈ 0.5 %)."""
    k = max(2, int(density * n))
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _lp_flatten_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    One LP step:
        maximise Σ g_j  s.t.  conv(f,g) ≤ M   (element‑wise) , g ≥ 0.
    Returns the optimal g (already normalised to sum 1).
    """
    n = f.shape[0]
    # Build the convolution matrix A (shape (2n‑1) × n) such that
    # A @ g == conv(f,g)
    col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = scipy.linalg.toeplitz(col, row)

    c = -np.ones(n, dtype=np.float64)            # maximise sum(g)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = scipy.optimize.linprog(
        c, A_ub=A, b_ub=b_ub, bounds=bounds,
        method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9}
    )
    if not res.success:
        # In the very unlikely case of a failure fall back to doing nothing.
        raise RuntimeError(f"LP failed: {res.message}")

    g = res.x
    if g.sum() == 0:
        raise RuntimeError("LP returned zero vector")
    return g / g.sum()


def _lp_flattening(f: np.ndarray, deadline: float,
                   best_seq: np.ndarray, best_score: float,
                   max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP flattening step, mixing the result back into f.
    Mixing factor τ starts at 0.5 and decays geometrically.
    """
    n = f.shape[0]
    cur = f.copy()
    tau0 = 0.5
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = _fft_self_convolve(cur).max()
        try:
            g = _lp_flatten_step(cur, M)
        except Exception as e:
            print(f"[{time.time():.2f}] LP step failed at iter {it}: {e}")
            break

        # geometric decay of mixing weight
        tau = max(tau0 * (0.9 ** it), 0.01)
        cur = (1.0 - tau) * cur + tau * g
        cur = _project_onto_simplex(cur)

        sc = _objective(cur)          # sum(cur)==1 ⇒ denominator =1
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = cur.copy()
            print(f"[{time.time():.2f}] LP iter {it} new best {best_score:.6f}")

        # early exit when nothing improves for a while
        if it % 5 == 4:
            # try a tiny step; if objective does not improve, stop
            test = (1.0 - tau) * cur + tau * cur
            test = _project_onto_simplex(test)
            if _objective(test) >= sc - 1e-12:
                break
    return best_seq, best_score

def _smooth_adam_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Adam optimisation on a log‑sum‑exp smooth approximation of max‑convolution."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    schedule = [
        (1.0, 1000),
        (5.0, 1000),
        (20.0, 1000),
        (100.0, 1000),
        (500.0, 1000),
        (2000.0, 1000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth max (size n)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + simplex projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a)

            if t % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-12:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] Adam α={alpha:.0f} step={t} best {best_score:.6f}")

    return best_seq, best_score

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a modest length – large enough to give freedom but small
    # enough that the LP matrix fits comfortably in memory.
    n = 1200

    # ------------------------------------------------------------------
    # Initial candidate: arcsine shape (good start for the LP phase)
    a = _distribution_arcsine(n)
    a = a / a.sum()

    overall_best_seq = a.copy()
    overall_best_score = _objective(a)
    print(f"[{time.time():.2f}] init length={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – linear‑programming flattening (multiple warm‑starts)
    # we perform a short LP loop, then a short random‑restart loop.
    a, overall_best_score = _lp_flattening(
        a, deadline, best_seq=overall_best_seq, best_score=overall_best_score,
        max_iters=25
    )

    # ------------------------------------------------------------------
    # Phase 2 – smoothed Adam descent for fine‑tuning
    a, overall_best_score = _smooth_adam_descent(
        a, deadline, best_seq=overall_best_seq, best_score=overall_best_score
    )
    overall_best_seq = a.copy()

    # ------------------------------------------------------------------
    # Additional random‑restart jitter if time permits (cheap perturbations)
    while time.time() < deadline:
        # create a tiny random perturbation and polishing it
        jitter = np.random.randn(n) * 1e-3
        cand = a + jitter
        cand = _project_onto_simplex(cand)

        # quick Adam run (a few hundred steps) on the perturbed start
        cand, cand_score = _smooth_adam_descent(
            cand, deadline, best_seq=cand, best_score=_objective(cand)
        )
        if cand_score < overall_best_score - 1e-12:
            overall_best_score = cand_score
            overall_best_seq = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] random jitter improved to {overall_best_score:.6f}")
        else:
            # no improvement – stop the cheap loop
            break

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
