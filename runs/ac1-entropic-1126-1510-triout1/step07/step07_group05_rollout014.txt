"""
Propose a step function with a low evaluation score.

Algorithm
---------
* Work on the probability simplex (a_i ≥ 0, Σ a_i = 1) – scaling of the
  sequence does not change the objective, but fixing Σa_i = 1 makes the
  optimisation easier.
* For a given length n we minimise the quantity 2·n·max_k (a*a)_k.
  The maximum of the self‑convolution is a convex function of a, therefore
  a (projected) sub‑gradient method converges.
* We use multiplicative (exponentiated) updates – i.e. mirror descent on the
  simplex – which often gives faster progress than an additive projected
  gradient.
* Each sub‑gradient is obtained from the active set of shifts attaining the
  current maximum.  The gradient is computed efficiently with two FFTs.
* Several random initialisations (uniform, Dirichlet with different
  concentrations and sparse “spiky’’ draws) are tried; the best sequence
  found is kept.
* After the mirror‑descent phase a short projected sub‑gradient refinement
  (the routine used in the reference solution) cleans the remaining
  plateau.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – a tiny tolerance scaled with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _mirror_descent_optimize(initial_a: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 30000,
                             eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) minimiser on the simplex.
    Returns the best a encountered and its *raw* score 2·n·max(a*a).
    """
    # keep everything in log‑space for numerical stability
    log_a = np.log(initial_a + 1e-30)
    best_a = None
    best_raw = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # actual probabilities
        a = np.exp(log_a - np.max(log_a))      # shift for stability
        a /= a.sum()

        # convolution and current raw score
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw = 2.0 * n * cur_max
        if raw < best_raw:
            best_raw = raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_raw:.6f}")

        # sub‑gradient of the max‑convolution
        grad, _ = _max_subgradient_step(a)

        # step‑size schedule (annealed)
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update in log‑space
        log_a -= eta * grad

        # occasional small random jitter to escape plateaus
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-6
            log_a += jitter

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_full(best_a).max()
    return best_a, best_raw


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent with exact max‑convolution.
    Returns refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_full(a).max()
    it = start_iter
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional tiny random perturbation
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals
