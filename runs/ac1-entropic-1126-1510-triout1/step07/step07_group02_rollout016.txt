"""
Search for a non‑negative step function (finite vector) that approximately
minimises the black‑box ``evaluate_sequence`` function.  The optimisation
pipeline is:

1.  Scan a wide collection of analytic families (uniform, arcsine, beta,
    exponential, power‑law, sine‑power, cosine windows) together with a few
    random Dirichlet draws for several lengths.  The best of these cheap
    candidates seeds the subsequent optimisation.

2.  Multi‑phase Adam descent on a smooth approximation of the
    ``max``‑convolution.  The softness parameter ``alpha`` is increased
    gradually, the step size is adapted by Adam and each iterate is
    projected onto the probability simplex.

3.  Exact max‑sub‑gradient descent (the same update that was used in the
    reference implementation) to push the solution further.

4.  A few diffusion‑smoothing passes (convolution with a small kernel) –
    this often reduces the peak of the self‑convolution without destroying
    the structure found by the gradient steps.

5.  Simulated‑annealing hill‑climbing that works directly on the full
    self‑convolution.  A mass transfer ``delta`` from a source index to a
    target index is applied with an O(n) incremental update of the
    convolution, and the move is accepted either if it improves the
    objective or with a temperature‑controlled probability.

All steps respect a hard time budget (default 1000 s).  The routine returns
the best sequence (as a plain list of floats) found within the budget.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_power_law(n, p):
    """Power‑law decay a_i ∝ 1/(i+1)^p, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = 1.0 / (idx ** p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()                     # linear part uses the *old* a
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta          # quadratic self‑term
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (approximately) minimises
    ``evaluate_sequence``.  The routine combines a large pool of analytic
    families, a multi‑phase Adam / sub‑gradient optimisation, optional
    diffusion smoothing, and a simulated‑annealing hill‑climber that works
    directly on the self‑convolution using O(n) incremental updates.
    It respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1. Scan analytic families + a few random draws for several lengths.
    lengths = [2000, 3000, 4000, 5000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break

        cand = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.003),
            _distribution_exponential(n, lam=0.02),
        ]

        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            cand.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            cand.append(_distribution_sine_power(n, pw))

        for c in (0.5, 0.46, 0.42):
            cand.append(_distribution_cosine_window(n, c))

        for p in (0.5, 1.0, 1.5, 2.0):
            cand.append(_distribution_power_law(n, p))

        # sparse random seed
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        cand.append(sparse)

        # a few Dirichlet draws
        for _ in range(3):
            x = np.random.gamma(0.5, 1.0, size=n)
            x /= x.sum()
            cand.append(x)

        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. If a previous solution exists, use it as an extra seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # Fallback – should never happen.
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Adam descent with a gradually sharpening soft‑max.
    a = best_seq.copy()
    n = a.shape[0]

    schedule = [
        (1.0,   2000),
        (5.0,   2000),
        (20.0,  4000),
        (100.0, 4000),
        (500.0, 4000),
        (2000.0,3000),
        (5000.0,2000)
    ]

    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_seq_adam = a.copy()
    best_score_adam = _max_convolution_score(a)

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if t % 250 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score_adam - 1e-12:
                    best_score_adam = cur
                    best_seq_adam = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

    a = best_seq_adam.copy()
    best_seq = a.copy()
    best_score = best_score_adam

    # ------------------------------------------------------------------
    # 4. Exact max‑sub‑gradient refinement.
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cur:.6f}")

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 5. Optional diffusion smoothing (8 short passes).
    if time.time() < deadline:
        kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
        a = best_seq.copy()
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, eps)
            a = a / a.sum()
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Diffusion smoothing new best {cur:.6f}")

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing hill‑climber on the full convolution.
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    cur_score = 2.0 * n * M
    best_score = cur_score
    best_seq = a.copy()
    it = 0
    it_no_improve = 0
    T = cur_score * 0.05          # initial temperature (≈5 % of current score)
    decay = 0.9995                # exponential cooling
    max_delta_factor = 0.02       # move at most 2 % of source mass each step

    while time.time() < deadline:
        it += 1
        # pick a source index proportional to its current mass
        source = np.random.choice(n, p=a)
        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * max_delta_factor, 0.05)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()

        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        new_score = 2.0 * n * new_M

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_score = new_score
            M = new_M
            it_no_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {new_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        T *= decay

        # occasional jitter if we are stuck
        if it % 5000 == 0 and it_no_improve > 2000:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            cur_score = 2.0 * n * M
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at SA iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
