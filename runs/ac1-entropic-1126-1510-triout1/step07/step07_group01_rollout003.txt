"""
Hybrid optimizer for the non‑negative step‑function problem.

Strategy
--------
1. **Rich seeding** – generate many analytic families (arcsine, beta,
   triangular, exponential, sine, cosine, uniform, power‑law) for a few
   plausible lengths together with random Dirichlet draws.  The cheap
   objective ``2·n·max(conv)`` (sum = 1) is computed for each candidate.

2. **Smoothed Adam descent** – starting from the best seeds we minimise a
   smooth approximation of the max‑convolution:
        S_T(a) = T·log∑_k exp(conv_k / T) .
   The gradient of S_T is obtained analytically via FFT‑based
   convolutions.  A temperature schedule (geometric cooling) turns the
   smooth problem into the original non‑smooth one.

3. **Population evolution** – the top refined vectors form a small
   population.  New individuals are created by convex combinations of
   two parents plus a tiny Gaussian jitter, projected back onto the
   probability simplex.  The population is iteratively improved by
   replacing its worst member with a better offspring.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a feasible sequence (non‑negative, sum > 0.01).  The
objective is scale‑invariant, therefore the returned vector is kept on
the simplex.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a uniform Dirichlet point on the simplex."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) density on the discrete simplex."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _powerlaw_edge_distribution(n: int, alpha: float) -> np.ndarray:
    """Weight ∝ ((i+0.5)*(n-i-0.5))^{-α} (α>0)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i * (n - i + 0.5)) ** (-alpha)
    return w / w.sum()

def _adam_subgrad_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 150_000,
                         step0: float = 0.2,
                         beta1: float = 0.9, beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimiser on the *exact* max‑convolution (sub‑gradient)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    n = len(a)
    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient (same routine as earlier code)
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps)
        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
        even = (active % 2 == 0)
        diag_idx = active[even] // 2
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_smooth_descent(a0: np.ndarray, deadline: float,
                         init_temp: float,
                         temp_decay: float = 0.92,
                         steps_per_temp: int = 4000,
                         max_iters: int = 200_000,
                         step0: float = 0.15,
                         beta1: float = 0.9,
                         beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smoothed version of the max‑convolution:
        S_T(a) = T·log∑ exp(conv_k / T) .
    Temperature is reduced geometrically after ``steps_per_temp`` steps.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    n = len(a)
    temp = max(init_temp, 1e-12)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # compute convolution and its smoothed weights
        conv = _fft_self_convolve(a)
        # stable softmax
        c_max = conv.max()
        exp_vals = np.exp((conv - c_max) / temp)
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth objective
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # evaluate cheap (exact) score
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()

        # cool down temperature
        if it % steps_per_temp == 0:
            temp *= temp_decay
            if temp < 1e-12:
                temp = 1e-12
    return best_a, best_score


# ----------------------------------------------------------------------
def _evolve_population(pop: list[tuple[np.ndarray, float]],
                       deadline: float,
                       jitter_scale: float = 5e-5,
                       replace_prob: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Simple evolutionary improvement.
    ``pop`` is a list of (vector, cheap_score) tuples; all vectors have the same length.
    """
    if not pop:
        raise ValueError("population empty")
    n = pop[0][0].size
    # pre‑compute selection probabilities (better = higher chance)
    scores = np.array([s for _, s in pop])
    # invert scores to get weights (add small epsilon for stability)
    weights = 1.0 / (scores + 1e-15)
    weights /= weights.sum()

    best_vec, best_score = min(pop, key=lambda x: x[1])

    while time.time() < deadline:
        # choose two parents
        idx = np.random.choice(len(pop), size=2, replace=False, p=weights)
        a1, _ = pop[idx[0]]
        a2, _ = pop[idx[1]]
        # convex blend
        w = np.random.rand()
        child = w * a1 + (1.0 - w) * a2
        # small jitter
        if np.random.rand() < replace_prob:
            child += np.random.randn(n) * jitter_scale
        child = _project_onto_simplex(child)

        child_score = _max_convolution_score(child)

        # possibly replace the worst individual
        worst_idx = np.argmax([s for _, s in pop])
        if child_score < pop[worst_idx][1]:
            pop[worst_idx] = (child, child_score)
            # update best tracking
            if child_score < best_score:
                best_score = child_score
                best_vec = child.copy()
            # recompute selection weights periodically
            scores = np.array([s for _, s in pop])
            weights = 1.0 / (scores + 1e-15)
            weights /= weights.sum()
    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # small safety margin

    # ------------------------------------------------------------------
    # 0. Determine a sensible base length
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # 1. Build a diverse set of analytic candidates
    lengths = set()
    for f in (0.8, 0.9, 1.0, 1.1, 1.2):
        lengths.add(int(base_n * f))
    for _ in range(3):
        lengths.add(np.random.randint(500, min(12000, base_n * 3) + 1))
    lengths = sorted(lengths)

    candidates: list[tuple[np.ndarray, float]] = []

    beta_ps = [0.3, 0.4, 0.5, 0.6, 0.8, 1.0, 1.2]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- analytic families ---------------------------------------
        fams = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.003), "exponential"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (lambda m: np.full(m, 1.0 / m), "uniform"),
            (lambda m: _powerlaw_edge_distribution(m, alpha=0.4), "powerlaw0.4"),
            (lambda m: _powerlaw_edge_distribution(m, alpha=0.6), "powerlaw0.6")
        ]
        for gen, name in fams:
            a = gen(n)
            sc = _max_convolution_score(a)
            candidates.append((a, sc))
            print(f"[{time.time():.1f}] n={n:4d} {name:12s} score={sc:.6f}")

        # ----- symmetric beta family ------------------------------------
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            sc = _max_convolution_score(a)
            candidates.append((a, sc))
            print(f"[{time.time():.1f}] n={n:4d} beta(p={p:.2f}) score={sc:.6f}")

        # ----- random Dirichlet draws ----------------------------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            a = _random_dirichlet(n)
            sc = _max_convolution_score(a)
            candidates.append((a, sc))
            print(f"[{time.time():.1f}] n={n:4d} random Dirichlet score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Include the previously best known construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            candidates.append((prev, sc))
            print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 15
    top_seeds = candidates[:top_k]
    print(f"Selected {len(top_seeds)} seeds for intensive refinement; best cheap score={top_seeds[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy refinement (Adam on exact sub‑gradient, then smoothed Adam)
    best_vec = top_seeds[0][0].copy()
    best_score = top_seeds[0][1]

    for idx, (seed_seq, _) in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # ---- Adam on the exact max (sub‑gradient) --------------------
        sub_deadline = deadline - 60.0          # reserve time for later phases
        seq, sc = _adam_subgrad_descent(seed_seq, sub_deadline,
                                        max_iters=80_000, step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_vec = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} Adam‑subgrad improved to {sc:.6f}")

        # ---- Smoothed Adam (annealed temperature) -------------------
        # initialise temperature using current max‑convolution
        init_temp = _fft_self_convolve(seq).max() * 0.3 + 1e-12
        seq2, sc2 = _adam_smooth_descent(seq, deadline,
                                         init_temp=init_temp,
                                         temp_decay=0.93,
                                         steps_per_temp=3000,
                                         max_iters=120_000,
                                         step0=0.18)
        if sc2 < best_score - 1e-14:
            best_score, best_vec = sc2, seq2.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_seeds)} smoothed‑Adam improved to {sc2:.6f}")

    # ------------------------------------------------------------------
    # 5. Evolutionary refinement on the best length
    n_best = best_vec.shape[0]
    pop_size = 30
    population: list[tuple[np.ndarray, float]] = []

    # fill with refined best, other top seeds of same length, and random draws
    population.append((best_vec, _max_convolution_score(best_vec)))
    for a, _ in top_seeds[1:]:
        if a.shape[0] == n_best:
            population.append((a, _max_convolution_score(a)))
        if len(population) >= pop_size:
            break
    while len(population) < pop_size and time.time() < deadline:
        a = _random_dirichlet(n_best)
        population.append((a, _max_convolution_score(a)))

    print(f"Starting population evolution (size={len(population)}) on length {n_best}")

    evo_deadline = deadline - 0.3   # leave a tiny margin for return
    best_vec, best_score = _evolve_population(population, evo_deadline,
                                              jitter_scale=4e-5)

    # ------------------------------------------------------------------
    # 6. Final tiny hill‑climb to clean up any remaining flatness
    a = best_vec.copy()
    cur_score = best_score
    iteration = 0
    max_transfer = 0.025

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 10000.0))

        i = np.random.randint(n_best)
        j = np.random.randint(n_best - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if iteration % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n_best) * 6e-5
            a = best_vec + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={n_best})")
    # The optimizer works on the simplex, which is scale‑invariant for
    # ``evaluate_sequence``.  Return the probability vector.
    return best_vec.tolist()
