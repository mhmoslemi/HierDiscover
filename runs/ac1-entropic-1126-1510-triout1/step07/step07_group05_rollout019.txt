def _build_conv_matrix(a):
    n = len(a)
    row = []
    col = []
    data = []
    for i in range(n):
        fi = a[i]
        # for each j index in b, row = i + j
        # vectorized: rows = i + np.arange(n)
        # columns = np.arange(n)
        # data = np.full(n, fi)
        rows = i + np.arange(n)
        cols = np.arange(n)
        row.extend(rows.tolist())
        col.extend(cols.tolist())
        data.extend([fi]*n)
    A = scipy.sparse.coo_matrix((data, (row, col)), shape=(2*n-1,n))
    return A.tocsr()
```

def _lp_optimize(a, M):
    n = len(a)
    A = _build_conv_matrix(a)
    ub = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n)  # maximize sum(b)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        return None, None
    b_opt = res.x
    sum_b = b_opt.sum()
    return b_opt, sum_b

def _max_convolution(a):

    return 2.0 * n * _max_convolution(a)

    s = v.sum()
    if s==0: return v
    return v / s

"""
Propose a candidate step function.

The search works on a small collection of simple parametric families
(linear, polynomial, exponential, Dirichlet draws, etc.).  Each
candidate is normalised to sum to one, its raw score
    raw = 2·n·max(conv(a,a))
is evaluated via an FFT, and the best candidate seen so far is kept.
Finally a short projected‑sub‑gradient refinement (the same routine
used by the reference solution) is applied to the best candidate.
All helper functions are top‑level – no closures – and the total run
time stays well below the 1000 s budget.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for the exact max‑convolution.
    Returns refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)
    it = start_iter
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Raw objective 2·n·max(a*a) for a normalised sequence (sum=1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)


def _make_candidate(shape: np.ndarray) -> np.ndarray:
    """Normalise a non‑negative shape to a probability vector."""
    a = np.maximum(shape, 0.0)
    s = a.sum()
    if s == 0.0:
        # fallback to uniform
        n = a.shape[0]
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑evaluation step function.

    The routine enumerates a handful of parametric families
    (linear, polynomial, exponential, Dirichlet draws) for several
    problem sizes, keeps the best raw score and finally runs a few
    sub‑gradient iterations to polish the solution.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for I/O

    best_seq = None
    best_raw = float("inf")
    best_n = None

    # problem sizes – moderate lengths are cheap and already give a good
    # approximation of the asymptotic optimum.
    lengths = [2000, 4000, 8000, 12000]

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Exploring length {n} ===")

        # ----------------------------------------------------------------
        # 1) a few deterministic shapes
        candidates = []

        # uniform (baseline)
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # linear decreasing (triangular)
        candidates.append(np.arange(n, 0, -1, dtype=np.float64))

        # quadratic decreasing
        candidates.append((np.arange(n, 0, -1, dtype=np.float64)) ** 2)

        # cubic decreasing
        candidates.append((np.arange(n, 0, -1, dtype=np.float64)) ** 3)

        # power‑law decreasing with several exponents
        for p in np.linspace(0.1, 3.0, 12):

        # geometric (exponential) families
        for r in np.linspace(0.90, 0.9999, 12):
            candidates.append((1.0 - r) * (r ** np.arange(n, dtype=np.float64)))

        # a handful of random Dirichlet draws (different concentrations)
        for _ in range(5):
            alpha = np.random.rand() * 4.0 + 0.1   # α ∈ (0.1,4.1)
            d = np.random.gamma(alpha, 1.0, size=n)
            candidates.append(d)

        # ----------------------------------------------------------------
        # 2) evaluate all candidates
        for idx, shape in enumerate(candidates):
            if time.time() >= deadline:
                break

            raw = _raw_score(a)
            if raw < best_raw:
                best_raw = raw
                best_seq = a.copy()
                best_n = n
                print(f"[{time.time():.1f}] new global best (len={n}) raw={best_raw:.6f}")

    # 3) short refinement with the exact sub‑gradient method
    if best_seq is not None and time.time() < deadline:

    # ------------------------------------------------------------------
    # 4) final evaluation through the supplied black‑box and return
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
