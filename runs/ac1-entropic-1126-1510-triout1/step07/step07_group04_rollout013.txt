"""
Propose a step‑function with a low evaluate_sequence score.

Algorithm
---------

* **Initialisation** – start from the best known sequence (`height_sequence_1`
  if it exists) else a symmetric Beta distribution.  The candidate is
  interpolated to a chosen length and normalised to Σ a = 1.

* **Smoothed‑gradient optimisation** – minimise the soft‑max
  ``fγ(a)= (1/γ)·log Σ_k exp(γ·(a*a)_k)`` whose gradient can be written in
  closed form:
      grad fγ(a) = 2·conv_rev(a, w) ,   w_k ∝ exp(γ·(a*a)_k) .
  The soft‑max makes the optimisation problem smoother than the original
  ``max_k (a*a)_k``.  We run a projected Adam optimiser on the simplex,
  gradually increasing the smoothing parameter γ (homotopy) until it is large
  enough that the smoothed objective approximates the true supremum.

* **Length‑escalation** – after optimisation at a moderate size (≈8 k) the
  best sequence is up‑sampled to larger lengths (12 k, 16 k, 20 k, …) and the
  same Adam–soft‑max routine is applied again.  Larger lengths reduce the
  discretisation error of the underlying continuous problem.

* **Final evaluation** – the best sequence found is fed to the provided
  ``evaluate_sequence`` (which rescales internally).  The routine prints
  progress and stops automatically before the time budget expires.

The implementation respects the 1000 s budget, uses only `numpy` and the
standard library and never writes to the filesystem.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)                     # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _raw_score(a: np.ndarray) -> float:
    """Proxy score for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    conv = _self_convolution_full(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, gamma: float):
    """
    Gradient of the smoothed objective
          fγ(a) = (1/γ) log Σ_k exp(γ·(a*a)_k)
    Returns (grad, max_conv, fγ).
    """
    n = a.shape[0]
    conv = _self_convolution_full(a)                # length 2n‑1
    maxc = conv.max()
    # stable softmax
    shifted = conv - maxc
    exp_term = np.exp(gamma * shifted)
    sum_exp = exp_term.sum()
    if sum_exp == 0.0:
        w = np.full_like(exp_term, 1.0 / exp_term.size)
    else:
        w = exp_term / sum_exp

    # gradient = 2 * (a_rev * w)   (full convolution, then slice)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # smoothed objective value (for diagnostics)
    if gamma > 0:
        f_val = maxc + (1.0 / gamma) * np.log(sum_exp)
    else:
        f_val = maxc
    return grad, maxc, f_val


# ----------------------------------------------------------------------
def _adam_refine(a: np.ndarray,
                 deadline: float,
                 rng: np.random.Generator,
                 eta0: float = 0.08,
                 gamma0: float = 0.2,
                 gamma_max: float = 20.0,
                 gamma_mul: float = 1.003) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex with a homotopy on the soft‑max smoothing.
    Returns the best sequence (and its raw score) observed before ``deadline``.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    gamma = gamma0

    while time.time() < deadline:
        t += 1
        grad, cur_max, _ = _softmax_gradient(a, gamma)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_onto_simplex(a_new)

        # occasional jitter to avoid stagnation
        if t % 2000 == 0 and rng.random() < 0.1:
            a_new += rng.normal(scale=1e-6, size=n)
            a_new = _project_onto_simplex(a_new)

        a = a_new

        # increase smoothing (homotopy)
        gamma = min(gamma * gamma_mul, gamma_max)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] adam len={n} it={t} score={best_val:.6f} γ={gamma:.2f}")

        # early exit when we already beat the target
        if best_val <= 1.503:
            break

    return best_a, best_val
