def _adam_subgrad_opt(a0, deadline, max_iters=200000):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    lr0=0.1
    best_a = a.copy()
    best_raw = 2.0*n*_fft_self_convolve_max(a)
    it=0
    while time.time() < deadline and it < max_iters:
        it+=1
        grad, cur_max = _max_subgradient_step(a)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        lr = lr0 * (np.sqrt(1-beta2**it) / (1-beta1**it))
        a = a - lr * m_hat/(np.sqrt(v_hat)+eps)
        a = _project_onto_simplex(a)
        # compute raw
        raw = 2.0*n*cur_max
        if raw < best_raw:
            best_raw = raw
            best_a = a.copy()
        # occasional jitter to avoid stagnation
        if it % 2000 == 0:
            # add small random noise
            a = a + np.random.randn(n)*1e-5
            a = _project_onto_simplex(a)
    # compute final max
    best_raw = 2.0*n*_fft_self_convolve_max(best_a)
    return best_a, best_raw
```

"""
Propose a step function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
* Warm‑start from the previously discovered sequence if it exists,
  otherwise try several random initialisations on the probability simplex.
* Perform a *projected Adam* optimisation where the gradient is the exact
  sub‑gradient of the max‑convolution (instead of the log‑sum‑exp surrogate).
  Adam’s adaptive learning‑rate together with periodic jitter helps escape
  plateaus.
* After the Adam phase a cheap random‑swap local search is applied to
  fine‑tune the solution on the simplex.
* The whole procedure respects the time budget (default 1000 s) and
  returns the best sequence found.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_subgrad_opt(a0: np.ndarray,
                      deadline: float,
                      max_iters: int = 500_000,
                      lr0: float = 0.10) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *exact* max‑convolution objective.
    Works directly on the simplex (a ≥ 0, Σa = 1).
    Returns the best sequence found and its raw score 2·n·max(a*a).
    """
    a = a0.copy()
    n = a.shape[0]

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)

    it = 0
    next_check = 0          # when to recompute the true objective

    while time.time() < deadline and it < max_iters:
        it += 1

        # exact sub‑gradient of the max‑convolution
        grad, cur_max = _max_subgradient_step(a)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (approximately constant)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a)       # keep on the simplex

        # Occasionally recompute the true objective (expensive)
        if it >= next_check:
            cur_score = 2.0 * n * _fft_self_convolve_max(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # print progress
                print(f"[{time.time():.1f}] Adam it={it}  raw={best_score:.6f}")

            # next check after a few hundred iterations (adaptive)
            next_check = it + max(200, int(0.01 * it))

        # jitter every few thousand steps to jump out of plateaus
        if it % 3000 == 0:
            a += np.random.randn(n) * 5e-5
            a = _project_onto_simplex(a)

    # final evaluation of the best vector
    best_score = 2.0 * n * _fft_self_convolve_max(best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
def _local_random_search(a0: np.ndarray,
                         deadline: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing on the simplex:
    repeatedly move a tiny amount of mass from one coordinate to another.
    Accept only moves that improve the objective.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            it += 1
            continue

        # transfer a small random fraction of a[i] to a[j]
        eps = np.random.rand() * min(a[i], 0.02)   # ≤2 % of a[i]
        if eps <= 0.0:
            it += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps

        cur_score = 2.0 * n * _fft_self_convolve_max(a_candidate)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            # occasional progress report
            if it % 5000 == 0:
                print(f"[{time.time():.1f}] swap it={it}  raw={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals
