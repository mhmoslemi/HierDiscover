                improved = True

"""
Propose a non‑negative step function (finite vector) minimising
``evaluate_sequence``.  The algorithm proceeds in several stages:

1.  Scan a wide collection of analytic families (uniform, arcsine‑like,
    beta, sine‑power, polynomial, exponential, cosine windows) for a few
    lengths.  The best candidate (according to the exact objective
    2·n·max(a*a) with sum(a)=1) is kept.

2.  A long Adam descent on a soft‑max approximation of the max‑convolution,
    followed by an exact sub‑gradient method, reduces the peak further.

3.  A linear‑programming refinement (SciPy `linprog`) searches for a
    direction g such that a * g never exceeds the current maximum M.
    Mixing a small amount of the normalised g into a provably lowers M.
    The LP is re‑solved on the updated a until no improvement is possible.

4.  A final O(n) hill‑climber works directly on the self‑convolution,
    moving tiny mass from the largest product pair to a random target,
    accepting moves that lower the peak.

All work respects a 1000 s budget (the function receives *budget_s* and
leaves a small safety margin).  The best sequence found is returned as a
list of floats (sum≈1, all entries ≥0).
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_arcsine_like(n, alpha):
    """Arcsine‑type shape with exponent α > 0."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-alpha)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_polynomial(n, p):
    """Symmetric polynomial shape a_i ∝ ((i+1)(n-i))^p."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 1) * (n - i)) ** p
    vals = np.maximum(vals, 0)
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j,
    updating both the vector `a` and its full self‑convolution `conv`
    in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # remove contributions of the moved mass from all entries of the convolution
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # add the quadratic self‑terms for the two positions
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ----------------------------------------------------------------------
def _build_conv_matrix(a):
    """
    Build the (2·n‑1)×n matrix A that satisfies
        (A @ g) == a * g   (full convolution)
    for a fixed vector a.  The matrix is sparse (CSR) and consists
    of a diagonal stack: column i contains a[i] on rows i … i+n‑1.
    """
    from scipy.sparse import diags
    n = a.shape[0]
    # each column i contributes a constant diagonal a[i] of length n
    diags_data = [np.full(n, ai, dtype=np.float64) for ai in a]
    offsets = np.arange(n)          # offsets 0,1,...,n‑1
    return diags(diags_data, offsets, shape=(2 * n - 1, n), format='csr')


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step function that minimises evaluate_sequence().
    The routine respects the 1000 s time limit and returns the best
    sequence found (list of non‑negative floats, sum≈1).
    """
    try:
        from scipy import sparse as sp
        from scipy.optimize import linprog
        SCIPY_AVAILABLE = True
    except Exception:
        SCIPY_AVAILABLE = False

    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float('inf')

    # ------------------------------------------------------------------
    # 1. Broad scan of analytic families for several lengths
    lengths = [800, 1200, 2000, 3000, 5000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # deterministic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.003))
        candidates.append(_distribution_exponential(n, lam=0.01))
        for p in (0.2, 0.5, 1.0, 2.0, 3.0, 5.0):
            candidates.append(_distribution_beta_sym(n, p))
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))
        for coeff in (0.5, 0.46, 0.42):
            candidates.append(_distribution_cosine_window(n, coeff))
        for p in (-0.5, -0.25, 0.25, 0.5):
            candidates.append(_distribution_polynomial(n, p))
        for alpha in (0.3, 0.5, 0.7, 1.0):
            candidates.append(_distribution_arcsine_like(n, alpha))

        # a few random sparse vectors
        for _ in range(3):
            sparse = np.zeros(n, dtype=np.float64)
            k = max(2, n // 200)
            idx = np.random.choice(n, k, replace=False)
            sparse[idx] = np.random.rand(k)
            sparse /= sparse.sum()
            candidates.append(sparse)

        # evaluate each candidate (cheap, no FFT needed for the scoring function)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] n={n} init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Use a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        n = 2000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # ------------------------------------------------------------------
    # 3. Adam descent on a smooth‐max approximation
    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.07
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_seq_adam = a.copy()
    best_score_adam = _max_convolution_score(a)

    # schedule of temperature (α) – higher α makes the soft‑max closer to max
    schedule = [
        (0.5, 4000),
        (2.0, 4000),
        (10.0, 6000),
        (50.0, 6000),
        (200.0, 6000),
        (800.0, 4000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w /= w.sum()
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if t % 200 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score_adam - 1e-12:
                    best_score_adam = cur
                    best_seq_adam = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

    a = best_seq_adam.copy()
    best_score = best_score_adam

    # ------------------------------------------------------------------
    # 4. Exact sub‑gradient refinement (as in the reference method)
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.2
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad iter={it} new best {cur:.6f}")

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 5. Linear‑programming refinement (optional, uses SciPy if available)
    if SCIPY_AVAILABLE:
        a = best_seq.copy()
        n = a.shape[0]
        while time.time() < deadline:
            b = _fft_self_convolve(a)
            M = b.max()
            A = _build_conv_matrix(a)          # sparse matrix (2n‑1)×n

            # maximise sum(g)  ⇔ minimise -sum(g)
            c = -np.ones(n, dtype=np.float64)
            ub = np.full(A.shape[0], M, dtype=np.float64)

            res = linprog(c, A_ub=A, b_ub=ub,
                          bounds=[(0, None)] * n,
                          method='highs')
            if res.status != 0:
                print(f"[{time.time():.1f}] LP failed (status {res.status})")
                break
            g = res.x
            sum_g = g.sum()
            if sum_g <= 1 + 1e-8:
                # no improving direction exists
                break
            g_norm = g / sum_g

            improved = False
            for t_mix in (0.05, 0.1, 0.2, 0.3):
                cand = (1 - t_mix) * a + t_mix * g_norm
                cand = _project_onto_simplex(cand, 1.0)
                cand = np.maximum(cand, eps)
                cand = cand / cand.sum()
                sc = _max_convolution_score(cand)
                if sc < best_score - 1e-12:
                    a = cand
                    best_seq = cand.copy()
                    best_score = sc
                    improved = True
                    print(f"[{time.time():.1f}] LP mix t={t_mix:.2f} new best {sc:.6f}")
                    break
            if not improved:
                break    # cannot improve further

    # ------------------------------------------------------------------
    # 6. Final hill‑climber that works directly on the convolution
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    it_no_improve = 0
    max_jitter = 1e-5

    while time.time() < deadline:
        it += 1
        # locate the index where the current peak occurs
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # among all pairs (i, k-i) that contribute to the peak,
        # pick the one with the largest product a[i]*a[j]
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            it_no_improve = 0
            if it % 100 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {best_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        # occasional jitter to escape plateaus
        if it % 2000 == 0 and it_no_improve > 500:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
