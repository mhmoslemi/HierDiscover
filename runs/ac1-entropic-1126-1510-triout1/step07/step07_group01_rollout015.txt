            candidates.append((prev, cheap))

        if time.time() >= deadline: break
        seq, sc = _exp_gradient_descent(seed_seq, deadline, max_iters=200_000, eta0=1.0)

            print(f"[{time.time():.1f}] final subgrad improved to {sc:.6f}")

    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02
    while time.time() < deadline:
        iteration += 1

        j = np.random.randint(n-1)

            j += 1

            a = a_new
            cur_score = new_score

                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

            jitter = np.random.randn(n) * 1e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

"""
Propose a low‑autocorrelation step function.

The algorithm works in three stages:
1. **Broad exploration** – generate a variety of analytic families
   (arcsine, uniform, beta, exponential, triangular, sine, cosine)
   for several lengths (≈500–15000).  Additionally generate simple
   mixtures of the best family (arcsine) with the others using a small
   mixing coefficient grid.  All candidates are evaluated with the cheap
   surrogate 2·n·max(a*a) (the objective of `evaluate_sequence` after normalising
   the sum to 1).  The best few seeds are retained.
2. **Intensive optimisation** – the top seed is refined by a long
   sub‑gradient descent on the simplex (exact sub‑gradient of the
   ℓ∞‑norm of the self‑convolution).  The descent runs until the time
   budget is almost exhausted, keeping the best iterate.
3. **Fine hill‑climbing** – tiny random mass transfers (and occasional
   jitter) polish the solution.

All operations keep the vector on the probability simplex, therefore the
objective of `evaluate_sequence` (scale‑invariant) is unchanged.
The routine respects the 1000 s budget and always returns a list of
non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap surrogate = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns (grad, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # indicator of active constraints
    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the linear functional w·(a*a)
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (the terms a_i*a_i appear only once in the sum)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 600_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a -= eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.003) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function minimising `evaluate_sequence`."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # base length – try to reuse a previously discovered size if available
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # lengths to examine (grid + a few random samples)
    lengths = set()
    for factor in (0.7, 0.85, 1.0, 1.15, 1.3, 1.6):
        n = int(base_n * factor)
        if n >= 500:
            lengths.add(n)
    # add a few deterministic larger sizes (up to ~15000)
    for n in (5000, 8000, 11000, 15000):
        if n <= base_n * 5:
            lengths.add(n)
    # a few random lengths
    for _ in range(6):
        n = np.random.randint(500, min(20000, base_n * 5) + 1)
        lengths.add(n)

    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Analytic families (all are already normalised)
    families = [
        (_arcsine_distribution, "arcsine"),
        (lambda m: np.full(m, 1.0 / m), "uniform"),
        (_triangular_distribution, "triangular"),
        (lambda m: _exponential_distribution(m, lam=0.001), "exponential"),
        (_sine_distribution, "sine"),
        (_cosine_distribution, "cosine"),
    ]

    # extra beta family (dense around the arcsine exponent)
    beta_ps = [0.45, 0.48, 0.5, 0.52, 0.55]

    candidates: list[tuple[np.ndarray, float]] = []   # (vector, cheap_score)

    # ------------------------------------------------------------------
    # 1. Broad sampling of pure families and simple mixtures
    for n in lengths:
        if time.time() >= deadline:
            break

        # generate pure shapes
        pure_shapes = {}
        for gen, name in families:
            a = gen(n)
            pure_shapes[name] = a
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} {name:12s} score={score:.6f}")

        # beta shapes
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            pure_shapes[f"beta(p={p:.2f})"] = a
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} beta(p={p:.2f}) score={score:.6f}")

        # random Dirichlet seeds
        for _ in range(2):
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:5d} random Dirichlet score={score:.6f}")

        # ------------------------------------------------------------------
        # 1b. Mixtures of arcsine with every other pure shape (small α grid)
        arcsine = pure_shapes["arcsine"]
        mix_alphas = np.linspace(0.0, 0.20, 11)  # 0,0.02,…,0.20
        for name, other in pure_shapes.items():
            if name == "arcsine":
                continue
            for alpha in mix_alphas:
                if alpha == 0.0:
                    continue
                mix = (1.0 - alpha) * arcsine + alpha * other
                # mixture of two normalised vectors is automatically normalised
                score = _max_convolution_score(mix)
                candidates.append((mix, score))
                print(f"[{time.time():.1f}] n={n:5d} mix arcsine+{name} α={alpha:.2f} score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Include the previous best construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            if np.any(prev < 0):
                prev = np.abs(prev)
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the top‑k seeds for final refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 8
    top_candidates = candidates[:top_k]
    if not top_candidates:
        raise RuntimeError("no candidate generated")
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]
    print(f"Selected {len(top_candidates)} seeds; current best cheap={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy sub‑gradient descent on the best seed (most time is here)
    subgrad_deadline = deadline - 30.0   # keep ~30 s for the hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=800_000, eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent improved to {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Fine‑grained hill climbing (tiny random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.025  # up‑to 2.5 % of total mass per move

    while time.time() < deadline:
        iteration += 1
        # geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 11000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score, best_seq = new_score, a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional Gaussian jitter to escape flat plateaux
        if iteration % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
