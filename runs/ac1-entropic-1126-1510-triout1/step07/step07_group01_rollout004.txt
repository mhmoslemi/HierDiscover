def _exp_gradient_smooth(a0, deadline, max_iters=50_000, beta=200, eta0=1.0):
   a = a0.copy()
   best = a.copy()
   best_score = _max_convolution_score(a)
   for it in range(1, max_iters+1):
       if time.time() >= deadline: break
       grad, smooth_obj = _smoothed_gradient(a, beta)
       eta = eta0 / np.sqrt(it)
       a = a * np.exp(-eta * grad)
       s = a.sum()
       if s == 0:
           a = np.full_like(a, 1.0 / a.size)
       else:
           a = a / s
       # evaluate cheap
       cur_score = _max_convolution_score(a)
       if cur_score < best_score:
           best_score, best = cur_score, a.copy()
   return best, best_score

def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]
```

    smooth_max = max_val + np.log(sum_w) / beta
    smooth_score = 2.0 * n * smooth_max
    return grad, smooth_score

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The algorithm expands the previous three‑phase
scheme with a continuation‑style smoothed optimisation:

1. **Wide sampling** – analytic families (arcsine, beta, triangular,
   exponential, sine, cosine, uniform) plus mixtures and random
   Dirichlet draws for several lengths around a known good size.
   Each candidate is scored by the cheap objective
   ``2·n·max(conv(a,a))`` (exact when sum(a)=1).

2. **Smoothed mirror‑descent** – a continuation method that first
   minimises a smoothed version of the max‑convolution
   (log‑sum‑exp with a temperature β) using exponentiated‑gradient
   and Adam updates on the simplex.  β is increased in stages
   (80 → 250) to approach the true max while staying in a smooth
   landscape.

3. **Exact refinement** – a plain sub‑gradient descent on the true
   max‑convolution, followed by a simulated‑annealing hill‑climb that
   performs tiny random mass transfers with occasional uphill moves.

All steps respect the overall time budget (default 1000 s) and keep
the vector on the probability simplex, which leaves the scale‑invariant
objective unchanged.  The best sequence found is returned as a plain
list of floats.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Gamma‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _smooth_grad_and_value(a: np.ndarray, beta: float):
    """
    Gradient and (smooth) objective for the log‑sum‑exp smoothing of the
    max‑convolution.  Returns (grad, smooth_score) where smooth_score =
    2·n·smooth_max with smooth_max = max(conv) + log(sum(exp(beta*(conv-max))))/beta.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # softmax weights (stable)
    w = np.exp(beta * (conv - max_val))
    z = w.sum()
    w /= z

    # gradient of the smoothed max
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    smooth_max = max_val + np.log(z) / beta
    smooth_score = 2.0 * n * smooth_max
    return grad, smooth_score


# ----------------------------------------------------------------------
def _exp_gradient_smooth(a0: np.ndarray, deadline: float,
                        max_iters: int = 60_000,
                        beta: float = 200.0,
                        eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the smoothed objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        grad, _ = _smooth_grad_and_value(a, beta)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a /= s
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_smooth(a0: np.ndarray, deadline: float,
                 max_iters: int = 60_000,
                 beta: float = 200.0,
                 step0: float = 0.2,
                 beta1: float = 0.9,
                 beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smoothed objective."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _ = _smooth_grad_and_value(a, beta)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for indices where i=j contributes only once
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent with a diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgradient(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _annealing_hill_climb(a0: np.ndarray, cur_score: float,
                          deadline: float) -> tuple[np.ndarray, float]:
    """Simulated‑annealing hill‑climbing on the exact cheap score."""
    a = a0.copy()
    best = a.copy()
    best_score = cur_score
    n = a.shape[0]
    start = time.time()
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # exponential cooling based on remaining time
        remaining_frac = (deadline - time.time()) / (deadline - start)
        temp = cur_score * (0.02 * remaining_frac + 1e-6)

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer up to a few percent of the mass at i
        max_transfer = 0.04
        delta = np.random.rand() * min(max_transfer * a[i], a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best = a_new.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] anneal iter={iteration} new best {best_score:.6f}")
        else:
            # occasional uphill acceptance
            if temp > 0 and np.random.rand() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
                cur_score = new_score

        # occasional jitter to escape flat plateaus
        if iteration % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function minimising evaluate_sequence."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Base length: keep the size of the previously discovered construction
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # Lengths to explore (around the base size and a few random larger ones)
    lengths = set()
    for f in (0.9, 0.95, 1.0, 1.05, 1.1, 1.2):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(8000, int(base_n * 2)) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Candidate generation
    beta_ps = [0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]
    alphas = [0.0, 0.05, 0.10, 0.15, 0.20]   # mixture weights
    candidates: list[tuple[np.ndarray, float]] = []

    for n in lengths:
        if time.time() >= deadline:
            break

        # Analytic families -------------------------------------------------
        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_exponential_distribution, "exponential"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_uniform_distribution, "uniform")
        ]
        for gen, name in families:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Symmetric beta family (wide range of exponents)
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Mixtures of arcsine with uniform / exponential
        arcsine = _arcsine_distribution(n)
        uniform = _uniform_distribution(n)
        expo = _exponential_distribution(n, lam=0.003)
        for alpha in alphas:
            # arcsine + uniform
            a = (1 - alpha) * arcsine + alpha * uniform
            a /= a.sum()
            score = _max_convolution_score(a)
            candidates.append((a, score))

            # arcsine + exponential
            a = (1 - alpha) * arcsine + alpha * expo
            a /= a.sum()
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Random Dirichlet draws
        for _ in range(6):
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))

    # Include the previously best construction, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 20
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, best cheap score={top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # ------------------------------------------------------------------
    # Heavy refinement of each seed
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        # allocate a fair budget for this seed
        remaining_seeds = len(top_candidates) - idx
        per_seed_deadline = time.time() + (deadline - time.time()) * 0.9 / remaining_seeds

        # 1) smoothed EG with moderate temperature
        seq, sc = _exp_gradient_smooth(seed_seq, per_seed_deadline,
                                       max_iters=30_000,
                                       beta=80.0, eta0=0.9)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} EGβ80 → {sc:.6f}")

        # 2) smoothed EG with larger temperature (closer to true max)
        seq, sc = _exp_gradient_smooth(seq, per_seed_deadline,
                                       max_iters=30_000,
                                       beta=250.0, eta0=0.5)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} EGβ250 → {sc:.6f}")

        # 3) Adam on the same smoothed objective
        seq, sc = _adam_smooth(seq, per_seed_deadline,
                               max_iters=30_000,
                               beta=250.0, step0=0.15)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adamβ250 → {sc:.6f}")

        # 4) Precise sub‑gradient descent on the true max
        seq, sc = _subgradient_descent(seq, per_seed_deadline,
                                       max_iters=200_000, eta0=0.6)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} sub‑grad → {sc:.6f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing hill‑climb with the remaining budget
    if time.time() < deadline:
        seq, sc = _annealing_hill_climb(best_seq, best_score, deadline)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] final annealing → {best_score:.6f}")

    print(f"Best cheap score after full search: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list – evaluate_sequence handles the scaling.
    return best_seq.tolist()
