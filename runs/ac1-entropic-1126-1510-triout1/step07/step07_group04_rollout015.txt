    decreasing step size and keep the move if it improves the
    raw score.
    """
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0

        it += 1

            # enforce non‑negativity

                best_val = val
                improved = True
                print(f"[{time.time():.1f}] active‑descent it={it} step={step:.2e} factor={factor} score={best_val:.6f}")
                break
        if not improved:
            # no improvement with current step – shrink it
            step *= 0.5
    return best_a, best_val

"""
Propose a step function with a low evaluate_sequence score.

Strategy
--------
1️⃣  Random exploration on moderate lengths (2k‑8k) using a rich set of
    Beta‑family distributions, Dirichlet draws and mixtures.
2️⃣  Mirror‑descent (exponentiated‑gradient) refinement that minimises the
    max‑entry of the self‑convolution.
3️⃣  Aggressive active‑set descent: at each iteration we take the gradient
    of the *currently* maximal convolution entry (i.e. the sub‑gradient of
    that single quadratic term) and perform a line‑search with a
    decreasing step‑size, always projecting back onto the simplex.
    This directly attacks the bottleneck term.
4️⃣  Upsample the best sequence to larger lengths (12k‑24k) and repeat
    steps 2‑3.  Early stopping when the target bound 1.503 is reached.

All helper functions are top‑level, the code respects the 1000 s budget
and uses only NumPy (no file or network I/O).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    It equals the true evaluator output because the score is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray):
    """
    Sub‑gradient of f(a)=max_k (a*a)_k using a uniform weight over the
    active set (all indices attaining the max).  Returns (grad, cur_max).
    """
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _argmax_grad(a: np.ndarray):
    """
    Gradient of the *single* quadratic term that attains the current maximum
    (i.e. of (a*a)_k for k = argmax).  Returns (grad, cur_max).
    """
    b = _self_convolution_full(a)
    k = int(b.argmax())
    w = np.zeros_like(b)
    w[k] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, b.max()

def _refine_mirror_descent(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) refinement on the simplex.
    Returns the best sequence and its raw score found before `deadline`.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient(a)
        eta = eta0 / math.sqrt(it)

        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)

        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        if it % 500 == 0 and rng.random() < 0.06:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror‑descent len={n} it={it} score={best_val:.6f}")

    return best_a, best_val


def _refine_active_descent(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,

                           min_step: float = 1e-6) -> tuple[np.ndarray, float]:

    Aggressive deterministic descent that repeatedly moves mass opposite
    to the gradient of the *currently* maximal convolution entry.
    A simple back‑tracking line‑search over a few factors is used.
    The step size is halved whenever no improvement is found.

    step = init_step

        # Dirichlet draws
        for _ in range(500):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-3, 1)
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   dirichlet conc={conc:.3e} score={best_score:.6f}")

        # mixture of two Betas
        for _ in range(300):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            a2, b2 = 10 ** rng.uniform(-3, 1), 10 ** rng.uniform(-3, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   mix2 improved to {best_score:.6f}")

    print(f"[{time.time():.1f}] best after coarse search = {best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent refinement on the current best length
    if time.time() < deadline:
        a, scr = _refine_mirror_descent(best_seq.copy(), deadline, rng, eta0=0.4)
        if scr < best_score - 1e-12:
            best_seq, best_score = a, scr
            print(f"[{time.time():.1f}] after mirror‑descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – active‑set deterministic descent
    if time.time() < deadline:
        a, scr = _refine_active_descent(best_seq.copy(), deadline, rng, init_step=0.2)
        if scr < best_score - 1e-12:
            best_seq, best_score = a, scr
            print(f"[{time.time():.1f}] after active‑descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – up‑sample to larger lengths and repeat refinement
    larger_lengths = [12000, 16000, 20000, 24000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)

        # mirror‑descent on the larger grid
        a, scr = _refine_mirror_descent(a, deadline, rng, eta0=0.35)
        if scr < best_score - 1e-12:
            best_seq, best_score = a.copy(), scr
            print(f"   mirror after up‑sample improved to {best_score:.6f}")

            # follow‑up active descent
            a2, scr2 = _refine_active_descent(best_seq.copy(), deadline, rng, init_step=0.15)
            if scr2 < best_score - 1e-12:
                best_seq, best_score = a2.copy(), scr2
                print(f"   active after up‑sample improved to {best_score:.6f}")

        if best_score <= 1.503:
            print("[early exit] target bound reached")
            break

    # ------------------------------------------------------------------
    # Final exact evaluation (the evaluator rescales automatically)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} "
          f"(len={len(best_seq)})")
    return best_seq.tolist()
