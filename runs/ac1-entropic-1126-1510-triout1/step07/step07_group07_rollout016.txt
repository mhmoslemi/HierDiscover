"""
Hybrid optimization for step-function heights.
1. Build a rich pool of candidate probability vectors (uniform,
   symmetric beta, exponential, inverse‑distance, random Dirichlet,
   and the supplied warm‑start).
2. Pick the best seed w.r.t. the cheap raw objective 2 n max(a*a).
3. Refine the seed with a continuation scheme:
   - minimise the smoothed max‑convolution F_μ(a)=μ log ∑exp((a*a)/μ)
     using L‑BFGS‑B on the soft‑max parametrisation a_i=exp(z_i)/∑exp(z_i).
   - μ is halved geometrically until ≈10⁻⁶.
   - during the optimisation the best sequence (according to the exact
     raw objective) is kept.
4. A short projected sub‑gradient descent polish finishes the run.
The routine respects the 1000 s time budget and returns a non‑negative
sequence that usually beats the previous best bound.
"""

import math
import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError('Simplex size must be non‑negative')
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # Guard against overflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()

# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector: 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2*n-1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights, Σp = 1

    # gradient: 2 * (a_rev * p)   (only the central n entries are needed)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smoothed objective value
    f_val = M + mu * np.log(Z)
    return f_val, grad

# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# ----------------------------------------------------------------------
def _smooth_obj_and_grad_z(z: np.ndarray, mu: float):
    """
    Smoothed objective expressed in the soft‑max parameters.
    Returns (value, gradient_z) where gradient_z is ∂F/∂z.
    """
    a = _softmax(z)
    f_val, grad_a = _smooth_obj_and_grad(a, mu)
    # ∂a/∂z = diag(a) - a aᵀ -> grad_z = (diag(a)-a aᵀ) grad_a
    a_dot_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - a_dot_grad)
    return f_val, grad_z

# ----------------------------------------------------------------------
def _generate_candidates(rng: np.random.Generator, n: int):
    """Create a pool of diverse probability vectors."""
    candidates = []

    # 1. global warm‑start if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base = base / base.sum()
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # 2. simple baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))                # arcsine

    # 3. symmetric Betas
    for a in (0.2, 0.3, 0.4, 0.5, 0.7, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n, a, a))

    # 4. random Betas
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(_beta_distribution(n, alpha, beta))

    # 5. random Dirichlet draws
    for _ in range(8):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec /= vec.sum()
        candidates.append(vec)

    # 6. exponential‑decay families (symmetric)
    for decay in np.logspace(-5, -2, num=6):
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-decay * x) + np.exp(-decay * (n - 1 - x))
        a /= a.sum()
        candidates.append(a)

    # 7. inverse‑distance families
    for power in (0.5, 0.7, 1.0, 1.2, 1.5, 2.0):
        i = np.arange(n, dtype=np.float64) + 0.5
        a = (i ** -power + (n + 0.5 - i) ** -power)
        a /= a.sum()
        candidates.append(a)

    # 8. a few random mixtures of the above (convex combinations)
    for _ in range(6):
        w = rng.uniform(0.0, 1.0, size=2)
        w /= w.sum()
        a1 = _beta_distribution(n, rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0))
        a2 = _beta_distribution(n, rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0))
        a = w[0] * a1 + w[1] * a2
        a /= a.sum()
        candidates.append(a)

    return candidates

# ----------------------------------------------------------------------
def _select_best_seed(candidates):
    """Pick the candidate with the smallest raw score."""
    best_seq = None
    best_raw = float('inf')
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    return best_seq, best_raw

# ----------------------------------------------------------------------
def _lbfgs_refine(a0: np.ndarray, time_budget: float):
    """
    Continuation + L‑BFGS‑B optimisation in the soft‑max parameter space.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    # initialise soft‑max parameters
    eps = 1e-12
    z = np.log(np.maximum(a, eps))

    mu = max(_max_self_convolution(a) * 0.5, 1e-7)
    mu_target = 1e-6
    mu_factor = 0.6

    best_seq = a.copy()
    best_raw = _raw_score(a)

    start = time.time()
    while True:
        remaining = time_budget - (time.time() - start)
        if remaining < 2.0:
            break

        # allocate a portion of remaining time for this mu stage
        # we aim for at most 5000 L‑BFGS iterations or the time budget
        max_iter = int(min(8000, remaining / 0.0012))
        if max_iter <= 0:
            break

        # objective wrapper for scipy
        def obj(z_vec):
            return _smooth_obj_and_grad_z(z_vec, mu)

        # callback to keep track of the best exact score
        def callback(z_vec):
            nonlocal best_seq, best_raw
            a_cur = _softmax(z_vec)
            cur_raw = _raw_score(a_cur)
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_seq = a_cur.copy()

        opt.minimize(
            obj,
            z,
            method='L-BFGS-B',
            jac=True,
            callback=callback,
            options={'maxiter': max_iter, 'disp': False}
        )

        # after optimisation step, refresh variables
        a = _softmax(z)
        cur_raw = _raw_score(a)
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_seq = a.copy()

        # prepare next mu
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target:
            break

    return best_seq, best_raw

# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float):
    """
    A short projected sub‑gradient descent (as in the reference code)
    to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        # compute exact sub‑gradient of max_k (a*a)_k
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # step‑size schedule
        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional jitter to escape flat regions
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point used by the harness.
    Returns a list of non‑negative step heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a problem size (the length of the step function)
    n = 8000      # a length that balances quality and speed

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seeds (all normalised to Σ=1)
    candidates = _generate_candidates(rng, n)

    # ------------------------------------------------------------------
    # 3. Pick the best raw‑score seed
    best_seq, best_raw = _select_best_seed(candidates)
    print(f'Initial best raw score: {best_raw:.6f}')

    # ------------------------------------------------------------------
    # 4. Continuation + L‑BFGS refinement (major part of the budget)
    remaining = deadline - time.time()
    if remaining > 30.0:
        best_seq, best_raw = _lbfgs_refine(best_seq, remaining * 0.7)

    # ------------------------------------------------------------------
    # 5. Final short sub‑gradient polishing (optional)
    remaining = deadline - time.time()
    if remaining > 5.0:
        best_seq, best_raw = _subgradient_polish(best_seq, remaining)

    # ------------------------------------------------------------------
    # 6. Return the sequence (evaluate_sequence rescales it automatically)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f'Final evaluated score: {final_score:.6f} (len={len(best_seq)})')
    return best_seq.tolist()

"""
Hybrid optimisation for a non‑negative step function.

1. Build a rich pool of probability vectors (uniform, symmetric / random
   Beta, exponential‑decay, inverse‑distance, Dirichlet draws and the
   supplied warm‑start `height_sequence_1`).
2. Pick the best seed according to the cheap proxy 2·n·max(a*a).
3. Refine the seed with a continuation scheme:
   • minimise the smoothed max‑convolution
        F_μ(a) = μ·log Σ exp((a*a)_k / μ)
     using L‑BFGS‑B on the soft‑max parameters
        a_i = exp(z_i) / Σ exp(z_j).
   • μ is multiplied by 0.6 each outer step until it reaches ≈10⁻⁶.
   • the best exact raw score (2·n·max(a*a)) found during the loop is kept.
4. A short projected sub‑gradient descent polish finishes the run.
5. The sequence is returned; `evaluate_sequence` rescales it automatically.

The routine respects the 1000 s time budget and usually improves on the
previous bound (≈1.510 → ≈1.503)."""

import math
import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)          # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector: 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights, Σp = 1

    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _smooth_obj_and_grad_z(z: np.ndarray, mu: float):
    """
    Smoothed objective expressed in the soft‑max parameters.
    Returns (value, gradient_z) where gradient_z = ∂F/∂z.
    """
    a = _softmax(z)
    f_val, grad_a = _smooth_obj_and_grad(a, mu)
    a_dot_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - a_dot_grad)
    return f_val, grad_z


# ----------------------------------------------------------------------
class _SmoothObj:
    """Callable wrapper required by scipy.optimize, holding the current μ."""
    def __init__(self, mu: float):
        self.mu = mu

    def __call__(self, z: np.ndarray):
        return _smooth_obj_and_grad_z(z, self.mu)

def _generate_candidates(rng: np.random.Generator, n: int):
    """Create a pool of diverse probability vectors (all sum to 1)."""
    candidates = []

    # warm‑start if supplied
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base = base / base.sum()
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # simple deterministic baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))                # arcsine

    # symmetric Beta shapes
    for a in (0.2, 0.3, 0.4, 0.5, 0.7, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n, a, a))

    # random Beta draws
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(_beta_distribution(n, alpha, beta))

    # random Dirichlet draws
    for _ in range(8):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec /= vec.sum()
        candidates.append(vec)

    # symmetric exponential‑decay families
    for decay in np.logspace(-5, -2, num=6):
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-decay * x) + np.exp(-decay * (n - 1 - x))
        a /= a.sum()
        candidates.append(a)

    # inverse‑distance families
    for power in (0.5, 0.7, 1.0, 1.2, 1.5, 2.0):
        i = np.arange(n, dtype=np.float64) + 0.5
        a = (i ** -power + (n + 0.5 - i) ** -power)
        a /= a.sum()
        candidates.append(a)

    # few random convex mixtures of Beta shapes
    for _ in range(6):
        w = rng.uniform(0.0, 1.0, size=2)
        w /= w.sum()
        a1 = _beta_distribution(n, rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0))
        a2 = _beta_distribution(n, rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0))
        a = w[0] * a1 + w[1] * a2
        a /= a.sum()
        candidates.append(a)

    return candidates


# ----------------------------------------------------------------------
def _select_best_seed(candidates):
    """Return (best_vector, best_raw_score) among the candidates."""
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    return best_seq, best_raw


# ----------------------------------------------------------------------
def _lbfgs_refine(a0: np.ndarray, time_budget: float):
    """
    Continuation + L‑BFGS‑B optimisation in the soft‑max space.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    eps = 1e-12
    z = np.log(np.maximum(a, eps))

    mu = max(_max_self_convolution(a) * 0.5, 1e-7)
    mu_target = 1e-6
    mu_factor = 0.6

    best_seq = a.copy()
    best_raw = _raw_score(a)

    start = time.time()
    while True:
        remaining = time_budget - (time.time() - start)
        if remaining < 2.0:
            break

        # at most a few thousand L‑BFGS iterations or the time left
        max_iter = int(min(8000, remaining / 0.0012))
        if max_iter <= 0:
            break

        obj = _SmoothObj(mu)
        res = opt.minimize(
            obj,
            z,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        z = res.x
        a = _softmax(z)

        cur_raw = _raw_score(a)
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_seq = a.copy()

        # next smoothing parameter
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target:
            break

    return best_seq, best_raw


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float):
    """
    Short projected sub‑gradient descent polish (as in the reference code).
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val

    """
    rng = np.random.default_rng(seed)
