assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function (a probability vector) that minimizes  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the full discrete self‑convolution.  
The objective is **scale‑invariant**, therefore we can restrict the search to vectors whose entries sum to 1.

The search consists of four stages:

1. **Rich initialisation** – a large collection of analytic families (uniform, arcsine, beta,
   exponential, Gaussian, two‑block spikes, alternating, random‑sparse, …) is generated for several
   lengths.  The best few candidates are kept.

2. **Smoothed Adam descent** – a smooth approximation of `max` is built with the
   log‑sum‑exp trick.  Adam optimisation with an increasing smoothness parameter
   (`α = 1 → 5 → 20 → 100 → 500 → 2000`) refines every kept candidate.  After each
   update the iterate is projected onto the simplex.

3. **Exact sub‑gradient refinement** – once the smoothing is large, the true
   sub‑gradient (active set of maximal convolution entries) is used together with
   a momentum scheme.

4. **Local greedy/annealing fine‑tuning** – a cheap active‑set swap and a short
   simulated‑annealing walk are applied to escape possible plateaux.

All stages respect a global time budget (default 1000 s).  
The best vector found is returned as a plain Python list.

"""
Non‑negative step‑function optimiser.

The algorithm builds many analytic candidates for several lengths,
keeps the most promising ones and refines them with
* smoothed Adam descent (log‑sum‑exp surrogate of the max‑convolution),
* exact sub‑gradient descent,
* a greedy active‑set swap,
* a short simulated‑annealing walk.

All operations are performed on the probability simplex (sum = 1, a_i ≥ 0).
The routine respects a global time budget (default 1000 s) and returns the
best sequence found as a plain Python list of floats.
"""

# ----------------------------------------------------------------------
# 1.  Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# 2.  Convolution utilities (real FFT, O(n log n))
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# 3.  Objective (sum‑normalised, i.e. exactly the value returned by evaluate_sequence)
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# 4.  Analytic families (all return a probability vector, sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta density with shape p (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric beta density with shapes p,q (>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float = 0.15) -> np.ndarray:
    """Discrete Gaussian (centred, periodic boundary ignored)."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((idx - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Uniform mass on the first and last `block_frac` fraction of indices."""
    k = max(1, int(round(block_frac * n)))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    """Uniform mass on every `stride`‑th coordinate (starting at 0)."""
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, k: int) -> np.ndarray:
    """k random positions get a random positive weight, rest zero."""
    if k >= n:
        k = n // 2
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k) + 0.1  # avoid exact zeros on selected spots
    a /= a.sum()
    return a

# 5.  Candidate generation for a given length
def _candidates_for_length(n: int) -> list[np.ndarray]:
    """Generate a diverse set of plausible start vectors for length n."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # symmetric beta (many shapes)
    for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0):
        cand.append(_distribution_beta_sym(n, p))

    # asymmetric beta (few combos)
    for p, q in ((0.1, 2.0), (2.0, 0.1), (0.5, 2.5), (2.5, 0.5)):
        cand.append(_distribution_beta_asym(n, p, q))

    # exponentials
    for lam in (0.001, 0.002, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # Gaussians (different widths)
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        cand.append(_distribution_gaussian(n, sigma))

    # alternating strides
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # two‑block spikes
    for frac in (0.03, 0.05, 0.07, 0.10, 0.15):
        cand.append(_distribution_two_blocks(n, block_frac=frac))

    # random sparse patterns
    for k in (max(2, n // 200), max(2, n // 100), max(2, n // 50)):
        cand.append(_distribution_random_sparse(n, k))

    return cand

# 6.  Smoothed Adam descent (log‑sum‑exp surrogate)
def _smooth_adam_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Adam optimisation of the log‑sum‑exp smoothed max‑convolution."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule – (α, number_of_steps)
    schedule = [
        (1.0, 3000),
        (5.0, 3000),
        (20.0, 3000),
        (100.0, 3000),
        (500.0, 3000),
        (2000.0, 3000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of smooth max (size n)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # simplex projection
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smooth α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score

# 7.  Exact sub‑gradient descent (true max‑convolution)
def _exact_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """Momentum sub‑gradient descent on the genuine max‑convolution objective."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        # sub‑gradient from active set
        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum + diminishing stepsize
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        # early stop if we already beat the target bound
        if best_score <= 1.503:
            break

        # cheap time‑check every few iterations
        if it % 500 == 0 and time.time() + 0.5 >= deadline:
            break

    return best_seq, best_score

# 8.  Greedy active‑set swapping
def _greedy_swap_opt(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from high‑gradient to low‑gradient indices while improving."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]

        j = int(np.argmin(grad))
        if i == j:
            break

        delta = min(a[i] * 0.5, 0.005)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")
        else:
            # try a smaller transfer
            delta *= 0.25
            if delta < 1e-7:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] tiny swap improved best {best_score:.6f}")

    return best_seq, best_score

# 9.  Simulated annealing fine‑tuning
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis‑type random walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    T = 0.04
    it = 0

    while time.time() < deadline and T > 1e-4:
        it += 1
        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {best_score:.6f}")

        if it % 2500 == 0:
            T *= 0.85

    return best_seq, best_score

# 10.  Main entry point
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    Returns the best vector encountered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # (a)  Lengths to explore – larger lengths give more freedom but cost more.
    lengths = [800, 1600, 3200, 6400, 8000]   # a mix of medium / large sizes

    # (b)  Keep a small pool of the most promising candidates (global)
    pool_best = []          # list of (score, vector)
    max_pool = 12           # keep at most this many vectors

    # ------------------------------------------------------------------
    # (c)  Scan analytic families for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        cand_list = _candidates_for_length(n)
        for cand in cand_list:
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()            # enforce sum = 1
            sc = _objective(cand)
            # maintain a size‑limited pool of best candidates
            if len(pool_best) < max_pool or sc < max(pool_best, key=lambda x: x[0])[0]:
                pool_best.append((sc, cand.copy()))
                pool_best.sort(key=lambda x: x[0])
                if len(pool_best) > max_pool:
                    pool_best.pop()   # discard worst

    # ------------------------------------------------------------------
    # (d)  Include the previous best if it exists (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if len(pool_best) < max_pool or sc < max(pool_best, key=lambda x: x[0])[0]:
                pool_best.append((sc, prev.copy()))
                pool_best.sort(key=lambda x: x[0])
                if len(pool_best) > max_pool:
                    pool_best.pop()
    except Exception:
        pass

    # ------------------------------------------------------------------
    # (e)  Pick the current overall best
    best_score, best_seq = pool_best[0]
    print(f"[{time.time():.1f}] Initial best score {best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # (f)  Main optimisation pipeline – treat each pooled candidate
    #     as a starting point for a full refinement.
    for idx, (init_score, init_vec) in enumerate(pool_best):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Refining candidate {idx+1}/{len(pool_best)} length={len(init_vec)}")
        a = init_vec.copy()
        # smoothed Adam descent
        a, best_score = _smooth_adam_descent(a, deadline, best_seq, best_score)
        # exact sub‑gradient refinement
        a, best_score = _exact_subgradient_descent(a, deadline, best_seq, best_score)
        # greedy swaps
        a, best_score = _greedy_swap_opt(a, deadline, best_seq, best_score)
        # simulated annealing
        a, best_score = _simulated_annealing(a, deadline, best_seq, best_score)

        # keep the absolute best across all candidates
        if best_score < init_score - 1e-13:
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Candidate {idx+1} yielded new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # (g)  Final report
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Return as plain Python list
    return best_seq.tolist()
