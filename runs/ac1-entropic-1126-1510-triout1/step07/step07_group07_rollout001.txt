"""
Hybrid continuation + high‑iteration FISTA optimizer for the step‑function
evaluation problem.

- Build a diverse set of seed probability vectors (uniform, Beta,
  Dirichlet, the supplied `height_sequence_1`).
- Pick the seed with the lowest raw score (2·n·max(a*a)).
- Run a continuation scheme: minimise the smoothed max‑convolution
  $F_μ(a)=μ·log∑exp((a*a)_k/μ)$.
  For each μ we run a long projected‑gradient (FISTA) loop with an
  aggressive back‑tracking line‑search and occasional random jitter.
  μ is decreased geometrically down to 1e‑9, which makes $F_μ$ arbitrarily
  close to the true max.
- Finish with a short projected sub‑gradient polishing stage.
- All iterates are projected onto the probability simplex, therefore the
  final sequence can be returned directly – `evaluate_sequence` rescales
  it automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    # Fast path for already feasible vectors
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on n equiprobable bins, normalised to sum=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf with safe handling of extreme α,β
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    # avoid -inf for x=0 or 1 when α<1 or β<1
    logx = np.log(np.clip(x, 1e-300, 1.0))
    log1mx = np.log(np.clip(1.0 - x, 1e-300, 1.0))
    logpdf = logC + (alpha - 1.0) * logx + (beta - 1.0) * log1mx
    # cap to avoid overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective 2·n·max(a*a) for a normalised probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution via log‑sum‑exp and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)  # length 2*n-1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)  # values in (0, 1]
    Z = exp_scaled.sum()
    p = exp_scaled / Z  # soft‑max weights, sum to 1

    # gradient: 2 * (a_rev * p). The central n entries correspond to
    # the derivative of each a_i.
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smoothed objective value
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation + long FISTA loop.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]

    # initial smoothing (half of the current max‑conv)
    mu = max(_max_self_convolution(a) * 0.5, 1e-12)
    mu_target = 1e-9
    mu_factor = 0.85          # slower reduction → more μ stages
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    # Outer continuation loop
    while True:
        # check remaining time
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        remaining = time_budget - elapsed

        # Choose Lipschitz constant for the current μ (the gradient of the
        # smoothed max has Lipschitz constant ≤ 2/μ)
        L = max(1.0, 2.0 / mu)

        # FISTA variables
        y = a.copy()
        t = 1.0

        # Determine a safe number of inner iterations – aim for at most
        # 0.0015 s per iteration; cap at 15000 iterations per μ‑stage.
        est_iter_time = 1.5e-3
        max_iters = int(min(15000, remaining / est_iter_time))
        if max_iters < 1:
            break

        for it in range(max_iters):
            # gradient of the smoothed objective at y
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # simple back‑tracking line‑search (max two doublings of L)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                # sufficient decrease condition for convex smooth func.
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # Every few iterations evaluate the true raw score
            if (it & 31) == 0:
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()

            # Occasionally inject a tiny random jitter to shake off plateaus
            if (it & 511) == 0 and np.random.rand() < 0.02:
                jitter = np.random.normal(scale=5e-7, size=n)
                a = _project_onto_simplex(a + jitter)
                y = a.copy()
                t = 1.0

        # Reduce μ for the next continuation stage
        mu = max(mu * mu_factor, mu_target)

        # Warm‑restart from the best sequence seen so far for the next μ
        a = best_seq.copy()

        # Stop if μ has reached the target (the objective is effectively the true max)
        if mu <= mu_target + 1e-12:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1

        # exact sub‑gradient of max_k (a*a)_k
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()

    return best_seq, best_val

    start = time.time()
    while True:
        # Time check
        if time.time() - start >= time_budget:
            break
        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break

        # Lipschitz constant for the smoothed objective (≤ 2/μ)
        L = max(1.0, 2.0 / mu)

        # FISTA state
        y = a.copy()
        t = 1.0

        # Estimate feasible inner‑iteration count (≈0.0015 s each)
        est_it_time = 1.5e-3
        max_iters = int(min(15000, remaining / est_it_time))
        if max_iters < 1:
            break

        for it in range(max_iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # Simple back‑tracking line‑search (max two doublings of L)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # Every 32 iterations evaluate the true raw score
            if (it & 31) == 0:
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()

            # Occasional tiny jitter to escape flat regions
            if (it & 511) == 0 and np.random.rand() < 0.02:
                jitter = np.random.normal(scale=5e-7, size=n)
                a = _project_onto_simplex(a + jitter)
                y = a.copy()
                t = 1.0

        # Reduce smoothing parameter
        mu = max(mu * mu_factor, mu_target)

        # Warm‑restart from the best point discovered for the new μ
        a = best_seq.copy()

        # Stop when μ is essentially at its target
        if mu <= mu_target + 1e-12:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1

        # Exact sub‑gradient of max_k (a*a)_k
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # Small jitter every few hundred iterations
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the competition harness.
    Returns a list of non‑negative step heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # 1. Build a small pool of seeds (all normalised to Σ=1)
    n = 8000                                 # length of the final answer
    candidates = []

    # a) global warm‑start if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base = base / base.sum()
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # b) deterministic baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))                # arcsine

    # c) symmetric Beta shapes
    for a_param in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n, a_param, a_param))

    # d) random Beta draws
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(_beta_distribution(n, alpha, beta))

    # e) random Dirichlet draws
    for _ in range(5):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n)
        vec /= vec.sum()
        candidates.append(vec)

    # 2. Pick the best raw‑score seed
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    print(f"Initial best raw score: {best_raw:.6f}")

    # 3. Continuation + long FISTA refinement (≈80 % of the remaining budget)
    remaining = deadline - time.time()
    if remaining > 15.0:                     # keep a few seconds for polishing
        refine_budget = remaining * 0.80
        best_seq, best_raw = _fista_refine(best_seq, refine_budget)

    # 4. Final short sub‑gradient polishing (if any time left)
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_raw = _subgradient_polish(best_seq, remaining)

    # 5. Return the sequence (evaluate_sequence rescales automatically)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
