# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # stable log‑pdf
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)          # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The value equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of   f(a) = max_k (a*a)_k .
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

def _refine_mirror_descent(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) refinement on the simplex."""
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        eta = eta0 / math.sqrt(it)

        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)

        s = a.sum()
        if s == 0.0:

        else:
            a /= s

            a = _project_onto_simplex(a)

            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror‑descent len={n} it={it} score={best_val:.6f}")

    return best_a, best_val

    Uses additive updates with projection onto the simplex.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_raw = _raw_score(a)

    epsilon = init_epsilon
    alpha = init_alpha
    it = 0

    while time.time() < deadline:
        it += 1

        # full self‑convolution
        b = _self_convolution_full(a)
        maxb = b.max()

        # soft‑max weights (stable)
        w_exp = np.exp(epsilon * (b - maxb))
        w = w_exp / w_exp.sum()

        # gradient of the smooth objective
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # additive projected step
        a_new = a - alpha * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new)

        a = a_new

        # occasional diagnostics and parameter decay
        if it % 200 == 0:
            raw = _raw_score(a)
            if raw + 1e-12 < best_raw:
                best_raw = raw
                best_a = a.copy()
                print(f"[{time.time():.1f}] smooth‑descent it={it} raw={best_raw:.6f}")
            epsilon = max(1.0, epsilon * epsilon_decay)
            alpha *= alpha_decay

        # occasional random jitter to escape plateaus
        if it % 5000 == 0 and rng.random() < 0.15:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

    return best_a, best_raw


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluate_sequence score.
    Returns the best sequence found within the allotted time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if base_seq.sum() == 0.0:
            raise ValueError
        base_seq = base_seq / base_seq.sum()
    except Exception:
        base_seq = None

    if base_seq is not None:
        best_seq = base_seq.copy()
    else:
        # start from the continuous arcsine density at a generous size
        n0 = 20000
        best_seq = _beta_distribution(n0, 0.5, 0.5)
        best_seq = _project_onto_simplex(best_seq)

    best_score = _raw_score(best_seq)
    best_len = len(best_seq)
    print(f"[{time.time():.1f}] initial best raw score = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 1 – broad stochastic exploration on several lengths
    exploration_lengths = [4000, 8000, 16000, 20000]
    for n in exploration_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploration length {n}")

        # start from the current best, interpolated
        a = _interpolate_to_length(best_seq, n)
        cur = _raw_score(a)
        if cur < best_score:
            best_seq, best_score, best_len = a, cur, n
            print(f"   interpolated improvement score={best_score:.6f}")

        # deterministic baselines
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        cur = _raw_score(uniform)
        if cur < best_score:
            best_seq, best_score, best_len = uniform.copy(), cur, n
            print(f"   uniform score={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        cur = _raw_score(arcsine)
        if cur < best_score:
            best_seq, best_score, best_len = arcsine.copy(), cur, n
            print(f"   arcsine score={best_score:.6f}")

        # symmetric Beta grid (now includes very sharp U‑shapes)
        alphas = np.concatenate((
            np.linspace(0.01, 0.2, 8),            # very edge‑heavy
            np.linspace(0.2, 0.8, 10),            # moderate
            np.linspace(0.8, 3.0, 12)             # bell‑shaped
        ))
        for a_alpha in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a_alpha, a_alpha)
            cur = _raw_score(vec)
            if cur < best_score:
                best_seq, best_score, best_len = vec.copy(), cur, n
                print(f"   symBeta a={a_alpha:.3f} score={best_score:.6f}")

        # random Beta draws (extended range)
        for _ in range(400):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-3, 1)   # α ∈ [1e‑3,10]
            beta = 10 ** rng.uniform(-3, 1)
            vec = _beta_distribution(n, alpha, beta)
            cur = _raw_score(vec)
            if cur < best_score:
                best_seq, best_score, best_len = vec.copy(), cur, n
                # occasional print (suppressing flood)
                if rng.random() < 0.02:
                    print(f"   randBeta a={alpha:.3e} b={beta:.3e} score={best_score:.6f}")

        # Dirichlet draws
        for _ in range(300):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-3, 1)   # concentration
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            cur = _raw_score(vec)
            if cur < best_score:
                best_seq, best_score, best_len = vec.copy(), cur, n
                if rng.random() < 0.02:
                    print(f"   dirichlet conc={conc:.3e} score={best_score:.6f}")

        # Mixture of two Betas
        for _ in range(200):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            cur = _raw_score(vec)
            if cur < best_score:
                best_seq, best_score, best_len = vec.copy(), cur, n
                # sparse logging
                if rng.random() < 0.02:
                    print(f"   mix2 λ={lam:.3f} score={best_score:.6f}")

    print(f"[{time.time():.1f}] after coarse search best raw = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max gradient descent on the current best length
    if time.time() < deadline:
        a = best_seq.copy()
        a, scr = _smooth_gradient_descent(a, deadline, rng,
                                          init_epsilon=12.0,
                                          init_alpha=0.6,
                                          epsilon_decay=0.9993,
                                          alpha_decay=0.999)
        if scr < best_score:
            best_seq, best_score = a.copy(), scr
            best_len = len(best_seq)
            print(f"[{time.time():.1f}] after smooth descent score={best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 3 – mirror‑descent refinement (adds fine‑grained adjustments)
    if time.time() < deadline:
        a = best_seq.copy()
        a, scr = _refine_mirror_descent(a, deadline, rng, eta0=0.35)
        if scr < best_score:
            best_seq, best_score = a.copy(), scr
            best_len = len(best_seq)
            print(f"[{time.time():.1f}] after mirror refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – up‑sample to larger sizes and repeat the two refinements
    upsample_lengths = [25000, 30000, 35000]
    for n in upsample_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)

        # smooth descent
        a, scr = _smooth_gradient_descent(a, deadline, rng,
                                          init_epsilon=8.0,
                                          init_alpha=0.5)
        # mirror descent (short)
        if time.time() < deadline:
            a, scr_md = _refine_mirror_descent(a, deadline, rng, eta0=0.3)
            scr = min(scr, scr_md)

        if scr + 1e-12 < best_score:
            best_seq, best_score, best_len = a.copy(), scr, n
            print(f"   new best after up‑sample score={best_score:.6f} (len={best_len})")
        if best_score <= 1.503:
            print("[early exit] target bound reached.")
            break

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluator (exact arithmetic, clipping, etc.)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
