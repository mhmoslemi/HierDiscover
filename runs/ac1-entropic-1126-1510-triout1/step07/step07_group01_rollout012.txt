"""
Improved step‑function optimizer.

The algorithm expands the three‑phase scheme used previously:

1. **Broad sampling** – generate a diverse pool of analytic families
   (beta, arcsine, triangular, exponential, sine, cosine, Gaussian)
   together with random Dirichlet draws.  All candidates are normalised to
   the probability simplex and ranked by the cheap objective
   ``2·n·max(a*a)``.

2. **Quasi‑Newton refinement** – the best seeds are optimised on the
   simplex using a smooth approximation of the “max” (soft‑max) and a
   L‑BFGS‑B solver.  The optimisation works in the log‑softmax
   parametrisation ``a_i = exp(x_i)/∑_j exp(x_j)`` so the simplex
   constraint is automatically satisfied; gradients are obtained from an
   exact FFT‑based convolution.

3. **Final polishing** – a short sub‑gradient descent (using a proper
   sub‑gradient of the true max) is followed by a tiny random‑mass‑
   transfer hill‑climb.  The whole routine respects the 1000 s budget
   and always returns a non‑negative list ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# FFT based convolutions (real, using rfft/irfft)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Cheap objective used for ranking (scale‑invariant)
def _max_convolution_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) for a on the simplex (sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / (sigma * n)) ** 2)
    return vals / vals.sum()


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

# Smooth “max” objective (log‑sum‑exp) and gradients
def _smooth_obj_and_grad(a: np.ndarray, beta: float):
    """
    Returns (smooth_obj, grad_a) where

        smooth_obj = (1/β)·log∑_k exp(β·b_k)   with b = a*a

    and grad_a = ∂smooth_obj/∂a (size n).  The implementation uses real FFTs.
    """
    b = _fft_self_convolve(a)                     # length 2n‑1
    b_max = b.max()
    # stable soft‑max
    exp_shift = np.exp(beta * (b - b_max))
    sum_exp = exp_shift.sum()
    smooth = b_max + np.log(sum_exp) / beta      # (1/β)·log‑sum‑exp(b)
    w = exp_shift / sum_exp                       # weights, sum to 1

    # gradient = Σ_k w_k · ∂b_k/∂a = 2·Σ_k w_k·a_{k‑i}
    conv_rev = _fft_convolve(a[::-1], w)        # full length n + (2n‑1) -1 = 3n‑1
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return smooth, grad

# L‑BFGS‑B optimisation in the log‑softmax parametrisation
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _lbfgsb_obj_grad(x: np.ndarray, beta: float, n: int):
    """
    Objective and gradient for L‑BFGS‑B.
    x are unconstrained logits; a = softmax(x) lies on the simplex.
    Returns (value, grad_x).
    """
    # softmax
    max_x = np.max(x)
    exp_x = np.exp(x - max_x)
    a = exp_x / exp_x.sum()                     # sum(a) = 1

    # smooth max on a*a
    smooth, grad_a = _smooth_obj_and_grad(a, beta)   # grad_a = ∂smooth/∂a

    # scale to the true evaluation objective (sum(a)=1 → factor 2·n)
    obj = 2.0 * n * smooth

    # chain rule: ∂obj/∂x = (∂obj/∂a)·∂a/∂x
    # ∂a_i/∂x_j = a_i (δ_{ij} – a_j)
    dot = np.dot(a, grad_a)                     # Σ_j a_j·grad_a_j
    grad_x = 2.0 * n * a * (grad_a - dot)       # shape (n,)
    return obj, grad_x


def _lbfgsb_optimize(a0: np.ndarray, beta: float,
                     max_iter: int = 200) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B from a0 (already on the simplex) with a given β."""
    if not _SCIPY_AVAILABLE:
        # fallback – simple exponentiated‑gradient for safety
        from math import sqrt

        a = a0.copy()
        best = a.copy()
        best_score = _max_convolution_score(a)
        for it in range(1, max_iter + 1):
            grad, _, _ = _subgrad_max(a)
            eta = 0.8 / np.sqrt(it)
            a = a * np.exp(-eta * grad)
            a = _project_onto_simplex(a, s=1.0)
            cur = _max_convolution_score(a)
            if cur < best_score:
                best_score, best = cur, a.copy()
        return best, best_score

    # start from log‑softmax
    x0 = np.log(a0 + 1e-12)

    # L‑BFGS‑B call (objective returns both value and gradient)
    res = minimize(_lbfgsb_obj_grad, x0,
                   args=(beta, a0.shape[0]),
                   method='L-BFGS-B',
                   jac=True,
                   options={'maxfun': max_iter, 'maxiter': max_iter,
                            'disp': False})
    max_res = np.max(res.x)
    exp_res = np.exp(res.x - max_res)
    a_opt = exp_res / exp_res.sum()
    cheap = _max_convolution_score(a_opt)
    return a_opt, cheap

# Exact sub‑gradient of the true max(b) (no diagonal correction)
def _subgrad_max(a: np.ndarray):
    """
    Returns (grad, cur_max, active) where
        grad = ∂max(b)/∂a  (sub‑gradient)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)
    if active.size == 0:
        w = np.zeros_like(b)
    else:
        w = np.zeros_like(b)
        w[active] = 1.0 / active.size                 # convex combination → valid sub‑gradient

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, cur_max, active


def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent on the exact max(b) objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad_max(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns a list of floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Base length (use previous construction if available)
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # ------------------------------------------------------------------
    # Lengths to explore (around the base size, plus a few random larger ones)
    lengths = set()
    for f in (0.9, 1.0, 1.1, 1.2, 1.3):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(4):
        n = np.random.randint(500, min(8000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1. Broad sampling
    candidates = []   # (array, cheap_score)

    beta_exponents = [0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5,
                     0.75, 1.0, 1.25, 1.5, 2.0, 3.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.003), "exponential"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_gaussian_distribution, "gaussian"),
            (lambda m: np.full(m, 1.0 / m), "uniform")
        ]

        for gen, name in families:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} {name:10s} score={score:.6f}")

        # symmetric beta family
        for p in beta_exponents:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} beta(p={p:4.2f}) score={score:.6f}")

        # random Dirichlet seeds
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} random Dirichlet score={score:.6f}")

    # ------------------------------------------------------------------
    # 2. Include previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep the best few seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 12
    top_candidates = candidates[:top_k]
    print(f"Selected {len(top_candidates)} seeds for refinement; best cheap score={top_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy quasi‑Newton refinement (smooth max with increasing β)
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        a_cur = seed_seq.copy()
        # schedule of β values – start smooth, finish close to true max
        for beta in (5.0, 15.0, 30.0, 50.0):
            if time.time() >= deadline:
                break
            a_cur, cur_score = _lbfgsb_optimize(a_cur, beta, max_iter=200)
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a_cur.copy()
                print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} β={beta:.0f} improved to {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Short sub‑gradient descent on the exact max (use remaining time)
    subgrad_deadline = deadline - 20.0     # keep a little margin for hill‑climb
    if time.time() < subgrad_deadline:
        a_cur, cur_score = _subgradient_descent(best_seq, subgrad_deadline,
                                                max_iters=400_000, eta0=0.5)
        if cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a_cur.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent improved to {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained hill‑climbing (tiny random mass transfers)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.025   # up to 2.5 % of total mass per move

    while time.time() < deadline:
        it += 1
        # geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (it / 12000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to escape flat regions
        if it % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # evaluate_sequence is scale‑invariant; we return the simplex vector.
    return best_seq.tolist()
