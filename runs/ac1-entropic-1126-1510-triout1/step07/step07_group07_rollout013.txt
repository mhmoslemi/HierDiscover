def _adam_optimize(a0, time_budget):
    a = a0.copy()
    n = a.shape[0]
    # hyperparameters:
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # smoothing schedule:
    mu = _max_self_convolution(a) * 0.5
    mu = max(mu, 1e-6)
    mu_target = 1e-6
    mu_factor = 0.7
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_raw = _raw_score(a)
    start = time.time()
    while True:
        # inner loop per mu
        L = max(1.0, 2.0 / mu) # not used in Adam but maybe step size scaling
        # number of iterations for this mu maybe (int) time dependent
        remaining = time_budget - (time.time() - start)
        if remaining <= 0:
            break
        # limit iterations:
        # each iteration ~ conv_time; we estimate 0.001 sec per iteration
        max_iters = int(min(8000, remaining / 0.0012))
        if max_iters <= 0:
            break
        for i in range(max_iters):
            t += 1
            f_val, grad = _smooth_obj_and_grad(a, mu)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = (0.5 / L) * m_hat / (np.sqrt(v_hat) + eps)  # Learning rate scaling
            a_new = a - step
            a_new = _project_onto_simplex(a_new)
            a = a_new
            # occasionally check raw
            if (i & 31) == 0:
                cur_raw = _raw_score(a)
                if cur_raw < best_raw:
                    best_raw = cur_raw
                    best_a = a.copy()
        # after inner loop reduce mu
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        # warm start for new mu: keep a as best_a (most improved)
        a = best_a.copy()
        # reset Adam's moments? Maybe keep but scaling may differ
        m[:] = 0.0
        v[:] = 0.0
        t = 0
    return best_a, best_raw

"""
Hybrid multistart optimiser for the step‑function evaluation problem.

Strategy
========
1. Build a modest library of *basis* probability vectors (uniform,
   arcsine, many symmetric/asymmetric Beta pdfs, random Dirichlet draws,
   and the provided global warm‑start if present).  All vectors sum to 1.

2. Treat a candidate sequence as a convex combination of those bases:
      a = w @ B      (w lies on the simplex, B is the matrix of bases).

   Optimise the low‑dimensional weight vector `w` with Adam on the
   smoothed max‑convolution objective
       f_μ(a) = μ·log Σ exp((a*a)_k / μ) .
   The smoothing parameter μ is reduced geometrically (continuation).

3. Keep the best full‑resolution sequence found during the low‑dimensional
   search.

4. Refine that sequence with the already‑tested continuation + FISTA
   method (projected accelerated gradient) and finish with a short
   projected sub‑gradient polish.

All operations respect the 1000 s budget and the sequence is returned as a
plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # protect against overflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective used while the vector is normalised (Σa=1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2*n-1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights, Σp = 1

    # gradient: 2 * (a_rev * p)   (central n entries)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1: len(a) - 1 + len(a)]

    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation + FISTA refinement of a probability vector.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]

    mu = _max_self_convolution(a) * 0.5
    mu = max(mu, 1e-7)
    mu_target = 1e-6
    mu_factor = 0.7

    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # ---- inner FISTA for current μ ---------------------------------
        L = max(1.0, 2.0 / mu)          # Lipschitz estimate
        y = a.copy()
        t = 1.0

        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        iters = int(min(3000, remaining / 0.0015))
        if iters <= 0:
            break

        for _ in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # back‑tracking (max two doublings)
            L_local = L
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L_local)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L_local) + 1e-12:
                    break
                L_local *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            if (_ & 31) == 0:          # every 32 steps
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()
        # ----------------------------------------------------------------
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        a = best_seq.copy()            # warm‑restart

    return best_seq, best_val

def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val

def _adam_optimize_basis(w0: np.ndarray,

    Low‑dimensional Adam optimisation of the basis weights w
    (B is K×n).  Returns (best_w, best_a, best_raw_score).
    """
    K, n = B.shape
    w = w0.copy()
    a = w @ B
    best_w = w.copy()
    best_a = a.copy()
    best_raw = _raw_score(a)

    beta1, beta2 = 0.9, 0.999

    v = np.zeros(K)

    mu_target = 1e-6
    mu_factor = 0.7

    start = time.time()
    while True:
        remaining = time_budget - (time.time() - start)
        if remaining <= 0:
            break

        # bound the number of inner Adam steps by remaining time

        if iters <= 0:
            break

        for _ in range(iters):
            t += 1
            a = w @ B                     # full‑resolution candidate
            _, grad_a = _smooth_obj_and_grad(a, mu)

            # gradient w.r.t. w (chain rule)
            w_grad = B @ grad_a           # shape (K,)

            # Adam update on w
            m = beta1 * m + (1.0 - beta1) * w_grad
            v = beta2 * v + (1.0 - beta2) * (w_grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = 0.5 * mu                 # step‑size proportional to the current μ
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            w_new = w - step
            w_new = _project_onto_simplex(w_new)

            w = w_new

            # occasional evaluation of the true (non‑smoothed) objective
            if (t & 31) == 0:

                cur_raw = _raw_score(a)
                if cur_raw < best_raw:

                    best_a = a.copy()

        m.fill(0.0)
        v.fill(0.0)
        t = 0

        w = best_w.copy()
