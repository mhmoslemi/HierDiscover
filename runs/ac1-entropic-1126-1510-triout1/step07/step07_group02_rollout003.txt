"""
Propose a candidate step‑function that minimises ``evaluate_sequence``.

Algorithm
---------

1. **Analytic initialisation** – For several lengths (2000 … 12000) a
   library of analytic probability distributions (uniform, arcsine,
   beta, exponential, sine‑power, cosine windows, random sparse) is
   generated, normalised to sum = 1 and evaluated cheaply by an FFT based

   In each move a small amount ``δ`` of mass is transferred from a
   “source’’ index that heavily contributes to the current peak to a
   random “target’’ index.  The convolution is updated in *O(n)* using an

   sum = 1, all entries ≥ 0) is returned as a plain Python list.  The

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear terms
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

                b_eq = np.array([1.0])

                # Objective: minimise M
                c = np.zeros(n + 1)
                c[-1] = 1.0

                # Solve
                res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                              A_eq=A_eq, b_eq=b_eq,
                              bounds=[(0, None)] * n + [(0, None)],
                              method='highs')
                if not res.success:
                    print(f"[{time.time():.1f}] LP failed at n={n} iter={lp_iter}")
                    break

                g = res.x[:n]
                M_opt = res.x[-1]

                # Mix the new distribution with the current one
                t_mix = 0.5
                new_f = (1.0 - t_mix) * f + t_mix * g
                new_f = np.maximum(new_f, eps)
                new_f = new_f / new_f.sum()

                # Check improvement with the true scoring function
                new_score = evaluate_sequence(new_f.tolist())
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_f.copy()
                    best_n = n
                    print(f"[{time.time():.1f}] LP iter {lp_iter} n={n} new best {new_score:.6f}")

                # Prepare for possible next LP iteration
                best_seq = new_f.copy()

    # ----------------------------------------------------------------------
    # 4. Final sanity checks and return
    # ----------------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution of a modest size
        best_seq = _distribution_uniform(2000)

    best_seq = np.maximum(best_seq, eps)
    best_seq = best_seq / best_seq.sum()
    print(f"[{time.time():.1f}] Returning best score ≈ {best_score:.6f} (length={best_n})")
    return best_seq.tolist()
