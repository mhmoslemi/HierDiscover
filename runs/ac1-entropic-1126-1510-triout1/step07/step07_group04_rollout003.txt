"""
Propose a low‑score step function.

The algorithm works in three stages:

1️⃣  **LP‐based flattening** – on a modest size (≈1500) we iteratively
    solve a linear program that maximises the ℓ₁‑mass of a candidate
    `g` while keeping the cross‑convolution `f∗g` below the current
    worst self‑convolution value `M`.  Mixing the current sequence `f`
    with the normalised optimiser `g` (with a decreasing step size)
    provably never increases `M`.  A handful of such iterations already
    push the shape far closer to the conjectured optimum.

2️⃣  **Mirror‑descent polishing** – an exponentiated‑gradient (multiplicative)
    sub‑gradient method on the convex proxy  `max_k (a∗a)_k` refines the
    result to high precision.

3️⃣  **Upsampling & final polishing** – the best sequence is linearly
    interpolated to larger lengths (4k → 8k → 12k) and again polished
    by mirror‑descent.  The final list is handed to `evaluate_sequence`,
    which rescales it automatically.

The whole search respects the 1000 s budget and prints intermediate
scores for monitoring.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)       # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The value equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of   f(a) = max_k (a*a)_k .
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_mirror_descent(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) refinement on the simplex.
    Returns the best sequence and its raw score found before `deadline`.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        eta = eta0 / math.sqrt(it)

        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)

        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.06:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror‑descent len={n} it={it} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (size (2n‑1)×n) such that (f∗g) = A·g.
    Row k corresponds to the k‑th entry of the full linear convolution.
    """
    n = f.size
    rows = 2 * n - 1
    A = sp.lil_matrix((rows, n), dtype=np.float64)

    # each column j contains the vector f shifted by +j positions
    for j in range(n):
        A[j : j + n, j] = f  # Python slicing creates a view of length n
    return A.tocsr()


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear program:
        maximise   Σ g_j
        subject to (f∗g)_k ≤ M  (for all k)
                   g_j ≥ 0
    Returns the optimiser g (may be identical to f if no improvement).
    """
    n = f.size
    A = _build_constraint_matrix(f)
    c = -np.ones(n)                     # maximise Σg  → minimise -Σg
    bounds = [(0.0, None)] * n
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # a tiny slack helps the solver with numerical noise
    b += 1e-14

    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method="highs", options={"presolve": True})

    if not res.success:
        # LP failed – fall back to the current sequence
        return f.copy()
    return res.x

    and up‑sampling to larger lengths.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Initialise from a previously found sequence if available.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if base_seq.sum() == 0.0:
            raise ValueError
        base_seq = base_seq / base_seq.sum()
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # Choose a modest size for the LP phase.
    n_small = 1500
    if base_seq is not None:
        seq = _interpolate_to_length(base_seq, n_small)
    else:
        # start from a symmetric Beta distribution (often good)
        seq = _beta_distribution(n_small, 0.35, 0.35)

    seq = _project_onto_simplex(seq)   # guarantee Σ=1
    best_seq = seq.copy()
    best_score = _raw_score(best_seq)
    print(f"[{time.time():.1f}] initial score = {best_score:.6f} (len={n_small})")

    # ------------------------------------------------------------------
    # Phase 1 – LP based flattening
    max_lp_iters = 12
    for it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        M = _max_self_convolution(best_seq)
        g = _lp_step(best_seq, M)
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        # Normalise g to unit ℓ₁‑mass
        g_norm = g / sum_g

        # Mixing coefficient – decreasing with iteration
        t = 0.5 / (it + 1)   # 0.5, 0.25, 0.166,… that stays ≤0.5
        new_seq = (1.0 - t) * best_seq + t * g_norm
        new_seq = _project_onto_simplex(new_seq)

        new_score = _raw_score(new_seq)
        print(f"[{time.time():.1f}] LP iter {it+1:2d}  t={t:.3f}  score={new_score:.6f}")

        if new_score + 1e-12 < best_score:
            best_seq, best_score = new_seq, new_score

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent polishing at the small size
    if time.time() < deadline:
        best_seq, best_score = _refine_mirror_descent(best_seq, deadline, rng, eta0=0.5)
        print(f"[{time.time():.1f}] after mirror‑descent: score={best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to larger lengths and polish again
    # We stop early if we already beat the target bound.
    target_lengths = [4000, 8000, 12000]
    for n in target_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")

        up_seq = _interpolate_to_length(best_seq, n)
        up_seq = _project_onto_simplex(up_seq)

        up_seq, up_score = _refine_mirror_descent(up_seq, deadline, rng, eta0=0.35)
        print(f"[{time.time():.1f}]   post‑refine score={up_score:.6f}")

        if up_score + 1e-12 < best_score:
            best_seq, best_score = up_seq, up_score
            print(f"   *** NEW GLOBAL BEST: {best_score:.6f} (len={n})")

        if best_score <= 1.503:
            print("Target bound reached – terminating early.")
            break

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluator (exact arithmetic, clipping, etc.)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
