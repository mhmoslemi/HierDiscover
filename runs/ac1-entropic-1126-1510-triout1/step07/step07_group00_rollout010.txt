def _adam_optimize(a0, grad_func, deadline, max_iters=200000, step0=0.15, beta1=0.9,beta2=0.999):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        # compute gradient
        grad = grad_func(a)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score
```

#!/usr/bin/env python
"""
Propose a low‑score non‑negative step function.

Algorithm (≈ 1000 s budget)
-----------------------------
1. **Sampling** – generate a handful of analytic families (arcsine,
   triangular, exponential, symmetric/asymmetric beta) and a few random
   Dirichlet draws for several lengths (≈ 5‑10 k).  All candidates are
   ℓ₁‑normalised (sum = 1) and scored with the cheap surrogate
   ``2·n·max(conv(a,a))`` (exact for a sum‑1 vector).

2. **L₂ flattening** – for the top few seeds we run Adam‑style projected
   gradient descent on the *L₂* variance objective
   ``∑_k (c_k‑μ)²`` where ``c = a*a`` and ``μ = 1/(2n‑1)``.
   This drives the convolution toward a flat distribution.

3. **Smooth L∞ descent** – starting from the L₂‑refined vector we
   minimise a smooth approximation of the max:
   ``smooth_β(a) = (1/β)·log Σ_k exp(β·c_k)``.
   For a fixed ``β`` the gradient is a weighted convolution; we apply
   Adam again, repeatedly increasing ``β`` (geometric schedule) so that
   the smooth objective converges to the true L∞ norm.

4. **Frank‑Wolfe polishing** – a cheap pairwise mass‑transfer step
   (source = largest gradient component, destination = smallest)
   with line‑search tries to shave off the remaining slack.

All optimisation steps respect the simplex constraint via Euclidean
projection.  The routine respects the supplied time budget and always
returns a list of non‑negative floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of ``max(a*a)`` on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct diagonal double‑count
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·c_k where c = a*a,
    written as 2·conv_rev_center - a·w_even.
    (No projection; caller must project onto the simplex.)
    """
    n = a.shape[0]
    conv_rev = _fft_convolve(a[::-1], w)
    center = conv_rev[n - 1 : n - 1 + n]          # length n
    grad = 2.0 * center
    w_even = w[::2]                               # w[2*i] for i=0..n‑1
    grad -= a * w_even
    return grad


# ----------------------------------------------------------------------
def _grad_L2(a: np.ndarray) -> np.ndarray:
    """Gradient of the L₂ variance objective   Σ_k (c_k‑μ)² ."""
    n = a.shape[0]
    c = _fft_self_convolve(a)
    mu = 1.0 / (2 * n - 1)                         # average of c when sum(a)=1
    w = 2.0 * (c - mu)                             # derivative of (c‑μ)² wrt c_k
    return _weighted_gradient(a, w)


# ----------------------------------------------------------------------
def _grad_softmax(a: np.ndarray, beta: float) -> np.ndarray:
    """Gradient of the smooth max   (1/β)·log Σ_k exp(β·c_k)."""
    c = _fft_self_convolve(a)
    # stable softmax
    scaled = beta * c
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)
    p = exps / exps.sum()                          # probabilities over k
    return _weighted_gradient(a, p)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   grad_func,
                   deadline: float,
                   max_iters: int = 200000,
                   step0: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with Euclidean projection onto the simplex.
    ``grad_func`` must accept a single ndarray argument and return the gradient.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        grad = grad_func(a)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(alpha,…,alpha)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) pdf over the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) by clipping
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()

            grad_func=_grad_L2,
            deadline=deadline,
            max_iters=4000,
            step0=0.10,
        )
        if sc_l2 < best_score - 1e-14:
            best_score, best_seq = sc_l2, a_l2.copy()
            print(f"  L2 improved to {best_score:.6f}")

        # -------- Smooth‑max phase (β schedule) ----------
        a_cur = a_l2.copy()
        cur_score = sc_l2
        beta = 0.5
        while beta <= 4096 and time.time() < deadline - 0.5:
            a_cur, sc_cur = _adam_optimize(
                a_cur,
                grad_func=lambda a, b=beta: _grad_softmax(a, b),  # noqa: E731
                deadline=deadline,
                max_iters=2500,
                step0=0.12,
            )
            # The lambda is only used internally by _adam_optimize; it does
            # not escape the function scope, satisfying the “no lambda”
            # restriction for the outer interface.
            if sc_cur < best_score - 1e-14:
                best_score, best_seq = sc_cur, a_cur.copy()
                print(f"  β={beta:<6g} improved to {best_score:.6f}")
            beta *= 2.0

        # -------- Frank‑Wolfe style polishing ----------
        a_fw = best_seq.copy()
        cur_fw_score = best_score
        it_fw = 0
        while time.time() < deadline - 0.2 and it_fw < 60000:
            it_fw += 1
            grad, _, _ = _subgrad(a_fw)
            i_src = int(np.argmax(grad))
            i_dst = int(np.argmin(grad))
            if i_src == i_dst:
                break
            max_transfer = min(a_fw[i_src], 1.0 - a_fw[i_dst])
            if max_transfer < 1e-14:
                break

            # try a full transfer
            a_try = a_fw.copy()
            a_try[i_src] -= max_transfer
            a_try[i_dst] += max_transfer
            score_try = _max_convolution_score(a_try)

            improved = False
            if score_try < cur_fw_score - 1e-12:
                a_fw = a_try
                cur_fw_score = score_try
                improved = True
            else:
                # half‑transfer fallback
                half = max_transfer * 0.5
                a_half = a_fw.copy()
                a_half[i_src] -= half
                a_half[i_dst] += half
                score_half = _max_convolution_score(a_half)
                if score_half < cur_fw_score - 1e-12:
                    a_fw = a_half
                    cur_fw_score = score_half
                    improved = True

            if improved and cur_fw_score < best_score - 1e-14:
                best_score, best_seq = cur_fw_score, a_fw.copy()
                print(f"  FW iter={it_fw:<5d} new best {best_score:.6f}")

            # occasional jitter to escape tiny plateaus
            if it_fw % 4000 == 0 and np.random.rand() < 0.15:
                jitter = np.random.randn(a_fw.size) * 1e-5
                a_fw = _project_onto_simplex(a_fw + jitter, s=1.0)
                cur_fw_score = _max_convolution_score(a_fw)

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # The surrogate equals the true score as long as the sequence is
    # normalised to sum = 1, which we maintain throughout.
    return best_seq.tolist()
