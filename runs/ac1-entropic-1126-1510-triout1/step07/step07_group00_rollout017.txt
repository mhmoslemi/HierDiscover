1. **Broad sampling** – generate many candidates from analytic families
   (arcsine, triangular, exponential, symmetric beta) for several
   sequence lengths together with random Dirichlet draws.  All candidates
   are projected onto the simplex (sum=1) and scored by the cheap surge
   ``score = 2·n·max(conv)``.

2. **Heavy local optimisation** – the best few seeds are polished with
   three complementary schemes:
   * Adam‑style adaptive gradient descent on the simplex.
   * Exponentiated‑gradient (KL‑mirror) descent.
   * An extended Frank‑Wolfe style mass‑transfer that tries a handful of
     source/destination pairs per iteration (instead of a single pair).

   Each optimiser runs until the global deadline is hit, so the budget
   is automatically split between the seeds.

3. **Jitter & finalisation** – occasional tiny Gaussian jitter is added
   to escape plateaus.  The best sequence found is returned as a plain
   ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Score = 2·n·max(a*a) – exact surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) on the simplex.
    Returns (grad, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (the contribution of a_i*a_i appears only once
    # in the convolution, but the formula 2·conv_rev counts it twice)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent (mirror descent w.r.t. KL)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0

    while time.time() < deadline:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style gradient descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0

    while time.time() < deadline:
        t += 1
        grad, _, _ = _subgrad(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a point on the simplex from Dirichlet(alpha,…,alpha)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _pairwise_frank_wolfe(a: np.ndarray,
                          global_best: dict,
                          deadline: float,
                          max_iters: int = 40000,
                          src_k: int = 4,
                          dst_k: int = 4) -> np.ndarray:
    """
    Extended Frank‑Wolfe style mass‑transfer.
    ``global_best`` is a dict with keys ``'seq'`` and ``'score'`` that is
    updated whenever a new overall best is discovered.
    """
    cur_score = _max_convolution_score(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)

        # pick a handful of promising sources / destinations
        srcs = np.argsort(-grad)[:src_k]
        dsts = np.argsort(grad)[:dst_k]

        improved = False
        best_local = a
        best_local_score = cur_score

        for src in srcs:
            for dst in dsts:
                if src == dst:
                    continue
                max_transfer = min(a[src], 1.0 - a[dst])
                if max_transfer < 1e-14:
                    continue

                # full transfer trial
                trial = a.copy()
                trial[src] -= max_transfer
                trial[dst] += max_transfer
                s_full = _max_convolution_score(trial)
                if s_full < best_local_score - 1e-12:
                    best_local_score = s_full
                    best_local = trial
                    improved = True

                # half‑transfer trial (fallback)
                half = max_transfer * 0.5
                trial2 = a.copy()
                trial2[src] -= half
                trial2[dst] += half
                s_half = _max_convolution_score(trial2)
                if s_half < best_local_score - 1e-12:
                    best_local_score = s_half
                    best_local = trial2
                    improved = True

        if improved:
            a = best_local
            cur_score = best_local_score
            if cur_score < global_best['score'] - 1e-14:
                global_best['score'] = cur_score
                global_best['seq'] = a.copy()
                print(f"[{time.time():.1f}] FW improved overall best to {cur_score:.6f}")
            continue

        # no improvement – occasionally add a tiny jitter to escape plateaus
        if it % 3000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter)
            cur_score = _max_convolution_score(a)

    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and always
    returns a feasible sequence."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a collection of sequence lengths to explore
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    lengths = set()
    for scale in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0):
        n = int(base_n * scale)
        if 500 <= n <= 20000:
            lengths.add(n)
    for _ in range(7):
        n = np.random.randint(500, min(20000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 2. Broad sampling of analytic families + Dirichlet draws
    candidates = []  # (vector, score)

    beta_ps = list(np.arange(0.02, 0.21, 0.02)) \
              + list(np.arange(0.25, 0.71, 0.05)) \
              + list(np.arange(0.80, 1.51, 0.10)) \
              + [2.0, 3.0]

    dirichlet_alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # analytic families ------------------------------------------------
        for gen, name in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exponential"),
        ):
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:<5d} {name:12s} score={score:.6f}")

        # symmetric beta family (fine grid)
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # random Dirichlet draws with various concentrations
        for alpha in dirichlet_alphas:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))

    # ------------------------------------------------------------------
    # 3. Insert the previous best (if available) – helps continuity
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Keep the top seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(10, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")
    print(f"Current best seed score = {top_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 5. Heavier local optimisation (Adam + ExpGrad) on each seed
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # container for global best that can be updated inside FW
    global_best = {"seq": best_seq.copy(), "score": best_score}

    for idx, (seed_seq, seed_score) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Adam descent – give it as much time as possible
        seq, sc = _adam_descent(seed_seq, deadline)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            global_best["seq"], global_best["score"] = best_seq.copy(), best_score
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam improved to {best_score:.6f}")

        # Exponentiated‑gradient descent – complementary sweep
        seq, sc = _exp_gradient_descent(seed_seq, deadline)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            global_best["seq"], global_best["score"] = best_seq.copy(), best_score
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} ExpGrad improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Extended Frank‑Wolfe pairwise mass transfer
    a = global_best["seq"].copy()
    a = _pairwise_frank_wolfe(a, global_best, deadline,
                              max_iters=50000,
                              src_k=5, dst_k=5)

    # ------------------------------------------------------------------
    # 7. Final jitter‑enhanced pass (helps escape last tiny plateaus)
    it_jit = 0
    while time.time() < deadline and it_jit < 8000:
        it_jit += 1
        if np.random.rand() < 0.07:
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter)
            cur = _max_convolution_score(a)
            if cur < global_best["score"] - 1e-14:
                global_best["score"], global_best["seq"] = cur, a.copy()
                print(f"[{time.time():.1f}] jitter improved overall best to {cur:.6f}")

    # ------------------------------------------------------------------
    final_seq = global_best["seq"]
    final_score = global_best["score"]
    print(f"Final best score: {final_score:.6f} (length={len(final_seq)})")
    # guarantee non‑negative, sum to 1 (already true) and return plain list
    return final_seq.tolist()
