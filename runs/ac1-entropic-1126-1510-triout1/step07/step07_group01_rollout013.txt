def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Choose length n between 4000 and 8000, maybe random but ensure >= 500.
    # Use base_n if defined else 4000.
    try:
        base_n = len(height_sequence_1)
    except:
        base_n = 4000
    # Explore lengths: a set [int(base_n*0.8), base_n, int(base_n*1.2), maybe larger random].
    lengths = [max(500, int(base_n*0.8)), base_n, max(500, int(base_n*1.2))]
    lengths += [np.random.randint(500, min(12000, base_n*3) + 1) for _ in range(3)]
    lengths = list(set(lengths))

    # store best vector and score
    best_vec = None
    best_score = np.inf

    # Helper function to evaluate cheap score given vector normalized.
    def cheap_score(v):
        # ensure nonneg and sum=1.
        if np.sum(v) == 0:
            return np.inf
        v = v / np.sum(v)
        # compute convolution via FFT
        n = len(v)
        size = 1 << ((2*n - 1) -1).bit_length()
        ft = np.fft.rfft(v, n=size)
        conv = np.fft.irfft(ft*ft, n=size)[:2*n-1]
        return 2.0 * n * conv.max()

    # Phase 1: generate seeds
    seeds = []

    for n in lengths:
        # arcsine
        i = np.arange(n, dtype=np.float64)
        f = 1.0 / np.sqrt((i+0.5)*(n - i - 0.5))
        f = f / f.sum()
        seeds.append((f.copy(), cheap_score(f)))
        # beta symmetric with p from set
        for p in [0.2,0.4,0.5,0.7,1.0,1.5,2.0]:
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            w = np.exp((p-1)*np.log(x) + (p-1)*np.log(1-x))

"""
Propose a step‑function (non‑negative vector) which minimises
``evaluate_sequence``.

Algorithm
---------
1. **Base families** – for a few lengths (≈500–3000) generate analytic
   candidates (arcsine, symmetric Beta, asymmetric Beta, random Dirichlet).
2. **Tilt search** – for each base candidate apply a multiplicative
   exponential tilt `exp(theta*(i-(n-1)/2))`.  Scan a dense set of `theta`
   values; keep the best (lowest cheap score = 2·n·max(f*f)).
3. **LP improvement** – starting from the best tilted seed, solve a
   linear program

        maximise   Σ g_j
        subject to   (f * g)_i ≤ M   for all i
                     g_j ≥ 0

   where `M = max(f*f)`.  The constraint matrix is a Toeplitz matrix built
   from `f`.  The LP yields a vector `g` that carries as much mass as
   possible while never exceeding the current worst convolution value.
   Rescale `g` to unit sum and replace `f ← (1−t) f + t g` (t≈0.5, then
   reduced).  Repeat a few times – this strictly reduces the objective.
4. **Noisy sub‑gradient descent** – from the LP‑improved vector perform
   many iterations of sub‑gradient descent on the exact objective
   `max(f*f)`.  A small amount of Gaussian noise is added each step,
   and a momentum term is used.  After every step the vector is projected
   back onto the probability simplex.
5. **Hill‑climbing** – final random mass‑transfer moves (tiny δ) and
   occasional jitter finish the optimisation.
6. Return the best vector found (any positive scaling gives the same
   value, we return the unit‑sum version).

All operations stay on the simplex, therefore the scale‑invariant
objective of ``evaluate_sequence`` is unchanged.
"""

import time
import numpy as np
import scipy.sparse

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    theta = 0.0 if rho.size == 0 else cssv[rho[-1]] / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.

    Returns:
        grad   – gradient of the active max‑convolution term,
        curmax – current maximum value of a*a,
        active – indices where the maximum is attained (within eps).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # Indicator vector for active constraints
    w = np.zeros_like(b)
    w[active] = 1.0

    # Convolution of a (reversed) with w gives gradient pre‑correction
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # Diagonal correction (indices where k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _beta_asymmetric_distribution(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta(a,b) on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5))."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _tilted_distribution(base: np.ndarray, theta: float) -> np.ndarray:
    """Multiply `base` by exp(theta*(i-(n-1)/2)) and re‑normalise."""
    n = base.shape[0]
    idx = np.arange(n) - (n - 1) / 2.0
    vec = base * np.exp(theta * idx)
    vec = np.maximum(vec, 0.0)
    s = vec.sum()
    if s == 0.0:
        return base.copy()
    return vec / s


# ----------------------------------------------------------------------
def _solve_lp_max_sum(f: np.ndarray, M: float):
    """
    Solve   maximise Σ g_j   s.t.   (f * g)_i ≤ M,   g_j ≥ 0.

    Returns the optimal g (may have sum > 0) or ``None`` on failure.
    """
    n = f.shape[0]
    # Build Toeplitz constraint matrix A where A[i,j] = f[i-j] if i>=j else 0.
    # This is a set of n diagonals with constant entries f[d] (d = i-j).
    data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)                     # 0 … n‑1
    A = scipy.sparse.diags(data, offsets,
                           shape=(2 * n - 1, n), format='csr')
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)                           # minimise -Σg  → maximise Σg
    bounds = [(0.0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b,
                                bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Length(s) to explore – around the size of the previously known best.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3000

    lengths = set()
    for factor in (0.8, 0.9, 1.0, 1.1, 1.2):
        n = int(base_n * factor)
        if n >= 500:
            lengths.add(n)
    # a few random larger lengths
    for _ in range(3):
        n = np.random.randint(500, min(8000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # 1. Broad generation of analytic families and random draws
    candidates = []   # (vector, cheap_score)

    beta_ps = [0.2, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # arcsine – our strongest analytic baseline
        base = _arcsine_distribution(n)
        score = _max_convolution_score(base)
        candidates.append((base.copy(), score))

        # symmetric beta family
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a.copy(), score))

        # asymmetric beta (random a,b)
        for _ in range(2):
            a_par = np.random.uniform(0.2, 3.0)
            b_par = np.random.uniform(0.2, 3.0)
            a_vec = _beta_asymmetric_distribution(n, a_par, b_par)
            score = _max_convolution_score(a_vec)
            candidates.append((a_vec.copy(), score))

        # a few random Dirichlet vectors
        for _ in range(2):
            a_vec = _random_dirichlet(n)
            score = _max_convolution_score(a_vec)
            candidates.append((a_vec.copy(), score))

        # 2. Tilt scan around the arcsine base (introduces asymmetry)
        max_theta = 5.0
        for theta in np.linspace(-max_theta, max_theta, 61):
            if time.time() >= deadline:
                break
            tilted = _tilted_distribution(base, theta)
            score = _max_convolution_score(tilted)
            candidates.append((tilted.copy(), score))

    # ------------------------------------------------------------------
    # 3. Include previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev.copy(), score))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Keep a handful of promising seeds
    candidates.sort(key=lambda x: x[1])
    top_k = 7
    seeds = candidates[:top_k]

    print(f"[{time.time():.1f}] kept {len(seeds)} seeds, best cheap score {seeds[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 5. LP‑based improvement for each seed (few iterations, cheap)
    def lp_improve(vec: np.ndarray, t0: float = 0.5, steps: int = 4) -> np.ndarray:
        """Iteratively improve `vec` using the LP trick described in the statement."""
        cur = vec.copy()
        for s in range(steps):
            if time.time() >= deadline:
                break
            # current M = max convolution value
            conv = _fft_self_convolve(cur)
            M = conv.max()
            g0 = _solve_lp_max_sum(cur, M)
            if g0 is None or g0.sum() == 0.0:
                break
            g = g0 / g0.sum()
            # convex update
            t = t0 * (0.9 ** s)          # slowly decaying step size
            cur = (1.0 - t) * cur + t * g
            cur = _project_onto_simplex(cur, s=1.0)
        return cur

    improved_seeds = []
    for idx, (vec, cheap) in enumerate(seeds):
        if time.time() >= deadline:
            break
        vec_imp = lp_improve(vec)
        new_cheap = _max_convolution_score(vec_imp)
        improved_seeds.append((vec_imp, new_cheap))
        if new_cheap < cheap - 1e-12:
            print(f"[{time.time():.1f}] seed {idx+1}/{len(seeds)} LP improved {cheap:.6f} → {new_cheap:.6f}")

    # ------------------------------------------------------------------
    # 6. Noisy sub‑gradient descent on the best seed
    best_vec, best_score = min(improved_seeds, key=lambda x: x[1])
    n = best_vec.shape[0]
    m = np.zeros_like(best_vec)          # momentum term
    beta = 0.9
    it = 0
    max_iters = 300_000

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(best_vec)
        # add diminishing isotropic noise to break symmetry
        noise_scale = 0.02 / np.sqrt(it)
        grad = grad + noise_scale * np.random.randn(n)

        # momentum update
        m = beta * m + (1.0 - beta) * grad

        # diminishing step size
        eta = 0.6 / np.sqrt(it)
        best_vec = best_vec - eta * m
        best_vec = _project_onto_simplex(best_vec, s=1.0)

        cur_score = _max_convolution_score(best_vec)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            print(f"[{time.time():.1f}] sub‑grad it {it} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Final hill‑climbing (tiny random mass transfers)
    a = best_vec.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02

    while time.time() < deadline:
        iteration += 1
        transfer = max_transfer * (0.5 ** (iteration / 12_000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_vec = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter {iteration} new best {best_score:.6f}")

        if iteration % 5_000 == 0 and np.random.rand() < 0.20:
            # occasional jitter to hop out of flat plateaus
            jitter = np.random.randn(n) * 5e-5
            a = best_vec + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"Final cheap score {best_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()
