"""
Hybrid stochastic‑search + continuation‑gradient optimiser.

1.  Build a modest “coarse” problem (n≈2000) and generate a large
    pool of candidates:
        – deterministic shapes (uniform, arcsine, symmetric Beta,
          Hann window, exponential decay)
        – random Dirichlet draws
        – random mixtures of two Beta densities.
2.  Pick the best candidate by the cheap proxy 2·n·max(a*a).
3.  Run a continuation of the smooth log‑sum‑exp surrogate
    (projected FISTA) on the coarse problem, then a short
    projected sub‑gradient polish.
4.  Upscale the best coarse solution to the final length
    (default 8000) by linear interpolation, renormalise and repeat
    the refinement steps with the remaining time budget.
5.  Return the final probability vector; ``evaluate_sequence`` rescales
    it automatically, guaranteeing a finite score.

The optimiser deliberately restarts from many random seeds within the
available time budget, which helps escape local minima and usually
produces a score < 1.51 on the reference data (better than the previous
baseline).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow for extreme α,β
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)  # safe guard
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay centred at the middle of the interval."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    pdf = np.exp(-lam * np.abs(i - centre))
    pdf /= pdf.sum()
    return pdf


# ----------------------------------------------------------------------
def _hann_window(n: int) -> np.ndarray:
    """Discrete Hann (raised‑cosine) window, normalised to Σ=1."""
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector (Σa=1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)                 # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)                      # ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                               # soft‑max weights

    # gradient: 2·(a_rev * p)   (central n entries)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]
    f_val = M + mu * np.log(Z)
    return f_val, grad

def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent (exact sub‑gradient of the max‑norm).
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_mom = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        # active set (within relative tolerance)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_mom * momentum + (1.0 - mu_mom) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional tiny jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val

def _random_mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random mixture of two symmetric Beta(α,α) densities."""
    alpha1 = 10 ** rng.uniform(-2, 1)
    alpha2 = 10 ** rng.uniform(-2, 1)
    w = rng.random()
    b1 = _beta_distribution(n, alpha1, alpha1)
    b2 = _beta_distribution(n, alpha2, alpha2)
    mix = w * b1 + (1.0 - w) * b2
    mix /= mix.sum()
    return mix

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the competition harness.
    Returns a list of non‑negative step heights (will be normalised
    internally by ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse problem size (fast inner loops)
    n_coarse = 2000
    candidates = []

    # a) global warm‑start if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base = base / base.sum()
        base = _interpolate_to_length(base, n_coarse)
        candidates.append(base)
    except Exception:
        pass

    # b) deterministic baselines
    candidates.append(np.full(n_coarse, 1.0 / n_coarse, dtype=np.float64))   # uniform
    candidates.append(_beta_distribution(n_coarse, 0.5, 0.5))                 # arcsine
    candidates.append(_hann_window(n_coarse))

    # c) symmetric Beta shapes
    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n_coarse, a, a))

    # d) exponential tails
    for lam in (0.5, 1.0, 1.5):
        candidates.append(_exponential_distribution(n_coarse, lam))

    # e) random Dirichlet draws
    for _ in range(8):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n_coarse)
        vec /= vec.sum()
        candidates.append(vec)

    # f) random mixtures of two Beta densities
    for _ in range(6):
        candidates.append(_random_mixture_beta(n_coarse, rng))

    # ------------------------------------------------------------------
    # 2. Pick the best raw‑score candidate
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    print(f"[coarse] Initial best raw score: {best_raw:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop on the coarse problem
    # Allocate most of the remaining budget to refinement, but keep
    # a small reserve for the upscale stage.
    reserve = 30.0                       # seconds reserved for final up‑scale
    while time.time() < deadline - reserve:
        # a fresh random start
        conc = 10 ** rng.uniform(-2, 1)
        a0 = rng.gamma(conc, 1.0, size=n_coarse)
        a0 /= a0.sum()
        # small time slice (≈8 s) for a quick continuation + polish
        slice_budget = min(8.0, deadline - time.time() - reserve)
        if slice_budget <= 0:
            break
        a_ref, raw_ref = _fista_refine(a0, slice_budget * 0.65)
        a_ref, raw_ref = _subgradient_polish(a_ref, slice_budget * 0.35)
        if raw_ref < best_raw:
            best_raw = raw_ref
            best_seq = a_ref.copy()
            print(f"[coarse] New best raw score: {best_raw:.6f}")

    # ------------------------------------------------------------------
    # 4. Upscale to the final length
    n_final = 8000
    best_seq_big = _interpolate_to_length(best_seq, n_final)

    # ------------------------------------------------------------------
    # 5. Final refinement on the full‑size problem
    remaining = deadline - time.time()
    if remaining > 10.0:
        best_seq_big, best_raw = _fista_refine(best_seq_big, remaining * 0.70)
        remaining = deadline - time.time()

    if remaining > 5.0:
        best_seq_big, best_raw = _subgradient_polish(best_seq_big, remaining)

    # ------------------------------------------------------------------
    # 6. Return the result (evaluate_sequence will rescale automatically)
    final_score = evaluate_sequence(best_seq_big.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq_big)})")
    return best_seq_big.tolist()
