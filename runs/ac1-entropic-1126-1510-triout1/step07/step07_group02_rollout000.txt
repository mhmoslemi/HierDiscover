(the same objective used by ``evaluate_sequence``).  The algorithm works
in three stages:

1. **Exploratory initialization** – try many analytic families
   (uniform, arcsine, triangular, exponential, beta, sine‑power,
   cosine windows) for several lengths and keep the best shape.

2. **Stochastic refinement** – a simulated‑annealing loop that repeatedly
   transfers a small amount of mass from a source index to a target index.
   Moves that lower the current max‑convolution are always accepted,
   otherwise they are accepted with probability `exp(-Δ/T)`.  The
   temperature ``T`` cools exponentially from an initial value to a
   very small final value, allowing the search to escape local minima
   early and to fine‑tune later.

3. **Deterministic polishing** – a short greedy hill‑climbing phase that
   only keeps improving moves, to clean up any remaining noise.

All updates of the self‑convolution are performed incrementally
(`_apply_transfer`) in O(n) time, and the algorithm respects the 1000 s
budget.  The best sequence found is returned as a plain ``list[float]``.
"""

import time
import random
import math
import numpy as np

# -------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# -------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()           # use the *old* a for the linear part
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# -------------------------------------------------------------
# Analytic families – each returns a non‑negative vector summing to 1
# -------------------------------------------------------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative step function (list of floats) that attempts to
    minimise the objective used by ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds (default 1000).
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 3.0          # safety margin

    eps = 1e-12

    # -----------------------------------------------------------------
    # 1.   Broad search over analytic families for several lengths
    # -----------------------------------------------------------------
    lengths = [600, 1200, 2000, 3000, 4000, 5000]
    best_seq = None
    best_score = float('inf')
    best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.003),
            _distribution_exponential(n, lam=0.010),
        ]

        # beta‑symmetric
        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            candidates.append(_distribution_beta_sym(n, p))

        # sine‑power windows
        for pw in (0.2, 0.5, 0.8, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))

        # cosine windows
        for c in (0.5, 0.46, 0.42, 0.38):
            candidates.append(_distribution_cosine_window(n, c))

        # a sparsish random start
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 150)                       # a few non‑zeros
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        candidates.append(sparse)

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} new init best {sc:.6f}")

    # -----------------------------------------------------------------
    # 2.   Incorporate a previously found solution, if present
    # -----------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best candidate {sc:.6f}")
    except Exception:
        pass

    # Fallback (should never happen)
    if best_seq is None:
        best_n = 2000
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3.   Stochastic refinement (simulated annealing)
    # -----------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max
    best_score = cur_score
    best_seq = a.copy()

    # Temperature schedule – exponential cooling from 10 % of the current
    # score down to a negligible value.
    T_start = best_score * 0.10
    T_end = best_score * 1e-5
    refine_start = time.time()
    total_refine = deadline - refine_start

    it = 0
    while time.time() < deadline:
        # ----- temperature based on elapsed refinement time -----
        elapsed = time.time() - refine_start
        frac = min(elapsed / total_refine, 1.0)   # 0 → 1
        T = T_start * ((T_end / T_start) ** frac)

        # ----- propose a move -------------------------------------------------
        # source drawn proportional to current mass (so that we move from
        # “big” entries more often); target uniform.
        if a.sum() == 0:
            break
        source = np.random.choice(n, p=a / a.sum())
        max_delta = min(a[source] * 0.5, 0.05)
        # shrink maximum delta as temperature cools
        max_delta = max_delta * (1.0 - frac) + eps
        delta = max_delta * np.random.rand()
        if delta < eps:
            continue
        target = np.random.randint(n)
        while target == source:
            target = np.random.randint(n)

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_max = conv.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - cur_score

        # ----- acceptance test -----------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-delta_score / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            cur_max = new_max
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                # occasional progress report
                if it % 5000 == 0:
                    print(f"[{time.time():.1f}] SA iter={it} best {best_score:.6f}")
        else:
            # reject: revert
            a = a_snap
            conv = conv_snap

        # occasional global jitter to escape plateaus
        if it % 20000 == 0 and random.random() < 0.15:
            sigma = 0.005 * (1.0 - frac) + 1e-4
            a_jitt = a + sigma * np.random.randn(n)
            a_jitt = np.maximum(a_jitt, eps)
            a_jitt = _project_onto_simplex(a_jitt, 1.0)
            conv = _fft_self_convolve(a_jitt)
            a = a_jitt
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

        it += 1

    # -----------------------------------------------------------------
    # 4.   Final greedy hill‑climbing (only improving moves)
    # -----------------------------------------------------------------
    # Give it a modest amount of time (≈10 % of the remaining budget)
    final_deadline = start + budget_s - 2.0
    while time.time() < final_deadline:
        # pick the pair (i,j) that currently contributes most to the maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]

        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta < eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_max = conv.max()
        if new_max < cur_max - 1e-13:
            cur_max = new_max
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
        else:
            a = a_snap
            conv = conv_snap

    print(f"Final best score ≈ {best_score:.6f} (length={n})")
    # Return as plain python list
    return best_seq.tolist()
