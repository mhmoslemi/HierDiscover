def _smooth_grad(a, beta):
    conv = _self_convolution_full(a)

    exp_beta = np.exp(beta * (conv - maxc))

"""
propose_candidate
-----------------
Search for a non‑negative step function that minimizes
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a multi‑phase stochastic optimizer:

1️⃣  **Coarse random sampling** – generate many candidates on moderate
    lengths (2k‑8k) from Dirichlet draws, Beta PDFs and simple
    mixtures.  Keep the best (lowest raw score, i.e. with Σa=1).

2️⃣  **Additive sub‑gradient descent** – minimise the exact sup‑norm
    using the sub‑gradient of   f(a)=max_k (a*a)_k .
    A diminishing step size η=η₀/√t and Euclidean projection onto the
    simplex are used.  Improvements are printed and the best vector is
    retained.

3️⃣  **Smoothed (log‑sum‑exp) gradient descent** – replace the non‑smooth
    max by the smooth surrogate
        g_β(a)= (1/β)·log Σ_k exp(β·(a*a)_k)
    whose gradient is a weighted convolution.  β starts small (≈0.5)
    and is multiplied by a factor every few hundred iterations, so the
    optimisation gradually focuses on the active constraints.
    Again a simple projected gradient step with η∝1/√t is performed.

4️⃣  **Upsampling** – the current best sequence is linearly interpolated
    to larger lengths (12k‑20k) and the smoothed descent is applied again.
    The best overall result (lowest exact score) is returned.

All updates keep the vector non‑negative and normalised (∑a=1), which
matches the scale‑invariance of `evaluate_sequence`.  The routine runs
until the supplied time budget (default 1000 s) expires and then calls
`evaluate_sequence` for the final exact score.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) PDF on n points, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable version
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The value equals the true objective because the evaluator rescales
    a to the same sum before computing the final score.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of   f(a) = max_k (a*a)_k .
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        g_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

    The gradient is 2·(a_rev * softmax_weights) restricted to the
    original length.
    """
    n = a.shape[0]
    conv = _self_convolution_full(a)          # length 2n‑1
    maxc = conv.max()
    # softmax with subtraction of maxc for numerical stability
    exp_beta = np.exp(beta * (conv - maxc))
    Z = exp_beta.sum()
    w = exp_beta / Z                         # non‑negative, sums to 1
    # convolution with the reversed vector
    grad_full = _fft_convolve(a[::-1], w)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _refine_additive_subgrad(a: np.ndarray,
                            deadline: float,
                            rng: np.random.Generator,
                            eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the exact sup‑norm.
    Returns the best vector found before `deadline` and its raw score.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        eta = eta0 / math.sqrt(it)

        # tentative update
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new)

        val = _raw_score(a_new)
        if val + 1e-12 < best_val:
            best_val = val
            best = a_new.copy()
            a = a_new
            print(f"[{time.time():.1f}] additive subgrad it={it} score={best_val:.6f}")
        else:
            # occasional uphill move to escape shallow plateaus
            if rng.random() < 0.01:
                a = a_new
        # jitter every few hundred steps
        if it % 500 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)
    return best, best_val


def _refine_smoothing(a: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      beta_start: float = 0.5,
                      beta_factor: float = 1.2,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the log‑sum‑exp surrogate.
    The smoothing parameter β is multiplied by `beta_factor` every
    `beta_step` iterations, gradually tightening the approximation to
    the true max.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = _raw_score(a)
    it = 0
    beta = beta_start
    beta_step = 800  # increase β every `beta_step` iterations
    while time.time() < deadline:
        it += 1
        if it % beta_step == 0:
            beta = min(beta * beta_factor, 300.0)

        grad = _smooth_grad(a, beta)
        eta = eta0 / math.sqrt(it)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new)

        val = _raw_score(a_new)
        if val + 1e-12 < best_val:
            best_val = val
            best = a_new.copy()
            a = a_new
            print(f"[{time.time():.1f}] smoothing β={beta:.2f} it={it} score={best_val:.6f}")
        else:
            # keep the older a (no acceptance)
            pass

        if it % 500 == 0 and rng.random() < 0.04:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)
    return best, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point: search for a low‑score step function
    within the allowed time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a possibly pre‑computed global variable
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if base_seq.sum() > 0.0:
            base_seq = base_seq / base_seq.sum()
        else:
            raise ValueError
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # Initialise best candidate
    if base_seq is not None:
        best_seq = base_seq.copy()
    else:
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)   # uniform 2k
    best_score = _raw_score(best_seq)
    best_len = len(best_seq)
    print(f"[{time.time():.1f}] initial best raw score = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 1 – coarse random exploration on moderate lengths
    coarse_lengths = [2000, 4000, 8000]
    for n in coarse_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] coarse exploration length={n}")

        # start from the current best, interpolated to the new length
        cand = _interpolate_to_length(best_seq, n)
        sc = _raw_score(cand)
        if sc < best_score:
            best_seq, best_score, best_len = cand, sc, n
            print(f"   interpolated best score={best_score:.6f}")

        # deterministic baselines
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_seq, best_score, best_len = uniform.copy(), sc, n
            print(f"   uniform score={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_seq, best_score, best_len = arcsine.copy(), sc, n
            print(f"   arcsine score={best_score:.6f}")

        # random Dirichlet draws
        for _ in range(250):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-2, 1)          # concentration in [0.01, 10]
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   dirichlet conc={conc:.3f} score={best_score:.6f}")

        # random mixtures of two Beta PDFs
        for _ in range(150):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mix2 λ={lam:.3f} score={best_score:.6f}")

        # mixture of three Betas (lighter sampling)
        for _ in range(100):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a3, b3 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            weights = rng.dirichlet([1, 1, 1])
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            p3 = _beta_distribution(n, a3, b3)
            vec = weights[0] * p1 + weights[1] * p2 + weights[2] * p3
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mix3 score={best_score:.6f}")

    print(f"[{time.time():.1f}] after coarse search best raw score = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 2 – additive sub‑gradient refinement on the current best length
    if time.time() < deadline:
        print(f"[{time.time():.1f}] starting additive sub‑gradient refinement (len={best_len})")
        refined, score = _refine_additive_subgrad(best_seq, deadline, rng, eta0=0.5)
        if score < best_score:
            best_seq, best_score, best_len = refined, score, len(refined)
            print(f"[{time.time():.1f}] after additive refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – smoothed log‑sum‑exp gradient descent (still on best_len)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] starting smoothed gradient refinement (len={best_len})")
        refined, score = _refine_smoothing(best_seq, deadline, rng,
                                           beta_start=0.5, beta_factor=1.2, eta0=0.6)
        if score < best_score:
            best_seq, best_score, best_len = refined, score, len(refined)
            print(f"[{time.time():.1f}] after smoothing refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – up‑sample to larger lengths and repeat smoothing refinement
    larger_lengths = [12000, 16000, 20000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)
        a, scr = _refine_smoothing(a, deadline, rng,
                                   beta_start=1.0, beta_factor=1.3, eta0=0.6)
        if scr + 1e-12 < best_score:
            best_seq, best_score, best_len = a.copy(), scr, n
            print(f"   new best after up‑sample: score={best_score:.6f} (len={best_len})")
        if best_score <= 1.503:
            print("[early exit] target bound reached.")
            break

    # ------------------------------------------------------------------
    # Final exact evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
