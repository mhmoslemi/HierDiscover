def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1 (exact surrogate for evaluate_sequence)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)

Propose a low‑score non‑negative step function.

The algorithm proceeds in three stages:

1. **Broad sampling** – a diversified set of analytic families
   (arcsine, triangular, (left‑hand) exponential, power‑law,
   centered Gaussian, symmetric beta) as well as a few Dirichlet draws
   are generated for a handful of lengths.  Every candidate is
   ℓ₁‑normalised (sum = 1) and scored with the cheap surrogate
   ``2·n·max(conv(a,a))`` (exact for sum = 1).

2. **Refinement** – the top‑k candidates are polished with three
   complementary first‑order schemes:
   * classic Adam on the sub‑gradient of the max‑convolution,
   * exponentiated‑gradient (KL‑mirror) descent,
   * *smoothed* Adam on a temperature‑controlled log‑sum‑exp
     approximation of the max.  The smoothed objective
     ``Fτ(a) = (1/τ)·log Σₖ exp(τ·(a*a)ₖ)`` allows a smoother
     landscape and often finds deeper minima.

3. **Pairwise Frank‑Wolfe (FW) sweeps** – the current best sequence is
   repeatedly improved by moving mass from the component with the
   largest sub‑gradient to the component with the smallest one, with
   a simple line‑search (full transfer, then half transfer) and occasional
   Gaussian jitter to escape plateaus.

All work respects a 1000 s budget and always returns a list of non‑negative
floats that can be passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1 (exact surrogate for evaluate_sequence)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (sum(a)=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (subtract a_i for even active indices)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150000,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style adaptive moment optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 120000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Mirror descent with KL divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray,
                         deadline: float,
                         max_iters: int = 100000,
                         tau_start: float = 0.5,
                         tau_end: float = 30.0,
                         eta0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed objective
      Fτ(a) = (1/τ)·log Σₖ exp(τ·(a*a)ₖ)
    with τ increasing from ``tau_start`` to ``tau_end``.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    beta1 = 0.9
    beta2 = 0.999
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # temperature schedule (log‑linear)
        tau = tau_start * (tau_end / tau_start) ** (t / max_iters)

        b = _fft_self_convolve(a)                     # shape (2n‑1,)
        maxb = b.max()
        # stable soft‑max
        shifted = tau * (b - maxb)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()           # probabilities, sum=1

        # gradient of the smoothed objective
        conv_rev = _fft_convolve(a[::-1], w)          # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
        diag_w = w[2 * np.arange(n)]                  # w at even positions
        grad -= a * diag_w

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = eta0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(alpha,…,alpha)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential centred at the middle."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _left_exponential_distribution(n: int, lam: float = 0.02) -> np.ndarray:
    """Monotonically decreasing exponential starting at the left edge."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, alpha: float = 1.0) -> np.ndarray:
    """Monotone power‑law a_i ∝ (i+1)^(-alpha)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    vals = idx ** (-alpha)
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian (σ given in absolute index units)."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The function respects the time budget and returns
    the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1. Choose a modest set of lengths (including the previous one)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 4000

    lengths = set()
    for scale in (0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0):
        n = int(base_n * scale)
        if 500 <= n <= 25000:
            lengths.add(n)
    # a few random lengths for diversity
    for _ in range(5):
        n = np.random.randint(500, min(25000, base_n * 4) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # --------------------------------------------------------------
    # 2. Broad sampling from analytic families + Dirichlet draws
    candidates = []   # list of (array, surrogate_score)

    beta_ps = list(np.arange(0.02, 0.21, 0.02)) \
        + list(np.arange(0.25, 0.71, 0.05)) \
        + list(np.arange(0.80, 1.51, 0.10)) \
        + [2.0]

    dirichlet_alphas = [0.2, 0.5, 1.0, 2.0]

    power_law_alphas = [0.1, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0]
    gaussian_sigmas = [0.05, 0.10, 0.15, 0.20]   # as a fraction of n
    left_exp_lams = [0.01, 0.02, 0.05]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Analytic families (always cheap)
        for gen, name in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.01), "sym‑exp"),
        ):
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:<5d} {name:12s} score={score:.6f}")

        # Left‑hand exponential (asymmetric)
        for lam in left_exp_lams:
            a = _left_exponential_distribution(n, lam=lam)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Power‑law family
        for alpha in power_law_alphas:
            a = _power_law_distribution(n, alpha=alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Centered Gaussian (different bandwidths)
        for sf in gaussian_sigmas:
            sigma = sf * n
            a = _gaussian_distribution(n, sigma=sigma)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Symmetric beta (dense grid)
        for p in beta_ps:
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # Random Dirichlet draws
        for alpha in dirichlet_alphas:
            a = _random_dirichlet_alpha(n, alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))

    # ------------------------------------------------------------------
    # 3. Include previously discovered best sequence (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Keep the top seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(8, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, kept top {top_k} for refinement.")
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]
    print(f"Current best seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Refinement of top seeds
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Adam on the crude sub‑gradient
        seq, sc = _adam_descent(seed_seq, deadline,
                                 max_iters=80000,
                                 step0=0.15)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam improved to {best_score:.6f}")

        # Exponentiated‑gradient (KL‑mirror)
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=60000,
                                        eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} ExpGrad improved to {best_score:.6f}")

        # Smoothed Adam (log‑sum‑exp approximation)
        seq, sc = _smooth_adam_descent(seed_seq, deadline,
                                       max_iters=60000,
                                       tau_start=0.5,
                                       tau_end=30.0,
                                       eta0=0.12)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} SmoothAdam improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Pairwise Frank‑Wolfe style mass‑transfer refinement
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 60000:
        it_fw += 1
        grad, cur_max, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # mass that harms the max most
        i_dst = int(np.argmin(grad))   # mass that helps the most
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-12:
            a = a_full
            cur_score = score_full
            improved = True
        else:
            # fallback: half transfer
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-12:
                a = a_half
                cur_score = score_half
                improved = True

        if improved:
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] FW iter={it_fw} new best {best_score:.6f}")
            continue

        # occasional jitter to escape tiny plateaus
        if it_fw % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 1e-6
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
