import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # Build A (2n-1 x n) as a dense numpy array maybe.
    # A[k, j] = f[k - j] if 0 <= k - j < n else 0
    # Then constraints A @ g <= M * np.ones(2n-1)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(n):
        # row indices k = j...j+n-1 inclusive
        k_start = j
        k_end = j + n  # exclusive
        A[k_start:k_end, j] = f
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver='ECOS', warm_start=True, max_iters=5000)
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        # fallback to zeros or uniform?
        return np.zeros_like(f)
    sol = g.value
    # ensure nonneg
    sol = np.maximum(sol, 0.0)
    return sol
```

def _lp_optimize(f, M):
    n = len(f)
    A = _conv_matrix(f)
    # bounds for variables (0, None)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve':True, 'time_limit': 20})
    if not res.success:
        return np.zeros(n)
    g = res.x
    # clip to nonneg
    g = np.maximum(g, 0)
    return g
```

            # if not improved, try smaller tau

        return np.zeros_like(f)
    g = np.maximum(res.x, 0.0)
    return g

"""
Propose a low‑score step function.

The algorithm consists of three stages:

1️⃣ **Coarse random search** on moderate lengths (2‑4 k).  A variety of
   Beta‑, Dirichlet‑ and mixture distributions are sampled, normalised to
   Σa=1 and scored with the exact `max(a*a)` (fast FFT).

2️⃣ **Refinement** – first a mirror‑descent (exponentiated‑gradient) on the
   non‑convex objective `max_k(a*a)_k`, then an *LP‑boost*:
   for the current vector `f` we solve the linear program
   `max Σ g_j   s.t.  f*g ≤ M ,   g≥0` where `M = max(f*f)`.
   The optimal `g` is mixed into `f` (convex combination) and the process
   is repeated a few times.  This step systematically lowers the
   convolution bound.

3️⃣ **Upsampling** – the best vector is interpolated to larger lengths
   (8 k, 12 k, …) and refined again with mirror‑descent (and a final LP
   step for the 8 k case).  The whole routine stops as soon as the
   1000 s budget is exhausted or the target bound `≤ 1.503` is reached.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on a grid of size n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)       # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Score for a vector already normalised to Σa=1 (scale‑invariant)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Sub‑gradient of f(a)=max_k (a*a)_k  (averaged over active set)."""
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * np.fft.irfft(
        np.fft.rfft(a_rev) * np.fft.rfft(w, n=1 << ((a_rev.size + w.size - 1 - 1).bit_length())),
        n=1 << ((a_rev.size + w.size - 1 - 1).bit_length())
    )[:n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_mirror_descent(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) descent on the simplex."""
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        eta = eta0 / math.sqrt(it)
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)
        a_sum = a.sum()
        if a_sum == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a_sum

        # occasional jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.06:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror‑descent len={n} it={it} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the (2n‑1)×n matrix A with A[k,j]=f[k‑j] (zero outside)."""
    n = f.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_maximize_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ g_j   s.t.   f * g ≤ M ,   g ≥ 0 .
    Returns the optimal g (may be all‑zero if the LP fails).
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –Σg  ≡ maximise Σg
    bounds = [(0, None)] * n
    # Highs is fast and open‑source; we give it a modest time limit.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={"presolve": True, "time_limit": 5.0})
    if not res.success or res.x is None:
        return np.zeros_like(f)
    g = np.maximum(res.x, 0.0)
    return g


def _lp_improve(a: np.ndarray, deadline: float,
                rng: np.random.Generator, max_iters: int = 15) -> tuple[np.ndarray, float]:
    """
    Repeated LP‑boost: solve the linear program for the current a,
    mix the solution back with a (convex combination) and iterate.
    Stops when time expires or no further improvement is observed.
    """
    best = a.copy()
    best_score = _raw_score(best)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        M = _max_self_convolution(best)
        g = _lp_maximize_sum(best, M)
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g

        improved = False
        tau = 0.5
        while tau > 1e-3:
            cand = (1.0 - tau) * best + tau * g_norm
            cand = _project_onto_simplex(cand)
            val = _raw_score(cand)
            if val + 1e-12 < best_score:
                best, best_score = cand, val
                improved = True
                print(f"[{time.time():.1f}] LP‑boost iter={it} τ={tau:.3f} score={best_score:.6f}")
                break
            tau *= 0.5
        if not improved:
            break
    return best, best_score
