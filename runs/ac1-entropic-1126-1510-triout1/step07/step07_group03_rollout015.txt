"""
Greedy active‑set optimizer for min‑max convolution.
Starts from a rich set of analytic distributions (uniform,
arcsine, symmetric beta with many shape parameters, exponential,
alternating patterns and random perturbations).  The best
distribution (normalised to sum = 1) is refined by an aggressive
pairwise transfer loop: at each step the convolution is computed,
the shift(s) attaining the maximum are identified and the
sub‑gradient is obtained analytically; mass is moved from the
index with largest gradient (where the max could increase most)
to the index with smallest gradient.  Full‑length FFT makes each
iteration cheap (≈ 0.2 ms for n≈ 3000) so thousands of iterations
fit comfortably into the 1000 s budget.  The algorithm also
occasionally performs a tiny random perturbation to escape
plateaus.  The best vector seen is returned as a plain list of
floats.
"""

import time, math, numpy as np

def _project_onto_simplex(v, s=1.0):
    # Euclidean projection of v onto the simplex {x >=0, sum(x)=s}
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _objective(a):
    # assumes sum(a)=1
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def _distribution_alternating(n, stride=2):
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a

def _initial_candidates(n, rng):
    cands = []
    cands.append(_distribution_uniform(n))
    cands.append(_distribution_arcsine(n))

    # beta with many shape parameters (including very small)
    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cands.append(_distribution_beta_sym(n, p))

    # exponentials
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cands.append(_distribution_exponential(n, lam))

    # alternating
    for stride in (2, 3, 4, 5):
        cands.append(_distribution_alternating(n, stride))

    # random sparse vectors (≈0.5 % non‑zero)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    rnd[idx] = rng.rand(k)
    rnd /= rnd.sum()
    cands.append(rnd)

    # small random perturbation of beta shapes
    for _ in range(3):
        p = rng.choice([0.01,0.03,0.05,0.1,0.2,0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.01, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert)
        cands.append(pert)

    return cands

def _greedy_local_search(a, deadline):
    """
    Aggressive pairwise transfer based on sub‑gradient of the active shift(s).
    Returns (best_vec, best_score).
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        # identify all shifts that are practically maximal
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # compute sub‑gradient g = 2 * sum_{k in active} a shifted by k
        g = np.zeros_like(a)
        for k in active_shifts:
            # i ranges where 0 <= k - i < n  -> i in [max(0, k-n+1), min(k, n-1)]
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        # source: index with largest gradient among those that actually hold mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        # destination: index with smallest gradient (could be zero)
        dest = int(np.argmin(g))

        if src == dest:
            # nothing to move – stop
            break

        # amount of mass to move – try the whole source mass first
        delta = a[src]
        # limit the step size to keep changes smooth
        # we keep a small fraction of the source mass to avoid oscillations
        delta = min(delta, 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta

        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            # accept improvement
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            # print progress occasionally
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy it={it} new best {best_score:.6f}")
            continue

        # try a smaller transfer if the full delta did not help
        delta *= 0.2
        if delta < 1e-9:
            # cannot make progress
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy small it={it} new best {best_score:.6f}")
            continue

        # occasional random jitter to escape plateaus
        if it % 500 == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                eps = min(a[i] * 0.05, 0.001)
                a_j = a.copy()
                a_j[i] -= eps
                a_j[j] += eps
                scr = _objective(a_j)
                if scr < best_score - 1e-14:
                    a = a_j
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector)
    that minimises ``evaluate_sequence``.  The routine creates a

    moves probability mass from the most offending indices to the

    """

    # choose a length; longer vectors give a larger denominator n
    # but also increase runtime of FFT; we use a moderate size.
    n = 3270   # close to the best length seen previously
    # ensure n is not too small for FFT padding
    if n < 2:
        n = 2

    # initialise candidates
    candidates = _initial_candidates(n, rng)

    # include previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append(prev)
    except Exception:
        pass

    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = cand / cand.sum()
        score = _objective(cand)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
            print(f"[{time.time():.1f}] init cand {idx} score={score:.6f}")

    # if for any reason we have no vector, fall back to uniform

        best_vec = _distribution_uniform(n)
        best_score = _objective(best_vec)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    best_vec, best_score = _greedy_local_search(best_vec, deadline)

    # final sanity – make sure sum = 1 (project onto simplex)
    best_vec = _project_onto_simplex(best_vec)
    final_score = _objective(best_vec)
    print(f"Final best score: {final_score:.6f} (length={len(best_vec)})")
    return best_vec.tolist()

"""
Greedy active‑set optimiser for the min‑max self‑convolution problem.

The algorithm works in three phases:

1. **Rich initialisation** – a catalogue of analytic families
   (uniform, arcsine, symmetric beta with many shape parameters,
   exponential, alternating, sparse random, plus noisy perturbations)
   is sampled for a fixed length `n`.  The best of these candidates
   (in terms of the objective 2·n·max(conv(a,a))) is retained.

2. **Aggressive greedy local search** – in each iteration the
   convolution is computed, the shift(s) attaining the maximal value
   are identified and the exact sub‑gradient for the active set is
   obtained analytically (no FFT required).  Mass is moved from the
   index with the largest gradient (where the max could increase most)
   to the index with the smallest gradient.  Full‑mass moves are
   attempted first; if they do not improve the objective a reduced
   transfer is tried.  Random jitter is injected periodically to
   escape plateaus.

3. **Simulated‑annealing fine‑tuning** – a cheap Metropolis walk
   performs small random transfers between positions while the
   temperature decays exponentially.  Improvements are kept.

All stages respect the global deadline (`budget_s`).  The final
vector is projected onto the simplex (non‑negative, sum = 1) and
returned as a plain Python list.  The implementation uses only NumPy
and runs comfortably within the 1000 s budget for `n≈3300`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a))  – assumes sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Produce a diverse set of starting vectors for a given length."""
    cand = []

    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    # symmetric beta – wide range of shape parameters (including very small)
    for p in (0.005, 0.01, 0.02, 0.03, 0.05, 0.07, 0.1,
              0.15, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – several decay rates
    for lam in (0.0005, 0.001, 0.002, 0.003, 0.005, 0.01):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4, 5):
        cand.append(_distribution_alternating(n, stride))

    # random sparse start (~0.5 % non‑zeros)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, k, replace=False)
    rnd[idx] = rng.random(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    # noisy perturbations of a few beta shapes
    for _ in range(3):
        p = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2, 0.5])
        base = _distribution_beta_sym(n, p)
        noise = rng.normal(scale=0.015, size=n)
        pert = base + noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert)
        cand.append(pert)

    return cand


# ----------------------------------------------------------------------
def _greedy_local_search(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Aggressive pairwise mass transfer based on the exact sub‑gradient of
    the active convolution shift(s).  Returns the best vector seen.
    """
    n = a.shape[0]
    best_vec = a.copy()
    best_score = _objective(a)
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active_shifts = np.where(b >= maxb - eps)[0]

        # sub‑gradient: g_i = 2 * sum_{k∈active} a_{k-i} (if the index exists)
        g = np.zeros_like(a)
        for k in active_shifts:
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            i = np.arange(i_low, i_high + 1, dtype=np.int64)
            j = k - i                     # the mirrored indices
            g[i] += 2.0 * a[j]

        # source – index with large gradient that actually carries mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]
        # destination – index with smallest gradient
        dest = int(np.argmin(g))

        if src == dest:
            break

        # try moving the whole source mass (capped to 0.5 for stability)
        delta = min(a[src], 0.5)

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)

        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy it={it} improved to {best_score:.6f}")
            continue

        # if full transfer fails, try a smaller fraction
        delta *= 0.2
        if delta < 1e-9:
            break
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dest] += delta
        new_score = _objective(a_new)
        if new_score < best_score - 1e-14:
            a = a_new
            best_vec = a.copy()
            best_score = new_score
            if it % 100 == 0:
                print(f"[{time.time():.1f}] greedy (small) it={it} improved to {best_score:.6f}")
            continue

        # occasional random jitter to shake the system out of plateaus
        if it % 500 == 0:
            i = rng.integers(n)
            if a[i] > 1e-12:
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                eps = min(a[i] * 0.04, 0.001)
                a_tmp = a.copy()
                a_tmp[i] -= eps
                a_tmp[j] += eps
                scr = _objective(a_tmp)
                if scr < best_score - 1e-14:
                    a = a_tmp
                    best_vec = a.copy()
                    best_score = scr
                    print(f"[{time.time():.1f}] jitter it={it} new best {best_score:.6f}")

    return best_vec, best_score

def _simulated_annealing(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Cheap Metropolis‑type random walk with exponential cooling.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)
    best_a = cur_a.copy()
    best_score = cur_score

    T = 0.03               # initial temperature (objective units)
    while time.time() < deadline and T > 1e-5:
        i = rng.integers(n)
        if cur_a[i] <= 1e-12:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer up to 10 % of source mass
        delta = rng.random() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[{time.time():.1f}] SA new best {best_score:.6f}")

        # exponential cooling
        T *= 0.9997

    return best_a, best_score
