def _simulated_annealing(a, score, deadline):
    cur_a = a.copy()
    cur_score = score
    best_a, best_score = a.copy(), score
    n = len(a)
    T0 = 0.01  # initial temperature
    # Use schedule T = T0 * (remaining_time / total_time)
    start = time.time()
    total = deadline - start
    iter = 0
    while time.time() < deadline:
        iter += 1
        # update temperature based on elapsed fraction
        elapsed = time.time() - start
        T = T0 * max(0.0, 1 - elapsed / total)
        # propose move: transfer mass from i to j
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(0.02, cur_a[i])  # adjust
        if delta <= 0:
            continue
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - cur_score) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur_a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score
```

def _simulated_annealing(a0, cur_score, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    iter = 0
    while time.time() < deadline:
        iter += 1
        # linear temperature schedule
        elapsed = time.time() - start
        T = max(1e-6, 0.02 * (1 - elapsed / total))  # start at 0.02 and decay
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * min(0.01, a[i])
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)
        if new_score < cur_score - 1e-15 or np.random.rand() < np.exp((cur_score - new_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score

"""
propose_candidate – a fast, multi‑phase optimizer for the step‑function
evaluation problem.

The optimizer works completely on the cheap proxy
    2·n·max(conv(a,a))      (a is assumed to sum to 1)
which is exactly the value returned by evaluate_sequence when the
sequence is normalised.  It consists of the following stages:

1.  **Broad sampling** – a rich set of analytic families (beta,
    arcsine, triangular, exponential, sine, cosine, uniform) and a few
    random Dirichlet draws are generated for several lengths n.
    The 10‑15 best candidates (according to the cheap score) are kept.

2.  **Accelerated sub‑gradient descent** – a Nesterov‑style
    sub‑gradient method on the true ‑∞ objective.  The method works
    directly on the simplex via Euclidean projection.

3.  **Smooth continuation** – the non‑smooth max is approximated by a
    soft‑max (log‑sum‑exp) with a schedule of increasing λ.  Gradient
    descent on this smooth surrogate quickly drives the autocorrelation
    towards a flat shape.

4.  **Adam refinement** – the Adam optimiser (projected on the simplex)
    gives a final fine‑grained descent.

5.  **Simulated‑annealing hill climbing** – tiny random mass transfers
    are accepted with a Metropolis criterion, allowing the search to
    escape shallow plateaus.

All phases respect the global 1000 s budget (a few seconds are saved
as a safety margin).  The returned list of non‑negative floats sums to 1,
so ``evaluate_sequence`` can be called on it directly.

The implementation avoids closures and lambdas, uses only ``numpy`` and
the standard library, and prints occasional progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # mask of active constraints
    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient before diagonal correction
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (once for k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _accelerated_subgradient(a0: np.ndarray,
                             deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Nesterov‑style sub‑gradient descent on the true L∞ objective.
    Returns the best vector seen and its cheap score.
    """
    a = a0.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        beta = (it - 1) / (it + 2)          # classic Nesterov schedule
        y = a + beta * (a - a_prev)        # extrapolation
        grad, _, _ = _subgrad(y)
        eta = eta0 / np.sqrt(it)
        a_new = _project_onto_simplex(y - eta * grad, s=1.0)
        a_prev = a
        a = a_new
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _softmax_gradient_descent(a0: np.ndarray,
                              deadline: float,
                              lambda_schedule=None,
                              max_iters_per: int = 80_000,
                              eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth log‑sum‑exp surrogate of the max.
    The λ‑parameter is increased (continuation) to approach the true max.
    """
    if lambda_schedule is None:
        lambda_schedule = [1.0, 5.0, 20.0, 80.0, 300.0, 1000.0]

    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]

    for lam in lambda_schedule:
        it = 0
        while time.time() < deadline and it < max_iters_per:
            it += 1
            conv = _fft_self_convolve(a)                     # length 2n‑1
            # stable soft‑max
            maxc = conv.max()
            exp_w = np.exp(lam * (conv - maxc))
            s = exp_w / exp_w.sum()                         # weights, sum=1

            # gradient = conv_rev(s) + a
            conv_rev = _fft_convolve(a[::-1], s)
            grad = conv_rev[n - 1 : n - 1 + n] + a

            eta = eta0 / np.sqrt(it)
            a = a - eta * grad
            a = _project_onto_simplex(a, s=1.0)

            cur = _max_convolution_score(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_a = a.copy()
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray,
                  deadline: float,
                  max_iters: int = 120_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimisation (projected onto the simplex) on the L∞ objective.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         cur_score: float,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Tiny random mass transfers accepted with a Metropolis rule.
    Temperature decays linearly; the routine finishes when time runs out.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = cur_score
    n = a.shape[0]
    start = time.time()
    total = deadline - start
    it = 0
    while time.time() < deadline:
        it += 1
        # linear temperature schedule (starts at ~0.02)
        elapsed = time.time() - start
        T = max(1e-9, 0.02 * (1.0 - elapsed / total))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(0.02, a[i])   # up to 2 % of the mass at i
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        # accept if better, otherwise with probability exp(-(Δ)/T)
        if new_score < cur_score - 1e-15 or np.random.rand() < np.exp((cur_score - new_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a_new.copy()
        # occasional jitter to avoid stagnation
        if it % 7_000 == 0 and np.random.rand() < 0.18:
            jitter = np.random.randn(n) * 4e-5
            a = best_a + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    return best_a, best_score
