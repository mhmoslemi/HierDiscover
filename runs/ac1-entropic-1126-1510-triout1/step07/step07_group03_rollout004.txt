"""
Propose a non‑negative step‑function (probability vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines a very rich pool of analytic start families,
multiple random restarts, an extended smoothed‑Adam continuation
phase, exact sub‑gradient descent, a multi‑pair greedy active‑set
swap, and a longer simulated‑annealing fine‑tuning.  All steps respect
a global time budget (default 1000 s).  The best vector found is
returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
#  Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Project v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
#  FFT based convolutions (fast, real‑valued)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# -------------------------------------------------------------
#  Objective (sum of the vector is assumed to be 1)
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# -------------------------------------------------------------
#  Analytic families (all return a probability vector, sum=1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential (peak at centre)."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_exponential_decreasing(n: int, lam: float = 0.02) -> np.ndarray:
    """Purely decreasing geometric‑like shape."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


def _distribution_linear_decreasing(n: int) -> np.ndarray:
    vals = np.arange(n, 0, -1, dtype=np.float64)
    vals = vals / vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Uniform mass on the first and last `block_frac` fraction of indices."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_random_dirichlet(n: int, concentration: float = 0.2) -> np.ndarray:
    """Sample from Dirichlet(α) with a small α → spiky vectors."""
    a = np.random.dirichlet(np.full(n, concentration))
    a /= a.sum()
    return a


def _distribution_random_sparse(n: int, k_frac: float = 0.005) -> np.ndarray:
    """Exactly k non‑zero entries, equal mass on them."""
    k = max(2, int(k_frac * n))
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = 1.0
    a /= a.sum()
    return a

#  Stage 1 – extended smoothed Adam
def _smooth_adam_descent_extended(a: np.ndarray, deadline: float,
                                  best_seq: np.ndarray,
                                  best_score: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max smoothed max‑convolution."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # more aggressive schedule (double steps per α compared to the baseline)
    schedule = [
        (1.0, 4000),
        (5.0, 4000),
        (20.0, 4000),
        (100.0, 4000),
        (500.0, 4000),
        (2000.0, 4000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1

            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max (log‑sum‑exp) weights
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth max (length n)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact check
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score

def _exact_subgradient_descent_ext(a: np.ndarray, deadline: float,
                                  best_seq: np.ndarray,
                                  best_score: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient method on the true max‑convolution objective."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0
    # allow many iterations but break early on time
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {best_score:.6f}")

        # simple stopping rule when improvements become marginal
        if it % 5000 == 0 and time.time() > deadline - 30:
            break

    return best_seq, best_score

def _greedy_swap_multi(a: np.ndarray, deadline: float,

    """Move mass from high‑gradient to low‑gradient indices, testing many pairs."""
    n = a.shape[0]
    K = 25        # how many candidates on each side
    delta_max = 0.005
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # source candidates (must have some mass)
        src_mask = a > 1e-12
        src_idx = np.where(src_mask)[0]
        if src_idx.size == 0:
            break
        src_order = src_idx[np.argsort(-grad[src_idx])]
        src_top = src_order[:K]

        dest_order = np.argsort(grad)[:K]

        improved = False
        best_local_score = best_score
        best_move = None

        for i in src_top:
            for j in dest_order:
                if i == j:
                    continue
                delta = min(a[i] * 0.5, delta_max)
                if delta <= 1e-12:
                    continue
                a_test = a.copy()
                a_test[i] -= delta
                a_test[j] += delta
                sc = _objective(a_test)
                if sc < best_local_score - 1e-13:
                    best_local_score = sc
                    best_move = (i, j, delta)

        if best_move is not None:
            i, j, delta = best_move
            a[i] -= delta
            a[j] += delta
            best_score = best_local_score
            best_seq = a.copy()
            improved = True
            print(f"[{time.time():.1f}] greedy multi improved best {best_score:.6f}")

        if not improved:
            break

    return best_seq, best_score

#  Stage 4 – simulated annealing with slower cooling and more moves
def _simulated_annealing_ext(a: np.ndarray, deadline: float,
                             best_seq: np.ndarray,
                             best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis random walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    # initial temperature scaled to the typical improvement magnitude
    T = 0.02
    iter_cnt = 0
    report_interval = 2500

    while time.time() < deadline and T > 1e-5:
        iter_cnt += 1

        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        # choose a destination different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction (up to 10 % of the source mass)
        delta = np.random.rand() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter_cnt} new best {best_score:.6f}")

        # cooling schedule
        if iter_cnt % report_interval == 0:
            T *= 0.85
            # occasional temperature reset if we are stuck
            if iter_cnt % (report_interval * 4) == 0:
                T = max(T, 0.0005)

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector found (plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0     # 1 s safety margin

    # ------------------------------------------------------------------
    # 1.  choose a series of lengths; larger lengths give more freedom.
    lengths = [1200, 2000, 3000, 5000, 8000, 12000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2.  Scan analytic families and random draws for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates_extended(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3.  Use any previously computed global solution if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4.  Fallback (should never happen).
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Main optimisation pipeline (all respect the global deadline).
    a = overall_best_seq.copy()

    # 5a – extended smoothed Adam descent
    a, overall_best_score = _smooth_adam_descent_extended(
        a, deadline, best_seq=a, best_score=overall_best_score
    )

    # 5b – exact sub‑gradient descent
    a, overall_best_score = _exact_subgradient_descent_ext(
        a, deadline, best_seq=a, best_score=overall_best_score
    )

    # 5c – greedy multi‑pair active‑set swaps
    a, overall_best_score = _greedy_swap_multi(
        a, deadline, best_seq=a, best_score=overall_best_score
    )

    # 5d – longer simulated annealing fine‑tuning
    a, overall_best_score = _simulated_annealing_ext(
        a, deadline, best_seq=a, best_score=overall_best_score
    )

    print(f"Final best score: {overall_best_score:.6f} (length={len(a)})")
    return a.tolist()
