Propose a low‑score step function.

The search consists of three stages

1️⃣  **Coarse random sampling**  (≈200 s)
    – a few classical families (uniform, Dirichlet, Beta, a spaced “comb”)
    – each candidate is normalised to Σ a = 1 and evaluated

    – a smoothed version of the supremum loss is used:
          Lα(a)= (1/α)·log Σ exp(α·(a*a)ₖ)   (α→∞ ⇒ maxₖ (a*a)ₖ).

      simplex.
    – α is increased exponentially (1 → 2 → 4 …) so the optimisation

      best refined sequence is retained.

3️⃣  **Final evaluation** – the surviving sequence is handed to
    ``evaluate_sequence`` (which rescales automatically).  The printed
    score equals the true objective.

The whole routine is bounded by the supplied time budget (default
1000 s).  All numbers are non‑negative; the simplex projection guarantees
Σ a > 0, so ``evaluate_sequence`` never returns ``inf``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy score for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_subgradient_step(a: np.ndarray, alpha: float):
    """
    Gradient of the smoothed loss Lα(a) = (1/α)·log Σ exp(α·(a*a)ₖ).

    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)                     # shape 2n‑1
    cur_max = b.max()
    # soft weights (softmax) – numerical stability via shifting by cur_max
    w = np.exp(alpha * (b - cur_max))
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= w_sum
    # gradient = 2·conv(reversed a, w) restricted to original n entries
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)               # length n+ (2n‑1)‑1 = 3n‑2
    grad = grad_full[n - 1 : n - 1 + n]               # the n entries we need
    return grad, cur_max


def _refine_smooth_mirror_descent(a0: np.ndarray,
                                  deadline: float,
                                  rng: np.random.Generator,
                                  eta0: float = 0.5,
                                  alpha0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed supremum loss.
    The smoothing parameter ``alpha`` is doubled every 2 000 iterations.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        it += 1
        # increase smoothing gradually
        if it % 2000 == 0:
            alpha = min(alpha * 2.0, 256.0)

        grad, cur_max = _smooth_subgradient_step(a, alpha)

        # standard exponentiated‑gradient step
        eta = eta0 / math.sqrt(it)
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)          # avoid overflow
        a = a * np.exp(exponent)

        # re‑project onto the simplex
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny jitter to escape plateaus
        if (it % 500 == 0) and (rng.random() < 0.08):
            a += rng.normal(scale=1e-6, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] improve (len={n}) it={it} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)   # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _comb_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    A “comb” – equally spaced spikes – normalised to Σ a = 1.
    The number of spikes is chosen near the theoretical optimum
    sqrt(4·n/target) (≈150 for n≈8000).
    """
    target_k = int(math.sqrt(4 * n / 1.5))
    if target_k < 2:
        target_k = 2
    step = max(1, n // target_k)
    idx = np.arange(0, n, step, dtype=int)
    a = np.zeros(n, dtype=float)
    a[idx] = 1.0
    a /= a.sum()
    return a

    """

    else:
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)
    best_score = _raw_score(best_seq)
    best_len = len(best_seq)
    print(f"[{time.time():.1f}] initial raw score = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 1 – cheap random exploration on a modest length (8000)
    n = 8000
    print(f"[{time.time():.1f}] coarse exploration, length {n}")

    # Uniform baseline
    cand = np.full(n, 1.0 / n, dtype=np.float64)
    sc = _raw_score(cand)
    if sc < best_score:
        best_seq, best_score = cand.copy(), sc
        print(f"   uniform new best score={best_score:.6f}")

    # Dirichlet draws (moderately sparse)
    for _ in range(150):
        if time.time() >= coarse_deadline:
            break
        conc = 10 ** rng.uniform(-2, 0)      # promote sparsity
        cand = rng.gamma(conc, 1.0, size=n)
        cand /= cand.sum()

        if sc < best_score:
            best_seq, best_score = cand.copy(), sc
            print(f"   dirichlet sparse score={best_score:.6f}")

    for _ in range(150):
        if time.time() >= coarse_deadline:
            break
        a = 10 ** rng.uniform(-2, 1)
        b = 10 ** rng.uniform(-2, 1)
        cand = _beta_distribution(n, a, b)
        sc = _raw_score(cand)
        if sc < best_score:
            best_seq, best_score = cand.copy(), sc
            print(f"   randBeta a={a:.3f} b={b:.3f} score={best_score:.6f}")

    # Comb pattern (equally spaced spikes)
    cand = _comb_initial(n, rng)
    sc = _raw_score(cand)
    if sc < best_score:
        best_seq, best_score = cand.copy(), sc
        print(f"   comb pattern score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] best after coarse phase = {best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # Phase 2 – long smoothed mirror‑descent refinement with restarts
    while time.time() < deadline:
        # pick an initial point: either the current best (to polish) or a fresh start
        if rng.random() < 0.7:
            a0 = best_seq.copy()
        else:
            # random Dirichlet with a bit of sparsity
            conc = 10 ** rng.uniform(-2, -0.5)
            a0 = rng.gamma(conc, 1.0, size=n)
            a0 /= a0.sum()
        # run the smoother mirror‑descent for a chunk of time
        sub_deadline = min(deadline, time.time() + 15.0)   # ~15 s per restart
        a_opt, scr = _refine_smooth_mirror_descent(a0, sub_deadline, rng)
        if scr + 1e-12 < best_score:
            best_seq, best_score = a_opt.copy(), scr
            print(f"[{time.time():.1f}] NEW GLOBAL BEST score={best_score:.6f}")

            print("[*] target bound reached – exiting early.")
