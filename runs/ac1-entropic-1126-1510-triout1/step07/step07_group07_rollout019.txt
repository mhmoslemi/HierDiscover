"""
Hybrid parametric + smoothed‑gradient optimiser for the step‑function challenge.

The optimiser works in three stages:

1.  **Parametric sweep** – generate a family of symmetric “edge‑heavy”
    distributions
        a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-p} .
    Together with a mixture with the uniform distribution a simple 3‑parameter
    search (p, δ, λ) is performed.  The best candidate (lowest raw score
    2·n·max(conv)) is kept.

2.  **Accelerated smoothed‑gradient refinement** – start from the best
    candidate and minimise the log‑sum‑exp smooth approximation of the max
    convolution using projected FISTA with a continuation on the smoothing
    parameter μ.  The code is a small modification of the reference
    implementation: more inner iterations are performed and the smoothing
    schedule is a bit more aggressive.

3.  **Short sub‑gradient polishing** – a few seconds of projected
    sub‑gradient descent are used as a final polish.

All vectors are projected onto the probability simplex {x≥0, Σx=1}.  The
resulting list is returned; ``evaluate_sequence`` rescales it automatically.
"""

import time
import math
import numpy as np
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Log‑sum‑exp smooth approximation of max(a*a) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights, Σp = 1

    # gradient: 2 * (a_rev * p)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    f_val = M + mu * np.log(Z)
    return f_val, grad


def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation + projected FISTA for minimising max‑convolution.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]

    # initialise smoothing parameter (half of the present max‑conv)
    mu = _max_self_convolution(a) * 0.5
    mu = max(mu, 1e-9)
    mu_target = 1e-7          # final smoothing ≈ true max
    mu_factor = 0.6           # more aggressive reduction

    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # Lipschitz over‑estimate for the smooth objective
        L = max(1.0, 2.0 / mu)

        y = a.copy()
        t = 1.0

        # allocate inner iterations – up to a few thousand per μ
        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        # empirical cost ≈ 0.001 s per 1000 iterations
        iters = int(min(5000, remaining / 0.0012))
        if iters <= 0:
            break

        for _ in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # simple back‑tracking line search (max two doublings)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                # sufficient decrease condition (quadratic model)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional exact assessment
            if (_ & 31) == 0:          # every 32 iterations
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()

        # reduce smoothing and continue
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        # warm‑restart from current best
        a = best_seq.copy()

    return best_seq, best_val


def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Projected sub‑gradient descent (a few seconds) to polish the final result.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        if it % 500 == 0 and np.random.rand() < 0.08:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _edge_heavy_shape(n: int, p: float, delta: float) -> np.ndarray:
    """
    Symmetric edge‑heavy distribution:
        a_i ∝ ((i+δ)*(n‑1‑i+δ))^{-p}
    Normalised to Σ=1.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + delta) * (n - 1 - i + delta)
    # avoid division by zero
    prod = np.maximum(prod, 1e-12)
    a = prod ** (-p)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0. Problem size – the larger the better, 8000 is a good compromise
    n = 8000

    # ------------------------------------------------------------------
    # 1. Gather a few deterministic / global starters
    candidates = []

    # b) simple baselines

    # ------------------------------------------------------------------
    # 2. Parametric sweep – edge‑heavy families
    best_seq = None

    p_vals = np.linspace(0.5, 2.5, 22)          # shape exponent

    # pre‑compute the uniform proxy once

        if time.time() > deadline - 8.0:

    # ------------------------------------------------------------------

    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
