"""
propose_candidate
=================

A multi‑restart, multi‑length local optimiser for the
``evaluate_sequence`` objective.

*   Randomly samples a length from a modest list (2000–12000).
*   Generates a diverse set of initial vectors (arcsine, beta,
    exponential, uniform, random mixtures, sparse spikes).
*   For each start runs a short but potent optimisation pipeline:
    1.  Smoothed Adam on a log‑sum‑exp approximation of the max‑convolution.
    2.  Exact sub‑gradient descent with momentum.
    3.  Greedy active‑set swaps.
    4.  A longer simulated‑annealing walk.
*   Each phase respects the global deadline; if time runs out it simply
    returns the current best.
*   The best vector seen across **all** restarts is returned as a plain
    Python list of floats.

The implementation is pure NumPy, uses FFTs for fast convolutions,
and projects onto the probability simplex after every update.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
# Analytic families – all return a probability vector (sum=1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape parameter p>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_sparse_random(n: int, density: float = 0.005) -> np.ndarray:
    k = max(2, int(density * n))
    a = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    a[idx] = np.random.rand(k)
    a /= a.sum()
    return a


def _random_mixed_distribution(n: int) -> np.ndarray:
    """Convex mix of three baseline families plus tiny perturbations."""
    w = np.random.dirichlet(np.ones(3))
    a1 = _distribution_arcsine(n)
    a2 = _distribution_uniform(n)
    a3 = _distribution_exponential(n, lam=np.random.uniform(0.001, 0.02))
    a = w[0] * a1 + w[1] * a2 + w[2] * a3
    # a little noise (up to 1 % of each coordinate)
    a += 0.01 * np.random.rand(n) * a
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a

# Optimisation phases (all respect a deadline)
def _adam_phase(a: np.ndarray, deadline: float,
                best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Log‑sum‑exp smoothed Adam descent."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.13
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule: (alpha, steps)
    schedule = [
        (1.0, 1500),
        (5.0, 1500),
        (20.0, 1500),
        (100.0, 1500),
        (500.0, 1500),
        (2000.0, 1500),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth max (size n)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam step
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            if t % 120 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam α={alpha:.0f} step={t} best={best_score:.6f}")

    return best_seq, best_score


def _subgradient_phase(a: np.ndarray, deadline: float,
                       best_seq: np.ndarray, best_score: float,
                       max_iter: int = 8000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent with momentum."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.35
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        if it % 200 == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Subgrad it={it} best={best_score:.6f}")

        # early stop if we already beat the target
        if best_score <= 1.503:
            break

    return best_seq, best_score


def _greedy_swap_phase(a: np.ndarray, deadline: float,
                       best_seq: np.ndarray, best_score: float,
                       max_swaps: int = 5000) -> tuple[np.ndarray, float]:
    """Move mass from high‑gradient to low‑gradient positions."""
    n = a.shape[0]
    swaps = 0
    while time.time() < deadline and swaps < max_swaps:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]
        j = int(np.argmin(grad))
        if i == j:
            break

        delta = min(a[i] * 0.5, 0.006)   # slightly larger move than before
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            swaps += 1
            print(f"[{time.time():.1f}] Greedy swap #{swaps} best={best_score:.6f}")
        else:
            delta *= 0.3
            if delta < 1e-8:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                swaps += 1
                print(f"[{time.time():.1f}] Tiny swap #{swaps} best={best_score:.6f}")

    return best_seq, best_score


def _simulated_annealing_phase(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray, best_score: float,
                               max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Simple Metropolis walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    T = 0.045
    it = 0

    while time.time() < deadline and it < max_iters and T > 1e-4:
        it += 1
        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(cur_a[i], 0.12 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={it} best={best_score:.6f}")

        if it % 2500 == 0:
            T *= 0.86

    return best_seq, best_score


# ----------------------------------------------------------------------
def _run_full_pipeline(a_init: np.ndarray, deadline: float,
                       global_best_seq: np.ndarray, global_best_score: float):
    """Execute the whole optimisation pipeline on a single start vector."""
    a = a_init.copy()
    # Phase 1 – Adam (shortened schedule)
    best_seq, best_score = _adam_phase(a, deadline, global_best_seq, global_best_score)

    # Phase 2 – exact sub‑gradient
    best_seq, best_score = _subgradient_phase(best_seq, deadline,
                                              best_seq, best_score,
                                              max_iter=12000)

    # Phase 3 – greedy swaps
    best_seq, best_score = _greedy_swap_phase(best_seq, deadline,
                                              best_seq, best_score,
                                              max_swaps=8000)

    # Phase 4 – simulated annealing (longer)
    best_seq, best_score = _simulated_annealing_phase(best_seq, deadline,
                                                      best_seq, best_score,
                                                      max_iters=50000)
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises evaluate_sequence."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Gather an initial seed (previous run, if any)
    overall_best_seq = None
    overall_best_score = float("inf")
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.clip(prev, 0, None)
            if prev.sum() == 0:
                prev = _distribution_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            overall_best_seq = prev.copy()
            overall_best_score = _objective(prev)
            print(f"[{time.time():.1f}] Using previous best score={overall_best_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2. Define a modest palette of lengths; larger lengths give a
    #    chance of lower ratio but cost more time.
    lengths = [2000, 2800, 3400, 4000, 5600, 8000, 12000]

    # ------------------------------------------------------------------
    # 3. Time‑limited restart loop
    restart_counter = 0
    while time.time() < deadline - 2.0:  # keep a tiny buffer
        restart_counter += 1
        n = int(np.random.choice(lengths))

        # Pick a random family (uniformly) and possibly add noise
        family = np.random.choice([
            "uniform", "arcsine", "beta", "exp", "two_blocks",
            "sparse", "mixed"
        ])

        if family == "uniform":
            a0 = _distribution_uniform(n)
        elif family == "arcsine":
            a0 = _distribution_arcsine(n)
        elif family == "beta":
            p = np.random.uniform(0.05, 5.0)
            a0 = _distribution_beta_sym(n, p)
        elif family == "exp":
            lam = np.random.uniform(0.001, 0.02)
            a0 = _distribution_exponential(n, lam)
        elif family == "two_blocks":
            a0 = _distribution_two_blocks(n, block_frac=np.random.uniform(0.04, 0.12))
        elif family == "sparse":
            a0 = _distribution_sparse_random(n, density=np.random.uniform(0.002, 0.015))
        else:  # mixed
            a0 = _random_mixed_distribution(n)

        # Inject a tiny random perturbation to avoid exact symmetry lock‑ups
        noise = (np.random.rand(n) - 0.5) * 0.002
        a0 = np.clip(a0 + noise, 0, None)
        if a0.sum() == 0:
            a0 = _distribution_uniform(n)
        else:
            a0 /= a0.sum()

        # Run the optimisation pipeline; each call respects the global deadline
        candidate_seq, candidate_score = _run_full_pipeline(
            a0, deadline, overall_best_seq if overall_best_seq is not None else a0,
            overall_best_score
        )

        if candidate_score < overall_best_score - 1e-13:
            overall_best_score = candidate_score
            overall_best_seq = candidate_seq.copy()
            print(f"[{time.time():.1f}] New global best after restart #{restart_counter}: "
                  f"score={overall_best_score:.6f} (len={len(candidate_seq)})")

        # If we have already crossed the target, we can quit early
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # final sanity – ensure we return a list of python floats
    if overall_best_seq is None:
        # fallback – a safe uniform vector
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] Fallback used, score={overall_best_score:.6f}")

    print(f"Finished. Best score={overall_best_score:.6f}, length={len(overall_best_seq)}")
    return overall_best_seq.tolist()
