"""
Propose a step‑function (non‑negative vector) that minimises
    evaluate_sequence(sequence)
The algorithm consists of three stages:

1.  **Exploratory initialisation** – for several lengths we build a
    library of analytic families (uniform, arcsine, beta, exponential,
    sine‑power, cosine windows, triangular, random sparse).  Their
    parameters (beta‑exponent, sine‑power, exponential rate, …) are
    swept on a coarse grid and the best single distribution is kept.
    Afterwards a *mixture* of the base families is sampled:
    a weight vector w~Dirichlet(m) (m = number of bases) yields a new
    distribution `mix = Σ w_i * base_i`.  Many mixtures are evaluated
    (using a fast FFT for the self‑convolution) and the best mixture
    replaces the current incumbent.

2.  **Continuous refinement** – starting from the best incumbent we run
    three optimisation blocks, all respecting the 1000 s overall budget:

    * **Adam descent** on a smooth approximation of the max‑convolution.
    * **Sub‑gradient descent** on the exact max‑convolution (active‑set).
    * **Hill‑climbing** that transfers a tiny amount of mass from the
      index that contributes most to the current maximal convolution
      value to a random target, updating the full self‑convolution in
      O(n) time.

    The best vector seen during any block is kept.

3.  The final vector (as a plain Python list) is returned.
All numbers are clipped to the interval [0,1000] implicitly by
`evaluate_sequence`, but the algorithm works with normalised vectors
(`sum = 1`) so scaling never changes the objective.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a):
    """
    Return the value of evaluate_sequence for a vector 'a' that already
    satisfies sum(a) = 1 (the function is homogeneous, so this is exact).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())

# Analytic families – all return a non‑negative vector with sum == 1.
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j,
    updating both vector a and its full self‑convolution `conv` in O(n).
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()                     # use old values for the linear part
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta          # quadratic self‑term
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

    lr = 0.12
    m_mt = np.zeros_like(a)
    v_mt = np.zeros_like(a)
    t = 0
    best_seq_adam = a.copy()
    best_score_adam = best_score

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1

            # self‑convolution
            b = _fft_self_convolve(a)

            # soft‑max over the convolution entries
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w /= w.sum()

            # gradient of the smooth max (convolution w.r.t a)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + projection onto simplex
            m_mt = beta1 * m_mt + (1 - beta1) * grad
            v_mt = beta2 * v_mt + (1 - beta2) * (grad * grad)
            m_hat = m_mt / (1 - beta1 ** t)
            v_hat = v_mt / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if t % 300 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score_adam - 1e-12:
                    best_score_adam = cur
                    best_seq_adam = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

    a = best_seq_adam.copy()
    best_score = best_score_adam
    best_seq = a.copy()

    # ---------- Exact sub‑gradient descent ----------
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.20
    sub_it = 0
    while time.time() < deadline:
        sub_it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(sub_it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad iter={sub_it} new best {cur:.6f}")

        if time.time() + 0.5 > deadline:
            break

    # ---------- Hill‑climbing on the convolution (incremental updates) ----------
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0
    stagnant = 0
    max_jitter = 1e-5

    while time.time() < deadline:
        hill_it += 1

        # location of the current maximal convolution value
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)

        # find the pair (i,j) that most contributes to this entry
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue

        # propose a small transfer
        max_delta = min(a[source] * 0.5, 0.03)
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            stagnant = 0
            if hill_it % 200 == 0:
                print(f"[{time.time():.1f}] hill_iter={hill_it} new best {best_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            stagnant += 1

        # occasional jitter to escape plateaux
        if hill_it % 2500 == 0 and stagnant > 500:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            stagnant = 0
            print(f"[{time.time():.1f}] jitter at iter {hill_it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
