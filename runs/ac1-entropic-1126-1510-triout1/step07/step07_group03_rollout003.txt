"""
Improved optimizer for the “step‑function” challenge.

The algorithm:
1. **Broad initial scan** – generate analytic families (uniform,
   arcsine, symmetric beta, two‑sided exponential spikes) for several
   lengths up to 20 000.  The best shape becomes a seed.
2. **Extended smoothed Adam** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  The smoothing parameter
   α is increased from 1 to 2000, and each stage runs for thousands of
   Adam steps (far more than the reference implementation).
3. **Exact sub‑gradient descent** – after the continuation phase switch
   to the true sub‑gradient (active set of maximal convolution entries)
   with a Nesterov‑style momentum and a slowly decaying step size.
4. **Greedy active‑set swaps** – repeatedly move a fraction of mass
   from a high‑gradient index to a low‑gradient index while the
   objective improves.
5. **Simulated‑annealing fine‑tuning** – a cheap Metropolis walk with an
   exponential cooling schedule helps to escape shallow plateaus.

All stages respect a global 1000 s budget and keep the best vector found.
The returned list of floats is a valid step‑function (non‑negative,
sum > 0) that aims to minimise `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Objective for a probability vector (sum = 1): 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑space to avoid under/overflow for extreme p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_exponential_edges(n: int, lam: float) -> np.ndarray:
    """Mass concentrated at the two ends, decaying exponentially inward."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.exp(-lam * np.minimum(idx, n - 1 - idx))
    dist /= dist.sum()
    return dist


def _mix_with_uniform(a: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination a*(1‑α) + uniform*α, α∈[0,1]."""
    n = a.shape[0]
    u = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = (1.0 - alpha) * a + alpha * u
    mixed /= mixed.sum()
    return mixed

# Stage 1 – extended smoothed Adam descent -----------------------------
def _extended_smooth_adam(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp smoothing of max‑convolution."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.15                     # a slightly larger learning rate than the reference
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # More aggressive schedule – thousands of steps per α
    schedule = [
        (1.0,   4000),
        (5.0,   4000),
        (20.0,  4000),
        (100.0, 4000),
        (500.0, 4000),
        (2000.0,4000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smoothed objective
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # maintain feasibility
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 500 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score

# Stage 2 – exact sub‑gradient descent ---------------------------------
def _exact_subgradient(a: np.ndarray, deadline: float,
                       best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with Nesterov‑style momentum."""
    n = a.shape[0]
    vel = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.45   # a slightly larger base step size
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        # sub‑gradient of the true max‑convolution
        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # Nesterov‑style momentum
        vel = beta * vel + (1.0 - beta) * grad
        eta = eta0 / math.sqrt(it)
        a = a - eta * vel
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        # early termination if the target is reached
        if best_score <= 1.503:
            break

        # cheap check to leave the loop before the deadline
        if time.time() + 0.3 >= deadline:
            break

    return best_seq, best_score

def _greedy_swap(a: np.ndarray, deadline: float,
                 best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from a high‑gradient index toward a low‑gradient index."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        src_cands = np.where(a > 1e-12)[0]
        if src_cands.size == 0:
            break
        i = src_cands[np.argmax(grad[src_cands])]
        j = int(np.argmin(grad))

        if i == j:
            break

        delta = min(a[i] * 0.4, 0.008)   # larger tentative move
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")
        else:
            # try a finer move
            delta *= 0.25
            if delta < 1e-7:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] tiny swap improved best {best_score:.6f}")

    return best_seq, best_score

# Stage 4 – simulated annealing -----------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray, best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis random walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    T = 0.05               # start temperature (in objective units)
    iter_cnt = 0

    while time.time() < deadline and T > 5e-5:
        iter_cnt += 1

        # pick a source with non‑negligible mass
        src = np.random.choice(np.where(cur_a > 1e-12)[0])
        # pick a distinct destination
        dst = np.random.randint(n - 1)
        if dst >= src:
            dst += 1

        # transfer a random fraction (up to 10 % of source mass)
        delta = np.random.rand() * min(cur_a[src], 0.1 * cur_a[src])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter_cnt} new best {best_score:.6f}")

        # exponential cooling every few thousand iterations
        if iter_cnt % 3000 == 0:
            T *= 0.82

    return best_seq, best_score

                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init beta p={p:.3f} n={n} score={sc:.6f}")

        # two‑sided exponential spikes at the ends
        for lam in (0.001, 0.003, 0.005, 0.01, 0.02, 0.05):
            cand = _distribution_exponential_edges(n, lam)
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init exp‑edge λ={lam:.3f} n={n} score={sc:.6f}")

        # small mixes with uniform (helps breaking symmetry)
        for alpha in (0.05, 0.10, 0.15):
            mixed = _mix_with_uniform(cand, alpha)   # use last cand (edge) as base
            sc = _objective(mixed)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = mixed.copy()
                print(f"[{time.time():.1f}] init mixed α={alpha:.2f} n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. If a previous solution exists, consider it.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.abs(prev)                     # ensure non‑negative
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never be needed.
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform n={n} score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline (time aware).
    a = overall_best_seq.copy()

    # 5a – extended smoothed Adam descent
    a, overall_best_score = _extended_smooth_adam(a, deadline,
                                                  best_seq=a,
                                                  best_score=overall_best_score)

    # 5b – exact sub‑gradient refinement
    a, overall_best_score = _exact_subgradient(a, deadline,
                                                best_seq=a,
                                                best_score=overall_best_score)

    # 5c – greedy active‑set swaps
    a, overall_best_score = _greedy_swap(a, deadline,
                                          best_seq=a,
                                          best_score=overall_best_score)

    # 5d – simulated‑annealing fine‑tuning
    a, overall_best_score = _simulated_annealing(a, deadline,
                                                  best_seq=a,
                                                  best_score=overall_best_score)

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(a)})")
    # Convert to plain Python list (float) – the evaluator will clamp
    # values to [0,1000] and enforce non‑negativity automatically.
    return a.tolist()
