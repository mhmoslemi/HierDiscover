"""
Step‑function optimiser for the “max self‑convolution” objective.

Approach
--------
1. Work first on a modest size (n_small = 2000) – cheap to explore.
2. Build a diverse pool of initial probability vectors:
   * uniform, arcsine (Beta(0.5,0.5)), several U‑shaped Betas (α<1),
   * a handcrafted edge‑heavy distribution, random Beta draws and
     Dirichlet draws.

3. Pick the best seed (according to the cheap proxy 2·n·max(a*a)).
4. Refine it with a continuation scheme based on a smooth
   approximation of the max (log‑sum‑exp).  The smooth problem is minimised
   by Adam (projected onto the simplex) while the smoothing parameter μ
   is geometrically reduced (μ←μ·μ_factor) until μ≈1e‑9.
5. Upsample the refined small‑n solution to the final length

6. Run the same Adam‑based continuation on the full‑size vector.
7. Finish with a short projected sub‑gradient descent (the method used

8. If time permits, perform a few random perturbations and a quick
   Adam run on each to escape possible local minima.
9. Return the final probability vector; `evaluate_sequence` rescales it

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow in exp, cap log‑pdf
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a∗a."""
    return _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy: 2·n·max(a∗a) – the numerator of the true objective."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)                     # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)                         # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                                   # soft‑max weights
    # gradient = 2 * (a_rev * p); keep the central n entries
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _adam_refine(a0: np.ndarray, time_budget: float,
                 mu_start: float = None,
                 mu_target: float = 1e-9,
                 mu_factor: float = 0.6,
                 max_iters_per_mu: int = 2000):
    """
    Continuation with decreasing μ, minimising the smooth objective
    by Adam (projected onto the simplex).
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    # initial smoothing – half of the current max‑conv
    if mu_start is None:
        mu = max(_max_self_convolution(a) * 0.5, 1e-7)
    else:
        mu = mu_start

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while True:
        # abort if we ran out of time
        if time.time() - start >= time_budget:
            break

        # a modest, μ‑independent learning rate
        lr = 0.08

        # inner optimisation for the current μ
        for it in range(max_iters_per_mu):
            if time.time() - start >= time_budget:
                break
            f_val, grad = _smooth_obj_and_grad(a, mu)

            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = _project_onto_simplex(a_new)

            # occasional evaluation of the true (non‑smoothed) objective
            if (it & 31) == 0:                     # every 32‑th iteration
                cur_raw = _raw_score(a_new)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a_new.copy()

            a = a_new

        # reduce μ – stop when the target has been reached
        if mu <= mu_target:
            break
        mu = max(mu * mu_factor, mu_target)

        # restart Adam momentum for the new smoothing level
        m.fill(0.0)
        v.fill(0.0)
        t = 0
        # warm‑start from the best point found so far
        a = best_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float):
    """
    Short projected sub‑gradient descent (as in the reference code)
    to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        # exact sub‑gradient of max_k (a*a)_k
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # step‑size schedule similar to the original code
        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional tiny jitter to escape flat regions
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _edge_heavy_distribution(n: int) -> np.ndarray:
    """Simple symmetric edge‑biased distribution: 1/(i+1) + 1/(n-i)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point – called by the competition harness.
    Returns a list of non‑negative step heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 1. Problem sizes
    n_big = 8000   # final length
    n_small = 2000  # cheap exploratory length

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of seed vectors (all normalised to Σ=1)
    candidates = []

    # deterministic baselines
    candidates.append(np.full(n_small, 1.0 / n_small))          # uniform
    candidates.append(_beta_distribution(n_small, 0.5, 0.5))   # arcsine

    # several U‑shaped Betas (α=β<1) – heavily weighted at the edges
    for a in (0.2, 0.3, 0.4, 0.1):
        candidates.append(_beta_distribution(n_small, a, a))

    # handcrafted edge‑heavy shape
    candidates.append(_edge_heavy_distribution(n_small))

    # random Beta draws
    for _ in range(6):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        candidates.append(_beta_distribution(n_small, alpha, beta))

    # random Dirichlet draws
    for _ in range(6):
        conc = 10 ** rng.uniform(-2, 1)
        vec = rng.gamma(conc, 1.0, size=n_small)
        vec /= vec.sum()
        candidates.append(vec)

    # warm‑start from the globally supplied construction, if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base = base / base.sum()
        base_small = _interpolate_to_length(base, n_small)
        candidates.append(base_small)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Pick the best raw‑score seed (cheap proxy)
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    print(f"Initial best raw (n={n_small}): {best_raw:.6f}")

    # ------------------------------------------------------------------
    # 4. Small‑scale Adam‑based continuation
    remaining = deadline - time.time()
    if remaining > 40.0:
        best_seq, best_raw = _adam_refine(best_seq,
                                          time_budget=remaining * 0.55)
        print(f"After small‑scale Adam refinement raw: {best_raw:.6f}")

    # ------------------------------------------------------------------
    # 5. Upscale to the final length
    a_big = _interpolate_to_length(best_seq, n_big)
    a_big = np.maximum(a_big, 0.0)
    a_big /= a_big.sum()
    best_raw_big = _raw_score(a_big)
    print(f"Upscaled to n={n_big}, raw: {best_raw_big:.6f}")

    # ------------------------------------------------------------------
    # 6. Large‑scale Adam continuation
    remaining = deadline - time.time()
    if remaining > 30.0:
        a_big, best_raw_big = _adam_refine(a_big,
                                           time_budget=remaining * 0.55)
        print(f"After large‑scale Adam refinement raw: {best_raw_big:.6f}")

    # ------------------------------------------------------------------
    # 7. Final sub‑gradient polishing
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_big, best_raw_big = _subgradient_polish(a_big,
                                                  time_budget=remaining)
        print(f"After sub‑gradient polish raw: {best_raw_big:.6f}")

    # ------------------------------------------------------------------
    # 8. Optional random perturbation restarts (if we still have time)
    remaining = deadline - time.time()
    if remaining > 8.0:
        # split the remaining budget over a few short restarts
        n_restart = 3
        per_restart = remaining / n_restart
        for r in range(n_restart):
            pert = a_big + rng.normal(scale=5e-4, size=n_big)
            pert = _project_onto_simplex(pert)
            pert_seq, pert_raw = _adam_refine(pert,
                                              time_budget=per_restart)
            if pert_raw < best_raw_big:
                best_raw_big = pert_raw
                a_big = pert_seq
                print(f"Restart {r+1} improved raw to {best_raw_big:.6f}")

    # ------------------------------------------------------------------
    # 9. Return the final sequence (evaluate_sequence rescales automatically)
    final_score = evaluate_sequence(a_big.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(a_big)})")
    return a_big.tolist()
