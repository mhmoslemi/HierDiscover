"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises the supplied ``evaluate_sequence`` objective.

The search proceeds in three stages:

1. **Broad analytic portfolio** – several known families (uniform,
   arcsine, triangular, exponential, symmetric/asymmetric beta,
   sine‑power and cosine windows) together with random sparse seeds are
   instantiated for a range of lengths.  Each candidate is normalised to
   sum = 1 and scored with a fast FFT based computation of
   ``2·n·max(a*a)`` (the exact objective when the sum is 1).  The best
   candidate becomes the start point for the optimisation.

2. **Exponentiated‑gradient (mirror‑descent) on a smooth max** – we
   replace the non‑smooth ``max`` by a soft‑max with a temperature
   parameter that is increased gradually.  The gradient of the smooth
   max is obtained by convolving the current vector with a soft weight
   vector.  An update of the form ``a ← a·exp(‑η·grad)`` followed by a
   simplex renormalisation (KL‑projection) preserves positivity and
   avoids the costly Euclidean projection used in the reference code.
   This stage runs as long as the global time budget permits.

3. **Targeted stochastic hill‑climber** – using the current vector and
   its full linear self‑convolution we repeatedly transfer a tiny amount
   of mass from the index that contributes most to the current maximal
   convolution entry to a random destination.  The convolution array is
   updated incrementally in O(n) time, and a move is accepted only if it
   strictly lowers the maximal convolution value.  After a long stretch
   without improvement a small random jitter is added to escape plateaux.

The algorithm respects a 1000 s time limit, prints intermediate scores
and finally returns the best sequence found (as a plain ``list[float]``).
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Simplex helpers
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based convolutions
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Objective for vectors that already sum to 1
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with parameters (p,p)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_general(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric beta (a,b) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    if w.sum() == 0 or np.any(np.isnan(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
# Incremental mass transfer (O(n) update of the self‑convolution)
def _apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Transfer a mass ``delta`` from index i to index j, updating both ``a``
    and its full self‑convolution array ``conv`` in O(n) time.
    """
    if delta <= 0 or i == j:
        return
    n = a.shape[0]
    # keep a copy of the *old* a for the linear terms
    a_old = a.copy()
    # linear part: (a_i - delta) interacts with all other entries,
    # which translates into a shift of the convolution by +/-delta*a_old
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms (i,i) and (j,j)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    # finally move the mass in ``a``
    a[i] -= delta
    a[j] += delta

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for the best non‑negative sequence under the 1000 s budget.
    Returns the sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # tiny safety margin
    eps = 1e-12

    # ------------------------------------------------------------------
    # 1) Broad sweep of analytic families + random sparse seeds
    lengths = [800, 1200, 2000, 4000, 8000, 12000, 16000, 20000]
    best_seq = None
    best_score = float("inf")
    best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break

        candidates = []

        # deterministic families
        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.003))
        candidates.append(_distribution_exponential(n, lam=0.01))

        # symmetric beta shapes
        for p in (0.2, 0.5, 1.0, 2.0, 3.0, 5.0):
            candidates.append(_distribution_beta_sym(n, p))

        # asymmetric beta – a few random draws
        for _ in range(4):
            a = 10 ** np.random.uniform(-2, 1)   # between 0.01 and 10
            b = 10 ** np.random.uniform(-2, 1)
            candidates.append(_distribution_beta_general(n, a, b))

        # sine‑power windows
        for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
            candidates.append(_distribution_sine_power(n, pw))

        # cosine (Hann‑like) windows
        for coeff in (0.5, 0.46, 0.42, 0.38):
            candidates.append(_distribution_cosine_window(n, coeff))

        # random sparse vectors (2–5 % non‑zeros)
        for _ in range(3):
            sparse = np.zeros(n, dtype=np.float64)
            k = max(2, n // random.randint(50, 200))
            idx = np.random.choice(n, k, replace=False)
            sparse[idx] = np.random.rand(k)
            sparse /= sparse.sum()
            candidates.append(sparse)

        # random convex mixtures of two previously generated candidates
        for _ in range(4):
            a, b = random.sample(candidates, 2)
            w = random.random()
            mix = w * a + (1 - w) * b
            mix /= mix.sum()
            candidates.append(mix)

        # evaluate all candidates (cheap – single FFT per candidate)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # ------------------------------------------------------------------
    # 2) If a previous solution is available, try it as an extra seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Fallback: uniform (should never happen)
    if best_seq is None:
        best_seq = _distribution_uniform(2000)
        best_score = _max_convolution_score(best_seq)
        best_n = 2000
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Mirror‑descent on a smooth approximation of the max‑convolution.
    a = best_seq.copy()
    n = a.shape[0]
    step = 0
    lr = 0.12                     # fixed learning rate for the exp‑gradient
    grad_clip = 500.0            # avoid overflow in the exponentials
    # schedule for the soft‑max temperature (alpha). We increase it
    # slowly so that the landscape becomes progressively sharper.
    while time.time() < deadline * 0.65:
        # full self‑convolution
        b = _fft_self_convolve(a)
        maxb = b.max()
        # temperature grows roughly linearly with the number of steps
        alpha = min(8000.0, 1.0 + step * 0.02)
        w = np.exp(alpha * (b - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth max (same as sub‑gradient for the true max)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # exponentiated‑gradient (KL‑projection onto the simplex)
        grad = np.clip(grad, -grad_clip, grad_clip)
        a = a * np.exp(-lr * grad)
        # renormalise to the simplex (KL‑projection)
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            a = _distribution_uniform(n)
        else:
            a = a / a_sum
        # enforce a tiny positivity floor
        a = np.maximum(a, eps)
        a = a / a.sum()

        if step % 400 == 0:
            cur = _max_convolution_score(a)
            if cur < best_score - 1e-12:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] exp‑grad step={step} new best {cur:.6f}")
        step += 1
        if time.time() >= deadline * 0.65:
            break

    # ------------------------------------------------------------------
    # 4) Stochastic hill‑climber that works directly on the convolution.
    a = best_seq.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_score = 2.0 * n * M
    best_seq = a.copy()
    it = 0
    it_no_improve = 0
    max_jitter = 1e-5

    while time.time() < deadline:
        it += 1
        # index of current maximal convolution entry
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # vectorised search of the pair (i,j) giving the largest product a_i a_j
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        jdx = k - idx
        prod = a[idx] * a[jdx]
        best_offset = int(prod.argmax())
        best_i = int(idx[best_offset])
        best_j = int(jdx[best_offset])

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        # Propose a small transfer. The cap avoids destabilising the
        # distribution while still allowing a meaningful move.
        max_delta = min(a[source] * 0.5, 0.025)
        if max_delta <= 0.0:
            continue
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        # Save snapshots (cheap for moderate dimensions)
        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-15:                   # genuine improvement
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            it_no_improve = 0
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {best_score:.6f}")
        else:
            # revert the move
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        # occasional jitter to escape flat plateaux
        if it % 2000 == 0 and it_no_improve > 500:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
