def _smooth_descent(a, deadline, rng, eta0=0.3, tau0=None, tau_decay=0.995, noise_prob=0.005, noise_scale=1e-7):

   n = len(a)

   best_a = a.copy(); best_val = _raw_score(a)

   it = 0

   if tau0 is None: initial tau = max(_self_convolution_full(a).max(), 1e-12) maybe.

   else tau = tau0

   while time.time() < deadline:

       it += 1

       # compute convolution and max

       c = _self_convolution_full(a)

       max_c = c.max()

       # compute weight vector w using softmax

       # stability: compute exponents exp((c - max_c)/tau)

       if tau <= 0:
           w = np.zeros_like(c)
           w[np.argmax(c)] = 1.0
       else:
           exp_vals = np.exp((c - max_c) / tau)
           sum_exp = exp_vals.sum()
           if sum_exp == 0 or np.isinf(sum_exp) or np.isnan(sum_exp):
               # fallback to active set uniform
               active = np.where(c >= max_c - 1e-12)[0]
               w = np.zeros_like(c)
               w[active] = 1.0/len(active)
           else:
               w = exp_vals / sum_exp

       # gradient
       # compute conv_rev with w and take center slice
       a_rev = a[::-1]
       conv_res = _fft_convolve(a_rev, w)
       grad = 2.0 * conv_res[n-1:n-1+n]

       # step size
       eta = eta0 / math.sqrt(it)

       # multiplicative update
       exponent = -eta * grad
       exponent = np.clip(exponent, -50, 50)
       a = a * np.exp(exponent)

       # ensure nonneg and normalise
       s = a.sum()
       if s == 0 or np.isnan(s):
           a[:] = 1.0 / n
       else:
           a /= s

       # small random jitter occasionally
       if it % 400 == 0 and rng.random() < noise_prob:
           a += rng.normal(scale=noise_scale, size=n)
           a = _project_onto_simplex(a)

       # evaluate score
       cur_val = 2.0 * n * max_c   # same as _raw_score

       if cur_val + 1e-12 < best_val:   # improvement
           best_val = cur_val
           best_a = a.copy()
           print(f"[{time.time():.1f}] smooth descent n={n} it={it} score={best_val:.6f}")

       # decay tau
       tau *= tau_decay

   return best_a, best_val

"""
Propose a step‑function with a very low evaluate_sequence score.

Algorithm
---------
1. **Warm‑start** – load the previous best sequence (if any) and normalise it.
2. **Coarse random search** on moderate lengths (2k‑8k).  Families:
   uniform, arcsine, symmetric Beta(α,α) on a dense grid, random
   Beta draws, Dirichlet draws and mixtures of two Betas.  Every candidate
   is projected onto the simplex (∑a=1) and scored by the cheap proxy
   `2·n·max(a*a)`.
3. **Smooth mirror‑descent** (exponentiated‑gradient with Adam) on the
   current best length.
   * The objective `max_k (a*a)_k` is smoothed with a temperature τ using
     a soft‑max weighting `w ∝ exp(c/τ)`.  The sub‑gradient of the
     smoothed objective is `∇ = 2·conv_rev(a, w)`.
   * Adam‑style moments are kept for each coordinate and a multiplicative
     update `a ← a·exp(-η·step)` preserves non‑negativity.
   * τ is multiplied by a factor <1 each iteration (gradual annealing).
4. **Upsample** the best vector to larger lengths (12k‑20k) and repeat the
   smooth mirror‑descent.  The up‑sample is performed by linear interpolation
   and re‑normalisation.
5. **Final evaluation** – the best vector is handed to the provided
   `evaluate_sequence` function (which rescales automatically).

The whole routine respects the given time budget (≈ 1000 s) and prints
progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable version
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # guard against overflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Proxy score for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    max_b = _self_convolution_full(a).max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
def _smooth_adam_descent(a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         eta0: float = 0.2,
                         tau0: float = None,
                         tau_decay: float = 0.996,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution objective.
    Returns the best vector (∑=1) found before ``deadline`` and its raw score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)

    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t = 0                         # iteration counter

    # initial temperature
    if tau0 is None:
        tau = _self_convolution_full(a).max()
        tau = max(tau, 1e-12)
    else:
        tau = tau0

    while time.time() < deadline:
        t += 1
        # self‑convolution and its max
        c = _self_convolution_full(a)
        max_c = c.max()

        # soft‑max weighting (smooth active set)
        if tau < 1e-14:                         # essentially max‑set
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            # shift by max for numerical stability
            exp_arg = (c - max_c) / tau
            exp_vals = np.exp(np.clip(exp_arg, -700, 700))
            s = exp_vals.sum()
            if s == 0.0 or not np.isfinite(s):
                # fallback to uniform active set
                active = np.where(c >= max_c - 1e-12)[0]
                w = np.zeros_like(c)
                w[active] = 1.0 / active.size
            else:
                w = exp_vals / s

        # gradient of the smoothed objective
        a_rev = a[::-1]
        conv_rev = _fft_convolve(a_rev, w)                 # length 3n‑2
        grad = 2.0 * conv_rev[n - 1: n - 1 + n]            # slice of length n

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size (inverse‑sqrt schedule)
        eta = eta0 / math.sqrt(t)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # multiplicative update, keep numbers in a reasonable range
        exponent = -step
        exponent = np.clip(exponent, -50.0, 50.0)
        a = a * np.exp(exponent)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny jitter to avoid stagnation
        if t % 400 == 0 and rng.random() < 0.04:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = 2.0 * n * max_c
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] adam‑descent n={n} it={t} score={best_val:.6f}")

        # anneal temperature
        tau *= tau_decay

    return best_a, best_val
