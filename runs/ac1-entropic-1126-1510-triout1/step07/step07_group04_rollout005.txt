#!/usr/bin/env python
"""
Propose a low‑score step function.

Algorithm
---------

1. **Coarse random search**  
   Explore a few moderate lengths (2k‑8k) using simple families:
   uniform, arcsine, symmetric Beta(α,α) on a grid, random Beta draws,
   Dirichlet draws, and mixtures of two/three Beta distributions.
   All candidates are projected onto the simplex (∑a=1) and evaluated
   with an exact proxy `2·n·max(a*a)` (identical to the true score).

2. **Smooth‑max mirror descent**  
   Starting from the best candidate, minimise the smooth approximation of
   the ∞‑norm:
        fβ(a) = (1/β)·log∑_k exp(β·(a*a)_k) .
   The gradient of `fβ` is obtained analytically via a second FFT‑based
   convolution.  An exponentiated‑gradient (multiplicative) update
   `a ← a·exp(-η·∇fβ)` keeps non‑negativity, after which we renormalise
   onto the simplex.  The “temperature’’ β is slowly increased,
   turning the smooth surrogate into the true max.

3. **Upsampling & refinement**  
   Interpolate the current best sequence to larger lengths
   (12 k – 20 k) and run the same smooth‑max descent again.
   Larger n reduces the theoretical lower bound `2n/(2n‑1) ≈ 1`,
   giving extra head‑room.

The procedure runs until the supplied time budget (≈1000 s) expires,
printing intermediate scores.  The final sequence is returned as a
plain Python list; `evaluate_sequence` will rescale it automatically.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution of length n, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)               # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The value equals the true objective because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _softmax_weights(c: np.ndarray, beta: float) -> np.ndarray:
    """Softmax of β·c, returns a probability vector that sums to 1."""
    z = beta * c
    # subtract max for numerical stability
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    p = exp_z / exp_z.sum()
    return p


# ----------------------------------------------------------------------
def _refine_softmax(a: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator,
                    eta0: float = 0.5,
                    beta0: float = 5.0,
                    beta_max: float = 500.0,
                    beta_mul: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Smooth‑max (softmax) mirror descent on the simplex.
    Returns the best sequence and its raw score before `deadline`.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    beta = beta0
    while time.time() < deadline:
        it += 1

        # 1) self‑convolution
        c_full = _self_convolution_full(a)

        # 2) softmax weights
        w = _softmax_weights(c_full, beta)

        # 3) gradient of the smooth max
        a_rev = a[::-1]
        grad = 2.0 * _fft_convolve(w, a_rev)[n - 1 : n - 1 + n]

        # 4) multiplicative update (exponentiated gradient)
        eta = eta0 / math.sqrt(it)
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)   # avoid overflow
        a = a * np.exp(exponent)

        # 5) re‑project onto the simplex
        s = a.sum()
        if s == 0.0:               # safety fallback
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional jitter to escape plateaus
        if it % 400 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=1e-6, size=n)
            a = _project_onto_simplex(a)

        # track the best true (max‑norm) score
        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] softmax len={n} it={it} β={beta:.2f} score={best_val:.6f}")

        # anneal β towards beta_max
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

    return best_a, best_val

        print(f"[{time.time():.1f}] exploring length {n}")

        # start from the current global best, interpolated to length n
        a = _interpolate_to_length(best_seq, n)
        sc = _raw_score(a)
        if sc < best_score:
            best_seq, best_score, best_len = a, sc, n
            print(f"   interpolated best score={best_score:.6f}")

        # deterministic baselines
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_seq, best_score, best_len = uniform.copy(), sc, n
            print(f"   uniform score={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_seq, best_score, best_len = arcsine.copy(), sc, n
            print(f"   arcsine score={best_score:.6f}")

        # symmetric Beta grid (coarse)
        alphas = np.concatenate((
            np.linspace(0.12, 0.6, 9, endpoint=False),
            np.linspace(0.6, 2.0, 12),
            np.linspace(2.0, 5.0, 7)
        ))
        for a_alpha in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a_alpha, a_alpha)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   symBeta a={a_alpha:.3f} score={best_score:.6f}")

        # random Beta draws
        for _ in range(250):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)
            beta = 10 ** rng.uniform(-2, 1)
            vec = _beta_distribution(n, alpha, beta)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   randBeta a={alpha:.3f} b={beta:.3f} score={best_score:.6f}")

        # Dirichlet draws
        for _ in range(250):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-2, 1)
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   dirichlet conc={conc:.3f} score={best_score:.6f}")

        # Mixture of two Betas
        for _ in range(150):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mix2 λ={lam:.3f} score={best_score:.6f}")

        # Mixture of three Betas
        for _ in range(120):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a2, b2 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            a3, b3 = 10 ** rng.uniform(-2, 1), 10 ** rng.uniform(-2, 1)
            weights = rng.dirichlet([1, 1, 1])
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            p3 = _beta_distribution(n, a3, b3)
            vec = (weights[0] * p1 + weights[1] * p2 + weights[2] * p3)
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mix3 score={best_score:.6f}")

        # Uniform + Beta mixture
        for _ in range(120):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 1)
            beta = 10 ** rng.uniform(-2, 1)
            beta_vec = _beta_distribution(n, alpha, beta)
            lam = rng.random()
            vec = lam * beta_vec + (1.0 - lam) * uniform
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   mixU λ={lam:.3f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] best score after coarse search = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max mirror descent on the current best length
    if time.time() < deadline:
        a = best_seq.copy()
        a, best_score = _refine_softmax(a, deadline, rng,
                                        eta0=0.5,
                                        beta0=5.0,
                                        beta_max=800.0,
                                        beta_mul=1.0015)
        best_seq = a.copy()
        best_len = len(a)
        print(f"[{time.time():.1f}] after refinement: score={best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 3 – up‑sample to larger lengths (reducing discretisation error)
    larger_lengths = [12000, 16000, 20000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)   # safety
        a, scr = _refine_softmax(a, deadline, rng,
                                 eta0=0.45,
                                 beta0=8.0,
                                 beta_max=1200.0,
                                 beta_mul=1.0012)
        if scr + 1e-12 < best_score:
            best_seq, best_score, best_len = a.copy(), scr, n
            print(f"   new best after up‑sample: score={best_score:.6f} (len={best_len})")
        # early exit if we already beat the target bound
        if best_score <= 1.503:
            print("Target bound reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluator (exact arithmetic, clipping, etc.)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
