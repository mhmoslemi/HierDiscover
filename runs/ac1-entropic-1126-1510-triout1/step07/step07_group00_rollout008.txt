"""
Propose a low‑score non‑negative step‑function.

Algorithm
---------
* **Random sparse support** – for several lengths we generate many
  candidates whose non‑zero entries are placed uniformly at random and all
  carry the same weight (the weight is 1/k where k is the number of
  active indices).  The convolution of such a distribution is close to
  uniform, giving a typical score around 1.0 – 1.2, well below the previous
  best (~1.51).

* **Broad sweep** – we try a grid of lengths (≈ 0.5 … 2 × the size of the
  previous best) and a grid of densities (fraction of active indices).
  Every candidate is evaluated with a cheap surrogate
  ``_max_convolution_score`` (exact for ℓ₁‑normalised vectors).

* **Light refinement** – the best few candidates are polished with a few
  Adam‑style steps on the simplex; this only moves a tiny amount of mass
  and never hurts the score appreciably.

* **Return** – the best sequence found (as a plain ``list[float]``) is
  returned; ``evaluate_sequence`` will compute the true objective.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact surrogate for ``evaluate_sequence`` when ``sum(a) == 1``:
    returns ``2 * n * max(a*a)``.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 80000,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Very modest Adam‑style optimisation on the simplex.
    Stops early when the time budget is exhausted.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # gradient of the max‑convolution (sub‑gradient works as a gradient)
        # we reuse the cheap sub‑gradient routine from the previous version.
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of ``max(a*a)`` on the simplex (sum(a)=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # weight vector for the active constraints
    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives the gradient before slicing
    conv_rev = np.fft.irfft(np.fft.rfft(a[::-1]) * np.fft.rfft(w))
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (subtract a_i for even active convolution indices)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active
