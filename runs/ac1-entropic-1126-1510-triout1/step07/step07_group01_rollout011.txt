"""
Propose a step‑function (non‑negative vector on the simplex) that minimises
``evaluate_sequence``.  

The search proceeds in several stages:

1. **Broad sampling** – analytical families (arcsine, beta, triangular,
   exponential, sine, cosine, uniform) together with random Dirichlet draws.
   For every generated distribution we also try a *spreading* permutation
   (large masses placed far apart) because the objective depends on the order.
   All candidates are normalised to sum = 1 and ranked by the cheap proxy
   `2·n·max(a*a)`.

2. **Smooth max descent** – for the best seeds we minimise a smooth
   approximation of the max‑convolution:
   ``f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k])``.
   Its gradient is obtained analytically (FFT‑based convolution) and an
   Adam‑style optimiser is applied.  The smoothing parameter λ is increased
   gradually (continuation) so that the algorithm first sees a gentle landscape
   and later focuses on the true `max`.

3. **Exact sub‑gradient descent** – a final plain sub‑gradient phase refines
   the solution with respect to the true nonsmooth objective.

4. **Fine‑grained hill‑climbing** – tiny random mass transfers (and occasional
   jitter) finish the optimisation.

The whole routine respects the supplied time budget (default 1000 s) and always
returns a list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a * a via real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors via real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap proxy = 2·n·max(a*a) (exact when Σa = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _arcsine_distribution(n: int) -> np.ndarray:
    """Symmetric arcsine law on {0,…,n−1}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Beta(p,p) law on the grid, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric triangle, normalised to sum = 1."""
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """Symmetric exponential centred in the middle, normalised."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a point on the simplex (Γ‑Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


def _spread_out(seq: np.ndarray) -> np.ndarray:
    """
    Place the largest masses as far apart as possible.
    The sorted values are written alternately from the left and the right.
    """
    n = seq.shape[0]
    order = np.argsort(seq)[::-1]          # descending indices
    sorted_vals = seq[order]
    out = np.empty_like(seq)
    left, right = 0, n - 1
    for k, v in enumerate(sorted_vals):
        if (k & 1) == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out

def _smooth_max_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
    init_lambda: float = 1.0,
    lambda_mult: float = 1.01,
    step0: float = 0.6,
):
    """
    Adam‑style optimisation of the smooth approximation
        f_λ(a) = (1/λ)·log Σ_k exp(λ·(a*a)[k])

    ``a`` stays on the simplex (Σ a = 1, a ≥ 0).  λ is increased
    geometrically (continuation) so that the algorithm gradually focuses
    on the true ``max``.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]

    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1, beta2 = 0.9, 0.999
    lam = init_lambda
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)               # length 2n‑1
        mx = conv.max()
        # numerically stable soft‑max weights
        exp_diff = np.exp(lam * (conv - mx))
        w = exp_diff / exp_diff.sum()               # length 2n‑1, Σ w = 1

        # gradient of the smooth objective
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n] - a * w[2 * np.arange(n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best = cur_score, a.copy()
            print(f"[{time.time():.1f}] smooth descent improved to {best_score:.6f}")

        # increase the smoothing parameter every few hundred steps
        if it % 3000 == 0:
            lam = min(lam * lambda_mult, 200.0)

    return best, best_score

def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.6,
):
    """Plain diminishing‑step‑size sub‑gradient descent on the exact max."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    n = a.shape[0]
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        # compute active set and sub‑gradient
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps)

        w = np.zeros_like(conv)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n] - a * w[2 * np.arange(n)]

        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-15:
            best_score, best = cur_score, a.copy()
            print(f"[{time.time():.1f}] sub‑grad improved to {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function minimising ``evaluate_sequence``."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # 1️⃣  Base length (default ≈3500, or reuse a previously discovered one)
    try:
        base_n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # --------------------------------------------------------------
    # 2️⃣  Lengths to explore
    lengths = set()
    for f in (0.9, 1.0, 1.1, 1.2, 1.3):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(12000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # --------------------------------------------------------------
    # 3️⃣  Broad sampling of analytic families + random Dirichlet
    candidates = []   # (array, cheap_score)

    beta_ps = [
        0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25,
        0.35, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0,
    ]

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- analytic families -------------------------------------------------
        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_exponential_distribution, "exponential"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (_uniform_distribution, "uniform"),
        ]
        for gen, name in families:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            # also try the spread‑out permutation
            a_sp = _spread_out(a)
            score_sp = _max_convolution_score(a_sp)
            candidates.append((a_sp, score_sp))
            print(f"[{time.time():.1f}] n={n:4d} {name:10s}  score={score:.6f}")

        # ----- symmetric beta ----------------------------------------------------
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            a_sp = _spread_out(a)
            candidates.append((a_sp, _max_convolution_score(a_sp)))
            print(f"[{time.time():.1f}] n={n:4d} beta(p={p:4.2f}) score={score:.6f}")

        # ----- random Dirichlet draws -------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            a_sp = _spread_out(a)
            candidates.append((a_sp, _max_convolution_score(a_sp)))
            print(f"[{time.time():.1f}] n={n:4d} random Dirichlet score={score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Include the previously best construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 5️⃣  Keep the top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 12
    top_candidates = candidates[:top_k]
    print(f"Selected {len(top_candidates)} seeds for refinement; best cheap score={top_candidates[0][1]:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Heavy optimisation (smooth max + sub‑gradient)
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        # --- smooth max descent -------------------------------------------------
        seq_s, sc_s = _smooth_max_descent(
            seed_seq,
            deadline,
            max_iters=200_000,
            init_lambda=1.0,
            lambda_mult=1.02,
            step0=0.7,
        )
        if sc_s < best_score - 1e-14:
            best_score, best_seq = sc_s, seq_s.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smooth‑descent → {sc_s:.6f}")

        # --- exact sub‑gradient descent -----------------------------------------
        seq_g, sc_g = _subgradient_descent(best_seq, deadline, max_iters=600_000, eta0=0.6)
        if sc_g < best_score - 1e-14:
            best_score, best_seq = sc_g, seq_g.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} sub‑grad → {sc_g:.6f}")

    # --------------------------------------------------------------
    # 7️⃣  Fine‑grained hill‑climbing on the current best
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.03   # up to 3 % of total mass initially

    while time.time() < deadline:
        iteration += 1
        # geometric decay of transfer size
        transfer = max_transfer * (0.5 ** (iteration / 12000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if iteration % 5000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # ``evaluate_sequence`` is scale‑invariant; returning the simplex vector suffices.
    return best_seq.tolist()
