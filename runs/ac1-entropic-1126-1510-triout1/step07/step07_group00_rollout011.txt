"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Initialisation** – start from a well‑behaved analytical shape
   (arcsine law) or from the previous best vector `height_sequence_1`.
   The vector is normalised to sum 1 (the objective is scale‑invariant).

2. **Active‑set linear‑programming refinement** – given the current
   distribution f, compute M = max (f∗f).  Solve the LP

        max  Σ b_j
        s.t. f∗b ≤ M      (2n‑1 linear inequalities)
             b_j ≥ 0

   The constraint matrix is the Toeplitz convolution matrix of f.
   The LP is solved with the HiGHS simplex implementation from ``scipy``.
   The optimiser returns a feasible b; we normalise it to a probability
   vector g = b / Σb and take a convex step
        f ← (1‑t) f + t g  (t≈0.02).
   This step is guaranteed to **strictly** decrease the maximal
   convolution value, reproducing the classic “alternating LP” scheme
   described in the statement.

3. **Iterate** – repeat the LP step while time permits (≈ 10–20
   iterations for n≈3000).  The LP quickly drives the convolution
   towards a flat profile and typically yields a measurable drop in the
   surrogate score `2·n·max(conv)`.

4. **Fine‑tuning** – once the LP stops making progress, run two
   mirror‑descent refinements on the simplex (Adam‑style and
   exponentiated‑gradient) for the remaining budget.  These cheap
   gradient steps clean up the remaining irregularities.

5. **Return** – the best vector obtained (as a plain ``list[float]``) is
   ready for ``evaluate_sequence``.

The whole routine respects the 1000 s wall‑clock limit and never
produces invalid numbers (negative entries are projected away,
values are kept in the [0, 1000] range automatically by the evaluator).
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Surrogate score   2·n·max(a*a)   assuming Σa = 1.
    This equals ``evaluate_sequence`` for unit‑sum vectors.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _conv_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n non‑negative matrix A such that  A·b = f∗b .
    Each diagonal offset i (0 ≤ i < n) contains the constant f[i].
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # data for each diagonal – a constant vector of length n
    data = [np.full(n, val, dtype=np.float64) for val in f]
    offsets = list(range(n))
    A = sp.diags(data, offsets, shape=(rows, n), format='csr')
    return A


# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, t: float, deadline: float):
    """
    Perform one LP‑based refinement step:
        max Σb   s.t. f∗b ≤ M,  b≥0 .
    Returns the updated distribution and the new surrogate score.
    """
    n = f.shape[0]
    conv_ff = _fft_self_convolve(f)
    M = conv_ff.max()

    # Build the Toeplitz constraint matrix
    A = _conv_constraint_matrix(f)
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear program: minimise -Σb  (i.e. maximise Σb)
    c = -np.ones(n, dtype=np.float64)

    # HiGHS dual‑simplex is very fast on this structure
    res = opt.linprog(c,
                      A_ub=A,
                      b_ub=ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True,
                               'dual_feasibility_tolerance': 1e-9,
                               'time_limit': max(0.0, deadline - time.time())})
    if res.status != 0:
        # LP failed (should not happen); abort the step
        return f, _max_convolution_score(f), False

    b0 = res.x
    sum_b0 = b0.sum()
    if sum_b0 <= 0:
        return f, _max_convolution_score(f), False

    g = b0 / sum_b0            # normalise to a probability vector
    f_new = (1.0 - t) * f + t * g
    f_new = _project_onto_simplex(f_new, s=1.0)

    new_score = _max_convolution_score(f_new)
    return f_new, new_score, True


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300000,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999):
    """Adam‑style mirror descent on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # sub‑gradient of the max‑convolution (exact)
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # compute sub‑gradient via the active set
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps)

        w = np.zeros_like(conv)
        w[active] = 1.0
        # gradient of max convex part
        grad = 2.0 * _fft_self_convolve(a * w)[:a.size]   # simplified, cheap
        # diagonal correction for even active shifts
        even_mask = (active % 2 == 0)
        diag_idx = (active[even_mask] // 2)
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)

        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200000,
                          eta0: float = 0.6):
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # sub‑gradient via active set (same as in Adam)
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - eps)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_self_convolve(a * w)[:a.size]
        even_mask = (active % 2 == 0)
        diag_idx = (active[even_mask] // 2)
        if diag_idx.size:
            grad[diag_idx] -= a[diag_idx]

        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _arcsine_distribution(n: int) -> np.ndarray:
    """U‑shaped arcsine law on {0,…,n‑1} (unit‑sum)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

            # either truncate or pad with zeros, then renormalise
            if prev.size > n:
                f = prev[:n]
            else:
                f = np.pad(prev, (0, n - prev.size), constant_values=0.0)
        else:
            f = prev.copy()
        f = np.maximum(f, 0.0)
        if f.sum() == 0:
            f = _arcsine_distribution(n)
        else:
            f = f / f.sum()
        print(f"[{time.time():.1f}] start from previous best, length={n}")
    else:
        f = _arcsine_distribution(n)
        print(f"[{time.time():.1f}] start from arcsine law, length={n}")

    best_f = f.copy()
    best_score = _max_convolution_score(f)
    print(f"[{time.time():.1f}] initial surrogate score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Active‑set LP refinement (≈10‑15 iterations, adaptive step size)
    t = 0.02                       # convex combination parameter
    lp_iter = 0
    while time.time() < deadline and lp_iter < 30:
        lp_iter += 1
        f, new_score, ok = _lp_step(f, t, deadline)
        if not ok:
            print("[LP] step failed – aborting LP loop")
            break
        if new_score < best_score - 1e-12:
            best_score, best_f = new_score, f.copy()
            print(f"[{time.time():.1f}] LP iter {lp_iter:2d} → new best {best_score:.6f}")
        # slowly increase the step size to accelerate later stages
        if lp_iter % 5 == 0:
            t = min(0.12, t * 1.15)

    # ------------------------------------------------------------------
    # 4. Fine‑tuning with Adam and exponentiated‑gradient
    f_adam, sc_adam = _adam_descent(best_f, deadline,
                                    max_iters=400000,
                                    step0=0.20)
    if sc_adam < best_score - 1e-14:
        best_score, best_f = sc_adam, f_adam.copy()
        print(f"[{time.time():.1f}] Adam polishing improved to {best_score:.6f}")

    f_exp, sc_exp = _exp_gradient_descent(best_f, deadline,
                                          max_iters=250000,
                                          eta0=0.8)
    if sc_exp < best_score - 1e-14:
        best_score, best_f = sc_exp, f_exp.copy()
        print(f"[{time.time():.1f}] Exp‑Grad polishing improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final safety projection (clipping just in case)
    best_f = np.maximum(best_f, 0.0)
    best_f = _project_onto_simplex(best_f, s=1.0)

    print(f"[{time.time():.1f}] final surrogate score = {best_score:.6f} (len={len(best_f)})")
    return best_f.tolist()
