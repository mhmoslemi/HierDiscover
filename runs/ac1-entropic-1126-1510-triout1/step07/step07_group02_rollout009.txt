"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.
The search proceeds in three stages:

1.  Analytic families (uniform, arcsine, beta, exponential, sine‑power,
    cosine windows) are evaluated for several lengths.  The best of these
    seeds is kept.

2.  A mirror‑descent optimiser works on the soft‑max approximation of the
    max‑convolution.  The iterates are kept on the probability simplex via
    the exponential‑map (log‑its).  A temperature schedule (α) turns the
    smooth max into the true max, and a decreasing learning‑rate
    (η∝1/√t) provides convergence.

3.  A fast O(n) hill‑climber directly updates the self‑convolution when a
    small amount of mass is moved from a “source’’ index to a “target’’
    index.  This stage refines the answer found by mirror‑descent.

The routine respects the 1000 s time limit and prints intermediate
progress.  The returned list of floats can be fed directly to
``evaluate_sequence``.
"""

import time
import random
import numpy as np

# ------------------------------------------------------------
# Simplex projection (not used by the mirror descent, kept for fallback)
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# FFT based convolutions
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ------------------------------------------------------------
# Objective for a vector already normalised to have sum = 1
def _max_convolution_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a simplex‑normalised vector."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ------------------------------------------------------------
# Incremental update of the full convolution when a small amount of mass
# is transferred from index i to index j.
def _apply_transfer(a: np.ndarray, conv: np.ndarray,
                    i: int, j: int, delta: float) -> None:
    """
    Transfer a mass `delta` from a[i] to a[j] and update the full
    self‑convolution `conv` in O(n) time.
    """
    if delta <= 0.0 or i == j:
        return
    a_old = a.copy()                      # old values needed for the linear part
    # linear contributions
    conv[i:i + a.size] -= 2.0 * delta * a_old
    conv[j:j + a.size] += 2.0 * delta * a_old
    # quadratic self‑terms (the i‑i and j‑j entries)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The algorithm combines analytic seeds,
    a mirror‑descent optimiser on a smoothed max‑convolution,
    and a fast hill‑climber that works directly on the full
    self‑convolution.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0          # spare margin for cleanup
    eps = 1e-12

    # -----------------------------------------------------------------
    # 1.  Analytic initial candidates (various lengths)
    lengths = [2000, 3000, 4000, 5000, 8000]
    best_seq = None
    best_score = float("inf")

    for n in lengths:
        if time.time() >= deadline:
            break

        cand = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.003),
            _distribution_exponential(n, lam=0.01),
        ]

        for p in (0.2, 0.5, 1.0, 2.0, 3.0):
            cand.append(_distribution_beta_sym(n, p))

        for pw in (0.5, 1.0, 2.0, 4.0, 8.0):
            cand.append(_distribution_sine_power(n, pw))

        for c in (0.5, 0.46, 0.42):
            cand.append(_distribution_cosine_window(n, c))

        # a random sparse vector (few non‑zero entries)
        k = max(2, n // 200)
        idx = np.random.choice(n, k, replace=False)
        sparse = np.zeros(n, dtype=np.float64)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        cand.append(sparse)

        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # -----------------------------------------------------------------
    # 2.  Use a previously stored construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev = prev / prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 3.  Fallback (should never happen)
    if best_seq is None:
        n = 3000
        best_seq = _distribution_uniform(n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # -----------------------------------------------------------------
    # 4.  Mirror descent optimisation (soft‑max of the convolution)
    a = best_seq.copy()
    n = a.shape[0]
    y = np.log(a + eps)                 # logits; a = softmax(y)
    t = 0
    best_seq_mirror = a.copy()
    best_score_mirror = best_score
    max_iter = 300_000                  # large cap; loop stops early by deadline

    while time.time() < deadline and t < max_iter:
        t += 1
        # temperature that slowly focuses on the true maximum
        alpha = 1.0 + 1999.0 * t / max_iter          # 1 → 2000 linearly
        # recover the probability vector from logits
        a = np.exp(y)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        maxb = conv.max()

        # soft‑max weights over all convolution entries
        w = np.exp(alpha * (conv - maxb))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # weighted gradient w.r.t. a
        conv_rev = _fft_convolve(a[::-1], w)
        grad_a = 2.0 * conv_rev[n - 1:n - 1 + n]

        # mirror‑gradient (w.r.t. logits)
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)

        # learning‑rate decays as 1/√t
        eta = 0.7 / np.sqrt(t)
        y -= eta * grad_y

        # occasional logging & best‑track
        if t % 2000 == 0:
            cur_score = 2.0 * n * maxb
            if cur_score < best_score_mirror - 1e-12:
                best_score_mirror = cur_score
                best_seq_mirror = a.copy()
                print(f"[{time.time():.1f}] Mirror iter={t} new best {cur_score:.6f}")

    # keep the best result from mirror descent
    if best_score_mirror < best_score:
        best_score = best_score_mirror
        best_seq = best_seq_mirror.copy()
        print(f"[{time.time():.1f}] Best after mirror {best_score:.6f}")

    # -----------------------------------------------------------------
    # 5.  Fast O(n) hill‑climbing refinement
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    it_no_improve = 0
    max_jitter = 1e-5

    while time.time() < deadline:
        it += 1
        # position of the current global maximum of the convolution
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # vectorised search for the pair (i,j) giving the largest product
        i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_range = k - i_range
        products = a[i_range] * a[j_range]
        best_idx = int(products.argmax())
        i_best = int(i_range[best_idx])
        j_best = int(j_range[best_idx])

        # move a small amount from the larger member of the pair to a random target
        source = i_best if a[i_best] >= a[j_best] else j_best
        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta < eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-14:
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            it_no_improve = 0
            if it % 100 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {best_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap
            it_no_improve += 1

        # occasional jitter to escape plateaux
        if it % 5000 == 0 and it_no_improve > 500:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * n * conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
