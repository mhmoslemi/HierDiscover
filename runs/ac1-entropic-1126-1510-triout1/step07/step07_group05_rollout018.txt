"""
Propose a step function with a low evaluate_sequence score.

Approach
--------
* Work on the probability simplex (ai ≥ 0, Σ ai = 1) by using a soft‑max
  (log‑sum‑exp) parametrisation.
* First explore a rich family of hand‑crafted shapes (linear, exponential,
  power‑law, sine‑based, piece‑wise uniform, random masks, Dirichlet draws).
  The best of these provides a strong warm‑start.
* Run a long Adam optimisation on a smooth approximation of the max‑convolution.
  The temperature (τ) is increased to a very large value (τ≈500) so that the
  surrogate becomes almost the true max.
* Finish with a projected sub‑gradient descent on the exact max‑convolution.
  The sub‑gradient step size follows a Polyak‑like rule for fast decay.
* If time allows, repeat the whole pipeline for a larger length (8000) using
  linear interpolation of the current best sequence as a start.
* The whole search is bounded by the supplied time budget (default 1000 s).

The returned list is directly usable by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 15000,
                            tau0: float = 0.5,
                            tau_max: float = 500.0,
                            lr0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max parametrisation, using a
    log‑sum‑exp (smooth‑max) surrogate for the convolution supremum.
    Returns the best sequence (a) encountered and its *raw* score
    2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- soft‑max (Σa = 1) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # ----- convolution & raw score -----
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        # ----- temperature schedule (slow increase) -----
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # ----- smooth‑max gradient -----
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ----- chain rule through the soft‑max (Manifold gradient) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update (Polyak‑style step size scaling) -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            x += np.random.randn(*x.shape) * 0.005

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the exact max‑convolution.
    Uses a Polyak step size ( (f‑f_best)/||g||² ) for faster convergence.
    Returns refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)
    it = start_iter
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        # Polyak step – protect against division by zero
        grad_norm2 = max(np.dot(grad, grad), 1e-12)
        step = max(0.0, (cur_max * 2 * n - best_score) / grad_norm2)
        # fall‑back to diminishing step if the Polyak step becomes too small
        if step < 1e-6:
            step = 0.07 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")
        # occasional tiny jitter
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
    return best_a, best_score

def _generate_initial_logits(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a list of log‑softmax initialisations from many handcrafted families."""
    logits = []

    # uniform
    uniform = np.full(n, 1.0 / n)
    logits.append(np.log(uniform + 1e-12))

    # linear decreasing (from 1 to 0)
    lin = np.linspace(1.0, 0.0, n)
    lin = np.maximum(lin, 0.0)
    lin /= lin.sum()
    logits.append(np.log(lin + 1e-12))

    # exponential decays
    for lam in [5e-4, 1e-3, 2e-3, 5e-3]:
        exp_seq = np.exp(-lam * np.arange(n))
        exp_seq /= exp_seq.sum()
        logits.append(np.log(exp_seq + 1e-12))

    # power‑law decays
    for p in [0.5, 1.0, 1.5, 2.0]:
        pw = (np.arange(1, n + 1, dtype=np.float64) ** -p)
        pw /= pw.sum()
        logits.append(np.log(pw + 1e-12))

    # sine‑based shapes (shifted to be non‑negative)
    for freq in [1, 2, 3]:
        s = np.sin(np.pi * (np.arange(n) + 0.5) * freq / n) + 1.0
        s = np.maximum(s, 0.0)
        s /= s.sum()
        logits.append(np.log(s + 1e-12))

    # piece‑wise uniform (support of varying length)
    for frac in [0.2, 0.4, 0.6, 0.8]:
        L = max(1, int(frac * n))
        piece = np.zeros(n)
        piece[:L] = 1.0 / L
        logits.append(np.log(piece + 1e-12))

    # random masks (few non‑zero entries)
    for frac in [0.05, 0.1, 0.2]:
        k = max(1, int(frac * n))
        mask = np.zeros(n)
        idx = rng.choice(n, k, replace=False)
        mask[idx] = 1.0 / k
        logits.append(np.log(mask + 1e-12))

    # Dirichlet random draws (moderate and spiky concentrations)
    for _ in range(6):
        alpha = rng.random() * 4.0 + 0.1         # α ∈ (0.1,4.1)
        d = rng.gamma(alpha, 1.0, size=n)
        d /= d.sum()
        logits.append(np.log(d + 1e-12))

    return logits

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the challenge harness.
    Returns a list of non‑negative heights that (hopefully) yields a low score.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # base length – 4000 gives a good trade‑off between resolution and speed
    n0 = 4000
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1. generate many handcrafted candidates and keep the best
    init_logits = _generate_initial_logits(n0, rng)

    print("\n=== Evaluating handcrafted candidates (length=%d) ===" % n0)
    for idx, logit in enumerate(init_logits):
        if time.time() >= deadline:
            break
        a = np.exp(logit - np.max(logit))
        a /= a.sum()
        cur_raw = 2.0 * n0 * _fft_self_convolve_max(a)
        if cur_raw < best_score:
            best_score = cur_raw
            best_seq = a.copy()
            print(f"[{time.time():.1f}] init {idx+1}/{len(init_logits)} raw={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam + sub‑gradient refinement on the current best (long run)
    if best_seq is not None and time.time() < deadline:
        print("\n=== Long Adam optimisation (length=%d) ===" % n0)
        logit0 = np.log(best_seq + 1e-12)
        seq, raw = _adam_optimize_smoothed(
            logit0, n0, deadline,
            max_iters=18000,       # more iterations than the previous baseline
            tau0=0.5, tau_max=800.0, lr0=0.07
        )
        seq, raw = _refine_by_subgradient(seq, deadline)
        if raw < best_score:
            best_score = raw
            best_seq = seq.copy()
            print(f">>> Improved after refinement: raw={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Optional second stage on a larger length if time permits
    if time.time() < deadline - 30:                # reserve a few seconds for final eval
        n1 = 8000
        print("\n=== Warm‑starting larger length (%d) ===" % n1)
        seq_big = _interpolate_to_length(best_seq, n1)
        logit_big = np.log(seq_big + 1e-12)

        # a short Adam (fewer iterations) because the dimension doubled
        seq_big, raw_big = _adam_optimize_smoothed(
            logit_big, n1, deadline,
            max_iters=12000,
            tau0=0.5, tau_max=800.0, lr0=0.07
        )
        seq_big, raw_big = _refine_by_subgradient(seq_big, deadline)
        if raw_big < best_score:
            best_score = raw_big
            best_seq = seq_big.copy()
            print(f">>> New global best from length {n1}: raw={best_score:.6f}")

    # ------------------------------------------------------------------
    # final clean‑up: a quick extra sub‑gradient sweep to squeeze out any
    # remaining improvement.
    if best_seq is not None and time.time() < deadline:
        best_seq, final_raw = _refine_by_subgradient(best_seq, deadline)
        best_score = final_raw

    # ------------------------------------------------------------------
    # Return the result as a plain Python list
    final_list = best_seq.tolist()
    # sanity check (the harness will do it again, but we log it here)
    try:
        final_eval = evaluate_sequence(final_list)  # type: ignore[name-defined]
        print(f"\nFinal evaluated score reported by evaluate_sequence: {final_eval:.6f}")
    except Exception as e:
        print("Evaluation failed:", e)
    return final_list
