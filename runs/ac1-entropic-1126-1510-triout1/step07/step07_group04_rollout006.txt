def _refine_exponentiated_gradient(a, deadline, rng, eta0=0.5, beta=0.9):
    n = len(a)
    best_a = a.copy()
    best_score = _raw_score(a)
    it = 0
    m = np.zeros_like(a)
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        m = beta * m + (1-beta) * grad
        eta = eta0 / math.sqrt(it)
        exponent = -eta * m
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)
        # reproject onto simplex
        a = _project_onto_simplex(a)  # ensures sum=1
        cur_score = _raw_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror-momentum len={n} it={it} score={best_score:.6f}")
        # occasional jitter
        if it % 500 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=1e-6, size=n)
            a = _project_onto_simplex(a)
    return best_a, best_score

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    v = v.astype(np.float64, copy=False)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete (mid‑point) Beta(α,β) density on {0,…,n‑1},
    normalised to Σ=1.  Numerically stable for extreme α,β.
    """
    # mid‑point locations on (0,1)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, avoiding overflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # Clip safely before exponentiation
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing distribution onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a * b via FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score when Σa = 1 (exact for the evaluator).
    Returns 2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of f(a) = max_k (a*a)_k.
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    # active set (indices attaining the maximum, up to a tolerance)
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    # gradient = 2 * (a_rev * w) restricted to length n
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w)
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, cur_max

def _refine_exponentiated_gradient(a: np.ndarray,
                                   deadline: float,
                                   rng: np.random.Generator,
                                   eta0: float = 0.5,
                                   beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) with momentum.
    Returns the best vector seen and its proxy score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    # momentum accumulator (same shape as a)
    m = np.zeros_like(a)

    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        # momentum update
        m = beta * m + (1.0 - beta) * grad
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient step (clipped to avoid overflow)
        exponent = -eta * m
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)

        # re‑project onto the simplex Σa = 1
        a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] EG‑momentum len={n} it={it} score={best_val:.6f}")

        # occasional jitter to escape plateaus
        if it % 500 == 0 and rng.random() < 0.07:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

    return best_a, best_val

def _local_hill_climb(a: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iter: int = 20000,
                      init_step: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climbing: move a small fraction of mass
    from one index to another and keep the move if the proxy score
    improves.  The step‑size decays like init_step/√t.
    """
    n = len(a)
    best_a = a.copy()
    best_val = _raw_score(a)
    eps = 1e-12
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # choose a source with non‑zero mass
        i = rng.integers(n)
        if best_a[i] < eps:
            continue
        # choose a distinct target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # adapt step size
        step = init_step / math.sqrt(it)
        delta = best_a[i] * step * rng.random()
        if delta <= eps:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # non‑negativity is guaranteed by construction, sum unchanged

        cur_val = _raw_score(cand)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = cand
            if it % 500 == 0:
                print(f"[{time.time():.1f}] hill‑climb it={it} score={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function that minimises the
    evaluate_sequence() score.  Returns the best sequence as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5    # safety margin for the final evaluation

    # ------------------------------------------------------------------
    # Attempt to warm‑start from a previously found solution.
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if base_seq.ndim == 1 and base_seq.sum() > 0.0:
            base_seq = base_seq / base_seq.sum()
        else:
            base_seq = None
    except Exception:
        base_seq = None

    # Initial best solution: either the warm‑start or a plain uniform.
    if base_seq is not None:
        best_seq = base_seq.copy()
        best_score = _raw_score(best_seq)
    else:
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)
        best_score = _raw_score(best_seq)

    print(f"[{time.time():.1f}] initial best raw score = {best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 1 – coarse random exploration on several lengths
    lengths = [2000, 4000, 8000, 16000, 32000]
    # a few α‑grid values for symmetric Beta
    sym_alpha_grid = np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0])

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # start from an interpolated version of the current best
        a = _interpolate_to_length(best_seq, n)
        sc = _raw_score(a)
        if sc < best_score:
            best_seq, best_score = a, sc
            print(f"   interp‑best score={best_score:.6f}")

        # deterministic baselines
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_seq, best_score = uniform.copy(), sc
            print(f"   uniform score={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_seq, best_score = arcsine.copy(), sc
            print(f"   arcsine score={best_score:.6f}")

        # symmetric Beta grid
        for alpha in sym_alpha_grid:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, alpha, alpha)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   symBeta α={alpha:.3f} score={best_score:.6f}")

        # random symmetric Beta draws (α between 1e‑4 and 10)
        for _ in range(250):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-4, 1)
            vec = _beta_distribution(n, alpha, alpha)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   randSymBeta α={alpha:.4f} score={best_score:.6f}")

        # random asymmetric Beta draws
        for _ in range(250):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-4, 1)
            beta = 10 ** rng.uniform(-4, 1)
            vec = _beta_distribution(n, alpha, beta)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   randBeta α={alpha:.4f} β={beta:.4f} score={best_score:.6f}")

        # Dirichlet draws – extremely concentrated and almost uniform
        for _ in range(250):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-4, 1)
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   dirichlet conc={conc:.4f} score={best_score:.6f}")

        # Mixture of two Betas
        for _ in range(150):
            if time.time() >= deadline:
                break
            a1, b1 = 10 ** rng.uniform(-4, 1), 10 ** rng.uniform(-4, 1)
            a2, b2 = 10 ** rng.uniform(-4, 1), 10 ** rng.uniform(-4, 1)
            lam = rng.random()
            p1 = _beta_distribution(n, a1, b1)
            p2 = _beta_distribution(n, a2, b2)
            vec = lam * p1 + (1.0 - lam) * p2
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   mix2 λ={lam:.3f} score={best_score:.6f}")

        # Mixture of three Betas
        for _ in range(100):
            if time.time() >= deadline:
                break
            alphas = 10 ** rng.uniform(-4, 1, size=3)
            betas = 10 ** rng.uniform(-4, 1, size=3)
            weights = rng.dirichlet([1, 1, 1])
            vec = np.zeros(n)
            for k in range(3):
                vec += weights[k] * _beta_distribution(n, alphas[k], betas[k])
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   mix3 score={best_score:.6f}")

        # Uniform + Beta mixture (often useful)
        for _ in range(100):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-4, 1)
            beta = 10 ** rng.uniform(-4, 1)
            beta_vec = _beta_distribution(n, alpha, beta)
            lam = rng.random()
            vec = lam * beta_vec + (1.0 - lam) * uniform
            vec = _project_onto_simplex(vec)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score = vec.copy(), sc
                print(f"   mixU λ={lam:.3f} score={best_score:.6f}")

    print(f"[{time.time():.1f}] best score after coarse phase = {best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement (momentum)
    if time.time() < deadline:
        a = best_seq.copy()
        a, best_score = _refine_exponentiated_gradient(a, deadline, rng, eta0=0.6, beta=0.9)
        best_seq = a.copy()
        print(f"[{time.time():.1f}] after EG‑momentum: score={best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill climbing
    if time.time() < deadline:
        a, best_score = _local_hill_climb(best_seq, deadline, rng,
                                          max_iter=15000, init_step=0.02)
        best_seq = a.copy()
        print(f"[{time.time():.1f}] after hill‑climb: score={best_score:.6f} (len={len(best_seq)})")

    # ------------------------------------------------------------------
    # Phase 4 – up‑sample to larger lengths (reduces discretisation error)
    up_lengths = [48000, 64000, 80000]
    for n in up_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)
        a, scr = _refine_exponentiated_gradient(a, deadline, rng,
                                                eta0=0.5, beta=0.9)
        if scr + 1e-12 < best_score:
            best_seq, best_score = a.copy(), scr
            print(f"   improved after up‑sample: score={best_score:.6f}")
        # tiny local search at the larger scale (fast)
        a, scr = _local_hill_climb(best_seq, deadline, rng,
                                   max_iter=5000, init_step=0.015)
        if scr + 1e-12 < best_score:
            best_seq, best_score = a.copy(), scr
            print(f"   improved after local climb (large): score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided evaluator (includes clipping, scaling, etc.)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
