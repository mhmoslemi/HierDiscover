Algorithm Overview
-----------------
* Work on the probability simplex (non‑negative entries summing to 1) using a
  softmax parametrisation.
* Optimise a smooth approximation of the max‑convolution by Adam.  The
  temperature of the log‑sum‑exp is increased during the run so that we start
  with a dense gradient and end with the true max.
* Several random initialisations (uniform, Dirichlet with several concentrations,
  very spiky draws) are tried for each length.
* After the Adam phase a long projected sub‑gradient descent fine‑tunes the exact
  max‑convolution.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

    v = np.zeros_like(x)

    eps = 1e-12
    best_a = None
    best_raw = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        max_x = np.max(x)

        a = exp_x / exp_x.sum()

        # exact convolution and raw score
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw = 2.0 * n * cur_max
        if raw < best_raw:

            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_raw:.6f}")

        # temperature schedule – linear increase
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # smoothed max via log‑sum‑exp
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)

        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))

        print(f"Baseline (length={len(prev)}) raw={raw:.6f}")
        if raw < best_raw:
            best_raw = raw
            best_seq = prev.copy()
            best_n = len(prev)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        init_logits = []

        # warm‑start from the global best (interpolated)
        if best_seq is not None:
            cand = _interpolate_to_length(best_seq, n)
            init_logits.append(np.log(cand + 1e-12))

        # uniform start
        init_logits.append(np.zeros(n, dtype=np.float64))

        # Dirichlet draws (moderate concentration)
        for _ in range(3):
            alpha = np.random.rand() * 5.0 + 0.2   # α ∈ (0.2, 5.2)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_logits.append(np.log(d + 1e-12))

        # Very spiky Dirichlet draws (tiny concentration)
        for _ in range(2):
            alpha = np.random.rand() * 0.09 + 0.001
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_logits.append(np.log(d + 1e-12))

        local_best_seq = None
        local_best_raw = float("inf")

        for idx, x0 in enumerate(init_logits):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] start {idx+1}/{len(init_logits)} for n={n}")

            # 1) Adam optimisation with smoothed max
            seq, raw_score = _adam_optimize_smoothed(
                x0, n, deadline,
                max_iters=12000, tau0=0.5, tau_max=300.0, lr0=0.12
            )

            # 2) Precise refinement with sub‑gradient descent
            seq, raw_score = _refine_by_subgradient(seq, deadline)

            if raw_score < local_best_raw:
                local_best_raw = raw_score
                local_best_seq = seq.copy()
                print(f"   *** new local best for n={n}: raw={local_best_raw:.6f}")

        if local_best_seq is not None and local_best_raw < best_raw:
            best_raw = local_best_raw
            best_seq = local_best_seq.copy()
            best_n = n
            print(f">>> Updated global best: raw={best_raw:.6f} (length={best_n})")

    # ------------------------------------------------------------------
    # Final long‑run sub‑gradient refinement on the global best.
    if best_seq is not None:
        a = best_seq.copy()
        a, final_raw = _refine_by_subgradient(a, deadline)
        best_seq = a.copy()
        best_raw = final_raw
    else:
        # Fallback: modest uniform sequence.
        n = 5000
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # supplied by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
