"""
Hybrid multi‑seed optimisation for the self‑convolution L∞ problem.

Algorithm
---------

1. Build a diverse pool of probability vectors (uniform, Beta‑families,
   Dirichlet draws, simple analytic shapes and the supplied `height_sequence_1`).

2. Normalise every candidate to have sum = 1 (the objective is
   scaling‑invariant).  Keep the best seed according to the cheap proxy
   `2·n·max(a*a)`.

3. Perform a quick continuation / FISTA refinement on the 5 most
   promising seeds, each with a modest time budget.

4. Take the best result so far and run a long continuation‑FISTA
   optimisation (smoothed log‑sum‑exp surrogate) using the full budget.

5. Finish with a short projected sub‑gradient descent followed by a
   simulated‑annealing polishing phase.

All iterates are projected onto the probability simplex, guaranteeing
non‑negativity and a non‑vanishing sum (the evaluator rejects sums < 0.01).

The returned list is a non‑negative step‑function; `evaluate_sequence`
will automatically compute the true score.
"""

import time
import math
import numpy as np
from operator import itemgetter
from typing import List

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf for numerical stability
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow; clip exponentials
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector (Σ a = 1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution via log‑sum‑exp and its gradient.
    Returns (value, grad) where grad has shape a.
    """
    conv = _self_convolution_full(a)                # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)                     # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                               # soft‑max weights, Σp = 1

    # gradient = 2 * (a_rev * p)   (central n entries)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation + FISTA refinement of a probability vector.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]

    mu = max(_max_self_convolution(a) * 0.5, 1e-7)
    mu_target = 1e-6
    mu_factor = 0.7

    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # ----- inner FISTA for the current mu --------------------------------
        L = max(1.0, 2.0 / mu)          # safe Lipschitz estimate
        y = a.copy()
        t = 1.0

        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        # generous cap – we adapt to the measured speed
        iters = int(min(6000, remaining / 0.0012))
        if iters <= 0:
            break

        for it in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # simple back‑tracking L‑adjustment (max two doublings)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional check of the true (non‑smoothed) objective
            if (it & 31) == 0:
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()
        # --------------------------------------------------------------------
        # Reduce mu and continue
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        a = best_seq.copy()   # warm‑restart
    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent (the method used previously) to
    polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # tiny jitter every few hundred steps
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=len(a))
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Simple SA optimiser: random Gaussian perturbations with projection onto
    the simplex and an exponential‑temperature schedule.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    # linear schedule from T0 down to Tmin
    T0 = 0.2
    Tmin = 1e-5

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        T = T0 * (1.0 - elapsed / time_budget) + Tmin * (elapsed / time_budget)
        sigma = T * 0.15  # perturbation size

        # propose a new point
        cand = a + np.random.normal(scale=sigma, size=a.shape)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand /= s

        cand_val = _raw_score(cand)
        cur_val = _raw_score(a)

        # accept if improvement or with Metropolis probability
        if cand_val < best_val:
            best_val = cand_val
            best_seq = cand.copy()
        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
            a = cand
    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> List[float]:
    """
    Entry point for the competition harness.
    Returns a list of non‑negative heights (a probability vector) that
    (approximately) minimizes `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1. Problem size
    n = 8000  # length of the final step function

    # --------------------------------------------------------------
    # 2. Build a diverse pool of seeds (all are normalised Σ=1)
    candidates = []

    # a) supplied warm‑start, if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base = np.maximum(base, 0.0)
        if base.sum() == 0.0:
            base[:] = 1.0 / len(base)
        else:
            base /= base.sum()
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # b) simple baselines
    candidates.append(np.full(n, 1.0 / n))                # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))    # arcsine

    # c) symmetric Beta shapes (log‑spaced)
    alphas = np.geomspace(0.05, 5.0, num=15)
    for a in alphas:
        candidates.append(_beta_distribution(n, a, a))

    # d) random Beta draws
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2, 2)
        beta = 10 ** rng.uniform(-2, 2)
        candidates.append(_beta_distribution(n, alpha, beta))

    # e) random Dirichlet draws (different concentration)
    for _ in range(12):
        conc = 10 ** rng.uniform(-2, 2)
        vec = rng.gamma(conc, 1.0, size=n)
        vec /= vec.sum()
        candidates.append(vec)

    # f) analytic shapes
    #   triangular (tent) shape
    idx = np.arange(n)
    centre = (n - 1) / 2.0
    tri = 1.0 - np.abs(idx - centre) / (centre + 1.0)
    tri = np.maximum(tri, 0.0)
    tri /= tri.sum()
    candidates.append(tri)

    #   half‑sine shape
    sinsh = np.sin(np.pi * (idx + 0.5) / n)
    sinsh = np.maximum(sinsh, 0.0)
    sinsh /= sinsh.sum()
    candidates.append(sinsh)

    # g) mixtures of uniform and arcsine
    for lam in rng.uniform(0.0, 1.0, size=5):
        mix = lam * np.full(n, 1.0 / n) + (1.0 - lam) * _beta_distribution(n, 0.5, 0.5)
        mix /= mix.sum()
        candidates.append(mix)

    # --------------------------------------------------------------
    # 3. Evaluate raw scores and keep the best seed
    scored = [(seq, _raw_score(seq)) for seq in candidates]
    scored.sort(key=itemgetter(1))
    best_seq, best_raw = scored[0]
    print(f"Initial best raw score: {best_raw:.6f}")

    # --------------------------------------------------------------
    # 4. Quick refinement on a handful of top candidates
    top_k = 5
    quick_budget = max(0.0, (deadline - time.time()) * 0.30)
    per_candidate = quick_budget / top_k if top_k else 0.0
    for i in range(min(top_k, len(scored))):
        cand_seq, cand_raw = scored[i]
        if per_candidate < 5.0:               # avoid too short runs
            continue
        refined_seq, refined_raw = _fista_refine(cand_seq, per_candidate * 0.8)
        print(f"  Quick‑refine {i+1}/{top_k}: raw {refined_raw:.6f}")
        if refined_raw < best_raw:
            best_raw = refined_raw
            best_seq = refined_seq.copy()

    # --------------------------------------------------------------
    # 5. Main continuation‑FISTA phase on the current best sequence
    remaining = deadline - time.time()
    if remaining > 8.0:
        best_seq, best_raw = _fista_refine(best_seq, remaining * 0.60)
        print(f"Long FISTA refinement raw: {best_raw:.6f}")

    # --------------------------------------------------------------
    # 6. Sub‑gradient polishing
    remaining = deadline - time.time()
    if remaining > 4.0:
        best_seq, best_raw = _subgradient_polish(best_seq, remaining * 0.40)
        print(f"Sub‑gradient polish raw: {best_raw:.6f}")

    # --------------------------------------------------------------
    # 7. Simulated‑annealing polishing (optional, very cheap)
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_raw = _simulated_annealing(best_seq, remaining)
        print(f"SA polish raw: {best_raw:.6f}")

    # --------------------------------------------------------------
    # 8. Final evaluation (the harness will also call evaluate_sequence)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
