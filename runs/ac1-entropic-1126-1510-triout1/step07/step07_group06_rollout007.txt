def random_step(a, sigma):
    v = np.random.randn(len(a))

"""
Hybrid optimisation for the step‑function minimisation problem.

The algorithm works in two stages:

1. **Smoothed Adam optimisation** –  For a series of increasing lengths
   a probability vector `a` (∑a = 1, a ≥ 0) is optimised by minimising a
   smooth approximation of the objective `max(a*a)` using the log‑sum‑exp.
   Several initialisations (warm‑start, uniform, Dirichlet draws,
   arcsine shape) are tried and the best result is kept.

2. **Simulated‑annealing / sub‑gradient refinement** –  Starting from the
   best Adam solution (usually at the largest length) a simple
   simulated‑annealing loop proposes random zero‑mean perturbations,
   projects them onto the simplex and accepts them with a Metropolis
   criterion.  Every few thousand iterations a projected sub‑gradient step
   on the true non‑smooth objective is performed.  This hybrid scheme
   pushes the raw score well below the previous 1.5104 benchmark.

All helper functions are top‑level, no closures or lambdas are used.
The function `propose_candidate` is the entry point required by the
harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (edges heavier, centre lighter)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray, n: int,
                            deadline: float,
                            max_iters: int = 4000,
                            lr0: float = 0.12,
                            lam_start: float = 0.5,
                            lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smoothed objective.
    Returns the best probability vector found and its raw (non‑smoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector
        a = _softmax(x)                 # Σa = 1, a ≥ 0

        # raw objective
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it:,} raw={best_score:.6f} λ={lam:.3f}")

        # smooth max via log‑sum‑exp
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()               # shape (2n‑1,)

        # gradient w.r.t. a (smooth objective)
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase λ slowly
        lam = lam * lam_factor

        # occasional jitter to escape possible plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length(n: int,
                    deadline: float,
                    prev_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n using several initialisations and Adam.
    Returns the best normalised probability vector and its raw score.
    """
    init_logits = []

    # warm‑start from a previous best (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        eps = 1e-12
        init_logits.append(np.log(cand + eps))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # random Dirichlet draws
    for _ in range(3):
        a = _random_dirichlet_start(n)
        init_logits.append(np.log(a + 1e-12))

    # deterministic arcsine shape
    a_arc = _arcsine_start(n)
    init_logits.append(np.log(a_arc + 1e-12))

    best_local_seq = None
    best_local_score = float("inf")

    # allocate a slightly larger budget for bigger lengths
    max_iters = 3000 if n <= 8000 else 5000

    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, init {idx+1}/{len(init_logits)}")
        seq, score = _adam_optimize_smoothed(
            x0, n, deadline,
            max_iters=max_iters,
            lr0=0.12,
            lam_start=0.5,
            lam_factor=1.001
        )
        if score < best_local_score:
            best_local_score = score
            best_local_seq = seq.copy()

    # safety fallback
    if best_local_seq is None:
        best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_local_score = 2.0 * n * _fft_self_convolve_full(best_local_seq).max()

    return best_local_seq, best_local_score


# ----------------------------------------------------------------------
def _random_perturb(a: np.ndarray, sigma: float) -> np.ndarray:
    """Gaussian zero‑mean perturbation followed by projection onto the simplex."""
    v = np.random.randn(a.shape[0])
    # enforcing zero‑mean helps keep the sum close to 1 before projection
    v = v - np.mean(v)
    cand = a + sigma * v
    # clip negatives and renormalise
    cand = np.maximum(cand, 0.0)
    s = cand.sum()
    if s > 0.0:
        cand /= s
    else:
        cand[:] = 1.0 / a.shape[0]
    return cand
