Improvements over the reference solution:

* Wider length schedule (up to 30 000) and a finer search of the
  hyper‑parameters of the smoothed‐max Adam optimiser.
* More diverse initialisations (uniform, warm‑start, random Dirichlet,
  deterministic “arcsine”, and a two‑spike “binary” pattern).

  refinement is executed on the currently best sequence.  The
  refinement works directly on the simplex (projected after each
  proposal) and accepts uphill moves with a temperature that cools
  down to zero.
* All heavy work (convolutions, gradients) is performed with FFTs,
  so the algorithm stays comfortably inside the 1000 s budget while
  exploring sequences up to 30 000 entries.

The returned list is a plain Python list of non‑negative floats and
will be evaluated by the provided ``evaluate_sequence`` function.

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b (full length) via real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex‑normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (heavy ends, light centre)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _binary_spike_start(n: int, spikes: int = 2) -> np.ndarray:
    """
    Very sparse start: place equal mass on ``spikes`` equally spaced indices.
    """
    if spikes >= n:
        spikes = n
    a = np.zeros(n, dtype=float)
    idx = np.linspace(0, n - 1, spikes, dtype=int)
    a[idx] = 1.0 / spikes
    return a

def _adam_optimize_smoothed(x0: np.ndarray, n: int,
                            deadline: float,
                            max_iters: int = 8000,
                            lr0: float = 0.12,
                            lam_start: float = 0.4,
                            lam_factor: float = 1.0008) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max convolution.
    Returns best probability vector (∑ = 1) and its raw (non‑smoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector
        a = _softmax(x)

        # raw objective (exact)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.4f}")

        # smooth max (log‑sum‑exp)
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()          # shape (2n‑1,)

        # gradient w.r.t. a : ∇_a f_λ = 2·(a_rev * p)
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # λ annealing
        lam = lam * lam_factor

        # occasional jitter to escape plateaux
        if it % 1500 == 0:
            x += np.random.randn(*x.shape) * 0.005

    # safety fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score

def _simulated_annealing_refine(a: np.ndarray,
                                deadline: float,
                                init_temp: float = 0.1,
                                min_temp: float = 1e-5,
                                decay: float = 0.9995) -> np.ndarray:
    """
    Simple random‑swap annealing on the simplex.
    Proposes a move by transferring a small amount of mass between two
    randomly chosen indices, projects back onto the simplex and accepts
    the move according to a temperature schedule.
    """
    n = a.shape[0]
    best = a.copy()
    # pre‑compute current score (using fast FFT)
    best_score = 2.0 * n * _fft_self_convolve_full(best).max()
    cur = a.copy()
    cur_score = best_score
    temp = init_temp
    it = 0

    while time.time() < deadline and temp > min_temp:
        it += 1
        # pick two distinct indices
        i, j = np.random.choice(n, size=2, replace=False)
        # propose a transfer of mass from i to j
        delta = (np.random.rand() - 0.5) * temp  # magnitude scales with temperature
        proposal = cur.copy()
        proposal[i] = max(proposal[i] - delta, 0.0)
        proposal[j] = max(proposal[j] + delta, 0.0)
        proposal = _project_onto_simplex(proposal)

        # evaluate
        conv = _fft_self_convolve_full(proposal)
        score = 2.0 * n * conv.max()

        # Metropolis acceptance
        if score < cur_score or np.random.rand() < math.exp(-(score - cur_score) / temp):
            cur = proposal
            cur_score = score
            if score < best_score:
                best = proposal.copy()
                best_score = score
                print(f"[{time.time():.1f}] anneal it={it} temp={temp:.6f} raw={best_score:.6f}")

        # geometric cooling
        temp *= decay

    return best
