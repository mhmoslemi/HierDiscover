def _distribution_edge_heavy(n, p):
    idx = np.arange(1, n+1)  # 1..n
    # Edge distance product (i)*(n-i+1)
    vals = (idx * (n - idx + 1)) ** (-p)   # heavy at edges
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

"""
Propose a step‑function (list of non‑negative heights) that minimises the
`evaluate_sequence` objective.

The search consists of three stages:

1. **Broad analytic sweep** – we generate many candidate shapes
   (uniform, triangular, exponential, beta‑sym, edge‑heavy power,
   mixtures) for a single length `n` (taken from the previous best
   solution when available, otherwise 3000).  The best of those
   candidates seeds the later optimisation.

2. **Continuous optimisation** – an Adam optimiser on a smooth
   approximation of the max‑convolution followed by a classic
   sub‑gradient descent (the same ideas as the reference code but with
   a finer schedule).

3. **Simulated‑annealing hill‑climber** – we repeatedly move a small
   amount of mass from one index to another while maintaining the sum
   constraint.  Deteriorating moves are occasionally accepted according
   to a temperature schedule, allowing the algorithm to escape local
   minima that the pure descent can get stuck in.

All operations respect the simplex constraint (`a >= 0`, `sum(a)=1`) and
clip tiny values for numerical stability.  The whole routine stops
short of the 1000 s budget, printing intermediate scores so the progress
can be monitored.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector whose sum is 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric beta(p,p) on the discrete grid {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n, p):
    """Heavy at the ends: a_i ∝ ((i+1)*(n-i))^{-p}  (p>0)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = (i * (n - i + 1.0)) ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _distribution_mixed_uniform_edge(n, w):
    """Mixture (1‑w)·uniform + w·edge_power(p=1)."""
    uni = _distribution_uniform(n)
    edge = _distribution_edge_power(n, p=1.0)
    vals = (1 - w) * uni + w * edge
    vals /= vals.sum()
    return vals


def _distribution_random_dirichlet(n, concentration):
    """Random Dirichlet draw with the same concentration for every component."""
    alpha = np.full(n, concentration, dtype=np.float64)
    vals = np.random.gamma(alpha, 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j.
    Updates both the vector `a` and its full self‑convolution `conv`
    in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # linear part (remove contributions of i, add contributions of j)
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    # quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

    # --------------------------------------------------------------
    # 1️⃣  Analytic sweep (single length – we use the length of the previous
    #      best if it exists, otherwise a modest 3000)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        n = prev.shape[0] if prev.ndim == 1 else 3000
    except Exception:
        n = 3000

    print(f"[{time.time():.1f}] Using length n={n}")

    candidates = []

    # core families
    candidates.append(_distribution_uniform(n))
    candidates.append(_distribution_triangular(n))
    candidates.append(_distribution_exponential(n, lam=0.003))
    candidates.append(_distribution_exponential(n, lam=0.01))

    # beta‑sym with a finer grid around the uniform point (p=1)
    for p in (0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85, 0.9):
        candidates.append(_distribution_beta_sym(n, p))

    # edge‑heavy power families
    for p in (0.5, 0.7, 0.9, 1.1, 1.3):
        candidates.append(_distribution_edge_power(n, p))

    # mixtures of uniform and edge‑heavy
    for w in np.linspace(0.05, 0.45, 9):
        candidates.append(_distribution_mixed_uniform_edge(n, w))

    # a few random Dirichlet draws (different concentrations)
    for conc in (0.3, 0.6, 1.0, 2.0):
        for _ in range(4):
            candidates.append(_distribution_random_dirichlet(n, conc))

    # a random sparse start (as in the reference code)
    sparse = np.zeros(n, dtype=np.float64)
    k = max(2, n // 200)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    candidates.append(sparse)

    best_seq = None
    best_score = float("inf")

    for vec in candidates:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
            print(f"[{time.time():.1f}] init candidate new best {sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Continuous optimisation – Adam on a smooth max
    a = best_seq.copy()
    n = a.shape[0]

    # Adam hyper‑parameters (a bit more aggressive than the reference)
    beta1, beta2, adam_eps = 0.9, 0.999, 1e-8
    lr = 0.16
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule: increase the “soft‑max sharpness” α gradually
    schedule = [
        (1.0, 3000),
        (5.0, 3000),
        (15.0, 6000),
        (60.0, 6000),
        (300.0, 6000),
        (1200.0, 4000),
    ]

    best_seq_adam = a.copy()
    best_score_adam = _max_convolution_score(a)

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            t += 1
            # self‑convolution
            b = _fft_self_convolve(a)

            # smooth max via soft‑max weighting
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            w /= w.sum()

            # gradient of the smooth max (same algebra as in the reference)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # project onto the simplex
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            if t % 250 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score_adam - 1e-12:
                    best_score_adam = cur
                    best_seq_adam = a.copy()
                    print(f"[{time.time():.1f}] Adam iter={t} new best {cur:.6f}")

    a = best_seq_adam.copy()
    best_score = best_score_adam
    best_seq = best_seq_adam.copy()

    # --------------------------------------------------------------
    # 3️⃣  Exact sub‑gradient descent (as in the reference)
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.22   # a slightly larger starting step
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Sub‑grad iter={it} new best {cur:.6f}")

        if time.time() + 0.5 > deadline:
            break

    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    # --------------------------------------------------------------
    # 4️⃣  Simulated‑annealing hill‑climber (accept occasional uphill moves)
    it = 0
    temperature = M * 0.6            # initial temperature
    cool_factor = 0.9999
    min_delta = 1e-6

    while time.time() < deadline:
        it += 1
        # pick a source (bias towards larger entries) and a random target
        src = np.random.choice(n, p=a)          # higher probability if a_i is big
        tgt = np.random.randint(n)
        if src == tgt:
            continue

        # propose a transfer; we allow up to 4 % of the source mass
        max_delta = min(a[src] * 0.9, 0.04)
        delta = max_delta * np.random.rand()
        if delta < min_delta:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, src, tgt, delta)

        new_M = conv.max()
        deltaM = new_M - M

        # accept if improvement, otherwise with probability exp(-Δ/Temp)
        if deltaM < -1e-12 or np.random.rand() < math.exp(-deltaM / max(temperature, 1e-12)):
            M = new_M
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA iter={it} new best {cur_score:.6f}")
        else:
            a = a_snap
            conv = conv_snap

        # temperature schedule
        temperature *= cool_factor
        if temperature < 1e-12:
            temperature = 1e-12

        # occasional jitter to break long plateaux
        if it % 4000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    final_score = 2.0 * n * conv.max()
    print(f"Finished – best score ≈ {final_score:.6f} (n={n})")
    return best_seq.tolist()
