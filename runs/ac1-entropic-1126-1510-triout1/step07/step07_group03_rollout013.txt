"""
Proposes a non‑negative step‑function (probability vector) that
minimises `evaluate_sequence`.  The method combines:

* a rich palette of analytic families (including novel monotone power‑law
  shapes) to obtain a good seed;
* a smoothed Adam optimiser on a log‑sum‑exp approximation of the
  max‑convolution;
* a short exact sub‑gradient descent;
* a greedy active‑set swap;
* a final pure hill‑climbing phase that moves tiny amounts of mass
  between random positions only when the objective improves.

All stages respect a global deadline (default 1000 s).  The best vector
found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ===== Utility functions =====
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast O(n log n) algorithm based on sorting.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _objective(a: np.ndarray) -> float:
    """Objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# ==== Analytic families ====

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals

def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute log to avoid underflow
    log_w = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(log_w - np.max(log_w))  # stabilise
    w /= w.sum()
    return w

def _distribution_exponential_sym(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

def _distribution_one_sided_power(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals

def _distribution_one_sided_exp(n: int, lam: float = 0.01) -> np.ndarray:
    """Monotone decreasing exponential."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals

def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a diverse set of start vectors for a given length."""
    cand = []
    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # beta family
    for p in (0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # symmetric exponential
    for lam in (0.001, 0.002, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_sym(n, lam))

    # monotone families – decreasing

        cand.append(_distribution_one_sided_power(n, p))
    for lam in (0.001, 0.003, 0.008, 0.015):
        cand.append(_distribution_one_sided_exp(n, lam))

    # random dense Dirichlet (full support)
    rnd = np.random.rand(n)
    cand.append(rnd)

    # random sparse (≈0.5 % non‑zeros)
    k = max(2, n // 200)

    cand.append(block)

def _smooth_adam_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth approximation of max‑convolution."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # schedule: larger smoothing at the end
    schedule = [
        (1.0, 1500),
        (5.0, 1500),
        (20.0, 1500),
        (100.0, 1500),
        (500.0, 1500),
        (2000.0, 1500),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights
            w = np.exp(alpha * (b - maxb))
            ws = w.sum()
            if ws == 0 or not np.isfinite(ws):
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= ws

            # gradient of the smoothed max (size n)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + simplex projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smooth α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score

def _exact_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the genuine max‑convolution objective."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        # sub‑gradient from active set
        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum + diminishing stepsize
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        if time.time() + 0.5 >= deadline:
            break
    return best_seq, best_score

def _greedy_swap_opt(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from high‑gradient to low‑gradient indices while improving."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]
        j = int(np.argmin(grad))

        if i == j:
            break

        delta = min(a[i] * 0.5, 0.005)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")
        else:
            # try a smaller delta
            delta *= 0.25
            if delta < 1e-8:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] tiny swap improved best {best_score:.6f}")

    return best_seq, best_score

def _hill_climb_refine(a: np.ndarray, deadline: float,
                       best_seq: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """Pure local search: move tiny mass between random indices if it improves."""
    n = a.shape[0]
    # pre‑compute a mask for indices that have non‑negligible mass
    min_mass = 1e-12
    rng = np.random.default_rng()
    while time.time() < deadline:
        # pick a source with some mass
        src_idxs = np.where(a > min_mass)[0]
        if src_idxs.size == 0:
            break
        i = rng.choice(src_idxs)
        j = rng.integers(0, n - 1)
        if j == i:
            continue
        # step size shrinks gradually
        max_delta = min(a[i] * 0.3, 0.01)

"""
Proposes a non‑negative step‑function (probability vector) that
minimises ``evaluate_sequence``.  The search combines several complementary
stages:

1. **Rich initialisation** – a wide palette of analytic families (uniform,
   arcsine, symmetric beta, symmetric exponential, monotone power‑law,
   monotone exponential, block spikes, random dense and sparse vectors) is
   generated for several lengths.  The best few candidates are kept as seeds.

2. **Smoothed Adam descent** – minimise a log‑sum‑exp smooth approximation
   of the max‑convolution.  The smoothing parameter α is increased
   progressively while the vector is projected onto the probability simplex.

3. **Exact sub‑gradient descent** – a classic sub‑gradient method with momentum
   refines the solution with respect to the true max‑convolution objective.

4. **Greedy active‑set swapping** – repeatedly move mass from an index with a
   large sub‑gradient to one with a small sub‑gradient while the objective
   improves.

5. **Annealed local search** – a cheap Metropolis‑type random walk with an
   exponential cooling schedule explores the remaining search space and
   captures the last few improvements.

All stages respect a global deadline (default 1000 s).  The best vector found
is returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families (used for initial candidates)
# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use logs for stability when p is small
    log_w = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(log_w - np.max(log_w))
    w /= w.sum()
    return w


def _distribution_exponential_sym(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_one_sided_power(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_one_sided_exp(n: int, lam: float = 0.01) -> np.ndarray:
    """Monotone decreasing exponential."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int) -> list[np.ndarray]:
    """Generate a diverse set of start vectors for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # symmetric beta (U‑shaped and bell‑shaped)
    for p in (0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # symmetric exponential (peaked centre)
    for lam in (0.001, 0.002, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential_sym(n, lam))

    # monotone decreasing families – often give a different trade‑off
    for p in (0.1, 0.3, 0.6, 1.0, 2.0, 3.0):
        cand.append(_distribution_one_sided_power(n, p))
    for lam in (0.001, 0.003, 0.008, 0.015):
        cand.append(_distribution_one_sided_exp(n, lam))

    # random dense vector (full support)
    cand.append(np.random.rand(n))

    # random sparse vector (≈0.5 % non‑zero)
    k = max(2, n // 200)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    cand.append(sparse)

    # two‑block spikes at the ends (often good for large n)
    k = max(1, int(0.08 * n))
    block = np.zeros(n, dtype=np.float64)
    block[:k] = 1.0
    block[-k:] = 1.0
    block /= block.sum()
    cand.append(block)

    return cand

# Optimisation kernels

def _smooth_adam_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp smooth approximation of the max‑convolution."""
    n = a.shape[0]
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    schedule = [
        (1.0, 1500),
        (5.0, 1500),
        (20.0, 1500),
        (100.0, 1500),
        (500.0, 1500),
        (2000.0, 1500),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score
            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            w = np.exp(alpha * (b - maxb))
            ws = w.sum()
            if ws == 0 or not np.isfinite(ws):
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= ws
            # gradient of smooth max (size n)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_onto_simplex(a, 1.0)

            # occasional exact check
            if t % 200 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smooth α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


def _exact_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent on the true max‑convolution objective."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        if time.time() + 0.5 >= deadline:
            break
    return best_seq, best_score


def _greedy_swap_opt(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from a high‑gradient index to a low‑gradient one while improving."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]
        j = int(np.argmin(grad))

        if i == j:
            break

        delta = min(a[i] * 0.5, 0.005)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")
        else:
            delta *= 0.25
            if delta < 1e-8:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] tiny swap improved best {best_score:.6f}")

    return best_seq, best_score


def _annealed_local_search(a: np.ndarray, deadline: float,
                           best_seq: np.ndarray,
                           best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis‑type walk with exponential cooling to polish the solution."""
    n = a.shape[0]
    cur = a.copy()
    cur_score = best_score
    rng = np.random.default_rng()

    # start with a modest temperature (a few percent of the current score)
    T = max(best_score * 0.02, 1e-6)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        i = rng.integers(0, n)
        if cur[i] <= 1e-14:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = min(cur[i], 0.05 * cur[i])
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta

        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_score = _objective(new)
        dE = new_score - cur_score

        if dE < 0 or rng.random() < math.exp(-dE / T):
            cur = new
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_seq = cur.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] anneal iter={iteration} new best {best_score:.6f}")

        # geometric cooling
        if iteration % 2500 == 0:
            T *= 0.85
            if T < 1e-8:
                T = 1e-8

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best vector found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # A handful of lengths – larger n gives a tighter theoretical bound.
    lengths = [1200, 2000, 3000, 5000, 8000, 12000]

    overall_best_seq = None
    overall_best_score = float("inf")
    # keep the top few seeds for multi‑restart optimisation
    top_seeds: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1. Scan analytic families for each length.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

            # maintain up to 5 best seeds
            if len(top_seeds) < 5:
                top_seeds.append((sc, cand.copy()))
                top_seeds.sort(key=lambda x: x[0])
            else:
                if sc < top_seeds[-1][0]:
                    top_seeds[-1] = (sc, cand.copy())
                    top_seeds.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Re‑use a previous solution if it exists.
    # ------------------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = _project_onto_simplex(prev, 1.0)
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never happen).
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑restart optimisation – run the pipeline on the best few seeds.
    # ------------------------------------------------------------------
    for seed_idx, (seed_score, seed_vec) in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        a = seed_vec.copy()
        cur_best_score = seed_score
        cur_best_seq = a.copy()

        # smoothed Adam descent
        a, cur_best_score = _smooth_adam_descent(a, deadline,
                                                 best_seq=a,
                                                 best_score=cur_best_score)

        # exact sub‑gradient descent
        a, cur_best_score = _exact_subgradient_descent(a, deadline,
                                                       best_seq=a,
                                                       best_score=cur_best_score)

        # greedy swap optimisation
        a, cur_best_score = _greedy_swap_opt(a, deadline,
                                             best_seq=a,
                                             best_score=cur_best_score)

        # final annealed local search (uses whatever time is left)
        a, cur_best_score = _annealed_local_search(a, deadline,
                                                   best_seq=a,
                                                   best_score=cur_best_score)

        if cur_best_score < overall_best_score:
            overall_best_score = cur_best_score
            overall_best_seq = cur_best_seq.copy()
            print(f"[{time.time():.1f}] seed {seed_idx} produced new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best vector found.
    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
