"""

Algorithm
---------
1.  Keep the best sequence found so far (initialised from the optional
    global ``height_sequence_1``).
2.  Optimise a family of increasing lengths.  For each length we

4.  The best raw score (``2·n·max(a*a)``) across all lengths is kept,

5.  The resulting list is returned – ``evaluate_sequence`` will be called

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    # gradient of weighted sum w.r.t a (see derivation in original code)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _shape_sin_pow(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p   (p ≥ 0)."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.sin(np.pi * x) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a[:] = 1.0 / n
    return a


def _shape_arcsine(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-alpha}   (alpha > 0, small alpha → flat)."""
    i = np.arange(n, dtype=float) + 0.5
    prod = i * (n - i)
    # avoid division by zero – prod is never zero because i∈(0,n)
    a = prod ** (-alpha)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a[:] = 1.0 / n
    return a


def _shape_triangle(n: int) -> np.ndarray:
    """Symmetric triangular distribution (linear rise then fall)."""
    m = n // 2
    # build increasing part, then decreasing
    inc = np.arange(1, m + 1, dtype=float)
    dec = np.arange(n - m, 0, -1, dtype=float)
    a = np.concatenate([inc, dec])
    if a.size > n:
        a = a[:n]
    a /= a.sum()
    return a

def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 6000,
                            tau0: float = 0.5,
                            tau_max: float = 100.0,
                            lr0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max (log‑sum‑exp) surrogate.
    Returns the best probability vector a (Σa = 1) encountered and its
    *raw* score 2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # soft‑max to obtain the probability vector
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # exact convolution max (for bookkeeping)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        # temperature schedule (linear increase)
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # smooth‑max weights (stable soft‑max)
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # gradient of the smooth max w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # chain rule through the soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _refine_by_subgradient_accelerated(a: np.ndarray,
                                      deadline: float,
                                      max_iters: int = 500000,
                                      step0: float = 0.07,
                                      momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with Nesterov‑type momentum on the exact
    max‑convolution.  Returns the best (a, raw_score) pair found.
    """
    n = a.shape[0]
    a_prev = a.copy()
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = step0 / math.sqrt(it)
        # Nesterov‑like extrapolation
        a_new = a - step * grad + momentum * (a - a_prev)
        a_new = _project_onto_simplex(a_new)
        a_prev, a = a, a_new

        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] acc sub‑grad it={it} raw={best_raw:.6f}")

        # occasional tiny random perturbation
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
            a_prev = a.copy()

    return best_a, best_raw

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the challenge harness.
    Returns a list of non‑negative heights (probability simplex) that aims
    to minimise ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # reserve a small margin for the final evaluation call
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    best_seq = None
    best_raw = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Lengths to explore – we increase gradually and stop early if time runs out
    lengths = [2000, 4000, 8000, 12000, 20000, 30000]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # --------------------------------------------------------------
        # Build a small pool of starting points (probability vectors)
        init_candidates = []

        # Warm‑start from the current global best (if any)
        if best_seq is not None:
            init_candidates.append(_interpolate_to_length(best_seq, n))

        # Uniform distribution
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # Deterministic shapes
        init_candidates.append(_shape_sin_pow(n, p=0.5))
        init_candidates.append(_shape_sin_pow(n, p=1.5))
        init_candidates.append(_shape_arcsine(n, alpha=0.25))
        init_candidates.append(_shape_arcsine(n, alpha=0.45))
        init_candidates.append(_shape_triangle(n))

        # Random Dirichlet draws (moderate concentration)
        for _ in range(3):
            alpha = np.random.rand() * 3.0 + 0.5   # α ∈ (0.5,3.5)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_candidates.append(d)

        # Very spiky draws (tiny concentration)
        for _ in range(2):
            alpha = np.random.rand() * 0.09 + 0.001   # α ∈ (0.001,0.091)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_candidates.append(d)

        # --------------------------------------------------------------
        # Parameterisation for the Adam stage – number of iterations scaled with n
        base_iters = 8000
        max_adam_iters = max(2000, int(base_iters * (4000.0 / n)))

        # Parameterisation for the sub‑gradient stage – also scaled with n
        base_sub_iters = 500000
        max_sub_iters = max(50000, int(base_sub_iters * (4000.0 / n)))

        local_best_seq = None
        local_best_raw = float("inf")

        for idx, a0 in enumerate(init_candidates):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] start {idx+1}/{len(init_candidates)} for n={n}")

            # Convert to logits for the Adam optimiser
            logits0 = np.log(a0 + 1e-12)

            # ---------------------------------------------------------
            # 1) Adam optimisation on the smooth‑max surrogate
            a_adam, raw_adam = _adam_optimize_smoothed(
                logits0, n, deadline,
                max_iters=max_adam_iters,
                tau0=0.5,
                tau_max=150.0,          # a slightly hotter final temperature
                lr0=0.12
            )

            # ---------------------------------------------------------
            # 2) Exact sub‑gradient refinement with momentum
            a_refined, raw_refined = _refine_by_subgradient_accelerated(
                a_adam, deadline, max_iters=max_sub_iters
            )

            if raw_refined < local_best_raw:
                local_best_raw = raw_refined
                local_best_seq = a_refined.copy()
                print(f"   *** new local best for n={n}: raw={local_best_raw:.6f}")

        # --------------------------------------------------------------
        # Update the global best if this length gave an improvement
        if local_best_seq is not None and local_best_raw < best_raw:
            best_raw = local_best_raw
            best_seq = local_best_seq.copy()
            best_n = n
            print(f">>> Updated global best: raw={best_raw:.6f} (length={best_n})")

    # ------------------------------------------------------------------
    # Final short momentum sub‑gradient refinement on the best sequence
    if best_seq is not None:
        a_final, raw_final = _refine_by_subgradient_accelerated(best_seq, deadline,
                                                                max_iters=200000)
        best_seq = a_final.copy()
        best_raw = raw_final
    else:
        # fallback – uniform distribution of moderate size
        n = 5000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_seq)

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # supplied by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
