import cvxpy as cp

def solve_lp_for_g(f, M):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    # conv(f,g) <= M for k=0..2n-2
    # conv formula: use Toeplitz matrix: for each k,
    # conv_k = sum_{i=0}^{n-1} f_i * g_{k - i} where index valid.
    # Equivalent: conv_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f_j * g_{k-j}
    # We'll create constraint with expression.
    # vectorized: cp.sum(cp.multiply(f, shift(g)))?
    # Simpler: Construct matrix A (size (2n-1) x n) where A[k,j] = f_{k-j} if 0 <= k-j < n else 0.
    # Build via Toeplitz.
    import numpy as np
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A[k,j] = f[i]
    constraints.append(A @ g <= M)
    # also trivial sum(g) <= something? We maximize sum(g) anyway.
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if g.value is None:
        # fallback: return uniform
        return np.full(n, 1/n)
    return np.array(g.value).clip(min=0)

def solve_lp_max_sum_under_conv(f, M):
    n = len(f)
    m = 2*n-1
    # Build constraints matrix
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    bounds = [(0, None)]*n
    # linprog for max sum => minimize -sum(g)
    res = scipy.optimize.linprog(-np.ones(n), A_ub=A, b_ub = np.full(m, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: uniform distribution
        return np.full(n, 1.0/n)
    return res.x

def lp_iterative_refinement(seq, deadline, max_iters=15, t=0.1):
    a = seq.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute M
        M = np.max(_fft_self_convolve(a))
        # solve LP to get g
        g = solve_lp_max_sum_under_conv(a, M)
        # normalize g to sum=1
        if g.sum() == 0:
            break
        g = g / g.sum()
        # mixture update
        a = (1 - t) * a + t * g
        # optionally renormalize to ensure sum 1 (should already)
        a = a / a.sum()
        # evaluate
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] LP iter {it+1} improved score {best_score:.6f}")
        # maybe adapt t (reduce if no improvement)
        # For simplicity keep t constant.
    return best_a, best_score

"""
Propose a step‑function (non‑negative list) that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Broad sampling** – generate many candidates from a few analytic families
   (arcsine, triangular, exponential, symmetric beta) of several lengths and
   a handful of random Dirichlet draws.  All candidates are ℓ₁‑normalised
   (sum = 1) and scored by the exact surrogate
   ``score = 2·n·max(conv(a,a))`` (identical to ``evaluate_sequence`` when the
   sum is 1).

2. **LP‑based refinement** – for each promising candidate of moderate size
   (≤ 1800 entries) we run a few iterations of the following:  
   *let M = max(conv(a,a))*; solve the linear program  

        maximise   Σ g_j
        subject to g_j ≥ 0 , (a * g)_k ≤ M   for all k

   (the convolution matrix is built explicitly).  The optimum g spreads the
   mass as uniformly as possible without increasing the current worst
   convolution value.  A convex combination ``a←(1‑t)a + t·ĝ`` (with a small
   ``t``) is taken, then projected back onto the simplex.

3. **Smoothed gradient descent** – on the simplex we minimise a
   differentiable surrogate of the max (log‑sum‑exp) using projected gradient
   descent with a step size proportional to ``1/√iteration``.  The gradient
   is obtained analytically via a convolution with the soft‑max weights.

4. **Pairwise Frank‑Wolfe polishing** – finally a cheap FW‑style move swaps
   a tiny amount of mass from the component with the largest sub‑gradient to
   the component with the smallest one; a full or a half transfer is kept
   when it improves the exact surrogate.

The whole routine respects the supplied time budget (≈ 1000 s) and always
returns a valid list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(a*a)  (assumes Σa = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric beta density on {0,…,n‑1} (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density to avoid underflow
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def _solve_lp_max_sum_under_conv(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   subject to g_j ≥ 0  and  (f * g)_k ≤ M.
    Returns the optimal g (un‑normalised) or None if the solver fails.
    """
    n = f.shape[0]
    m = 2 * n - 1
    # Build the (m × n) convolution matrix A, where (A @ g)_k = (f*g)_k
    A = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    c = -np.ones(n)                     # minimise -Σ g_j  → maximise Σ g_j
    bounds = [(0, None)] * n
    res = opt.linprog(
        c,
        A_ub=A,
        b_ub=np.full(m, M, dtype=np.float64),
        bounds=bounds,
        method="highs",
        options={"presolve": True, "symm": False},
    )
    if not res.success:
        return None
    return np.asarray(res.x, dtype=np.float64)

def _smooth_max_and_grad(a: np.ndarray, alpha: float = 30.0):
    """
    Compute a smooth approximation of max(a*a) using log‑sum‑exp
    and its gradient w.r.t. a.
    Returns (smoothed_max, gradient).
    """
    b = _fft_self_convolve(a)                     # full convolution (len 2n‑1)
    max_b = b.max()
    # stable soft‑max weights
    y = alpha * (b - max_b)                        # ∈ (‑inf, 0]
    exp_y = np.exp(y)
    w = exp_y / exp_y.sum()                       # size 2n‑1, Σw = 1
    # gradient = 2 * (a_rev * w) on the slice [n‑1 … n‑1+n‑1]
    conv_rev = _fft_convolve(a[::-1], w)          # length n + (2n‑1) – 1 = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    # smoothed maximum value
    smooth_max = max_b + np.log(exp_y.sum()) / alpha
    return smooth_max, grad

        cand_score = _max_convolution_score(cand)
        improved = False
        if cand_score < cur_score - 1e-12:
            cur, cur_score = cand, cand_score
            improved = True
        else:
            # try half‑transfer
            half = 0.5 * transfer
            cand2 = cur.copy()
            cand2[i_src] -= half
            cand2[i_dst] += half
            cand2_score = _max_convolution_score(cand2)
            if cand2_score < cur_score - 1e-12:
                cur, cur_score = cand2, cand2_score
                improved = True
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, cur.copy()
            print(f"[{time.time():.1f}] FW iter {it} new best {best_score:.6f}")
        # occasional jitter to escape tiny plateaus
        if it % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 1e-5
            cur = _project_onto_simplex(cur + jitter, s=1.0)
            cur_score = _max_convolution_score(cur)
    return best_seq, best_score


def _subgrad(a: np.ndarray):
    """
    Exact (binary) sub‑gradient of max(a*a) on the simplex.
    Returns (grad, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (each a_i^2 appears only once in the true sub‑gradient)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the supplied time budget and always returns a list
    of non‑negative floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1. Choose a set of lengths to explore
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 1800
    lengths = set()
    for scale in (0.5, 0.75, 1.0, 1.25, 1.5):
        n = int(base_n * scale)
        if 500 <= n <= 3500:
            lengths.add(n)
    for _ in range(7):
        n = np.random.randint(500, min(3500, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # --------------------------------------------------------------
    # 2. Broad sampling – analytic families + Dirichlet draws
    candidates = []          # [(array, score), …]

    beta_ps = (
        list(np.arange(0.02, 0.21, 0.02))
        + list(np.arange(0.25, 0.71, 0.05))
        + list(np.arange(0.80, 1.51, 0.10))
        + [2.0, 0.01, 0.005]   # extra extreme U‑shapes
    )
    dirichlet_alphas = [0.2, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # analytic families
        for gen, name in (
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.005), "exponential"),
        ):
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} {name:12s} score={score:.6f}")

        # symmetric beta family (dense grid)
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))

        # random Dirichlet draws
        for alpha in dirichlet_alphas:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            score = _max_convolution_score(a)
            candidates.append((a, score))

    # --------------------------------------------------------------
    # 3. Include the previous best (if present)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Keep the top seeds for intensive refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(8, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k} for refinement.")
    print(f"Current best seed score = {top_candidates[0][1]:.6f}")

    # --------------------------------------------------------------
    # 5. Main refinement loop
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, seed_score) in enumerate(top_candidates, 1):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]
        print(f"[{time.time():.1f}] refining seed {idx}/{top_k} (n={n})")
        cur_seq = seed_seq.copy()

        # 5a) LP‑based refinement for moderate sizes
        if n <= 1800:
            cur_seq, lp_score = _lp_refinement(cur_seq, deadline,
                                               max_iters=8, t_start=0.30)
            if lp_score < best_score - 1e-14:
                best_score, best_seq = lp_score, cur_seq.copy()
                print(f"[{time.time():.1f}] after LP improvement best={best_score:.6f}")

        # 5b) Smoothed gradient descent (always run)
        cur_seq, gd_score = _smooth_gradient_descent(cur_seq, deadline,
                                                     max_iters=150_000,
                                                     alpha=30.0,
                                                     eta0=0.35)
        if gd_score < best_score - 1e-14:
            best_score, best_seq = gd_score, cur_seq.copy()
            print(f"[{time.time():.1f}] after GD improvement best={best_score:.6f}")

        # 5c) Pairwise Frank‑Wolfe polishing (cheap but effective)
        cur_seq, fw_score = _fw_pairwise_refine(cur_seq, deadline,
                                                max_iters=30_000)
        if fw_score < best_score - 1e-14:
            best_score, best_seq = fw_score, cur_seq.copy()
            print(f"[{time.time():.1f}] after FW improvement best={best_score:.6f}")

    # --------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # The surrogate is exactly the same as ``evaluate_sequence`` when
    # the sum of the vector equals 1.  Normalise once more for safety.
    best_seq = best_seq / best_seq.sum()
    return best_seq.tolist()
