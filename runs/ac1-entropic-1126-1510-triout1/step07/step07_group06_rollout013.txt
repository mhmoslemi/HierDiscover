"""
Propose a low‑score step function.

Algorithm
---------
1.  Load any previously discovered construction (`height_sequence_1`).
2.  For several candidate lengths (up to 20 000) perform a *tiny*
    parametric search over a one‑parameter family

        a_i ∝ ((i+½)(n‑½‑i) + ε)^{−p}      (0<p<1)

    which contains the known “arcsine’’ shape (p=½).  The family is
    normalised to a probability vector (∑a_i=1) and its raw score
    `2·n·max(a*a)` is evaluated with FFT‑based convolutions.

3.  Keep the best vector found in step 2 (across all lengths).

4.  Refine that vector with a smooth‑max Adam optimiser
    (log‑sum‑exp surrogate) – the optimiser works on the logits
    of a soft‑max parametrisation and respects the simplex constraint
    automatically.

5.  Finish with a few thousand projected sub‑gradient steps that
    directly minimise the true `max(a*a)`.

All heavy work uses O(n log n) FFTs; a 20 000‑long vector can be
optimised well within the 1000 s budget.  The routine prints progress
and finally returns a plain Python list ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np
import sys

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _shape_exponent(p: float, n: int, eps: float = 1e-12) -> np.ndarray:
    """
    Symmetric Beta‑like shape:
        a_i ∝ ((i+½)(n‑½‑i) + eps)^{−p}
    Returns a normalised probability vector (sum = 1).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) * (n - 0.5 - i) + eps
    a = np.power(x, -p)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a[:] = 1.0 / n
    return a


def _raw_score(a: np.ndarray) -> float:
    """Exact (non‑smooth) objective 2·n·max(a*a) for a normalised vector."""
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _grid_search_shape(
    n: int,
    p_min: float = 0.2,
    p_max: float = 0.8,
    coarse_steps: int = 31,
    fine_factor: int = 4,
) -> tuple[np.ndarray, float, float]:
    """
    Coarse‑to‑fine search of the exponent p for a given length n.
    Returns the best vector, its raw score and the optimal p.
    """
    best_raw = float("inf")
    best_a = None
    best_p = None

    # ------------------------------------------------------------------
    # coarse sweep
    ps = np.linspace(p_min, p_max, coarse_steps)
    for p in ps:
        a = _shape_exponent(p, n)
        raw = _raw_score(a)
        if raw < best_raw:
            best_raw = raw
            best_a = a
            best_p = p

    # ------------------------------------------------------------------
    # fine sweep around the best coarse p
    delta = (p_max - p_min) / (coarse_steps - 1)
    lo = max(p_min, best_p - delta)
    hi = min(p_max, best_p + delta)
    fine_steps = coarse_steps * fine_factor
    ps_fine = np.linspace(lo, hi, fine_steps)
    for p in ps_fine:
        a = _shape_exponent(p, n)
        raw = _raw_score(a)
        if raw < best_raw:
            best_raw = raw
            best_a = a
            best_p = p

    return best_a, best_raw, best_p


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(a*a).
    Parameters are the logits `x0` (so that a = softmax(x0) and Σa = 1).
    Returns the best probability vector found together with its *raw* score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_raw = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector
        a = _softmax(x)

        # raw objective for book‑keeping
        conv = _fft_self_convolve_full(a)
        cur_raw = 2.0 * n * conv.max()
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it:5d} raw={best_raw:.6f} λ={lam:.3f}")

        # smooth max (log‑sum‑exp)
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()                     # shape (2n‑1,)

        # gradient of smoothed objective w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain‑rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # schedule λ upwards
        lam = lam * lam_factor

        # occasional jitter to shake out of plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = _raw_score(best_a)

    return best_a, best_raw


# ----------------------------------------------------------------------
def _subgradient_refine(a: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑objective.
    Very cheap – a few thousand iterations are enough for the last digits.
    """
    best_a = a.copy()
    best_raw = _raw_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()

        # indices where the max is attained (within tolerance)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        # gradient of the true max
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # diminishing step size
        step = 0.07 / math.sqrt(it)

        a = a - step * grad
        a = _project_onto_simplex(a)

        # small random shake now and then
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it:7d} raw={best_raw:.6f}")

    return best_a, best_raw


# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------

    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None
