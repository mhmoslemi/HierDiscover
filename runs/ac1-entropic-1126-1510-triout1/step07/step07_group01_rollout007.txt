    n = a.shape[0]

"""
Propose a step‑function that (approximately) minimizes
evaluate_sequence().

Main ideas
---------
1. Generate a diverse collection of candidate shapes (analytical families,
   Beta‑distributions, random Dirichlet draws and a two‑spike pattern).
2. Keep the top few (≈12) seeds by the cheap proxy
   ``score = 2·n·max(a*a)`` when sum(a)=1.
3. Refine each seed with a *smooth‑max* annealing schedule.
   For a temperature α we minimise the soft‑max
   ``M_α(a) = (1/α)·log Σ exp(α·(a*a))`` using Adam on the simplex.
   α is increased (2 → 5 → 10 → 20 → 40 → 80) so the optimisation
   gradually focuses on the true maximum.
4. Finish refinement with a true sub‑gradient step (max‑convolution)
   again using Adam.
5. End with a cheap hill‑climbing mass‑transfer phase.

All operations stay on the probability simplex, therefore the
scale‑invariant objective of ``evaluate_sequence`` is unchanged.
The routine respects the 1000‑s budget and always returns a list of
non‑negative floats ready for evaluation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap proxy = 2·n·max(a*a) (assumes Σa = 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _edge_exponential_distribution(n: int, lam: float = 0.015) -> np.ndarray:
    """Large weights near the two ends, small in the middle."""
    idx = np.arange(n, dtype=np.float64)
    dist = np.exp(-lam * np.minimum(idx, n - 1 - idx))
    return dist / dist.sum()


# ----------------------------------------------------------------------
def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _two_spike_distribution(n: int) -> np.ndarray:
    """Two equal spikes at random positions (ensuring they differ)."""
    a = np.zeros(n, dtype=np.float64)
    i, j = np.random.choice(n, size=2, replace=False)
    a[i] = a[j] = 0.5
    return a


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Stable softmax of α·b (weights sum to 1)."""
    max_b = b.max()
    exp_term = np.exp(alpha * (b - max_b))
    return exp_term / exp_term.sum()


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, alpha: float) -> np.ndarray:
    """∇ of the soft‑max proxy (α‑smoothed max of a*a)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    w = _softmax_weights(b, alpha)
    conv_rev = _fft_convolve(a[::-1], w)
    return 2.0 * conv_rev[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
def _max_subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Uses an equal convex combination of all active constraints.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)
    if active.size == 0:
        w = np.zeros_like(b)
    else:
        w = np.zeros_like(b)
        w[active] = 1.0 / active.size        # equal weights -> convex combo
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_opt_smooth(a0: np.ndarray, deadline: float,
                     max_iters: int, step0: float,
                     alpha: float, beta1: float = 0.9,
                     beta2: float = 0.999):
    """Adam optimisation of the α‑smoothed proxy."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_gradient(a, alpha)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, 1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _adam_opt_subgrad(a0: np.ndarray, deadline: float,
                     max_iters: int, step0: float,
                     beta1: float = 0.9, beta2: float = 0.999):

    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
