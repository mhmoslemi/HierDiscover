def run_eg(a, deadline, initial_eta=0.3):
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    eta = initial_eta
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _, _ = _subgrad(a)   # subgrad
        # step
        a = a * np.exp(-eta * grad)
        # renorm
        a_sum = a.sum()
        if a_sum == 0: 
            a = np.full_like(a, 1.0/a.size)
        else:
            a /= a_sum
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            # optionally increase eta if big improvement?

        # jitter occasionally
        if it % 20000 == 0 and np.random.rand() < 0.2:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-6, 1.0)
    return best_a, best_score

"""
search for a low‑score non‑negative step function.

The algorithm builds several educated seeds (arcsine, beta‑U‑shape,
triangular, exponential, Dirichlet draws and, if available,
an up‑sampled previous best).  The best few seeds are then refined
with two complementary mirror‑descent schemes – Adam (additive) and
Exponentiated‑Gradient (multiplicative) – for as long as the time
budget allows.  A final Frank‑Wolfe style pairwise mass‑transfer
finetunes the solution.  All operations are performed in the
frequency domain (real FFT) for speed; the whole routine respects
the supplied time budget and always returns a simplex‑projected
vector ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """
    Surrogate score for a simplex‑normalised vector: 2·n·max(a*a).
    (equal to ``evaluate_sequence`` up to the normalisation factor.)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(a*a) at a (a lies on the simplex).
    Returns (gradient, current maximum, active indices of b).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (the term a_i*a_i appears only once)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with all concentration parameters = alpha."""
    if alpha <= 0:
        raise ValueError("alpha must be > 0")
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full_like(x, 1.0 / n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) density on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style mirror descent on the simplex (additive updates)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline:
        t += 1
        grad, _, _ = _subgrad(a)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best = a.copy()
    return best, best_score


# ----------------------------------------------------------------------
def _exp_grad_descent(a0: np.ndarray, deadline: float,
                      eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (KL‑mirror) descent."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _, _ = _subgrad(a)
        # Clip gradient to avoid overflow in the exponent
        grad = np.clip(grad, -5.0, 5.0)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-14:
            best_score = cur
            best = a.copy()
        # occasional tiny jitter to shake out of plateaus
        if it % 50000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(a.size) * 1e-6
            a = _project_onto_simplex(a + jitter, s=1.0)
    return best, best_score

    for alpha in (0.02, 0.05, 0.1, 0.3, 0.7):
        seeds.append(_random_dirichlet_alpha(n, alpha))

    # previous best (if present) – up‑sample / interpolate to the new length
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 1:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            interp = np.interp(x_new, x_old, prev)
            if interp.sum() == 0:
                interp = np.full_like(interp, 1.0 / n)
            else:
                interp = interp / interp.sum()
            seeds.append(interp)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Evaluate seeds, keep the top few
    scored = []
    for a in seeds:
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        sc = _max_convolution_score(a)
        scored.append((a, sc))

    scored.sort(key=lambda x: x[1])
    top_k = min(5, len(scored))
    top_candidates = scored[:top_k]

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]
    print(f"[{time.time():.1f}] initial best score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 4. Intensive refinement of the top seeds
    for idx, (seed_seq, seed_score) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] refining seed {idx + 1}/{top_k} (score={seed_score:.6f})")
        # split the remaining budget roughly equally among the seeds
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # give 60 % to Adam, 30 % to EG, 10 % spare for FW later
        adam_dead = time.time() + remaining * 0.55
        seq_adam, sc_adam = _adam_descent(seed_seq, adam_dead, step0=0.2)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}]   Adam improvement → {best_score:.6f}")

        eg_dead = time.time() + remaining * 0.35
        seq_eg, sc_eg = _exp_grad_descent(seed_seq, eg_dead, eta0=0.6)
        if sc_eg < best_score - 1e-14:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}]   EG improvement   → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final Frank‑Wolfe–style pairwise mass‑transfer
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    while time.time() < deadline and fw_iter < 200000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # mass to move away from
        i_dst = int(np.argmin(grad))   # mass to move towards
        if i_src == i_dst:
            break
        # maximum admissible transfer
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta < 1e-14:
            break

        # try a full transfer
        a_tmp = a.copy()
        a_tmp[i_src] -= delta
        a_tmp[i_dst] += delta
        sc_tmp = _max_convolution_score(a_tmp)

        improved = False
        if sc_tmp < cur_score - 1e-12:
            a = a_tmp
            cur_score = sc_tmp
            improved = True
        else:
            # try half transfer as a cheaper fallback
            half = delta * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            sc_half = _max_convolution_score(a_half)
            if sc_half < cur_score - 1e-12:
                a = a_half
                cur_score = sc_half
                improved = True

        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter {fw_iter} new best {best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if fw_iter % 4000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 1e-6
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    print(f"[{time.time():.1f}] final best score = {best_score:.6f} (length={len(best_seq)})")
    # Ensure the returned list respects the evaluation function's caps
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # tiny safeguard – sum must be > 0.01
    if best_seq.sum() < 0.01:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    return best_seq.tolist()
