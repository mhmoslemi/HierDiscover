"""
Propose a step function with a low evaluation score.

Algorithm
---------

* Work on the probability simplex via a soft‑max parametrisation
  a_i = exp(x_i) / Σ exp(x_i).  Hence Σ a_i = 1 and a_i ≥ 0 automatically.
* Minimise a *smooth* approximation of the max‑convolution:
        f_τ(a) = 2·n·(1/τ)·log Σ_k exp(τ·(a*a)_k) .
  For τ → ∞ the smooth value converges to the true objective.
* For each τ we solve the smooth problem with a limited L‑BFGS‑B run
  (no closures – a small callable class is used).  τ follows a
  continuation schedule 0.5 → … → 100, which gradually sharpens the
  surrogate.
* The best vector obtained for the current length is refined by a
  classic projected sub‑gradient method for the exact max‑convolution.
* We explore several lengths (from a few thousand up to a few tens of
  thousands) and several random initialisations (uniform, Dirichlet,
  spiky, triangular).  Warm‑starts are obtained by linear interpolation
  of the best‑known shape to the new length.
* All steps respect the overall time budget (≈1000 s).  The final
  sequence is returned as a plain Python list for the harness.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           max_iters: int = None) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for the exact max‑convolution.
    Returns (best_a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional tiny perturbation
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")
        if max_iters is not None and it >= max_iters:
            break
    return best_a, best_score

class _SmoothObjective:
    """
    Callable returning (value, gradient) of the τ‑smoothed max‑convolution
    objective for a given length n and temperature τ.
    """
    __slots__ = ("n", "tau")

    def __init__(self, n: int, tau: float):
        self.n = n
        self.tau = tau

    def __call__(self, x: np.ndarray):
        a = _softmax(x)                                    # Σa = 1
        conv = _fft_self_convolve_full(a)                  # length 2n‑1
        max_c = conv.max()
        # stable log‑sum‑exp
        diff = conv - max_c                                 # ≤ 0
        exp_scaled = np.exp(self.tau * diff)
        sum_exp = exp_scaled.sum()
        smooth = max_c + np.log(sum_exp) / self.tau
        f = 2.0 * self.n * smooth

        # weights for the smooth max (they sum to 1)
        w = exp_scaled / sum_exp
        # gradient w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[self.n - 1 : self.n - 1 + self.n]
        # chain rule through the soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        return f, grad_x


def _optimize_with_continuation(x0: np.ndarray,
                               n: int,
                               deadline: float) -> np.ndarray:
    """
    Runs a continuation of L‑BFGS‑B optimisations for a schedule of τ values.
    Returns a probability vector a (Σa = 1, a ≥ 0).
    """
    x = x0.copy()
    tau_schedule = [0.5, 2.0, 5.0, 10.0, 30.0, 100.0]
    for tau in tau_schedule:
        if time.time() >= deadline:
            break
        # modest number of L‑BFGS steps – enough for a decent descent
        max_iter = 200
        obj = _SmoothObjective(n, tau)
        try:
            res = minimize(obj, x, method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter,
                                    'ftol': 1e-12,
                                    'gtol': 1e-12,
                                    'disp': False})
            x = res.x
        except Exception as exc:  # pragma: no cover
            print("L‑BFGS failure:", exc)
            # keep the current x and continue
    a = _softmax(x)
    # tiny random jitter + projection to stay on the simplex
    a += np.random.randn(n) * 1e-7
    a = _project_onto_simplex(a)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point required by the challenge harness.
    Returns a list of non‑negative heights that (hopefully) yields a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0                     # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if it exists.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # Lengths to explore – we start small and gradually increase.
    lengths = [4000, 8000, 16000, 32000, 50000]

    best_seq = None
    best_score = float("inf")
    best_n = None

    # Baseline from a previous solution (if any)
    if prev is not None:
        raw = 2.0 * len(prev) * _fft_self_convolve_max(prev)
        print(f"Baseline (length={len(prev)}) raw score = {raw:.6f}")
        if raw < best_score:
            best_score = raw
            best_seq = prev.copy()
            best_n = len(prev)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")
        # ------------------------------------------------------------------
        # Assemble a collection of starting logits.
        init_logits = []

        # Warm‑start from the current global best (interpolated)
        if best_seq is not None:
            cand = _interpolate_to_length(best_seq, n)
            init_logits.append(np.log(cand + 1e-12))

        # Uniform start
        init_logits.append(np.zeros(n, dtype=np.float64))

        # Dirichlet draws – moderate concentration
        for _ in range(2):
            alpha = np.random.rand() * 4.0 + 0.1   # α ∈ (0.1, 4.1)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_logits.append(np.log(d + 1e-12))

        # Dirichlet draws – very spiky
        for _ in range(2):
            alpha = np.random.rand() * 0.09 + 0.001   # α ∈ (0.001, 0.091)
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            init_logits.append(np.log(d + 1e-12))

        # Simple triangular shape (decreasing to zero)
        tri = np.linspace(1.0, 0.0, n)
        tri = np.maximum(tri, 0.0)
        tri /= tri.sum()
        init_logits.append(np.log(tri + 1e-12))

        local_best_seq = None
        local_best_score = float('inf')

        # ------------------------------------------------------------------
        for idx, logits in enumerate(init_logits):
            if time.time() >= deadline:
                break
            print(f"[{time.time():.1f}] start {idx+1}/{len(init_logits)} for n={n}")

            # 1) L‑BFGS‑B with continuation (smooth max)
            a = _optimize_with_continuation(logits, n, deadline)

            # 2) Short exact‑max refinement (projected sub‑gradient)
            sub_deadline = min(deadline, time.time() + 25.0)   # give ≤25 s per candidate
            a, raw_score = _refine_by_subgradient(a, sub_deadline)

            if raw_score < local_best_score:
                local_best_score = raw_score
                local_best_seq = a.copy()
                print(f"   *** new local best for n={n}: raw={local_best_score:.6f}")

        # ------------------------------------------------------------------
        if local_best_seq is not None and local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.copy()
            best_n = n
            print(f">>> Updated global best: raw={best_score:.6f} (length={best_n})")

    # ------------------------------------------------------------------
    # Final refinement on the global best (time permitting)
    if best_seq is not None:
        a = best_seq.copy()
        a, final_raw = _refine_by_subgradient(a, deadline)
        best_seq = a.copy()
        best_score = final_raw
    else:
        # fallback – a modest uniform distribution
        n = 5000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
