"""
optimised_step_function.py

Search for a non‑negative step‑function (a probability vector) that
minimises ``evaluate_sequence``.  The algorithm works in four stages:

1. **Sampling of analytical families** – arcsine, beta, triangular,
   exponential, sine, cosine, harmonic, geometric and several random
   Dirichlet draws with a wide range of concentration parameters.
   Every candidate is normalised to sum = 1 and cheap‑scored by
   ``2·n·max(conv(a,a))`` (the exact objective for a normalised vector).

2. **Intensive mirror‑descent** – the best few candidates are refined by
   exponentiated‑gradient (KL‑mirror‑descent) and Adam optimisation on the
   simplex.  The optimiser works directly on the objective ``max(conv)``.

3. **Permutation / transfer hill‑climbing** – the currently best vector is
   further improved by a long greedy search that combines
   *mass transfers* (as in the reference implementation) and *pairwise
   swaps* of entries.  The search is performed until the time budget is
   exhausted, always keeping the best vector seen.

4. **Return** – the best probability vector (list of ``float``) is returned.
   ``evaluate_sequence`` is scale‑invariant, therefore the vector is
   already suitable for the final evaluation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a)  (assumes sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex.
    Returns grad, current maximum of the convolution and the active indices.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # mask of active constraints
    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient before diagonal correction
    conv_rev = np.fft.irfft(np.fft.rfft(a[::-1]) * np.fft.rfft(w))
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (only once for k = 2*i)
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.9):
    """Exponentiated‑gradient (KL‑mirror descent) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a_sum
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 150_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Dirichlet draw with concentration parameter alpha (alpha>0)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        x = np.full_like(x, 1.0 / n)
    else:
        x = x / x.sum()
    return x


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    return (vals / vals.sum())


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    return (vals / vals.sum())


def _harmonic_distribution(n: int) -> np.ndarray:
    """a_i ∝ 1/(i+1) (harmonic)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = 1.0 / i
    return vals / vals.sum()


def _geometric_distribution(n: int, r: float = 0.999) -> np.ndarray:
    """Geometric‑like decay."""
    i = np.arange(n, dtype=np.float64)
    vals = (1 - r) * (r ** i)
    return vals / vals.sum()


# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray, deadline: float) -> np.ndarray:
    """
    Greedy hill‑climbing that mixes mass‑transfer moves and pairwise swaps.
    The procedure runs until ``deadline`` and always keeps the best vector.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution_score(cur)
    best = cur.copy()
    best_score = cur_score
    it = 0
    # parameters controlling the size of mass transfers
    max_transfer = 0.04                       # up to 4 % of total mass initially
    scale = 10_000.0

    while time.time() < deadline:
        it += 1
        # exponential decay of transfer magnitude
        transfer = max_transfer * (0.5 ** (it / scale))

        if np.random.rand() < 0.55:          # ------- mass transfer -------
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            if cur[i] <= 0:
                continue
            delta = np.random.rand() * min(transfer, cur[i])
            if delta <= 0:
                continue
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        else:                                 # ------- pairwise swap -------
            i, j = np.random.choice(n, size=2, replace=False)
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]

        new_score = _max_convolution_score(cand)

        if new_score < cur_score - 1e-15:
            # accept improvement
            cur = cand
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best = cand.copy()
                best_score = new_score
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best {best_score:.6f}")

        # occasional jitter to avoid plateaus
        if it % 6_000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            cur = _project_onto_simplex(best + jitter, s=1.0)
            cur_score = _max_convolution_score(cur)

    return best

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence``.  The routine uses a mixture of analytic families,
    mirror‑descent optimisation and a long greedy hill‑climb.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 0. Determine a base length.
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    lengths = {base_n, int(base_n * 1.05), int(base_n * 0.95)}
    # add a few random lengths (between 500 and 1.5·base_n)
    for _ in range(5):
        lengths.add(np.random.randint(500, int(base_n * 1.5) + 1))
    lengths = sorted(lengths)

    # --------------------------------------------------------------
    # 1. Broad sampling of analytic families and random Dirichlet draws.
    cand_arrays = []   # list of (np.ndarray, cheap_score)

    beta_ps = [0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5,
               0.75, 1.0, 1.25, 1.5, 2.0, 3.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        families = [
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (lambda m: _exponential_distribution(m, lam=0.003), "exponential"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine"),
            (lambda m: np.full(m, 1.0 / m), "uniform"),
            (_harmonic_distribution, "harmonic"),
            (_geometric_distribution, "geometric")
        ]
        for gen, name in families:
            a = gen(n)
            score = _max_convolution_score(a)
            cand_arrays.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} {name:11s} score={score:.6f}")

        # beta‑symmetric family
        for p in beta_ps:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            cand_arrays.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} beta(p={p:4.2f}) score={score:.6f}")

        # Dirichlet draws with several concentration parameters
        for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            score = _max_convolution_score(a)
            cand_arrays.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} Dirichlet(alpha={alpha:.2f}) score={score:.6f}")

    # --------------------------------------------------------------
    # 2. Include the previously best sequence, if available.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            cand_arrays.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Keep the top‑k for heavy refinement.
    cand_arrays.sort(key=lambda x: x[1])          # cheap score ascending
    top_k = 20
    top_cands = cand_arrays[:top_k]
    print(f"--- phase 1 complete: {len(top_cands)} seeds selected (best cheap={top_cands[0][1]:.6f})")

    # --------------------------------------------------------------
    # 4. Heavy refinement (exp‑grad + Adam) on the selected seeds.
    best_seq = top_cands[0][0].copy()
    best_score = top_cands[0][1]

    for idx, (seed_seq, _) in enumerate(top_cands):
        if time.time() >= deadline:
            break

        # exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=80_000,
                                        eta0=0.9)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_cands)} exp‑grad improved to {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                max_iters=80_000,
                                step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_cands)} Adam improved to {sc:.6f}")

    # --------------------------------------------------------------
    # 5. Optional sub‑gradient refinement on the current best (time‑budget).
    subgrad_deadline = deadline - 30.0
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=400_000,
                                       eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient refined best to {sc:.6f}")

    # --------------------------------------------------------------
    # 6. Long hill‑climbing with swaps + transfers.
    final_seq = _hill_climb(best_seq, deadline)

    final_score = _max_convolution_score(final_seq)
    print(f"=== final cheap score: {final_score:.6f} (length={len(final_seq)})")
    # ``evaluate_sequence`` is scale‑invariant; returning the probability vector is enough.
    return final_seq.tolist()
