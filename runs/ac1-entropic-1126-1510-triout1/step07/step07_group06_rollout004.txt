    i = np.arange(n)
    eps = 1e-12

    # if p=0 yields uniform.
    # if p negative yields center heavier.
    # ensure positivity.
    vals = np.maximum(vals, 0.0)
    sum_ = vals.sum()
    if sum_ > 0:
        vals /= sum_
    return vals

def _power_shape(n: int, exponent: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    vals = 1.0 / (( (i + 0.5) * (n - 0.5 - i) ) + eps) ** exponent
    # For exponent small (<0) the denominator to negative power yields larger values near edges maybe huge, but okay.
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals

"""
A fast multi‑scale optimisation for the step‑function L∞ self‑convolution
problem.

The algorithm works in three stages:

1.  **Shape exploration** – for each candidate length `n` we sweep a
    one‑parameter family  a_i ∝ ((i+½)(n‑½‑i))⁻ᵖ  (the arcsine shape
    corresponds to p=0.5).  This inexpensive grid search often finds a very
    good starting point.

2.  **Smooth‑max Adam optimisation** – the probability vector
    a = softmax(z) (Σa=1, a≥0) minimises the log‑sum‑exp of the full
    self‑convolution.  Gradients are obtained by a single FFT per
    iteration.  An early‑stopping criterion prevents wasted work on very
    large `n`.

3.  **Projected sub‑gradient polishing** – once the smooth optimisation
    has converged we switch to the true max‑objective and perform a few
    hundred sub‑gradient steps (with Euclidean projection onto the simplex)
    to clean up the last digits.

The search proceeds over an increasing schedule of lengths
[2000, 5000, 10000, 20000, 40000].  Results are warm‑started from the
previous best sequence (interpolated to the new length).  The whole
procedure respects the supplied time budget (default 1000 s) and prints
intermediate progress.

The returned list can be fed directly to the supplied
`evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a different length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    # mid‑point positions of bins
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1, alpha_high: float = 4.0) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (edges heavier, centre lighter)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _power_shape(n: int, p: float) -> np.ndarray:
    """One‑parameter family a_i ∝ ((i+½)(n‑½‑i))⁻ᵖ  (p=0.5 → arcsine)."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    vals = denom ** (-p)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _search_power_shapes(
    n: int,
    deadline: float,
    exponents: np.ndarray,
) -> tuple[np.ndarray | None, float]:
    """Return the best power‑shape for the given length."""
    best_seq = None
    best_score = float("inf")
    for p in exponents:
        if time.time() >= deadline:
            break
        seq = _power_shape(n, float(p))
        conv = _fft_self_convolve_full(seq)
        cur_max = conv.max()
        score = 2.0 * n * cur_max
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
    early_stop_patience: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp smoothed max.
    Returns the best probability vector found and its raw (non‑smoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start
    no_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                     # Σa = 1, a ≥ 0

        # raw objective (exact max)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            # occasional progress print
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.3f}")
        else:
            no_improve += 1
            if no_improve >= early_stop_patience:
                # stop early – further progress unlikely
                break

        # smooth max via log‑sum‑exp
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()          # shape (2n‑1,)

        # gradient w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase λ slowly
        lam *= lam_factor

        # occasional jitter to escape plateaus
        if it % 800 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # Fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length `n`.
    Uses a quick parametric sweep followed by a few Adam runs.
    """
    # ----- parametric sweep -------------------------------------------------
    # exponents around the classic arcsine (p = 0.5)
    exponents = np.concatenate([
        np.linspace(0.10, 1.00, 10),   # coarse sweep
        np.linspace(0.30, 0.70, 9),    # finer near 0.5
        np.linspace(-0.20, 0.00, 5),   # allow slight centre‑heavy shapes
    ])

    power_seq, power_score = _search_power_shapes(n, deadline, exponents)

    # ----- build list of initial logits ------------------------------------
    init_logits = []

    # warm‑start from previous best (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        init_logits.append(np.log(cand + 1e-12))

    # start from the best parametric shape found above
    if power_seq is not None:
        init_logits.append(np.log(power_seq + 1e-12))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # deterministic arcsine shape
    init_logits.append(np.log(_arcsine_start(n) + 1e-12))

    # a few random Dirichlet draws
    for _ in range(2):
        a = _random_dirichlet_start(n)
        init_logits.append(np.log(a + 1e-12))

    # ----- Adam optimisation ------------------------------------------------
    # iteration budget adapts to problem size
    if n <= 5000:
        max_iters = 2500
    elif n <= 15000:
        max_iters = 1500
    else:
        max_iters = 800   # larger n – fewer passes

    best_local_seq = None
    best_local_score = float("inf")

    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, init {idx+1}/{len(init_logits)}")
        seq, score = _adam_optimize_smoothed(
            x0,
            n,
            deadline,
            max_iters=max_iters,
            lr0=0.10,
            lam_start=0.5,
            lam_factor=1.002,
            early_stop_patience=25,
        )
        if score < best_local_score:
            best_local_score = score
            best_local_seq = seq.copy()

    # safety fallback
    if best_local_seq is None:
        best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_local_score = 2.0 * n * _fft_self_convolve_full(best_local_seq).max()

    return best_local_seq, best_local_score
