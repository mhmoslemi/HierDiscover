def optimize_adam(a0, time_budget):
    # returns best sequence and raw.
    a = a0.copy()
    best_seq = a.copy()
    best_raw = _raw_score(a)
    # Adam hyperparameters
    lr = 0.5   # step size can be tuned
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # mu schedule
    mu = _max_self_convolution(a) * 0.5
    mu = max(mu, 1e-7)
    mu_target = 1e-8
    mu_factor = 0.5
    start = time.time()
    while True:
        # break if time insufficient
        if time.time() - start > time_budget:
            break
        # compute gradient for current mu
        f_val, grad = _smooth_obj_and_grad(a, mu)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # project onto simplex
        a_new = _project_onto_simplex(a_new)
        # evaluate raw occasionally
        cur_raw = _raw_score(a_new)
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_seq = a_new.copy()
        a = a_new
        # schedule update mu maybe every few iterations, or based on improvement
        if t % 200 == 0:
            if mu > mu_target:
                mu = max(mu * mu_factor, mu_target)
            else:
                # maybe stop earlier
                pass
    return best_seq, best_raw

"""
Improved optimizer for the step‑function max‑convolution problem.

Strategy
--------
1. Build a large pool of seed probability vectors.
   • Uniform, arcsine, a dense grid of symmetric and asymmetric Beta PDFs,
     random Dirichlet draws and the supplied `height_sequence_1`.
2. Keep the best few seeds according to the cheap proxy
   `2·n·max(a*a)` (the objective when the vector is already ℓ₁‑normalised).
3. Refine each seed with a continuation method:
   minimise the smooth max–convolution
        μ·log Σ exp( (a*a)_k / μ )
   using Adam (momentum) + Euclidean projection onto the simplex.
   μ is halved repeatedly until ≈1e‑8.
4. After the continuation stages run a short projected sub‑gradient
   descent (exact max) for polishing.
5. Return the best sequence found (the harness rescales it automatically).

All heavy work is performed with FFT‑based convolutions, which are
~10⁻³ s for n≈8000, allowing many thousands of optimisation steps
within the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, ℓ₁‑normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf is more stable for very small α/β
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # Guard against overflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Cheap proxy (exact when Σa=1): 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution via log‑sum‑exp.
    Returns (value, gradient) where gradient has the same shape as a.
    """
    conv = _self_convolution_full(a)                 # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)                     # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                               # soft‑max weights, Σp = 1

    # gradient = 2·(a_rev * p)   (central n entries needed)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation + Adam optimiser on the smoothed objective.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_raw = _raw_score(a)

    # Adam hyper‑parameters
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule
    mu = _max_self_convolution(a) * 0.5
    mu = max(mu, 1e-7)
    mu_target = 1e-8
    mu_factor = 0.5

    t = 0
    start = time.time()
    while True:
        # time check
        if time.time() - start > time_budget:
            break

        # compute gradient for current μ
        f_val, grad = _smooth_obj_and_grad(a, mu)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_onto_simplex(a_new)

        cur_raw = _raw_score(a_new)
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_seq = a_new.copy()

        a = a_new

        # occasional μ reduction (every few hundred iterations)
        if t % 300 == 0 and mu > mu_target:
            mu = max(mu * mu_factor, mu_target)

    return best_seq, best_raw


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent on the true max‑convolution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mom_coef = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mom_coef * momentum + (1 - mom_coef) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()

    return best_seq, best_val
