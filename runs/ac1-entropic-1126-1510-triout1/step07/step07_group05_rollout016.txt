   def projected_gradient_descent(a, deadline, max_iters=2000):
       best = a.copy()
       best_score = raw_score(a)
       it = 0
       while time.time() < deadline and it < max_iters:
            it += 1
            # compute conv and gradient
            conv = fft_self_convolve_full(a)
            maxc = conv.max()
            # tolerance for active set:
            tol = maxc * 1e-12
            active = np.where(conv >= maxc - tol)[0]
            w = np.zeros_like(conv)
            if active.size:
                w[active] = 1.0 / active.size
            # gradient
            a_rev = a[::-1]
            grad = 2.0 * _fft_convolve(a_rev, w)[len(a)-1 : len(a)-1 + len(a)]
            # step size
            # compute avg conv as sum(conv)/len(conv)
            avgc = conv.mean()
            # compute base step
            step_base = (maxc - avgc) / (np.linalg.norm(grad) ** 2 + 1e-12)
            # scale factors
            # try scaling values in [0.5, 1.0, 2.0]
            best_local = None
            best_local_score = float('inf')
            a_candidate = None
            for scale in (0.5, 1.0, 2.0):
                step = scale * step_base
                a_new = a - step * grad
                a_new = project_onto_simplex(a_new)
                score = raw_score(a_new)
                if score < best_local_score:
                    best_local_score = score
                    a_candidate = a_new
            # update a
            a = a_candidate
            # update best global
            if best_local_score < best_score:
                best_score = best_local_score
                best = a.copy()
                # optionally print progress
                print(f"[{time.time():.1f}] it={it} raw={best_score:.6f}")
            # occasional jitter
            if it % 200 == 0:
                a = a + np.random.randn(len(a))*1e-4
                a = project_onto_simplex(a)
       return best, best_score

"""
Hybrid optimisation for the step–function attack.

The algorithm proceeds in three phases for a chosen length `n`:

1. **Initialisation** – a handful of diverse starting points on the simplex
   (uniform, warm‑started previous best, random Dirichlet draws and random
   logits).

2. **Smooth max minimisation** – Adam optimisation on a soft‑max
   parametrisation of the sequence, using a log‑sum‑exp surrogate for the
   max‑convolution.  The temperature is ramped up to a large value
   (`tau_max≈800`) so that the surrogate closely follows the true max.

3. **Exact refinement** – projected sub‑gradient descent on the true
   max‑convolution, followed by a short stochastic hill‑climb.  Both steps
   keep the sequence on the probability simplex.

The routine adapts its running time to the provided budget (default 1000 s)
and returns the best sequence it discovers.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) (sum(a)=1 ⇒ same as evaluate_sequence)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 20000,
                            tau0: float = 0.5,
                            tau_max: float = 800.0,
                            lr0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max parametrisation.
    Returns the best sequence (on the simplex) and its raw score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # --- soft‑max to enforce simplex ---
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # --- compute convolution and current raw score ---
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        # temperature schedule (linear)
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # smooth‑max weights (log‑sum‑exp)
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # gradient of the smooth max w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # chain rule through the soft‑max
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat regions
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of  max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scaled with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0,
                           step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for the exact max‑convolution.
    Returns the refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _raw_score(a)
    it = start_iter

    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = step0 / math.sqrt(it)          # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a)

        # tiny random perturbation every few hundred iterations
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _hill_climb(a: np.ndarray,
                deadline: float,
                sigma_init: float = 0.01,
                max_trials: int = 200000) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climb on the simplex."""
    n = a.shape[0]
    best = a.copy()
    best_score = _raw_score(best)
    sigma = sigma_init
    it = 0

    while time.time() < deadline and it < max_trials:
        it += 1
        cand = best + sigma * np.random.randn(n)
        cand = _project_onto_simplex(cand)
        s = _raw_score(cand)
        if s < best_score:
            best = cand
            best_score = s
            sigma = sigma_init                 # reset step size
            print(f"[{time.time():.1f}] hill‑climb it={it} raw={best_score:.6f}")
        else:
            sigma *= 0.9995                    # slowly decay

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the challenge harness.
    Returns a list of non‑negative heights that (hopefully) gives a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Determine a working length. Larger ``n`` gives a richer space but
    # costs a bit more per iteration.
    if budget_s > 600:
        n = 8000
    else:
        n = 4000
    print(f"Optimising sequences of length n={n}")

    # --------------------------------------------------------------
    # Warm‑start from a previously found construction if it exists.
    prev_seq = None
    try:
        # ``height_sequence_1`` is injected by the harness if available.
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev_seq = prev / s
    except Exception:
        prev_seq = None

    # --------------------------------------------------------------
    # Build a small pool of diverse initial logits (log‑softmax params).
    init_logits = []

    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        init_logits.append(np.log(cand + 1e-12))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # Dirichlet draws with various concentrations
    for alpha in [0.05, 0.2, 1.0, 5.0]:
        d = np.random.dirichlet(np.full(n, alpha))
        init_logits.append(np.log(d + 1e-12))

    # a few random normal logits (purely exploratory)
    for _ in range(2):
        init_logits.append(np.random.randn(n) * 0.5)

    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    for idx, logits in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"\n--- Starting restart {idx + 1}/{len(init_logits)} ---")

        # 1) Adam optimisation with a very high temperature schedule
        seq, raw = _adam_optimize_smoothed(
            logits,
            n,
            deadline,
            max_iters=20000,
            tau0=0.5,
            tau_max=800.0,
            lr0=0.12,
        )

        # 2) Exact sub‑gradient refinement
        seq, raw = _refine_by_subgradient(seq, deadline, step0=0.5)

        if raw < best_score:
            best_score = raw
            best_seq = seq.copy()
            print(f">>> New global best (restart {idx + 1}): raw={best_score:.6f}")

    # --------------------------------------------------------------
    # Final stochastic hill‑climb if we still have time left.
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final hill‑climb phase ---")
        best_seq, best_score = _hill_climb(best_seq, deadline,
                                           sigma_init=0.01,
                                           max_trials=200000)

    # --------------------------------------------------------------
    # Fallback in the extremely unlikely event that nothing was found.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_seq)

    # --------------------------------------------------------------
    final_list = best_seq.tolist()
    final_eval = evaluate_sequence(final_list)  # provided by the harness
    print(f"\nFinal evaluated score: {final_eval:.6f} (length={len(final_list)})")
    return final_list
