"""
Optimise a non‑negative step function (probability vector) to minimise
the black‑box ``evaluate_sequence`` score.

Main ideas
----------
* Parameterise the step heights by logits → softmax, which enforces the
  simplex constraints automatically.
* Minimise a smooth approximation of the max‑convolution
  (log‑sum‑exp) using Adam.  Several hyper‑parameter settings,
  initialisations and random restarts are tried.
* Warm‑start larger problems by linear interpolation of the best
  solution found for a smaller length.
* After the Adam stage a cheap projected (sub‑)gradient descent and a
  short simulated‑annealing phase are applied to polish the solution.
* The whole search respects the given time budget (default 1000 s).

The returned list is a normalised probability vector; ``evaluate_sequence``
is scale‑invariant, so any positive scaling would give the same score.
"""

import time
import math
import numpy as np
from typing import List, Tuple, Optional

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (outputs a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1, alpha_high: float = 4.0) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic U‑shaped arcsine‑like shape."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(
    logits0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth log‑sum‑exp objective.
    Returns the best probability vector (sum = 1) and its *raw* score
    (2·n·max(conv)).
    """
    x = logits0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a: Optional[np.ndarray] = None
    best_raw = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector on the simplex
        a = _softmax(x)

        # raw objective (exact)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_a = a.copy()
            # log progress – the time stamps are useful for debugging
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_raw:.6f} λ={lam:.3f}")

        # smooth approximation (log‑sum‑exp) and gradient
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)              # shift for stability
        p = exp_tmp / exp_tmp.sum()                  # softmax weights, shape (2n‑1,)

        # ∇_a f_λ = 2 * (a_rev * p)  – we need only the central n entries
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain‑rule through softmax: ∂x = a * (grad_a – <a,grad_a>)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase smoothness
        lam = lam * lam_factor

        # occasional jitter (helps to escape plateaus)
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_full(best_a).max()
    return best_a, best_raw


# ----------------------------------------------------------------------
def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: Optional[np.ndarray] = None,
) -> Tuple[np.ndarray, float]:
    """
    Optimise a step function of length ``n``.
    Several initialisations and hyper‑parameter configurations are tried.
    The best probability vector (sum = 1) and its raw score are returned.
    """
    # ---------- build initial logit vectors ----------
    init_logits: List[np.ndarray] = []

    # warm start (interpolated from previous best)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        init_logits.append(np.log(cand + 1e-12))

    # plain uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # a couple of Dirichlet draws
    for _ in range(3):
        a = _random_dirichlet_start(n)
        init_logits.append(np.log(a + 1e-12))

    # deterministic arcsine shape
    init_logits.append(np.log(_arcsine_start(n) + 1e-12))

    # ---------- search over starts & Adam hyper‑parameters ----------
    best_seq: Optional[np.ndarray] = None
    best_score = float("inf")

    # Helper to run Adam with given hyper‑parameters
    def run_adam(logits, max_iters, lr0, lam_start, lam_factor):
        return _adam_optimize_smoothed(
            logits,
            n,
            deadline,
            max_iters=max_iters,
            lr0=lr0,
            lam_start=lam_start,
            lam_factor=lam_factor,
        )

    # First pass over the deterministic starts
    for idx, logits0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, deterministic start {idx+1}/{len(init_logits)}")
        seq, score = run_adam(
            logits0,
            max_iters=5000,
            lr0=0.12,
            lam_start=0.5,
            lam_factor=1.001,
        )
        if score < best_score:
            best_score, best_seq = score, seq.copy()

    # Random restarts with varied hyper‑parameters until time runs out
    while time.time() < deadline - 2.0:
        # random initial logits (Gaussian)
        logits0 = np.random.randn(n) * 0.5
        # random hyper‑parameters
        max_iters = int(np.random.uniform(3000, 6000))
        lr0 = 0.08 + 0.1 * np.random.rand()          # 0.08 … 0.18
        lam_start = 10 ** np.random.uniform(-1, 0)   # 0.1 … 1.0
        lam_factor = 1.001 + 0.009 * np.random.rand()  # 1.001 … 1.010
        seq, score = run_adam(logits0, max_iters, lr0, lam_start, lam_factor)
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"[{time.time():.1f}] length {n} new overall best raw={best_score:.6f}")

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_seq).max()

    return best_seq, best_score

def _subgradient_refinement(a: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns an improved vector and its raw score.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_full(best).max()
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        # active set (indices attaining the max, up to numerical tolerance)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        # sub‑gradient of the exact objective
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.15 / math.sqrt(it)          # a slightly larger step than before
        a = a - step * grad
        a = _project_onto_simplex(a)

        # occasional tiny random perturbation
        if it % 600 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score, best = cur_score, a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} raw={best_score:.6f}")

    return best, best_score

def _simulated_annealing(a0: np.ndarray, deadline: float) -> Tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing on the simplex (logit space).
    Accepts worse moves with a temperature that decays exponentially.
    """
    n = a0.shape[0]
    # work in logit space to keep the simplex constraint cheap
    x_cur = np.log(a0 + 1e-12)
    a_cur = _softmax(x_cur)
    cur_score = 2.0 * n * _fft_self_convolve_full(a_cur).max()
    best_a, best_score = a_cur.copy(), cur_score

    temp = 0.02            # initial temperature
    decay = 0.9995         # exponential decay factor

    while time.time() < deadline:
        # propose a new point by Gaussian jitter in logit space
        x_new = x_cur + np.random.randn(n) * (temp * 0.6)
        a_new = _softmax(x_new)
        new_score = 2.0 * n * _fft_self_convolve_full(a_new).max()

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / max(temp, 1e-9))
            if np.random.rand() < prob:
                accept = True

        if accept:
            x_cur, a_cur, cur_score = x_new, a_new, new_score
            if cur_score < best_score:
                best_score, best_a = cur_score, a_cur.copy()
                print(f"[{time.time():.1f}] SA new best raw={best_score:.6f}")

        temp *= decay
        if temp < 1e-8:
            temp = 1e-8

    return best_a, best_score
