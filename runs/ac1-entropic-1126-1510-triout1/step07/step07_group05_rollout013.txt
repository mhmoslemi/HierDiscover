"""
Propose a step function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
* Load the previously best known sequence if it exists.
* Work on the probability simplex (a_i ≥ 0, Σ a_i = 1) – scaling does not change
  the score, so we keep the sum equal to one throughout the search.
* For a sequence length n we run three phases:
    1. **Adam on a smooth‑max surrogate** – the objective
       2·n·max(a*a) is replaced by a temperature‑controlled log‑sum‑exp.
       The soft‑max parametrisation a = softmax(logits) guarantees feasibility.
    2. **Exact sub‑gradient descent** – after the smooth phase we
       switch to the true max‑convolution and perform a long projected
       sub‑gradient descent (the implementation from the reference solution).
    3. **Gradient‑guided mass transfers** – the sub‑gradient gives a
       direction indicating which coordinates increase the current maximal
       convolution.  We repeatedly move a tiny amount of probability mass
       from the index with the largest gradient to the index with the
       smallest gradient; this is a cheap “mirror‑descent” step that often
       pushes the max a little lower after the plateau of the pure sub‑gradient
       method.
* The optimisation is performed on a schedule of increasing lengths
  (4000 → 8000 → 12000 …).  After each length we interpolate the current
  best distribution to the new size (preserving the shape) and repeat the
  three phases.
* Finally a short random‑perturbation hill‑climb is executed to squeeze out
  any remaining improvement.
* The best sequence found is returned; the harness will evaluate it with
  ``evaluate_sequence``.
"""

import math

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    The gradient corresponds to the average over all active (maximal) shifts.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 8000,
                            tau0: float = 0.1,
                            tau_max: float = 200.0,
                            lr0: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parametrisation with a log‑sum‑exp
    surrogate for the convolution supremum.
    Returns the best sequence encountered (sum = 1) and the *raw* score
    2·n·max(a*a).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # ---- soft‑max (stable) ----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # ---- convolution & raw score ----
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[ {time.time():.1f} ] n={n} it={it}  raw={best_score:.6f}")

        # ---- temperature schedule (linear) ----
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # ---- smooth‑max weights (stable softmax) ----
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # ---- gradient of the smoothed max w.r.t. a ----
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ---- chain rule through the soft‑max ----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for the exact max‑convolution.
    Returns refined (a, raw_score) where raw_score = 2·n·max(a*a).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)
    it = start_iter
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)
        # occasional tiny perturbation
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
def _gradient_transfer_refine(a: np.ndarray,
                             deadline: float,
                             max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Very cheap refinement: move a tiny amount of probability mass from the
    index with the largest sub‑gradient to the one with the smallest.
    This often reduces the maximal convolution after the heavy sub‑gradient
    phase has stalled.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(best_a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max = _max_subgradient_step(best_a)

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if i_max == i_min:
            break

        # amount to transfer – shrink as we progress
        eps = min(best_a[i_max] * 0.4, 0.001 / math.sqrt(it))
        if eps <= 0:
            break

        new_a = best_a.copy()
        new_a[i_max] -= eps
        new_a[i_min] += eps
        new_a = _project_onto_simplex(new_a)

        new_raw = 2.0 * n * _fft_self_convolve_max(new_a)
        if new_raw < best_raw - 1e-12:
            best_a = new_a
            best_raw = new_raw
            if it % 500 == 0:
                print(f"[grad‑refine it={it}] raw={best_raw:.6f}")
        # if no improvement we keep the old vector (simple hill‑climb)
    return best_a, best_raw


# ----------------------------------------------------------------------
def _random_local_search(a: np.ndarray,
                         deadline: float,
                         max_iters: int = 3000,
                         sigma_start: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Small stochastic perturbations of the current best vector.
    Each perturbation is multiplicative (to stay on the simplex after renorm).
    Accepted only if the raw score improves.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(best_a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # slowly decay the perturbation magnitude
        sigma = sigma_start / math.sqrt(it)
        perturb = np.exp(sigma * np.random.randn(n))
        cand = best_a * perturb
        cand = _project_onto_simplex(cand)
        cand_raw = 2.0 * n * _fft_self_convolve_max(cand)
        if cand_raw < best_raw - 1e-12:
            best_a = cand
            best_raw = cand_raw
            print(f"[rand‑search it={it}] raw={best_raw:.6f}")
    return best_a, best_raw


# ----------------------------------------------------------------------
def _run_optimisation_on_length(base_seq: np.ndarray,
                                n: int,
                                deadline: float) -> tuple[np.ndarray, float]:
    """
    Optimises a sequence of length n using the three phases described
    in the module docstring.
    """
    seq = _interpolate_to_length(base_seq, n)

    # --- Phase 1 : Adam on smooth‑max -------------------------------
    logits = np.log(seq + 1e-12)
    seq, raw = _adam_optimize_smoothed(
        logits, n, deadline, max_iters=8000,
        tau0=0.1, tau_max=200.0, lr0=0.20
    )
    if time.time() >= deadline:
        return seq, raw

    # --- Phase 2 : exact sub‑gradient descent -----------------------
    seq, raw = _refine_by_subgradient(seq, deadline)
    if time.time() >= deadline:
        return seq, raw

    # --- Phase 3 : gradient‑guided mass transfers -------------------
    seq, raw = _gradient_transfer_refine(seq, deadline)
    if time.time() >= deadline:
        return seq, raw

    # --- Phase 4 : tiny random hill‑climb --------------------------
    seq, raw = _random_local_search(seq, deadline)
    return seq, raw

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point required by the challenge harness.
    Returns a list of non‑negative heights that (hopefully) gives a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Try to reuse a previously discovered construction if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # --------------------------------------------------------------
    # start with a reasonable length – 4000 is the size used in the
    # reference run and gives a good trade‑off between quality and speed
    n0 = 4000
    if prev is not None and len(prev) >= n0:
        best_seq = prev[:n0].copy()
    else:
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)

    best_raw = 2.0 * n0 * _fft_self_convolve_max(best_seq)
    print(f"Initial raw score (length={n0}) = {best_raw:.6f}")

    # --------------------------------------------------------------
    # Length schedule – gradually increase the support,
    # always warm‑starting from the current best.
    lengths = [4000, 8000, 12000, 20000]   # the last one may be skipped if time runs out

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n--- Optimising length {n} ---")
        best_seq, best_raw = _run_optimisation_on_length(best_seq, n, deadline)
        print(f">>> After length {n}: raw={best_raw:.6f}")

    # --------------------------------------------------------------
    # Final polish – a short extra gradient‑transfer refinement (if time)
    if time.time() < deadline:
        best_seq, best_raw = _gradient_transfer_refine(best_seq, deadline)

    # --------------------------------------------------------------
    # Return the sequence in plain Python list form.
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (raw≈{best_raw:.6f}, length={len(final_list)})")
    return final_list
