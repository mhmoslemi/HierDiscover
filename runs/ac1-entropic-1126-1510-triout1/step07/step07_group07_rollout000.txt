"""
Propose a step‑function sequence with a low evaluate_sequence score.

Algorithm
---------
1.   Choose a large length `n` (20000).  A longer discretisation of the
     optimal continuous arcsine density reduces the discretisation error
     and thus the final bound.

2.   Build a modest pool of candidate probability vectors (uniform,
     arcsine β(0.5,0.5), several symmetric β‑distributions, a few random
     Dirichlet draws and, if present, the supplied `height_sequence_1`).
     All vectors are normalised to have L¹‑norm = 1.

3.   Pick the candidate with the smallest raw objective
     `2·n·max(a*a)`.

4.   Refine the best candidate with a continuation method:
     *Smooth maximum*: `F_μ(a)= μ·log Σ exp((a*a)_k/μ)`.  For a fixed `μ`
     we minimise `F_μ` by projected (accelerated) gradient descent
     (FISTA).  The gradient is obtained via FFT‑based convolutions.
     The smoothing parameter is halved repeatedly until `μ≈10⁻⁶`.

5.   After the continuation stage, perform a short projected sub‑gradient
     descent to polish the last digits.

6.   Finally, try a few random perturbations of the current best vector
     (Gaussian noise, re‑normalisation) and run a brief continuation on
     each to escape possible local minima.

7.   Return the best vector found; `evaluate_sequence` rescales it
     automatically.

All arithmetic is performed in double precision; projection onto the
simplex guarantees feasibility (non‑negative entries summing to 1).  The
implementation respects the 1000 s time budget, repeatedly checking the
remaining time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete β(α,β) density on n bins, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf of Beta, computed in a stable way
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # Guard against overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector:
       2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights

    # Gradient: 2·(a_rev * p)  – keep only the central n entries
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # Smoothed objective value
    f_val = M + mu * np.log(Z)
    return f_val, grad

def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Refine a probability vector with continuation + FISTA.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    mu = max(_max_self_convolution(a) * 0.5, 1e-7)
    mu_target = 1e-6
    mu_factor = 0.5                # more aggressive reduction
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # ----- inner FISTA for the current μ -----------------------------
        L = max(1.0, 2.0 / mu)      # safe Lipschitz estimate

        y = a.copy()
        t = 1.0

        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        # Allow up to ~5000 iterations per μ‑stage (adjusted to time left)
        iters = int(min(5000, remaining / 0.0012))
        if iters <= 0:
            break

        for _ in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # back‑tracking line‑search (max two doublings of L)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2.0 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional evaluation of the true (non‑smoothed) objective
            if (_ & 31) == 0:               # every 32 iterations
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()

        # ---------------------------------------------------------------
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        # Warm‑restart from the best sequence at the new μ
        a = best_seq.copy()

    return best_seq, best_val

def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional jitter to escape flat plateaus

            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val
