"""
Propose a step‑function (non‑negative vector) that minimizes the
evaluate_sequence objective.

The algorithm proceeds in several stages:

1.  Initialise many candidate vectors using analytic families
    (uniform, arcsine, beta‑sym, exponential, cosine‑window, …) and
    random Dirichlet draws for several lengths.  The best of these
    initialise the current global best.

2.  For the most promising initial vectors run a specialised
    hill‑climber that works directly on the *linear* self‑convolution.
    The move is a tiny transfer of mass from a “source’’ index to a
    “target’’ index.  The convolution array is updated in O(n) time,
    the current maximum is examined, and any improving move is kept.
    If no improvement occurs for a while the step size is reduced and
    a small random jitter is added to escape plateaux.

3.  With the time that remains a number of random restarts are performed:
    a fresh random Dirichlet vector is generated, hill‑climbed and, if
    better, stored as the new global optimum.

4.  The previously discovered sequence (``height_sequence_1``) is also
    refined, so the search never forgets a good starting point.

All loops respect the supplied time budget (default 1000 s) – the
function always returns the best vector found before the deadline.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating both `a`
    and its full self‑convolution array `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

def _hillclimb_local(a, deadline, max_iters=200_000):
    """
    Deterministic hill‑climber that works directly on the convolution.
    Returns the best vector found (sum = 1) and its score.
    """
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    max_delta_factor = 0.5          # start with a relatively large step size
    it_no_improve = 0
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # 1) locate the shift where the convolution attains its maximum
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # 2) among the contributing pairs (i,j) with i+j=k find the one
        #    with the largest product a[i]*a[j] – this pair is the most
        #    responsible for the peak.
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # 3) move a small amount of mass from the larger partner of that pair
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue

        # propose a transfer
        max_delta = min(a[source], max_delta_factor)
        if max_delta <= eps:
            continue
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        # keep snapshots to be able to revert a non‑improving move
        a_snapshot = a.copy()
        conv_snapshot = conv.copy()

        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        new_score = 2.0 * n * new_M

        if new_score < best_score - 1e-12:
            # accept improvement
            best_score = new_score
            best_a = a.copy()
            M = new_M
            it_no_improve = 0
        else:
            # revert the move
            a = a_snapshot
            conv = conv_snapshot
            it_no_improve += 1

        # if we have been stuck for a while, shrink the step size and shake the vector
        if it_no_improve > 2000:
            max_delta_factor *= 0.5
            it_no_improve = 0
            jitter = (np.random.rand(n) - 0.5) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            max_delta_factor = min(max_delta_factor * 2, 0.5)

        if max_delta_factor < 1e-7:
            break

    return best_a, best_score

                best_n = n
                print(f"[{time.time():.1f}] n={n} init new best {sc:.6f}")

        # ------------------------------------------------------------------
        # Pick the two most promising for a short hill‑climb.
        scored = [(v, _max_convolution_score(v)) for v in candidates]
        scored.sort(key=lambda x: x[1])
        top_vecs = [v for v, _ in scored[:2]]

        for vec in top_vecs:
            if time.time() >= deadline:
                break
            refined_vec, refined_score = _hillclimb_local(vec.copy(), deadline,
                                                         max_iters=150_000)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined_vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] hill‑climb n={n} new best {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Random restarts on the currently best length (or a default length)
    while time.time() < deadline - 5.0:
        n = best_n if best_n is not None else 5000
        a = np.random.rand(n)
        a = np.maximum(a, eps)
        a /= a.sum()
        refined_vec, refined_score = _hillclimb_local(a, deadline, max_iters=120_000)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_vec.copy()
            best_n = n
            print(f"[{time.time():.1f}] random‑restart new best {refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final sanity check with a fresh convolution (full FFT) and return.
    if best_seq is None:
        # extreme fallback – should never happen
        best_seq = _distribution_uniform(2000)
        best_n = 2000
        best_score = _max_convolution_score(best_seq)

    final_conv = _fft_self_convolve(best_seq)
    final_score = 2.0 * best_n * final_conv.max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_seq.tolist()
