def _smooth_grad(a, epsilon):
    b = _fft_self_convolve(a)
    maxb = b.max()
    # w_i = exp((b_i - maxb)/epsilon)
    w = np.exp((b - maxb) / epsilon)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[len(a)-1 : len(a)-1 + len(a)]
    # subtract diagonal correction
    # even indices in b (i even) -> diag positions 2*k = i -> k = i//2
    even = (np.arange(b.size) % 2 == 0)   # or using range
    diag_idx = np.arange(b.size)[even] // 2
    grad[diag_idx] -= w[2*diag_idx] * a[diag_idx]
    return grad

def _smooth_adam_descent(a0, deadline, max_iters=150_000, step0=0.2, beta1=0.9, beta2=0.999, epsilon=0.001):

   a = a0.copy()
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   best_a = a.copy()
   best_score = _max_convolution_score(a)   # cheap score

   t = 0
   while time.time() < deadline and t < max_iters:
       t += 1
       grad = _smooth_grad(a, epsilon)
       # maybe decrease epsilon gradually
       if t % 50_000 == 0: epsilon = epsilon * 0.5
       # update as Adam
       m = beta1 * m + (1.0 - beta1) * grad
       v = beta2 * v + (1.0 - beta2) * (grad ** 2)
       m_hat = m / (1.0 - beta1 ** t)
       v_hat = v / (1.0 - beta2 ** t)
       eta = step0 / (np.sqrt(v_hat) + 1e-8)
       a = a - eta * m_hat
       a = _project_onto_simplex(a, s=1.0)
       cur = _max_convolution_score(a)
       if cur < best_score - 1e-15:
           best_score = cur

"""
Propose a step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------

1. **Broad sampling** – generate many candidate probability vectors
   (analytic families: arcsine, beta, triangular, exponential,
   Gaussian, Cauchy, semicircle, uniform; random Dirichlet draws with
   several concentration parameters).  All candidates are normalised
   to sum = 1 and scored by the cheap quantity
   ``2·n·max(conv)`` (identical to ``evaluate_sequence`` for sum = 1).

2. **Select seeds** – keep the `top_k` candidates with smallest cheap
   scores.

3. **Heavy refinement** – for every seed run three first‑order
   optimisation schemes (exponentiated‑gradient, Adam, and a *smooth*
   Adam that minimises a log‑sum‑exp approximation of the max‑convolution).
   Each routine respects the global deadline and stops early if time runs out.

4. **Long sub‑gradient descent** – allocate the bulk of the remaining run‑time
   to a plain projected sub‑gradient method (the exact sub‑gradient of the
   max‑convolution) with a diminishing step size.

5. **Fine‑grained hill‑climbing** – use tiny mass‑transfer moves that are
   guided by the current sub‑gradient (move mass from the index with the
   largest positive gradient to the one with the most negative gradient)
   together with occasional random jitter.

The whole procedure works on the probability simplex, therefore the
objective is scale‑invariant and the returned list can be fed directly to
``evaluate_sequence``.  All operations are vectorised and use FFT‑based
convolutions for speed.

The function respects the 1000 s budget (a small safety margin is
subtracted) and always returns a feasible sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, v.size + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap score = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where i = 2*k
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _smooth_grad(a: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed max using log‑sum‑exp:
    f_eps(a) = eps * log Σ_i exp(conv_i/eps).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)          # length 2n‑1
    maxc = conv.max()
    # weights for the softmax
    w = np.exp((conv - maxc) / eps)
    w_sum = w.sum()
    w /= w_sum

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction: each diagonal entry (i=2k) was counted twice
    diag_indices = np.arange(conv.size)[(np.arange(conv.size) % 2) == 0] // 2
    grad[diag_indices] -= w[2 * diag_indices] * a[diag_indices]

    return grad


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 0.9, decay: float = 0.9999):
    """Mirror descent with KL‑divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.25,
                  beta1: float = 0.9, beta2: float = 0.999):
    """Adam‑style adaptive gradient optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_adam_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 200_000,
                         eps0: float = 1e-3,
                         step0: float = 0.25,
                         beta1: float = 0.9, beta2: float = 0.999):
    """Adam optimiser on the log‑sum‑exp smoothing of max(conv)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    t = 0
    eps = eps0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad = _smooth_grad(a, eps)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        # occasionally tighten the smoothing
        if t % 50_000 == 0:
            eps = eps * 0.5 if eps > 1e-6 else eps

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

def _subgradient_descent(a0: np.ndarray, deadline: float,
                         max_iters: int = 1_200_000,
                         eta0: float = 0.6):
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score, best_a = cur, a.copy()
    return best_a, best_score

# Analytic families -------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    return w / w.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _gaussian_distribution(n: int, sigma_frac: float = 0.30) -> np.ndarray:
    """σ = sigma_frac * n   (relative width)."""
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if vals.sum() == 0:
        return np.full_like(vals, 1.0 / n)
    return vals / vals.sum()


def _cauchy_distribution(n: int, scale_frac: float = 0.30) -> np.ndarray:
    scale = scale_frac * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = 1.0 / (1.0 + ((i - centre) / scale) ** 2)
    return vals / vals.sum()


def _semicircle_distribution(n: int) -> np.ndarray:
    """Discrete Wigner semicircle on [-1,1]."""
    i = np.arange(n, dtype=np.float64)
    x = (2.0 * i) / (n - 1) - 1.0            # map to [-1,1]
    vals = np.sqrt(np.clip(1.0 - x * x, 0.0, None))
    return vals / vals.sum()


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def _random_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw with concentration α (α=1 → uniform on simplex)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()

    # 2. Include the previously best sequence, if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() == 0:
                prev = _uniform_distribution(prev.size)
            else:
                prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best score={score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Keep top‑k seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = 20
    top_candidates = candidates[:top_k]
    print(f"Selected {len(top_candidates)} seeds for refinement; best cheap score={top_candidates[0][1]:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy refinement (exp‑grad, Adam, smooth Adam)
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # Exponentiated‑gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=100_000,
                                        eta0=0.9,
                                        decay=0.9999)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} exp‑grad → {sc:.6f}")

        # Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                max_iters=100_000,
                                step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam → {sc:.6f}")

        # Smooth Adam
        seq, sc = _smooth_adam_descent(seed_seq, deadline,
                                       max_iters=80_000,
                                       eps0=1e-3,
                                       step0=0.2)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smooth‑Adam → {sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Long sub‑gradient descent on the current best
    subgrad_deadline = deadline - 30.0          # keep ~30 s for hill‑climb
    if time.time() < subgrad_deadline:
        seq, sc = _subgradient_descent(best_seq, subgrad_deadline,
                                       max_iters=1_200_000,
                                       eta0=0.6)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent → {sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Fine‑grained hill‑climbing (guided transfers + random jitter)
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    iteration = 0
    max_transfer = 0.02                       # up to 2 % of total mass initially

    while time.time() < deadline:
        iteration += 1
        # geometric decay of the transfer magnitude
        transfer = max_transfer * (0.5 ** (iteration / 10_000.0))

        # Gradient‑guided transfer (if gradient is cheap to compute)
        if iteration % 7 == 0:
            g, _, _ = _subgrad(a)
            i = int(np.argmax(g))
            j = int(np.argmin(g))
            if i == j:
                i = np.random.randint(n)
                j = (i + np.random.randint(1, n)) % n
        else:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={iteration} new best {best_score:.6f}")

        # occasional jitter to escape plateaus
        if iteration % 5_000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 5e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    print(f"Final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list – ``evaluate_sequence`` expects a list.
    return best_seq.tolist()
