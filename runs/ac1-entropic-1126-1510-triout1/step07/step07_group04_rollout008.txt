"""
Polynomial-time subgradient optimization for minimizing max self‑convolution.

The routine searches a few discrete lengths, performs a fine‑grained
grid search over symmetric Beta distributions, and then refines the best
candidate with an exponentiated‑gradient (mirror‑descent) method that
uses the sub‑gradient of the *single* active convolution entry (the
current maximum).  Several random restarts with tiny jitter are run
until the time budget expires.  The final sequence is returned as a
list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1} normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable version to avoid overflow/underflow
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    # avoid log(0) when x=0 or 1 and alpha/beta<1; clip x away from 0/1
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1.0 - eps)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # cap exponent to avoid overflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf

# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing distribution onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()

# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score when Σa = 1.  Returns 2·n·max(a*a) which equals the exact
    evaluate_sequence output because the evaluator is scale‑invariant.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

# ----------------------------------------------------------------------
def _max_subgradient_one(a: np.ndarray):
    """
    Sub‑gradient of   f(a) = max_k (a*a)_k , taking the gradient of a single
    active index (the current maximum).  Returns (grad, cur_max).
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    # choose one of the maximal indices (if many, pick one at random)
    idxs = np.flatnonzero(b == cur_max)
    idx = idxs[0]  # deterministic; could randomise if desired
    w = np.zeros_like(b)
    w[idx] = 1.0
    a_rev = a[::-1]
    # convolution of a_rev (length n) with w (length 2n‑1); keep central n entries
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max

# ----------------------------------------------------------------------
def _refine_exponential(a: np.ndarray, deadline: float,
                         rng: np.random.Generator,
                         eta0: float = 0.45) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the simplex using the
    single‑max sub‑gradient.  Stops when the deadline is reached.
    Returns the best vector (Σ=1) and its raw score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_one(a)
        eta = eta0 / math.sqrt(it)

        # multiplicative update; clamp the exponent to avoid overflow/underflow
        exponent = -eta * grad
        exponent = np.clip(exponent, -45, 45)
        a = a * np.exp(exponent)

        # re‑normalise onto the simplex Σa = 1
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny random perturbation to jump out of plateaus
        if it % 800 == 0 and rng.random() < 0.07:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] it={it} new best={best_val:.6f} len={n}")

    return best_a, best_val

    x = np.clip(x, eps, 1.0 - eps)
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)        # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a∗a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a∗a."""
    return _self_convolution_full(a).max()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score when Σa = 1.  Returns 2·n·max(a∗a); for such normalized
    vectors this equals the exact value returned by evaluate_sequence.
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_one(a: np.ndarray):
    """
    Sub‑gradient of f(a)=maxₖ (a∗a)ₖ using the gradient of a *single*
    active index (the current maximiser).  Returns (grad, cur_max).
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    # pick the first maximal index (deterministic, fast)
    idx = np.flatnonzero(b == cur_max)[0]
    w = np.zeros_like(b)
    w[idx] = 1.0
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_exponential(a: np.ndarray, deadline: float,
                         rng: np.random.Generator,
                         eta0: float = 0.45) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the simplex using the
    single‑max sub‑gradient.  Stops when the deadline is reached.
    Returns the best vector (Σ=1) and its raw score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_one(a)
        eta = eta0 / math.sqrt(it)

        # multiplicative update; keep exponent bounded
        exponent = -eta * grad
        exponent = np.clip(exponent, -45, 45)
        a = a * np.exp(exponent)

        # re‑normalise onto the simplex Σa = 1
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny jitter to escape flat regions
        if it % 800 == 0 and rng.random() < 0.07:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] it={it} new best={best_val:.6f} len={n}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluate_sequence
    score.  Stages:

    * warm‑start (previous best or uniform);
    * coarse global search (dense Beta grid + Dirichlet draws);
    * intensive mirror‑descent refinement with occasional restarts;
    * optional up‑sampling and final refinement.

    Returns the best sequence found (plain Python list).
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for final evaluation

    # ---------- Warm‑start ----------
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if base_seq.size == 0 or base_seq.sum() == 0.0:
            raise ValueError
        base_seq = base_seq / base_seq.sum()
    except Exception:
        base_seq = np.full(4000, 1.0 / 4000, dtype=np.float64)

    best_seq = base_seq.copy()
    best_score = _raw_score(best_seq)
    best_len = len(best_seq)
    print(f"[{time.time():.1f}] initial raw score={best_score:.6f} (len={best_len})")

    # ---------- Phase 1: global exploration ----------
    lengths = [2000, 4000, 8000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] exploring length {n}")

        # Interpolate current best to this length
        a = _interpolate_to_length(best_seq, n)
        cur = _raw_score(a)
        if cur < best_score:
            best_seq, best_score, best_len = a, cur, n
            print(f"   interpolated best score={best_score:.6f}")

        # Dense symmetric Beta grid
        alphas = np.linspace(0.18, 0.72, 111)   # step ≈0.005
        for a_alpha in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a_alpha, a_alpha)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec, sc, n
                print(f"   symBeta α={a_alpha:.3f} score={best_score:.6f}")

        # Random Dirichlet draws for extra diversity
        for _ in range(200):
            if time.time() >= deadline:
                break
            conc = 10 ** rng.uniform(-2, 1)   # concentration ∈[0.01,10]
            vec = rng.gamma(conc, 1.0, size=n)
            vec /= vec.sum()
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec, sc, n
                print(f"   dirichlet conc={conc:.3f} score={best_score:.6f}")

    print(f"[{time.time():.1f}] after coarse search best raw={best_score:.6f} (len={best_len})")

    # ---------- Phase 2: intensive mirror‑descent ----------
    if time.time() < deadline:
        a = best_seq.copy()
        a, refined = _refine_exponential(a, deadline, rng, eta0=0.45)
        if refined < best_score:
            best_seq, best_score = a, refined
            best_len = len(a)
        print(f"[{time.time():.1f}] after refinement score={best_score:.6f} (len={best_len})")

    # ---------- Phase 3: short restarts with jitter ----------
    for restart in range(5):
        if time.time() >= deadline:
            break
        pert = best_seq + rng.normal(scale=1e-6, size=best_len)
        pert = _project_onto_simplex(pert)
        a, scr = _refine_exponential(pert, deadline, rng, eta0=0.42)
        if scr < best_score:
            best_seq, best_score = a, scr
            print(f"[{time.time():.1f}] restart {restart} improved score={best_score:.6f}")

    # ---------- Phase 4: up‑sampling and final refinement ----------
    larger_lengths = [12000, 16000, 20000]
    for n in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to {n}")
        a = _interpolate_to_length(best_seq, n)
        a = _project_onto_simplex(a)
        a, scr = _refine_exponential(a, deadline, rng, eta0=0.38)
        if scr < best_score:
            best_seq, best_score, best_len = a, scr, n
            print(f"   new best after up‑sample score={best_score:.6f} (len={best_len})")
        if best_score <= 1.503:
            print("[*] target bound reached – exiting early")
            break

    # ---------- Final exact evaluation ----------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
