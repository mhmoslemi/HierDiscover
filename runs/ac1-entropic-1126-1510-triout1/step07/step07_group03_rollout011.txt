def _lp_refine(a, deadline, best_seq, best_score):
    while time.time() < deadline:
        n = a.shape[0]
        b = _fft_self_convolve(a)
        M = b.max()
        # Build constraint matrix A (2n-1, n)
        n_const = 2*n - 1
        # Use sparse construction: each row corresponds to a shift of a.
        # Build data arrays:
        rows = []
        cols = []
        data = []
        for i in range(n):
            # a[i] contributes to constraints with offset i in rows: row = i + j
            # For each j (col index) from 0 to n-1:
            # convolution term a_{row - j} => row = i + j => constraint coefficient for g_j = a_i
            # So coefficient a_i appears in each row = i + j for column j.
            # So we can fill for each i: for j in 0..n-1: rows append i+j, cols append j, data append a[i]
            # This is O(n^2). Might be okay for n up to 1500, as n^2=2.25M entries, not huge.
            for j in range(n):
                rows.append(i + j)
                cols.append(j)
                data.append(a[i])
        # Build sparse matrix
        A = csr_matrix((data, (rows, cols)), shape=(n_const, n))
        # b_ub = M * ones
        b_ub = np.full(n_const, M)
        c = -np.ones(n)  # maximize sum(g)
        bounds = [(0, None)]*n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()
        # Evaluate conv of g alone
        score_g = _objective(g)
        # Update best if improved
        if score_g < best_score - 1e-12:
            best_score = score_g
            best_seq = g.copy()
            print(f"[LP] new best {best_score:.6f}")
        # Mix current a with g using small steps or optimal t
        improved = False
        for t in (0.5,0.2,0.1,0.05):

                break
        if not improved:
            # If mixing didn't improve, break out.
            break
        # Time check
        if time.time() + 1.0 >= deadline:
            break
    return best_seq, best_score

def _lp_refine(a, deadline, best_seq, best_score):
    import scipy.optimize
    import scipy.linalg
    while time.time() < deadline:
        n = a.shape[0]
        b = _fft_self_convolve(a)
        M = b.max()
        # construct Toeplitz matrix T of shape (2n-1, n)
        n_const = 2*n - 1
        first_col = np.concatenate([a, np.zeros(n_const - n, dtype=a.dtype)])
        first_row = np.zeros(n, dtype=a.dtype)
        first_row[0] = a[0]
        T = scipy.linalg.toeplitz(first_col, first_row)  # dense matrix
        # Linear program: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=a.dtype)
        bounds = [(0, None)] * n
        res = scipy.optimize.linprog(c, A_ub=T, b_ub=np.full(n_const, M), bounds=bounds,
                                    method='highs')
        if not res.success:
            print(f"[LP] linprog failed: {res.message}")
            break
        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()
        # Evaluate g directly
        score_g = _objective(g)
        if score_g < best_score - 1e-12:
            best_score = score_g
            best_seq = g.copy()

"""
Propose a non‑negative step‑function (probability vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a multi‑stage optimiser:

1. **Broad initialisation** – many analytic families (uniform, arcsine,
   symmetric beta, exponential, alternating, two‑block, sparse random)
   are generated for several lengths.  The best of those seeds the search.

2. **Smoothed Adam descent** – minimise a smooth log‑sum‑exp approximation
   of the max‑convolution.  The smoothness parameter α is increased
   progressively (1 → 5 → 20 → 100 → 500 → 2000 → 20000) while Adam updates
   are projected onto the simplex.

3. **Linear‑program refinement** – for a current vector *f* we solve the LP  

        maximise   Σ g_j
        subject to  f * g ≤ M   (pointwise),   g ≥ 0,

   where *M = max(f*f)*.  The optimal *g* is mixed with *f* (convex
   combination) and projected onto the simplex.  This active‑set step
   often reduces the peak convolution further.

4. **Exact sub‑gradient descent** – true sub‑gradient of the max‑convolution
   is used with momentum and a diminishing step size.

5. **Greedy active‑set swaps** – mass is transferred from indices with a
   large sub‑gradient to those with a small one whenever this improves the
   objective.

6. **Simulated‑annealing fine‑tuning** – a cheap Metropolis walk with
   exponential cooling helps to escape shallow plateaux.

All stages respect a global time budget (default 1000 s).  The best vector
found is returned as a plain Python list of floats.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector normalised to Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    # symmetric beta with shape parameter p (p>0)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    """Uniform mass on every `stride`‑th coordinate (starting at 0)."""
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    """Uniform mass on the first and last `block_frac` fraction of indices."""
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int) -> list[np.ndarray]:
    """Generate a broad set of plausible start vectors for a given length."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # beta family – many shape parameters
    for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials – wide range
    for lam in (0.001, 0.002, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4):
        cand.append(_distribution_alternating(n, stride))

    # two‑block spikes at the ends
    cand.append(_distribution_two_blocks(n, block_frac=0.05))
    cand.append(_distribution_two_blocks(n, block_frac=0.10))

    # random sparse start (~0.5 % non‑zeros)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    return cand


# ----------------------------------------------------------------------
# Stage 1 – smoothed Adam descent --------------------------------------
def _smooth_adam_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth approximation of max‑convolution."""
    n = a.shape[0]

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # increasingly sharp smooth‑max
    schedule = [
        (1.0, 2000),
        (5.0, 2000),
        (20.0, 2000),
        (100.0, 2000),
        (500.0, 2000),
        (2000.0, 2000),
        (20000.0, 2000),   # extra refinement phase
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights (log‑sum‑exp)
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth max (cross‑correlation)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto simplex
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] Adam α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – linear‑program improvement ---------------------------------
def _lp_refine(a: np.ndarray, deadline: float,
               best_seq: np.ndarray,
               best_score: float) -> tuple[np.ndarray, float]:
    """
    One‑shot LP refinement:
       maximise Σ g_j  subject to  f * g ≤ M, g ≥ 0.
    The solution g is mixed back into f (convex combination) and the
    simplex‑projected result is kept if it improves the objective.
    """
    try:
        from scipy.optimize import linprog
        from scipy.linalg import toeplitz
    except Exception as exc:
        # SciPy unavailable – skip this stage.
        print("[LP] SciPy not available, skipping LP refinement.")
        return best_seq, best_score

    n = a.shape[0]
    # For very large n the dense Toeplitz matrix becomes huge; abort.
    if n > 2000:
        return best_seq, best_score

    while time.time() < deadline:
        # current peak M
        b = _fft_self_convolve(a)
        M = b.max()

        # Build the (2n‑1)×n Toeplitz constraint matrix for f * g .
        n_const = 2 * n - 1
        first_col = np.concatenate([a, np.zeros(n_const - n, dtype=a.dtype)])
        first_row = np.zeros(n, dtype=a.dtype)
        first_row[0] = a[0]
        A = toeplitz(first_col, first_row)  # dense (2n‑1)×n matrix

        # Linear program: maximise Σ g   ⇔ minimise –Σ g
        c = -np.ones(n, dtype=a.dtype)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(n_const, M), bounds=bounds,
                      method='highs')
        if not res.success:
            print(f"[LP] linprog failed: {res.message}")
            break

        g = res.x
        if g.sum() == 0:
            break
        g = g / g.sum()          # renormalise to unit sum

        # Direct evaluation of the LP solution
        score_g = _objective(g)
        if score_g < best_score - 1e-12:
            best_score = score_g
            best_seq = g.copy()
            print(f"[LP] direct new best {best_score:.6f}")

        # Try to improve by mixing g back into the current a
        improved = False
        for t in (0.5, 0.2, 0.1, 0.05):
            a_new = (1.0 - t) * a + t * g
            a_new = _project_onto_simplex(a_new, 1.0)
            s = _objective(a_new)
            if s < best_score - 1e-12:
                a = a_new
                best_score = s
                best_seq = a.copy()
                improved = True
                print(f"[LP] mix t={t:.2f} new best {best_score:.6f}")
                break
        if not improved:
            # No improvement → stop LP refinement.
            break

        # safety time check
        if time.time() + 2.0 >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 3 – exact sub‑gradient descent ---------------------------------
def _exact_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the genuine max‑convolution objective."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.30
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        # sub‑gradient from active set
        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # momentum + diminishing stepsize
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        if best_score <= 1.503:
            break

        # cheap early‑exit check
        if time.time() + 0.5 >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 4 – greedy active‑set swapping ---------------------------------
def _greedy_swap_opt(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Move mass from high‑gradient to low‑gradient indices while improving."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        # sub‑gradient for the active set
        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]

        j = int(np.argmin(grad))
        if i == j:
            break

        delta = min(a[i] * 0.5, 0.005)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")
        else:
            # try a smaller transfer
            delta *= 0.25
            if delta < 1e-7:
                break
            a_new[i] = a[i] - delta
            a_new[j] = a[j] + delta
            cur = _objective(a_new)
            if cur < best_score - 1e-13:
                a = a_new
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] tiny swap improved best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 5 – simulated annealing -----------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis‑type random walk with exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    T = 0.04                     # initial temperature (objective units)
    iter_cnt = 0

    while time.time() < deadline and T > 1e-4:
        iter_cnt += 1

        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue

        # pick a different destination
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random transfer, up to 10 % of source mass
        delta = np.random.rand() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter_cnt} new best {best_score:.6f}")

        # exponential cooling
        if iter_cnt % 2500 == 0:
            T *= 0.85

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best vector (list of floats).
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a set of lengths – a mixture of short and long vectors.
    lengths = [800, 1200, 2000, 3000, 5000, 8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # 2. Scan analytic families for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates_extended(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 3. If a previous solution exists (global variable ``height_sequence_1``),
    #    reuse it as a seed.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 4. Fallback – should never fire, but keep a safe default.
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline.
    a = overall_best_seq.copy()

    # 5a – smoothed Adam descent (most work‑intensive stage)
    a, overall_best_score = _smooth_adam_descent(
        a, deadline,
        best_seq=a,
        best_score=overall_best_score,
    )

    # 5b – linear‑program refinement (only for moderate lengths)
    a, overall_best_score = _lp_refine(
        a, deadline,
        best_seq=a,
        best_score=overall_best_score,
    )

    # 5c – exact sub‑gradient refinement
    a, overall_best_score = _exact_subgradient_descent(
        a, deadline,
        best_seq=a,
        best_score=overall_best_score,
    )

    # 5d – greedy active‑set swapping
    a, overall_best_score = _greedy_swap_opt(
        a, deadline,
        best_seq=a,
        best_score=overall_best_score,
    )

    # 5e – simulated annealing fine‑tuning
    a, overall_best_score = _simulated_annealing(
        a, deadline,
        best_seq=a,
        best_score=overall_best_score,
    )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(a)})")
    # Return as plain Python list (evaluate_sequence expects a list of floats)
    return a.tolist()
