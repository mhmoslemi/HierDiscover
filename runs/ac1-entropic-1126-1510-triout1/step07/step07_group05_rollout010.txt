    best_n = None

    for n in lengths:
        if time.time() >= deadline: break
        print(f"\n=== Optimising length {n} ===")
        # generate candidate initial sequences normalized
        candidates = []

        # uniform
        candidates.append(np.full(n, 1.0/n))

        # warm start from previous best if available
        if best_seq is not None:
            cand = _interpolate_to_length(best_seq, n)
            candidates.append(cand)

        # parametric shapes
        # exponent p for polynomial shape
        ps = [0.3, 0.5, 0.7, 1.0, 1.3, 1.5, 2.0]
        for p in ps:
            # polynomial shape (i+0.5)*(n-i-0.5)
            i = np.arange(n, dtype=np.float64) + 0.5
            shape = (i * (n - i)) ** p
            shape /= shape.sum()
            candidates.append(shape)

        # sine shape
        for p in [0.5, 1.0, 2.0, 4.0]:
            theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
            shape = np.sin(theta) ** p
            shape /= shape.sum()
            candidates.append(shape)

        # mixture with uniform
        for lam in [0.2, 0.4, 0.6, 0.8]:
            for p in [0.5, 1.0, 1.5]:
                i = np.arange(n, dtype=np.float64) + 0.5
                shape = ((i * (n - i)) ** p)
                shape /= shape.sum()
                mix = lam * (1.0/n) + (1 - lam) * shape
                mix /= mix.sum()
                candidates.append(mix)

        # few Dirichlet random draws
        for _ in range(5):
            alpha = np.random.rand() * 4.0 + 0.1
            d = np.random.gamma(alpha, 1.0, size=n)
            d /= d.sum()
            candidates.append(d)

        # evaluate each candidate quickly and keep best
        for a in candidates:
            raw = 2.0 * n * _fft_self_convolve_max(a)  # fast conv
            if raw < best_raw:
                best_raw = raw
                best_seq = a.copy()
                best_n = n
                print(f"New best raw={best_raw:.6f} at length={n}")

        # refine best for this length using subgradient
        if best_seq is not None and best_seq.shape[0]==n:
            a, raw = _refine_by_subgradient(best_seq, deadline, start_iter=0)
            if raw < best_raw:

    # final refinement on best overall
    if best_seq is not None:

"""
Propose a low‑score step function.

Algorithm
---------
* Choose a target length (here 8000 – large enough to get close to the
  asymptotic optimum, still cheap to convolve with an FFT).
* Generate many diverse starting points:
    - uniform distribution,
    - warm‑start from a previously known solution (if any),
    - Dirichlet draws with a wide range of concentration parameters,
    - a few handcrafted shapes (polynomial / sine based) for extra
      coverage of the search space.
* For each start run a short Adam optimisation on a smooth‑max surrogate
  of the convolution maximum (log‑sum‑exp with a linearly increasing
  temperature).  The Adam routine respects the probability simplex via
  a soft‑max parametrisation.
* Follow the Adam phase by a projected sub‑gradient descent on the exact
  max‑convolution (the same routine used in the reference solution).
* Keep the best sequence seen over all starts.
* Finally perform a longer sub‑gradient refinement on the global best.
* Return the raw sequence (the evaluation routine clips negative values,
  rescales overly large entries, and is scale‑invariant).

The whole optimisation stops before the given time budget (default 1000 s)
to avoid a forced termination with –∞ points.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 6000,
                            tau0: float = 0.5,
                            tau_max: float = 300.0,
                            lr0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max (log‑sum‑exp) surrogate of the
    convolution supremum.  The parametrisation is a soft‑max over
    unconstrained logits `x`.  Returns the best raw score (2·n·max) seen.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        # ----- soft‑max (stable) -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # ----- convolution and smooth max -----
        conv = _fft_self_convolve_full(a)        # length 2n‑1
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        # temperature schedule (linear increase)
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # smooth‑max weights (stable softmax)
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # gradient of the smooth max w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # ----- chain rule through the soft‑max -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent for the exact max‑convolution.
    Returns the refined (a, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_max(a)
    it = start_iter
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.07 / math.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)
        # tiny random perturbations every few hundred iterations
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")
    return best_a, best_score

        print(f"\n=== Final long‑run refinement (length={n_target}) ===")
        best_seq, best_raw = _refine_by_subgradient(best_seq, deadline, start_iter=0)
        print(f"Final raw after refinement: {best_raw:.6f}")
    else:
        # fallback – uniform distribution of moderate size
        n_fallback = 5000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # ------------------------------------------------------------------
    # Convert to plain Python list and obtain the official score.
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # provided by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
