def _lp_optimize(a, max_iter=1):
    n = len(a)
    # Build constraint matrix for each k.
    rows = []
    cols = []
    data = []
    nob = 2 * n - 1
    for k in range(nob):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # For i in range(i_start, i_end+1):
        # j = k - i
        # coefficient a[i] on variable b[j]
        # we can do vectorized: i_range = np.arange(i_start, i_end+1)
        # j_range = k - i_range
        # these slices length L
        L = i_end - i_start + 1
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        rows.extend([k] * L)
        cols.extend(j_range.tolist())
        data.extend(a[i_range].tolist())
        # M coefficient
        rows.append(k)
        cols.append(n)  # index for M
        data.append(-1.0)
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(nob, n+1))
    b_ub = np.zeros(nob)
    # Equality constraint for sum(b) = 1.
    A_eq = scipy.sparse.csr_matrix(([1.0]*n, (np.zeros(n), np.arange(n))), shape=(1, n+1))
    # extend with zero for M column
    A_eq = scipy.sparse.hstack([A_eq, scipy.sparse.csr_matrix([[0.0]])])
    b_eq = np.array([1.0])
    # bounds: b_i >=0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M

    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                 bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback to current a; return a copy and max conv.
        return a.copy(), _fft_self_convolve(a).max()
    b = res.x[:n]
    M = res.x[-1]
    return b, M
```

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence().  The algorithm combines:

1. A quick scan of a few analytic families (beta‑U‑shapes, sine‑powers,
   exponential, etc.) over several lengths to pick a promising start.
2. An alternating linear‑programming refinement:
   – For a fixed a, solve   min M   s.t. a * b ≤ M  (convolution),
     b ≥ 0, Σb = 1.
   – Mix a ← (1‑t)·a + t·b (t≈0.5) and renormalise.
   – Repeat a few times.
3. A lightweight hill‑climber that moves tiny mass chunks while
   updating the full self‑convolution in O(n) per move.

All operations stay on the simplex (Σa = 1, a_i ≥ 0) so the objective
reduces to 2·n·max(a*a).  The routine respects a 1000 s wall‑clock limit,
prints intermediate scores and finally returns the best vector found.
"""

import time
import random
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Project vector v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Return 2·n·max(a*a) for a vector a normalised to Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families (all return a vector with Σ = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric beta density on {0,…,n‑1} with both shape parameters = p."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j, updating a and its
    full self‑convolution `conv` in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()
    # remove contribution of transferred mass from all positions involving i
    conv[i:i + n] -= 2.0 * delta * a_old
    # add contribution of transferred mass to all positions involving j
    conv[j:j + n] += 2.0 * delta * a_old
    # the pure quadratic self‑terms
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

def _lp_optimize(a):
    """
    Solve   min M   s.t.   a * b ≤ M   (full convolution),
    b ≥ 0, Σb = 1.
    Returns the optimal b and the achieved M.
    """
    n = a.shape[0]
    # number of convolution entries
    n_out = 2 * n - 1

    # Build sparse A_ub matrix (shape n_out × (n+1))
    rows = []
    cols = []
    data = []

    for k in range(n_out):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        rows.extend([k] * i_range.size)
        cols.extend(j_range.tolist())
        data.extend(a[i_range].tolist())
        # coefficient for M (last variable)
        rows.append(k)
        cols.append(n)          # index of M
        data.append(-1.0)

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(n_out, n + 1))
    b_ub = np.zeros(n_out)

    # Equality Σb = 1  (M has zero coefficient)
    eq_data = np.ones(n)
    eq_rows = np.zeros(n, dtype=int)
    eq_cols = np.arange(n, dtype=int)
    A_eq = sp.csr_matrix((eq_data, (eq_rows, eq_cols)), shape=(1, n + 1))
    b_eq = np.array([1.0])

    # Bounds: b_i ≥ 0, M ≥ 0
    bounds = [(0.0, None)] * n + [(0.0, None)]

    # Objective: minimise M
    c = np.zeros(n + 1)
    c[-1] = 1.0

    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})
    if not res.success:
        # fallback: return the input unchanged and its current max
        return a.copy(), _fft_self_convolve(a).max()
    b_opt = res.x[:n]
    M_opt = res.x[-1]
    return b_opt, M_opt

def _hill_climb(a, deadline, max_no_improve=800):
    """
    Simple hill‑climber that repeatedly moves a tiny amount of mass
    from the entry that most contributes to the current maximum towards
    a random location.  Updates the convolution incrementally.
    Returns the (possibly) improved a and the corresponding score.
    """
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    no_improve = 0

    while time.time() < deadline and it < 200000:
        it += 1
        # index of current maximum of the convolution
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i, j) inside the slice that gives the largest product a[i]*a[j]
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # source is the larger of the two, target is random
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a_snap = a.copy()
        conv_snap = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            no_improve = 0
        else:
            a = a_snap
            conv = conv_snap
            no_improve += 1

        if no_improve > max_no_improve:
            # jitter to escape flat regions
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            no_improve = 0

        if it % 5000 == 0 and time.time() < deadline:
            print(f"[{time.time():.1f}] hill_it={it} cur {2.0 * n * M:.6f}")

    return a, 2.0 * n * M

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function (finite vector) that minimises
    evaluate_sequence().  The routine respects the 1000 s time limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 1. Quick scan over a handful of lengths / analytic shapes
    lengths = [1800, 2500, 3000, 3500]
    beta_ps = [0.15, 0.2, 0.25, 0.3, 0.35, 0.4, 0.45, 0.5]
    best_a = None
    best_score = float('inf')
    best_n = None
    eps = 1e-12

    for n in lengths:
        if time.time() >= deadline:
            break
        for p in beta_ps:
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_a = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] length={n} beta={p:.2f} init best {sc:.6f}")

    # fallback to a uniform start if nothing selected (should not happen)
    if best_a is None:
        best_n = 2500
        best_a = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_a)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    a = best_a.copy()
    n = best_a.shape[0]

    # ------------------------------------------------------------------
    # 2. Alternating LP refinement (few outer rounds)
    outer_iters = 6        # number of LP+mix cycles
    mix_param = 0.5        # convex combination weight
    for outer in range(outer_iters):
        if time.time() >= deadline:
            break
        # solve LP: find b that minimises max convolution with current a
        b, M_opt = _lp_optimize(a)
        conv_a = _fft_self_convolve(a)
        M_cur = conv_a.max()
        print(f"[{time.time():.1f}] LP round {outer+1}/{outer_iters} M_cur={M_cur:.6e} M_opt={M_opt:.6e}")

        # If LP does not improve, stop early
        if M_opt >= M_cur - 1e-12:
            print("LP unable to improve – stopping LP phase.")
            break

        # Mix and renormalise
        a = (1.0 - mix_param) * a + mix_param * b
        a = np.maximum(a, eps)
        a /= a.sum()

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"  --> new best after mix {cur_score:.6f}")

        # 3. Small hill‑climb to clean up any leftover spikes
        a, hill_score = _hill_climb(a, deadline, max_no_improve=500)
        if hill_score < best_score - 1e-12:
            best_score = hill_score
            best_a = a.copy()
            print(f"  --> new best after hill {hill_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final hill‑climb (more aggressive) until time runs out
    a, final_score = _hill_climb(best_a, deadline, max_no_improve=800)
    if final_score < best_score:
        best_a = a.copy()
        best_score = final_score

    # ------------------------------------------------------------------
    # 5. Report and return the best sequence found
    print(f"Final best score ≈ {best_score:.6f} (length={best_a.shape[0]})")
    return best_a.tolist()
