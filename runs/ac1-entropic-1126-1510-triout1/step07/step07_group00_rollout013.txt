def _sgd_subgrad(a0, deadline, max_iters=200000, step0=0.5):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0

        it += 1

        a = a - step * grad

            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _beta_distribution(n, alpha, beta):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Handle potential overflow to -inf
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or w.sum() == 0 or np.isinf(w).any():
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w.sum()
    return w

Now Gaussian distribution:

def _gaussian_distribution(n, sigma_factor=0.2):
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w = w / w_sum
    return w

def _lp_refine(a, deadline):
    # solve linear program to find g maximizing sum(g) under (a*g) <= M
    n = a.shape[0]
    cur_max = _subgrad(a)[1]  # returns cur_max
    M = cur_max
    # Build constraint matrix A (2n-1) x n as sparse
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    # iterate k rows
    for k in range(2*n-1):
        # indices j where 0 <= k-j < n => max(0, k-(n-1)) <= j <= min(k, n-1)
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # for each j, coeff is a_{k-j}
        # we can vectorize: for j in range(j_start, j_end+1):
        #   rows.append(k)
        #   cols.append(j)
        #   vals.append(a[k-j])
        # let's use loops
        for j in range(j_start, j_end + 1):
            rows.append(k)

"""
A robust time‑bounded optimiser for the step‑function score.

The algorithm proceeds in three phases:

1. **Broad sampling** – generate many analytic families (arcsine,
   triangular, exponential, symmetric/unsymmetric beta, Gaussian,
   Cauchy) on a variety of lengths and a few random Dirichlet draws.
   Every candidate is ℓ₁‑normalised (sum = 1) and scored with the cheap
   surrogate  ``2·n·max(conv(a,a))``  which equals the true
   ``evaluate_sequence`` value for a normalised vector.

2. **Intensive refinement** – the best few candidates are polished by
   three complementary mirror‑descent schemes (Adam, exponentiated‑
   gradient, simple sub‑gradient) with aggressive iteration limits.
   A Frank‑Wolfe style pairwise mass‑transfer loop finishes the local
   optimisation.

3. **Linear‑programming boost (optional)** – for moderate lengths a
   linear programme (maximise ∑g_j subject to (a∗g)_k ≤ M) is solved
   (highs solver).  The resulting direction is blended into the current
   best vector and a short SGD refinement is performed.

All steps respect the global time budget (≈ 1000 s).  The function
returns the best normalised sequence found as a plain ``list[float]``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for evaluate_sequence when sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (sum(a)=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active

# analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8,
                          decay: float = 0.9999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 200_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _lp_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """One‑shot linear‑programming improvement (used only for modest n)."""
    n = a.shape[0]
    if n > 2000:                     # keep the LP cheap
        return a, _max_convolution_score(a)

    # current maximal convolution value
    cur_max = _subgrad(a)[1]

    # Build sparse constraint matrix for (a * g) ≤ cur_max
    from scipy.sparse import csr_matrix
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, cur_max, dtype=np.float64)

    # maximise ∑ g_j  → minimise −∑ g_j
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n

    from scipy.optimize import linprog
    remaining = max(0.0, deadline - time.time())
    res = linprog(c,
                  A_ub=A, b_ub=b,
                  bounds=bounds,
                  method='highs',
                  options={'time_limit': remaining, 'presolve': True})
    if not res.success:
        return a, _max_convolution_score(a)

    g = res.x
    if g.sum() > 0:
        g = g / g.sum()
    else:
        g = a.copy()

    # Blend a small amount of g into a and polish with a quick SGD run
    t = 0.2
    a_new = (1 - t) * a + t * g
    a_new = _project_onto_simplex(a_new, s=1.0)
    # a few cheap SGD steps
    a_ref, sc_ref = _sgd_subgrad(a_new, deadline, max_iters=60_000, step0=0.4)
    return a_ref, sc_ref

        # unsymmetric beta -----------------------------------------------
        for alpha in beta_params:
            for beta in beta_params:
                if time.time() >= deadline:
                    break
                a = _beta_distribution(n, alpha, beta)
                candidates.append((a, _max_convolution_score(a)))

        # power‑transformed variants (applied to arcsine & beta) -------
        for base_func, base_name in (
            (_arcsine_distribution, "arcsine"),
            (lambda m: _beta_symmetric_distribution(m, 0.5), "beta0.5"),
            (lambda m: _beta_symmetric_distribution(m, 1.0), "beta1.0"),
        ):
            base = base_func(n)
            for pwr in power_exps:
                if time.time() >= deadline:
                    break
                a = np.power(base, pwr)
                a = a / a.sum()
                candidates.append((a, _max_convolution_score(a)))

        # random Dirichlet draws -----------------------------------------
        for alpha in dirichlet_alphas:
            if time.time() >= deadline:
                break
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # --------------------------------------------------------------
    # 2. inject previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            print(f"[{time.time():.1f}] added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. keep top seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(8, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Current best seed score = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # --------------------------------------------------------------
    # 4. intensive refinement of each top seed
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Adam
        seq, sc = _adam_descent(seed_seq, deadline,
                                 max_iters=300_000,
                                 step0=0.2)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam -> {best_score:.6f}")

        # Exponentiated gradient
        seq, sc = _exp_gradient_descent(seed_seq, deadline,
                                        max_iters=150_000,
                                        eta0=0.8)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} ExpGrad -> {best_score:.6f}")

        # Simple sub‑gradient SGD (lots of cheap steps)
        seq, sc = _sgd_subgrad(seed_seq, deadline,
                               max_iters=200_000,
                               step0=0.5)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} SGD -> {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Frank‑Wolfe style pairwise mass transfer on the global best
    a = best_seq.copy()
    cur_score = best_score
    it_fw = 0
    while time.time() < deadline and it_fw < 80_000:
        it_fw += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-14:
            break

        # try full transfer
        a_full = a.copy()
        a_full[i_src] -= max_transfer
        a_full[i_dst] += max_transfer
        score_full = _max_convolution_score(a_full)

        improved = False
        if score_full < cur_score - 1e-12:
            a = a_full
            cur_score = score_full
            improved = True
        else:
            # try half‑transfer as fallback
            half = max_transfer * 0.5
            a_half = a.copy()
            a_half[i_src] -= half
            a_half[i_dst] += half
            score_half = _max_convolution_score(a_half)
            if score_half < cur_score - 1e-12:
                a = a_half
                cur_score = score_half
                improved = True

        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={it_fw} new best {best_score:.6f}")

        # occasional tiny jitter to escape plateaus
        if it_fw % 5000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 1e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    # 6. linear‑programming boost (if time permits)
    if time.time() < deadline - 2.0:
        try:
            best_seq, best_score = _lp_refine(best_seq, deadline)
            print(f"[{time.time():.1f}] LP refinement gave new best {best_score:.6f}")
        except Exception as e:
            print("LP refinement failed:", e)

    # --------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
