"""
Hybrid random‑search + continuation optimisation for the
step‑function evaluation problem.

The algorithm proceeds in four stages:

1.  Build a diverse pool of seed distributions (uniform, arcsine,
    Dirichlet draws, Beta‑pdfs and a few simple mixtures) and pick the
    best according to the cheap proxy 2·n·max(a*a).

2.  Run a time‑limited random search over a parametrised family of
    mixtures of Beta‑pdfs (with optional uniform component).  The
    objective is the exact raw score 2·n·max_self_convolution.

3.  Refine the current best sequence with a continuation method:
    minimise a smooth approximation of the max‑convolution
    (log‑sum‑exp) using projected FISTA.  The smoothing parameter μ is
    reduced geometrically until it reaches ≈1e‑6.

4.  Polish the final sequence with a short projected sub‑gradient
    descent.

All operations respect the non‑negativity and simplex constraints
required by `evaluate_sequence`.  The routine returns the best
sequence found before the time budget (≈1000 s) expires.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf to avoid overflow for extreme parameters
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # Clip to avoid overflow in exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """Power‑law distribution a_i ∝ (i+1)^{-p}, normalised to Σ=1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    pdf = 1.0 / np.power(idx, p)
    pdf /= pdf.sum()
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Exact proxy: 2·n·max(a*a).  Input must already be a probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # values ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights, Σp = 1

    # gradient: 2 * (a_rev * p)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smoothed objective value (log‑sum‑exp of conv)
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Refine a probability vector with continuation + FISTA.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]

    mu = _max_self_convolution(a) * 0.5
    mu = max(mu, 1e-7)
    mu_target = 1e-6
    mu_factor = 0.7          # geometric reduction

    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # ---- inner FISTA for current μ ---------------------------------
        # Lipschitz estimate (safe upper bound)
        L = max(1.0, 2.0 / mu)

        y = a.copy()
        t = 1.0

        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        # Estimate ~0.001 s per 1000 iterations (empirical)
        iters = int(min(5000, remaining / 0.0012))
        if iters <= 0:
            break

        for it in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # simple back‑tracking line‑search (max two doublings of L)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional exact evaluation
            if (it & 31) == 0:
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()
        # ----------------------------------------------------------------
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-12:
            break
        # Warm‑restart from the best point seen for the new μ
        a = best_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish the final solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # occasional jitter to escape plateaus
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point for the competition harness.  Returns a list of non‑negative
    step heights whose evaluation score is as low as possible.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # safety margin
    n = 8000                                            # length of final vector
    TARGET = 1.503                                      # optional early‑stop target

    # --------------------------------------------------------------
    # 1. Build initial pool of seed sequences (all are normalised to sum=1)
    candidates = []

    # a) warm‑start from the global solution if it exists
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if base.ndim != 1:
            base = base.ravel()
        base = base / base.sum()
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # b) deterministic baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_beta_distribution(n, 0.5, 0.5))                # arcsine

    # c) a few symmetric Beta shapes
    for a in (0.2, 0.3, 0.4, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n, a, a))

    # d) random Dirichlet draws
    for _ in range(6):
        conc = 10 ** rng.uniform(-2, 1)            # concentration in [0.01,10]
        vec = rng.gamma(conc, 1.0, size=n)
        vec /= vec.sum()
        candidates.append(vec)

    # e) power‑law family (useful for very skewed shapes)
    for p in (0.0, 0.3, 0.6, 1.0, 1.5, 2.0):
        candidates.append(_powerlaw_distribution(n, p))

    # --------------------------------------------------------------
    # 2. Evaluate pool and keep the best
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        raw = _raw_score(seq)
        if raw < best_raw:
            best_raw = raw
            best_seq = seq.copy()
    print(f"[Init] best raw score from pool: {best_raw:.6f}")

    # --------------------------------------------------------------
    # 3. Random search over mixtures of Beta‑pdfs (and optional uniform)
    uniform_vec = np.full(n, 1.0 / n, dtype=np.float64)
    while time.time() < deadline - 30:          # keep ~30 s for polishing
        r = rng.random()
        if r < 0.45:
            # single Beta
            alpha = 10 ** rng.uniform(-2, 1)
            beta = 10 ** rng.uniform(-2, 1)
            seq = _beta_distribution(n, alpha, beta)
        elif r < 0.70:
            # Beta + uniform mixture
            alpha = 10 ** rng.uniform(-2, 1)
            beta = 10 ** rng.uniform(-2, 1)
            w = rng.random()
            seq = w * _beta_distribution(n, alpha, beta) + (1 - w) * uniform_vec
        else:
            # mixture of two Betas
            alpha1 = 10 ** rng.uniform(-2, 1)
            beta1 = 10 ** rng.uniform(-2, 1)
            alpha2 = 10 ** rng.uniform(-2, 1)
            beta2 = 10 ** rng.uniform(-2, 1)
            w = rng.random()
            seq = w * _beta_distribution(n, alpha1, beta1) + (1 - w) * _beta_distribution(n, alpha2, beta2)

        # normalise (numerical safety)
        seq = seq / seq.sum()
        raw = _raw_score(seq)
        if raw < best_raw:
            best_raw = raw
            best_seq = seq.copy()
            print(f"[Random] new best raw = {best_raw:.6f} (time {time.time()-start_time:.1f}s)")

        # early stop if target already reached
        if best_raw <= TARGET:
            print("[Random] target reached, exiting random search")
            break

    # --------------------------------------------------------------
    # 4. Continuation + FISTA refinement (if time permits)
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 20.0:
        print(f"[FISTA] starting refinement, time left {remaining:.1f}s")
        best_seq, best_raw = _fista_refine(best_seq, remaining * 0.6)
        print(f"[FISTA] best raw after refinement: {best_raw:.6f}")

    # --------------------------------------------------------------
    # 5. Final short sub‑gradient polishing
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 5.0:
        print(f"[Polish] starting sub‑gradient polish, time left {remaining:.1f}s")
        best_seq, best_raw = _subgradient_polish(best_seq, remaining)
        print(f"[Polish] best raw after polish: {best_raw:.6f}")

    # --------------------------------------------------------------
    # 6. Return the answer (evaluate_sequence rescales automatically)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
