def _smooth_max_gradient(a, gamma):
    n = len(a)
    b = _fft_self_convolve(a)   # length 2n-1

    max_b = np.max(b)   # to avoid overflow

    w_sum = np.sum(w)
    w = w / w_sum

"""
Optimise a non‑negative step function (probability vector) to minimise
the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)²)     (scale‑invariant)

The implementation follows a multi‑scale projected‑Adam scheme:

1. Generate a small set of analytic seeds (arcsine, triangular,
   exponential, symmetric beta) and a few Dirichlet draws.
2. Refine each seed with a massive Adam optimiser on the true
   sub‑gradient of the max‑convolution (the same routine used in the
   reference solution, but with many more iterations).
3. Upsample the current best distribution to a larger resolution by
   linear interpolation and continue optimisation – a finer grid
   permits a lower max‑convolution value.
4. Perform a second Adam phase that minimises a smooth approximation
   of the max (soft‑max with a gradually increasing temperature) to
   obtain a more accurate descent direction.
5. Finish with a cheap pair‑wise Frank‑Wolfe (mass‑transfer) step that
   moves probability mass from the component with the largest
   sub‑gradient to the one with the smallest, always respecting the
   simplex constraints.
6. Return the best vector found (as a plain list of floats).

The routine respects the supplied time‑budget (≈ 1000 s) and always
produces a valid sequence that can be fed directly to ``evaluate_sequence``.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate score: 2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (Σa = 1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (subtract a_i for even active indices)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 300000,
                  step0: float = 0.12,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Projected Adam on the true sub‑gradient of max‑convolution."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = 1e-8

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + eps)
        a -= eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 80000,
                          eta0: float = 0.5,
                          decay: float = 0.9999) -> tuple[np.ndarray, float]:

def _smooth_max_gradient(a: np.ndarray, gamma: float) -> np.ndarray:
    """
    Gradient of the smooth (soft‑max) approximation
        f_γ(a) = (1/γ) log Σ_k exp(γ·(a*a)[k]).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)                     # length 2·n‑1
    # soft‑max weights, numerically stabilised
    b_shift = b - b.max()
    w = np.exp(gamma * b_shift)
    w /= w.sum()

    # convolution of reversed a with the weight vector
    conv_rev = _fft_convolve(a[::-1], w)
    slice_rev = conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for the terms i = k/2
    diag = np.zeros_like(a)
    idx = np.arange(n)
    even = (2 * idx) < w.size
    diag[even] = w[2 * idx[even]]

    grad = 2.0 * slice_rev - a * diag
    return grad


def _adam_smooth(a0: np.ndarray, deadline: float,
                 max_iters: int = 200000,
                 step0: float = 0.10,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 gamma0: float = 5.0,
                 gamma_max: float = 200.0,
                 ramp_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smooth soft‑max surrogate.
    The temperature γ is increased linearly from ``gamma0`` to
    ``gamma_max`` during the first ``ramp_iters`` steps.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    eps = 1e-8

    while time.time() < deadline and t < max_iters:
        t += 1
        gamma = gamma_max if t > ramp_iters else gamma0 + (gamma_max - gamma0) * (t / ramp_iters)
        grad = _smooth_max_gradient(a, gamma)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + eps)
        a -= eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(alpha,…,alpha)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0:
        new_a = np.full(new_n, 1.0 / new_n)
    else:
        new_a /= s
    return new_a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Pick a base length; use a previous solution if available
    try:
        base_n = len(height_sequence_1)    # type: ignore[name-defined]
    except Exception:
        base_n = 2000

    n_small = max(500, min(4000, int(base_n * 0.8)))

    # ------------------------------------------------------------------
    # Build a modest collection of seeds
    seeds = [
        _arcsine_distribution(n_small),
        _triangular_distribution(n_small),
        _exponential_distribution(n_small, lam=0.01),
        _beta_symmetric_distribution(n_small, p=0.35),
    ]
    for alpha in (0.2, 0.5, 1.0):
        seeds.append(_random_dirichlet_alpha(n_small, alpha))

    # include the previous best if it matches the size
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, 0.0)
            if prev.sum() > 0:
                prev = prev / prev.sum()
                if prev.shape[0] == n_small:
                    seeds.append(prev)
    except Exception:
        pass

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – heavy Adam on the true sub‑gradient
    for idx, a0 in enumerate(seeds):
        if time.time() >= deadline:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            continue
        a0 = a0 / a0.sum()
        seq, sc = _adam_descent(a0, deadline, max_iters=300000, step0=0.12)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] Seed {idx+1}/{len(seeds)} Adam => {best_score:.6f}")
        # a quick Exp‑Grad polish
        seq2, sc2 = _exp_gradient_descent(seq, deadline, max_iters=80000, eta0=0.5)
        if sc2 < best_score - 1e-14:
            best_score, best_seq = sc2, seq2.copy()
            print(f"[{time.time():.1f}] Seed {idx+1}/{len(seeds)} ExpGrad => {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upscale to a finer grid if time permits
    if best_seq is not None and time.time() < deadline:
        # pick a larger length, roughly double the current one but capped
        n_big = min(12000, max(6000, best_seq.shape[0] * 2))
        a_big = _interpolate_distribution(best_seq, n_big)

        # Adam on the sub‑gradient (larger problem)
        seq_big, sc_big = _adam_descent(a_big, deadline,
                                        max_iters=200000, step0=0.12)
        if sc_big < best_score - 1e-14:
            best_score, best_seq = sc_big, seq_big.copy()
            print(f"[{time.time():.1f}] Upscaled Adam => {best_score:.6f}")

        # Adam on the smooth soft‑max surrogate (helps escape plateaus)
        seq_smooth, sc_smooth = _adam_smooth(best_seq, deadline,
                                            max_iters=150000,
                                            step0=0.10,
                                            gamma0=5.0, gamma_max=200.0,
                                            ramp_iters=2000)
        if sc_smooth < best_score - 1e-14:
            best_score, best_seq = sc_smooth, seq_smooth.copy()
            print(f"[{time.time():.1f}] Smooth‑Adam => {best_score:.6f}")

        # One more Exp‑Grad polish on the refined vector
        seq2, sc2 = _exp_gradient_descent(best_seq, deadline, max_iters=80000, eta0=0.5)
        if sc2 < best_score - 1e-14:
            best_score, best_seq = sc2, seq2.copy()
            print(f"[{time.time():.1f}] Final ExpGrad => {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – cheap Frank‑Wolfe style mass transfer
    if best_seq is not None:
        a = best_seq.copy()
        cur_score = best_score
        it_fw = 0
        while time.time() < deadline and it_fw < 50000:
            it_fw += 1
            grad, _, _ = _subgrad(a)
            i_src = int(np.argmax(grad))
            i_dst = int(np.argmin(grad))
            if i_src == i_dst:
                break
            max_transfer = min(a[i_src], 1.0 - a[i_dst])
            if max_transfer < 1e-14:
                break

            # try moving the whole possible mass
            a_try = a.copy()
            a_try[i_src] -= max_transfer
            a_try[i_dst] += max_transfer
            scr = _max_convolution_score(a_try)

            improved = False
            if scr < cur_score - 1e-12:
                a, cur_score = a_try, scr
                improved = True
            else:
                # half‑transfer fallback
                half = max_transfer * 0.5
                a_half = a.copy()
                a_half[i_src] -= half
                a_half[i_dst] += half
                scr_h = _max_convolution_score(a_half)
                if scr_h < cur_score - 1e-12:
                    a, cur_score = a_half, scr_h
                    improved = True

            if improved and cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] FW iter={it_fw} new best {best_score:.6f}")

            # occasional jitter to get out of tiny plateaus
            if it_fw % 5000 == 0 and np.random.rand() < 0.12:
                jitter = np.random.randn(a.size) * 1e-5
                a = _project_onto_simplex(a + jitter, s=1.0)
                cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n)
        best_score = _max_convolution_score(best_seq)

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    return best_seq.tolist()
