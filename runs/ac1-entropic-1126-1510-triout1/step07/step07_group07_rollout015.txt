import time, math, numpy as np

def _self_convolution_full(a):
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:2 * n - 1]

def _fft_convolve(a,b):
    size = 1 << ((a.size + b.size - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:a.size + b.size -1]

def _max_self_convolution(a):
    return _self_convolution_full(a).max()

def _raw_score(a):
    # a assumed normalized sum=1
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)

def _smooth_obj_and_grad_y(y, mu):
    # softmax
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    a = exp_y / exp_y.sum()
    # convolution
    conv = _self_convolution_full(a)
    M = conv.max()
    # stable log-sum-exp softmax over conv entries
    scaled = (conv - M) / mu
    # to avoid overflow, clip scaled min maybe -700? Not needed, exp small.
    exp_scaled = np.exp(scaled)
    Z = exp_scaled.sum()
    p = exp_scaled / Z  # probabilities for each conv index

    # gradient wrt a
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad_a = 2.0 * tmp[a.shape[0] - 1 : a.shape[0] - 1 + a.shape[0]]

    # gradient wrt y
    G = np.dot(grad_a, a)   # sum_i a_i * grad_a_i
    grad_y = a * (grad_a - G)

    # smoothed objective value
    f_val = M + mu * np.log(Z)

    return f_val, grad_y, a, M

def _beta_distribution(n, alpha, beta):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute log pdf safely
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    # avoid log(0) by clamping x away from 0 and 1:
    eps = np.finfo(np.float64).tiny
    x_clipped = np.clip(x, eps, 1 - eps)
    logpdf = logC + (alpha - 1.0) * np.log(x_clipped) + (beta - 1.0) * np.log(1.0 - x_clipped)
    # subtract max to prevent overflow
    logpdf = logpdf - np.max(logpdf)
    pdf = np.exp(logpdf)
    pdf_sum = pdf.sum()
    if pdf_sum == 0:
        pdf[:] = 1.0 / n
    else:
        pdf /= pdf_sum
    return pdf

def _interpolate_to_length(old_seq, new_n):
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

def _subgradient_polish(a0, time_budget):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _raw_score(a)
    start = time.time()
    it = 0
    # momentum parameters
    momentum = np.zeros_like(a)
    mu_momentum = 0.9
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        it += 1
        # exact subgradient of maxconv
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        # gradient
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        # projection onto simplex (ensure sum=1)
        # simple projection onto simplex
        # Using same projection method as earlier
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1.0
        rho = np.where(u - cssv / np.arange(1, len(a)+1) > 0)[0]
        if rho.size:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        else:
            theta = 0.0
        a = np.maximum(a - theta, 0.0)
        # evaluate
        cur_score = _raw_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            # projection again
            u = np.sort(a)[::-1]
            cssv = np.cumsum(u) - 1.0
            rho = np.where(u - cssv / np.arange(1, len(a)+1) > 0)[0]
            if rho.size:
                rho = rho[-1]
                theta = cssv[rho] / (rho + 1)
            else:
                theta = 0.0
            a = np.maximum(a - theta, 0.0)
    return best_a, best_score

def _adam_optimize(a_init, time_budget):
    # a_init normalized sum=1
    n = a_init.shape[0]
    # initialize y = log probabilities
    eps = 1e-30
    y = np.log(a_init + eps)
    best_y = y.copy()
    best_a = a_init.copy()
    best_score = _raw_score(best_a)
    start = time.time()
    # smoothing schedule
    maxconv0 = _max_self_convolution(a_init)
    mu = max(maxconv0 * 0.5, 1e-6)
    mu_min = 1e-6
    mu_factor = 0.5
    mu_vals = []
    while mu > mu_min:
        mu_vals.append(mu)
        mu = max(mu * mu_factor, mu_min)
    mu_vals.append(mu_min)

    # adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.2
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for mu_cur in mu_vals:
        # reset Adam moments for each mu stage
        m[:] = 0.0
        v[:] = 0.0
        while True:
            if time.time() - start >= time_budget:
                break
            f_val, grad_y, a, M = _smooth_obj_and_grad_y(y, mu_cur)
            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + epsilon)

            # periodically evaluate true raw score
            if (t & 31) == 0:  # every 32 steps
                cur_score = 2.0 * n * M   # M is maxconv
                if cur_score < best_score:
                    best_score = cur_score
                    best_y = y.copy()
                    best_a = a.copy()
        # after mu stage, reinitialize y with best_y for next mu
        y = best_y.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Generate a step‑function sequence minimizing the evaluation objective.
    Uses a softmax‑parametrised Adam optimizer on a smoothed max‑convolution
    objective, followed by a short projected sub‑gradient polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # choose length; larger n gives finer resolution but higher cost
    n = 12000

    candidates = []

    # 0) warm‑start from previous solution if available
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base = np.maximum(base, 0.0)
        base_sum = base.sum()
        if base_sum == 0.0:
            base[:] = 1.0 / base.size
        else:
            base /= base_sum
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # 1) basic deterministic baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_beta_distribution(n, 0.6, 0.6))               # near‑uniform
    candidates.append(_beta_distribution(n, 0.8, 0.8))
    candidates.append(_beta_distribution(n, 1.2, 1.2))

    # 2) symmetric beta families
    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n, a, a))

    # 3) random Dirichlet draws with varied concentration
    for _ in range(8):
        conc = 10 ** rng.uniform(-2, 1)      # between 0.01 and 10
        vec = rng.gamma(conc, 1.0, size=n)
        vec_sum = vec.sum()
        if vec_sum == 0.0:
            vec[:] = 1.0 / n
        else:
            vec /= vec_sum
        candidates.append(vec)

    # 4) a few random mixtures of two beta shapes
    for _ in range(4):
        a1 = 10 ** rng.uniform(-1, 1)
        a2 = 10 ** rng.uniform(-1, 1)
        w = rng.random()
        mix = w * _beta_distribution(n, a1, a1) + (1 - w) * _beta_distribution(n, a2, a2)
        mix_sum = mix.sum()
        if mix_sum == 0.0:
            mix[:] = 1.0 / n
        else:
            mix /= mix_sum
        candidates.append(mix)

    # pick best raw‑score seed
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        scr = _raw_score(seq)
        if scr < best_raw:
            best_raw = scr
            best_seq = seq.copy()
    print(f"Initial best raw score (probability vector): {best_raw:.6f}")

    # time remaining for refinement
    remaining = deadline - time.time()
    if remaining > 20.0:
        # run Adam optimizer
        best_seq, best_raw = _adam_optimize(best_seq, remaining * 0.65)

    # final polishing sub‑gradient
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_raw = _subgradient_polish(best_seq, remaining)

    # final evaluation (evaluate_sequence rescales automatically)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()

"""
Propose a step‑function (non‑negative heights) that minimises

    2·n·max(a*a) / (sum(a)**2)

where “*” is linear convolution.  The optimisation works as follows:

1.  Build a diverse pool of probability vectors (uniform, several symmetric
    Beta‑distributions, random Dirichlet draws and a safe interpolation of the
    supplied `height_sequence_1`).
2.  Pick the best seed according to the **raw** objective.
3.  Parametrise a probability vector with a soft‑max (`a_i = exp(y_i)/∑exp(y)`),
    then run a continuation of Adam optimisation on the *smoothed* max‑convolution
    `F_μ(a)= μ·log(∑exp((a*a)_k/μ))`.  The smoothing parameter μ is halved
    repeatedly (geometric schedule) until μ≈1e‑6.
4.  Finish with a short projected sub‑gradient descent that directly minimises
    the true max‑convolution.
5.  Return the final probability vector; `evaluate_sequence` rescales it
    automatically and computes the true score.

The whole procedure respects the 1000‑second budget, prints a few progress
messages and stays within the required API.
"""

# FFT based convolutions (fast, O(n log n))
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # next power of two ≥2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using real FFT."""
    size = 1 << ((a.size + b.size - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector (∑a=1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. the soft‑max params y
def _smooth_obj_and_grad_y(y: np.ndarray, mu: float):
    """
    Returns (value, grad_y, a, max_conv) where
        a = softmax(y)   (∑a = 1, a_i ≥ 0)
        value = M + mu·log(∑exp((a*a - M)/mu))
        grad_y = ∂value/∂y
    """
    # soft‑max (numerically stable)
    y_shift = y - np.max(y)
    exp_y = np.exp(y_shift)
    a = exp_y / exp_y.sum()

    # self‑convolution of a
    conv = _self_convolution_full(a)               # length 2·n‑1
    M = conv.max()                                 # true max

    # log‑sum‑exp smooth approximation of the max
    scaled = (conv - M) / mu                       # ≤ 0
    exp_scaled = np.exp(scaled)                   # the largest entry is 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                             # soft‑max weights over indices

    # gradient w.r.t. a : 2·(a_rev * p)   (central n entries)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad_a = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # chain rule for the soft‑max parametrisation
    G = np.dot(grad_a, a)                          # Σ_i a_i·grad_a_i
    grad_y = a * (grad_a - G)

    # smoothed objective value
    f_val = M + mu * np.log(Z)

    return f_val, grad_y, a, M

# Simple Beta pdf (safe, no overflow)
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised symmetric Beta(α,β) density, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = np.finfo(np.float64).tiny
    x = np.clip(x, eps, 1 - eps)
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # subtract max to avoid overflow, then exponentiate
    logpdf -= np.max(logpdf)
    pdf = np.exp(logpdf)
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= pdf_sum
    return pdf

# Truncated arcsine (Beta(½,½)) – computed directly, no overflow
def _arcsine_distribution(n: int) -> np.ndarray:
    """U‑shaped arcsine density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    pdf = 1.0 / (np.pi * np.sqrt(x * (1.0 - x)))
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= pdf_sum
    return pdf

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals

# Project onto the simplex (used by the sub‑gradient polish)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    return w

    y = np.log(a_init + eps)

    # best‑so‑far bookkeeping
    best_y = y.copy()
    best_a = a_init.copy()
    best_score = _raw_score(best_a)

    # smoothing schedule (geometric halving)
    maxconv0 = _max_self_convolution(a_init)
    mu = max(maxconv0 * 0.5, 1e-6)
    mu_min = 1e-6
    mu_factor = 0.5
    mus = []
    while mu > mu_min:
        mus.append(mu)
        mu = max(mu * mu_factor, mu_min)
    mus.append(mu_min)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.2
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    start = time.time()

    for mu_cur in mus:
        # reset the moments for each new smoothing level
        m.fill(0.0)
        v.fill(0.0)
        while True:
            if time.time() - start >= time_budget:
                break
            f_val, grad_y, a, M = _smooth_obj_and_grad_y(y, mu_cur)

            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + epsilon)

            # every few iterations evaluate the true (non‑smoothed) score
            if (t & 31) == 0:                # each 32 steps
                cur_score = 2.0 * n * M      # because Σa = 1
                if cur_score < best_score:
                    best_score = cur_score
                    best_y = y.copy()
                    best_a = a.copy()
        # continue with the best parameters found for this μ
        y = best_y.copy()
    return best_a, best_score

# Short sub‑gradient polish (identical to the one used before)
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns (best_a, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        cur_score = _raw_score(a)
        if cur_score < best_val:
            best_val = cur_score
            best_a = a.copy()

        # occasional tiny random jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point – creates a step sequence, optimises it,
    and returns the list of heights.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin for the harness

    # --------------------------------------------------------------
    # 1. Choice of length (larger n gives finer shape, n≈12000 is a good trade‑off)
    n = 12000

    # --------------------------------------------------------------
    # 2. Assemble a pool of candidate probability vectors (all sum to 1)
    candidates = []

    # a) warm‑start from the global solution, if present
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        base = np.maximum(base, 0.0)
        base_sum = base.sum()
        if base_sum == 0.0:
            base[:] = 1.0 / base.size
        else:
            base /= base_sum
        if base.shape[0] != n:
            base = _interpolate_to_length(base, n)
        candidates.append(base)
    except Exception:
        pass

    # b) elementary baselines
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))          # uniform
    candidates.append(_arcsine_distribution(n))                     # arcsine (β=½)

    # c) symmetric Beta families
    for a in (0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0):
        candidates.append(_beta_distribution(n, a, a))

    # d) random Dirichlet draws with varied concentration
    for _ in range(8):
        conc = 10 ** rng.uniform(-2, 1)      # between 0.01 and 10
        vec = rng.gamma(conc, 1.0, size=n)
        vec_sum = vec.sum()
        if vec_sum == 0.0:
            vec[:] = 1.0 / n
        else:
            vec /= vec_sum
        candidates.append(vec)

    # e) random mixtures of two Beta shapes
    for _ in range(4):
        a1 = 10 ** rng.uniform(-1, 1)
        a2 = 10 ** rng.uniform(-1, 1)
        w = rng.random()
        mix = w * _beta_distribution(n, a1, a1) + (1.0 - w) * _beta_distribution(n, a2, a2)
        mix_sum = mix.sum()
        if mix_sum == 0.0:
            mix[:] = 1.0 / n
        else:
            mix /= mix_sum
        candidates.append(mix)

    # --------------------------------------------------------------
    # 3. Pick the seed with the lowest *raw* objective
    best_seq = None
    best_raw = float("inf")
    for seq in candidates:
        sc = _raw_score(seq)
        if sc < best_raw:
            best_raw = sc
            best_seq = seq.copy()
    print(f"Initial best raw score (probability vector): {best_raw:.6f}")

    # --------------------------------------------------------------
    # 4. Adam optimisation on the smoothed problem (majority of the budget)
    remaining = deadline - time.time()
    if remaining > 20.0:
        best_seq, best_raw = _adam_optimize(best_seq, remaining * 0.65)

    # --------------------------------------------------------------
    # 5. Final short projected sub‑gradient polishing
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_seq, best_raw = _subgradient_polish(best_seq, remaining)

    # --------------------------------------------------------------
    # 6. Evaluate with the supplied evaluator (it rescales automatically)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Final evaluated score: {final_score:.6f} (len={len(best_seq)})")

    # --------------------------------------------------------------
    return best_seq.tolist()
