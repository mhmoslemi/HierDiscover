    mu0 = max(_max_self_convolution(initial_a) * 0.5, 1e-7)

    start = time.time()
    iteration = 0
    best_seq = None; best_score = np.inf

    while time.time() - start < time_budget:
        iteration += 1
        # compute mu based on iteration schedule:
        # Use exponentially decaying mu: mu = max(mu_target, mu0 * (0.98**iteration))
        mu = max(mu_target, mu0 * (0.98**iteration))

        # compute a via softmax
        max_x = np.max(x)  # for stability
        exps = np.exp(x - max_x)
        a = exps / exps.sum()

        # compute conv and softmax p for gradient: _smooth_obj_and_grad returns f, grad

        x = x - step

def _adam_refine(a0, time_budget):
    n = len(a0)
    # guarantee a0 sum to 1 (already but ensure)
    a0 = np.maximum(a0, 0)
    total = a0.sum()
    if total == 0: a0 = np.full(n, 1/n)
    else: a0 = a0 / total

    epsilon = 1e-12
    x = np.log(a0 + epsilon)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # initial smoothing parameter
    mu0 = max(_max_self_convolution(a0) * 0.5, 1e-7)
    mu_target = 1e-8
    decay = 0.9995  # per iteration

    best_seq = a0.copy()
    best_raw = _raw_score(a0)  # raw score = 2 n * max_b

    start = time.time()
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        it += 1
        # compute mu
        mu = max(mu_target, mu0 * (decay ** it))

        # compute a via softmax
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # compute smoothed gradient
        _, grad_a = _smooth_obj_and_grad(a, mu)

        # gradient w.r.t. x
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # learning rate schedule: could be decreasing
        lr = 0.5 / (1 + 0.0001 * it)  # start fairly large

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x = x - step

        # Evaluate raw score occasionally
        if (it % 50) == 0:
            # compute actual max conv for a (maybe with direct max)
            max_conv = _max_self_convolution(a)
            raw = 2 * n * max_conv
            if raw < best_raw:
                best_raw = raw
                best_seq = a.copy()
                # print progress?
    # after loop, return best_seq, best_raw
    return best_seq, best_raw

"""
Hybrid multiscale Adam + sub‑gradient optimizer for the step‑function
evaluation problem.

1.  Build a collection of simple probability vectors (uniform, several
    Beta‑distributions, random Dirichlet draws and the supplied warm‑start).
2.  Pick the best seed w.r.t. the cheap proxy 2·n·max(a*a).
3.  Refine the seed on a *small* problem size (n≈2000) with an
    Adam‑style optimiser that works on the soft‑max (logit) parametrisation
    of the simplex and uses a log‑sum‑exp smoothing of the max‑convolution.
    The smoothing parameter μ is decreased geometrically.
4.  Upsample the obtained shape to the final length (default n=8000) and
    continue the Adam refinement.
5.  Finish with a short projected sub‑gradient descent (the routine used
    previously) to polish the last digits.
6.  Return the probability vector; ``evaluate_sequence`` rescales it
    automatically (the objective is scale‑invariant).
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow – clip the exponent before exp
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a * a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector (sum = 1)."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)          # length 2·n‑1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)               # all ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                        # soft‑max weights

    # gradient: 2 * (a_rev * p)   (central n entries)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _adam_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Adam optimiser on the soft‑max (logit) parametrisation of the simplex.
    Returns the best sequence found and its raw score.
    """
    # make sure a0 is a proper probability vector
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s == 0.0:
        a0 = np.full_like(a0, 1.0 / len(a0))
    else:
        a0 = a0 / s

    n = a0.shape[0]
    eps_log = 1e-16
    x = np.log(a0 + eps_log)                 # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing schedule
    mu0 = max(_max_self_convolution(a0) * 0.5, 1e-7)
    mu_target = 1e-6
    mu_decay = 0.9995

    best_seq = a0.copy()
    best_raw = _raw_score(a0)

    start = time.time()
    it = 0
    while True:
        if time.time() - start >= time_budget:
            break
        it += 1

        # current smoothing parameter
        mu = max(mu_target, mu0 * (mu_decay ** it))

        # soft‑max to obtain a
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()

        # smoothed objective & gradient w.r.t. a
        _, grad_a = _smooth_obj_and_grad(a, mu)

        # chain‑rule: gradient w.r.t. logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple learning‑rate schedule
        lr = 0.5 / (1.0 + 1e-4 * it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional assessment of the true (non‑smooth) objective
        if (it & 31) == 0:                     # every 32 iterations
            cur_raw = _raw_score(a)
            if cur_raw < best_raw:
                best_raw = cur_raw
                best_seq = a.copy()
    return best_seq, best_raw


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """Short projected sub‑gradient descent (as used in the reference code)."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # tiny jitter every 500 steps to escape flat regions
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val
