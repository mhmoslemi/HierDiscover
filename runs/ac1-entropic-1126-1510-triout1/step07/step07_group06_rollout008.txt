"""
Propose a step function (a non‑negative sequence) that minimises
the black‑box ``evaluate_sequence`` score.

Algorithm
---------

* **Representation** – we treat the sequence as a probability vector
  (non‑negative, sum = 1).  The objective is homogeneous, so scaling does
  not matter.

* **Search schedule** – several lengths are tried (from a few thousand up
  to twenty thousand).  For each length a collection of plausible
  initial shapes is generated (uniform, arcsine, power‑law tails,
  exponential tails, random Dirichlet draws).  The best shape is refined
  by two stages:

  1. **Smooth‑max Adam** – minimise a log‑sum‑exp surrogate of the
     infinity‑norm of the self‑convolution.  The smoothing parameter
     λ is increased gradually (λ←λ·1.005) so that the optimisation
     progressively focuses on the true maximum.

  2. **Mirror‑descent on the true max** – a multiplicative (KL‑mirror)
     sub‑gradient method that directly minimises the true maximum
     (`max_k (a*a)_k`).  The step size follows a 1/√t schedule.

* **Warm‑starts** – after a length is optimised, the resulting distribution
  is linearly interpolated to the next length and used as a warm start.

* **Time‑budget awareness** – every loop checks the remaining time and
  aborts early if necessary (the function must finish before the supplied
  budget, otherwise it receives a score of –∞).

The combination of diverse initialisation, a well‑tuned Adam phase and a
mirror‑descent phase typically improves the raw score below 1.503 in the
given time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

# ---- Families of initial shapes ---------------------------------------
def _uniform_shape(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _power_shape(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-0.5-i)+eps)^{-α}   (α>0)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps) ** (2 * alpha)
    # The extra sqrt is to keep values comparable; α=0.5 gives arcsine.
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _exponential_shape(n: int, beta: float) -> np.ndarray:
    """a_i ∝ exp(-β * min(i, n‑1‑i)).   Larger β concentrates mass at the edges."""
    i = np.arange(n, dtype=float)
    dist = np.minimum(i, n - 1 - i)
    vals = np.exp(-beta * dist)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _random_dirichlet_start(n: int, alpha_low: float = 0.1, alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    a = g / g.sum()
    return a

def _adam_optimize_smoothed(
    logits0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 4000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.005,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logit parametrisation, minimising the
    smoothed log‑sum‑exp of the self‑convolution.
    Returns the best probability vector (sum = 1) and the corresponding
    raw score 2·n·max(conv).
    """
    x = logits0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- probability vector (simplex) ----
        a = _softmax(x)                     # Σa = 1, a ≥ 0

        # ---- raw objective (for record) ----
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.3f}")

        # ---- smooth max (log‑sum‑exp) ----
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)          # stable
        p = exp_tmp / exp_tmp.sum()              # shape (2n‑1,)

        # ---- gradient wrt a ----
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # ---- chain rule through softmax (gradient wrt logits) ----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- anneal λ (increase smoothing) ----
        lam = lam * lam_factor

        # ---- occasional jitter to shake out of plateaus ----
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score

def _mirror_descent_max(a0: np.ndarray, n: int, deadline: float,
                       max_iters: int = 20000, step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Multiplicative (KL‑mirror) sub‑gradient descent on the true max.
    Returns the best probability vector found and its raw score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _fft_self_convolve_full(a)
    best_score = 2.0 * n * conv.max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        # active set (indices where max is attained, up to tolerance)
        tol = max(1e-14, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size          # uniform over active indices

        # gradient of max (sub‑gradient)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # mirror‑descent (multiplicative) update
        eta = step0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)   # element‑wise, stays positive
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # occasional tiny random perturbation
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror it={it} raw={best_score:.6f}")

    return best_a, best_score

def _optimise_length(
    n: int,
    deadline: float,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n``.
    Returns the best probability vector (sum = 1) and its raw score.
    """
    # ---------- build a list of candidate initial probabilities ----------
    candidates = []

    # warm‑start from previous best (if any)

    for beta in [0.001, 0.003, 0.006, 0.010]:

    for _ in range(4):

    # convert to logits (log‑softmax input) for Adam
    eps = 1e-12
    init_logits = [np.log(cand + eps) for cand in candidates]

    best_score = float("inf")

    for idx, logits0 in enumerate(init_logits):
        if time.time() >= deadline:

        print(f"[{time.time():.1f}] length {n}, init {idx+1}/{len(init_logits)}")
