    velocity = np.zeros_like(a)

    step0 = 0.2
    it = 0
    while time.time() < deadline:

        cur_max = b.max()

def hill_climb(a, n, deadline, best_seq, best_score):
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0
    no_improve = 0
    max_jitter = 1e-5
    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i = i_start
        best_j = k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j
        source = best_i if a[best_i] >= a[best_j] else best_j
        # pick target with low a
        target = np.random.randint(n)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue
        a_snapshot = a.copy()
        conv_snapshot = conv.copy()
        _apply_transfer(a, conv, source, target, delta)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            no_improve = 0
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {best_score:.6f}")
        else:
            a = a_snapshot
            conv = conv_snapshot
            no_improve += 1
        # occasional jitter
        if it % 3000 == 0 and no_improve > 500:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at iter {it}")
    return best_seq, best_score

"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises the objective
``evaluate_sequence``.  The algorithm consists of four stages:

1.  **Broad analytic initialization** – many families of distributions
    (uniform, arcsine, triangular, exponential, beta, sine‑power,
    cosine, Tukey, DPSS) are generated for a set of lengths and the
    best score is kept.

2.  **Mirror‑descent on a soft‑max surrogate** – the max‑convolution
    is replaced by a smooth soft‑max (parameter α).  An exponentiated‑gradient
    (multiplicative) update is performed with a decaying step size and
    projection onto the simplex.

3.  **Nesterov‑style sub‑gradient phase** – the exact active set of the
    current maximal convolution entry is used to compute a sub‑gradient.
    Momentum and a slowly decreasing step size accelerate convergence.

4.  **Targeted hill‑climbing** – the convolution is updated incrementally.
    Mass is transferred from the pair that contributes most to the current
    peak to a random low‑mass index; occasional jitter helps escape plateaus.

All stages respect a 1000 s wall‑clock limit and print intermediate
statistics.  The returned list contains non‑negative numbers and can be
used directly with ``evaluate_sequence``.
"""

import time

import random

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a):
    """Full self‑convolution a*a using real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """
    Fast objective for a vector already normalised to sum = 1:
    2·n·max(a*a)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _apply_transfer(a, conv, i, j, delta):
    """
    Transfer a mass `delta` from index i to index j,
    updating both a and its full self‑convolution array `conv`
    in O(n) time.
    """
    n = a.shape[0]
    if delta <= 0 or i == j:
        return
    a_old = a.copy()                     # linear part uses the *old* a
    conv[i:i + n] -= 2.0 * delta * a_old
    conv[j:j + n] += 2.0 * delta * a_old
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    a[i] -= delta
    a[j] += delta

# Analytic families – each returns a vector with sum = 1

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    """Generalised cosine window: coeff - coeff·cos(2π·i/(n‑1))."""
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_tukey_window(n, alpha=0.5):
    """Tukey window (cosine‑tapered rectangle)."""
    try:
        from scipy.signal import windows
        vals = windows.tukey(n, alpha, sym=False)
    except Exception:
        # fallback manual implementation
        if alpha <= 0:
            vals = np.ones(n)
        elif alpha >= 1:
            i = np.arange(n, dtype=np.float64)
            vals = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        else:
            i = np.arange(n, dtype=np.float64)
            period = n - 1
            vals = np.ones(n, dtype=np.float64)
            first = i < alpha * period / 2
            vals[first] = 0.5 * (1 + np.cos(np.pi * (2 * i[first] / (alpha * period) - 1)))
            last = i >= (1 - alpha / 2) * period
            vals[last] = 0.5 * (1 + np.cos(np.pi * (2 * i[last] / (alpha * period) - 2 / alpha + 1)))
    vals = np.maximum(vals, 0)
    if vals.sum() == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / vals.sum()
    return vals


def _distribution_dpss(n, NW=2.5):
    """Discrete Prolate Spheroidal Sequence (first DPSS)."""
    try:
        from scipy.signal import windows
        vec = windows.dpss(n, NW, Kmax=1, sym=False)[0]
    except Exception:
        # fallback: simple Gaussian‑like shape
        idx = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        sigma = n / 5.0
        vec = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    vec = np.maximum(vec, 0.0)
    if vec.sum() == 0:
        vec = np.full(n, 1.0 / n)
    else:
        vec = vec / vec.sum()
    return vec

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    Search for a step‑function that minimises `evaluate_sequence`.
    The algorithm combines analytic seeds, mirror‑descent, sub‑gradient
    refinement and a targeted hill‑climber while respecting a 1000 s
    wall‑clock budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0  # spare margin for finalisation
    eps = 1e-12

    # --------------------------------------------------------
    # 1. Analytic families – choose the best seed
    # --------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 10000, 12000]
    best_seq = None
    best_score = float('inf')
    best_n = None

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- generate a rich set of candidates -----
        cand = [
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.002),
            _distribution_exponential(n, lam=0.01),
        ]

        # beta shapes
        for p in (0.2, 0.5, 1.0, 2.0, 3.0, 5.0):
            cand.append(_distribution_beta_sym(n, p))

        # sine‑power windows
        for pw in (0.2, 0.5, 1.0, 2.0, 4.0, 8.0):
            cand.append(_distribution_sine_power(n, pw))

        # cosine‑window coefficients
        for c in (0.3, 0.42, 0.5, 0.58, 0.66):
            cand.append(_distribution_cosine_window(n, c))

        # Tukey windows
        for a in (0.1, 0.3, 0.5, 0.7, 0.9):
            cand.append(_distribution_tukey_window(n, a))

        # DPSS (if SciPy is present)
        for NW in (2.0, 2.5, 3.0, 3.5):
            cand.append(_distribution_dpss(n, NW))

        # a random sparse start
        sparse = np.zeros(n, dtype=np.float64)
        k = max(2, n // 250)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k)
        sparse /= sparse.sum()
        cand.append(sparse)

        # random convex combinations of two families
        for _ in range(4):
            f1, f2 = random.sample(cand, 2)
            w = random.random()
            combo = w * f1 + (1 - w) * f2
            combo = np.maximum(combo, eps)
            combo /= combo.sum()
            cand.append(combo)

        # ----- evaluate all candidates (fast) -----
        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] n={n} new init best {sc:.6f}")

    # --------------------------------------------------------
    # 2. Try to reuse a previous construction if available
    # --------------------------------------------------------
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = len(prev)
                print(f"[{time.time():.1f}] previous best {sc:.6f}")
    except Exception:
        pass

    # Fallback – should never trigger.
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform {best_score:.6f}")

    # --------------------------------------------------------
    # 3. Mirror‑descent with soft‑max surrogate
    # --------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # schedule of (α, steps) – α grows to make the soft‑max sharper
    schedule = [
        (1.0,   4000),
        (5.0,   4000),
        (20.0,  8000),
        (100.0, 8000),
        (500.0, 8000),
        (2000.0,8000),
        (5000.0,4000),
    ]

    eta0 = 0.30                # initial multiplicative step size
    step_counter = 0
    best_seq_md = a.copy()
    best_score_md = best_score

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                break
            step_counter += 1

            # self‑convolution
            b = _fft_self_convolve(a)
            maxb = b.max()

            # soft‑max weights over convolution entries
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                continue
            w /= w_sum

            # gradient of the smooth max (mirror‑gradient)
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # multiplicative (exponentiated) update
            eta = eta0 / (math.sqrt(step_counter) + 1e-12)
            a = a * np.exp(-eta * grad)

            # project back onto the simplex and clean tiny values
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a = a / a.sum()

            # occasional bookkeeping
            if step_counter % 250 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score_md - 1e-12:
                    best_score_md = cur
                    best_seq_md = a.copy()
                    print(f"[{time.time():.1f}] MD step={step_counter} α={alpha} new best {cur:.6f}")

    a = best_seq_md.copy()
    best_seq = best_seq_md.copy()
    best_score = best_score_md

    # --------------------------------------------------------
    # 4. Exact max‑sub‑gradient refinement (Nesterov‑style)
    # --------------------------------------------------------
    velocity = np.zeros_like(a)
    beta = 0.9
    step0 = 0.20
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps_rel = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(b >= cur_max - eps_rel)

        w = np.zeros_like(b)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / (it ** 0.70)            # slowly decaying step size
        a = a - eta * velocity

        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a = a / a.sum()

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-12:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] Subgrad iter={it} new best {cur:.6f}")

        # leave a few seconds for the final hill‑climber
        if time.time() + 2.0 >= deadline:
            break

    # --------------------------------------------------------
    # 5. Targeted hill‑climber working on the convolution directly
    # --------------------------------------------------------
    conv = _fft_self_convolve(a)          # full self‑convolution (length 2·n‑1)
    M = conv.max()
    it = 0
    it_no_improve = 0
    max_jitter = 1e-5

    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())            # index of current peak

        i_start = max(0, k - (n - 1))
        i_end   = min(n - 1, k)

        # find the pair (i, j) that contributes most to the peak
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.random.randint(n)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * np.random.rand()
        if delta <= eps:
            continue

        a_snapshot = a.copy()
        conv_snapshot = conv.copy()
        _apply_transfer(a, conv, source, target, delta)

        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            best_score = 2.0 * n * M
            best_seq = a.copy()
            it_no_improve = 0
            if it % 200 == 0:
                print(f"[{time.time():.1f}] hill_iter={it} new best {best_score:.6f}")
        else:
            a = a_snapshot
            conv = conv_snapshot
            it_no_improve += 1

        # occasional jitter to get out of plateaux
        if it % 3000 == 0 and it_no_improve > 500:
            jitter = np.random.randn(n) * max_jitter
            a += jitter
            a = np.maximum(a, eps)
            a = a / a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            it_no_improve = 0
            print(f"[{time.time():.1f}] jitter applied at iter {it}")

    final_score = 2.0 * best_n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_n})")
    return best_seq.tolist()
