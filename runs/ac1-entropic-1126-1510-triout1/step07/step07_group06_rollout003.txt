"""
Multi‑stage optimisation for the step‑function attack.

Stage‑1 : Warm‑started Adam on a smooth approximation of
          maxₖ (a*a)ₖ (log‑sum‑exp).  Several initialisations
          (uniform, Dirichlet, arcsine, power‑law) are tried for a
          sequence length that grows from a few thousand up to ~5·10⁴.
          The Adam loop checks the real objective (max‑convolution)
          and remembers the best iterate.

Stage‑2 : Projected sub‑gradient descent on the true max‑convolution.
          Uses the active set of indices where the convolution attains
          its maximum; a decreasing step‑size 0.2/√t is employed.
          Random jitter occasionally perturbates the iterate to escape
          plateaus.

All operations respect the time budget (default 1000 s).  The final
sequence is returned as a plain Python list ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returning a probability vector."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (edges heavier, centre lighter)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _powerlaw_start(n: int, p: float) -> np.ndarray:
    """Power‑law family a_i ∝ ((i+½)(n‑½‑i))^{-p}."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    base = (i + 0.5) * (n - 0.5 - i) + eps
    vals = base ** (-p)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray, n: int,
                            deadline: float,
                            max_iters: int = 8000,
                            lr0: float = 0.15,
                            lam_start: float = 0.2,
                            lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smooth approximation of
    maxₖ (a*a)ₖ.  Returns the best probability vector found and its
    unsmoothed raw score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector on the simplex
        a = _softmax(x)

        # raw objective (no smoothing)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it} raw={best_score:.6f} λ={lam:.3f}")

        # smooth max via log‑sum‑exp
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()           # shape (2n‑1,)

        # gradient of smooth max w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # chain rule through soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase λ
        lam = lam * lam_factor

        # occasional jitter to help escape flat regions
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.008

    # fallback in the unlikely case that no improvement was recorded
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()

    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length(n: int, deadline: float,
                     prev_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a distribution of length `n` using several initialisations
    and the Adam optimiser above.  Returns the best sequence (as a
    probability vector) together with its raw score.
    """
    init_logits = []

    # Warm‑start from a previous solution if available
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        init_logits.append(np.log(cand + 1e-12))

    # Uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # Dirichlet draws
    for _ in range(3):
        a = _random_dirichlet_start(n)
        init_logits.append(np.log(a + 1e-12))

    # Deterministic arcsine shape
    init_logits.append(np.log(_arcsine_start(n) + 1e-12))

    # A few random power‑law exponents
    for _ in range(2):
        p = np.random.uniform(0.2, 1.0)
        init_logits.append(np.log(_powerlaw_start(n, p) + 1e-12))

    best_seq = None
    best_score = float("inf")

    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length={n}, init #{idx + 1}/{len(init_logits)}")
        seq, score = _adam_optimize_smoothed(
            x0, n, deadline,
            max_iters=8000,
            lr0=0.15,
            lam_start=0.2,
            lam_factor=1.001,
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"   >>> new local best raw={best_score:.6f}")

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_seq).max()

    return best_seq, best_score

        else:

    except Exception:

    # ------------------------------------------------------------------
    # Length schedule – progressively larger allows more freedom.
    # The final step may be skipped if time runs out.

        print(f"  -> length {n} finished, best raw = {score:.6f}")
        if score < best_score:

            best_seq = seq.copy()
            best_n = n
        # warm‑start for the next (larger) length

    # ------------------------------------------------------------------
    # Final fine‑tuning with a projected sub‑gradient method on the true

                best_seq = a.copy()
                print(f"[{time.time():.1f}] post‑refine it={it} raw={best_score:.6f}")

    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
