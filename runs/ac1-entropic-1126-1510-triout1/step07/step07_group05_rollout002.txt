"""
Propose a step function (non‑negative heights) with a low evaluation score.

Algorithm
---------
* Work on the probability simplex (sum = 1) – scaling does not affect the
  score.
* For each chosen length we run several starts:
    - Warm‑start from the current best (linear interpolation).
    - Uniform and Dirichlet draws (moderate and very spiky).
* Each start undergoes two optimisation phases:
    1. Adam on a smooth approximation of the max‑convolution (log‑sum‑exp)
       with a temperature that is increased linearly up to a large value.
       The parameters (iterations, learning‑rate, final temperature) are
       adapted to the sequence length.
    2. Projected sub‑gradient descent on the exact max‑convolution.
* After the two phases a cheap local random‑move hill‑climber tries to
  push the raw score a little further.
* The best sequence found over all lengths is returned.
* The whole routine is time‑boxed (≈ budget_s seconds).  All heavy loops
  check the deadline and stop early if necessary.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power‑of‑two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution a*b using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised vector onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Exact raw score = 2·n·max(a*a) (sum a is assumed 1)."""
    n = a.shape[0]
    return 2.0 * n * _fft_self_convolve_max(a)


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(logits: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 6000,
                            tau0: float = 0.5,
                            tau_max: float = 100.0,
                            lr0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parametrisation with a log‑sum‑exp
    surrogate for the max‑convolution.
    Returns the best found probability vector and its raw score.
    """
    x = logits.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = None
    best_score = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # -- soft‑max to obtain a probability vector --
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()          # Σa = 1, a ≥ 0

        # -- exact max‑convolution (used for temperature schedule) --
        conv = _fft_self_convolve_full(a)      # length 2n‑1
        cur_max = conv.max()
        raw_score = 2.0 * n * cur_max
        if raw_score < best_score:
            best_score = raw_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}")

        # -- temperature scheduling (linear increase) --
        tau = tau0 + (tau_max - tau0) * (it - 1) / max_iters

        # -- smooth‑max weights (stable log‑sum‑exp) --
        max_conv = cur_max
        exps = np.exp(tau * (conv - max_conv))
        w = exps / exps.sum()

        # -- gradient of smooth‑max w.r.t. a --
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # -- chain rule through the soft‑max (mirror‑gradient) --
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # -- Adam update --
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to avoid plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback – should never happen
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of max_k (a*a)_k.
    Returns (grad, cur_max) where grad has shape a.shape.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance scales with the current maximum
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_by_subgradient(a: np.ndarray,
                           deadline: float,
                           start_iter: int = 0,
                           max_iters: int = 200000,
                           step0: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the refined vector and its raw score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _raw_score(a)
    it = start_iter

    while time.time() < deadline and it < max_iters:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = step0 / math.sqrt(it)          # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad it={it}  raw={best_score:.6f}")

        # tiny random perturbations help cross flat regions
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

    return best_a, best_score

def _local_random_tweak(a: np.ndarray,
                        deadline: float,
                        cur_raw: float,
                        max_trials: int = 5000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: repeatedly move a small mass from a high‑value
    index to a low‑value index and keep the move if the raw score improves.
    """
    n = len(a)
    best_a = a.copy()
    best_score = cur_raw
    trials = 0
    while time.time() < deadline and trials < max_trials:
        trials += 1
        # pick source index proportionally to its weight (bias towards large entries)
        src = np.random.choice(n, p=best_a)
        dst = np.random.randint(n)
        if src == dst:
            continue
        # propose moving a modest fraction (up to 10 %) of the source mass
        frac = np.random.rand() * 0.1
        delta = best_a[src] * frac
        if delta <= 0:
            continue
        new_a = best_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        # ensure non‑negative (src already checked)
        raw_new = _raw_score(new_a)
        if raw_new < best_score:
            best_score = raw_new
            best_a = new_a
            # optional progress print; keep it sparse
            if trials % 1000 == 0:
                print(f"[{time.time():.1f}] tweak trial={trials} raw={best_score:.6f}")
    return best_a, best_score

                print(f"   *** new local best for n={n}: raw={local_best_raw:.6f}")

        # Update global best if this length produced improvement.
        if local_best_seq is not None and local_best_raw < best_raw:
            best_raw = local_best_raw
            best_seq = local_best_seq.copy()
            best_n = n
            print(f">>> Updated global best: raw={best_raw:.6f} (length={best_n})")

    # ------------------------------------------------------------------
    # Final refinement on the current global best (if any).
    if best_seq is not None:
        print("\n=== Final refinement on global best ===")
        best_seq, best_raw = _refine_by_subgradient(best_seq, deadline)
        best_seq, best_raw = _local_random_tweak(best_seq, deadline, best_raw, max_trials=5000)
    else:
        # Fallback to a modest uniform sequence.
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Convert to plain Python list and evaluate through the supplied black‑box.
    final_list = best_seq.tolist()
    final_score = evaluate_sequence(final_list)   # supplied by the harness
    print(f"\nFinal evaluated score: {final_score:.6f} (length={len(final_list)})")
    return final_list
