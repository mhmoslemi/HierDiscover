def _simulated_annealing(a, deadline, best_seq, best_score):
    n = a.shape[0]

"""
Multi‑stage optimizer for the step‑function evaluation problem.

The algorithm proceeds as follows:

1. **Rich initialisation** – generate a large set of analytic families
   (uniform, arcsine, triangular, symmetric beta, exponential,
   alternating, two‑block, power‑law, sparse random) for several
   lengths (up to 20 000).  The best vector (in terms of the exact
   objective 2·n·max(conv)/sum²) is kept as a seed.

2. **Extended smoothed Adam** – minimise a smooth approximation of
   max‑convolution using the log‑sum‑exp trick.  The smoothing parameter
   α is increased progressively (1 → 5 → 20 → 100 → 500 → 2000 →
   5000 → 10000).  Each stage performs a few thousand Adam steps,
   always projecting onto the probability simplex.

3. **Exact sub‑gradient descent** – switch to the genuine sub‑gradient
   of the max‑convolution and continue optimisation with momentum and a
   diminishing stepsize.

4. **Aggressive greedy swaps** – move a sizable fraction of mass from
   the index with the largest active‑set gradient to the index with the
   smallest gradient, accepting only improving moves.

5. **Large‑move simulated annealing** – a Metropolis walk that may
   transfer the entire mass of a source index to a destination index.
   The temperature is cooled exponentially.

6. **Random restarts** – if time permits, perturb the current best
   solution slightly and run a short sub‑gradient phase; keep any
   improvement.

All stages respect a global deadline (default 1000 s).  The function
`propose_candidate` returns the best sequence found as a plain Python
list of floats.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta with shape p (p>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_alternating(n: int, stride: int = 2) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    a[::stride] = 1.0
    if a.sum() == 0:
        a[0] = 1.0
    a /= a.sum()
    return a


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_power_law(n: int, s: float = 0.5) -> np.ndarray:
    """Symmetric power‑law: weight ∝ (distance to nearest end)^(−s)."""
    idx = np.arange(n, dtype=np.float64)
    d = np.minimum(idx + 1, n - idx)  # 1‑based distance to nearest end
    vals = d ** (-s)
    vals /= vals.sum()
    return vals


def _initial_candidates_extended(n: int) -> list[np.ndarray]:
    """Broad set of plausible start vectors for length n."""
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # symmetric beta – many shapes
    for p in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # exponentials
    for lam in (0.001, 0.002, 0.003, 0.005, 0.01, 0.02):
        cand.append(_distribution_exponential(n, lam))

    # alternating patterns
    for stride in (2, 3, 4):
        cand.append(_distribution_alternating(n, stride))

    # two‑block spikes at the ends
    cand.append(_distribution_two_blocks(n, block_frac=0.05))
    cand.append(_distribution_two_blocks(n, block_frac=0.10))

    # power‑law families (U‑shaped)
    for s in (0.2, 0.5, 0.8, 1.2):
        cand.append(_distribution_power_law(n, s))

    # random sparse start (≈0.5 % non‑zeros)
    k = max(2, n // 200)
    rnd = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    rnd[idx] = np.random.rand(k)
    rnd /= rnd.sum()
    cand.append(rnd)

    return cand

# Stage 1 – smoothed Adam descent --------------------------------------
def _smooth_adam_descent(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp smooth approximation of max‑convolution."""
    n = a.shape[0]

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # smoothing schedule – progressively tighter approximation
    schedule = [
        (1.0,   2000),
        (5.0,   2000),
        (20.0,  2000),
        (100.0, 2000),
        (500.0, 2000),
        (2000.0,2000),
        (5000.0,2000),
        (10000.0,2000),
    ]

    for alpha, steps in schedule:
        for _ in range(steps):
            if time.time() >= deadline:
                return best_seq, best_score

            t += 1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # soft‑max weights (log‑sum‑exp)
            w = np.exp(alpha * (b - maxb))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(b, 1.0 / b.size)
            else:
                w /= w_sum

            # gradient of the smooth max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # simplex projection
            a = _project_onto_simplex(a, 1.0)

            # occasional exact evaluation
            if t % 150 == 0:
                cur = _objective(a)
                if cur < best_score - 1e-13:
                    best_score = cur
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] smooth α={alpha:.0f} step={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 2 – exact sub‑gradient descent ---------------------------------
def _exact_subgradient_descent(a: np.ndarray, deadline: float,
                               best_seq: np.ndarray,
                               best_score: float) -> tuple[np.ndarray, float]:
    """Sub‑gradient descent on the genuine max‑convolution objective."""
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.8            # a bit less momentum than Adam
    step0 = 0.50          # larger initial step
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (b >= cur_max - eps).astype(float)

        # sub‑gradient from the active set
        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        # momentum + diminishing stepsize
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = step0 / math.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, 1.0)

        cur = _objective(a)
        if cur < best_score - 1e-13:
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it} new best {best_score:.6f}")

        # early exit if we have already reached the target
        if best_score <= 1.503:
            break

        # cheap time check
        if time.time() + 0.5 >= deadline:
            break

    return best_seq, best_score

# Stage 3 – greedy active‑set swapping ---------------------------------
def _greedy_swap_opt(a: np.ndarray, deadline: float,
                     best_seq: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """Move a sizable fraction of mass from high‑gradient to low‑gradient indices."""
    n = a.shape[0]
    while time.time() < deadline:
        b = _fft_self_convolve(a)
        maxb = b.max()
        eps = maxb * 1e-12 + 1e-12
        active = (b >= maxb - eps).astype(float)

        conv_rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = src_candidates[np.argmax(grad[src_candidates])]
        j = int(np.argmin(grad))

        if i == j:
            break

        # transfer up to half of the source mass (but never more than 0.02)
        delta = min(a[i] * 0.7, 0.02)
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        cur = _objective(a_new)

        if cur < best_score - 1e-13:
            a = a_new
            best_seq = a.copy()
            best_score = cur
            print(f"[{time.time():.1f}] greedy swap improved best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stage 4 – simulated annealing -----------------------------------------
def _simulated_annealing(a: np.ndarray, deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Metropolis walk with large mass transfers and exponential cooling."""
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _objective(cur_a)

    T = 0.20                     # initial temperature (objective units)
    iter_cnt = 0

    while time.time() < deadline and T > 1e-5:
        iter_cnt += 1

        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        # pick a destination different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to the whole source mass)
        delta = np.random.rand() * cur_a[i]
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter={iter_cnt} new best {best_score:.6f}")

        # exponential cooling
        if iter_cnt % 2500 == 0:
            T *= 0.85

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best vector found as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # 1. Choose a set of lengths – longer vectors tend to approach the
    #    continuous optimum but are slower to optimise.
    lengths = [2000, 4000, 8000, 12000, 16000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # 2. Scan analytic families for each length.
    for n in lengths:
        if time.time() >= deadline:
            break
        for cand in _initial_candidates_extended(n):
            if time.time() >= deadline:
                break
            cand = cand / cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # 3. Incorporate any previously discovered sequence (if available).
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
    except Exception:
        pass

    # 4. Fallback – never should happen, but keep a safe default.
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Main optimisation pipeline (all stages are time‑aware).
    a = overall_best_seq.copy()

    # 5a – smoothed Adam descent (more aggressive schedule)
    a, overall_best_score = _smooth_adam_descent(a, deadline,
                                                 best_seq=a,
                                                 best_score=overall_best_score)

    # 5b – exact sub‑gradient refinement
    a, overall_best_score = _exact_subgradient_descent(a, deadline,
                                                       best_seq=a,
                                                       best_score=overall_best_score)

    # 5c – greedy active‑set swapping
    a, overall_best_score = _greedy_swap_opt(a, deadline,
                                             best_seq=a,
                                             best_score=overall_best_score)

    # 5d – simulated annealing fine‑tuning
    a, overall_best_score = _simulated_annealing(a, deadline,
                                                 best_seq=a,
                                                 best_score=overall_best_score)

    # 6. Random restarts with short sub‑gradient bursts (if time remains)
    while time.time() + 5 < deadline:
        # small random perturbation of the current best
        pert = a.copy()
        noise = (np.random.rand(*pert.shape) - 0.5) * 0.02
        pert += noise
        pert = np.maximum(pert, 0.0)
        pert = _project_onto_simplex(pert, 1.0)

        # short sub‑gradient run (≈30 s)
        sub_deadline = min(deadline, time.time() + 30.0)
        pert, _ = _exact_subgradient_descent(pert, sub_deadline,
                                             best_seq=pert,
                                             best_score=_objective(pert))

        sc = _objective(pert)
        if sc < overall_best_score - 1e-13:
            overall_best_score = sc
            a = pert.copy()
            print(f"[{time.time():.1f}] random restart improved best {overall_best_score:.6f}")

    print(f"Final best score: {overall_best_score:.6f} (length={len(a)})")
    return a.tolist()
