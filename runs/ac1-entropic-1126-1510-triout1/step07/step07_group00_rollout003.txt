"""
Propose a non‑negative step function (list of floats) that minimises
``evaluate_sequence``.  The algorithm consists of several phases:

1. **Broad candidate generation** – analytical families (arcsine, triangular,
   exponential, symmetric and asymmetric Beta, cosine, power‑law) and random
   Dirichlet draws for a handful of lengths.  All candidates are ℓ₁‑normalised.
2. **Seed selection** – keep the top‑K candidates (by the surrogate
   `2·n·max(conv(a,a))`, exact when the sum equals 1).
3. **Intensive refinement** – for each seed run Adam and exponentiated‑gradient
   mirror descent on the simplex, followed by a cheap random‑perturbation
   hill‑climb.  The best sequence is updated whenever a lower score appears.
4. **Pairwise Frank‑Wolfe mass transfer** – repeatedly move probability mass
   from the index with the largest sub‑gradient to the one with the smallest,
   accepting full or half transfers when they decrease the surrogate.
5. **Final output** – return the best sequence found before the deadline.

The whole routine respects the supplied time budget (≈ 1000 s) and prints
progress information.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used in the core loop


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate score: 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max(a*a) on the simplex (sum(a)=1).
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives gradient before slicing
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction (subtract a_i for even active indices)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 200000,
                  step0: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam‑style adaptive moment optimisation on the simplex."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 200000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Mirror descent with KL divergence (exponentiated‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(alpha,…,alpha)."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow
    log_w = (p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x)
    w = np.exp(log_w - np.max(log_w))  # shift for numerical stability
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_general_distribution(n: int, a: float, b: float) -> np.ndarray:
    """Asymmetric Beta distribution on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    log_w = (a - 1) * np.log(x) + (b - 1) * np.log(1.0 - x)
    w = np.exp(log_w - np.max(log_w))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _cosine_distribution(n: int) -> np.ndarray:
    """Half‑period cosine, positive on (0,π)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(np.pi * i / n)
    w[w < 0] = 0.0
    w /= w.sum()
    return w


def _powerlaw_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law: w_i ∝ ((i+0.5)*(n-i-0.5))^{alpha}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = ((i) * (n - i + 0.5)) ** alpha
    w[w < 0] = 0.0
    w /= w.sum()
    return w

def _local_random_search(a0: np.ndarray, deadline: float,
                         attempts: int = 20000,
                         sigma_start: float = 0.01) -> tuple[np.ndarray, float]:
    """Simple hill‑climb: random zero‑sum perturbations, accept if they improve."""
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(best)
    sigma = sigma_start
    it = 0
    while it < attempts and time.time() < deadline:
        it += 1
        delta = np.random.randn(best.size) * sigma
        delta -= delta.mean()                     # enforce zero sum
        cand = best + delta
        cand = _project_onto_simplex(cand, s=1.0)
        score = _max_convolution_score(cand)
        if score < best_score - 1e-14:
            best = cand
            best_score = score
            sigma *= 0.94                         # zoom in after success
        else:
            if it % 500 == 0:
                sigma *= 1.02                     # occasional expansion
    return best, best_score
