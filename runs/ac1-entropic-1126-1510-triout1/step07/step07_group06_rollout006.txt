"""
Propose a step function (non‑negative heights) that minimizes
evaluate_sequence().  The algorithm mixes three ideas:

1.  A cheap parametric sweep (power‑law, exponential, arcsine) to obtain
    good baselines for each length.
2.  A smoothed‑max Adam optimiser (log‑sum‑exp surrogate of the
    ℓ∞‑norm of the self‑convolution).  The optimiser is run from the
    best baselines and from a warm‑started interpolation of the previous
    best sequence.
3.  A final projected sub‑gradient polishing stage that works directly on
    the true max‑convolution objective.

Time is split roughly 70 % for the Adam stage (several lengths, a few
restarts each) and the remaining 30 % for the sub‑gradient refinement.
All vectors are kept on the probability simplex (∑a = 1), therefore the
raw objective 2·n·max(a*a) equals the value returned by
evaluate_sequence().
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int,
                            alpha_low: float = 0.1,
                            alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (edges heavier, centre lighter)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


def _power_shape(n: int, beta: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-0.5-i))^{-beta}."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    denom = (i + 0.5) * (n - 0.5 - i) + eps
    vals = 1.0 / np.power(denom, beta)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _exp_shape(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c * distance_to_nearest_edge)."""
    i = np.arange(n, dtype=float)
    dist = np.minimum(i + 0.5, n - 0.5 - i)
    vals = np.exp(-c * dist)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a probability vector a (∑a=1)."""
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 3000,
                            lr0: float = 0.12,
                            lam_start: float = 0.5,
                            lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate of the ℓ∞‑norm.
    Returns the best probability vector found and its raw (unsmoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                         # probability vector
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}  λ={lam:.2f}")

        # ----- smooth max (log‑sum‑exp) -----
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)          # safe exponentiation
        p = exp_tmp / exp_tmp.sum()              # shape (2n‑1,)

        # ----- gradient w.r.t. a (from smooth max) -----
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- anneal λ -----
        lam = lam * lam_factor

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:          # should never happen
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_a)
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_refine(a: np.ndarray,
                        deadline: float,
                        init_step: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Stops when time runs out.  Returns the best vector seen.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _raw_score(best_a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        # indices where the max is attained (within tiny tolerance)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        # gradient of the max (sub‑gradient)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = init_step / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)

        # occasional tiny random perturbation
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it}  raw={best_score:.6f}")

    return best_a, best_score
