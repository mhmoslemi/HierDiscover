def generate_power_shape(n, p, eps=1e-12):
    i = np.arange(n, dtype=float) + 0.5
    denom = i * (n + 0.5 - i)
    vals = np.power(denom, -p, out=np.empty_like(denom))
    vals = np.maximum(vals, eps)
    vals_sum = vals.sum()
    if vals_sum > 0.0:
        vals /= vals_sum
    else:

"""
Propose a step function (non‑negative heights) that minimises
evaluate_sequence.  The algorithm consists of three stages:

1.  A fast parametric sweep.  For several lengths we generate
    families of analytic shapes (Beta‑type, power‑law “arcsine‑like”)
    with a small number of parameters (α or p).  Each shape is
    normalised to a probability vector and its raw score
    2·n·max(conv) is evaluated via an FFT‑based self‑convolution.
    The best shape found so far becomes the warm‑start for the next
    length.

2.  A short Adam optimisation on the smooth log‑sum‑exp
    approximation of the max‑convolution.  The optimisation works
    on the soft‑max logits, respects the simplex constraint and
    uses a continuation on λ (the smoothness parameter).

3.  A cheap hill‑climbing phase that repeatedly mixes the current
    best vector with a fresh Dirichlet sample (convex combination
    stays on the simplex).  If the mixed vector yields a lower raw
    score it is accepted.  This often reduces the last few
    thousandths of the objective.

All work respects a global deadline (default 1000 s) and prints
progress information.  The returned list can be fed directly to
evaluate_sequence.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max, Σsoftmax = 1; all entries non‑negative."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing normalised sequence to a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


# ----------------------------------------------------------------------
def _random_dirichlet_start(n: int, alpha_low: float = 0.1, alpha_high: float = 4.1) -> np.ndarray:
    """Sample a random Dirichlet vector (≈probability distribution)."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Deterministic Beta(α,α) shape (symmetric, edges heavier for α<1)."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    eps = 1e-12
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals = np.maximum(vals, eps)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _power_shape(n: int, p: float) -> np.ndarray:
    """Arcsine‑like family a_i ∝ [(i+½)(n‑½‑i)]^{‑p} (symmetric)."""
    i = np.arange(n, dtype=float) + 0.5
    denom = i * (n + 0.5 - i)
    eps = 1e-12
    vals = np.power(denom, -p, out=np.empty_like(denom))
    vals = np.maximum(vals, eps)
    s = vals.sum()
    if s > 0.0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(conv) for a normalised probability vector."""
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(
    x0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 3000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth max) of the self‑convolution.
    Returns the best probability vector found and its *raw* (unsmoothed) score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                     # probability vector
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_score:
            best_score = cur_raw
            best_a = a.copy()
            # occasional logging
            print(f"[{time.time():.1f}] n={n} it={it}  raw={best_score:.6f}  λ={lam:.3f}")

        # ----- smooth max (log‑sum‑exp) -----
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()                      # shape (2n‑1,)

        # ----- gradient w.r.t. a -----
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # ----- chain rule through soft‑max (logits) -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- increase smoothness -----
        lam = lam * lam_factor

        # occasional jitter to avoid stagnation
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_a)

    return best_a, best_score
