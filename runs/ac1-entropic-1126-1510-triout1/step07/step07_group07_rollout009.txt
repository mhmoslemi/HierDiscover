"""
Improved search for the step‑function heights.

Algorithm
---------

* Start from a small pool of simple probability vectors (uniform,
  arcsine, several symmetric Beta PDFs, random Dirichlet draws and the
  optional global warm‑start `height_sequence_1`).  All vectors are
  normalised to sum = 1.

* Keep the best seed according to the *raw* objective  
        2·n·max(a*a)  (this equals the value returned by evaluate_sequence).

* Refine the best seed by a continuation / FISTA scheme on the smooth
  “log‑sum‑exp” approximation of the max‑convolution:
        F_µ(a) = µ·log Σ_k exp( (a*a)_k / µ )
  The algorithm repeatedly halves the smoothing parameter µ (down to 1e‑6)
  and, for each µ, performs a large number of projected FISTA iterations.
  The projection is the Euclidean projection onto the probability simplex.

  Compared with the previous baseline we allow many more inner iterations
  (up to 20000 per smoothing level) and use a faster reduction
  factor µ←0.5·µ.  This gives the optimiser far more time to converge
  at each smoothing stage.

* After the smoothing phase a short projected sub‑gradient descent is
  applied (the same routine as in the baseline) to polish the final
  solution.

* The final sequence is returned as a plain list; `evaluate_sequence`
  will automatically clip and rescale it.

The whole routine respects the 1000 s time budget and prints intermediate
progress (raw scores and timings) for diagnostics.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Proxy objective for a normalised probability vector."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Returns (value, grad) where grad has the same shape as a.
    """
    conv = _self_convolution_full(a)                 # length 2*n-1
    M = conv.max()
    # stable log‑sum‑exp
    scaled = (conv - M) / mu
    exp_scaled = np.exp(scaled)                     # ≤ 1
    Z = exp_scaled.sum()
    p = exp_scaled / Z                               # soft‑max, Σp = 1

    # gradient: 2·(a_rev * p)  (central n entries only)
    a_rev = a[::-1]
    tmp = _fft_convolve(a_rev, p)
    grad = 2.0 * tmp[len(a) - 1 : len(a) - 1 + len(a)]

    # smoothed value
    f_val = M + mu * np.log(Z)
    return f_val, grad


# ----------------------------------------------------------------------
def _fista_refine(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Continuation + projected FISTA with many inner iterations.
    Returns (best_sequence, best_raw_score).
    """
    a = a0.copy()
    n = a.shape[0]

    mu = max(_max_self_convolution(a) * 0.5, 1e-9)   # initial smoothing
    mu_target = 1e-6
    mu_factor = 0.5                                 # halve µ each outer step
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    while True:
        # Lipschitz constant for the μ‑smoothed objective (safe bound)
        L = max(1.0, 2.0 / mu)

        y = a.copy()
        t = 1.0

        # number of inner FISTA steps – we allow up to 20000
        # or as many as the remaining time permits.
        remaining = time_budget - (time.time() - start)
        if remaining <= 0.0:
            break
        # empirical cost ≈ 0.001 s per iteration for n≈8000
        iters = int(min(20000, max(2000, remaining / 0.0009)))
        if iters <= 0:
            break

        for _ in range(iters):
            f_val, grad = _smooth_obj_and_grad(y, mu)

            # simple back‑tracking (max two doublings of L)
            for _ in range(2):
                a_new = _project_onto_simplex(y - grad / L)
                f_new, _ = _smooth_obj_and_grad(a_new, mu)
                if f_new <= f_val - (np.linalg.norm(grad) ** 2) / (2 * L) + 1e-12:
                    break
                L *= 2.0

            a_prev = a
            a = a_new

            # FISTA extrapolation
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = a + ((t - 1.0) / t_new) * (a - a_prev)
            t = t_new

            # occasional true‑objective check
            if (_ % 31) == 0:
                cur_raw = _raw_score(a)
                if cur_raw < best_val:
                    best_val = cur_raw
                    best_seq = a.copy()

        # Reduce µ, stop when target reached
        mu = max(mu * mu_factor, mu_target)
        if mu <= mu_target + 1e-15:
            break
        # warm‑restart from the best sequence observed at this µ
        a = best_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
def _subgradient_polish(a0: np.ndarray, time_budget: float) -> tuple:
    """
    Short projected sub‑gradient descent to polish the solution.
    Returns (best_seq, best_raw_score).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _raw_score(a)

    start = time.time()
    it = 0
    momentum = np.zeros_like(a)
    mu_momentum = 0.9

    while time.time() - start < time_budget:
        it += 1
        # exact sub‑gradient of max_k (a*a)_k
        conv = _self_convolution_full(a)
        cur_max = conv.max()
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]

        # step‑size schedule
        step = 0.25 / math.sqrt(it)
        momentum = mu_momentum * momentum + (1.0 - mu_momentum) * grad
        a = a - step * momentum
        a = _project_onto_simplex(a)

        # tiny jitter every 500 iterations to escape flats
        if (it % 500) == 0 and np.random.rand() < 0.07:
            a += np.random.normal(scale=5e-7, size=len(a))
            a = _project_onto_simplex(a)

        cur_raw = _raw_score(a)
        if cur_raw < best_val - 1e-12:
            best_val = cur_raw
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Mid‑point discretisation of Beta(α,β) on n bins, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)          # avoid overflow
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals
