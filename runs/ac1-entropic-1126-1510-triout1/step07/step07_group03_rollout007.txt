"""
A fast LP‑based refinement for the “min‑max self‑convolution” problem.

The algorithm:
1. Initialise a probability vector (sum = 1) by a classical analytic shape
   (arcsine) or by the previous global solution `height_sequence_1`.
2. Repeatedly solve the linear programme

       maximise   Σ b_j
       subject to  (a * b)_k ≤ M   for all k,
                   b_j ≥ 0,

   where * denotes the ordinary convolution, a is the current
   vector and M = max_k (a * a)_k.  The LP increases the total mass
   that fits under the same peak M.
   The optimal b is normalised to a probability vector g = b / Σb.
   Replacing a by g strictly reduces the peak M (unless a is already
   optimal).  Only a few LP steps are needed – each solves a dense
   (2n‑1)×n system (n ≈ 2000) with HiGHS, which is well under a second.

3. A short exact sub‑gradient descent cleans up any remaining irregularities.

4. A light simulated‑annealing walk performs a final fine‑tuning.

All steps respect the global time‑budget (default 1000 s).  The best
vector found is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _distribution_arcsine(n: int) -> np.ndarray:
    """Classic arcsine density discretised on n points (Σ = 1)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _build_conv_constraints(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A where (A @ b)_k = (a * b)_k.
    Each column i contains the constant a[i] on rows i … i+n‑1.
    """
    n = a.shape[0]
    K = 2 * n - 1
    A = np.empty((K, n), dtype=np.float64)
    A.fill(0.0)
    for i in range(n):
        A[i:i + n, i] = a[i]
    return A


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  The routine mixes a tiny LP‑based refinement
    with sub‑gradient descent and a short simulated‑annealing phase.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0                     # safety margin

    # ------------------------------------------------------------------
    # 0. Choose a working length.  2000 points give a good trade‑off
    #    between resolution and LP size.
    n = 2000

    # 1. Initialise from the global best if available, otherwise arcsine.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim != 1:
            raise ValueError
        L = prev.size
        if L >= n:
            # random down‑sample without replacement, keep mass where taken
            idx = np.random.choice(L, n, replace=False)
            a = prev[idx].copy()
        else:
            # pad with zeros if the previous vector is shorter
            a = np.concatenate([prev, np.zeros(n - L, dtype=np.float64)])
    except Exception:
        a = _distribution_arcsine(n)

    # Normalise to a probability vector (sum = 1) and protect against
    # pathological zero‑vectors.
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    best_seq = a.copy()
    best_score = _objective(a)
    print(f"[{time.time():.1f}] initial score {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 2. LP‑based refinement (the core of the new method).
    #    We perform a few iterations; each iteration solves a dense LP.
    max_lp_iters = 7
    try:
        from scipy.optimize import linprog
    except Exception as exc:
        print("SciPy not available – skipping LP stage:", exc)
        linprog = None

    if linprog is not None:
        for it in range(1, max_lp_iters + 1):
            if time.time() >= deadline:
                break

            # current peak M
            conv_a = _fft_self_convolve(a)
            M = conv_a.max()

            # build constraints A b ≤ M
            A = _build_conv_constraints(a)
            K = A.shape[0]
            ub = np.full(K, M, dtype=np.float64)

            # linear programme: maximise Σ b_j   ⇔   minimise −Σ b_j
            c = -np.ones(n, dtype=np.float64)

            lp_res = linprog(
                c,
                A_ub=A,
                b_ub=ub,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
            )

            if not lp_res.success:
                print(f"[{time.time():.1f}] LP iteration {it} failed – {lp_res.message}")
                break

            b = lp_res.x
            b_sum = b.sum()
            if b_sum <= 1e-12:
                print(f"[{time.time():.1f}] LP iteration {it} produced zero mass")
                break

            g = b / b_sum                     # new probability vector
            new_score = _objective(g)

            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = g.copy()
                a = g                               # replace current a
                print(f"[{time.time():.1f}] LP iter {it} → new best {best_score:.6f}")
                # if we already beat the target we can stop early
                if best_score <= 1.503:
                    break
            else:
                # No improvement → stop the LP loop
                print(f"[{time.time():.1f}] LP iter {it} no improvement (score {new_score:.6f})")
                break

    # ------------------------------------------------------------------
    # 3. Exact sub‑gradient descent (few quick steps to clean up).
    a = best_seq.copy()
    best_score = _objective(a)
    grad_vel = np.zeros_like(a)
    beta = 0.9
    step0 = 0.3

    for sub_it in range(1, 2001):
        if time.time() >= deadline:
            break
        conv_a = _fft_self_convolve(a)
        cur_max = conv_a.max()
        eps = 1e-12 * (abs(cur_max) + 1.0)
        active = (conv_a >= cur_max - eps).astype(float)

        # sub‑gradient from the active set
        rev = _fft_convolve(a[::-1], active)
        grad = 2.0 * rev[n - 1 : n - 1 + n]

        # momentum + diminishing stepsize
        grad_vel = beta * grad_vel + (1.0 - beta) * grad
        eta = step0 / math.sqrt(sub_it)
        a = a - eta * grad_vel
        a = _project_onto_simplex(a, 1.0)

        cur_score = _objective(a)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] sub‑grad {sub_it} → new best {best_score:.6f}")
            if best_score <= 1.503:
                break

        # occasional early exit when the improvement becomes tiny
        if sub_it % 500 == 0 and best_score <= 1.51:
            # nothing to do, just keep going a bit longer
            pass

    # ------------------------------------------------------------------
    # 4. Light simulated‑annealing fine‑tuning (helps escape tiny plateaux).
    cur_a = best_seq.copy()
    cur_score = _objective(cur_a)
    T = 0.03
    anneal_iter = 0
    while time.time() < deadline and T > 1e-4:
        anneal_iter += 1

        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        # pick a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a modest fraction of the source mass
        delta = np.random.rand() * min(cur_a[i], 0.1 * cur_a[i])
        if delta <= 0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _objective(new_a)

        dE = new_score - cur_score
        if dE < 0 or np.random.rand() < math.exp(-dE / T):
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_seq = cur_a.copy()
                print(f"[{time.time():.1f}] SA iter {anneal_iter} → new best {best_score:.6f}")

        if anneal_iter % 2500 == 0:
            T *= 0.85

    # ------------------------------------------------------------------
    print(f"Final best score {best_score:.6f} (length={len(best_seq)})")
    # The evaluation routine rescales internally, so returning a probability
    # vector (sum≈1) is safe.
    return best_seq.tolist()
