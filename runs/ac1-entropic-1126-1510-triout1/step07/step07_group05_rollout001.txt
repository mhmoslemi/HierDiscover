"""
Propose a low‑score step function.

Algorithm
---------
1. Work on the probability simplex (non‑negative numbers that sum to 1)
   using a soft‑max (log‑its) parametrisation.
2. Optimise a smooth approximation of the max‑convolution:
      max(c) ≈ (1/τ)·log Σ exp(τ·c_i)
   with an exponentially increasing temperature τ.
   Gradient is obtained analytically via FFTs; Adam updates the logits.
3. After the smooth phase perform a short exact‑max sub‑gradient descent
   (projected onto the simplex) to polish the solution.
4. Explore several sequence lengths (4000 → 20000).  For each length
   warm‑start from the best known shape (linear interpolation) and also
   try a handful of random Dirichlet initialisations.
5. Allocate the remaining budget to a final sub‑gradient refinement of the
   globally best sequence.

The implementation respects the 1000 s time limit, uses only NumPy
(and optionally SciPy if available) and prints progress information.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_self_convolve_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _fft_self_convolve_full(a).max()


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, tau: float) -> tuple[float, np.ndarray]:
    """
    Compute a smooth approximation of max(convolution) and its gradient
    w.r.t. a using temperature `tau`.
    Returns (smooth_max, grad_a) where grad_a has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)          # length 2n‑1
    # soft‑max weights
    max_c = conv.max()
    exps = np.exp(tau * (conv - max_c))
    w = exps / exps.sum()
    # gradient of smooth‑max w.r.t. a
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    smooth = (max_c + (1.0 / tau) * np.log(w.sum()))  # = (1/τ)·log Σexp
    return smooth, grad


# ----------------------------------------------------------------------
def _adam_optimize(
    logits0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 8000,
    tau_start: float = 0.2,
    tau_end: float = 800.0,
    lr_start: float = 0.15,
    lr_end: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max parametrisation.
    Returns the best probability vector `a` (sums to 1) and the *raw* score
    2·n·max(a*a).
    """
    x = logits0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_raw = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # ----- soft‑max to obtain a probability vector -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)
        a = exp_x / exp_x.sum()                # a ≥ 0, Σa = 1

        # ----- evaluate raw objective -----
        cur_max = _fft_self_convolve_max(a)
        raw = 2.0 * n * cur_max
        if raw < best_raw:
            best_raw = raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} it={it:5d} raw={best_raw:.6f}")

        # ----- temperature schedule (exponential) -----
        tau = tau_start * (tau_end / tau_start) ** ((it - 1) / max_iters)

        # ----- smooth‑max and its gradient -----
        _, grad_a = _smooth_max_and_grad(a, tau)

        # ----- chain rule through the soft‑max -----
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)           # ∂L/∂logits

        # ----- Adam learning‑rate schedule (exponential) -----
        lr = lr_start * (lr_end / lr_start) ** ((it - 1) / max_iters)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 1200 == 0:
            x += np.random.randn(*x.shape) * 0.008

    # fallback – uniform distribution (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_raw = 2.0 * n * _fft_self_convolve_max(best_a)

    return best_a, best_raw


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Compute a sub‑gradient of max_k (a*a)_k.
    Returns (grad, current_max) where grad has the same shape as a.
    """
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    cur_max = conv.max()
    # active set – tolerance proportional to cur_max
    tol = max(1e-14, cur_max * 1e-12)
    active = np.where(conv >= cur_max - tol)[0]
    w = np.zeros_like(conv)
    if active.size:
        w[active] = 1.0 / active.size
    # gradient = 2 * a_rev * w (restricted to indices of a)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_by_subgradient(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 8000,
    step0: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the exact max‑convolution.
    Returns (a_refined, raw_score).
    """
    n = a.shape[0]
    best_a = a.copy()
    best_raw = 2.0 * n * _fft_self_convolve_max(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = step0 / math.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a)

        # tiny random perturbations now and then
        if it % 1000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 3e-7
            a = _project_onto_simplex(a)

        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw - 1e-12:
            best_raw = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] sub‑grad n={n} it={it:5d} raw={best_raw:.6f}")

    return best_a, best_raw


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals

    print(f"\nFinal raw value (2·n·max_conv): {best_raw:.6f}  (length={len(final_list)})")
    return final_list
