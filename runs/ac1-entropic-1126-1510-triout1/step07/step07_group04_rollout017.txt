exponential‑gradient refinement before the final mirror‑descent stage.

2. Coarse random exploration on lengths 2000‑8000 (as before) to obtain
   a decent baseline.

6. Optionally up‑sample to a larger size (12000‑20000) and repeat
   steps 4‑5.
The whole routine respects the global time budget (≈1000 s) and

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete (mid‑point) Beta(α,β) distribution, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf, stable
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    # avoid overflow
    logpdf = np.minimum(logpdf, 700.0)
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a sequence onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """
    Proxy score for a vector that already satisfies Σa = 1.
    The evaluator’s objective equals 2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Sub‑gradient of f(a)=max_k (a*a)_k  (used by mirror‑descent)."""
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_mirror_descent(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the ℓ∞‑objective."""
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        eta = eta0 / math.sqrt(it)
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)
        a = _project_onto_simplex(a)

        # occasional jitter
        if it % 500 == 0 and rng.random() < 0.06:
            a += rng.normal(scale=5e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] mirror‑descent n={n} it={it} score={best_val:.6f}")
    return best_a, best_val


def _refine_l2_exponential(a: np.ndarray, deadline: float,
                           rng: np.random.Generator,
                           eta0: float = 0.12) -> tuple[np.ndarray, float]:
    """
    Exponential‑gradient descent on the L2 loss
        L = Σ_k ( (a*a)_k – t )²,
    where t = 1/(2n‑1) is the uniform target.
    """
    n = a.shape[0]
    t = 1.0 / (2 * n - 1)
    best_a = a.copy()
    best_val = _raw_score(a)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _self_convolution_full(a)
        diff = conv - t
        a_rev = a[::-1]
        grad = 4.0 * _fft_convolve(a_rev, diff)[n - 1 : n - 1 + n]
        eta = eta0 / math.sqrt(it)
        exponent = -eta * grad
        exponent = np.clip(exponent, -50, 50)
        a = a * np.exp(exponent)
        a = _project_onto_simplex(a)

        # occasional jitter to escape shallow plateaux
        if it % 400 == 0 and rng.random() < 0.05:
            a += rng.normal(scale=7e-7, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[{time.time():.1f}] L2‑refine n={n} it={it} score={best_val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluate_sequence score.
    The routine runs a coarse random search, a fine Beta‑grid search,
    an L2‑exponential‑gradient polishing step, and finally the original
    mirror‑descent refinement.  If time permits it up‑samples to larger
    lengths and repeats the polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # ------------------------------------------------------------------
    # Warm‑start from previous solution if available
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if base_seq.sum() == 0.0:
            raise ValueError
        base_seq = base_seq / base_seq.sum()
    except Exception:
        base_seq = None

    if base_seq is not None:
        best_seq = base_seq.copy()
    else:
        best_seq = np.full(2000, 1.0 / 2000, dtype=np.float64)

    best_score = _raw_score(best_seq)
    best_len = len(best_seq)
    print(f"[{time.time():.1f}] initial raw score = {best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 1 – coarse random exploration (same families as before)
    base_lengths = [2000, 4000, 8000]
    for n in base_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] coarse exploration, length {n}")

        # start from current best, interpolated
        a = _interpolate_to_length(best_seq, n)
        sc = _raw_score(a)
        if sc < best_score:
            best_seq, best_score, best_len = a, sc, n
            print(f"   interpolated best score={best_score:.6f}")

        # deterministic baselines
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        sc = _raw_score(uniform)
        if sc < best_score:
            best_seq, best_score, best_len = uniform.copy(), sc, n
            print(f"   uniform score={best_score:.6f}")

        arcsine = _beta_distribution(n, 0.5, 0.5)
        sc = _raw_score(arcsine)
        if sc < best_score:
            best_seq, best_score, best_len = arcsine.copy(), sc, n
            print(f"   arcsine score={best_score:.6f}")

        # symmetric Beta grid (moderate density)
        alphas = np.linspace(0.30, 0.70, 21)  # 21 values
        for a_alpha in alphas:
            if time.time() >= deadline:
                break
            vec = _beta_distribution(n, a_alpha, a_alpha)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   symBeta α={a_alpha:.3f} score={best_score:.6f}")

        # mixture Beta+Uniform grid (dense)
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        for a_alpha in alphas:
            if time.time() >= deadline:
                break
            beta_vec = _beta_distribution(n, a_alpha, a_alpha)
            for lam in np.linspace(0.0, 0.5, 11):   # λ from 0 to 0.5
                mix = lam * beta_vec + (1.0 - lam) * uniform
                mix = _project_onto_simplex(mix)   # just to be safe
                sc = _raw_score(mix)
                if sc < best_score:
                    best_seq, best_score, best_len = mix.copy(), sc, n
                    print(f"   mixU α={a_alpha:.3f} λ={lam:.2f} score={best_score:.6f}")

        # random draws – a light‑weight supplement
        for _ in range(200):
            if time.time() >= deadline:
                break
            alpha = 10 ** rng.uniform(-2, 0.3)   # focus on small alphas
            vec = _beta_distribution(n, alpha, alpha)
            sc = _raw_score(vec)
            if sc < best_score:
                best_seq, best_score, best_len = vec.copy(), sc, n
                print(f"   randBeta α={alpha:.4f} score={best_score:.6f}")

    print(f"[{time.time():.1f}] after coarse phase best score={best_score:.6f} (len={best_len})")

    # ------------------------------------------------------------------
    # Phase 2 – fine grid on the best length found so far
    n = best_len
    if time.time() < deadline:
        print(f"[{time.time():.1f}] fine grid search on length {n}")
        uniform = np.full(n, 1.0 / n, dtype=np.float64)

        alphas_fine = np.concatenate([
            np.linspace(0.30, 0.70, 81),   # very fine linear grid
            10 ** np.linspace(-2, -0.1, 40)  # log‑grid for very small alphas
        ])
        for a_alpha in alphas_fine:
            if time.time() >= deadline:
                break
            beta_vec = _beta_distribution(n, a_alpha, a_alpha)
            # pure Beta
            sc = _raw_score(beta_vec)
            if sc < best_score:
                best_seq, best_score = beta_vec.copy(), sc
                print(f"   fineBeta α={a_alpha:.4f} score={best_score:.6f}")

            # mixture with uniform (λ∈[0,0.4])
            for lam in np.linspace(0.0, 0.4, 9):
                mix = lam * beta_vec + (1.0 - lam) * uniform
                mix = _project_onto_simplex(mix)
                sc = _raw_score(mix)
                if sc < best_score:
                    best_seq, best_score = mix.copy(), sc
                    print(f"   fineMix α={a_alpha:.4f} λ={lam:.2f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – L2 exponential‑gradient polishing
    if time.time() < deadline:
        print(f"[{time.time():.1f}] L2 exponential‑gradient polishing (len={best_len})")
        best_seq, best_score = _refine_l2_exponential(best_seq, deadline, rng, eta0=0.12)
        print(f"[{time.time():.1f}] after L2 polish score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final mirror‑descent refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final mirror‑descent refinement")
        best_seq, best_score = _refine_mirror_descent(best_seq, deadline, rng, eta0=0.42)
        print(f"[{time.time():.1f}] after mirror‑descent score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – optional up‑sampling to larger lengths (if time permits)
    larger_lengths = [12000, 16000, 20000]
    for n_big in larger_lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] up‑sampling to length {n_big}")
        a_big = _interpolate_to_length(best_seq, n_big)
        a_big = _project_onto_simplex(a_big)
        # a quick L2 polish followed by mirror descent
        a_big, scr = _refine_l2_exponential(a_big, deadline, rng, eta0=0.10)
        a_big, scr = _refine_mirror_descent(a_big, deadline, rng, eta0=0.35)
        if scr + 1e-12 < best_score:
            best_seq, best_score = a_big.copy(), scr
            print(f"   improved after up‑sample: score={best_score:.6f} (len={n_big})")
        if best_score <= 1.503:
            print("Target bound reached – exiting early.")
            break

    # ------------------------------------------------------------------
    # Final exact evaluation (clipping, scaling, etc. are handled internally)
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.1f}] final evaluated score: {final_score:.6f} (len={len(best_seq)})")
    return best_seq.tolist()
