   simple analytic shapes  
   `a_i(p) ∝ ((i+0.5)*(n-0.5-i)+ε)^{-p}` (the classic arcsine corresponds to
   `p=0.5`).  Each shape is mixed with the uniform distribution using a
   weight `w`.  All candidates are normalised to sum = 1 and the raw
   score `2·n·max(conv(a,a))` is evaluated via an FFT‑based convolution.
   The best candidate from this cheap scan is taken as a warm‑start.

2. **Smooth‑max Adam optimisation** – Starting from the best grid point
   we run Adam on the logits of the probability vector.  The non‑smooth
   max is approximated by a log‑sum‑exp with an annealing temperature
   `λ`.  The gradient of the smoothed objective is
   `∇ = 2·conv_rev(softmax(λ·conv(a)))`.  `λ` is increased geometrically up
   to a ceiling (`λ_max≈50`) so that the optimisation eventually targets
   the true maximum.

3. **Projected sub‑gradient polishing** – After Adam we switch to a
   true sub‑gradient method for the non‑smooth objective.  The active
   set (indices where the convolution attains its maximum) receives a
   uniform weight and the gradient `∇ = 2·conv_rev(weight)` is used.
   A diminishing step size `α_t = 0.07/√t` and Euclidean projection onto the
   probability simplex keep the iterates feasible.

4. **Fallback & final evaluation** – If any stage fails to produce a
   candidate we fall back to the uniform distribution.  The best sequence
   found (as a plain list) is returned; the harness will call
   `evaluate_sequence` on it.

The whole routine respects the supplied time budget (default 1000 s)
by checking the wall‑clock after each major loop.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _raw_score(a: np.ndarray) -> float:
    """Raw objective 2·n·max(conv(a,a)) for a normalised (Σa=1) vector."""
    n = a.shape[0]
    conv = _fft_self_convolve_full(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _power_shape(p: float, n: int) -> np.ndarray:
    """Return a normalised shape a_i ∝ ((i+0.5)*(n‑0.5‑i)+ε)^{-p}."""
    i = np.arange(n, dtype=float)
    eps = 1e-18
    vals = ((i + 0.5) * (n - 0.5 - i) + eps) ** (-p)
    s = vals.sum()
    if s == 0:
        vals[:] = 1.0 / n
    else:
        vals /= s
    return vals


# ----------------------------------------------------------------------
def _adam_optimize(
    logits0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 20000,
    lr0: float = 0.12,
    lam_start: float = 0.5,
    lam_factor: float = 1.005,
    lam_max: float = 50.0,
) -> tuple[np.ndarray, float]:
    """
    Adam on the logits of a probability vector, minimising the
    smooth log‑sum‑exp of the self‑convolution.
    Returns the best probability vector (Σ=1) and its raw score.
    """
    x = logits0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # probability vector (softmax ensures Σ=1, a≥0)
        a = _softmax(x)

        # raw score (for bookkeeping)
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_score:
            best_score = cur_raw
            best_a = a.copy()
            print(
                f"[{time.time():.1f}] Adam n={n} it={it:5d} raw={best_score:.6f} λ={lam:.3f}"
            )

        # ---------- smooth max (log‑sum‑exp) ----------
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()  # shape (2n‑1,)

        # ---------- gradient w.r.t. a ----------
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1 : n - 1 + n]

        # ---------- chain rule through soft‑max ----------
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- anneal λ ----------
        lam = min(lam * lam_factor, lam_max)

        # occasional jitter to escape plateaus
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    # fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _raw_score(best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_refine(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method on the true (non‑smooth) objective.
    Returns an improved probability vector and its raw score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = _raw_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        # active set (within relative tolerance)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        step = 0.07 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a)

        cur_raw = 2.0 * n * cur_max
        if cur_raw + 1e-12 < best_score:
            best_score = cur_raw
            best_a = a.copy()
            print(f"[{time.time():.1f}] subgrad it={it:5d} raw={best_score:.6f}")

        # tiny random shake every few hundred iterations
        if it % 800 == 0 and np.random.rand() < 0.06:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Main entry point.  Returns a list of non‑negative heights that (hopefully)
    achieves a low value of `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Try to fetch a previously discovered construction (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # Choose a target length.  Larger lengths give more freedom but
    # increase per‑iteration cost.  Empirically n≈20000–25000 works well.
    target_n = 24000

    # ------------------------------------------------------------------
    # 1) Simple shape grid search
    best_seq = None
    best_raw = float("inf")

    # include a warm‑start from a previous sequence (interpolated)
    if prev is not None:
        cand = np.interp(
            np.linspace(0, 1, target_n, endpoint=False),
            np.linspace(0, 1, prev.shape[0], endpoint=False),
            prev,
        )
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        raw = _raw_score(cand)
        if raw < best_raw:
            best_raw = raw
            best_seq = cand.copy()
        print(f"[{time.time():.1f}] prev‑interp raw={raw:.6f}")

    # Parameter grids
    p_vals = np.linspace(0.25, 0.75, 11)          # exponent p
    w_vals = np.linspace(0.0, 1.0, 9)            # mixture weight with uniform
    uniform = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    for p in p_vals:
        shape_p = _power_shape(p, target_n)
        for w in w_vals:
            if time.time() >= deadline:
                break
            a = w * shape_p + (1.0 - w) * uniform
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum
            else:
                a[:] = 1.0 / target_n
            raw = _raw_score(a)
            if raw < best_raw:
                best_raw = raw
                best_seq = a.copy()
                print(f"[{time.time():.1f}] grid p={p:.3f} w={w:.3f} raw={raw:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam optimisation (starting from the best grid point)
    if best_seq is None:
        # fallback: uniform
        best_seq = uniform.copy()
        best_raw = _raw_score(best_seq)

    # Convert to logits (log‑softmax inverse) for Adam
    eps = 1e-12
    init_logits = np.log(best_seq + eps)
    adam_seq, adam_raw = _adam_optimize(
        init_logits,
        target_n,
        deadline,
        max_iters=18000,
        lr0=0.12,
        lam_start=0.5,
        lam_factor=1.005,
        lam_max=50.0,
    )
    if adam_raw < best_raw:
        best_raw = adam_raw
        best_seq = adam_seq.copy()
        print(f"[{time.time():.1f}] after Adam raw={best_raw:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient polishing (optional, time‑aware)
    sub_seq, sub_raw = _subgradient_refine(best_seq, deadline, max_iters=8000)
    if sub_raw < best_raw:
        best_raw = sub_raw
        best_seq = sub_seq.copy()
        print(f"[{time.time():.1f}] after sub‑gradient raw={best_raw:.6f}")

    # ------------------------------------------------------------------
    # Final conversion to plain Python list (evaluate_sequence will
    # renormalise anyway, but we keep sum≈1 for consistency).
    final = best_seq.tolist()
    final_score = evaluate_sequence(final)
    print(
        f"\nFinal evaluated score: {final_score:.6f} (raw≈{best_raw:.6f}, length={len(final)})"
    )
    return final
