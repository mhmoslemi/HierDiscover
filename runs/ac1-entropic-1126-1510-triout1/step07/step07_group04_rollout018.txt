"""
A fast multi‑phase optimizer for the step‑function score.

Phase 1 – a cheap random scan over a few lengths (2000‑8000) using
Beta, Dirichlet and mixture families.  
Phase 2 – a refined exponentiated‑gradient (mirror‑descent) optimisation
with momentum (ADAGRAD‑style) on the current best length.  The update
keeps the sequence on the probability simplex, adds occasional tiny
jitter and uses a decaying learning rate ηₜ=η₀/√t.  
Phase 3 – up‑sample the best vector to larger lengths (12 k‑32 k) and
run another short mirror‑descent, which dramatically reduces the
sup‑norm of the self‑convolution.  The whole routine stops at the
global time budget (≈1000 s) and finally returns the best list of
heights found.  The score printed during the run is the exact value
returned by ``evaluate_sequence``.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on n points, normalised to Σ=1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑pdf – the log‑constant is safe for a wide range of parameters
    logC = math.lgamma(alpha + beta) - math.lgamma(alpha) - math.lgamma(beta)
    logpdf = logC + (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    logpdf = np.minimum(logpdf, 700.0)          # avoid overflow in exp
    pdf = np.exp(logpdf)
    pdf = np.maximum(pdf, 0.0)
    total = pdf.sum()
    if total == 0.0:
        pdf[:] = 1.0 / n
    else:
        pdf /= total
    return pdf


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0.0:
        new_vals[:] = 1.0 / new_n
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _self_convolution_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (linear) self‑convolution a*a."""
    return _self_convolution_full(a).max()


def _raw_score(a: np.ndarray) -> float:
    """Score for a vector that already satisfies Σa = 1."""
    n = a.shape[0]
    return 2.0 * n * _max_self_convolution(a)


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Sub‑gradient of f(a)=max_k (a*a)_k .
    Returns (grad, cur_max).  grad has the same shape as a.
    """
    n = a.shape[0]
    b = _self_convolution_full(a)
    cur_max = b.max()
    # tolerance for the active set
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / float(active.size)
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_expgd(
    a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    eta0: float = 0.8,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) with ADAGRAD‑style momentum.
    Stops when ``time.time()`` exceeds *deadline*.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_val = _raw_score(a)
    # ADAGRAD accumulator (actually a momentum‑averaged gradient)
    g_acc = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1
        grad, _ = _max_subgradient_step(a)
        # momentum update
        g_acc = momentum * g_acc + (1.0 - momentum) * grad
        eta = eta0 / math.sqrt(it)

        exponent = -eta * g_acc
        exponent = np.clip(exponent, -50, 50)          # avoid overflow
        a = a * np.exp(exponent)

        # re‑normalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # occasional tiny jitter to escape plateaus
        if (it % 400 == 0) and (rng.random() < 0.07):
            a += rng.normal(scale=1e-6, size=n)
            a = _project_onto_simplex(a)

        cur_val = _raw_score(a)
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(
                f"[{time.time():.1f}] exp‑gd len={n} it={it} score={best_val:.6f}"
            )
        # early exit if we are already under the target
        if best_val <= 1.503:
            break
    return best_a, best_val
