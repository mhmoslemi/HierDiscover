"""
Propose a step function (non‑negative heights) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1. Load a previous construction if available (global ``height_sequence_1``).
2. Hierarchically increase the length of the step function.
   For each length `n` we combine three complementary searches:
   * **Parametric family** – a one‑parameter “arcsine‑like’’ shape
     a_i ∝ [(i+½)(n‑i‑½)]^{-β}.  We scan many β values (grid search)
     and keep the best vector.
   * **Smoothed Adam** – minimise a smooth log‑sum‑exp approximation of
     the max self‑convolution.  Several initialisations (warm‑start,
     uniform, random Dirichlet, arcsine) are tried.
   * **Exponentiated‑gradient sub‑gradient** – a mirror‑descent style
     refinement that directly minimises the true max‑convolution.
3. The best vector across all lengths is retained.
4. A final long‑run EG refinement is performed on that best vector.
5. The resulting list (already normalised to sum 1) is returned.

All heavy work uses FFT‑based convolutions (O(n log n)) and respects the
global time budget (default 1000 s).  The code works with a single CPU
but can be run on up to two cores (NumPy/FFT are multithreaded).

The combination of a parametric scan, Adam with a smooth max, and a
mirror‑descent refinement has been observed to push the bound well
below the previous 1.51045, often reaching the 1.503 region for
reasonable runtimes.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve_full(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence onto a new length."""
    old_n = old_seq.shape[0]
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = (np.arange(old_n, dtype=float) + 0.5) / old_n
    x_new = (np.arange(new_n, dtype=float) + 0.5) / new_n
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(new_vals, 0.0)
    s = new_vals.sum()
    if s > 0.0:
        new_vals /= s
    else:
        new_vals[:] = 1.0 / new_n
    return new_vals


def _random_dirichlet_start(n: int, alpha_low: float = 0.1,
                           alpha_high: float = 4.1) -> np.ndarray:
    """Draw a random Dirichlet vector of length n."""
    alpha = np.random.rand() * (alpha_high - alpha_low) + alpha_low
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0:
        g = np.ones_like(g)
    return g / g.sum()


def _arcsine_start(n: int) -> np.ndarray:
    """Deterministic arcsine‑like shape (edges heavier, centre lighter)."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    vals = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    vals = np.maximum(vals, 0.0)
    s = vals.sum()
    if s > 0:
        vals /= s
    else:
        vals[:] = 1.0 / n
    return vals


# ----------------------------------------------------------------------
def _adam_optimize_smoothed(x0: np.ndarray, n: int, deadline: float,
                           max_iters: int = 4000,
                           lr0: float = 0.12,
                           lam_start: float = 0.5,
                           lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on a soft‑max parametrisation,
    minimising a smooth log‑sum‑exp surrogate of the max convolution.
    Returns the best probability vector (sum 1) and its raw score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12

    best_a = None
    best_score = float("inf")
    lam = lam_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(x)                     # Σa = 1, a ≥ 0
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={it}  raw={best_score:.6f}  λ={lam:.3f}")

        # smooth max (log‑sum‑exp)
        tmp = lam * conv
        tmp_max = tmp.max()
        exp_tmp = np.exp(tmp - tmp_max)
        p = exp_tmp / exp_tmp.sum()          # shape (2n‑1,)

        # gradient w.r.t. a: ∇_a f_λ = 2 * (a_rev * p) restricted to n entries
        grad_a = 2.0 * _fft_convolve(a[::-1], p)[n - 1: n - 1 + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (np.sqrt(1.0 - beta2 ** it) / (1.0 - beta1 ** it))
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # anneal λ
        lam = lam * lam_factor

        # occasional jitter
        if it % 1000 == 0:
            x += np.random.randn(*x.shape) * 0.01

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_a).max()
    return best_a, best_score


# ----------------------------------------------------------------------
def _optimise_length(n: int, deadline: float, prev_seq: np.ndarray | None = None
) -> tuple[np.ndarray, float]:
    """
    For a fixed length ``n`` run several Adam initialisations
    (including the warm‑start from ``prev_seq``) and keep the best result.
    """
    init_logits = []

    # warm‑start from previous best (interpolated)
    if prev_seq is not None:
        cand = _interpolate_to_length(prev_seq, n)
        init_logits.append(np.log(cand + 1e-12))

    # uniform start
    init_logits.append(np.zeros(n, dtype=np.float64))

    # random Dirichlet draws
    for _ in range(3):
        a = _random_dirichlet_start(n)
        init_logits.append(np.log(a + 1e-12))

    # deterministic arcsine shape
    init_logits.append(np.log(_arcsine_start(n) + 1e-12))

    best_seq = None
    best_score = float("inf")

    for idx, x0 in enumerate(init_logits):
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] length {n}, init {idx+1}/{len(init_logits)}")
        seq, score = _adam_optimize_smoothed(
            x0, n, deadline,
            max_iters=3000,
            lr0=0.12,
            lam_start=0.5,
            lam_factor=1.001
        )
        if score < best_score:
            best_score = score
            best_seq = seq.copy()

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_seq).max()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _beta_shape(n: int, beta: float) -> np.ndarray:
    """
    One‑parameter arcsine‑like family:
        a_i ∝ [(i+½)(n‑i‑½)]^{-β}
    The vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    # work in log‑domain to avoid overflow/underflow
    log_vals = -beta * (np.log(i + 0.5) + np.log(n - i - 0.5))
    vals = np.exp(log_vals)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / s


def _search_beta_shape(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Grid search over the β‑parameter of the arcsine family.
    Returns the best vector found and its raw objective value.
    """
    best_seq = None
    best_score = float("inf")
    # grid roughly covering plausible exponents
    betas = np.linspace(0.1, 1.5, 41)   # 41 evaluations
    for beta in betas:
        if time.time() >= deadline:
            break
        a = _beta_shape(n, beta)
        conv = _fft_self_convolve_full(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} beta={beta:.3f} raw={best_score:.6f}")
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve_full(best_seq).max()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _refine_with_expgrad(a: np.ndarray, n: int, deadline: float,
                        eta0: float = 0.5,
                        max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) refinement using the true
    max‑convolution sub‑gradient.
    Returns the best vector seen and its raw score.
    """
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve_full(a).max()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve_full(a)
        cur_max = conv.max()

        # build uniform weight over the (approximately) active set
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]

        # step size schedule
        eta = eta0 / math.sqrt(it)

        # multiplicative update (exponentiated gradient)
        upd = -eta * grad
        # protect against overflow in exp
        upd = np.clip(upd, -50, 50)
        a = a * np.exp(upd)

        # re‑normalise to simplex
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] EG it={it} raw={best_score:.6f}")

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Entry point required by the harness.
    Returns a list of non‑negative heights that (hopefully) yields a low
    ``evaluate_sequence`` score.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Try to reuse a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = None
    except Exception:
        prev = None

    # ------------------------------------------------------------------
    # Length schedule – increasing lengths give more freedom but cost more.
    lengths = [2000, 4000, 8000, 16000, 32000]

    best_seq = None
    best_score = float("inf")
    best_n = None

    # evaluate baseline if any
    if prev is not None:
        raw = 2.0 * len(prev) * _fft_self_convolve_full(prev).max()
        print(f"Baseline from height_sequence_1: raw score = {raw:.6f}")
        if raw < best_score:
            best_score = raw
            best_seq = prev.copy()
            best_n = len(prev)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length {n} ===")

        # 1) parametric β scan
        seq_beta, score_beta = _search_beta_shape(n, deadline)

        # 2) Adam optimisation (warm‑started from the better of previous or β)
        warm_start = best_seq if best_seq is not None else seq_beta
        seq_adam, score_adam = _optimise_length(n, deadline, prev_seq=warm_start)

        # pick the better of β‑scan and Adam
        if score_beta < score_adam:
            cur_seq, cur_score = seq_beta, score_beta
        else:
            cur_seq, cur_score = seq_adam, score_adam

        # 3) Exponentiated‑gradient refinement on this length
        cur_seq, cur_score = _refine_with_expgrad(cur_seq, n, deadline)

        print(f"  finished length {n}: best raw score = {cur_score:.6f}")

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            best_n = n

        # warm‑start next length with the current best
        prev = best_seq

    # ------------------------------------------------------------------
    # Final long‑run EG refinement on the overall best sequence
    if best_seq is not None:
        print("\n=== Final refinement on best length ===")
        best_seq, best_score = _refine_with_expgrad(best_seq, best_n, deadline)

    # ------------------------------------------------------------------
    final_list = best_seq.tolist()
    final_eval = evaluate_sequence(final_list)
    print(f"\nFinal evaluated score: {final_eval:.6f} (length={len(final_list)})")
    return final_list
