"""
Improved step‑function optimizer.

The algorithm augments the previous three‑phase scheme with a
*log‑sum‑exp smoothing* of the max‑convolution objective.
A schedule over the smoothing parameter α (starting from a very
smooth approximation and ending with a near‑exact max) is used
inside an Adam optimiser.  This yields more informative gradients
than the raw sub‑gradient and helps escape shallow plateaus.
The search proceeds as follows:

1. **Generate a rich pool of seeds** – analytical families
   (arcsine, beta, triangular, exponential, sine, cosine,
   uniform) together with a few random Dirichlet draws and the
   best previously discovered sequence (if available).

2. **Heavy refinement** – for the top‑k inexpensive seeds run
   the smoothed Adam optimiser with an annealing schedule on α.
   The best result from this phase is kept.

3. **Exact sub‑gradient polishing** – a long plain sub‑gradient
   descent on the true max‑convolution objective finishes the
   optimisation.

4. **Fine‑grained hill climbing** – tiny random mass transfers
   and occasional jitter polish the final candidate.

All updates keep the vector on the probability simplex, therefore
the scale‑invariant evaluation function `evaluate_sequence` can be
called directly on the returned list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Cheap objective = 2·n·max(a*a) assuming sum(a)=1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _smooth_score_and_grad(a: np.ndarray, alpha: float):
    """
    Returns the smoothed objective
        f(a) = (1/α) log Σ exp(α·(a*a)_k)
    and its gradient ∇f(a).

    The gradient uses the identity
        ∂f/∂a_i = Σ_k w_k (a_{k-i}+a_i·1_{0≤k-i<n}),
    where w_k = exp(α·b_k)/Σ exp(α·b).
    """
    n = a.shape[0]

    # self‑convolution (length 2n‑1)
    b = _fft_self_convolve(a)

    # log‑sum‑exp in a numerically stable way
    max_b = b.max()
    exp_shift = np.exp(alpha * (b - max_b))
    Z = exp_shift.sum()
    w = exp_shift / Z                       # normalized weights, size 2n‑1
    smooth_obj = max_b + np.log(Z) / alpha  # (1/α)·log Σ exp(α b) = max_b + log(Z)/α

    # term1 = Σ_k w_k a_{k-i}   → correlation of a and w
    conv_rev = _fft_convolve(a[::-1], w)   # length 2n‑1 + 2n‑1 -1 = 4n‑3, but we only need slice
    term1 = conv_rev[n - 1 : n - 1 + n]    # size n, exactly Σ_j a_j w_{i+j}

    # term2 = a_i * Σ_{k=i}^{i+n-1} w_k   → sliding window sum of w
    ones = np.ones(n, dtype=np.float64)
    w_window = _fft_convolve(w, ones)      # length (2n‑1)+n‑1 = 3n‑2
    term2 = w_window[n - 1 : n - 1 + n]    # Σ_{j=0}^{n-1} w_{i+j}

    grad = term1 + a * term2
    return smooth_obj, grad


# ----------------------------------------------------------------------
def _adam_smooth_opt(a0: np.ndarray,
                     deadline: float,
                     alpha_start: float = 5.0,
                     alpha_end: float = 4000.0,
                     alpha_steps: int = 6,
                     iters_per_alpha: int = 30_000,
                     step0: float = 0.20,
                     beta1: float = 0.90,
                     beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed log‑sum‑exp objective.
    The smoothing parameter α is increased geometrically from
    alpha_start to alpha_end in `alpha_steps` stages.
    Returns the best sequence found (in terms of exact cheap score)
    and its cheap score.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = _max_convolution_score(a)

    # geometric schedule for α
    if alpha_steps == 1:
        alphas = [alpha_end]
    else:
        alphas = np.geomspace(alpha_start, alpha_end, num=alpha_steps)

    t = 0  # global iteration counter for bias correction
    for stage, alpha in enumerate(alphas):
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score
            t += 1
            _, grad = _smooth_score_and_grad(a, alpha)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            eta = step0 / (np.sqrt(v_hat) + 1e-8)
            a = a - eta * m_hat
            a = _project_onto_simplex(a, s=1.0)

            if t % 250 == 0:
                cur = _max_convolution_score(a)
                if cur < best_score - 1e-15:
                    best_score = cur
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Adam (α={alpha:.1f}) improved cheap score to {cur:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray):
    """Exact sub‑gradient of max(a*a) on the simplex."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # weight vector for active constraints (0/1)
    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution with reversed a gives Σ_j a_j w_{i+j}
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for positions where 2*i is active
    even = (active % 2 == 0)
    diag_idx = active[even] // 2
    if diag_idx.size:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """Plain sub‑gradient descent with diminishing step size."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgradient(a)
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        a = _project_onto_simplex(a, s=1.0)

        if it % 500 == 0:
            cur = _max_convolution_score(a)
            if cur < best_score - 1e-15:
                best_score = cur
                best_a = a.copy()
    return best_a, best_score

# Analytic families ------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    return vals / vals.sum()


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    return vals / vals.sum()


def _exponential_distribution_fixed_lam(lam: float, n: int) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    return vals / vals.sum()


def _exponential_distribution_0_003(n: int) -> np.ndarray:
    return _exponential_distribution_fixed_lam(0.003, n)


def _exponential_distribution_0_001(n: int) -> np.ndarray:
    return _exponential_distribution_fixed_lam(0.001, n)


def _sine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n)
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _cosine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (i - (n - 1) / 2) / n) + 1.0
    vals = np.maximum(vals, 0)
    return vals / vals.sum()


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a uniform point on the simplex (Γ–Dirichlet, α=1)."""
    x = np.random.gamma(1.0, 1.0, size=n)
    return x / x.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    The routine combines analytic seed generation, a smoothed‑Adam phase,
    exact sub‑gradient polishing and a final hill‑climb.  It respects the
    supplied time budget and returns a list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 0. Base length – try to stay near the previously best construction.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 3500

    # --------------------------------------------------------------
    # 1. Choose a handful of lengths to explore.
    lengths = set()
    for f in (0.9, 1.0, 1.1, 1.2, 1.3):
        n = int(base_n * f)
        if n >= 500:
            lengths.add(n)
    for _ in range(5):
        n = np.random.randint(500, min(8000, base_n * 3) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # --------------------------------------------------------------
    # 2. Broad sampling of analytic families + random Dirichlet draws.
    candidates = []   # list of (array, cheap_score)

    beta_exponents = [0.02, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5,
                      0.75, 1.0, 1.25, 1.5, 2.0, 3.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families -------------------------------------------------
        families = [
            (_uniform_distribution, "uniform"),
            (_arcsine_distribution, "arcsine"),
            (_triangular_distribution, "triangular"),
            (_exponential_distribution_0_003, "exp(0.003)"),
            (_exponential_distribution_0_001, "exp(0.001)"),
            (_sine_distribution, "sine"),
            (_cosine_distribution, "cosine")
        ]
        for gen, name in families:
            a = gen(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} {name:12s} cheap={score:.6f}")

        # symmetric beta
        for p in beta_exponents:
            if time.time() >= deadline:
                break
            a = _beta_symmetric_distribution(n, p)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} beta(p={p:4.2f}) cheap={score:.6f}")

        # random Dirichlet draws
        for _ in range(4):
            if time.time() >= deadline:
                break
            a = _random_dirichlet(n)
            score = _max_convolution_score(a)
            candidates.append((a, score))
            print(f"[{time.time():.1f}] n={n:4d} random Dirichlet cheap={score:.6f}")

    # --------------------------------------------------------------
    # 3. Include the best sequence discovered so far (if any).
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            score = _max_convolution_score(prev)
            candidates.append((prev, score))
            print(f"[{time.time():.1f}] previous best cheap={score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 4. Keep the top‑k seeds for heavy refinement.
    candidates.sort(key=lambda x: x[1])
    top_k = 12
    top_candidates = candidates[:top_k]
    print(f"--- selected {len(top_candidates)} seeds for refinement (best cheap={top_candidates[0][1]:.6f})")

    # --------------------------------------------------------------
    # 5. Heavy refinement – smoothed Adam with α‑annealing.
    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break

        seq, sc = _adam_smooth_opt(seed_seq, deadline,
                                   alpha_start=5.0,
                                   alpha_end=4000.0,
                                   alpha_steps=6,
                                   iters_per_alpha=20_000,
                                   step0=0.25)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} smoothed‑Adam improved cheap={sc:.6f}")

    # --------------------------------------------------------------
    # 6. Exact sub‑gradient polishing on the current best.
    sub_deadline = deadline - 30.0   # reserve a few seconds for hill‑climb
    if time.time() < sub_deadline:
        seq, sc = _subgradient_descent(best_seq, sub_deadline,
                                       max_iters=700_000,
                                       eta0=0.55)
        if sc < best_score - 1e-14:
            best_score, best_seq = sc, seq.copy()
            print(f"[{time.time():.1f}] sub‑gradient descent finished cheap={sc:.6f}")

    # --------------------------------------------------------------
    # 7. Fine‑grained hill climbing (tiny mass transfers, occasional jitter).
    a = best_seq.copy()
    cur_score = best_score
    n = a.shape[0]
    it = 0
    max_transfer = 0.025  # start with 2.5 % of total mass

    while time.time() < deadline:
        it += 1
        # geometric decay of transfer magnitude
        transfer = max_transfer * (0.5 ** (it / 8000.0))

        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * min(transfer, a[i])
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] hill‑climb iter={it} new best cheap={best_score:.6f}")

        # occasional jitter to hop off flat plateaus
        if it % 7000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(n) * 6e-5
            a = best_seq + jitter
            a = _project_onto_simplex(a, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    print(f"=== final cheap score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list – `evaluate_sequence` will normalise the sum.
    return best_seq.tolist()
