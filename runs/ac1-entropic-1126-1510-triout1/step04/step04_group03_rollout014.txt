"""
Propose a step function (list of non‑negative heights) that minimizes
`evaluate_sequence`.  The algorithm:

1. Choose a relatively large length (≈2 × the length of the
   previously found sequence, capped at 25 000).  A longer sequence
   allows the theoretical lower bound 2·n·max(b) → 1 to be approached.

2. Initialise several candidates (previous best up‑sampled, arcsine,
   uniform, random, and a two‑spike pattern).  All are projected onto
   the probability simplex (sum = 1).

3. For each candidate run a projected sub‑gradient descent on the
   simplex:
   * the objective is `max_k (a*a)_k`;
   * the sub‑gradient of `max` is obtained by taking the (average)
     convolution of the reversed vector with an indicator at the
     current maximal index (or all almost‑maximal indices);
   * Nesterov‑style momentum and a diminishing step size
     ηₜ = η₀/√t are used;
   * occasional tiny random jitter helps escape plateaus.

4. After the gradient phase a cheap stochastic local‑search swaps a
   random fraction of mass between two positions and keeps the move if
   the objective improves.

5. The best sequence found over all restarts is refined once more with
   the local‑search procedure.

The whole search respects the 1000 s budget, leaves a small safety
margin and prints intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                      # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length n (probability vector)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _upsample_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(old)
    if old_n == new_n:
        return old.astype(np.float64, copy=True)

    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(new_x, old_x, old)
    return new_vals.astype(np.float64)


# ----------------------------------------------------------------------
def _gradient_of_max(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k.
    The gradient is 2 * conv(w, rev(a)) where w is the (averaged)
    indicator of the maximal indices of b.
    """
    n = a.shape[0]
    max_val = b.max()
    # treat all entries within a tiny tolerance as active
    tol = 1e-12
    ks = np.where(np.abs(b - max_val) <= tol)[0]
    if ks.size == 0:
        ks = np.array([np.argmax(b)], dtype=int)

    rev = a[::-1]
    g = np.zeros_like(a)
    for k in ks:
        shift = n - 1 - k                # shift needed on rev
        idx = shift + np.arange(n)       # indices in rev that contribute
        mask = (idx >= 0) & (idx < n)
        g[mask] += 2.0 * rev[idx[mask]]
    g /= ks.size
    return g


# ----------------------------------------------------------------------
def _optimize_length_subgrad(init_a: np.ndarray,
                             n: int,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex to minimise
    max(self‑convolution).  Returns the best sequence discovered for
    this length and its internal score 2·n·max_b (which equals the
    official `evaluate_sequence` value when the sum of the coefficients
    equals 1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # hyper‑parameters
    beta = 0.9            # momentum coefficient
    eta0 = 0.8            # base step size
    velocity = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # sub‑gradient of the max‑convolution
        grad = _gradient_of_max(a, b)

        # Nesterov‑style momentum
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny random jitter to escape flat regions
        if it % 3000 == 0 and np.random.rand() < 0.25:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic pairwise mass‑swap local search.
    Repeatedly moves a random fraction (≤30 %) of mass from one index
    to another and keeps the move if the max‑convolution improves.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] <= 1e-12:
            continue
        frac = np.random.rand() * 0.30          # up to 30 % transfer
        delta = best_seq[i] * frac
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # projection is not needed – the total sum stays 1 and all entries remain ≥0
        b = _fft_self_convolve(cand)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_seq = cand

        # tiny jitter occasionally
        if it % 5000 == 0:
            jitter = best_seq + np.random.randn(n) * 1e-6
            jitter = _project_onto_simplex(jitter, s=1.0)
            b = _fft_self_convolve(jitter)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_seq = jitter

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point.  Searches for a non‑negative step function
    (list of heights) with a low `evaluate_sequence` score.
    """
    np.random.seed(seed)

    # give a small safety margin for the final printing
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine a sensible target length.
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # aim for roughly twice the previous length, but not larger than 25 000
    n_target = int(min(25000, max(3000, base_n * 2)))
    n_target = max(300, n_target)                # enforce a minimum

    print(f"Target length = {n_target}")

    # ------------------------------------------------------------------
    # Build a list of diverse initialisations (all sum to 1).
    init_candidates = []

    # 1) previous best, up‑sampled if length differs
    try:
        old = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        up = _upsample_sequence(old, n_target)
        up = _project_onto_simplex(up, s=1.0)
        init_candidates.append(("prev_up", up))
    except Exception:
        pass

    # 2) arcsine distribution
    arcs = _arcsine_distribution(n_target)
    init_candidates.append(("arcsine", arcs))

    # 3) uniform distribution
    uni = np.full(n_target, 1.0 / n_target, dtype=np.float64)
    init_candidates.append(("uniform", uni))

    # 4) random Dirichlet‑like vector
    rnd = np.random.rand(n_target)
    rnd /= rnd.sum()
    init_candidates.append(("random", rnd))

    # 5) two‑spike pattern (mass near both ends)
    spike = np.zeros(n_target, dtype=np.float64)
    pos1 = np.random.randint(max(1, n_target // 10))
    pos2 = np.random.randint(n_target - max(1, n_target // 10), n_target)
    spike[pos1] = 0.5
    spike[pos2] = 0.5
    spike = _project_onto_simplex(spike, s=1.0)
    init_candidates.append(("spike", spike))

    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    for kind, init_a in init_candidates:
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate a fair slice of the remaining time to this candidate
        slice_deadline = time.time() + max(0.05, remaining * 0.25)

        print(f"--- Optimising length {n_target}, init = {kind}")

        # gradient‑based optimisation
        seq, int_score = _optimize_length_subgrad(init_a, n_target, slice_deadline)

        # short local‑search phase (if any time left)
        left = slice_deadline - time.time()
        if left > 0.5:
            seq, int_score = _local_search(seq, n_target,
                                           time.time() + min(left, 4.0))

        # evaluate with the official black‑box (more reliable)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = int_score

        print(f"Result for init={kind}: score={final_score:.6f}")

        if final_score < global_best_score:
            global_best_score = final_score
            global_best_seq = seq.copy()
            print(f"*** New global best ← {final_score:.6f}")

        # early exit if we already beat the target
        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # final refinement on the overall best (if any time left)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _local_search(global_best_seq,
                                       len(global_best_seq),
                                       deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if global_best_seq is None:
        n = 3000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
