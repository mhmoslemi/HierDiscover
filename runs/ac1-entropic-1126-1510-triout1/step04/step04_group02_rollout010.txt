def _optimize_adam(a0, n, deadline):
    a = a0.copy()
    best_seq = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2 * n * conv.max()
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 1.0  # base learning rate; will be scaled.
    it = 0
    alpha = 5.0
    alpha_inc = 1.005
    alpha_max = 5000.0
    size_fft = 1 << ((3 * n - 2).bit_length())
    while time.time() < deadline:
        it += 1
        # compute convolution for gradient
        b = _fft_self_convolve(a)
        cur_score = 2 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")
        # gradient
        # compute weights softmax
        # we may use softmax gradient like earlier
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0 else np.eye(len(b))[np.argmax(b)]
        # compute gradient: g_i = 2 * sum_{k} w_k * a_{k-i} ? Actually earlier gradient formula is 2 * (w * a_rev) conv. They compute by convolution of w with reversed a.
        # same as earlier: g = 2 * conv(w, a_rev) at center.
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        conv_w_a = np.fft.irfft(W * A_rev, n=size_fft)
        start = a.shape[0] - 1
        g = 2.0 * conv_w_a[start:start + a.shape[0]]
        # Adam update for log-domain: a_new = a * exp(-lr * (m_hat/(sqrt(v_hat)+eps))) where m_hat, v_hat are bias-corrected.
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        # step size schedule: step = lr / (np.sqrt(it))
        step = lr / (np.sqrt(it))
        update = - step * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(update)
        # ensure non-negativity
        a = np.maximum(a, 1e-12)
        # renorm to sum=1
        a = a / a.sum()
        # increase alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_inc)
        # random jitter occasionally
        if it % 2000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0)
            a = a / a.sum()
    return best_seq, best_score

"""
Propose candidate step‑function via Adam mirror‑descent and simulated annealing.

The routine works on probability vectors (∑a_i = 1) because the evaluation
`evaluate_sequence` is homogeneous with respect to scaling.  For a chosen
length *n* an initial distribution (arcsine, beta‑skewed, random …) is
optimised with an Adam‑style exponentiated‑gradient (mirror‑descent) on the
simplex.  The objective is a smooth log‑sum‑exp surrogate of the
peak of the self‑convolution; the temperature is raised gradually so that
the surrogate converges to the true max.  After the main phase a short
simulated‑annealing / hill‑climbing refinement moves a fraction of mass
between indices and occasionally smooths small windows.

Several lengths (up to 20 000) and initialisations are tried within the
time budget (default 1000 s).  The best sequence found is returned as a
plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]               # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_skewed_distribution(n: int) -> np.ndarray:
    a = np.random.beta(0.5, 0.5, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = _arcsine_distribution(n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a = a / a.sum()
    elif kind == "beta":
        a = _beta_skewed_distribution(n)
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]

        a = np.maximum(a, 0.0)
        a = a / a.sum()
    else:
        a = _arcsine_distribution(n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _optimize_adam(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) optimiser with Adam‑style
    moment adaptation.  Returns the best sequence seen and its internal
    `2·n·max(conv)` score.
    """
    a = a0.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr = 1.0

    it = 0
    alpha = 5.0               # soft‑max temperature start
    alpha_mul = 1.02
    alpha_max = 5000.0
    size_fft = 1 << ((3 * n - 2).bit_length())

    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient computation
        if alpha < alpha_max:                         # smooth surrogate
            shift = b - b.max()
            w = np.exp(alpha * shift)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(w)
                w[np.argmax(b)] = 1.0
            else:
                w /= w_sum

            a_rev = a[::-1]
            W = np.fft.rfft(w, n=size_fft)
            A_rev = np.fft.rfft(a_rev, n=size_fft)
            conv = np.fft.irfft(W * A_rev, n=size_fft)
            start = n - 1
            grad = 2.0 * conv[start:start + n]
        else:                                         # pure sub‑gradient
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)

        # Adam moment updates (in log‑space)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr / (np.sqrt(it) + 1e-12)

        # multiplicative (exponentiated‑gradient) update
        update = -step * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(update)
        a = np.maximum(a, eps)          # keep strictly positive
        a /= a.sum()                    # project back onto simplex

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter
        if it % 2500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, eps)
            a /= a.sum()

    return best_seq, best_score

    """

    """

    best_seq = a.copy()

    init_temp = cur_score * 0.005   # about 0.5 % of the current score

    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = max(0.0, 1.0 - elapsed / total)
        T = max(1e-12, init_temp * frac)

        # --- random mass transfer -------------------------------------------------
        i = np.random.randint(0, n)
        j = np.random.randint(0, n - 1)
        if j >= i:
            j += 1

        a_candidate[i] -= eps
