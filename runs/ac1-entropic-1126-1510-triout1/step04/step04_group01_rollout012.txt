def _lp_step(f, M, active_idx):
    n = len(f)
    # Build sparse matrix A_ub (len(active_idx) x n) with rows for each idx j
    data = []
    rows = []
    cols = []
    for r, j in enumerate(active_idx):
        lo = max(0, j - (n-1))
        hi = min(j, n-1)
        # i from lo..hi
        # For each i, var index k = j - i
        # coeff = f[i]
        # add entries
        for i in range(lo, hi+1):
            k = j - i
            data.append(f[i])
            rows.append(r)
            cols.append(k)
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), M)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g0 = res.x
    return g0
```

"""
Hybrid LP‑gradient optimizer for the self‑convolution peak problem.

The algorithm proceeds in three stages for each candidate length:

1. **Initialisation** – a probability vector on the simplex is created.
   Several simple seeds are tried (arcsine, uniform, random, previous best).

2. **LP improvement** – using the current vector `f` we form the bound
   `M = max (f*f)`.  A linear programme
        maximise   Σ g_i
        subject to g_i ≥ 0,
                   (f * g)_j ≤ M   for a small set of “active’’ convolution
                   indices,
   is solved (SciPy’s HiGHS).  The solution `g₀` is scaled to the simplex
   (`g = λ·g₀` with λ = 1/Σ g₀) which guarantees a strictly smaller
   self‑convolution peak.  A convex combination `f ← (1‑t)·f + t·g`
   (t≈0.3) is taken and the process is repeated a few times.

3. **Fine‑tuning** – the result of the LP phase is polished with the
   Adam optimiser on a smooth soft‑max surrogate (as in the reference
   code) and a short exact‑max sub‑gradient descent.  The best candidate
   found across all lengths and seeds is returned.

The whole search respects the global time budget (default 1000 s) and
falls back to a uniform distribution if nothing is produced.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex utilities ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Distributions --------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine‑like probability vector of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
# FFT‑based convolutions ------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by Adam) ---------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max objective.
    Returns the best sequence encountered and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        lr = lr0 / np.sqrt(t)

        # update & project
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # tiny random jitter every few hundred iterations
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
# Exact max sub‑gradient (used for polishing) -------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the true ℓ∞ objective.
    Uses a uniform weight over all currently active indices.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short exact‑max polishing using sub‑gradient descent."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
# Linear‑programming improvement ---------------------------------------
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
    _LP_AVAILABLE = True
except Exception:               # pragma: no cover
    _LP_AVAILABLE = False


def _lp_step(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """Solve  max Σ g_i  s.t. g≥0 and (f*g)_j ≤ M for all j∈active_idx."""
    if not _LP_AVAILABLE:
        return None

    n = f.shape[0]
    rows, cols, data = [], [], []
    for r, j in enumerate(active_idx):
        lo = max(0, j - (n - 1))
        hi = min(j, n - 1)
        # i runs from lo..hi, variable index k = j-i
        for i in range(lo, hi + 1):
            k = j - i
            rows.append(r)
            cols.append(k)
            data.append(f[i])
    A_ub = _sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -Σg_i  → maximise Σg_i
    bounds = [(0, None)] * n
    res = _opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                       method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x


def _lp_improve(f0: np.ndarray,
                t: float = 0.3,
                max_iter: int = 6,
                deadline: float = 1e30,
                top_k: int = 200) -> np.ndarray:
    """
    Repeated LP‑based improvement.
    At each iteration we maximise total mass under the current maximum‑convolution
    bound and move partially towards the scaled solution.
    """
    f = f0.copy()
    n = f.shape[0]
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        b = _fft_self_convolve(f)
        M = b.max()
        # pick the k largest entries of b as the active set
        if top_k >= len(b):
            active = np.arange(len(b))
        else:
            # argpartition gives the k largest (unsorted) indices
            active = np.argpartition(-b, top_k - 1)[:top_k]
        g0 = _lp_step(f, M, active)
        if g0 is None:
            break
        s = g0.sum()
        if s <= 0:
            break
        lam = 1.0 / s                # scaling to keep Σg = 1
        # do not scale up – that would increase the bound
        if lam > 1.0:
            lam = 1.0
        g = lam * g0
        # convex combination
        f = (1.0 - t) * f + t * g
        f = _project_onto_simplex(f, s=1.0)
    return f

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function (probability vector) with a low
    `evaluate_sequence` score.  Combines a short LP‑based
    improvement phase with Adam optimisation on a soft‑max surrogate
    and a final exact‑max polishing.
    """
    import numpy as np
    import time

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # 1 s safety margin

    # try to reuse the globally supplied construction if it exists
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:

    # ------------------------------------------------------------------
    # candidate lengths (a handful around a historically good size)
    base_n = len(base_seq) if base_seq is not None else 4000
    lengths = sorted({max(500, int(base_n * f)) for f in (0.6, 0.8, 1.0, 1.2, 1.5)
                     if int(base_n * f) >= 500})

    # shuffle to diversify early exploration
    np.random.shuffle(lengths)

    overall_best_seq = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a rough per‑length budget (equal split)
        length_deadline = time.time() + (deadline - time.time()) / (len(lengths) - lengths.index(n) or 1)

        # ----------------------------------------------------------------
        # different initialisation schemes
        init_strategies = ["prev", "arcsine", "uniform", "random"]
        for kind in init_strategies:
            if time.time() >= deadline or time.time() >= length_deadline:
                break

            # ---- initialise ------------------------------------------------
            if kind == "prev" and base_seq is not None:
                a0 = _interpolate_to_length(base_seq, n)
            elif kind == "arcsine":
                a0 = _arcsine_distribution(n)
            elif kind == "uniform":
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            else:   # random
                a0 = np.random.rand(n)
                a0 /= a0.sum()

            a0 = _project_onto_simplex(a0, s=1.0)

            # ---- LP improvement phase --------------------------------------
            # give each length roughly 40 % of its budget for the LP phase
            lp_deadline = time.time() + min(8.0, (length_deadline - time.time()) * 0.4)
            a_lp = _lp_improve(a0, t=0.3, max_iter=8,
                               deadline=lp_deadline, top_k=200)

            # ---- Adam on smooth‑max surrogate ------------------------------
            adam_deadline = time.time() + min(12.0,
                                             (length_deadline - time.time()) * 0.4)
            a_adam, _ = _adam_optimize(a_lp, adam_deadline)

            # ---- Exact‑max polishing --------------------------------------
            refine_deadline = min(length_deadline, time.time() + 3.0)
            a_refined = _refine_max(a_adam, refine_deadline)

            # ---- fast score (no clipping, just the formula) ---------------
            max_b = _fft_self_convolve(a_refined).max()
            fast_score = 2.0 * n * max_b
            if fast_score < overall_best_score:
                overall_best_score = fast_score
                overall_best_seq = a_refined.copy()
                print(f"[{time.time():.1f}] len={n} init={kind} fast_score={fast_score:.6f}")

            # ---- true black‑box evaluation ---------------------------------
            try:
                true_score = evaluate_sequence(overall_best_seq.tolist())
                if true_score < overall_best_score:
                    overall_best_score = true_score
                    overall_best_seq = overall_best_seq.copy()
                    print(f"[{time.time():.1f}] *** New true best: len={n} init={kind} "
                          f"score={true_score:.6f}")
            except Exception as exc:
                print("evaluate_sequence raised:", exc)

            # early exit if we have reached the target
            if overall_best_score <= 1.503:
                break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback (should never happen)
    if overall_best_seq is None:

        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback uniform used.")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
