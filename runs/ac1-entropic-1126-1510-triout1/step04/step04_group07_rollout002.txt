import time
import math
import numpy as np
import scipy.optimize as opt

def _project_to_simplex(v, z):
    """Project v onto simplex {x>=0, sum x = z}."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _softmax(y):
    """Stable softmax to probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()

def _compute_f_and_grad(y, mu, n, S):
    """Smooth‑max (mu>0) or hard‑max (mu=0) value and gradient wrt logits y."""
    p = _softmax(y)
    a = S * p
    conv = np.convolve(a, a)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    inner = np.dot(p, g_a)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)

class _SmoothMaxObjective:
    __slots__ = ("mu","n","S")
    def __init__(self, mu, n, S):
        self.mu = mu; self.n = n; self.S = S
    def __call__(self, y):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)

    beta2 = 0.999
    eps = 1e-8

        # probability and heights

        a = S * p
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        conv_wr = np.convolve(w, a[::-1])
        g_a = 2.0 * conv_wr[n-1:n-1+n]
        inner = np.dot(p, g_a)
        grad_y = S * p * (g_a - inner)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)

    improve convergence near the non‑differentiable optimum.

After the continuous phases we perform a cheap random‑block local search,
a small power‑law polishing sweep, and finally a monotone‑sorting
heuristic.  The routine respects the supplied time budget (default
1000 s) and returns the best feasible sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (mu>0) or hard‑max (mu=0) value and gradient w.r.t.
    the free logits y.
    """
    p = _softmax(y)                # probabilities, Σp = 1
    a = S * p                      # actual heights, Σa = S
    conv = np.convolve(a, a)       # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                           # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (sub‑gradient when μ=0)
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]    # slice length n

    # chain rule through the soft‑max
    inner = np.dot(p, g_a)                     # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for ``scipy.optimize.minimize`` returning (value, grad)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _adam_optimize(
    y0: np.ndarray,
    n: int,
    S: float,
    max_iter: int,
    deadline: float,
    best_a: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam‑style refinement on the *hard* max objective (μ = 0).
    Returns the (possibly) improved best_a, best_score and the final logits y.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.15                      # base step size (empirically good)

    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        p = _softmax(y)
        a = S * p
        conv = np.convolve(a, a)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        conv_wr = np.convolve(w, a[::-1])
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]
        inner = np.dot(p, g_a)
        grad_y = S * p * (g_a - inner)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional evaluation
        if t % 10 == 0 or t == max_iter:
            a_curr = S * _softmax(y)
            sc = evaluate_sequence(a_curr.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a_curr.copy()
                print(f"[adam] iter {t:5d} new best score = {best_score:.6f}")

    return best_a, best_score, y


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low
    ``evaluate_sequence`` score (i.e. low max autoconvolution).

    The method combines a smooth‑max continuation (L‑BFGS‑B),
    an Adam refinement on the true max, a large random‑block search,
    a power‑law polishing sweep and a monotone‑sorting heuristic.
    The whole routine respects the given time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # problem size (use reference if available)
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3500
        have_ref = False

    S = math.sqrt(2.0 * n)   # ℓ¹‑norm we enforce via the soft‑max

    # bookkeeping
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    def _maybe_update(a_vec: np.ndarray) -> None:
        """Evaluate a candidate and keep it if it improves the best score."""
        nonlocal best_score, best_a
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # generate a collection of initial logits
    init_ys = []

    # 0 – uniform (y = 0)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – decreasing exponential
    alpha = rng.uniform(0.001, 0.008) * n / 1000.0
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # 2 – increasing exponential (mirror of the above)
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 3 – random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 4 – random Dirichlet (typically sparse)
    init_ys.append(np.log(rng.dirichlet(np.ones(n)) + 1e-12))

    # 5 – reference if we have one
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        eps = 1e-12
        init_ys.append(np.log(a_ref + eps))

    # --------------------------------------------------------------
    # continuation parameters (smooth‑max)
    mu_factor = 0.6               # slower cooling than before
    mu_min = 1e-8
    maxiter_per_mu = 300
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}]")
        y = y0.copy()

        # initialise temperature from the current max‑convolution
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        mu = max(mu, 1.0)                # safeguard against a zero start

        # ---- smooth‑max continuation (L‑BFGS‑B) --------------------
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)

            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

        # ---- hard‑max refinement with Adam ------------------------
        if time.time() < deadline:
            best_a, best_score, y = _adam_optimize(
                y,
                n,
                S,
                max_iter=2000,
                deadline=deadline,
                best_a=best_a,
                best_score=best_score,
            )

    # --------------------------------------------------------------
    # cheap random block search (many cheap trials)
    remaining = deadline - time.time()
    if remaining > 2.0:
        max_trials = int(min(20000, remaining * 200))
        print(f"[block] performing {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(2, 15)  # number of constant blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # --------------------------------------------------------------
    # power‑law polishing: try a_i^p for several p‑values
    if time.time() < deadline:
        for p in [0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0]:
            cand = np.power(best_a, p)
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # --------------------------------------------------------------
    # monotone‑sorting heuristic (often useful)
    if time.time() < deadline:
        sorted_desc = np.sort(best_a)[::-1]
        sorted_desc = _project_to_simplex(sorted_desc, S)
        _maybe_update(sorted_desc)

        sorted_asc = np.sort(best_a)
        sorted_asc = _project_to_simplex(sorted_asc, S)
        _maybe_update(sorted_asc)

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
