A fast stochastic optimizer for the self‑convolution L∞ problem.

The score (with a unit‑sum vector) equals 2·n·max(a*a).
We minimise max(a*a) by projected Adam on the probability simplex.

1. **Soft‑max smoothing** – for the first few hundred iterations we replace

2. **Exact sub‑gradient** – after the smoothing phase we switch to the true
   sub‑gradient of the max (indicator of the active convolution entries),
   with the usual diagonal correction for i=j.

The algorithm explores many lengths (500 – 8000) and a variety of
initialisations (uniform, arcsine, sparse Dirichlet, previous best).
Each run is limited by the remaining time budget; the best candidate seen
so far (according to the black‑box `evaluate_sequence`) is returned.

All updates are projected onto the simplex with the Euclidean projection.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path for already feasible vectors
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    needed = 2 * n - 1
    size = 1 << ((needed - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[:needed]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    needed = a.size + b.size - 1
    size = 1 << ((needed - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[:needed]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)) on n points, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initialize_vector(kind: str, n: int) -> np.ndarray:
    """Return a feasible probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "random_sparse":
        # Dirichlet with small concentration → sparse vectors
        conc = 10 ** np.random.uniform(-2, -0.2)   # between 0.01 and ~0.63
        a = np.random.gamma(conc, 1.0, size=n)
        a /= a.sum()
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                # simple linear interpolation to the required length
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    # Ensure exact feasibility
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv_self: np.ndarray,
                      alpha: float, cur_max: float) -> np.ndarray:
    """Gradient of the soft‑max surrogate of max(conv_self)."""
    # w_i = exp(alpha*(b_i - cur_max)) / sum_j exp(...)
    exp_term = np.exp(alpha * (conv_self - cur_max))
    w = exp_term / exp_term.sum()
    conv_rev = _fft_convolve(a[::-1], w)
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv_self: np.ndarray,
                   eps: float = 1e-12) -> np.ndarray:
    """Exact sub‑gradient of max(conv_self) w.r.t. a."""
    cur_max = conv_self.max()
    active = np.where(conv_self >= cur_max - eps)[0]
    w = np.zeros_like(conv_self)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1:n - 1 + n]

    # Diagonal correction for terms where i=j (indices k=2i)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Stochastic projected‑Adam search for a low‑score step‑function.
    Runs for at most `budget_s` seconds and returns a list of non‑negative
    heights (sum is not required to be 1 – the evaluator rescales).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    # ------------------------------------------------------------------
    # Lengths to explore – a mix of powers of two and neighbours of the
    # historically best length (if known).
    lengths = [512, 768, 1024, 1536, 2048, 3072, 4096, 6144, 8192]
    try:
        base = len(height_sequence_1)   # type: ignore[name-defined]
        lengths += [int(base * f) for f in (0.75, 1.0, 1.25) if int(base * f) > 0]
    except Exception:
        pass
    lengths = sorted(set(l for l in lengths if l >= 500))

    # Global best
    best_seq = None           # type: np.ndarray | None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Main exploration loop – keep launching runs while time permits.
    while time.time() < deadline:
        # Randomly pick a length (biased towards smaller lengths for speed)
        n = int(np.random.choice(lengths))
        # Pick an initialisation strategy
        init_kind = np.random.choice(
            ["uniform", "arcsine", "random", "random_sparse", "prev"]
        )
        a = _initialize_vector(init_kind, n)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps_adam = 1e-8
        lr0 = 0.12

        # Soft‑max schedule
        alpha = 1.0               # initial temperature
        alpha_factor = 1.02
        alpha_max = 2000.0
        softmax_iters = 350       # after this we switch to exact sub‑gradient

        # Local best (proxy based on max conv)
        conv_self = _fft_self_convolve(a)
        cur_max = conv_self.max()
        best_local_max = cur_max
        best_local_seq = a.copy()

        iteration = 0
        slice_deadline = time.time() + min((deadline - time.time()) * 0.4, 18.0)

        while time.time() < slice_deadline:
            iteration += 1

            # ---- compute objective proxy (max of autocorrelation) ----
            conv_self = _fft_self_convolve(a)
            cur_max = conv_self.max()

            if cur_max < best_local_max - 1e-14:
                best_local_max = cur_max
                best_local_seq = a.copy()
                print(
                    f"[{time.time():.1f}] n={n} init={init_kind} iter={iteration} "
                    f"maxB={best_local_max:.8e}"
                )

            # ---- gradient -------------------------------------------------
            if iteration <= softmax_iters and alpha < alpha_max:
                grad = _softmax_gradient(a, conv_self, alpha, cur_max)
                alpha = min(alpha_max, alpha * alpha_factor)
            else:
                grad = _exact_gradient(a, conv_self)

            # ---- Adam update + simplex projection -------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            lr = lr0 / np.sqrt(iteration)
            step = lr * m / (np.sqrt(v) + eps_adam)
            a = a - step
            a = _project_onto_simplex(a, s=1.0)

            # ---- occasional jitter to escape tiny plateaus ---------------
            if iteration % 1800 == 0 and np.random.rand() < 0.08:
                jitter = np.random.randn(n) * 2e-6
                a += jitter
                a = _project_onto_simplex(a, s=1.0)

            # early exit if proxy already beats global best by a comfortable margin
            if best_score != float("inf"):
                # proxy score = 2 * n * best_local_max (since sum=1)
                proxy_score = 2.0 * n * best_local_max
                if proxy_score < best_score * 0.99:
                    break

        # ------------------------------------------------------------------
        # Evaluate the best local candidate with the true black‑box.
        try:
            candidate = best_local_seq.tolist()
            final_score = evaluate_sequence(candidate)
        except Exception:
            final_score = float("inf")

        if final_score < best_score:
            best_score = final_score
            best_seq = best_local_seq.copy()
            print(
                f"*** New global best → n={n} init={init_kind} "
                f"score={best_score:.6f}"
            )
            # Stop if we have reached the target bound.
            if best_score <= 1.503:
                break

        # If time is almost exhausted, break the outer loop.
        if time.time() + 2.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe default.
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    print(
        f"Final best score: {best_score:.6f} "
        f"(length = {len(best_seq)})"
    )
    return best_seq.tolist()
