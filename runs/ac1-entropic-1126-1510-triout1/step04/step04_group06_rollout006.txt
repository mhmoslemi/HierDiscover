def descend(a, deadline, n, size_fft, target_sum, max_iters=100000):
    a_cur = a.copy()
    best = a_cur.copy()
    best_score = quick_score(a_cur, n, size_fft)
    # momentum
    v = np.zeros_like(a_cur)
    mu = 0.9
    # learning rate schedule, start from eta0, decays.
    eta0 = target_sum * 0.2 / n   # maybe.
    it=0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a_cur, size_fft)
        M = conv.max()
        cur_score = M  # after normalization
        if cur_score < best_score:
            best_score = cur_score
            best = a_cur.copy()
        # compute gradient as described
        k = int(conv.argmax())
        start = max(0, k - (n-1))
        end = min(k, n-1)
        # compute gradient
        grad = np.zeros_like(a_cur)
        j = np.arange(start, end+1)
        grad[j] = 2.0 * a_cur[k - j]
        # step
        eta = eta0 / (1.0 + it * 1e-5)  # decays
        v = mu * v - eta * grad
        a_new = a_cur + v
        # random noise occasionally
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
        # projection
        a_new = _project_to_simplex(a_new, target_sum)
        a_cur = a_new
        it += 1
    return best, best_score

"""
Search for a non‑negative step‑function (list of heights) that minimises
the self‑convolution peak defined by `evaluate_sequence`.

Algorithm
---------

1. **Generate a rich pool of analytic candidates**  
   Families include uniform, inverse‑sqrt, double‑inverse‑sqrt, power‑law,
   exponential decay, cosine / sine shapes, Gaussian bumps, triangular,
   beta‑shaped, monotone decreasing power laws and random block patterns.
   Every candidate is projected onto the simplex
   {a ≥ 0, Σa = √(2 n)} and its exact score (the maximal convolution entry)
   is obtained by an FFT‑based full convolution.

2. **Keep the K best raw candidates** and improve each of them by a
   projected sub‑gradient descent on the objective  
   `M(a) = max_k (a*a)[k]`.  At each iteration we
   * compute the full convolution,
   * locate the index `k*` of the maximum,
   * form the sub‑gradient `g_j = 2·a_{k*‑j}` for all feasible `j`,
   * take a momentum step (`v ← μ·v – η·g`) and project back onto the simplex.
   The learning‑rate decays slowly, and a small Gaussian kick is added
   every few thousand iterations to escape plateaus.

3. **Final random‑block sweep** – with the remaining time budget we
   generate many random block candidates (few constant blocks, possibly
   monotone) and keep any improvement.

The best sequence found before the 1000 s budget expires is returned.
All operations respect the security checks of `evaluate_sequence`.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a using FFT (returns length 2·n‑1)."""
    fu = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fu * fu, n=size_fft)
    n = a.shape[0]
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, size_fft: int) -> float:
    """Exact objective value (max entry of a*a) for a sequence already normalised."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on M(a) = max_k (a*a)[k].
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, size_fft)

    # momentum buffer
    v = np.zeros_like(a)
    mu = 0.9
    # base learning rate – a fraction of the typical coefficient size
    eta0 = 0.6 * target_sum / n

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        # index of the maximal convolution entry
        k = int(conv.argmax())

        # sub‑gradient of the max entry w.r.t. a
        grad = np.zeros_like(a)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        if start <= end:
            j = np.arange(start, end + 1, dtype=int)
            grad[j] = 2.0 * a[k - j]

        # learning‑rate decay
        eta = eta0 / (1.0 + it * 1e-5)

        # momentum update (v stores the negative direction)
        v = mu * v - eta * grad
        a_new = a + v

        # occasional random kick to diversify the search
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise

        # projection onto the simplex Σa = √(2 n)
        a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    return best_a, best_score

        decay_r = np.random.rand() * 0.015 + 0.0002
        a = np.exp(-decay_l * idx) + np.exp(-decay_r * (n - 1 - idx))
        add_candidate(a)

    # (g) Cosine‑based (large at ends, small in the middle)
    for _ in range(120):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        add_candidate(a)

    # (h) Sine‑based (zero at ends, peak in the middle)
    for _ in range(120):
        p = np.random.rand() * 6.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** p
        add_candidate(a)

    # (i) Gaussian‑like bump (centered)
    for _ in range(120):
        sigma = (np.random.rand() * 0.25 + 0.05) * n
        centre = (n - 1) / 2.0 + (np.random.rand() - 0.5) * n * 0.1
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        add_candidate(a)

    # (j) Triangular (symmetric peak at centre)
    for _ in range(100):
        peak = np.random.rand() * 3.0 + 0.5
        a = np.minimum(idx + 1, n - idx) ** peak
        add_candidate(a)

    # (k) Beta‑shaped (a_i ∝ (i+1)^α * (n-i)^β)
    for _ in range(100):
        alpha = np.random.rand() * 3.0 + 0.2
        beta = np.random.rand() * 3.0 + 0.2
        a = (idx + 1) ** alpha * (n - idx) ** beta
        add_candidate(a)

    # (l) Monotone decreasing power‑law (a_i ∝ (n‑i)^p)
    for _ in range(100):
        p = np.random.rand() * 2.5 + 0.1
        a = (n - idx) ** p
        add_candidate(a)

    # (m) Random monotone sequence (sorted decreasing random numbers)
    for _ in range(120):
        a = np.sort(np.random.rand(n))[::-1]
        add_candidate(a)

    # (n) Random block candidates (few large constant blocks, possibly asymmetric)
    for _ in range(400):
        nb = np.random.randint(2, 9)                 # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        add_candidate(cand)

    # (o) Reference sequence, if available
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        add_candidate(a_ref)

    # ------------------------------------------------------------------
    best_candidates.sort(key=lambda x: x[0])
    print(f"[exploration] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # ---------- 2. REFINEMENT (sub‑gradient descent) -----------------
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    for rank, (raw_score, a0) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank+1} (raw score {raw_score:.6f})")
        remaining = deadline - time.time()
        max_iters = int(min(200_000, remaining * 1500))   # generous allocation
        refined_a, refined_score = _subgradient_descent(
            a0, deadline, n, size_fft, target_sum, max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---------- 3. FINAL RANDOM BLOCK SWEEP -------------------------
    while time.time() < deadline:
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")
        # early exit if time is about to run out
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
