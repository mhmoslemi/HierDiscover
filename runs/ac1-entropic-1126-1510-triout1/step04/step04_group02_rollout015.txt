    a = a0.copy()
    best_seq = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    eta0 = 0.5
    alpha = 5.0

    alpha_max = 300.0  # keep moderate
    size_fft = 1 << ((3 * n - 2).bit_length())
    it = 0
    while time.time() < deadline:
        it += 1
        # current
        conv = _fft_self_convolve(a)
        cur_max = conv.max()

            # optionally print

"""
Propose a step‑function (non‑negative list) with a low evaluation score.

The algorithm consists of three stages:

1. **Coarse shape search** – for each candidate length we test a
   one‑parameter family a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ (p≈0.5 gives the arcsine
   law).  A small mixture with the uniform distribution is also tried.
   All shapes are normalised to sum 1 and scored by a fast FFT based
   evaluation of 2·n·max(conv).

2. **Exponential‑gradient refinement** – the best few candidates from the
   coarse search are polished with a mirror‑descent / exponentiated‑gradient
   method.  The gradient of a smooth log‑sum‑exp (soft‑max) surrogate for the
   L∞‑norm is used while the temperature is moderate; once it reaches a
   threshold we fall back to the true sub‑gradient of the max entry.
   The step‑size decays as η₀ / √t.

3. **Local random‑move hill climbing** – a final stochastic refinement
   moves a tiny amount of mass between two indices whenever this reduces
   the true evaluation score.

The routine respects the 1000 s budget, repeatedly updates the best
sequence found, and finally returns it as a plain Python list.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of *v* onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    theta = 0.0 if rho_arr.size == 0 else cssv[rho_arr[-1]] / (rho_arr[-1] + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (length 2·n‑1) convolution of *a* with itself via a real FFT."""
    n = a.shape[0]
    # smallest power‑of‑two ≥ 2·n‑1
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max entry of a*a at index *k*."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """Gradient of (1/α)·log∑exp(α·b) w.r.t. a (soft‑max surrogate)."""
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                         # fallback – pure max
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]                          # needed because convolution is not a correlation
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

def _initial_sequence(n: int, kind: str) -> np.ndarray:

    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a = np.maximum(a, 0.0)

    elif kind == "beta":

        a = np.maximum(a, 0.0)

        a = np.full(n, 1.0 / n) if s == 0 else a / s
    elif kind == "arcsine":
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a /= a.sum()
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]

    else:

def _beta_power_shape(n: int, p: float) -> np.ndarray:
    """Distribution a_i ∝ ((i+½)(n‑i‑½))⁻ᵖ  (symmetric, p>0)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _asymmetric_power_shape(n: int, pl: float, pr: float) -> np.ndarray:
    """Distribution a_i ∝ (i+½)⁻ˡ · (n‑i‑½)⁻ʳ ."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (-pl) * (n - i - 0.5) ** (-pr)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _make_candidate(n: int, p: float, w: float) -> np.ndarray:
    """
    Mixture of the power‑law shape with the uniform distribution.
    w = weight of the uniform part (0 → pure power law, 1 → pure uniform).
    """
    shape = _beta_power_shape(n, p)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    cand = (1.0 - w) * shape + w * uniform
    # both components already sum to 1 → cand also sums to 1
    return _project_onto_simplex(cand, s=1.0)


# ----------------------------------------------------------------------
def _grid_search_best(n: int) -> tuple[np.ndarray, float]:
    """
    Exhaustively scan a small grid of (p,w) and a few asymmetric families.
    Returns the best vector found (as a numpy array) together with its
    internal score 2·n·max(conv).
    """
    best_seq = None
    best_score = np.inf

    # ----- symmetric power‑law + uniform mixture -----------------------
    p_vals = np.linspace(0.30, 0.70, 9)          # coarse around the arcsine exponent 0.5
    w_vals = np.linspace(0.0, 1.0, 11)          # mixture weight
    for p in p_vals:
        shape = _beta_power_shape(n, p)
        for w in w_vals:
            cand = (1.0 - w) * shape + w * np.full(n, 1.0 / n)
            cand = _project_onto_simplex(cand, s=1.0)
            score = 2.0 * n * _fft_self_convolve(cand).max()
            if score < best_score:
                best_score, best_seq = score, cand.copy()

    # ----- asymmetric two‑parameter family ----------------------------
    pl_vals = np.linspace(0.20, 0.80, 4)   # left exponent
    pr_vals = np.linspace(0.20, 0.80, 4)   # right exponent
    for pl in pl_vals:
        for pr in pr_vals:
            cand = _asymmetric_power_shape(n, pl, pr)
            score = 2.0 * n * _fft_self_convolve(cand).max()
            if score < best_score:
                best_score, best_seq = score, cand.copy()

    # ----- fallback: pure arcsine (p=0.5) -----------------------------
    arcsine = _initial_sequence(n, "arcsine")
    score = 2.0 * n * _fft_self_convolve(arcsine).max()
    if score < best_score:
        best_score, best_seq = score, arcsine.copy()

    return best_seq, float(best_score)


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient refinement.
    Returns the best vector seen and its internal (2·n·max) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # hyper‑parameters
    eta0 = 0.5                     # initial learning‑rate factor
    alpha = 5.0                    # start with a moderate temperature
    alpha_mul = 1.02
    alpha_max = 300.0              # keep the surrogate smooth
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()

        # gradient (soft‑max while temperature small, then true sub‑gradient)
        if alpha < alpha_max:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
            alpha = min(alpha_max, alpha * alpha_mul)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # step‑size – diminishing
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # safeguard against under‑flow / negativity
        a = np.maximum(a, 0.0)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # occasional tiny jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a fraction of mass from one index
    to another whenever the true evaluation score improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.10
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
            # optional progress output
            # print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence``.  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # 1. Choose a set of candidate lengths (moderate to allow many
    #    iterations yet still large enough to approach the theoretical
    #    limit).  We keep the list short to spend most time on optimisation.
    base_n = 8000
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        pass
    lengths = sorted({max(500, int(base_n * f))
                      for f in (0.75, 1.0, 1.25, 1.5, 2.0)
                      if max(500, int(base_n * f)) <= 20000})
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503

    # --------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline - 5.0:
            break

        # allocate a fair slice of the remaining budget to this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(0.2, remaining * 0.35)

        # -----------------------------------------------------------------
        # Produce a list of promising initial vectors for this length.
        init_candidates = []

        # (a) simple predefined seeds
        for kind in ("prev", "arcsine", "beta", "random", "uniform"):
            try:
                init_candidates.append(_initial_sequence(n, kind))
            except Exception:
                pass

        # (b) best shape found by a cheap grid search
        shape_best, _ = _grid_search_best(n)
        init_candidates.append(shape_best)

        # keep only the few brightest candidates (according to the cheap score)
        def cheap_score(v):
            return 2.0 * n * _fft_self_convolve(v).max()
        init_candidates = sorted(init_candidates, key=cheap_score)[:4]

        # -----------------------------------------------------------------
        # Optimise each candidate with the exponentiated‑gradient routine.
        for a0 in init_candidates:
            if time.time() >= slice_deadline:
                break
            seq, internal_score = _optimize_exponential(a0, n, slice_deadline)

            # Convert to Python list before calling the black‑box validator.
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = internal_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq.copy()
                print(f"*** New global best → len={n} score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break
        if overall_best_score <= target_score:
            break

    # --------------------------------------------------------------
    # Final stochastic refinement on the globally best vector (if any).
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n = len(overall_best_seq)
        seq, _ = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = _optimize_exponential(seq, n, deadline)[1]   # fallback
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq.copy()
            print(f"*** After local refinement score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # Safety net – should never be needed.
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Safety fallback used.")

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
