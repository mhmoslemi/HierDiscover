def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate value and gradient w.r.t. the free parameters y.

    Parameters
    ----------
    y : np.ndarray            – unconstrained log‑softmax parameters (len = n)
    mu : float                – temperature (>0); µ=0 gives the hard max.
    n : int                   – length of the step function.
    S : float                 – target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val   – surrogate objective value.
    grad_y  – gradient ∂f/∂y (shape (n,)).
    """
    # probabilities on the simplex and actual heights (sum = S)
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    conv = np.convolve(a, a)                     # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        # avoid overflow in exp
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:          # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (see derivation in the original code)
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule through the soft‑max
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j·g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize: returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """
    Exact evaluate_sequence value for a non‑negative vector a.
    Uses the analytical formula to avoid the overhead of the public wrapper.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_c / (sum_a ** 2))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluate_sequence score.
    The routine respects the given time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size (use reference length if available)
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000                      # a comfortable size for the budget
        have_ref = False

    S = math.sqrt(2.0 * n)            # required ℓ¹‑norm

    # bookkeeping
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    def _maybe_update(a_vec: np.ndarray) -> None:
        """Update the incumbent if the new candidate improves the score."""
        nonlocal best_score, best_a
        if time.time() > deadline:
            return
        sc = _score(a_vec)
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # initialise a few sensible candidates
    initial_as = []

    # 0 – reference (if any)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        initial_as.append(a_ref)

    # 1 – uniform
    initial_as.append(np.full(n, S / n))

    # 2 – decreasing exponential (large on the left)
    alpha = rng.uniform(0.002, 0.01)
    y_exp = -alpha * np.arange(n, dtype=float)
    initial_as.append(S * _softmax(y_exp))

    # 3 – increasing exponential (large on the right)
    y_rev = -alpha * np.arange(n - 1, -1, -1, dtype=float)
    initial_as.append(S * _softmax(y_rev))

    # 4 – random Gaussian
    y_gauss = rng.standard_normal(n) * 0.5
    initial_as.append(S * _softmax(y_gauss))

    # 5‑9 – random block patterns
    for _ in range(5):
        nb = rng.integers(2, 12)
        cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(np.arange(n), cuts)
        vals = rng.random(nb)
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, S)
        initial_as.append(cand)

    # ------------------------------------------------------------------
    # Turn the initial heights into log‑softmax parameters y = log(a+ε)
    init_ys = [np.log(a + 1e-12) for a in initial_as]

    # ------------------------------------------------------------------
    # Smooth‑max continuation/L‑BFGS phase
    mu_factor = 0.85                 # slower cooling → finer optimisation
    mu_min = 1e-9
    maxiter_per_mu = 400
    restart_counter = 0

    while time.time() < deadline and restart_counter < 30:
        # pick a starting point: either from the pool or a jittered copy of the best so far
        if restart_counter < len(init_ys):
            y = init_ys[restart_counter].copy()
        else:
            # jitter around the current best solution
            jitter = rng.normal(scale=0.2, size=n)
            y = np.log(best_a + 1e-12) + jitter

        # initial temperature based on the current max‑convolution
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = max(conv_cur.max(), 1.0)   # ensure positive

        # continuation loop for this restart
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

        restart_counter += 1
        print(f"[restart] completed {restart_counter}/{30}")

    # ------------------------------------------------------------------
    # Large‑scale random block search (more trials now that we have time left)
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_trials = int(min(30_000, remaining * 200))
        print(f"[block] performing up to {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(2, 15)
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing around the best candidate
    remaining = deadline - time.time()
    if remaining > 5.0:
        a = best_a.copy()
        it_hc = 0
        while time.time() < deadline and it_hc < 8000:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if a[i] <= 0:
                continue
            # move a small random fraction of mass from i to j
            delta = a[i] * rng.uniform(0.001, 0.03)
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # projection guarantees feasibility
            a_new = _project_to_simplex(a_new, S)
            sc = _score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                print(f"[hill] improved to {best_score:.6f} (it {it_hc})")
            it_hc += 1

    # ------------------------------------------------------------------
    # Final projected sub‑gradient descent with back‑tracking line search
    remaining = deadline - time.time()
    if remaining > 2.0:
        a = best_a.copy()
        it = 0
        while time.time() < deadline and it < 20_000:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            # gradient of the hard max w.r.t. a
            g_a = 2.0 * np.convolve(w, a[::-1])[n - 1:n - 1 + n]

            # back‑tracking line search
            step = 1.0
            while True:
                a_new = a - step * g_a
                a_new = _project_to_simplex(a_new, S)
                sc_new = _score(a_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    a = a_new
                    print(f"[grad] improved to {best_score:.6f} (it {it})")
                    break
                step *= 0.5
                if step < 1e-12:
                    break
            it += 1

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
