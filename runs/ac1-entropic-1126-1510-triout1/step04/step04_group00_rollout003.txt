    lengths = [2000, 4000, 8000, 16000]

    if pdf_sum == 0:
        return np.full(n, 1.0/n)
    pdf = pdf / pdf_sum
    return pdf

"""
Propose a low‑score step‑function.

The program searches for a probability vector `a` (non‑negative,
∑a_i = 1) that minimises

        score = 2·n·max(a * a)            (n = len(a))

i.e. the maximal value of the self‑convolution.
The search is performed in three stages:

1. **Low‑dimensional global search** –  
   For a given length `n` the distribution is represented as a mixture

        a = w·Beta(p,p) + (1-w)·Uniform ,

   where `Beta(p,p)` is the symmetric beta‑distribution discretised on the
   mesh `{(i+½)/n}`.  The two parameters `(p,w) ∈ [0.1,2]×[0,1]` are
   optimised with a simple Differential Evolution (DE) routine.  The
   objective is evaluated by a fast FFT‑based `score_fast` function
   (exactly the same as `evaluate_sequence` for normalised vectors).

2. **Local refinement** –  
   Starting from the best mixture found, a momentum‑based gradient descent
   is executed.  For low “temperature’’ a smooth soft‑max surrogate of the
   max‑operator is used, later the true sub‑gradient of the maximal
   convolution entries.  After each step the iterate is projected back onto
   the probability simplex.

3. **Best‑ever tracking** –  
   The global best vector (any length) is kept, and the previous construction
   `height_sequence_1` (if present) is used as an additional starting point.

The algorithm respects the 1000 s time budget, prints progress, and finally
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, p: float) -> np.ndarray:
    """
    Symmetric Beta(p,p) density on points x_i = (i+0.5)/n,
    normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Guard against p very close to 0 (x**(p-1) may blow up)
    # Use log‑space formulation for stability.
    log_pdf = (p - 1) * (np.log(x) + np.log(1.0 - x))
    # Replace -inf by a large negative number (happens only at the ends)
    log_pdf = np.where(np.isfinite(log_pdf), log_pdf, -1e300)
    pdf = np.exp(log_pdf)
    total = pdf.sum()
    if total == 0 or not np.isfinite(total):
        # fall back to uniform if numerical issues arise
        return np.full(n, 1.0 / n, dtype=np.float64)
    return pdf / total


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _score_fast(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a normalised vector (sum(a)=1):
        score = 2·n·max(a*a)
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Momentum based gradient descent with a soft‑max surrogate that later
    switches to the exact sub‑gradient.
    Returns the best vector seen and its fast score.
    """
    a = a0.copy()
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta = 0.9                # momentum coefficient
    eta0 = 0.5                # initial step‑size factor
    alpha = 5.0               # temperature for soft‑max
    alpha_factor = 1.02
    alpha_max = 500.0

    best_a = a.copy()
    best_score = _score_fast(a)

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.1f}] refine n={n} iter={iteration} "
                  f"score≈{best_score:.6f}")

        # ---- gradient computation ----
        if alpha < alpha_max:
            # smooth approximation of the max
            exp_term = np.exp(alpha * (b - cur_max))
            w = exp_term / exp_term.sum()
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
            alpha = min(alpha_max, alpha * alpha_factor)
        else:
            # exact sub‑gradient from active entries
            eps = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(b >= cur_max - eps)
            w = np.zeros_like(b)
            w[active] = 1.0
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            # diagonal correction (terms i=j)
            even_mask = (active % 2 == 0)
            diag_idx = (active[even_mask] // 2)
            if diag_idx.size > 0:
                grad[diag_idx] -= a[diag_idx]

        # ---- momentum + step ----
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(iteration)
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional jitter to escape plateaux
        if iteration % 2000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

        # early stop if we are already below target
        if best_score <= 1.503:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) with a
    score below the target 1.503.  Uses a mixture of global DE on
    a two‑parameter beta‑distribution and a local gradient refinement.
    Returns a plain Python list.
    """
    np.random.seed(seed)
    # reserve a couple of seconds for final housekeeping
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Lengths to try – a few powers of two give a good trade‑off.
    lengths = [4000, 8000, 16000]
    # Guarantee we have at least one length
    if not lengths:
        lengths = [8000]

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    # If a previous construction exists, use it as a baseline.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = prev / np.sum(prev)  # normalise to the simplex
        prev_score = _score_fast(prev)
        if prev_score < overall_best_score:
            overall_best_score = prev_score
            overall_best_seq = prev.copy()
            print(f"[{time.time():.1f}] baseline previous seq score={prev_score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget to this length
        remaining = max(0.0, deadline - time.time())
        slice_deadline = time.time() + min(remaining * 0.35, 250.0)

        # ---- Differential Evolution on (p,w) -------------------------
        popsize = 20
        # population: columns = [p, w]
        pop = np.empty((popsize, 2), dtype=np.float64)
        pop[:, 0] = 0.1 + np.random.rand(popsize) * (2.0 - 0.1)   # p ∈ [0.1,2]
        pop[:, 1] = np.random.rand(popsize)                     # w ∈ [0,1]

        def eval_params(params: np.ndarray) -> float:
            p, w = params
            beta_seq = _beta_distribution(n, p)
            uniform_seq = np.full(n, 1.0 / n, dtype=np.float64)
            a = w * beta_seq + (1.0 - w) * uniform_seq
            # normalise (should already be 1, but keep numerics safe)
            a /= a.sum()
            return _score_fast(a)

        scores = np.empty(popsize, dtype=np.float64)
        for i in range(popsize):
            scores[i] = eval_params(pop[i])

        best_idx = int(np.argmin(scores))
        best_params = pop[best_idx].copy()
        best_score = float(scores[best_idx])

        generation = 0
        while time.time() < slice_deadline and generation < 80:
            generation += 1
            for i in range(popsize):
                # mutation: DE/rand/1
                idxs = list(range(popsize))
                idxs.remove(i)
                a, b, c = np.random.choice(idxs, 3, replace=False)
                mutant = pop[a] + 0.8 * (pop[b] - pop[c])
                # respect bounds
                mutant[0] = np.clip(mutant[0], 0.1, 2.0)
                mutant[1] = np.clip(mutant[1], 0.0, 1.0)

                # crossover
                cross = np.random.rand(2) < 0.7
                if not cross.any():
                    cross[np.random.randint(0, 2)] = True
                trial = np.where(cross, mutant, pop[i])

                trial_score = eval_params(trial)

                if trial_score < scores[i]:
                    pop[i] = trial
                    scores[i] = trial_score
                    if trial_score < best_score:
                        best_score = trial_score
                        best_params = trial.copy()

            # intermittent progress report
            if generation % 5 == 0:
                print(f"[{time.time():.1f}] n={n} gen={generation} "
                      f"DE best_score≈{best_score:.6f}")

            if best_score <= 1.503:
                break

        # ------------------------------------------------------------------
        # Build the best mixture from DE and launch local refinement.
        p_opt, w_opt = best_params
        beta_seq = _beta_distribution(n, p_opt)
        uniform_seq = np.full(n, 1.0 / n, dtype=np.float64)
        a0 = w_opt * beta_seq + (1.0 - w_opt) * uniform_seq
        a0 /= a0.sum()

        # Give the local optimiser the remaining time of this slice.
        refine_deadline = min(slice_deadline, time.time() + 120.0)
        a_refined, refined_score = _refine_sequence(a0, refine_deadline)

        # Compare with the current overall best.

            overall_best_score = refined_score
            overall_best_seq = a_refined.copy()
            print(f"*** New global best → n={n} score={overall_best_score:.6f}")

        # Early exit if we already beat the target.
        if overall_best_score <= 1.503:
            print("Target achieved – terminating search.")
            break

    # ------------------------------------------------------------------
    # Fallback to a uniform vector if everything else failed.
    if overall_best_seq is None:

    try:

        # In rare cases rounding may differ – keep the better of the two.
        if final_score < overall_best_score:

          f"(length = {len(overall_best_seq)})")
    # Return a plain Python list as required by the harness.
    return overall_best_seq.tolist()
