"""
A fast, multi‑restart sub‑gradient optimiser for the self‑convolution peak.
The idea:
    • Initialise a diverse pool of candidates using many analytic families
      (uniform, power‑law, exponential, logistic, random monotone, block).
    • Project each candidate onto the simplex {a ≥ 0, Σa = √(2 n)}.
    • For the best few candidates run a pure sub‑gradient descent on the
      exact max‑convolution (no soft‑max smoothing).  The sub‑gradient at the
      current maximal index can be computed in O(n) by simple slicing.
    • After a short Adam‑based refinement (same as the reference code) the
      best sequence found before the deadline is returned.
The optimiser respects the 1000 s time budget, prints progress and works
with any `height_sequence_1` that may be present in the global namespace.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution via FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution a * a."""
    return _conv_fft(a, a, size_fft)


def _grad_of_conv_at_idx(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of the k‑th convolution entry (k = idx) w.r.t. a.
    For the max‑entry this is a valid sub‑gradient.
    """
    n = a.shape[0]
    # exploit symmetry: b_k == b_{2n-2-k}
    if idx > 2 * n - 2 - idx:
        idx = 2 * n - 2 - idx
    g = np.zeros_like(a)

    if idx < n:
        # part from the left side of the overlap
        g[idx:] += a[: n - idx]
        # part from the right side (reversed)
        g[: idx + 1] += a[idx :: -1]
    else:
        # idx >= n : map to the mirrored index (should not happen after the
        # previous reduction, but kept for completeness)
        k = 2 * n - 2 - idx
        g[k:] += a[: n - k]
        g[: k + 1] += a[k :: -1]
    return g


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective (no security checks) used during the search."""
    conv = _self_convolution(a, size_fft)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    The Adam‑based smoothing optimiser from the reference solution.
    Returns the best sequence observed and its (exact) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    eta0 = target_sum * 0.5 / n
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max smoothing of the max -----
        shifted = conv - conv.max()
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        a_rev = a[::-1]
        conv_w = np.fft.rfft(w, n=size_fft) * np.fft.rfft(a_rev, n=size_fft)
        grad = 2.0 * np.fft.irfft(conv_w, n=size_fft)[: a.size]

        # ----- Adam update -----
        eta = eta0 / (1.0 + it * 1e-5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = _project_to_simplex(a_new + noise, target_sum)

        a = a_new
        T *= 0.9997
        it += 1

    return best_a, best_score

        t = (idx / (n - 1) + shift) * 2 - 1        # map into [‑1, 1]
        a = 1.0 / (1.0 + np.exp(-k * t))
        add_candidate(a)
        add_candidate(a[::-1])

    # random monotone (sorted uniform random numbers)
    for _ in range(80):
        rand_vals = np.random.rand(n)
        a = np.sort(rand_vals)          # increasing
        add_candidate(a)
        add_candidate(a[::-1])          # decreasing

    # random block candidates (few constant blocks)
    for _ in range(250):
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        add_candidate(cand)

    # reference sequence if it exists
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        add_candidate(a_ref)

    best_candidates = sorted(best_candidates, key=lambda x: x[0])
    print(f"[exploration] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # ---- 2. sub‑gradient descent on the exact max convolution  ---------
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    for rank, (raw_score, a0) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[subgrad start] candidate #{rank+1} (raw score {raw_score:.6f})")
        a = a0.copy()
        # number of inner iterations depends on remaining time
        remaining = deadline - time.time()
        max_iters = int(min(40_000, remaining * 800))   # about 800 iters/sec
        eta = target_sum * 0.5 / n                     # base step

        for it in range(max_iters):
            conv = _self_convolution(a, size_fft)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max / (a.sum() ** 2)

            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best_a = a.copy()
                print(f"[global] new best (subgrad) = {global_best_score:.6f}")

            idx_max = int(conv.argmax())
            grad = _grad_of_conv_at_idx(a, idx_max)

            # simple SGD step + projection
            a = a - eta * grad
            a = _project_to_simplex(a, target_sum)

            # decay step size
            eta *= 0.9995

            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # ---- 3. short Adam polishing of the current best  -----------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = int(min(150_000, remaining * 1400))
        refined_a, refined_score = _refine_candidate_adam(
            global_best_a, deadline, n, size_fft, target_sum, max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] improved after Adam = {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
