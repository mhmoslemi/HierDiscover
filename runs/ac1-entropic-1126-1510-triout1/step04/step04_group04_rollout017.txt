                    target_sum, deadline, max_iters):

    best_a = a.copy()
    best_max = _max_convolution(a, size_fft_conv)

    n = a.size
    # Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # schedule of p values
    p_schedule = [2,4,8,16,32]
    it = 0
    for p in p_schedule:
        # assign a limited number of iters per p, maybe max_iters // len(p_schedule)
        iters_p = max_iters // len(p_schedule)
        for _ in range(iters_p):
            it += 1
            if time.time() >= deadline:
                break
            # compute convolution a*a
            conv = _full_conv(a, a, size_fft_conv)  # shape 2n-1
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = float(cur_max)
                best_a = a.copy()
                print(f"[opt p{p}] it {it} new best = {best_max:.6f}")

            # p-norm objective
            # sum_pow could become huge; use np.float64
            sum_pow = np.sum(conv**p)
            if sum_pow == 0:

            w = conv ** (p - 1)
            w_rev = w[::-1]

            m = beta1 * m + (1 - beta1) * grad

"""
A fast, gradient‑based optimizer for the self‑convolution peak.
The method builds a small pool of promising random candidates,
projects each onto the ℓ₁‑simplex (Σa = √(2 n)), and then refines
them with an Adam‑style descent on a smooth p‑norm surrogate of the
max‑convolution (p = 2, 4, 8, 16, 32).  The surrogate gradient is
computed using two FFT‑based convolutions per iteration, which is
fast enough to perform many thousands of updates within the 1000 s
budget.  After the refinement a short hill‑climbing on the ordering
of the values is performed.  The returned sequence is non‑negative
and has the best self‑convolution peak found within the time limit.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, b: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a * b using FFT, returning the exact length."""
    fa = np.fft.rfft(a, n=size_fft)
    fb = np.fft.rfft(b, n=size_fft)
    conv = np.fft.irfft(fa * fb, n=size_fft)
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of a * a."""
    conv = _full_conv(a, a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _pnorm_optimize(
    a0: np.ndarray,
    size_fft_conv: int,
    size_fft_cross: int,
    target_sum: float,
    deadline: float,
    max_iters: int,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style minimisation of the p‑norm surrogate of max(a*a).
    Returns the best sequence (lowest exact max‑convolution) seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_max = _max_convolution(a, size_fft_conv)

    n = a.size
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # schedule of p values – start smooth, end sharp
    p_schedule = [2, 4, 8, 16, 32]
    it = 0
    it_per_stage = max(1, max_iters // len(p_schedule))

    for p in p_schedule:
        for _ in range(it_per_stage):
            it += 1
            if time.time() >= deadline:
                break

            # ----- convolution a*a -------------------------------------------------
            conv = _full_conv(a, a, size_fft_conv)          # length 2n‑1
            cur_max = conv.max()
            if cur_max < best_max:
                best_max = float(cur_max)
                best_a = a.copy()
                print(f"[pnorm p{p}] it {it:,} new best = {best_max:.6f}")

            # ----- p‑norm value ----------------------------------------------------
            sum_pow = np.sum(conv ** p)
            if sum_pow == 0.0:
                continue
            f_val = sum_pow ** (1.0 / p)

            # ----- gradient of p‑norm ----------------------------------------------
            w = conv ** (p - 1)                              # length 2n‑1
            w_rev = w[::-1]
            conv_w_rev_a = _full_conv(w_rev, a, size_fft_cross)  # length 3n‑2
            # slice (n‑1) … (2n‑2)  → length n
            start = n - 1
            sub = conv_w_rev_a[start : start + n]           # shape (n,)
            # reversed to obtain Σ_j w_{i+j} a_j
            grad = (2.0 / (f_val ** (p - 1))) * sub[::-1]

            # ----- Adam update -----------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # learning rate decays like 1/√t, scale with average entry size
            eta0 = target_sum * 0.5 / n
            eta = eta0 / math.sqrt(it)

            a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

            # ----- project onto simplex ---------------------------------------------
            a = _project_to_simplex(a, target_sum)

    return best_a, best_max


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation score (i.e. the peak of the self‑convolution).
    The function respects a 1000 s budget and returns the best sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # ----- problem size -------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        ref_seq = np.array(globals()["height_sequence_1"], dtype=float)
        have_ref = True
    else:
        n = 3000
        ref_seq = None
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # Σa = √(2 n)
    size_fft_conv = _fft_size(2 * n - 1)     # for a*a
    size_fft_cross = _fft_size(3 * n - 2)    # for w_rev * a (gradient)

    # ----- helpers ------------------------------------------------------------
    def quick_score(a: np.ndarray) -> float:
        return _max_convolution(a, size_fft_conv)

    # store top‑K raw candidates (sorted by score)
    K = 12
    raw_candidates: list[tuple[float, np.ndarray]] = []

    def add_candidate(a: np.ndarray) -> None:
        a = _project_to_simplex(a, target_sum)
        sc = quick_score(a)
        raw_candidates.append((sc, a))
        raw_candidates.sort(key=lambda x: x[0])
        if len(raw_candidates) > K:
            raw_candidates.pop()   # keep only K best

    # ----- generate a wide variety of seeds ----------------------------------
    idx = np.arange(n, dtype=float)

    # uniform baseline
    add_candidate(np.ones(n))

    # reference, if present
    if have_ref:
        add_candidate(ref_seq)

    # Dirichlet draws with several concentration parameters
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(10):
            probs = np.random.dirichlet(np.full(n, alpha))
            add_candidate(probs * target_sum)

    # Power‑law decreasing shapes
    for _ in range(20):
        p = np.random.rand() * 3.0 + 0.2
        add_candidate(1.0 / (idx + 1) ** p)

    # Exponential decay shapes
    for _ in range(20):
        dec = np.random.rand() * 0.015 + 0.001
        add_candidate(np.exp(-dec * idx))

    # Random block‑wise constant sequences (2–6 blocks)
    for _ in range(80):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        add_candidate(cand)

    print(f"[sampling] best raw score = {raw_candidates[0][0]:.6f}")

    # ----- optimisation of the best raw candidates ---------------------------
    global_best_a = raw_candidates[0][1].copy()
    global_best_score = raw_candidates[0][0]

    for cand_idx, (init_score, a0) in enumerate(raw_candidates):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest budget per candidate
        max_iters = int(min(200_000, remaining * 2000))
        print(f"[refine] candidate #{cand_idx+1} init_score={init_score:.6f} max_iters={max_iters}")

        a_opt, sc_opt = _pnorm_optimize(
            a0,
            size_fft_conv,
            size_fft_cross,
            target_sum,
            deadline,
            max_iters,
        )
        if sc_opt < global_best_score:
            global_best_score = sc_opt
            global_best_a = a_opt.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ----- final permutation hill‑climbing -----------------------------------
    a = global_best_a.copy()
    best_score = global_best_score

    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        a_swapped = a.copy()
        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
        sc = _max_convolution(a_swapped, size_fft_conv)
        if sc < best_score:
            best_score = sc
            a = a_swapped
            global_best_a = a_swapped.copy()
            global_best_score = sc
            print(f"[swap] improved to {best_score:.6f}")

        # tiny random perturbation to escape plateaus
        if np.random.rand() < 0.001:
            i2 = np.random.randint(n)
            delta = (np.random.rand() * 2.0 - 1.0) * 0.02 * target_sum
            if a[i2] - delta > 0:
                a2 = a.copy()
                a2[i2] -= delta
                a2 = _project_to_simplex(a2, target_sum)
                sc2 = _max_convolution(a2, size_fft_conv)
                if sc2 < best_score:
                    best_score = sc2
                    a = a2
                    global_best_a = a2.copy()
                    global_best_score = sc2
                    print(f"[perturb] improved to {best_score:.6f}")

        # early exit if only a few milliseconds left
        if time.time() + 0.01 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
