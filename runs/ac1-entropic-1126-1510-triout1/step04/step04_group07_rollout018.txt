"""
Improved step‑function search.

The evaluation score equals the maximal coefficient of the a*a convolution
when the ℓ¹‑norm of a is √(2·n).  The algorithm proceeds as follows:

1.  Choose a problem size n (the length of the reference sequence if it
    exists, otherwise a default of 3500).  The target ℓ¹‑norm is
    S = √(2·n).

2.  Build a small collection of initial candidates:
    – analytic “Catalan” shape (coefficients of (1‑x)⁻¹ᐟ²),
    – uniform vector,
    – decreasing exponential,
    – random positive vector.
    All are renormalised to have ℓ¹‑norm S.

3.  For each start run a cheap sub‑gradient descent on the true hard‑max
    objective  M(a)=max_k (a*a)[k].
    The sub‑gradient for an active shift k is
        g_i = 2·a_{k‑i}   (if the index is valid, else 0).
    A simple projected gradient step
        a ← a – η·g ,  clip≥0, rescale to sum S
    is performed many times (linear‑time per iteration).  Random
    mass‑transfer perturbations are injected periodically to escape
    shallow minima.

4.  After the sub‑gradient phase the best vector found is polished with
    the smooth‑max / L‑BFGS‑B continuation used in the baseline
    (soft‑max parametrisation on the simplex) – but with a slower cooling
    schedule and more iterations per temperature.

5.  A final cheap random‑block search is executed on the incumbent.

All steps respect the global time budget (≈ 1000 s) and the best feasible
sequence discovered is returned.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip negatives to zero and renormalise to have ℓ¹‑norm = target."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, target / v.shape[0])
    return v * (target / s)


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real‑valued)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> tuple[np.ndarray, float]:
    """Return (full convolution, maximal coefficient)."""
    conv = _fft_convolve(a, a)
    return conv, float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_hard_max(a: np.ndarray, conv: np.ndarray, max_c: float) -> np.ndarray:
    """
    Sub‑gradient of M(a)=max(conv) (average over all active shifts).
    """
    n = a.shape[0]
    active = np.flatnonzero(conv == max_c)
    # average over active shifts
    w = np.zeros_like(conv, dtype=float)
    w[active] = 1.0 / active.size
    # compute g = 2 * (w * rev(a))_{n-1 : n-1+n}
    rev = a[::-1]
    tmp = np.convolve(w, rev)  # length = 3n‑2
    g = 2.0 * tmp[n - 1 : n - 1 + n]
    return g


# ----------------------------------------------------------------------
def _smooth_max_objective(y: np.ndarray, mu: float, n: int, S: float):
    """Value and gradient of the smooth‑max surrogate (log‑sum‑exp)."""
    # probability on simplex
    p = _softmax(y)
    a = S * p
    conv = np.convolve(a, a)          # dense for simplicity (n≈3‑4k)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        fval = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        fval = max_c
    # gradient w.r.t. a
    rev = a[::-1]
    tmp = np.convolve(w, rev)
    g_a = 2.0 * tmp[n - 1 : n - 1 + n]
    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(fval), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluate_sequence() score.  The routine respects the given time budget
    (seconds) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3500
        have_ref = False

    S = math.sqrt(2.0 * n)               # target ℓ¹‑norm

    # bookkeeping
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # --------------------------------------------------------------
    def _maybe_update(a_vec: np.ndarray) -> None:
        nonlocal best_score, best_a
        # use the official evaluator – it also applies the security caps
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # ------------------------------------------------------------------
    # 1. Generate initial candidates
    # ------------------------------------------------------------------
    init_candidates = []

    # (a) Catalan / (1‑x)^‑1/2 coefficients  (analytic solution of constant conv)
    cat = np.empty(n, dtype=float)
    cat[0] = 1.0
    for i in range(1, n):
        # c_{i} = c_{i‑1} * ((2i)*(2i‑1)) / (4*i*i)
        cat[i] = cat[i - 1] * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
    cat = _project_to_sum(cat, S)
    init_candidates.append(cat)

    # (b) uniform
    uniform = np.full(n, S / n, dtype=float)
    init_candidates.append(uniform)

    # (c) decreasing exponential
    alpha = rng.uniform(0.001, 0.008) * n / 1000.0
    exp_desc = np.exp(-alpha * np.arange(n, dtype=float))
    exp_desc = _project_to_sum(exp_desc, S)
    init_candidates.append(exp_desc)

    # (d) random positive vector
    rand_vec = rng.random(n, dtype=float)
    rand_vec = _project_to_sum(rand_vec, S)
    init_candidates.append(rand_vec)

    # (e) reference if available
    if have_ref:
        ref = np.asarray(height_sequence_1, dtype=float)
        ref = _project_to_sum(ref, S)
        init_candidates.append(ref)

    # --------------------------------------------------------------
    # 2. Sub‑gradient descent on the hard max for each start
    # --------------------------------------------------------------
    for rid, a_start in enumerate(init_candidates):
        if time.time() > deadline:
            break
        print(f"[restart {rid+1}/{len(init_candidates)}] sub‑gradient phase")
        a = a_start.copy()
        it = 0
        eta0 = 0.5 * S / n          # initial step size (empirical)
        decay = 5e-5
        while time.time() < deadline and it < 5000:
            # compute convolution and its maximum
            conv, max_c = _max_convolution(a)

            # record improvement
            _maybe_update(a)

            # sub‑gradient (average over all active shifts)
            g = _subgradient_hard_max(a, conv, max_c)

            # step‑size schedule
            eta = eta0 / (1.0 + decay * it)
            a = a - eta * g
            a = _project_to_sum(a, S)

            # occasional random mass transfer to improve exploration
            if it % 250 == 0 and it > 0:
                i, j = rng.choice(n, 2, replace=False)
                delta = min(a[i], a[i] * 0.05)   # move up to 5 % of a[i]
                a[i] -= delta
                a[j] += delta
                a = _project_to_sum(a, S)

            it += 1

        # after this start keep the best discovered so far
        _maybe_update(a)

    # --------------------------------------------------------------
    # 3. Polishing with smooth‑max/L‑BFGS‑B continuation (similar to baseline)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[polish] L‑BFGS‑B continuation")
        # initialise from the best a we have
        y = np.log(best_a + 1e-12)                 # free parameters
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = max(conv_cur.max(), 1.0)               # start temperature

        mu_factor = 0.8
        mu_min = 1e-8
        maxiter_per_mu = 300

        while mu > mu_min and time.time() < deadline:
            # closure required by scipy – we use a thin wrapper
            def obj(z):
                return _smooth_max_objective(z, mu, n, S)

            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # --------------------------------------------------------------
    # 4. Final cheap random‑block search (many trials, low cost)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        if remaining > 2.0:
            max_trials = int(min(8000, remaining * 180))
            print(f"[block] {max_trials} random block trials")
            for _ in range(max_trials):
                nb = rng.integers(2, 9)                 # 2‑8 blocks
                cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
                blocks = np.split(np.arange(n), cuts)
                vals = rng.random(nb)
                cand = np.empty(n, dtype=float)
                for blk, v in zip(blocks, vals):
                    cand[blk] = v
                cand = _project_to_sum(cand, S)
                _maybe_update(cand)

    # --------------------------------------------------------------
    # 5. Return the best sequence discovered
    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
