"""
Propose a non‑negative step function (list of heights) that minimises the
evaluation score.

Strategy
--------
1. Choose a problem size `n` (use the length of the provided
   `height_sequence_1` if it exists, otherwise fall back to 3000).  The
   ℓ₁‑norm is fixed to `S = sqrt(2·n)` – scaling does not affect the final
   score, but the simplex projection makes the optimisation well‑behaved.

2. **Broad sampling** – generate a large pool of raw candidates:
      * Dirichlet draws with many concentration parameters (very sparse to
        almost uniform)
      * Power‑law and exponential decreasing shapes
      * Random block‑wise constant vectors
      * Random permutations of the best known sequence (if any)
   Every candidate is projected onto the simplex and its exact peak of the
   self‑convolution is evaluated.  The `K` best raw candidates (by actual
   max‑convolution) are retained.

3. **Smooth‑max gradient descent** – for each retained candidate we run an
   Adam optimiser on the *smooth* approximation of the max‑convolution

        L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

   The gradient is `∇L = 2·w ∗ a_rev` where `w` are the softmax weights.
   The parameter `α` starts small (≈0.1) and is multiplied by 1.05 every
   2 000 iterations, ending at 10.  After each update the vector is projected
   back onto the simplex.  The best solution seen during the run is kept.

4. **Local permutation & mass‑transfer hill‑climbing** – with the remaining
   time we improve the current global best by
      * swapping two positions,
      * moving a tiny amount of mass from one index to another,
   accepting a change only if it lowers the true peak of the convolution.
   Both operations are cheap (a single FFT‑based convolution per trial).

5. Return the best sequence discovered before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for FFT efficiency)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray, size_fft: int) -> np.ndarray:
    """Full convolution of x and y using FFT (length = len(x)+len(y)-1)."""
    fx = np.fft.rfft(x, n=size_fft)
    fy = np.fft.rfft(y, n=size_fft)
    conv = np.fft.irfft(fx * fy, n=size_fft)
    return conv[: x.size + y.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution a * a."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _smooth_max_and_weights(conv: np.ndarray, alpha: float):
    """
    Compute the smooth max L_α and the associated soft‑max weights w.
    Returns (L, w) where w sums to 1 and has the same shape as conv.
    """
    max_c = conv.max()
    shifted = conv - max_c                     # for numerical stability
    exp_vals = np.exp(alpha * shifted)
    sum_exp = exp_vals.sum()
    L = max_c + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    return L, w


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, w: np.ndarray,
                      size_fft_cross: int) -> np.ndarray:
    """
    Gradient of the smooth max w.r.t. a.
    g = 2 * (w * a_rev) where * denotes linear convolution.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft_cross)
    start = a.size - 1
    g = 2.0 * conv[start:start + a.size]
    return g


# ----------------------------------------------------------------------
def _refine_softmax(a0: np.ndarray,
                    size_fft_self: int,
                    size_fft_cross: int,
                    target_sum: float,
                    deadline: float,
                    max_iters: int = 30_000,
                    alpha_start: float = 0.1,
                    alpha_factor: float = 1.05,
                    alpha_step: int = 2_000,
                    alpha_max: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smoothed max‑convolution.
    Returns the best sequence (lowest true max‑convolution) and its max value.
    """
    a = a0.copy()
    n = a.size

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning‑rate (order of magnitude of one entry)
    eta0 = target_sum * 0.5 / n

    best_a = a.copy()
    best_max = _max_convolution(a, size_fft_self)

    alpha = alpha_start

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # 1) self‑convolution
        conv = _full_conv(a, size_fft_self)
        cur_max = float(conv.max())

        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[refine] it {it:,} new best max = {best_max:.6f}")

        # 2) smooth objective & weights
        L, w = _smooth_max_and_weights(conv, alpha)

        # 3) gradient of the smooth objective
        g = _gradient_softmax(a, w, size_fft_cross)

        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)          # 1/√t decay
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # 5) keep the ℓ₁‑norm fixed and stay non‑negative
        a = _project_to_simplex(a, target_sum)

        # 6) slowly sharpen the soft‑max
        if it % alpha_step == 0:
            alpha = min(alpha * alpha_factor, alpha_max)

    return best_a, best_max

            a2[j] += delta
            # projection is not needed – the ℓ₁‑norm is preserved and no
            # negativity was introduced
            cur_max = true_max(a2)
            if cur_max < best_max:
                best_max = cur_max
                a = a2
                global_best_a = a.copy()
                global_best_max = cur_max
                print(f"[transfer] improved to {best_max:.6f}")

        # tiny random perturbation every now and then to escape plateaus
        if np.random.rand() < 0.002:
            i = np.random.randint(n)
            delta = (np.random.rand() * 2 - 1) * 0.02 * target_sum
            if a[i] - delta > 0:
                a2 = a.copy()
                a2[i] -= delta
                a2 = _project_to_simplex(a2, target_sum)
                cur_max = true_max(a2)
                if cur_max < best_max:
                    best_max = cur_max
                    a = a2
                    global_best_a = a2.copy()
                    global_best_max = cur_max
                    print(f"[perturb] improved to {best_max:.6f}")

        # safety break if we are within a few ms of the deadline
        if time.time() + 0.01 > deadline:
            break

    # --------------------------------------------------------------
    # Final reporting
    final_score = 2.0 * n * best_max / (target_sum ** 2)
    print(f"[final] best max‑conv = {best_max:.6f}, score = {final_score:.6f}")
    return global_best_a.tolist()
