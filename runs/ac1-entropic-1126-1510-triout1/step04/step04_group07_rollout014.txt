def _compute_f_and_grad_monotone(u, mu, n, S):
    # compute y = -cumsum(exp(u))  # shape (n,)
    exp_u = np.exp(np.clip(u, -50, 50))
    y = -np.cumsum(exp_u)
    # compute p = softmax(y)
    y_max = np.max(y)
    e = np.exp(y - y_max)
    p = e / e.sum()
    a = S * p

    conv = np.convolve(a, a)
    max_c = conv.max()
    if mu > 0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700, 700)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n-1:n-1+n]

    # gradient w.r.t y:
    inner = np.dot(g_a, p)  # scalar
    grad_y = S * p * (g_a - inner)

    # gradient w.r.t u: chain rule
    # y_i = -sum_{j=0}^{i} exp(u_j)
    # dy_i/du_j = -exp(u_j) if j <= i else 0

    suffix_sum = np.cumsum(grad_y[::-1])[::-1]   # sum from i=j to end
    grad_u = -exp_u * suffix_sum

    return float(f_val), grad_u.astype(np.float64)

"""
monotone‑L‑BFGS optimiser for the step‑function attack.

We directly optimise a *monotonically decreasing* non‑negative sequence.
A vector `u` ∈ ℝⁿ parametrises a monotone distribution via

    z_i = exp(u_i)               (z_i > 0)
    y_i = -cum_sum(z)            (y_0 ≥ y_1 ≥ … ≥ y_{n‑1})
    p   = softmax(y)             (Σ p_i = 1,  p_i ≥ 0)
    a   = S·p                     (Σ a_i = S = √(2·n))

The objective is    f(a) = max_k (a * a)_k  (the smooth‑max surrogate is
used for differentiability).  After a standard continuation (decreasing
temperature μ) we polish the result by a pure sub‑gradient descent and a
final random‑block search.  The whole routine respects the 1000 s budget
and returns the best feasible list for `evaluate_sequence`.

No closures or lambdas are used – all helpers are top‑level functions.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _a_from_u(u: np.ndarray, S: float) -> np.ndarray:
    """
    Build the step‑function heights from the optimisation variable `u`.

    Parameters
    ----------
    u : np.ndarray
        Unconstrained parameters (length n).
    S : float
        Desired ℓ¹‑norm ( √(2·n) ).

    Returns
    -------
    a : np.ndarray
        Non‑negative vector with Σ a = S.
    """
    # keep exponentials bounded to avoid overflow
    exp_u = np.exp(np.clip(u, -50.0, 50.0))
    y = -np.cumsum(exp_u)                      # decreasing
    p = _softmax(y)                            # Σ p = 1
    return S * p


# ----------------------------------------------------------------------
def _smooth_max_and_grad(u: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max surrogate (value and gradient) for the monotone parametrisation.

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_u : np.ndarray
        Gradient w.r.t. `u`.
    """
    # ---- build a from u -------------------------------------------------
    exp_u = np.exp(np.clip(u, -50.0, 50.0))
    y = -np.cumsum(exp_u)          # decreasing
    y_max = np.max(y)
    e = np.exp(y - y_max)
    p = e / e.sum()
    a = S * p

    # ---- convolution ----------------------------------------------------
    conv = np.convolve(a, a)       # length 2·n‑1
    max_c = conv.max()

    # ---- smooth‑max weights ---------------------------------------------
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # guard overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                               # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # ---- gradient w.r.t. a ------------------------------------------------
    # g_a[i] = 2 * Σ_k w_k * a_{k‑i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # ---- gradient w.r.t. y (soft‑max Jacobian) -------------------------
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)            # ∂f/∂y

    # ---- chain rule  y_i = -Σ_{j≤i} exp(u_j) ---------------------------
    # dy_i/du_j = -exp(u_j)  if j ≤ i else 0
    suffix_sum = np.cumsum(grad_y[::-1])[::-1]   # Σ_{i=j}^{n-1} grad_y[i]
    grad_u = -exp_u * suffix_sum

    return float(f_val), grad_u.astype(np.float64)


# ----------------------------------------------------------------------
class _MonotoneSmoothObj:
    """Callable required by `scipy.optimize.minimize` (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, u: np.ndarray):
        return _smooth_max_and_grad(u, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    `evaluate_sequence` score.  A monotone L‑BFGS continuation,
    followed by a sub‑gradient polish and a random‑block refinement,
    is performed within the allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size – reuse the reference length if it exists
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 3000                     # a comfortable size
    S = math.sqrt(2.0 * n)          # target ℓ¹‑norm (makes score = max convolution)

    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # --------------------------------------------------------------
    def _maybe_update(a_vec: np.ndarray) -> None:
        """Evaluate `a_vec` and keep it if it improves the incumbent."""
        nonlocal best_score, best_a
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # generate a few diverse starting points for `u`
    init_us = []

    # random Gaussian vectors
    for _ in range(4):
        init_us.append(rng.normal(loc=0.0, scale=0.7, size=n))

    # a very flat start (all u ≈ 0 → y ≈ -cumulative 1 → exponential decay)
    init_us.append(np.full(n, -0.5))

    # if a reference sequence is available, derive a monotone start from it
    try:
        a_ref = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, S)
        # we simply take the log‑softmax of the reference as a heuristic start
        eps = 1e-12
        y_ref = np.log(a_ref + eps)
        # convert that y into a feasible u (invert the cum‑sum construction)
        # we recover z = -diff(y); enforce positivity with a log‑transform
        dy = -np.diff(np.concatenate(([y_ref[0]], y_ref)))
        dy = np.maximum(dy, 1e-12)          # avoid zeros
        u_ref = np.log(dy)
        # pad to length n (the first element of u corresponds to -y0)
        u_ref = np.concatenate(([ -y_ref[0] ], u_ref))
        init_us.append(u_ref)
    except Exception:
        pass

    # --------------------------------------------------------------
    # continuation parameters (smooth‑max)
    mu_factor = 0.6          # slower cooling → finer optimisation
    mu_min = 1e-9
    maxiter_per_mu = 400
    n_restart = len(init_us)

    for rid, u0 in enumerate(init_us):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] initialising")

        u = u0.copy()
        a_cur = _a_from_u(u, S)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        mu = max(mu, 1.0)

        while mu > mu_min and time.time() < deadline:
            obj = _MonotoneSmoothObj(mu, n, S)

            try:
                res = opt.minimize(
                    obj,
                    u,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failure: {exc}")
                break

            u = res.x
            a = _a_from_u(u, S)
            _maybe_update(a)

            mu *= mu_factor

    # --------------------------------------------------------------
    # sub‑gradient descent on the true hard‑max (μ = 0)
    if time.time() < deadline:
        a = best_a.copy()
        step0 = S * 0.6 / n
        decay = 6e-5
        it = 0
        while time.time() < deadline and it < 8000:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            conv_wr = np.convolve(w, a[::-1])
            g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

            eta = step0 / (1.0 + decay * it)
            a_new = a - eta * g_a
            a_new = _project_to_simplex(a_new, S)

            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"[subgrad] improved to {best_score:.6f}")

            a = a_new
            it += 1

    # --------------------------------------------------------------
    # random‑block search (many cheap trials)
    remaining = deadline - time.time()
    if remaining > 3.0:
        max_trials = int(min(8000, remaining * 200))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(2, 15)                  # number of blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb) * S * 0.5            # keep values moderate
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # --------------------------------------------------------------
    # final monotone‑sorting heuristic (descending order is usually best)
    if time.time() < deadline:
        desc = np.sort(best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(desc)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
