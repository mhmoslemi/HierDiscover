"""
Propose a non‑negative step‑function (a list of heights) with a very low
self‑convolution peak.

The algorithm consists of three phases:

1. **Broad sampling** – generate many candidates from several families
   (Dirichlet, power‑law, exponential, block‑wise constant, binary).  All
   candidates are projected onto the simplex {x≥0, Σx = √(2 n)} and the best
   `K` (by max‑convolution) are kept.

2. **Sub‑gradient refinement** – each of the top candidates is improved with an
   Adam‑style projected sub‑gradient descent on the exact L∞‑norm of the
   self‑convolution.

3. **Permutation annealing** – the values of the current best candidate are
   rearranged.  A simulated‑annealing hill‑climber repeatedly swaps two
   positions; swaps that lower the peak are always accepted, otherwise they are
   accepted with a temperature‑dependent probability.  The best ordering found
   in the remaining time is returned.

All steps respect the 1000 s budget and the required non‑negativity /
normalisation constraints.
"""
import math
import time

import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Smallest power‑of‑2 ≥ required (efficient FFT size)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Full linear convolution a * a using FFT (length = 2 n‑1).
    Returned array has length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution (no clipping)."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)_k   at the current maximal index k_max.
    For non‑negative a this is the exact sub‑gradient of the L∞‑norm.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        # indices 0 … k_max receive the reversed tail
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        # k_max ≥ n
        start = k_max - n + 1                     # first index that participates
        length = 2 * n - 1 - k_max                # how many terms are used
        g[start:] = 2.0 * a[:length][::-1]

    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the exact L∞‑norm
    of the self‑convolution.
    Returns the best sequence seen and its peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base step size – roughly the size of a typical entry
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            # occasional progress report
            if it % 5_000 == 0:
                print(f"[refine] it {it:,} new best = {best_score:.6f}")

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning rate decay ~ 1/√t
        eta = eta0 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex (keeps Σa = target_sum and a ≥ 0)
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def _permutation_annealing(
    init_arr: np.ndarray, size_fft: int, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climber on the permutation of the values.
    Swaps two positions at a time; uphill moves are accepted with a
    temperature that decays exponentially.
    Returns the best arrangement seen and its peak value.
    """
    n = init_arr.shape[0]
    cur_arr = init_arr.copy()
    cur_max = _max_convolution(cur_arr, size_fft)

    best_arr = cur_arr.copy()
    best_max = cur_max

    start_time = time.time()
    total_budget = max(deadline - start_time, 1e-6)

    # temperature schedule
    Tmax = cur_max * 0.2 + 1e-6
    Tmin = cur_max * 1e-5 + 1e-12

    while time.time() < deadline:
        elapsed = time.time() - start_time
        fraction = min(elapsed / total_budget, 1.0)
        # exponential decay
        T = Tmax * ((Tmin / Tmax) ** fraction)

        # propose a swap
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # perform swap
        cur_arr[i], cur_arr[j] = cur_arr[j], cur_arr[i]

        new_max = _max_convolution(cur_arr, size_fft)

        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # uphill acceptance probability
            if T > 0:
                accept = np.random.rand() < math.exp(-(new_max - cur_max) / T)

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_arr = cur_arr.copy()
                # early stop if we have already beaten the target
                if best_max <= 1.503:
                    break
        else:
            # revert swap
            cur_arr[i], cur_arr[j] = cur_arr[j], cur_arr[i]

        # small safeguard to avoid an extremely long last loop
        if time.time() + 0.01 > deadline:
            break

    return best_arr, best_max

        k = np.random.randint(1, n // 2)
        a = np.zeros(n)
        a[np.random.choice(n, k, replace=False)] = 1.0
        a = _project_to_simplex(a, target_sum)
        push_candidate(_max_convolution(a, size_fft), a)

    print(f"[sampling] best raw score = {best_raw[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2. refinement ---------------------------------------------------
    # allocate roughly 40 % of the remaining budget for refinement
    remaining = deadline - time.time()
    refine_deadline = time.time() + remaining * 0.40

    max_refine_candidates = min(10, len(best_raw))
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    for rank in range(max_refine_candidates):
        if time.time() >= refine_deadline:
            break
        raw_sc, a0 = best_raw[rank]
        # compute a safe iteration budget from the time left for this candidate
        time_left = refine_deadline - time.time()
        max_iters = int(min(120_000, max(1_000, time_left * 1500)))
        print(f"[refine] candidate #{rank+1} (raw score {raw_sc:.6f}, max_iters {max_iters})")
        refined_a, refined_sc = _refine_subgrad(
            a0, size_fft, target_sum, refine_deadline, max_iters
        )
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # --------------------------------------------------------------
    # 3. permutation annealing ----------------------------------------
    print("[annealing] start")
    # use the values of the current best candidate
    values = global_best_a.copy()

    # create a “zig‑zag” start: highest values at the two ends and moving inward
    order = np.empty(n, dtype=int)
    left, right, pos = 0, n - 1, 0
    while left <= right:
        order[pos] = left
        pos += 1
        if left != right:
            order[pos] = right
            pos += 1
        left += 1
        right -= 1
    sorted_vals = np.sort(values)[::-1]            # descending
    init_arr = np.empty_like(values)
    init_arr[order[: len(sorted_vals)]] = sorted_vals

    # run annealing in the remaining time
    best_arr, best_max = _permutation_annealing(init_arr, size_fft, deadline)

    # final projection (numerical drift may have occurred)
    best_arr = _project_to_simplex(best_arr, target_sum)

    print(f"[final] best peak = {best_max:.6f}")
    return best_arr.tolist()
