    a = a0.copy()
    best_seq = a.copy()
    best_score = 2 * n * np.max(_fft_self_convolve(a))
    for iter in range(1, max_iter):
        if time.time() >= deadline:
            break
        b = _fft_self_convolve(a)
        max_idx = int(np.argmax(b))
        cur_score = 2 * n * b[max_idx]
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # compute subgradient
        grad = _subgradient_max(a, max_idx)
        # step size: maybe 1/sqrt(iter)
        eta = 0.7 / np.sqrt(iter)
        # gradient descent
        a = a - eta * grad
        # clip negatives to zero
        a = np.maximum(a, 0.0)
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
    return best_seq, best_score

"""
Propose a step‑function (non‑negative height sequence) that improves the
known best bound 1.512… .

The algorithm works as follows:

1. Choose a length `n` (based on the previous solution if any, otherwise a
   safe value ~8000, never exceeding 20000).

2. Generate a family of *monotone* candidate distributions
   a_i ∝ (i+½)^p   (p = 0.8, 1.0, 1.5, 2.0).  After normalisation the vector
   lies on the probability simplex (sum = 1).

3. For each candidate run a projected sub‑gradient descent on the convex
   objective   F(a)=2·n·max_k (a∗a)[k] .
   The sub‑gradient of the active shift `k* = argmax (a∗a)` is
   g_i = 2·a_{k*‑i}   for all i that contribute to that shift.
   After each step we project back onto the simplex.

   The step size is η_t = η0 / sqrt(t) (η0≈0.8) – a standard diminishing
   schedule for sub‑gradient methods on convex problems.

4. Keep the best sequence seen (lowest internal score 2·n·max_b).

5. After the main passes use a cheap random‑move hill‑climb that moves a
   small amount of mass from one index to another; accept only improving
   moves.  This often removes tiny plateaus.

6. Return the best sequence as a plain Python list.  All numbers are
   non‑negative and the routine respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _make_monotone_distribution(n: int, power: float) -> np.ndarray:
    """
    Return a monotone increasing probability vector:
        a_i ∝ (i + 0.5) ^ power ,   i = 0 … n‑1
    The vector is projected onto the simplex so the sum equals 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    a = np.power(i, power)
    a = a / a.sum()
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the objective
        F(a) = 2·n·max_k (a∗a)[k] .
    Returns the best sequence seen and its internal score (2·n·max_b).
    """
    a = a0.copy()
    best_seq = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))
        cur_score = 2.0 * n * conv[k]
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] subgrad it={it} score={best_score:.6f}")

        # sub‑gradient of the active shift
        g = _subgradient_max(a, k)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step and projection onto simplex
        a = a - eta * g
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _random_local_refinement(a_start: np.ndarray,
                             n: int,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑move hill‑climbing: move a small fraction of mass
    from one index to another and keep the improvement.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        eps = a[i] * np.random.rand() * 0.05
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        cur_score = 2.0 * n * np.max(_fft_self_convolve(cand))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score
