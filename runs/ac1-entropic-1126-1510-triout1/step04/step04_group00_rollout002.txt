"""
Propose a low‑score step function.

The continuous arcsine distribution  f(x)=1/(π·√(x(1‑x)))  (Beta(½,½))
has the remarkable property that its self‑convolution is the uniform
distribution on [0,2].  After a fine discretisation the discrete
probability vector a_i ≈ f(i/n)/n yields a self‑convolution a∗a whose
entries are almost constant; consequently

    score = 2·n·max(a∗a) / (∑a)²  ≈ 2·n·(½/n) = 1.

The algorithm therefore:
1. builds discrete arcsine vectors for a handful of lengths,
2. evaluates a cheap FFT‑based estimate of the score,
3. picks the length with the smallest estimate,
4. returns the corresponding normalised arcsine vector.

All numbers are non‑negative and sum to one, so the evaluator’s clipping
does not change the sequence.  The whole search finishes well within the
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) of length n, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    # raw (unnormalised) weights ∝ 1/√((i+½)(n‑i‑½))
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()                      # normalise to a probability vector
    # final safeguard – exact simplex projection removes possible round‑off drift
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    # next power‑of‑two size for efficient FFT
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative height sequence that minimises the evaluation score.

    The function builds discrete arcsine vectors for several lengths,
    estimates the score with an FFT‑based convolution, picks the best
    length and returns that vector.
    """
    np.random.seed(seed)            # seed is kept for API compatibility
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a few candidate lengths.  If a previous construction is
    # available we scale from it, otherwise we start around 8000.
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Candidate multipliers – keep the resulting n modest (≤ 18000) so that
    # the evaluator’s O(n²) convolution stays fast.
    factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    cand_ns = sorted(
        set(
            max(500, int(base_n * f))
            for f in factors
            if max(500, int(base_n * f)) <= 18000
        )
    )
    if not cand_ns:
        cand_ns = [8000]

    best_n = cand_ns[0]
    best_score_est = float("inf")
    # ------------------------------------------------------------------
    for n in cand_ns:
        if time.time() >= deadline:
            break
        a = _arcsine_distribution(n)
        b = _fft_self_convolve(a)
        est_score = 2.0 * n * b.max()          # sum(a) = 1, so denominator = 1
        print(f"[{time.time():.1f}] n={n:<5d} est_score≈{est_score:.6f}")
        if est_score < best_score_est:
            best_score_est = est_score
            best_n = n

    # ------------------------------------------------------------------
    # Build the final sequence for the selected length.
    final_seq = _arcsine_distribution(best_n).tolist()
    # Final safeguard – make sure the returned list is plain Python floats.
    final_seq = [float(x) for x in final_seq]

    print(
        f"Chosen length = {best_n}, estimated score = {best_score_est:.6f}"
    )
    return final_seq
