    v = np.zeros_like(y)
    beta1=0.9

        m_hat = m / (1 - beta1**t)

"""
Propose a low‑score non‑negative step‑function.

The algorithm consists of three stages, all confined to the time budget:

1. **Fast simulated‑annealing (SA)** – cheap random mass moves on the
   simplex.  This quickly discovers a good region of the search space.

2. **Adam optimisation of a smooth‑max surrogate** – we minimise a
   differentiable approximation of the true objective
   ``max_k (a*a)[k]`` using the Adam optimiser.  The surrogate temperature
   *mu* is geometrically cooled (0.7‑factor) and the optimiser is warm‑started
   from the SA result.  The soft‑max re‑parametrisation
   ``a = S·softmax(y)`` guarantees non‑negativity and the ℓ¹‑norm constraint
   (∑a = S = √(2·n)) automatically.

3. **Projected sub‑gradient polishing** – the exact hard‑max objective is
   tackled with a simple sub‑gradient descent and Euclidean projection onto
   the simplex.  The step‑size decays slowly, allowing a fine‑grained final
   improvement.

All helper routines are top‑level, there are no closures, and the code
conforms to the 1000 s wall‑clock limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector (sum = 1)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for FFT‑based convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT (fast)."""
    return float(_full_convolution(a, size).max())


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (soft‑max surrogate) and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Logit vector (real numbers) such that a = S·softmax(y) satisfies the
        simplex constraint.
    mu : float
        Temperature of the soft‑max; mu=0 yields the exact hard max.
    n : int
        Length of the sequence.
    S : float
        Desired ℓ¹‑norm (normally sqrt(2·n)).

    Returns
    -------
    f_val : float
        Smoothed maximum of the convolution.
    grad_y : np.ndarray
        Gradient of f_val w.r.t. y.
    """
    p = _softmax(y)               # Σp = 1, p_i ≥ 0
    a = S * p                     # Σa = S

    # exact convolution (dense, because we need it for the gradient anyway)
    conv = np.convolve(a, a)      # length 2·n‑1
    max_c = conv.max()

    # soft‑max (log‑sum‑exp) approximation of max if mu>0
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # exact max – distribute weight equally over all maximal entries
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (standard formula for a*a)
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule through the soft‑max parametrisation
    inner = np.dot(g_a, p)                     # scalar = Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Very cheap SA on the simplex – repeatedly move a random amount of mass
    from one index to another.  The score is the true maximum convolution.
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    # temperature schedule – start moderately high, cool to near zero
    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-6
    it = 0

    while time.time() - start_time < budget:
        # linear cooling
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of a[i] (up to 30 %)
        delta = np.random.rand() * a[i] * 0.3
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = diff < 0.0
        if not accept and T > 0.0 and np.random.rand() < math.exp(-diff / T):
            accept = True

        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score

def _adam_optimize(y_start: np.ndarray, S: float, n: int,
                   size: int, budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.  The temperature ``mu`` is
    reduced geometrically (factor 0.7) and the inner loop runs a few hundred
    Adam steps per mu.  The best hard‑max value found is returned.
    """
    # initialise
    y = y_start.copy()
    best_a = S * _softmax(y)
    best_score = _max_convolution(best_a, size)

    # start temperature – use the current hard max as a scale
    mu = max(best_score, 1.0)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0                         # global step counter

    # outer loop over mu
    mu_min = 1e-6
    while mu > mu_min and time.time() - start_time < budget:
        # inner Adam steps for this temperature
        inner_iters = 300
        for _ in range(inner_iters):
            if time.time() - start_time >= budget:
                break

            f, g = _compute_f_and_grad(y, mu, n, S)
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            # simple decay of the learning rate
            lr = 0.5 / (np.sqrt(t) + 1.0)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            # evaluate the hard max (no smoothing)
            a = S * _softmax(y)
            cur = _max_convolution(a, size)
            if cur < best_score:
                best_score = cur
                best_a = a.copy()
                print(f"[Adam] mu {mu:.6f} step {t:,} score = {best_score:.6f}")

        # reduce mu and (optionally) inject a tiny random jitter to escape shallow minima
        mu *= 0.7
        y += np.random.normal(scale=0.02, size=n)   # jitter keeps exploration alive

    return best_a, best_score

def _final_subgradient(a_start: np.ndarray, S: float,
                       size: int, budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Hard‑max sub‑gradient descent with Euclidean projection onto the simplex.
    """
    n = len(a_start)
    a = a_start.copy()
    best_a = a.copy()
    best_score = _max_convolution(best_a, size)

    # step‑size parameters – heuristic values that work well for this problem size
    eta0 = S * 0.5 / n
    decay = 1e-5
    it = 0

    while time.time() - start_time < budget and it < 15000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = np.isclose(conv, max_c, atol=1e-12)
        cnt = mask.sum()
        w = np.zeros_like(conv, dtype=float)
        if cnt:
            w[mask] = 1.0 / cnt

        # sub‑gradient of the hard max (same formula as in the smooth case)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a_rev, n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        cur = _max_convolution(a_new, size)
        if cur < best_score:
            best_score = cur
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.
    Returns the best sequence (list of non‑negative floats) found before the
    deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 2500
        have_ref = False

    S = math.sqrt(2.0 * n)               # ℓ¹‑norm which makes the metric = max‑conv
    size = _fft_size(n)

    # --------------------------------------------------------------
    # initialise a
    if have_ref:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = _project_to_simplex(a0, S)
    else:
        a0 = np.full(n, S / n, dtype=float)   # uniform start

    best_a = a0.copy()
    best_score = _max_convolution(best_a, size)
    print(f"[init]   score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – quick simulated annealing
    remaining = deadline - time.time()
    if remaining > 30.0:
        sa_time = remaining * 0.25
        print(f"[SA] running for {sa_time:.1f}s")
        best_a, best_score = _simulated_annealing(best_a, S, sa_time, time.time())
        print(f"[SA]    best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on smooth‑max surrogate
    remaining = deadline - time.time()
    if remaining > 10.0:
        print(f"[Adam] optimizing for {remaining:.1f}s")
        # initialise logits y from the current best a
        p0 = best_a / S
        y0 = np.log(p0 + 1e-12)
        best_a, best_score = _adam_optimize(y0, S, n, size,
                                            remaining, time.time())
        print(f"[Adam]   best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – hard‑max sub‑gradient polish
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[final] polishing for {remaining:.1f}s")
        best_a, best_score = _final_subgradient(best_a, S, size,
                                                remaining, time.time())
        print(f"[final]  best = {best_score:.6f}")

    # safety fallback (should never be needed)
    if not np.isfinite(best_score):
        best_a = S * np.ones(n) / n
        best_score = _max_convolution(best_a, size)
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
