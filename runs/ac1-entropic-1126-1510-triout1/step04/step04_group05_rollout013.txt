def _egd_refine(a_start, S, budget, start_time):
    n = len(a_start)
    size = _fft_size(n)
    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    # Convert to probability:
    p = a / S
    # initial step size:
    eta0 = 0.1  # maybe small
    decay = 0.001  # adjust
    it = 0
    while time.time() - start_time < budget and it < 20000:
        # compute conv for grad
        conv = _full_convolution(a, size)
        max_c = conv.max()
        # compute gradient g_a
        g_a = _gradient_max_convolution(a, conv, size)  # shape n
        # compute gradient w.r.t p
        gp = S * g_a
        # compute logp update
        logp = np.log(p + 1e-12)
        # step size schedule
        eta = eta0 / (1.0 + decay * it)
        # update
        logp = logp - eta * gp
        # compute softmax to get new p
        max_log = np.max(logp)
        exp_logp = np.exp(logp - max_log)
        p = exp_logp / exp_logp.sum()
        a = S * p
        # evaluate slowly
        if it % 100 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[EGD improve] iter {it} score {best_score}")
        it += 1

    return best_a, best_score

def _l_bfgs_refine_long(a_start, S, budget, start_time):
   n = len(a_start)
   p0 = a_start / S
   eps = 1e-12
   y = np.log(p0 + eps).astype(np.float64)

   # compute initial mu as max conv of a_start
   conv0 = np.convolve(a_start, a_start)
   mu = float(conv0.max())
   if mu <= 0: mu = 1.0

   best_a = a_start.copy()
   best_score = evaluate_sequence(best_a.tolist())

   mu_factor = 0.8
   mu_min = 1e-7
   maxiter_per_mu = 500
   while mu > mu_min and time.time() - start_time < budget:
       obj = _SmoothMaxObjective(mu, n, S)

                              jac=True,

   return best_a, best_score

"""
Search for a low‑score non‑negative step function.

The algorithm consists of three stages:

1. **Random restarts** – many short L‑BFGS runs on the smooth‑max
   surrogate (softmax of the convolution).  Each restart starts from a
   different random simplex point (uniform, Dirichlet‑sparse or a
   few equal‑mass spikes).  The best solution of these explorations is
   kept.

2. **Focused L‑BFGS refinement** – the current best candidate is refined
   with a long continuation schedule: the smoothing parameter μ is
   decreased geometrically (μ←0.8 μ) down to μ≈10⁻⁷ while the L‑BFGS
   optimizer performs up to 200 inner iterations per μ.  This yields a
   very accurate approximation of the true maximum‑convolution objective.

3. **Polishing** – a projected sub‑gradient descent (hard max) finishes
   the remaining time budget, guaranteeing that the final sequence
   satisfies the simplex constraints exactly.

All helper functions are top‑level; no closures or lambdas are used.
The provided ``evaluate_sequence`` function is called only for the
true score (the maximum entry of the self‑convolution) and for keeping
track of the best solution found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0
    return conv


def _gradient_max_convolution(a: np.ndarray, conv: np.ndarray,
                             size: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    """
    max_c = conv.max()
    mask = np.isclose(conv, max_c, atol=1e-12)
    cnt = mask.sum()
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * len(a) - 1]
    g_a = 2.0 * conv_wr[len(a) - 1 : len(a) - 1 + len(a)]
    return g_a


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. y."""
    p = _softmax(y)               # Σp = 1, p_i ≥ 0
    a = S * p                     # Σa = S

    conv = np.convolve(a, a)      # dense convolution, length 2·n‑1
    max_c = conv.max()

    # smooth‑max via softmax over the convolution values
    diff = (conv - max_c) / mu
    diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
    exp_diff = np.exp(diff)
    Z = exp_diff.sum()
    w = exp_diff / Z
    f_val = max_c + mu * math.log(Z)

    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # chain rule to y (y -> p -> a)
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def _l_bfgs_refine_long(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation on the smooth‑max surrogate with a long schedule.
    Starts from a_start and reduces the smoothing parameter μ geometrically.
    """
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    # initialise μ from the current max‑convolution
    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    mu = max(mu, 1.0)

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())

    mu_factor = 0.8          # slower cooling than the baseline
    mu_min = 1e-7
    maxiter_per_mu = 200    # more interior steps per μ

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu,
                         "disp": False,
                         "gtol": 1e-12}
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS refine] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


# ----------------------------------------------------------------------
def _egd_refine(a_start: np.ndarray, S: float,
                budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the hard max (no smoothing).
    Works directly on the probability vector p = a / S.
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    p = a / S
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    eta0 = 0.5          # initial step size for log‑prob updates
    decay = 5e-4        # decay of the step size
    it = 0
    while time.time() - start_time < budget and it < 20000:
        conv = _full_convolution(a, size)
        g_a = _gradient_max_convolution(a, conv, size)

        # gradient w.r.t. the probability vector
        gp = S * g_a

        # mirror‑descent update in log‑space
        logp = np.log(p + 1e-12)
        eta = eta0 / (1.0 + decay * it)
        logp = logp - eta * gp

        # re‑normalise by soft‑max (ensures p ≥ 0, Σp = 1)
        max_log = np.max(logp)
        exp_log = np.exp(logp - max_log)
        p = exp_log / exp_log.sum()
        a = S * p

        if it % 200 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[EGD improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _random_init(n: int, S: float) -> np.ndarray:
    """
    Produce a random point on the simplex Σa = S, a ≥ 0.
    Three regimes are mixed:
      * uniform distribution,
      * Dirichlet with a small concentration (sparse),
      * a few equal‑mass spikes.
    """
    r = np.random.rand()
    if r < 0.3:
        # uniform
        return np.full(n, S / n, dtype=float)
    elif r < 0.8:
        # Dirichlet‑type sparse point
        alpha = np.random.uniform(0.01, 0.5)
        g = np.random.gamma(alpha, 1.0, size=n)
        p = g / g.sum()
        return S * p
    else:
        # few spikes with equal mass
        k = np.random.randint(2, min(200, max(2, n // 10)))
        idx = np.random.choice(n, size=k, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = S / k
        return a


# ----------------------------------------------------------------------
def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true hard max.
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # step‑size parameters
    eta0 = S * 0.5 / n
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 10000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # gradient of the hard max
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step‑function within the allotted time.
    Returns the best sequence found (list of non‑negative floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # -------------------- problem size --------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 2000
        have_ref = False

    S = math.sqrt(2.0 * n)                         # target ℓ¹‑norm

    # -------------------- initial sequence --------------------
    if have_ref:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = _project_to_simplex(a0, S)
    else:
        a0 = np.full(n, S / n, dtype=float)

    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init]   score = {best_score:.6f}")

    # -------------------- Phase 1: random restarts (short L‑BFGS) --------------------
    MAX_RESTARTS = 12
    for _ in range(MAX_RESTARTS):
        remaining = deadline - time.time()
        if remaining < 30.0:
            break
        # allocate a modest slice of the remaining budget to this restart
        restart_budget = min(25.0, remaining * 0.3)

        a_start = _random_init(n, S)
        a_opt, sc_opt = _l_bfgs_refine_long(a_start, S, restart_budget, time.time())
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[restart] new best = {best_score:.6f}")

    # -------------------- Phase 2: focused long L‑BFGS refinement --------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print(f"[L‑BFGS final] refining for {remaining:.1f}s")
        best_a, best_score = _l_bfgs_refine_long(best_a, S, remaining, time.time())
        print(f"[L‑BFGS final] best = {best_score:.6f}")

    # -------------------- Phase 3: exponentiated‑gradient polishing --------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[EGD] polishing for {remaining:.1f}s")
        a_opt, sc_opt = _egd_refine(best_a, S, remaining, time.time())
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[EGD] new best = {best_score:.6f}")

    # -------------------- Phase 4: final projected sub‑gradient polish --------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        print(f"[final subgrad] polishing for {remaining:.1f}s")
        a_opt, sc_opt = _final_subgradient(best_a, S, remaining, time.time())
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[final subgrad] new best = {best_score:.6f}")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
