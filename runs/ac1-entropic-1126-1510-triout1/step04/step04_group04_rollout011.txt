def _smooth_adam_opt(a0, size_self, size_corr, target_sum, deadline,
                     max_iters=100000, p_start=2.0, p_end=20.0):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_self)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9; beta2=0.999; eps = 1e-8
    # base step
    eta0 = target_sum * 0.5 / a.size
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # schedule p
        p = p_start + (p_end - p_start) * (it - 1) / max_iters
        # compute conv and gradient
        conv, grad = _grad_Lp(a, size_self, size_corr, p)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[smoothAdam] it {it:,} p={p:.2f} best={best_score:.6f}")
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        # projection
        a = _project_to_simplex(a, target_sum)
    return best_a, best_score

"""
Propose a non‑negative step function (vector) that minimises the self‑convolution peak.
The algorithm combines:

1. Diverse random initialisations (Dirichlet, power‑law, exponential, blockwise)
   projected onto the ℓ₁‑simplex Σa = √(2 n).

2. A continuation‑style Adam optimiser that minimises an Lᵖ‑norm of the
   self‑convolution (p grows from 2 up to ~20).  The gradient is obtained
   analytically via FFT based correlation, and each step is projected back
   onto the simplex.

3. A short Adam‑style sub‑gradient phase on the exact L∞‑objective for fine‑tuning.

4. A fast hill‑climbing permutation search (random swaps and tiny perturbations)
   to improve the ordering of the final values.

All heavy work is done with FFT (O(n log n) per iteration) and the search respects
the 1000 s time budget.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (good for FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Exact linear self‑convolution a * a (length = 2 n‑1)."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Peak value of the self‑convolution (no clipping)."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)_k   at the current maximal index k_max.
    This delivers the exact sub‑gradient of the L∞‑norm of the convolution.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        start = k_max - n + 1
        length = 2 * n - 1 - k_max
        g[start:] = 2.0 * a[:length][::-1]

    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent on the hard max‑convolution (L∞).
    Returns the best sequence and its peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[subgrad] it {it:,} best = {best_score:.6f}")

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _grad_Lp(
    a: np.ndarray,
    size_self: int,
    size_corr: int,
    p: float,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient of the Lᵖ‑norm of the self‑convolution.
    Returns (conv, grad) where conv is the full self‑convolution
    (length = 2 n‑1) and grad has length n.
    """
    # self‑convolution
    conv = _full_conv(a, size_self)
    conv = np.maximum(conv, 0.0)                     # numerical safety

    # denominator = (∑ conv_i^p)^{(p‑1)/p}
    conv_p = conv ** p
    sum_pow = conv_p.sum()
    denom = sum_pow ** ((p - 1.0) / p) if sum_pow > 0 else 1.0

    # weight = conv^{p‑1}
    weight = conv ** (p - 1.0)

    # correlation weight * a (valid part) via FFT
    a_rev = a[::-1]
    fw = np.fft.rfft(weight, n=size_corr)
    fa_rev = np.fft.rfft(a_rev, n=size_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=size_corr)

    n = a.size
    grad = 2.0 * corr_full[n - 1 : 2 * n - 1] / denom
    return conv, grad


# ----------------------------------------------------------------------
def _smooth_adam_opt(
    a0: np.ndarray,
    size_self: int,
    size_corr: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 150_000,
    p_start: float = 2.0,
    p_end: float = 20.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth Lᵖ‑approximation of the max‑convolution.
    Returns the best vector seen and its actual max‑convolution value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_self)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # linear schedule for the smoothing exponent
        p = p_start + (p_end - p_start) * (it - 1) / max_iters
        conv, grad = _grad_Lp(a, size_self, size_corr, p)

        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[smoothAdam] it {it:,} p={p:.2f} best={best_score:.6f}")

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # back to the simplex
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score
