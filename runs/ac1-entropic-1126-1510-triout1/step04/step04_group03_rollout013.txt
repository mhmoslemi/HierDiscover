"""
Propose a non‑negative step function that minimizes `evaluate_sequence`.

Algorithm
---------
* **Adam on the simplex** – a modern adaptive gradient method.
  * The gradient is obtained from a soft‑max (log‑sum‑exp) approximation of the
    max‑convolution, the same surrogate used in the previous solution.
  * After each Adam step the vector is Euclidean‑projected onto the probability
    simplex so that all heights stay non‑negative and sum to one.
* **Length schedule** – start from the best known length (if any) and
  progressively enlarge it (≈ ×1.2, ×1.5, ×2).  When moving to a larger `n`,
  the current best sequence is up‑sampled by simple repetition + renormalisation.
* **Multiple restarts** – for each length we try a few diverse initialisations
  (previous best, arcsine, uniform, random).
* **Intensive local search** – after the Adam phase we perform a cheap
  stochastic “mass‑transfer” hill‑climb that repeatedly moves a random
  fraction of probability mass from one index to another and keeps the move
  if the true `evaluate_sequence` score improves.
* **Time management** – the overall budget (≈ 1000 s) is split among the
  lengths, each stage receiving a fraction of the remaining time.  The code
  always returns the best sequence found before the deadline.

The whole pipeline runs without any filesystem or network access and uses only
NumPy (and the provided `evaluate_sequence`).  It is deliberately written
without closures or lambdas so that the grader can import it directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         fft_size: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·b_k) w.r.t. a.
    For very large α this approximates a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # --- soft‑max weights (or uniform over all almost‑maximal entries) ---
    if alpha > 2000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # --- convolution w * a_rev (a_rev = a reversed) ---
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_size)
    A_rev = np.fft.rfft(a_rev, n=fft_size)
    conv = np.fft.irfft(W * A_rev, n=fft_size)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _adam_optimize_one_length(init_a: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 10**9) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected optimisation on the simplex for a fixed length.
    Returns the best sequence found (still on the simplex) and the internal
    score 2·n·max(b) (i.e. the exact value of evaluate_sequence when the sum
    of the coefficients equals 1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta = 0.1                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.06
    alpha_max = 20000.0

    fft_size = 1 << ((3 * n - 2).bit_length())   # enough for w (2n‑1) + a_rev
    t = 0

    while time.time() < deadline and t < max_iters:
        t += 1

        # convolution & score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            # occasional log
            if t % 2000 == 0:
                print(f"[{time.time():.2f}] n={n} it={t} score={best_score:.6f}")

        # gradient (soft‑max)
        grad = _gradient_via_softmax(a, b, alpha, fft_size)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a, s=1.0)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: repeatedly move a random fraction of mass
    from one index to another and keep the move if the true evaluate_sequence
    score improves.  The routine stops when the deadline is reached.
    """
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] <= 1e-14:
            continue
        # random fraction of the mass at i (up to 30 %)
        frac = np.random.rand() * 0.30
        delta = best_seq[i] * frac
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        # project onto simplex to guard against tiny numerical drift
        cand = _project_onto_simplex(cand, s=1.0)
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand

        # tiny jitter every few thousand steps
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-6
            cand = best_seq + jitter
            cand = _project_onto_simplex(cand, s=1.0)
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand

    return best_seq, best_score


# ----------------------------------------------------------------------
def _upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Return a version of ``seq`` with length ``new_n``.
    If the new length is an integer multiple of the old one we simply repeat
    each entry and split its mass evenly; otherwise we line‑interpolate and
    finally project back onto the simplex.
    """
    old_n = len(seq)
    if new_n % old_n == 0:
        factor = new_n // old_n
        up = np.repeat(seq, factor) / factor
        return up
    # fallback: linear interpolation followed by simplex projection
    old_pos = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_pos = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(new_pos, old_pos, seq)
    up = _project_onto_simplex(up, s=1.0)
    return up

            break

        # allocate roughly 20 % of the remaining budget to this length
        slice_deadline = time.time() + max(0.1, remaining * 0.20)

        # initialise with several diverse seeds
        init_kinds = ("prev", "arcsine", "uniform", "random", "spike")
        for kind in init_kinds:
            if time.time() >= slice_deadline:
                break

            if kind == "prev":
                if n == base_n:
                    a0 = base_seq.copy()
                else:
                    a0 = _upsample_sequence(base_seq, n)
            elif kind == "spike":
                a0 = np.zeros(n, dtype=np.float64)
                # two spikes (half the mass each) placed far apart
                pos1 = np.random.randint(max(1, n // 8))
                pos2 = np.random.randint(n - max(1, n // 8), n)
                a0[pos1] = 0.5
                a0[pos2] = 0.5
                a0 = _project_onto_simplex(a0, s=1.0)
            elif kind == "uniform":
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
                a0 = _project_onto_simplex(a0, s=1.0)
            elif kind == "random":
                a0 = np.random.rand(n)
                a0 /= a0.sum()
                a0 = _project_onto_simplex(a0, s=1.0)
            else:   # arcsine
                a0 = _arcsine_distribution(n)

            # --------------------------------------------------------------
            # Adam‑based optimisation on the simplex
            seq, int_score = _adam_optimize_one_length(a0, n, slice_deadline)

            # --------------------------------------------------------------
            # short local‑search phase (if any time left)
            time_left = slice_deadline - time.time()
            if time_left > 0.5:
                seq, int_score = _local_search(seq, n,
                                               time.time() + min(time_left, 5.0))

            # evaluate with the official black‑box (more reliable)
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = int_score

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq.copy()
                print(f"*** New global best → len={n} init={kind} "
                      f"score={global_best_score:.6f}")

            # early exit if we already beat the target
            if global_best_score <= 1.503:
                break

        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # final refinement on the overall best (if time permits)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _local_search(global_best_seq,
                                       len(global_best_seq),
                                       deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if global_best_seq is None:
        n = 3000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
