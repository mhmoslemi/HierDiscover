"""
Propose a low‑score step‑function.

The algorithm works in four stages:

1.  Choose a problem size (default 8000, larger than the reference) and
    initialise a feasible point on the simplex Σa_i = √(2·n).

2.  Perform several cheap random‑Dirichlet restarts and keep the best
    candidate found (fast max‑convolution via FFT).

3.  Run a long Simulated‑Annealing on the simplex using the true
    max‑convolution as objective.

4.  Refine the best point with a continuation scheme based on a smooth
    max (soft‑max) surrogate solved by L‑BFGS‑B, followed by a projected
    sub‑gradient polish.

All convolutions are performed with the FFT (O(n log n)).  The code
contains only top‑level functions (no closures) and obeys the required
interface `propose_candidate(seed:int=42, budget_s:int=1000) -> list[float]`.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _fft_size(length: int) -> int:
    """Smallest power of two ≥ length (used for FFT‑based convolutions)."""
    return 1 << ((length - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0                # numerical safety
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT (fast, returns a float)."""
    conv = _full_convolution(a, size)
    return float(conv.max())


def _gradient_max_convolution(a: np.ndarray, conv: np.ndarray,
                             size: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a (ordered‑pair version).
    """
    max_c = conv.max()
    mask = np.isclose(conv, max_c, atol=1e-12)
    cnt = mask.sum()
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * len(a) - 1]
    g_a = 2.0 * conv_wr[len(a) - 1 : len(a) - 1 + len(a)]
    return g_a


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (soft‑max) objective and gradient w.r.t. log‑probabilities y.
    """
    p = _softmax(y)          # Σp = 1, p_i ≥ 0
    a = S * p                # Σa = S

    # convolution a*a (FFT)
    size_a = _fft_size(n)
    conv = _full_convolution(a, size_a)
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        # avoid overflow/underflow in the exponentials
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (using FFT for the inner convolution)
    a_rev = a[::-1]
    size_wr = _fft_size(len(w) + len(a_rev) - 1)
    fw = np.fft.rfft(w, n=size_wr)
    fa_rev = np.fft.rfft(a_rev, n=size_wr)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size_wr)[: len(w) + len(a_rev) - 1]
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # chain rule: grad_y = S * p * (g_a - <p, g_a>)
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g

def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: move a random fraction of the mass from one
    index to another and accept with a Metropolis rule based on the true
    max‑convolution (the exact objective because Σa = S).
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    # temperature schedule – start fairly high, cool to near zero
    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-8

    it = 0
    while time.time() - start_time < budget:
        # linear cooling
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        # pick a different index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of a[i] (up to 50 %)
        delta = np.random.rand() * a[i] * 0.5
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = False
        if diff < 0.0:
            accept = True
        else:
            if T > 0.0 and np.random.rand() < math.exp(-diff / T):
                accept = True

        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score


def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation on the smooth‑max surrogate (soft‑max).  Starts from the
    supplied point, runs an outer loop decreasing the smoothing parameter
    μ and an inner L‑BFGS‑B optimisation for each μ.
    """
    n = len(a_start)
    eps = 1e-12
    p0 = a_start / S
    y = np.log(p0 + eps).astype(np.float64)

    # initialise μ from the current max‑convolution
    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = _max_convolution(best_a, _fft_size(n))

    mu_factor = 0.7
    mu_min = 1e-7
    maxiter_per_mu = 500

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu,
                         "disp": False,
                         "gtol": 1e-12}
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = _max_convolution(a, _fft_size(n))
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true hard max (no smoothing).
    Uses a diminishing step‑size and stops when the time budget is exhausted.
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = _max_convolution(best_a, size)

    # step‑size parameters
    eta0 = S * 0.5 / n            # a typical magnitude of a_i
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 30000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # sub‑gradient of the max
        g_a = _gradient_max_convolution(a, conv, size)

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = _max_convolution(a_new, size)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step‑function within the allotted time.
    Returns the best sequence found (list of non‑negative floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # problem size – we deliberately use a larger n than the reference
    # ------------------------------------------------------------------
    try:
        n_ref = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n_ref = None
        have_ref = False

    # target size: at least 8000, otherwise twice the reference length
    n = max(8000, (n_ref or 0) * 2)

    S = math.sqrt(2.0 * n)            # ℓ¹‑norm that makes the score = max‑conv

    # ------------------------------------------------------------------
    # initial feasible point
    # ------------------------------------------------------------------
    if have_ref:
        a0 = np.zeros(n, dtype=float)
        a0[:n_ref] = np.asarray(height_sequence_1, dtype=float)
        a0 = _project_to_simplex(a0, S)
    else:
        a0 = np.full(n, S / n, dtype=float)   # uniform start

    best_a = a0.copy()
    best_score = _max_convolution(best_a, _fft_size(n))
    print(f"[init]   score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap random Dirichlet restarts (exploration)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        rr_budget = min(30.0, remaining * 0.20)
        t0 = time.time()
        print(f"[rand] exploring for {rr_budget:.1f}s")
        while time.time() - t0 < rr_budget:
            g = np.random.gamma(1.0, 1.0, size=n)
            a_rand = S * g / g.sum()
            sc = _max_convolution(a_rand, _fft_size(n))
            if sc < best_score:
                best_score = sc
                best_a = a_rand.copy()
                print(f"[rand] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long simulated annealing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        sa_time = remaining * 0.55
        print(f"[SA] running for {sa_time:.1f}s")
        best_a, best_score = _simulated_annealing(best_a, S, sa_time, time.time())
        print(f"[SA]    best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – smooth‑max L‑BFGS refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print(f"[L‑BFGS] refining for {remaining:.1f}s")
        best_a, best_score = _l_bfgs_refine(best_a, S, remaining, time.time())
        print(f"[L‑BFGS] best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[subgrad] polishing for {remaining:.1f}s")
        best_a, best_score = _final_subgradient(best_a, S, remaining, time.time())
        print(f"[subgrad]  best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # safety fallback (should never fire)
    # ------------------------------------------------------------------
    if best_score == float("inf") or np.isnan(best_score):
        best_a = np.full(n, S / n, dtype=float)
        best_score = _max_convolution(best_a, _fft_size(n))
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    # convert to plain Python list for the caller
    return best_a.tolist()
