def _conv_fft(u, v, size_fft):
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[:len(u)+len(v)-1]

def _self_convolution(a, size_fft):
    return _conv_fft(a, a, size_fft)

def _weighted_gradient(a, w, size_fft):
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)

    grad_len = len(a)
    grad = 2.0 * conv[start:start + grad_len]
    return grad

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
self‑convolution peak (the value returned by `evaluate_sequence`).

Algorithm
---------
1. **Exploration** – generate many candidates from a rich collection of
   analytic families (inverse‑sqrt, power‑law, exponential, cosine,
   Gaussian, beta‑shaped, random block, etc.).  Each candidate is projected
   onto the simplex {a ≥ 0, Σa = √(2 n)} and its exact score (max of the
   self‑convolution) is computed with an FFT‑based convolution.  The best
   `K` candidates are retained.

2. **Refinement (sub‑gradient descent)** – each retained candidate is
   improved by a projected sub‑gradient descent on the true objective.
   The sub‑gradient is the derivative of the active convolution term
   (the index attaining the maximum).  A diminishing step‑size and
   Euclidean projection keep the iterate feasible.

3. **Annealing fine‑tuning** – starting from the best refined candidate,
   a simulated‑annealing style random‑mass‑exchange walk is performed.
   Small amounts of mass are moved between two random positions;
   moves are accepted if they improve the score or with a temperature‑
   dependent probability.  This helps to escape shallow local minima.

The whole search respects a 1000 s wall‑clock budget and returns the best
sequence found.
"""

import math
import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution u * v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _quick_score(a: np.ndarray, size_fft: int) -> float:
    """Exact objective value (max of self‑convolution)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k at the active index `idx`.
    Implements the formula:
        g_i = a_{idx-i} (if valid) + a_{i-idx} (if valid)
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    # first term: a_{idx-i}
    i_min = max(0, idx - n + 1)
    i_max = min(idx, n - 1)
    if i_min <= i_max:
        i_vals = np.arange(i_min, i_max + 1)
        j_vals = idx - i_vals
        g[i_vals] += a[j_vals]

    # second term: a_{i-idx}
    i_min2 = max(idx, 0)
    i_max2 = min(idx + n - 1, n - 1)
    if i_min2 <= i_max2:
        i_vals2 = np.arange(i_min2, i_max2 + 1)
        j_vals2 = i_vals2 - idx
        g[i_vals2] += a[j_vals2]

    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The search respects a 1000 s time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size --------------------------------------------------------
    try:
        n = len(height_sequence_1)       # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)               # normalisation used by the paper
    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for all convolutions we perform

    # ------------------------------------------------------------------
    # Helpers -------------------------------------------------------------
    idx_array = np.arange(n)

    def consider(best_list: list[tuple[float, np.ndarray]],
                 score: float, a: np.ndarray, K: int) -> None:
        """Maintain a sorted list of the top‑K candidates (ascending score)."""
        if len(best_list) < K:
            best_list.append((score, a.copy()))
            best_list.sort(key=lambda x: x[0])
        elif score < best_list[-1][0]:
            best_list[-1] = (score, a.copy())
            best_list.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 1. PARAMETRIC EXPLORATION -------------------------------------------
    K_keep = 25
    raw_best: list[tuple[float, np.ndarray]] = []   # (score, a)

    def add_candidate(arr: np.ndarray) -> None:
        arr = _project_to_simplex(arr.astype(float), target_sum)
        sc = _quick_score(arr, size_fft)
        consider(raw_best, sc, arr, K_keep)

    # (a) uniform
    add_candidate(np.ones(n))

    # (b) inverse‑sqrt (single‑ended)
    for _ in range(200):
        beta = np.random.rand() * 12.0
        a = 1.0 / np.sqrt(idx_array + beta + 1.0)
        add_candidate(a)

    # (c) double‑ended inverse‑sqrt
    for _ in range(200):
        beta = np.random.rand() * 12.0
        a = (1.0 / np.sqrt(idx_array + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx_array) + beta + 1.0))
        add_candidate(a)

    # (d) power‑law
    for _ in range(200):
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 3.0 + 0.1   # exponent in [0.1,3.1]
        a = 1.0 / (idx_array + beta + 1.0) ** pexp
        add_candidate(a)

    # (e) exponential decay (single)
    for _ in range(200):
        decay = np.random.rand() * 0.02 + 0.0001
        a = np.exp(-decay * idx_array)
        add_candidate(a)

    # (f) exponential decay (double‑ended, possibly asymmetric)
    for _ in range(200):
        decay_l = np.random.rand() * 0.02 + 0.0001
        decay_r = np.random.rand() * 0.02 + 0.0001
        a = np.exp(-decay_l * idx_array) + np.exp(-decay_r * (n - 1 - idx_array))
        add_candidate(a)

    # (g) cosine‑based (large at ends, small in the middle)
    for _ in range(150):
        p = np.random.rand() * 8.0 + 0.5
        a = np.abs(np.cos(np.pi * idx_array / (n - 1))) ** p
        add_candidate(a)

    # (h) sine‑based (zero at ends, peak in the middle)
    for _ in range(150):
        p = np.random.rand() * 8.0 + 0.5
        a = np.sin(np.pi * idx_array / (n - 1)) ** p
        add_candidate(a)

    # (i) Gaussian‑like bump (centered)
    for _ in range(200):
        sigma = np.random.rand() * 0.3 + 0.05
        sigma = sigma * n
        centre = (n - 1) / 2.0 + (np.random.rand() - 0.5) * n * 0.2
        a = np.exp(-0.5 * ((idx_array - centre) / sigma) ** 2)
        add_candidate(a)

    # (j) triangular (symmetric peak at centre)
    for _ in range(120):
        peak = np.random.rand() * 4.0 + 0.2
        a = np.minimum(idx_array + 1, n - idx_array) ** peak
        add_candidate(a)

    # (k) Beta‑shaped (a_i ∝ (i+1)^α (n-i)^β)
    for _ in range(120):
        alpha = np.random.rand() * 5.0 + 0.1
        beta = np.random.rand() * 5.0 + 0.1
        a = (idx_array + 1) ** alpha * (n - idx_array) ** beta
        add_candidate(a)

    # (l) Random block candidates (few large constant blocks)
    for _ in range(500):
        nb = np.random.randint(2, 9)                 # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx_array, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        add_candidate(cand)

    # (m) Random simplex point (purely random feasible vector)
    for _ in range(200):
        rand_vec = np.random.rand(n)
        add_candidate(rand_vec)

    # (n) Reference sequence, if it exists
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        add_candidate(ref)

    raw_best.sort(key=lambda x: x[0])
    print(f"[exploration] best raw score = {raw_best[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. REFINEMENT – projected sub‑gradient descent -----------------------
    global_best_a = raw_best[0][1].copy()
    global_best_score = raw_best[0][0]

    # we will refine the top few (e.g. 7) raw candidates
    top_to_refine = raw_best[:7]

    for rank, (raw_score, a0) in enumerate(top_to_refine):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank + 1} (raw score {raw_score:.6f})")

        # initialise parameters for this run
        a = a0.copy()
        cur_score = _quick_score(a, size_fft)
        best_local_a = a.copy()
        best_local_score = cur_score

        # step size schedule
        step = target_sum * 0.02 / n    # initial magnitude
        it = 0

        # allocate a fraction of remaining time to this candidate
        # (roughly equal share)
        while time.time() < deadline:
            conv = _self_convolution(a, size_fft)
            idx_max = int(conv.argmax())

            # sub‑gradient at the active index
            grad = _subgradient_max(a, idx_max)

            # diminishing step
            step_adj = step / (1.0 + it * 1e-5)

            a_new = a - step_adj * grad
            a_new = _project_to_simplex(a_new, target_sum)

            new_score = _quick_score(a_new, size_fft)

            if new_score < best_local_score:
                best_local_score = new_score
                best_local_a = a_new.copy()
                # early reporting
                print(f"   [improve] new local best {new_score:.6f}")

            # accept if improves, otherwise keep old (no uphill moves here)
            if new_score <= cur_score:
                a = a_new
                cur_score = new_score

            # decay step size
            step *= 0.9997
            it += 1

            # stop early if step becomes tiny
            if step < 1e-9:
                break

        if best_local_score < global_best_score:
            global_best_score = best_local_score
            global_best_a = best_local_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. FINAL ANNEALING/TUNING ------------------------------------------
    # Use simulated‑annealing‑style random mass exchanges on the best vector.
    cur = global_best_a.copy()
    cur_score = _quick_score(cur, size_fft)
    best_seq = cur.copy()
    best_score = cur_score

    temperature = cur_score * 0.1 + 0.01   # initial temperature
    decay = 0.99995
    max_frac = 0.12                      # maximal fraction of a[i] moved per step

    print("[annealing] start fine‑tuning")
    while time.time() < deadline:
        # pick two distinct indices
        i = random.randrange(n)
        if cur[i] <= 0.0:
            continue
        j = random.randrange(n)
        while j == i:
            j = random.randrange(n)

        delta = cur[i] * random.random() * max_frac
        # propose new vector
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta

        # (sum is preserved, non‑negativity guaranteed)
        cand_score = _quick_score(cand, size_fft)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            print(f"[annealing] improvement to {best_score:.6f}")

        # Metropolis acceptance
        if cand_score < cur_score or random.random() < math.exp(-(cand_score - cur_score) / max(temperature, 1e-12)):
            cur = cand
            cur_score = cand_score

        temperature *= decay
        # safety break if temperature collapses
        if temperature < 1e-8:
            temperature = 1e-8

    print(f"[final] best score = {best_score:.6f}")
    return best_seq.tolist()
