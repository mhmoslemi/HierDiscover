def _adam_optimize(a0, n, deadline, max_iter=100000):
    a = a0.copy()
    # convert to logits z
    eps = 1e-20
    z = np.log(a + eps)   # note sum(exp(z)) = roughly 1 but not exactly; compute a from softmax each iteration.
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05
    it = 0
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()
    alpha = 5.0
    alpha_mul = 1.08
    alpha_max = 20000.0
    size_fft = 1 << ((3 * n - 2).bit_length())
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute a via softmax
        max_z = np.max(z)
        expz = np.exp(z - max_z)
        a = expz / expz.sum()
        # symmetrize optional
        if it % 500 == 0:
            a = (a + a[::-1]) / 2.0
            # recompute z
            z = np.log(a + eps)
        # conv and max
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
        # compute w (softmax weights) for gradient
        if alpha > 2000:
            max_b = b.max()
            mask = np.abs(b - max_b) < 1e-12
            w = mask.astype(float)
            cnt = w.sum()
            if cnt > 0:
                w /= cnt
            else:
                w[np.argmax(b)] = 1.0
        else:
            shift = b - b.max()
            w = np.exp(alpha * shift)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.zeros_like(b)
                w[np.argmax(b)] = 1.0
            else:
                w /= w_sum
        # gradient w.r.t a
        grad_a = _gradient_via_softmax(a, b, alpha, size_fft)  # returns 2*conv(w, a_rev)
        # compute grad_z
        dot_ag = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_ag)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # anneal alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)
    return best_seq, best_score

Advanced optimizer for the step‑function challenge.

The algorithm works on the probability simplex (sum = 1, a_i ≥ 0) by
parameterising the vector with logits z (soft‑max).  A Adam optimizer
with an annealed soft‑max temperature α is applied to minimise a smooth
approximation of max(a∗a).  After each Adam run a cheap deterministic
mass‑rebalancing step (based on the exact sub‑gradient of the max‑norm)
tries to shave the remaining peak.  The whole procedure is repeated
for several lengths and initial shapes; the best sequence seen is
returned.

Key ideas:
* Soft‑max (log‑sum‑exp) surrogate for the max‑convolution and its
  analytically available gradient (via FFT).
* Adam on logits → natural mirror‑descent on the simplex.
* Periodic symmetrisation (a ← (a+a_rev)/2) – the optimal shape is
  symmetric.
* Deterministic “rebalancing” local search using the true sub‑gradient
  of the max‑convolution (uniform weight on all active indices).
* Time‑budget aware multi‑restart over a range of lengths.

The objective that is minimised is
    2·n·max(a∗a)            (the value returned by evaluate_sequence when ∑a=1)
which is invariant to scaling of a, so we keep the vector normalised to
sum‑one throughout.
"""

import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·b_k) w.r.t. a.
    For huge α this becomes a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # ---- compute soft‑max weights ----
    if alpha > 2000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ---- conv(w, a_rev) ----
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]          # ∂/∂a of the smooth max
    return g


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation on the simplex using Adam over logits.
    Returns the best probability vector found and its internal score
    2·n·max(a∗a) (i.e. the value of evaluate_sequence when sum=1).
    """
    eps = 1e-20
    # initialise logits from a0 (soft‑max will be used later)
    z = np.log(a0 + eps)

    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05

    # soft‑max temperature
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- obtain a from logits (soft‑max) -----
        max_z = np.max(z)
        expz = np.exp(z - max_z)
        a = expz / expz.sum()

        # occasional symmetrisation (preserves sum)
        if it % 500 == 0:
            a = (a + a[::-1]) / 2.0
            z = np.log(a + eps)

        # ----- evaluate current objective -----
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            # progress print (quiet enough)
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # ----- gradient of the smooth max -----
        grad_a = _gradient_via_softmax(a, b, alpha, size_fft)   # 2·conv(w,a_rev)

        # ----- map gradient to logits (mirror‑descent) -----
        dot_ag = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot_ag)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- anneal α -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # safety break if no time left
        if time.time() >= deadline:
            break

    if best_seq is None:       # should never happen
        best_seq = a0.copy()
        best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    return best_seq, best_score


# ----------------------------------------------------------------------
def _deterministic_rebalance(a: np.ndarray,
                             n: int,
                             deadline: float,
                             eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Very cheap local search that repeatedly moves a small fraction of mass
    from the index with the largest contribution to the current max‑convolution
    (i.e. highest sub‑gradient component) to the index with the smallest
    contribution.  Stops when no improvement is seen or time runs out.
    Returns the improved sequence and its internal score.
    """
    best_seq = a.copy()
    best_b = _fft_self_convolve(best_seq)
    best_max = best_b.max()

    size_fft = 1 << ((3 * n - 2).bit_length())
    a_rev = best_seq[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=size_fft)

    while time.time() < deadline:
        # active indices of the convolution peak
        active = np.where(best_b >= best_max - eps)[0]
        if active.size == 0:
            break

        # uniform weight on active indices → sub‑gradient contributions
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active] = 1.0 / active.size

        # convolution w * a_rev (full) via FFT
        w_fft = np.fft.rfft(w, n=size_fft)
        conv = np.fft.irfft(w_fft * a_rev_fft, n=size_fft)
        start = n - 1
        contrib = conv[start:start + n]               # non‑scaled sub‑gradient

        i_high = np.argmax(contrib)
        i_low = np.argmin(contrib)
        if i_high == i_low or best_seq[i_high] < 1e-12:
            break

        # move a modest fraction of mass
        delta = min(best_seq[i_high] * 0.20, 0.005)   # never too large
        a_new = best_seq.copy()
        a_new[i_high] -= delta
        a_new[i_low] += delta

        b_new = _fft_self_convolve(a_new)
        max_new = b_new.max()
        if max_new < best_max - 1e-14:
            best_seq = a_new
            best_b = b_new
            best_max = max_new
            # update cached FFT of the reversed vector for next iteration
            a_rev = best_seq[::-1]
            a_rev_fft = np.fft.rfft(a_rev, n=size_fft)
        else:
            # no improvement → stop
            break

    return best_seq, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point.  Searches over several lengths and initialisations,
    using Adam on logits plus deterministic rebalancing.
    Returns the best list of heights found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Determine a set of lengths to explore (capped at 20000 for speed)
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    length_factors = (0.7, 0.85, 1.0, 1.2, 1.5, 2.0, 2.5)
    lengths = sorted({
        max(300, min(20000, int(base_n * f)))
        for f in length_factors
    })

    global_best_seq = None
    global_best_score = float("inf")
    target_score = 1.503

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly 20 % of remaining time to this length
        slice_deadline = time.time() + max(0.1, remaining * 0.20)

        for init_kind in ("prev", "arcsine", "uniform", "random", "spike"):
            if time.time() >= slice_deadline:
                break

            # ----- generate initial probability vector -----
            if init_kind == "spike":
                a0 = np.zeros(n, dtype=np.float64)
                pos1 = np.random.randint(max(1, n // 10))
                pos2 = np.random.randint(n - max(1, n // 10), n)
                a0[pos1] = 0.5
                a0[pos2] = 0.5
                a0 = _project_onto_simplex(a0, s=1.0)
            else:
                a0 = _initial_sequence(n, init_kind)

            # ----- Adam optimisation on logits -----
            seq_adam, _ = _adam_optimize(a0, n, slice_deadline)

            # ----- deterministic rebalancing (if time permits) -----
            time_left = slice_deadline - time.time()
            if time_left > 0.2:
                seq_bal, _ = _deterministic_rebalance(seq_adam, n,
                                                      time.time() + min(time_left, 4.0))
            else:
                seq_bal = seq_adam

            # Evaluate with the official scorer (robust to clipping)
            try:
                final_score = evaluate_sequence(seq_bal.tolist())
            except Exception:
                # fallback to internal score if evaluate_sequence misbehaves
                max_b = _fft_self_convolve(seq_bal).max()
                final_score = float(2.0 * n * max_b)

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq_bal.copy()
                print(f"*** New global best → len={n} init={init_kind} "
                      f"score={global_best_score:.6f}")

            # Early exit if target already reached
            if global_best_score <= target_score:
                break

        if global_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Final refinement on the overall best (if any time left)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _deterministic_rebalance(
            global_best_seq,
            len(global_best_seq),
            deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
