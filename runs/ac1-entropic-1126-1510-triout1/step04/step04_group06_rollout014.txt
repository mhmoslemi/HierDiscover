def _refine_candidate_subgrad(a0,..., max_iters=None):
   a = a0.copy()
   best_a = a.copy()
   best_score = quick_score(a,...)
   # step size schedule based on target_sum/n
   base_lr = target_sum * 0.3 / n
   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8
   it = 0
   while time.time() < deadline and (max_iters is None or it < max_iters):
       conv = _self_convolution(a, size_fft)
       max_val = conv.max()
       # Determine active set: indices where conv within (1+tol) of max
       tol = 0.005
       idx_active = np.where(conv >= max_val * (1 - tol))[0]
       if idx_active.size == 0:
           w = np.zeros_like(conv)
           w[conv.argmax()] = 1.0
       else:
           w = np.zeros_like(conv)
           w[idx_active] = 1.0 / idx_active.size
       grad = _weighted_gradient(a, w, size_fft)
       # Adam update
       lr = base_lr / (1 + it * 1e-5)
       m = beta1 * m + (1 - beta1) * grad
       v = beta2 * v + (1 - beta2) * (grad * grad)
       m_hat = m / (1 - beta1 ** (it + 1))
       v_hat = v / (1 - beta2 ** (it + 1))
       a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
       a_new = _project_to_simplex(a_new, target_sum)
       # occasional random perturb
       if it % 3000 == 0 and it > 0:
           noise = np.random.randn(n) * target_sum * 0.0005
           a_new = _project_to_simplex(a_new + noise, target_sum)
       a = a_new
       cur_score = quick_score(a, n, size_fft)
       if cur_score < best_score:
           best_score = cur_score
           best_a = a.copy()
       it += 1
   return best_a, best_score

"""Search for a step‑function that minimizes the self‑convolution peak.

The algorithm consists of three phases (all respecting a 1000 s time
budget):
1. Exploration – generate thousands of candidates from a wide collection
   of analytic families (inverse‑sqrt, power‑law, exponential,
   cosine, Gaussian, triangular, beta, random block).  Each candidate is
   projected onto the simplex {a ≥ 0, Σa = √(2 n)} and the exact score
   2·n·max(a∗a)/Σa² is computed with FFTs.  The K = 30 best raw
   candidates are kept.
2. Refinement – the retained candidates are improved using two
   gradient‑based schemes:
   * an Adam optimiser applied to a smooth‑max (soft‑max) approximation
     of the peak,
   * a sub‑gradient descent that focuses on the active convolution
     indices (those attaining the current maximum).  Both updates are
   projected onto the simplex.
3. Final sweep – the current global best is refined for the remaining
   time with the sub‑gradient method, and a quick random‑block sweep
   updates the best solution whenever an improvement is found.

All operations respect the security checks of `evaluate_sequence`,
and the function returns the best sequence discovered before the time

import time

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.ifft(fu * fv).real
    # `conv` has length `size_fft`; we need only the first len(u)+len(v)-1 entries
    return conv[: len(u) + len(v) - 1]

def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a∗a."""
    return _conv_fft(a, a, size_fft)

def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    (w must sum to 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)
    start = len(w) - len(a)  # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad

def _quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective value without security checks."""
    conv = _self_convolution(a, size_fft)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser with a soft‑max approximation of the max‑peak."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    eta0 = target_sum * 0.5 / n
    # initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights (smooth approximation of max) -----
        shifted = conv - conv.max()
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam step -----
        eta = eta0 / (1.0 + it * 1e-5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9997
        it += 1

    return best_a, best_score

def _refine_candidate_subgrad(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = None,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent that concentrates on the active set
    (indices where the convolution attains its current maximum).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _quick_score(a, n, size_fft)

    # Adam‑style state for stability
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    base_lr = target_sum * 0.3 / n
    it = 0
    while time.time() < deadline and (max_iters is None or it < max_iters):
        conv = _self_convolution(a, size_fft)
        max_val = conv.max()

        # active set – all points within 1 % of the peak
        tol = 0.01
        active = np.where(conv >= max_val * (1 - tol))[0]
        if active.size == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size

        grad = _weighted_gradient(a, w, size_fft)

        lr = base_lr / (1.0 + it * 1e-5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random perturbation to escape plateaus
        if it % 3000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.0005
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        cur_score = _quick_score(a, n, size_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score

def _evaluate_and_update(
    cand: np.ndarray,

    target_sum: float,

    else:
        if sc < best_list[-1][0]:
            best_list[-1] = (sc, proj.copy())
            best_list.sort(key=lambda x: x[0])

"""Search for a step‑function that minimizes the self‑convolution peak.

The algorithm works in three phases (all respecting a 1000 s time budget):

1️⃣ Exploration – generate thousands of candidates from a wide collection of
   analytic families (inverse‑sqrt, power‑law, exponential, cosine,
   Gaussian, triangular, beta‑shaped, random block).  Each candidate is
   projected onto the simplex {a ≥ 0, Σa = √(2 n)} and its exact score
   2·n·max(a∗a)/Σa² is computed with FFTs.  The K = 30 best raw candidates
   are retained.

2️⃣ Refinement – each retained candidate is improved using two gradient‑

     peak,

     indices where the convolution attains its current maximum.

3️⃣ Final finetuning – the current global best is refined for the remaining
   time with the sub‑gradient method, and a quick random‑block sweep updates

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    target_sum = math.sqrt(2.0 * n)
    size_fft = 1 << ((3 * n - 2).bit_length())
    idx = np.arange(n)

    K = 30                         # retain the K best raw candidates
    best_candidates: list[tuple[float, np.ndarray]] = []

    # ---------------------- 1. EXPLORATION -------------------------
    _evaluate_and_update(np.ones(n), n, size_fft, target_sum,
                         best_candidates, K)

    # inverse‑sqrt (single‑ended)
    for _ in range(300):
        if time.time() >= deadline:
            break
        beta = np.random.rand() * 12.0
        cand = 1.0 / np.sqrt(idx + beta + 1.0)
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # double‑ended inverse‑sqrt
    for _ in range(300):
        if time.time() >= deadline:
            break
        beta = np.random.rand() * 12.0
        cand = (1.0 / np.sqrt(idx + beta + 1.0) +
                1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # power‑law
    for _ in range(300):
        if time.time() >= deadline:
            break
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 1.8 + 0.2
        cand = 1.0 / (idx + beta + 1.0) ** pexp
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # exponential decay (single‑ended)
    for _ in range(200):
        if time.time() >= deadline:
            break
        decay = np.random.rand() * 0.015 + 0.0002
        cand = np.exp(-decay * idx)
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # exponential decay (double‑ended)
    for _ in range(200):
        if time.time() >= deadline:
            break
        decay_l = np.random.rand() * 0.015 + 0.0002
        decay_r = np.random.rand() * 0.015 + 0.0002
        cand = np.exp(-decay_l * idx) + np.exp(-decay_r * (n - 1 - idx))
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # cosine‑based (ends large)
    for _ in range(200):
        if time.time() >= deadline:
            break
        p = np.random.rand() * 6.0 + 0.5
        cand = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # sine‑based (middle peak)
    for _ in range(200):
        if time.time() >= deadline:
            break
        p = np.random.rand() * 6.0 + 0.5
        cand = np.sin(np.pi * idx / (n - 1)) ** p
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # Gaussian bump
    for _ in range(200):
        if time.time() >= deadline:
            break
        sigma_frac = np.random.rand() * 0.25 + 0.05
        sigma = sigma_frac * n
        centre = (n - 1) / 2.0 + (np.random.rand() - 0.5) * n * 0.1
        cand = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # triangular shape
    for _ in range(150):
        if time.time() >= deadline:
            break
        peak = np.random.rand() * 3.0 + 0.5
        cand = np.minimum(idx + 1, n - idx) ** peak
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # beta‑shaped
    for _ in range(150):
        if time.time() >= deadline:
            break
        alpha = np.random.rand() * 3.0 + 0.2
        beta = np.random.rand() * 3.0 + 0.2
        cand = (idx + 1) ** alpha * (n - idx) ** beta
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # random block candidates (few constant blocks)
    for _ in range(800):
        if time.time() >= deadline:
            break
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        cand = np.empty(n, dtype=float)
        start = 0
        values = np.random.rand(nb) + 0.1
        for cut, v in zip(np.append(cuts, n), values):
            cand[start:cut] = v
            start = cut
        _evaluate_and_update(cand, n, size_fft, target_sum,
                             best_candidates, K)

    # reference sequence, if available
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        _evaluate_and_update(a_ref, n, size_fft, target_sum,
                             best_candidates, K)

    print(f"[explore] collected {len(best_candidates)} candidates, best raw score {best_candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ REFINEMENT – Adam then sub‑gradient on the top candidates
    best_candidates.sort(key=lambda x: x[0])
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    refine_limit = min(10, len(best_candidates))

    for rank in range(refine_limit):
        if time.time() >= deadline:
            break
        a_start = best_candidates[rank][1]
        raw_sc = best_candidates[rank][0]
        print(f"[refine] candidate #{rank+1} (raw {raw_sc:.6f})")
        remaining = deadline - time.time()
        max_iters = int(min(200_000, remaining * 1300))

        # Adam on smooth max
        a_adam, sc_adam = _refine_candidate_adam(
            a_start, deadline, n, size_fft, target_sum, max_iters)
        if sc_adam < global_best_score:
            global_best_score = sc_adam
            global_best_a = a_adam.copy()
            print(f"[global] improved after Adam → {global_best_score:.6f}")

        # Sub‑gradient fine‑tuning
        if time.time() < deadline:
            a_sub, sc_sub = _refine_candidate_subgrad(
                a_adam, deadline, n, size_fft, target_sum,
                max_iters=int(remaining * 500))
            if sc_sub < global_best_score:
                global_best_score = sc_sub
                global_best_a = a_sub.copy()
                print(f"[global] improved after sub‑grad → {global_best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ FINAL LONG FINETUNING ON THE CURRENT BEST
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"[final‑finetune] remaining time {remaining:.2f}s")
        a_final, sc_final = _refine_candidate_subgrad(
            global_best_a, deadline, n, size_fft, target_sum,
            max_iters=int(remaining * 1200))
        if sc_final < global_best_score:
            global_best_score = sc_final
            global_best_a = a_final.copy()
            print(f"[final] best after long finetune = {global_best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ QUICK RANDOM BLOCK SWEEP (if time permits)
    while time.time() < deadline:
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        cand = np.empty(n, dtype=float)
        start = 0
        values = np.random.rand(nb) + 0.1
        for cut, v in zip(np.append(cuts, n), values):
            cand[start:cut] = v
            start = cut
        cand = _project_to_simplex(cand, target_sum)
        sc = _quick_score(cand, n, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[block‑sweep] improved to {global_best_score:.6f}")

        if time.time() + 0.02 > deadline:
            break

    print(f"[finished] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
