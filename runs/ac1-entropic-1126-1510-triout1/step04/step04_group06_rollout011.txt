```python
"""
Propose a non‑negative step function that reduces the self‑convolution peak.

Algorithm
---------
1. **Exploratory generation** – sample a large set of candidate
   sequences from several analytic families (uniform, inverse‑sqrt,
   power‑law, exponentials, cosine, Gaussian, triangular, beta, random
   block, equally‑spaced spikes, random spikes, mixtures of Gaussians).
   Each candidate is projected onto the simplex Σa = √(2 n) and scored
   with an exact FFT‑based `quick_score`.  The best K raw candidates are
   retained.

2. **Intensive Adam‑based refinement** – the current global best is
   refined for the remaining time with a smooth‑max objective.  The
   temperature is slowly cooled, gradients are computed via FFT and the
   iterate is projected onto the simplex after every step.  Random
   perturbations are injected every few thousand iterations to escape
   plateaus.

3. **Light final sweep** – a quick pass of random‑block candidates
   updates the global best if a lower score is found.

The function respects the 1000 s budget and returns the best sequence
found.
"""

import math, time, random
import numpy as np

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x>=0, sum(x)=z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    v = v.astype(float, copy=True)
    n = v.shape[0]
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would be zero, put all mass in first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using FFT, returns full length result."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective without the security wrapper."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """Gradient of Σ_k w_k·(a*a)_k w.r.t a."""
    # conv of w with reversed a
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)      # length = len(w)+len(a)‑1 = 3n‑2
    start = len(w) - len(a)                   # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


def _refine_with_adam(a0: np.ndarray,
                      deadline: float,
                      n: int,
                      size_fft: int,
                      target_sum: float,
                      max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Smooth‑max Adam optimisation of the self‑convolution peak."""
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # learning‑rate scale: roughly target_sum / n
    eta0 = target_sum * 0.4 / n

    # initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.6)

    it = 0
    while time.time() < deadline and it < max_iters:
        # compute convolution and current (exact) score
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights – smoother when T is large, sharper as T → 0
        shifted = conv - conv.max()          # ≤ 0
        exp_arg = shifted / max(T, 1e-12)
        # avoid overflow
        exp_arg = np.clip(exp_arg, -60, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        grad = _weighted_gradient(a, w, size_fft)

        # Adam update
        eta = eta0 / (1.0 + it * 5e-5)        # very slow decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # project back to simplex
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random disturbance to avoid stagnation
        if it % 4000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.0008
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        # cool temperature – exponential decay, never below a floor
        T = max(1e-6, T * 0.9995)
        it += 1

    return best_a, best_score

"""
Propose a non‑negative step function that minimizes the self‑convolution peak.

Algorithm
---------
1. **Exploratory sweep** – generate thousands of candidates from a rich
   collection of analytic families (inverse‑sqrt, power‑law,
   exponential, cosine, Gaussian, triangular, beta‑shaped,
   random blocks, evenly‑spaced spikes, random spikes, mixtures of
   Gaussians, …).  Each candidate is projected onto the simplex
   {a ≥ 0, Σa = √(2 n)} and its exact score is computed with an FFT‑based
   routine.  The best `K` raw candidates are retained.

2. **Intensive Adam refinement** – the current global best is refined for
   the remaining time with a smooth‑max (soft‑max) objective.
   Gradient of the smoothed objective is obtained analytically via FFTs.
   Adam updates are followed by a Euclidean projection onto the simplex.
   Temperature is slowly cooled; occasional random kicks help escape
   plateaus.  A few secondary raw candidates are also refined briefly
   to avoid stagnation.

3. **Final random‑block sweep** – a quick pass of random‑block candidates
   updates the global best whenever a lower score is found.

All operations respect the security checks of `evaluate_sequence`.
The function returns the best sequence discovered before the 1000 s
budget expires.
"""

import math
import time
import random

import numpy as np


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    v = v.astype(float, copy=True)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective used during the search (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    (w must sum to 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _refine_with_adam(a0: np.ndarray,
                      deadline: float,
                      n: int,
                      size_fft: int,
                      target_sum: float,
                      max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Smooth‑max Adam optimisation of the self‑convolution peak.
    Returns the best sequence observed and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning rate (scaled to problem size)
    eta0 = target_sum * 0.4 / n

    # initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.6)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights (smooth approximation of max) -----
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-12)
        exp_arg = np.clip(exp_arg, -60, 0)         # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam step -----
        eta = eta0 / (1.0 + it * 5e-5)               # very gentle decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random kick to escape plateaus -----
        if it % 4000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.0008
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T = max(1e-6, T * 0.9995)                  # exponential cooling
        it += 1

    return best_a, best_score
```
