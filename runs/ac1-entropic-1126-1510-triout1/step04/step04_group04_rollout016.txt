def _smooth_descent(a0, size_fft, target_sum, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = np.max(_full_conv(a, size_fft))
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    base_lr = target_sum * 0.5 / a.size
    it = 0
    tau = 0.5
    while time.time() < deadline:
        # update tau schedule
        # maybe increase tau slowly each 1000 iterations
        if it % 1000 == 0 and tau < 10:
            tau = min(10, tau * 1.2)
        # compute gradient
        conv = _full_conv(a, size_fft)  # shape (2n-1)
        # compute smoothmax as log-sum-exp (optional)
        # compute weights
        maxc = conv.max()
        exp_weights = np.exp(tau * (conv - maxc))
        p = exp_weights / exp_weights.sum()
        # compute gradient via conv of a and reversed p
        rev_p = p[::-1]
        conv_rev = _conv_fft(a, rev_p, size_fft)[: len(a) + len(p) - 1]
        # compute gradient g
        n = a.size
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * conv_rev[idx]
        # Adam update
        it += 1
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        eta = base_lr / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = _project_to_simplex(a, target_sum)
        # track best max conv
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[smooth] iter {it} best {best_score:.6f} tau {tau:.3f}")
        # check deadline occasionally
        if it % 1000 == 0 and time.time() >= deadline:
            break
    return best_a, best_score

"""
Search for a non‑negative step‑function that minimizes the peak of its
self‑convolution (the evaluation score).  The algorithm consists of

1.  A broad random sampling of several families of vectors and keeping the
    best K raw candidates (projected onto the simplex Σa = √(2 n)).
2.  For each candidate a short “smooth‑max” Adam optimisation:
    * minimise a smooth approximation of the max‑convolution using the
      log‑sum‑exp (soft‑max) with an annealing temperature τ.
    * the gradient is obtained analytically via a second FFT‑based convolution.
3.  A final permutation hill‑climb (random swaps and tiny perturbations)
    on the globally best vector.
All steps respect the global 1000 s time budget and keep the vector in the
feasible region (non‑negative, fixed ℓ₁‑norm).  The routine returns the
best sequence found as a plain list of floats.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Smallest power‑of‑2 ≥ required (for FFT speed)."""
    return 1 << ((required - 1).bit_length())

# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a (length 2 n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]

# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())

# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    size_fft_conv: int,
    size_fft_grad: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the *smooth* max‑convolution.
    The smooth objective is  (1/τ)·log(∑exp(τ·conv)), whose gradient can be
    expressed through a second convolution.  τ is annealed from 0.5 up to ~8.
    Returns the best vector seen and its (non‑smoothed) peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft_conv)

    n = a.size
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    # base learning rate – roughly the size of a single entry
    eta0 = target_sum * 0.5 / n

    tau = 0.5
    it = 0
    while time.time() < deadline and it < max_iters:
        # increase temperature slowly
        if it % 2000 == 0 and tau < 8.0:
            tau = min(8.0, tau * 1.2)

        # ----- convolution a*a -------------------------------------------------
        conv = _full_conv(a, size_fft_conv)           # length 2n‑1

        # ----- soft‑max weights -------------------------------------------------
        maxc = conv.max()
        expw = np.exp(tau * (conv - maxc))
        p = expw / expw.sum()                         # length 2n‑1, sums to 1

        # ----- gradient = 2 * (a * reverse(p)) -------------------------------
        rev_p = p[::-1]
        # convolution a * rev_p (full) using FFT
        fa = np.fft.rfft(a, n=size_fft_grad)
        fp = np.fft.rfft(rev_p, n=size_fft_grad)
        conv_rev_full = np.fft.irfft(fa * fp, n=size_fft_grad)
        # indices (2n‑2‑j) give the needed slice of length n
        idx = (2 * n - 2) - np.arange(n)
        g = 2.0 * conv_rev_full[idx]

        # ----- Adam update ----------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex ------------------------------------
        a = _project_to_simplex(a, target_sum)

        # ----- keep track of the true (non‑smooth) peak -----------------------
        cur_max = _max_convolution(a, size_fft_conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[smooth] iter {it:>6d} best {best_score:.6f} τ={tau:.2f}")

        # occasional early break if we are out of time
        if it % 5000 == 0 and time.time() >= deadline:
            break

    return best_a, best_score

# ----------------------------------------------------------------------
def _push_candidate(pool: list[tuple[float, np.ndarray]], entry: tuple[float, np.ndarray], K: int) -> None:
    """Insert entry into a size‑limited sorted pool (ascending by score)."""
    if len(pool) < K:
        pool.append(entry)
        pool.sort(key=lambda x: x[0])
    else:
        if entry[0] < pool[-1][0]:
            pool[-1] = entry
            pool.sort(key=lambda x: x[0])

    The function respects the global budget (default 1000 s) and returns the
    best sequence discovered as a plain list of floats.
    """

        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    # ---- broad random sampling -------------------------------------------
    K = 12

    idx = np.arange(n)

    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:

            a = probs * target_sum

    for _ in range(30):

        score = _max_convolution(a, size_fft_conv)
        _push_candidate(best_raw, (score, a), K)

    for _ in range(150):

    # ---- smooth‑max refinement of the top candidates -----------------------
    global_best_a = best_raw[0][1].copy()

        remaining = deadline - time.time()

        maxc = conv.max()
        expw = np.exp(tau * (conv - maxc))
        p = expw / expw.sum()                         # length 2n‑1

        # ----- gradient = 2 * (a * reverse(p)) -------------------------------
        rev_p = p[::-1]
        # convolution a * rev_p (full) via FFT
        fa = np.fft.rfft(a, n=size_fft_grad)
        fp = np.fft.rfft(rev_p, n=size_fft_grad)
        conv_rev_full = np.fft.irfft(fa * fp, n=size_fft_grad)

        # pick the slice that corresponds to the gradient
        idx = (2 * n - 2) - np.arange(n)               # descending indices
        g = 2.0 * conv_rev_full[idx]

        # ----- Adam update ----------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- keep feasibility ------------------------------------------------
        a = _project_to_simplex(a, target_sum)

        # ----- evaluate true peak ---------------------------------------------
        cur_max = _max_convolution(a, size_fft_conv)
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[smooth] iter {it:>6d} best {best_score:.6f} τ={tau:.2f}")

        if it % 5000 == 0 and time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def _push_candidate(pool: list[tuple[float, np.ndarray]],
                    entry: tuple[float, np.ndarray],
                    K: int) -> None:
    """Insert entry into a size‑limited sorted pool (ascending by score)."""
    if len(pool) < K:
        pool.append(entry)
        pool.sort(key=lambda x: x[0])
    else:
        if entry[0] < pool[-1][0]:
            pool[-1] = entry
            pool.sort(key=lambda x: x[0])

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low self‑convolution peak.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ---- problem size ----------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # Σa = √(2 n)
    size_fft_conv = _fft_size(2 * n - 1)     # for a*a
    size_fft_grad = _fft_size(3 * n - 2)     # for a * rev(p)

    # ---- broad random sampling -------------------------------------------
    K = 12
    best_raw: list[tuple[float, np.ndarray]] = []   # (score, vector)
    idx = np.arange(n)

    # reference vector (if supplied)
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        score = _max_convolution(a_ref, size_fft_conv)
        _push_candidate(best_raw, (score, a_ref), K)
        print(f"[init] reference score = {score:.6f}")

    # uniform baseline
    a_uni = np.ones(n)
    a_uni = _project_to_simplex(a_uni, target_sum)
    score = _max_convolution(a_uni, size_fft_conv)
    _push_candidate(best_raw, (score, a_uni), K)
    print(f"[init] uniform score = {score:.6f}")

    # Dirichlet draws with several concentrations
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(12):
            probs = np.random.dirichlet(np.full(n, alpha))
            a = probs * target_sum
            score = _max_convolution(a, size_fft_conv)
            _push_candidate(best_raw, (score, a), K)

    # Power‑law decreasing shapes
    for _ in range(30):
        p = np.random.rand() * 3.0 + 0.2
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        score = _max_convolution(a, size_fft_conv)
        _push_candidate(best_raw, (score, a), K)

    # Exponential decay shapes
    for _ in range(30):
        dec = np.random.rand() * 0.015 + 0.001
        a = np.exp(-dec * idx)
        a = _project_to_simplex(a, target_sum)
        score = _max_convolution(a, size_fft_conv)
        _push_candidate(best_raw, (score, a), K)

    # Random block‑wise constant structures (2‑6 blocks)
    for _ in range(150):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        score = _max_convolution(cand, size_fft_conv)
        _push_candidate(best_raw, (score, cand), K)

    print(f"[sampling] best raw score = {best_raw[0][0]:.6f}")

    # ---- smooth‑max refinement of top candidates -------------------------
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    for rank, (raw_sc, a0) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        print(f"[smooth start] candidate #{rank+1} raw score {raw_sc:.6f}")

        # allocate a slice of the remaining budget (≤120 s per candidate)
        remaining = deadline - time.time()
        cand_deadline = time.time() + min(120.0, remaining * 0.6)

        refined_a, refined_sc = _smooth_descent(
            a0, size_fft_conv, size_fft_grad, target_sum, cand_deadline
        )
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] new best after smooth = {global_best_score:.6f}")

    # ---- permutation hill‑climbing ---------------------------------------
    print("[local swap] start")
    a = global_best_a.copy()
    best_sc = global_best_score

    while time.time() < deadline:
        # random swap (preserves ℓ₁‑norm automatically)
        i, j = np.random.choice(n, 2, replace=False)
        a_swapped = a.copy()
        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
        sc = _max_convolution(a_swapped, size_fft_conv)
        if sc < best_sc:
            best_sc = sc
            a = a_swapped
            global_best_a = a.copy()
            global_best_score = sc
            print(f"[swap] improved to {best_sc:.6f}")

        # occasional tiny mass‑reallocation move
        if np.random.rand() < 0.001:
            i2 = np.random.randint(n)
            delta = (np.random.rand() * 2 - 1) * 0.02 * target_sum
            if a[i2] - delta > 0:
                a2 = a.copy()
                a2[i2] -= delta
                j2 = np.random.randint(n)
                a2[j2] += delta
                a2 = _project_to_simplex(a2, target_sum)
                sc2 = _max_convolution(a2, size_fft_conv)
                if sc2 < best_sc:
                    best_sc = sc2
                    a = a2
                    global_best_a = a2.copy()
                    global_best_score = sc2
                    print(f"[perturb] improved to {best_sc:.6f}")

        # leave a tiny margin before the hard deadline
        if time.time() + 0.01 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
