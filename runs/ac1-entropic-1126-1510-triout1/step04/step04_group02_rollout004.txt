def cheap_objective(seq):

    n = len(seq)

    # ensure non-negatives maybe not needed.

    # compute conv via FFT with size next power-of-two

    size = 1 << ((2*n - 1) - 1).bit_length()

    a = np.fft.rfft(seq, n=size)

    b = a * a

    conv = np.fft.irfft(b, n=size)[:2*n-1]

    max_b = np.max(conv)

    sum_a = np.sum(seq)

    # Protect sum small (like earlier). If sum_a < 0.01: return inf

    if sum_a < 0.01:

        return float('inf')

    return 2 * n * max_b / (sum_a**2)

"""
Propose a candidate step‑function (list of non‑negative heights) that
minimises ``evaluate_sequence``.

Algorithm
---------

1. Choose a set of candidate lengths (based on the previous best length)
   and a time budget.

2. For each length ``n``:
   * Target L1‑norm is fixed to ``sqrt(2·n)`` – this makes the official
     objective equal to the maximal self‑convolution value.
   * Randomly sample parametrised families of distributions:
       – Beta(α,β) PDFs discretised on the grid {0,…,n‑1}
       – Mixture of a Beta PDF with the uniform distribution (mixing weight *p*).
     α,β are drawn log‑uniformly in a wide range, *p* uniformly.
     The best few samples are kept.
   * From the top samples launch a mirror‑descent / exponentiated‑gradient
     optimiser (entropy mirror map) that directly minimises the smoothed
     max‑convolution (soft‑max with a growing temperature, later the exact
     sub‑gradient of the max).  The optimiser works on the *fixed* L1‐norm
     simplex.
   * The best sequence found for the current length is compared with the
     global best (using the official ``evaluate_sequence`` for the final
     check).

3. After all lengths have been processed, a cheap hill‑climbing
   refinement moves a small fraction of mass between two coordinates.

All heavy computations (convolution, gradients) use an FFT of a size that
is a power‑of‑two, which makes the whole search fast enough to stay well
within the 1000 s budget.

The code returns a plain Python list of heights (non‑negative, sum ≈
sqrt(2·n)).
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _cheap_objective(seq: np.ndarray) -> float:
    """Fast surrogate for ``evaluate_sequence`` (no clipping, assumes safe values)."""
    n = len(seq)
    if np.sum(seq) < 0.01:
        return float('inf')
    conv = _fft_self_convolve(seq)
    max_b = np.max(conv)
    s = np.sum(seq)
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the (hard) max at index k: 2·a_{k‑i} for admissible i."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray, b: np.ndarray,
                     alpha: float, fft_size: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b_k) w.r.t. a.
    Returns 2·conv(w, rev(a)) where w are the soft‑max weights.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=fft_size)
    A_rev = np.fft.rfft(a_rev, n=fft_size)
    conv = np.fft.irfft(W * A_rev, n=fft_size)[: 2 * a.shape[0] - 1]
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

def _exponential_optimize(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser on the simplex.
    The L1‑norm is kept equal to sqrt(2·n) (target_sum) throughout.
    Returns the best sequence seen and its *internal* score (2·n·max_b).
    """
    target_sum = math.sqrt(2.0 * n)
    a = a0.copy()
    # enforce exact target sum
    a = a * (target_sum / a.sum())

    best_seq = a.copy()
    best_score = _cheap_objective(a)

    # --- hyper‑parameters -------------------------------------------------
    eta0 = 0.8               # initial learning‑rate factor
    alpha = 1.0              # temperature for soft‑max
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0       # switch to hard max‑subgradient beyond this
    fft_size = 1 << ((3 * n - 2).bit_length())
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] exp‑opt n={n} it={it} score={best_score:.6f}")

        # gradient
        if alpha > alpha_thr:
            # hard max sub‑gradient
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            # smooth soft‑max gradient
            grad = _gradient_softmax(a, b, alpha, fft_size)

        # step‑size (diminishing)
        eta = eta0 / math.sqrt(it)

        # exponentiated‑gradient update (entropy mirror map)
        a = a * np.exp(-eta * grad)

        # protect against under‑flow / negativity
        a = np.maximum(a, 0.0)

        # renormalise to target L1‑norm
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, target_sum / n, dtype=np.float64)
        else:
            a = a * (target_sum / s)

        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5 * target_sum
            a = np.maximum(a, 0.0)
            a = a * (target_sum / a.sum())

    return best_seq, best_score

def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: move a small fraction of mass from one coordinate
    to another.  Stops when time runs out.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _cheap_objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        score = _cheap_objective(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score

def _beta_pdf_grid(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta(α,β) density on the centred grid
    x_i = (i+0.5)/n, i=0,…,n‑1.  Returns a non‑negative vector (not normalised).
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # Guard against overflow/underflow by working in log‑space when necessary
    # but for the ranges we use plain powers are safe.
    pdf = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # Replace NaNs (possible when alpha<1 and x==0) by a large finite number
    pdf = np.where(np.isnan(pdf), 0.0, pdf)
    return pdf

        # Allocate a slice of the remaining budget to this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(8.0, remaining * 0.35)

        target_sum = math.sqrt(2.0 * n)   # we keep L1‑norm = sqrt(2n)

        # ------------------------------------------------------------------
        #   A. Random search in a parametrised family (Beta + uniform mix)
        # ------------------------------------------------------------------
        candidates = []                # store a few promising vectors
        # Seed with pure arcsine (Beta(0.5,0.5)) and uniform
        arcsine_seq = _beta_pdf_grid(n, 0.5, 0.5)
        arcsine_seq = arcsine_seq / arcsine_seq.sum() * target_sum
        uniform_seq = np.full(n, target_sum / n, dtype=np.float64)
        candidates.append(arcsine_seq)
        candidates.append(uniform_seq)

        # Random sampling loop (time‑bounded)
        random_iter = 0
        while time.time() < slice_deadline and random_iter < 5000:
            random_iter += 1

            # log‑uniform draw for α,β in [0.1, 5]
            log_alpha = np.random.uniform(-1.0, math.log10(5.0))
            log_beta  = np.random.uniform(-1.0, math.log10(5.0))
            alpha = 10 ** log_alpha
            beta  = 10 ** log_beta

            # mixing weight between Beta and uniform
            p = np.random.rand()

            beta_vals = _beta_pdf_grid(n, alpha, beta)
            mix = (1.0 - p) * beta_vals + p * np.ones_like(beta_vals)
            mix = mix / mix.sum() * target_sum

            score = _cheap_objective(mix)

            if score < overall_best_score:
                overall_best_score = score
                overall_best_seq = mix.copy()
                print(f"[{time.time():.2f}] n={n} NEW GLOBAL (random) "
                      f"score={overall_best_score:.6f} α={alpha:.3f} β={beta:.3f} p={p:.3f}")

            # keep a rolling set of the best ~30 candidates
            candidates.append(mix)
            if len(candidates) > 30:
                # prune by cheap score
                cand_scores = [( _cheap_objective(c), c) for c in candidates]
                cand_scores.sort(key=lambda x: x[0])
                candidates = [c for _, c in cand_scores[:30]]

        # ------------------------------------------------------------------
        #   B. Pick the top few candidates and run the exponential optimiser
        # ------------------------------------------------------------------
        cand_scores = [( _cheap_objective(c), c) for c in candidates]
        cand_scores.sort(key=lambda x: x[0])
        top_candidates = [c for _, c in cand_scores[:3]]

        for init_seq in top_candidates:
            if time.time() >= deadline - 1.0:
                break
            seq_opt, int_score = _exponential_optimize(init_seq, n, slice_deadline)
            # final official evaluation (more reliable, includes clipping)
            try:
                final_score = evaluate_sequence(seq_opt.tolist())
            except Exception:
                final_score = int_score   # fallback

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq_opt.copy()
                print(f"[{time.time():.2f}] n={n} EXP‑OPT IMPROVED "
                      f"score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break
        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    #   C. Final local refinement (if time permits)
    # ------------------------------------------------------------------
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n = len(overall_best_seq)
        seq_ref, _ = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq_ref.tolist())
        except Exception:
            final_score = _cheap_objective(seq_ref)
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq_ref.copy()
            print(f"[{time.time():.2f}] AFTER REFINEMENT best score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    #   D. Safety net
    # ------------------------------------------------------------------
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
