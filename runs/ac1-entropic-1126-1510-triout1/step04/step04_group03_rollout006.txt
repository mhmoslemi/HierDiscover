`evaluate_sequence`.

Algorithm
---------
1.  Choose several candidate lengths around the previously known best length.
2.  For each length generate a small library of parametric shapes:
        * Symmetric Beta(α,α) (α ranging from 0.3 to 2.0)
        * Sine‑power   a_i ∝ sin(π(i+0.5)/n)^p   (p ranging from 0.5 to 3)
        * Simple triangular shape.
        * Uniform / random.
3.  Every shape is normalised to sum 1 (the objective is scale‑invariant).
4.  Starting from each shape run a *projected Adam* optimiser on the
    simplex (log‑softmax parametrisation).  The loss is the maximal entry
    of the self‑convolution.  Gradients are obtained from the exact

7.  The globally best sequence found (by the official `evaluate_sequence`)

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on a discrete grid of size n."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
    # protect against overflow for very small alpha
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
def _sine_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π(i+0.5)/n)^p, symmetric and positive."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(angles) ** p
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Symmetric linear ramp up to the centre and down again."""
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    a = np.where(i <= half, i + 1.0, n - i)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
def _subgradient_of_max(conv: np.ndarray, n: int) -> np.ndarray:
    """
    Returns the sub‑gradient of max(conv) w.r.t. the original vector a.
    The gradient is 2 * (w * a_rev) where w is the uniform weight over all
    active (maximal) indices of the convolution.
    """
    max_val = conv.max()
    mask = np.abs(conv - max_val) < 1e-12
    w = mask.astype(float)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    w = w / w_sum

    # convolution w * a_rev (a_rev = a[::-1])
    # size needed: len(w)+len(a_rev)-1 = (2n-1)+(n)-1 = 3n-2
    size_fft = 1 << ((3 * n - 2).bit_length())
    # a_rev will be supplied by the caller; we only need to compute once per iteration
    # Here we just return w and size_fft; the caller will finish the convolution
    return w, size_fft


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iter: int = 500,
                   lr: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex using a log‑softmax parametrisation.
    Returns the best vector found and its internal score 2·n·max(conv).
    """
    # initialise logits (log‑softmax)
    eps_log = 1e-30
    z = np.log(a0 + eps_log)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = None
    best_score = float("inf")

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # obtain the probability vector from logits
        a = np.exp(z)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # convolution and current score
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} iter={it} score={best_score:.6f}")

        # sub‑gradient of max(conv) w.r.t a
        w, size_fft = _subgradient_of_max(conv, n)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        tmp = np.fft.irfft(W * A_rev, n=size_fft)
        start = n - 1
        grad_a = 2.0 * tmp[start:start + n]

        # centre the gradient (adds invariance to a constant shift of logits)
        grad_centered = grad_a - grad_a.mean()

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_centered
        v = beta2 * v + (1.0 - beta2) * (grad_centered ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        z = z - step

        # tiny jitter to escape plateaus
        if it % 200 == 0:
            z += np.random.randn(n) * 1e-4

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _fft_self_convolve(best_a).max()
    return best_a, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: repeatedly move a random
    fraction of mass from one index to another and keep the move if the
    max‑convolution improves.  Stops when the deadline is reached.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] <= 1e-12:
            continue
        frac = np.random.rand() * 0.30
        delta = best_seq[i] * frac
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        score = 2.0 * n * _fft_self_convolve(cand).max()
        if score < best_score:
            best_score = score
            best_seq = cand

        if it % 5000 == 0:
            jitter = best_seq + np.random.randn(n) * 1e-6
            jitter = _project_onto_simplex(jitter, s=1.0)
            score = 2.0 * n * _fft_self_convolve(jitter).max()
            if score < best_score:
                best_score = score
                best_seq = jitter

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point used by the grader.  It searches for a step‑function
    (list of non‑negative heights) with a low `evaluate_sequence` score.
    """
    np.random.seed(seed)
    # safety margin so that printing etc. does not eat into the budget
    deadline = time.time() + budget_s - 4.0

    # ------------------------------------------------------------------
    # length selection
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # we explore a modest set of lengths (no more than 25 000 items)
    length_factors = (0.6, 0.85, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(300, int(base_n * f))
        for f in length_factors
        if max(300, int(base_n * f)) <= 25000
    })

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # give each length roughly 20 % of the remaining time
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(0.1, remaining * 0.20)

        # ------------------------------------------------------------------
        # generate a small library of parametric initial shapes
        initial = []

        # Beta family (symmetric)
        for a in (0.3, 0.4, 0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0):
            try:
                init = _beta_distribution(n, a)
                initial.append((f"beta_{a}", init))
            except Exception:
                pass

        # Sine‑power family
        for p in (0.5, 0.75, 1.0, 1.2, 1.5, 2.0, 3.0):
            try:
                init = _sine_power_distribution(n, p)
                initial.append((f"sine_{p}", init))
            except Exception:
                pass

        # deterministic shapes
        initial.append(("triangular", _triangular_distribution(n)))
        init = np.full(n, 1.0 / n, dtype=np.float64)
        initial.append(("uniform", init))
        init = np.random.rand(n)
        init /= init.sum()
        initial.append(("random", init))

        # ------------------------------------------------------------------
        for tag, a0 in initial:
            if time.time() >= slice_deadline:
                break

            # ---- short Adam optimisation ---------------------------------
            max_iter = 300                     # cheap but enough to move
            seq_opt, _ = _adam_optimize(a0, n, slice_deadline,
                                        max_iter=max_iter, lr=0.2)

            # ---- (optional) local search --------------------------------
            time_left = slice_deadline - time.time()
            if time_left > 0.5:
                seq_opt, _ = _local_search(seq_opt, n,
                                           time.time() + min(2.0, time_left))

            # ---- official evaluation ------------------------------------
            try:
                score = evaluate_sequence(seq_opt.tolist())
            except Exception:
                # fallback to internal estimate (identical mathematically)
                score = 2.0 * n * _fft_self_convolve(seq_opt).max()

            if score < global_best_score:
                global_best_score = score
                global_best_seq = seq_opt.copy()
                print(f"*** New global best via {tag} (n={n}) → score={global_best_score:.6f}")

            # early exit if we already pass the target
            if global_best_score <= 1.503:
                break

        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # final refinement on the currently best sequence (if time permits)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _local_search(global_best_seq,
                                       len(global_best_seq),
                                       deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
