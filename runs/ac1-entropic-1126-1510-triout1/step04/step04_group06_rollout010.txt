maximum entry of the self‑convolution a∗a.  The optimiser works in three
phases:

1. **Exploration** – a fast sweep of analytic families (exponential,
   power‑law, double‑ended, random block, etc.) and a projection onto the
   simplex {a ≥ 0, Σa = √(2 n)}.  The best few raw candidates are kept.

2. **Deep refinement** – the best candidate is improved by a long
   Adam‑based descent on a smooth approximation of the max‑convolution.
   The optimiser runs until almost the whole time budget is exhausted.

3. **Linear‑programming boost** – a few iterations of the classic
   “LP‑step” from the literature are performed.  For the current a,
   we solve the LP

        maximise   Σ g_j
        subject to (a * g)_k ≤ M   ∀k ,   g ≥ 0

   where M = max(a∗a).  The solution g₀ is rescaled to the required
   normalisation and mixed with a (convex combination).  This step
   often yields a noticeable drop in the peak value.

The programme respects the 1000 s budget, projects onto the simplex
whenever needed and returns the best sequence found.
"""

import math
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution using FFT (output length len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear self‑convolution a*a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Score used during the search (exact value of the evaluation
    function when Σa = √(2 n))."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ w_k·(a*a)_k w.r.t. a.  The weights w sum to 1.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution (soft‑max).
    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = target_sum * 0.5 / n

    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # soft‑max weights approximating the max
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-12)
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        grad = _weighted_gradient(a, w, size_fft)

        # Adam step
        eta = eta0 / (1.0 + it * 1e-5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9997
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _solve_lp_for_g(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   Σ g_j
        subject to (a * g)_k ≤ M   for all k,
                    g_j ≥ 0 .
    Returns the optimal g (as a NumPy vector) or None if the solver fails.
    """
    # Small epsilon to avoid zero columns (otherwise the problem is unbounded)
    eps = 1e-12
    a_pos = np.maximum(a, eps)

    n = a_pos.size
    conv_len = 2 * n - 1

    # Build the constraint matrix A (shape conv_len × n) where
    #   A[k, j] = a_pos[k - j]   if 0 ≤ k - j < n, else 0.
    A = np.zeros((conv_len, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a_pos

    b = np.full(conv_len, M, dtype=np.float64)

    # maximise Σ g_j  → minimise –Σ g_j
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None) for _ in range(n)]

    # Solve with HiGHS (fast default interior‑point method)
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising the self‑convolution
    peak.  The function respects the 1000 s limit and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 2500
        have_ref = False

    target_sum = math.sqrt(2.0 * n)
    size_fft = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # 1. PARAMETRIC EXPLORATION
    best_raw: list[tuple[float, np.ndarray]] = []

    def consider(score: float, arr: np.ndarray) -> None:
        """Maintain a sorted list of the top‑K raw candidates."""
        K = 20
        if len(best_raw) < K:
            best_raw.append((score, arr.copy()))
            best_raw.sort(key=lambda x: x[0])
        else:
            if score < best_raw[-1][0]:
                best_raw[-1] = (score, arr.copy())
                best_raw.sort(key=lambda x: x[0])

    idx = np.arange(n)

    def add_candidate(arr: np.ndarray) -> None:
        arr = _project_to_simplex(arr, target_sum)
        sc = quick_score(arr, n, size_fft)
        consider(sc, arr)

    # (a) Uniform
    add_candidate(np.ones(n))

    # (b) Single‑ended exponential decay
    for _ in range(150):
        decay = np.random.rand() * 0.015 + 0.001
        a = np.exp(-decay * idx)
        add_candidate(a)

    # (c) Double‑ended exponential (possibly asymmetric)
    for _ in range(150):
        decay_l = np.random.rand() * 0.015 + 0.001
        decay_r = np.random.rand() * 0.015 + 0.001
        a = np.exp(-decay_l * idx) + np.exp(-decay_r * (n - 1 - idx))
        add_candidate(a)

    # (d) Power‑law
    for _ in range(150):
        beta = np.random.rand() * 8.0 + 2.0
        pexp = np.random.rand() * 1.8 + 0.2
        a = 1.0 / (idx + beta) ** pexp
        add_candidate(a)

    # (e) Inverse‑sqrt (double‑ended)
    for _ in range(150):
        beta = np.random.rand() * 12.0
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        add_candidate(a)

    # (f) Random block structures (few constant blocks)
    for _ in range(400):
        nb = np.random.randint(2, 9)               # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        add_candidate(cand)

    # (g) Binary‑type (m ones)
    m = int(round(target_sum))
    cand = np.zeros(n, dtype=float)
    ones_pos = np.random.choice(n, m, replace=False)
    cand[ones_pos] = 1.0
    add_candidate(cand)

    # reference sequence, if available
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        add_candidate(a_ref)

    # ------------------------------------------------------------------
    # pick the best few raw candidates
    best_raw.sort(key=lambda x: x[0])
    top_candidates = best_raw[:5]
    print(f"[exploration] best raw score = {top_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. DEEP ADAM REFINEMENT ON THE CURRENT BEST
    best_a = top_candidates[0][1].copy()
    best_score = top_candidates[0][0]

    # allocate the majority of the budget to Adam (≈80 % of total time)
    adam_deadline = start_time + 0.78 * budget_s
    if adam_deadline > deadline - 5:
        adam_deadline = deadline - 5

    refined_a, refined_score = _refine_candidate_adam(
        best_a,
        adam_deadline,
        n,
        size_fft,
        target_sum,
        max_iters=10_000_000,          # effectively bounded by the deadline
    )
    if refined_score < best_score:
        best_score = refined_score
        best_a = refined_a.copy()
        print(f"[adam] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. LP‑BOOST PHASE (few iterations, remaining time)
    while time.time() < deadline - 2.0:
        conv = _self_convolution(best_a, size_fft)
        M = float(conv.max())

        # solve the LP
        g = _solve_lp_for_g(best_a, M)
        if g is None:
            break

        # rescale g to the required normalisation
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g * (target_sum / sum_g)

        # convex combination (t can be tuned)
        t = 0.3
        new_a = (1 - t) * best_a + t * g
        new_a = _project_to_simplex(new_a, target_sum)

        new_score = quick_score(new_a, n, size_fft)
        if new_score < best_score - 1e-6:
            best_score = new_score
            best_a = new_a.copy()
            print(f"[lp] improved to {best_score:.6f}")

        # occasional random kicks to escape possible stalls
        if np.random.rand() < 0.15:
            noise = np.random.randn(n) * target_sum * 0.001
            cand = best_a + noise
            cand = _project_to_simplex(cand, target_sum)
            sc = quick_score(cand, n, size_fft)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
                print(f"[perturb] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
