def _subgradient_descent(a0, S, size, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _max_convolution(a, size)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        # compute active mask
        mask = np.isclose(conv, max_c, atol=1e-12)
        # compute w
        w = np.zeros_like(conv)
        cnt = mask.sum()
        if cnt > 0:
            w[mask] = 1.0 / cnt
        # gradient
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a_rev, n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[:2*len(a)-1]
        grad = 2.0 * conv_wr[len(a)-1 : len(a)-1 + len(a)]
        # descent direction
        direction = -grad
        # line search
        step = max(1e-5, np.mean(a) * 0.1)  # guess
        improved = False
        cur_val = max_c
        while step > 1e-12:
            a_new = a + step * direction
            a_new = _project_to_simplex(a_new, S)
            new_val = _max_convolution(a_new, size)

"""
Multiple‑restart projected sub‑gradient descent with line‑search
followed by a short L‑BFGS polishing phase.

* The search works directly on the probability simplex
  Σa_i = √(2·n) (the objective then equals the maximum
  entry of a∗a).  Projection onto the simplex is performed by
  the Euclidean routine `_project_to_simplex`.

* Convolutions and sub‑gradients are computed with FFTs
  (`_full_convolution`, `_gradient_max_convolution`) – this makes
  each iteration O(n log n) even for n≈3000.

* Several simple initial shapes are tried (uniform, random Dirichlet,
  a “V‑shape” that concentrates mass at the ends, a two‑block edge
  pattern).  For each start a projected sub‑gradient descent with
  back‑tracking line search is run for a short, fixed budget.
  The best sequence seen over all restarts is kept.

* After the restart loop a final L‑BFGS optimisation on a smooth
  ‑max surrogate (`_SmoothMaxObjective`) is performed to polish the
  result.

The whole routine respects the 1000 s time limit and returns a
list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for all FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0                     # numerical cleanup
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


def _gradient_max_convolution(a: np.ndarray, conv: np.ndarray, size: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    `conv` must be the full convolution a*a (length 2·n‑1).
    """
    max_c = conv.max()
    mask = np.isclose(conv, max_c, atol=1e-12)
    cnt = int(mask.sum())
    w = np.zeros_like(conv)
    if cnt > 0:
        w[mask] = 1.0 / cnt

    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * len(a) - 1]
    g_a = 2.0 * conv_wr[len(a) - 1 : len(a) - 1 + len(a)]
    return g_a.astype(np.float64)


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max objective (soft‑max surrogate) and its gradient w.r.t. y.
    The variables are y = log(p) where p lives on the simplex.
    """
    p = _softmax(y)                # Σp = 1, p_i ≥ 0
    a = S * p                      # Σa = S

    conv = np.convolve(a, a)       # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g

def _subgradient_refine(a_start: np.ndarray, S: float,
                        size: int, deadline: float):
    """
    Projected sub‑gradient descent on the true hard‑max objective.
    Uses a back‑tracking line search and occasional random perturbation
    to escape shallow local minima.
    Returns the best sequence and its max‑convolution value.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_val = _max_convolution(best_a, size)

    it = 0
    max_iters = 5000
    min_step = 1e-12
    while time.time() < deadline and it < max_iters:
        # full convolution and active set
        conv = _full_convolution(a, size)
        max_c = conv.max()

        # gradient of the hard max
        g_a = _gradient_max_convolution(a, conv, size)

        # descent direction
        direction = -g_a

        # back‑tracking line search
        step = max(1e-5, np.mean(a) * 0.2)
        improved = False
        cur_val = max_c

        while step > min_step:
            a_candidate = a + step * direction
            a_candidate = _project_to_simplex(a_candidate, S)
            cand_val = _max_convolution(a_candidate, size)
            if cand_val < cur_val - 1e-14:
                a = a_candidate
                cur_val = cand_val
                improved = True
                if cand_val < best_val - 1e-14:
                    best_val = cand_val
                    best_a = a.copy()
                    print(f"[subgrad improve] iter {it:,} max = {best_val:.6f}")
                break
            step *= 0.5

        if not improved:
            # small stochastic nudge to shake out of plateaus
            noise = np.random.randn(len(a)) * 0.02 * np.mean(a)
            a = _project_to_simplex(a + noise, S)

    return best_a, best_val


def _l_bfgs_refine(a_start: np.ndarray, S: float,

    """

    """
