def beta_distribution(n, alpha):
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x**(alpha-1)) * ((1-x)**(alpha-1))
    # handle alpha<1 may produce large values at edges; ensure no overflow
    # clip minimal small before computing maybe add epsilon.
    vals[vals == 0] = np.finfo(np.float64).tiny

def power_law_distribution(n, p):
    i = np.arange(n, dtype=np.float64) + 0.5
    # symmetric shape: weight = (i)^(-p) + (n+1-i)^(-p)
    w = (i)**(-p) + (n+1-i)**(-p)
    w /= w.sum()
    return w

def exponential_distribution(n, lam):
    i = np.arange(n, dtype=np.float64)
    center = (n-1)/2
    w = np.exp(-lam * np.abs(i - center))
    w /= w.sum()
    return w

"""
search for a low‑score step‑function.

The algorithm works in three stages:

1.  **Broad parametric sweep** – generate many candidate probability vectors
    from a handful of one‑parameter families (Uniform, arcsine, Beta,
    power‑law, exponential and mixtures with a uniform component).  Each
    candidate is normalised to sum 1 and its *fast* score
    `2·n·max(conv(a,a))` is evaluated via an FFT‑based convolution.
    The best few candidates are kept.

2.  **Intensive local optimisation** – the best candidate from stage 1 is
    refined with a two‑phase gradient descent:
      * Adam optimiser on a smooth log‑sum‑exp surrogate of the peak
        of the self‑convolution (temperature slowly annealed up to a very
        large value, i.e. the surrogate approaches the true max).
      * A short projected sub‑gradient descent on the exact ℓ∞‑objective
        to polish the solution.

    Several random jitter restarts are performed while time permits.

3.  **Final evaluation** – the best refined sequence seen so far is
    evaluated with the black‑box `evaluate_sequence`.  The resulting list
    of non‑negative heights is returned.

The whole search respects the supplied time budget (`budget_s`, default
1000 s) and aborts gracefully before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k).

    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    init_alpha: float = 1.0,
    alpha_factor: float = 1.005,
    alpha_max: float = 20000.0,
    lr0: float = 0.25,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best iterate encountered and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny noise to escape plateaus
        if t % 400 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the true ℓ∞‑objective.
    Uniform weight on the active set (all indices attaining the current max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short refinement using the exact ℓ∞‑objective.
    Diminishing step‑size sub‑gradient descent with projection onto the simplex.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.015 / np.sqrt(it)          # slowly vanishing
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 300 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 3e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(0.5,0.5)) distribution of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) distribution (discretised on mid‑points)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    log_vals = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    vals = np.exp(log_vals)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ i^{‑p} + (n‑i+1)^{‑p}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = (i) ** (-p) + (n + 1 - i) ** (-p)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _exponential_distribution(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(‑λ·|i‑center|)."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    w = np.exp(-lam * np.abs(i - center))
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
def _mix_with_uniform(base: np.ndarray, eps: float) -> np.ndarray:
    """Convex mixture (1‑ε)·base + ε·uniform, renormalised."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    seq = (1.0 - eps) * base + eps * uniform
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score step‑function.
    Returns a plain Python list that will be fed to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # base sequence (if it exists) – we will reuse it as a further candidate
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 4000

    # ------------------------------------------------------------------
    # candidate lengths – modest range around the historic size, plus a larger one
    length_factors = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]
    lengths = sorted({max(500, int(base_n * f)) for f in length_factors})
    # also allow a single big length (up to ~15000) if time permits
    if time.time() < deadline - 300:
        lengths.append(min(15000, int(base_n * 2.5)))

    # ------------------------------------------------------------------
    # quick parametric sweep (stage 1)
    sweep_deadline = time.time() + min(30.0, (deadline - start_time) * 0.07)
    candidates = []

    betas = [0.30, 0.35, 0.40, 0.45, 0.50, 0.55, 0.60, 0.70]
    powers = [0.5, 1.0, 1.5, 2.0, 3.0]
    lambdas = [0.001, 0.003, 0.006, 0.01, 0.02, 0.04]
    mix_eps = [0.02, 0.05, 0.10, 0.20]

    for n in lengths:
        if time.time() >= sweep_deadline:
            break

        # uniform baseline
        candidates.append((n, np.full(n, 1.0 / n, dtype=np.float64)))

        # arcsine
        candidates.append((n, _arcsine_distribution(n)))

        # beta family
        for a in betas:
            if time.time() >= sweep_deadline:
                break
            candidates.append((n, _beta_distribution(n, a)))

        # power‑law
        for p in powers:
            if time.time() >= sweep_deadline:
                break
            candidates.append((n, _power_law_distribution(n, p)))

        # exponential
        for lam in lambdas:
            if time.time() >= sweep_deadline:
                break

        # mixtures of arcsine with uniform
        arc = _arcsine_distribution(n)
        for eps in mix_eps:

        # interpolate previous best if available

    # ------------------------------------------------------------------

        seq /= s

        fast_score = 2.0 * n * conv.max()
        scored.append((fast_score, n, seq))
