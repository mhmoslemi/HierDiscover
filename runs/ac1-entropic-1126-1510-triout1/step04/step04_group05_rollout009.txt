"""
Improved search for the minimal‑max self‑convolution step function.

The algorithm consists of three stages:

1️⃣  Quick simulated‑annealing (SA) to break the uniform symmetry and obtain
    a non‑uniform feasible point (∑a = √(2·n), a ≥ 0).

2️⃣  Iterated linear‑programming refinement.
    For the current vector f we solve
        minimise   t
        s.t.       (f * g)[k] ≤ t      for all k
                   ∑ g = √(2·n) , g ≥ 0 .
    This LP finds a distribution *g* that makes the (f‑g) convolution as flat
    as possible.  We then move from f towards g (full or mixed) and repeat.
    The LP is built with a sparse Toeplitz matrix and solved by HiGHS.

3️⃣  Final projected sub‑gradient polish on the true hard‑max objective.

All helpers are top‑level functions; the search runs for the allotted
`budget_s` seconds and returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
#  Utility: projection onto the simplex {x ≥ 0, Σx = z}
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex Σx = z, x ≥ 0."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Convolution helpers (FFT based for speed)
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0                     # clip tiny negatives
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


# ----------------------------------------------------------------------
#  Simple simulated‑annealing on the simplex
def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap SA that moves a random fraction of mass between two indices."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    # temperature schedule (linear cooling)
    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-6
    it = 0
    while time.time() - start_time < budget:
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * a[i] * 0.3   # move up to 30 % of a[i]
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = diff < 0.0 or (T > 0.0 and np.random.rand() < math.exp(-diff / T))
        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} max = {best_score:.6f}")
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
#  Projected sub‑gradient polishing (hard max)
def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    eta0 = S * 0.5 / n           # rough step size
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 5000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        if cnt:
            w[mask] = 1.0 / cnt

        # sub‑gradient of the hard max
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
#  Linear‑programming refinement (core of this version)
def _lp_refine(a_start: np.ndarray, S: float,
               budget: float, start_time: float,
               best_score: float, best_a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Iteratively solve the LP:

        minimise   t
        s.t.       (f * g)[k] ≤ t   ∀k
                   ∑ g = S , g ≥ 0

    and move the current vector `f` towards the LP solution `g`.
    """
    n = len(a_start)
    size = _fft_size(n)

    # Pre‑compute the index pattern for the convolution constraints.
    # Row index = i + j , column = i (for coefficient f_i)
    row_idx = np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n)
    col_idx = np.repeat(np.arange(n, dtype=np.int32), n)          # column = i
    # column for the extra variable t (last column)
    row_t = np.arange(2 * n - 1, dtype=np.int32)
    col_t = np.full(2 * n - 1, n, dtype=np.int32)
    data_t = -np.ones(2 * n - 1, dtype=np.float64)

    # Equality: Σ g_i = S
    eq_data = np.concatenate([np.ones(n, dtype=np.float64), np.array([0.0])])
    eq_row = np.zeros(n + 1, dtype=np.int32)
    eq_col = np.arange(n + 1, dtype=np.int32)
    A_eq = sp.csr_matrix((eq_data, (eq_row, eq_col)), shape=(1, n + 1))
    b_eq = np.array([S])

    # Objective: minimise t
    c = np.concatenate([np.zeros(n, dtype=np.float64), np.array([1.0])])
    bounds = [(0, None)] * (n + 1)        # g_i ≥ 0 , t ≥ 0

    f = a_start.copy()
    curr_max = _max_convolution(f, size)

    it = 0
    while time.time() - start_time < budget and it < 100:
        # Build the sparse matrix A_ub = [A_f   -1]
        data_f = np.repeat(f, n)                     # f_i repeated n times
        A_f = sp.csr_matrix((data_f, (row_idx, col_idx)), shape=(2 * n - 1, n))
        A_t = sp.csr_matrix((data_t, (row_t, col_t)), shape=(2 * n - 1, n + 1))
        A_ub = sp.hstack([A_f, A_t], format="csr")
        b_ub = np.zeros(2 * n - 1, dtype=np.float64)

        remaining = max(0.0, (budget + start_time) - time.time())
        try:
            res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                              A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds,
                              method="highs",
                              options={"presolve": True,
                                       "time_limit": remaining})
        except Exception as exc:
            print(f"[LP] exception: {exc}")
            break

        if res.status != 0:
            print(f"[LP] solver status {res.status}: {res.message}")
            break

        g = res.x[:n]
        t_opt = res.x[-1]

        # If the LP cannot improve the current max we stop.
        if t_opt >= curr_max - 1e-9:
            print(f"[LP] it {it}: t_opt ({t_opt:.6f}) ≥ current max ({curr_max:.6f}) → stop")
            break

        # Clip tiny negatives that may appear due to numerical noise.
        g = np.maximum(g, 0.0)

        # ------------------------------------------------------------------
        #  Try different ways to obtain a better vector.
        # ------------------------------------------------------------------
        improved = False

        # (a) Directly adopt g if its own self‑convolution is lower.
        g_max = _max_convolution(g, size)
        if g_max < curr_max - 1e-12:
            f = g
            curr_max = g_max
            improved = True
            print(f"[LP] it {it}: take g directly, new max = {curr_max:.6f}")

        else:
            # (b) Mix f and g with a few trial step‑sizes.
            for lam in (0.2, 0.5, 0.8):
                cand = (1.0 - lam) * f + lam * g
                cand_max = _max_convolution(cand, size)
                if cand_max < curr_max - 1e-12:
                    f = cand
                    curr_max = cand_max
                    improved = True
                    print(f"[LP] it {it}: mix λ={lam:.2f}, new max = {curr_max:.6f}")
                    break

        if improved:
            # evaluate the true score (should equal curr_max when ∑a = √(2n))
            sc = evaluate_sequence(f.tolist())
            if sc < best_score:
                best_score = sc
                best_a = f.copy()
                print(f"[LP] global best improved → {best_score:.6f}")
        else:
            # no improvement from mixing – terminate early
            print(f"[LP] it {it}: mixing failed → stop")
            break

        it += 1

    return f, best_score, best_a
