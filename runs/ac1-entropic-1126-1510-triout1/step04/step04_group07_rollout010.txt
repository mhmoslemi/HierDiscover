        max_c = conv.max()

        # subgradient:
        mask = conv == max_c
        cnt = mask.sum()

        # gradient of max w.r.t a

"""
Low‑score step‑function search.

The routine builds several initial candidates (uniform, exponentials,
random Dirichlet, reference) and refines each with a smooth‑max L‑BFGS
continuation.  The best result is then polished by a long projected
Adam‑style sub‑gradient descent on the true ℓ∞‑norm of the
autoconvolution, followed by a gradient‑guided pairwise mass‑transfer
phase, a cheap random‑block search and a monotone‑sorting heuristic.
All phases respect the supplied time budget (seconds) and the function
returns the best sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = z} (z ≥ 0).
    """
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate (value & gradient) for the ℓ∞‑norm of a*a.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained parameters (length n).
    mu : float
        Temperature (>0). Larger µ → smoother approximation.
    n : int
        Length of the step‑function.
    S : float
        Desired ℓ¹‑norm (Σa = S).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : np.ndarray
        Gradient ∂f/∂y (shape (n,)).
    """
    # probabilities on the simplex and actual heights
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    # full convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weights w_k ∝ exp((conv_k - max)/μ)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # prevent overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                               # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a  (g_a[i] = 2 Σ_k w_k a_{k‑i})
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize: returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)

    while time.time() < deadline:

    # --------------------------------------------------------------
