def _anneal_permutation(a, size_fft, best_score, deadline):
    n = len(a)
    current_a = a.copy()
    current_score = best_score
    temperature = best_score * 0.1  # initial temp
    # number of iterations based on remaining time
    while time.time() < deadline:
        # try swap
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        a_new = current_a.copy()
        a_new[i], a_new[j] = a_new[j], a_new[i]
        new_score = _max_convolution_fft(a_new, size_fft)
        delta = new_score - current_score
        if delta < 0 or np.random.rand() < np.exp(-delta/temperature):
            current_a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = current_a.copy()
        # cool down
        temperature *= 0.9999
        # occasional random perturbation: try value transfer
        if np.random.rand() < 0.001:
            # choose source i where a_i > 1e-6, target j
            src = np.random.choice(np.where(current_a > 1e-6)[0])
            dst = np.random.randint(n)
            delta_val = current_a[src] * np.random.rand() * 0.1  # up to 10% transfer
            a_temp = current_a.copy()
            a_temp[src] -= delta_val
            a_temp[dst] += delta_val
            # ensure nonnegative
            if a_temp[src] >= 0:
                # No projection needed because sum conserved
                new_score2 = _max_convolution_fft(a_temp, size_fft)
                delta2 = new_score2 - current_score
                if delta2 < 0 or np.random.rand() < np.exp(-delta2/temperature):
                    current_a = a_temp
                    current_score = new_score2
                    if new_score2 < best_score:
                        best_score = new_score2
                        best_a = current_a.copy()
        # break if time almost up
        if time.time() + 0.01 > deadline:
            break
    return best_a, best_score

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Determine n
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)
            have_ref=True
        except:
            n = 3000
            have_ref=False
    else:
        n = 3000
        have_ref=False

    target_sum = math.sqrt(2.0 * n)
    size_fft = _fft_size(2 * n - 1)

    # Helper compute score
    def score_of(a):
        return _max_convolution_fft(a, size_fft)

    # pool of candidates
    best_raw = []  # list of (score, array)
    K = 10

    # push function
    def push(sc, a):
        nonlocal best_raw
        if len(best_raw) < K:
            best_raw.append((sc, a.copy()))
            best_raw.sort(key=lambda x: x[0])
        elif sc < best_raw[-1][0]:
            best_raw[-1] = (sc, a.copy())
            best_raw.sort(key=lambda x: x[0])

    # add reference if present
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, target_sum)
        push(score_of(a_ref), a_ref)

    # uniform baseline
    a_uni = np.ones(n)
    a_uni = _project_to_simplex(a_uni, target_sum)
    push(score_of(a_uni), a_uni)

    # Add monotone decreasing shapes
    idx = np.arange(n)
    # exponential decays
    for _ in range(30):
        c = np.random.rand() * 0.02 + 0.001
        a = np.exp(-c * idx)
        a = _project_to_simplex(a, target_sum)
        push(score_of(a), a)

    # power law
    for _ in range(30):
        p = np.random.rand() * 3.0 + 0.2
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        push(score_of(a), a)

    # Gaussian
    for _ in range(20):
        sigma = np.random.rand() * 400 + 200   # sigma in [200,600]
        mu = np.random.rand() * (n-1)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        a = _project_to_simplex(a, target_sum)
        push(score_of(a), a)

    # Sine-shaped
    for _ in range(20):
        # use sin(pi*(i+1)/(n+1))^q
        q = np.random.rand() * 4 + 0.5   # exponent between 0.5 and 4.5
        a = np.sin(np.pi * (idx + 1) / (n + 1)) ** q
        a = _project_to_simplex(a, target_sum)
        push(score_of(a), a)

    # Random Dirichlet with varying concentration
    for alpha in [0.02,0.05,0.1,0.2,0.5,1.0,2.0]:
        for _ in range(8):
            probs = np.random.dirichlet(np.full(n, alpha))
            a = probs * target_sum
            push(score_of(a), a)

    # Random block structures (higher number of blocks)
    for _ in range(200):
        nb = np.random.randint(3, 12)  # 3-11 blocks
        cuts = np.sort(np.random.choice(n-1, nb-1, replace=False)+1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        push(score_of(cand), cand)

    print(f"[sampling] top raw score: {best_raw[0][0]:.6f}")

    # 2. refinement via subgradient Adam
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    # refine each of top candidates
    for rank, (sc, a0) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate iteration budget proportionally or limited
        max_iters = int(min(200_000, remaining * 2000))  # e.g., 2000 iterations per second
        print(f"[refine] candidate {rank+1}/{len(best_raw)} raw {sc:.6f}, max_iters {max_iters}")
        refined_a, refined_sc = _refine_subgrad(a0, size_fft, target_sum, deadline, max_iters)
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] new best after refine {global_best_score:.6f}")

    # 3. permutation annealing on best sequence
    print("[annealing] starting permutation annealing")
    # allocate remaining time for annealing
    annealing_deadline = deadline
    best_a, best_score = _anneal_permutation(global_best_a, size_fft, global_best_score, annealing_deadline)
    # update global
    global_best_a = best_a
    global_best_score = best_score
    print(f"[final] best conv peak = {global_best_score:.6f}")

    # Convert to list and return
    return global_best_a.tolist()

"""
Propose a non‑negative step‑function (sequence) that minimizes the
self‑convolution peak.  

Algorithm
---------
1. **Problem size** – use the length of ``height_sequence_1`` when it exists,
   otherwise fall back to ``n = 3000``.  All candidates are projected onto the
   simplex Σ a_i = √(2 n).

2. **Broad sampling** – generate many raw candidates from a variety of
   families (uniform, exponential, power‑law, Gaussian, sine, Dirichlet,
   random block‑wise constants and monotone shapes).  Keep the best ``K``
   candidates according to the exact convolution peak.

3. **Refinement** – for each of the top ``K`` candidates run a projected
   Adam‑style sub‑gradient descent on the *hard* L∞‑norm of the
   self‑convolution.  The sub‑gradient is the exact gradient at the worst
   index.  The step‑size decays like 1/√t and every iterate is projected back
   onto the simplex.  The best iterate of each run updates the global best.

4. **Permutation annealing** – with the remaining time perform a simulated
   annealing on the ordering of the values (swaps) together with occasional
   tiny mass‑transfer moves.  The temperature starts proportional to the
   current objective and is multiplied by a factor < 1 each step.  Any move
   that lowers the peak is always accepted; otherwise it is accepted with
   probability exp(−Δ/temperature).

5. Return the best sequence discovered before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a using FFT (length = 2 n‑1)."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution (no clipping)."""
    conv = _full_convolution_fft(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of   max_k (a*a)_k   at the current maximal index k_max.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        # overlapping part is a[0..k_max] with reversed a[k_max::-1]
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        # k_max ≥ n
        start = k_max - n + 1                # first i that participates
        length = 2 * n - 1 - k_max           # number of terms
        g[start:] = 2.0 * a[:length][::-1]

    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the L∞‑norm of the
    self‑convolution.  Returns the best (lowest peak) sequence and its
    convolution peak.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_fft(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning rate (approximately the magnitude of a single entry)
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and current worst index
        conv_full = _full_convolution_fft(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[refine] it {it:,} new best = {best_score:.6f}")

        # sub‑gradient at the worst index
        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # decaying step size
        eta = eta0 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection to the simplex (keeps Σa = target_sum)
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _anneal_permutation(
    a0: np.ndarray,
    size_fft: int,
    init_score: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing on the permutation of the values.
    Swaps (and occasional tiny mass‑transfer perturbations) are tried;
    improvements are kept, worsening moves are accepted with probability
    exp(-Δ/temperature).  Returns the best sequence encountered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = init_score
    cur_score = init_score

    n = a.shape[0]
    temperature = init_score * 0.1 + 1e-6   # small positive start value
    decay = 0.9999

    while time.time() < deadline:
        # ---- swap two positions -------------------------------------------------
        i, j = np.random.choice(n, 2, replace=False)
        a_swapped = a.copy()
        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
        new_score = _max_convolution_fft(a_swapped, size_fft)

        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / temperature):
            a = a_swapped
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[anneal] swap improved to {best_score:.6f}")

        # ---- tiny mass‑transfer move --------------------------------------------
        if np.random.rand() < 0.001:
            src_candidates = np.where(a > 1e-12)[0]
            if src_candidates.size > 0:
                src = np.random.choice(src_candidates)
                dst = np.random.randint(n)
                if src != dst:
                    delta_val = a[src] * (np.random.rand() * 0.2)   # up to 20 % transfer
                    a2 = a.copy()
                    a2[src] -= delta_val
                    a2[dst] += delta_val
                    # sum conserved, non‑negativity already guaranteed
                    new_score2 = _max_convolution_fft(a2, size_fft)
                    delta2 = new_score2 - cur_score
                    if delta2 < 0 or np.random.rand() < math.exp(-delta2 / temperature):
                        a = a2
                        cur_score = new_score2
                        if new_score2 < best_score:
                            best_score = new_score2
                            best_a = a.copy()
                            print(f"[anneal] perturb improved to {best_score:.6f}")

        # ---- temperature update -------------------------------------------------
        temperature *= decay
        # safety break
        if time.time() + 0.01 > deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimizes the
    self‑convolution peak while respecting a 1000 s budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ---- problem size ---------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)   # type: ignore[name-defined]
            have_ref = True
        except Exception:
            n = 3000
            have_ref = False
    else:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)                 # Σa = √(2 n)
    size_fft = _fft_size(2 * n - 1)                 # FFT size for full convolution

    # ---- helper to evaluate a candidate ---------------------------------------
    def _score(a: np.ndarray) -> float:
        return _max_convolution_fft(a, size_fft)

    # ---- keep the best K raw candidates --------------------------------------
    K = 10
    best_raw: list[tuple[float, np.ndarray]] = []   # (score, a)

    def _push(sc: float, a: np.ndarray) -> None:
        if len(best_raw) < K:
            best_raw.append((sc, a.copy()))
            best_raw.sort(key=lambda x: x[0])
        elif sc < best_raw[-1][0]:
            best_raw[-1] = (sc, a.copy())
            best_raw.sort(key=lambda x: x[0])

    # ---- reference (if any) ---------------------------------------------------
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        _push(_score(a_ref), a_ref)
        print(f"[init] reference score = {_score(a_ref):.6f}")

    # ---- simple baselines -----------------------------------------------------
    a_uni = np.ones(n)
    a_uni = _project_to_simplex(a_uni, target_sum)
    _push(_score(a_uni), a_uni)
    print(f"[init] uniform score = {_score(a_uni):.6f}")

    idx = np.arange(n)

    # ---- exponential decays ---------------------------------------------------
    for _ in range(40):
        c = np.random.rand() * 0.02 + 0.001
        a = np.exp(-c * idx)
        a = _project_to_simplex(a, target_sum)
        _push(_score(a), a)

    # ---- power‑law shapes -----------------------------------------------------
    for _ in range(40):
        p = np.random.rand() * 3.0 + 0.2
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        _push(_score(a), a)

    # ---- Gaussian bumps --------------------------------------------------------
    for _ in range(30):
        sigma = np.random.rand() * 400 + 200          # roughly 200‑600
        mu = np.random.rand() * (n - 1)
        a = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        a = _project_to_simplex(a, target_sum)
        _push(_score(a), a)

    # ---- sine‑shaped families -------------------------------------------------
    for _ in range(30):
        q = np.random.rand() * 4 + 0.5                # exponent for shaping
        a = np.sin(np.pi * (idx + 1) / (n + 1)) ** q
        a = _project_to_simplex(a, target_sum)
        _push(_score(a), a)

    # ---- Dirichlet draws -------------------------------------------------------
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(12):
            probs = np.random.dirichlet(np.full(n, alpha))
            a = probs * target_sum
            _push(_score(a), a)

    # ---- random block‑wise constants (more blocks) ----------------------------
    for _ in range(250):
        nb = np.random.randint(3, 12)                  # 3‑11 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        _push(_score(cand), cand)

    print(f"[sampling] best raw score = {best_raw[0][0]:.6f}")

    # ---- refinement of the top raw candidates ---------------------------------
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    for rank, (raw_sc, a0) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        # allocate a modest budget for each candidate
        remaining = deadline - time.time()
        max_iters = int(min(180_000, remaining * 1500))
        print(f"[refine] candidate #{rank+1} (raw {raw_sc:.6f}) max_iters={max_iters}")
        refined_a, refined_sc = _refine_subgrad(
            a0, size_fft, target_sum, deadline, max_iters
        )
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ---- permutation annealing on the current best ----------------------------
    print("[anneal] start permutation annealing")
    best_a, best_score = _anneal_permutation(
        global_best_a, size_fft, global_best_score, deadline
    )
    global_best_a = best_a
    global_best_score = best_score
    print(f"[final] best convolution peak = {global_best_score:.6f}")

    # ---- return the sequence ---------------------------------------------------
    return global_best_a.tolist()
