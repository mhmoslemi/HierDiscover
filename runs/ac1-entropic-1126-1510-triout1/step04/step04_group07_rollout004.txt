√(2·n).  Hence we minimise  max⁡(a*a) under the simplex constraint.
The search proceeds in four phases:

1.  Simple parametric families (uniform, decreasing exponential,
    power‑law) are scanned; the best shape is kept.
2.  A continuation method with a smoothed max (µ‑smooth) and L‑BFGS‑B
    refines the best candidate.
3.  Hard‑max sub‑gradient descent on the logits (soft‑max
    parametrisation) using an Adam optimiser further lowers the maximum.
4.  A cheap random‑block perturbation followed by a monotone sorting
    heuristic explores the neighbourhood.

All steps respect a user‑supplied time budget (default 1000 s) and use only
NumPy/SciPy.  The best feasible sequence found is returned as a plain
list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _hard_max_grad(y: np.ndarray, n: int, S: float):
    """
    Sub‑gradient of the hard max objective with respect to the logits y.

    Returns
    -------
    grad_y : np.ndarray          shape (n,)
        Gradient of max(a*a) where a = S*softmax(y).
    max_val : float
        Current value of max(a*a).
    """
    p = _softmax(y)              # Σp = 1, p ≥ 0
    a = S * p                    # Σa = S

    # full convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)
    max_val = conv.max()
    max_idx = np.where(conv == max_val)[0]      # could be several

    # sub‑gradient w.r.t. a : g_a[i] = 2 * Σ_{k∈max_idx} a_{k-i}
    g = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        # a_slice = a[k-i] for i = i_start … i_end, taken in reverse order
        a_slice = a[k - i_start : k - i_end - 1 : -1]
        g[i_start:i_end + 1] += a_slice
    g_a = 2.0 * g

    # chain rule through the softmax (logits y)
    inner = np.dot(g_a, p)               # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return grad_y.astype(np.float64), max_val


def _smooth_max_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max (log‑sum‑exp) surrogate and its gradient w.r.t. y.
    """
    p = _softmax(y)
    a = S * p
    conv = np.convolve(a, a)
    max_c = conv.max()
    diff = (conv - max_c) / mu
    diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
    exp_diff = np.exp(diff)
    Z = exp_diff.sum()
    w = exp_diff / Z                     # smooth weights (sum to 1)
    f_val = max_c + mu * math.log(Z)

    # gradient w.r.t. a
    # g_a = 2 * Σ_k w_k * a_{k-i}
    # we reuse the same O(n·|supp(w)|) routine as in the hard case
    g = np.zeros_like(a)
    max_idx = np.where(w > 0)[0]
    for k in max_idx:
        coeff = w[k]
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        a_slice = a[k - i_start : k - i_end - 1 : -1]
        g[i_start:i_end + 1] += coeff * a_slice
    g_a = 2.0 * g

    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for ``scipy.optimize.minimize`` – returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _smooth_max_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) with a low
    ``evaluate_sequence`` score.  The routine respects the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3500
        have_ref = False

    S = math.sqrt(2.0 * n)                  # target ℓ¹‑norm (makes score = max conv)

    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # ------------------------------------------------------------------
    # helper to update the incumbent
    def _maybe_update(a_vec: np.ndarray) -> None:
        nonlocal best_score, best_a
        # ``evaluate_sequence`` works with a plain list
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – parametric scan
    print("[phase 1] parametric scan")
    # uniform
    a_uni = np.full(n, 1.0, dtype=float)
    a_uni = a_uni / a_uni.sum() * S
    _maybe_update(a_uni)

    # decreasing exponential family
    for beta in np.linspace(0.0, 0.015, 31):
        if time.time() > deadline:
            break
        a = np.exp(-beta * np.arange(n, dtype=float))
        a = a / a.sum() * S
        _maybe_update(a)

    # power‑law family a_i = 1/(i+1)^p
    for p in np.linspace(0.0, 2.0, 21):
        if time.time() > deadline:
            break
        a = 1.0 / (np.arange(1, n + 1, dtype=float) ** p)
        a = a / a.sum() * S
        _maybe_update(a)

    # mixed exponential‑power shapes (coarse grid)
    for beta in np.linspace(0.001, 0.008, 5):
        for p in np.linspace(0.5, 1.5, 5):
            if time.time() > deadline:
                break
            a = np.exp(-beta * np.arange(n, dtype=float)) / (np.arange(1, n + 1) ** p)
            a = a / a.sum() * S
            _maybe_update(a)

    # reference if available
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        _maybe_update(a_ref)

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max continuation + L‑BFGS‑B
    print("[phase 2] smooth‑max L‑BFGS")
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)      # logits for the current best
        a_cur = S * _softmax(y0)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = max(conv_cur.max(), 1.0)    # start temperature
        mu_min = 1e-8
        mu_factor = 0.7
        maxiter_per_mu = 300

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y0,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break
            y0 = res.x
            a_cur = S * _softmax(y0)
            _maybe_update(a_cur)
            mu *= mu_factor

    # ------------------------------------------------------------------
    # Phase 3 – hard‑max Adam sub‑gradient descent on the logits
    print("[phase 3] Adam on hard max")
    if time.time() < deadline:
        # initialise Adam from the best logits obtained so far
        y = np.log(best_a + 1e-12)
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        alpha = 0.05                 # base learning rate for logits
        it = 0
        while time.time() < deadline and it < 20000:
            grad_y, max_val = _hard_max_grad(y, n, S)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            step = alpha * m_hat / (np.sqrt(v_hat) + eps_adam)
            y = y - step

            # occasional evaluation
            if (it + 1) % 100 == 0:
                a = S * _softmax(y)
                _maybe_update(a)

            it += 1

    # ------------------------------------------------------------------
    # Phase 4 – cheap random block perturbations + monotone sorting
    print("[phase 4] random block search")
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_trials = int(min(8000, remaining * 180))
        for _ in range(max_trials):
            nb = rng.integers(2, 13)                     # number of blocks 2‑12
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # final monotone‑sorting heuristic
    if time.time() < deadline:
        # descending order
        cand_desc = np.sort(best_a)[::-1]
        cand_desc = _project_to_simplex(cand_desc, S)
        _maybe_update(cand_desc)
        # ascending order (mostly for curiosity)
        cand_asc = np.sort(best_a)
        cand_asc = _project_to_simplex(cand_asc, S)
        _maybe_update(cand_asc)

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
