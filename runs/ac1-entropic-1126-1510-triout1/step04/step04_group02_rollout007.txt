def _optimize_exponential_momentum(a0, n, deadline):
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2 * n * max(conv)

    # parameters
    eta0 = 1.0
    alpha0 = 0.5  # replicates soft max smoothing factor small
    alpha_mul = 1.05
    alpha_max = 20000.0
    alpha_thr = 2000.0
    momentum = 0.9
    prev_grad = np.zeros_like(a)

    size_fft = 1 << ((3*n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and max
        conv = _fft_self_convolve(a)
        cur_score = 2 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            if best_score < 1.4: maybe break? but keep.

        # compute gradient
        if alpha < alpha_thr:
            # use smooth gradient
            shift = conv - conv.max()
            w = np.exp(alpha * shift)
            w /= w.sum()
            # compute gradient via convolution of reversed a with w as before
            a_rev = a[::-1]
            W = np.fft.rfft(w, n=size_fft)
            A_rev = np.fft.rfft(a_rev, n=size_fft)
            conv_grad = np.fft.irfft(W * A_rev, n=size_fft)
            grad = 2 * conv_grad[n-1:n-1+n]  # same as earlier
        else:
            # subgradient at max index
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # momentum
        grad = momentum * prev_grad + (1 - momentum) * grad
        prev_grad = grad

        # step size
        eta = eta0 / np.sqrt(it)

        # exponentiated gradient update
        a = a * np.exp(-eta * grad)

        # clip, renorm
        a = np.maximum(a, 0.0)
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full(n, 1/n)
        else:
            a = a / sum_a

        # schedule alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional random jitter
        if it % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score

def _local_annealing(a_start, n, deadline):

    best_seq = a.copy()
    best_score = 2 * n * np.max(_fft_self_convolve(a))

    T0=1e-3 (temperature) decreasing linearly to near 0 at deadline.

    while time.time() < deadline:

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises
`evaluate_sequence`.  The search uses:

* a large problem size (up to 20 000 entries) – the objective 2·n·max
  scales like 1/n, so longer vectors give a better theoretical bound;
* several simple initialisers (arcsine, uniform, random);
* an exponentiated‑gradient (mirror‑descent) optimiser on the probability
  simplex with momentum, a soft‑max surrogate for the ∞‑norm and a
  sub‑gradient phase;
* a final stochastic hill‑climbing refinement.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Fast projection (see Duchi et al. 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond].max() - 1          # last true index (0‑based)
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine / Beta(0.5,0.5) distribution of length n (sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _uniform_distribution(n: int) -> np.ndarray:
    """Uniform distribution of length n."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
def _random_distribution(n: int) -> np.ndarray:
    """Random Dirichlet‑type distribution (sum = 1)."""
    a = np.random.rand(n)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform_distribution(n)
    return a / a_sum


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using an FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·conv) w.r.t. a.
    """
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_grad = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    grad = 2.0 * conv_grad[start:start + a.shape[0]]
    return grad

def _optimize_exponential_momentum(a0: np.ndarray,
                                   n: int,
                                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser with momentum.
    Returns the best sequence seen (as a probability vector) and its
    internal 2·n·max(conv) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()

    # hyper‑parameters (tuned for speed / quality)
    eta0 = 0.8                     # initial step‑size factor
    alpha = 0.5                    # temperature for the soft‑max surrogate
    alpha_mul = 1.07
    alpha_max = 8000.0
    alpha_thr = 1500.0            # switch to pure sub‑gradient after this
    momentum = 0.85               # exponential moving‑average of the gradient
    size_fft = 1 << ((3 * n - 2).bit_length())

    g_avg = np.zeros_like(a)      # momentum accumulator
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution & score
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient (soft‑max or sub‑gradient)
        if alpha < alpha_thr:
            grad = _gradient_softmax(a, conv, alpha, size_fft)
        else:
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)

        # momentum smoothing
        g_avg = momentum * g_avg + (1.0 - momentum) * grad

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * g_avg)

        # avoid numerical under‑flow
        a = np.maximum(a, 0.0)

        s = a.sum()
        if s == 0.0:
            a = _uniform_distribution(n)
        else:
            a = a / s

        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 5e-6
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: move a random fraction of mass from
    one index to another and keep the move if it improves the score.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 12 % of the source mass
        eps = a[i] * np.random.rand() * 0.12
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = 2.0 * n * np.max(_fft_self_convolve(a_candidate))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point.  Searches for a step‑function that minimises
    `evaluate_sequence`.  Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    # safety margin for printing / final evaluation
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # pick a set of lengths (up to 20 000).  Larger lengths give a better
    # asymptotic bound, but we also keep a few smaller values for quick
    # exploratory runs.
    lengths = [5000, 8000, 12000, 16000, 20000]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503          # stop early if we reach the target

    # ------------------------------------------------------------------
    # Phase 1 – quick runs on the smaller lengths
    for n in lengths[:-1]:  # all except the largest (reserved for deep optimisation)
        remaining = deadline - time.time()
        if remaining < 8.0:
            break
        # give each small problem a modest time slice (≈15 % of remaining)
        slice_deadline = time.time() + max(5.0, remaining * 0.15)

        for init_fun in (_arcsine_distribution,
                         _uniform_distribution,
                         _random_distribution):
            if time.time() >= slice_deadline:
                break

            a0 = init_fun(n)

            seq, _ = _optimize_exponential_momentum(a0, n, slice_deadline)

            # evaluate with the official black‑box (once per candidate)
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                # fallback to the internal surrogate score
                final_score = 2.0 * n * np.max(_fft_self_convolve(seq))

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq.copy()
                print(f"*** New global best → len={n} init={init_fun.__name__} "
                      f"score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break
        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Phase 2 – deep optimisation on the largest length (20 000)
    if overall_best_score > target_score and time.time() < deadline - 5.0:
        n = lengths[-1]
        # allocate the rest of the budget (leaving a couple of seconds for finalisation)
        slice_deadline = deadline - 2.0

        # several restarts with slightly different initialisers
        init_functions = (_arcsine_distribution,
                          _uniform_distribution,
                          _random_distribution)

        for init_fun in init_functions:
            if time.time() >= slice_deadline:
                break
            a0 = init_fun(n)

            seq, _ = _optimize_exponential_momentum(a0, n, slice_deadline)

            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = 2.0 * n * np.max(_fft_self_convolve(seq))

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq.copy()
                print(f"*** New global best (large) → len={n} init={init_fun.__name__} "
                      f"score={overall_best_score:.6f}")

            if overall_best_score <= target_score:
                break

    # ------------------------------------------------------------------
    # Final stochastic hill‑climbing on the best candidate found so far
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n = len(overall_best_seq)
        seq, score = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = score
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq.copy()
            print(f"*** After local refinement score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – should we still have nothing, fall back to a uniform vector
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _uniform_distribution(n)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
