"""
Simple multi‑stage optimiser for `evaluate_sequence`.

Stage 1 : generate a portfolio of parametric candidate distributions
          (arcsine, uniform, mixtures, power‑law and random beta).

Stage 2 : pick the best candidate (lowest internal bound 2·n·max(conv))
          and polish it with an exponentiated‑gradient optimiser
          (mirrored on the simplex).

Stage 3 : short stochastic hill‑climbing refinement.

The whole routine respects a user supplied time budget (default 1000 s)
and returns the best sequence found as a plain list.
"""
import time
import numpy as np

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a by FFT (real). Returns array of length 2·n‑1."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max of a∗a at index k (2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b_k) w.r.t. a using the soft‑max weights.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

                          n: int,
                          deadline: float,
                          max_iter: int = None) -> tuple[np.ndarray, float]:

    Exponentiated‑gradient optimisation on the probability simplex.
    Returns the best sequence seen and its internal score (2·n·max_conv).
    """

    best_seq = a.copy()

    eta0 = 0.7
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0          # switch to pure sub‑gradient beyond this

    it = 0
    while time.time() < deadline:
        it += 1
        if max_iter is not None and it > max_iter:

        b = _fft_self_convolve(a)
        cur_max = b.max()

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} best_score={best_score:.6f}")

        # gradient
        if alpha > alpha_thr:

            grad = _subgradient_max(a, k)
        else:

            grad = _gradient_softmax(a, b, alpha, size_fft)

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # protect against numerical underflow / negativity
        a = np.maximum(a, 0.0)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to break symmetry
        if it % 3000 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score

Multi‑stage optimiser for `evaluate_sequence`.

1️⃣  Generate a modest portfolio of parametric candidates (arcsine,
    uniform, mixtures, power‑law, random Beta) for several lengths.

2️⃣  Pick the best candidate (lowest internal bound 2·n·max(conv)) and
    refine it with an exponentiated‑gradient optimiser that works on the
    probability simplex (mirrored update, soft‑max → sub‑gradient).

3️⃣  Finish with a short stochastic hill‑climbing (random mass moves).

The routine respects a time budget (default 1000 s) and returns the best

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length n (probability vector)."""
    i = np.arange(n, dtype=np.float64)

    vals /= vals.sum()
    return vals

    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""

def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]

    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g

def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:

    Gradient of (1/α)·log∑exp(α·b_k) w.r.t. a using the soft‑max weights.

        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0

        w /= w_sum

    A_rev = np.fft.rfft(a_rev, n=size_fft)
