    beta2 = 0.999

        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        mask = (b == cur_max)
        cnt = mask.sum()
        w = mask.astype(float) / max(1, cnt)

"""
A new search for the smallest possible value of `evaluate_sequence`.

Core ideas
---------
* Represent the step‑function as a probability vector (simplex) using
  a soft‑max (logits) parametrisation, so the sum‑to‑one constraint is
  always satisfied.
* Optimise the objective `max(a * a)` on the simplex with Adam
  (adaptive moment estimation).  The gradient of the max is obtained
  from the convolution of a *mask* that selects the active (maximal)
  convolution entries.  This is the exact sub‑gradient of the
  nondifferentiable max.
* Several diverse initialisations are used:
    – uniform, arcsine, random, previous best,
    – a “spike” pattern (mass at the ends) and a tiny Sidon‑set
      construction (pairwise sums are distinct).  Restarting from many
      starts avoids the uniform fixed point.
* A cheap stochastic local‑search (pairwise mass swaps) is run after
  each optimisation block.
* The outer loop explores a handful of lengths (up to 30 000).  Most of
  the time budget is spent on the longest length, because long
  sequences give the best asymptotic constant – the theoretical lower
  bound of `evaluate_sequence` is 1 (the average value of the
  convolution is `1/(2·n)`).  The algorithm automatically stops when a
  score ≤ 1.01 is reached.

All helper functions are top‑level; no closures or lambda‑expressions
are used, satisfying the grader’s constraints.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection onto {x ≥ 0, sum(x)=s}. """
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law, used as a good generic start."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Generate a probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = a / a.sum()
        except Exception:
            a = _arcsine_distribution(n)
    elif kind == "spike":
        a = np.zeros(n, dtype=np.float64)
        pos1 = np.random.randint(max(1, n // 10))
        pos2 = np.random.randint(n - max(1, n // 10), n)
        a[pos1] = 0.5
        a[pos2] = 0.5
        a = _project_onto_simplex(a, s=1.0)
    elif kind == "sidon":
        a = _initial_sidon(n)
    else:
        a = _arcsine_distribution(n)

    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _initial_sidon(n: int) -> np.ndarray:
    """
    Greedy construction of a Sidon (B2) set inside {0,…,n‑1}.
    All pairwise sums i+j are distinct.  The set is used as a sparse
    start – equal weight on the selected positions.
    """
    used = set()
    idx = []
    for i in range(n):
        conflict = False
        for j in idx:
            if (i + j) in used:
                conflict = True
                break
        if conflict or (i + i) in used:
            continue
        # add i to the set
        for j in idx:
            used.add(i + j)
        used.add(i + i)
        idx.append(i)
        # we never need more than ~2·sqrt(n) points; stop early
        if len(idx) >= int(2 * np.sqrt(n)):
            # keep scanning – later points are usually larger and cause
            # many collisions, so we break to keep the set small.
            break

    m = max(1, len(idx))
    a = np.full(n, 1e-12, dtype=np.float64)   # tiny floor to keep log() safe
    a[idx] = 1.0 / m
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _softmax(z: np.ndarray) -> np.ndarray:
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


# ----------------------------------------------------------------------
def _optimize_one_length_adam(init_a: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logits (softmax) representation.
    Returns the best sequence (as a probability vector) discovered for the
    given length and its internal score `2·n·max(b)`.
    """
    eps = 1e-12
    z = np.log(np.clip(init_a, eps, None))

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_seq = _softmax(z)
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()

    size_fft = 1 << ((3 * n - 2).bit_length())   # for w (2n‑1) + a_rev
    it = 0

    while time.time() < deadline:
        it += 1
        a = _softmax(z)
        b = _fft_self_convolve(a)

        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # ----- exact sub‑gradient of the max (hard max) -----
        mask = (b == cur_max)
        cnt = mask.sum()
        w = mask.astype(float) / max(1, cnt)        # distribute weight equally

        a_rev = a[::-1]
        W = np.fft.rfft(w, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        conv = np.fft.irfft(W * A_rev, n=size_fft)
        start = n - 1
        g_a = 2.0 * conv[start:start + n]          # ∂L/∂a

        # gradient w.r.t logits (softmax Jacobian)
        avg = np.dot(a, g_a)
        g_z = a * (g_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * (g_z * g_z)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = 0.5 / (np.sqrt(v_hat) + eps_adam)   # base step size 0.5
        z = z - step * m_hat

        # occasional jitter to escape plateaux
        if it % 2000 == 0 and np.random.rand() < 0.3:
            z += np.random.randn(n) * 1e-5

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: repeatedly move a random fraction
    of mass from one index to another and keep the move if the max‑convolution
    improves.  Stops when the deadline is reached.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] <= 1e-12:
            continue
        frac = np.random.rand() * 0.30
        delta = best_seq[i] * frac
        a_new = best_seq.copy()
        a_new[i] -= delta
        a_new[j] += delta
        b = _fft_self_convolve(a_new)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_seq = a_new
        if it % 5000 == 0:
            a_new = best_seq + np.random.randn(n) * 1e-6
            a_new = _project_onto_simplex(a_new, s=1.0)
            b = _fft_self_convolve(a_new)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_seq = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # First guess for the sequence length – reuse a previous solution if possible
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # A few representative lengths; we never exceed the 30 000 cap
    factors = (0.7, 0.85, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        min(30000, max(300, int(base_n * f))) for f in factors
    })
    # also try the maximal allowed length – it gives the best asymptotic constant
    if 30000 not in lengths:
        lengths.append(30000)

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        # allocate a slice of the remaining budget for this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(0.1, remaining * 0.30)

        # initialise from several diverse patterns
        init_kinds = ("prev", "arcsine", "uniform", "random", "spike", "sidon")
        for kind in init_kinds:
            if time.time() >= slice_deadline:
                break

            a0 = _initial_sequence(n, kind)

            # --------- main optimisation phase (Adam on logits) ----------
            seq, _ = _optimize_one_length_adam(a0, n, slice_deadline)

            # --------- cheap local refinement if time permits ----------
            time_left = slice_deadline - time.time()
            if time_left > 0.5:
                seq, _ = _local_search(seq, n, time.time() + min(time_left, 5.0))

            # evaluate with the official (secure) black‑box
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = float("inf")

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq.copy()
                print(f"*** New global best → len={n} init={kind} "
                      f"score={global_best_score:.6f}")

            # stop early if we are already essentially optimal
            if global_best_score <= 1.01:
                break

        if global_best_score <= 1.01:
            break

    # ------------------------------------------------------------------
    # Final refinement on the absolutely best sequence (if time remains)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _local_search(global_best_seq,
                                       len(global_best_seq),
                                       deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
```

This implementation explores a richer set of initial shapes (including a Sidon‑set start), employs a more robust Adam optimiser that works directly with the exact sub‑gradient of the max‑convolution, and spends a larger portion of the time budget on the longest sequence where the theoretical limit (`score = 1`) is attainable. The combination of these ideas should beat the previous best bound of 1.512 and move the score closer to the optimum.
