def _beta_distribution(n, alpha):
    x = (np.arange(n)+0.5)/n
    vals = np.power(x, alpha-1) * np.power(1-x, alpha-1)
    vals_sum = vals.sum()
    if vals_sum == 0:
        # fallback uniform
        return np.full(n, 1.0/n)
    return vals/vals_sum

def _optimize_adagrad(a0, n, deadline):
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2.0*n*np.max(_fft_self_convolve(a))

    eta0 = 0.5
    epsilon = 1e-12
    G = np.zeros_like(a)   # sum of squares of gradients

    alpha = 0.1
    alpha_mul = 1.03
    alpha_max = 1e4
    alpha_thr = 2000   # maybe for subgradient switch

    size_fft = 1<<((3*n -2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        b = _fft_self_convolve(a)
        cur_score = 2.0*n*b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} best={best_score:.6f}")

        # gradient
        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a,k)
        else:
            grad = _gradient_softmax(a,b,alpha,size_fft)

        # accumulate squared grads
        G += grad**2

        # adjust per-coordinate step
        step = eta0 / (np.sqrt(G) + epsilon)

        # exponentiated gradient update multiplied
        a = a * np.exp(-step * grad)

        # ensure nonneg and renormalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a/s
        else:
            a = np.full(n, 1.0/n)

        # update alpha
        alpha = min(alpha_max, alpha*alpha_mul)

        # optional random jitter
        if it % 5000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n)*1e-5
            a = np.maximum(a,0)
            a /= a.sum()

    return best_seq, best_score

"""
Propose a low‑auto‑correlation step‑function.

Algorithm
---------
1. **Parameter sweep** – generate a family of symmetric Beta‑shaped
   distributions  a_i ∝ x^{α‑1}(1‑x)^{α‑1}  (x = (i+½)/n) for several
   values of α (<1 gives heavy tails, α≈0.5 is the classic arcsine).
   For a few candidate lengths (≈ 0.7–1.8× the previous best length) we
   evaluate the true objective and keep the best one.  The previous
   solution (``height_sequence_1``) is also tried.

2. **Adaptive mirror descent** – starting from the best candidate, run a
   projected exponentiated‑gradient (mirror) iteration on the simplex.
   * The gradient is the exact sub‑gradient of the max‑convolution while
     a smooth “soft‑max’’ gradient (temperature α) is used early on.
   * An *Adagrad*‑style per‑coordinate stepsize (η_i = η₀/(√G_i+ε))
     speeds up convergence.
   * The temperature α is increased slowly (α←α·1.03) – we begin with a
     very smooth surrogate (α≈0.1) and end with a hard max (α≈10⁴).

3. **Local random refinement** – after the main phase a cheap hill‑climbing
   that moves a small fraction of mass between two random indices is run
   until the time budget expires.

The whole routine respects the 1000 s budget, prints progress, and returns
the best sequence found (as a plain Python list of non‑negative heights).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Deterministic symmetric Beta(α,α) density on n points:
    a_i ∝ ((i+0.5)/n)^{α‑1}·(1‑(i+0.5)/n)^{α‑1}.
    For α = 0.5 this is the classic arcsine law.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid 0^neg by keeping x strictly inside (0,1)
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # normalise to sum = 1
    total = vals.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a via a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(b) where b = a*a, at the index k of the maximum."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b_k) w.r.t. a using the soft‑max weights.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_adagrad(a0: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) with Adagrad step‑size
    and a temperature schedule for the soft‑max surrogate.
    Returns the best sequence seen and its internal 2·n·max(b) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    # hyper‑parameters
    eta0 = 0.5
    eps = 1e-12
    G = np.zeros_like(a)                # accumulated squared grads

    alpha = 0.1                         # start very smooth
    alpha_mul = 1.03
    alpha_max = 1e4
    alpha_thr = 2000.0                  # when to switch to true sub‑grad

    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and (internal) score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} best={best_score:.6f}")

        # gradient (soft‑max or exact sub‑gradient)
        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        # Adagrad accumulation & per‑coordinate step
        G += grad ** 2
        step = eta0 / (np.sqrt(G) + eps)

        # exponentiated‑gradient update
        a = a * np.exp(-step * grad)

        # protect against numerical issues & re‑project onto simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: move a small proportion of mass from one index
    to another and keep the move if it improves the objective.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer at most 10 % of source mass
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = 2.0 * n * np.max(_fft_self_convolve(a_candidate))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score
