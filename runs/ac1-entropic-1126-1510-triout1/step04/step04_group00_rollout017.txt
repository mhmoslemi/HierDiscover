def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2
    # choose length
    lengths = [2000,4000,8000,16000] ... ensure within limit.

    overall_best_seq = None; best_score = inf

    for n in lengths:
        # allocate portion of remaining time
        remaining = deadline - time.time()
        if remaining <= 5: break
        slice_end = time.time() + min(remaining*0.9, 300)  # 90% but not exceed 300 secs per n.

        # param search over p
        best_local_a = None
        best_local_score = float('inf')
        # grid p
        for p in np.linspace(0.1, 0.9, 17): # 0.1 step about
            a = ((np.arange(n)+0.5)*(n - np.arange(n)-0.5))**(-p)
            a = a / a.sum()
            # compute conv and max
            b = _fft_self_convolve(a)
            maxb = b.max()
            score = 2 * n * maxb   # because sum=1
            if score < best_local_score:
                best_local_score = score
                best_local_a = a.copy()
        # attempt gradient descent from best_local_a
        a = best_local_a
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1=0.9; beta2=0.999; eps=1e-8
        iteration = 0
        beta = 5.0   # smoothing parameter initially
        while time.time() < slice_end:
            iteration +=1
            b = _fft_self_convolve(a)
            maxb = b.max()
            # compute w (softmax)
            if beta < 2000:
                # update beta gradually
                beta = min(2000, beta * 1.002)
            exp_shift = np.exp(beta * (b - maxb))
            w = exp_shift / exp_shift.sum()

            # gradient
            conv_slice = _fft_convolve(a[::-1], w)[n-1:n-1+n]
            grad = a + conv_slice   # gradient of smoothed max

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1**iteration)
            v_hat = v / (1 - beta2**iteration)
            step = 0.5 / (np.sqrt(iteration) + 1)  # base step decaying
            a = a - step * m_hat / (np.sqrt(v_hat) + eps)

            # project onto simplex
            a = _project_onto_simplex(a, s=1.0)

            # record best seen
            if maxb < best_local_score:
                best_local_score = maxb
                best_local_a = a.copy()
                # convert to score for evaluation
                score_val = 2 * n * maxb
                print(time.time(), f"n={n} iter={iteration} score≈{score_val:.6f}")

                if score_val <= 1.503:
                    overall_best_seq = a.copy()
                    best_score = score_val
                    print("Target achieved")
                    return overall_best_seq.tolist()

        # finalize this length
        # evaluate candidate
        seq = best_local_a
        try:
            score_val = evaluate_sequence(seq.tolist())
        except Exception:
            # fallback compute
            score_val = 2 * n * _fft_self_convolve(seq).max()
        if score_val < best_score:
            best_score = score_val
            overall_best_seq = seq.copy()
            print(f"New best overall score {best_score:.6f} at n={n}")

    # fallback
    if overall_best_seq is None:
        overall_best_seq = np.full(1000, 1/1000)
        best_score = evaluate_sequence(overall_best_seq.tolist())
    print(f"Final best score: {best_score:.6f}, length={len(overall_best_seq)}")
    return overall_best_seq.tolist()

"""
Propose a low‑score step‑function.

The algorithm works on the probability simplex Σ a_i = 1 (the evaluation
score is invariant to scaling).  It consists of three stages:

1.  A fast parametric scan.  Vectors of the form
       a_i ∝ ((i+0.5)*(n‑i‑0.5))^{-p}
    are generated for many values of p∈[0.1,0.9] and the one with the
    smallest 2·n·max(a*a) is kept as a warm start.

2.  A few rounds of random jitter + projection are applied to break
    possible symmetry.

3.  Projected Adam optimisation with a soft‑max (log‑sum‑exp) surrogate
    of the max‑convolution.  The temperature parameter β is increased
    slowly so the surrogate converges to the true max.  The iterate is
    kept on the simplex by a Euclidean projection after each update.
    The best vector seen during the whole run is returned.

The whole search respects the supplied time budget (default 1000 s)
and stops early as soon as a score ≤ 1.503 is found.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# --------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

        base_n = 8000
    # cap the length to keep convolutions fast but still expressive
    n = max(2000, min(20000, base_n))

    # ------------------------------------------------------------------
    # 1️⃣  Parametric warm‑start (grid over exponent p)
    best_a = None
    best_score = float("inf")
    p_grid = np.linspace(0.10, 0.90, 81)          # 81 values → fine scan
    idx = np.arange(n, dtype=np.float64)
    for p in p_grid:
        vals = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
        a = vals / vals.sum()
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    # ------------------------------------------------------------------
    # 2️⃣  Small random jitter to avoid symmetries
    jitter = np.random.randn(n) * 1e-4
    best_a = _project_onto_simplex(best_a + jitter, s=1.0)

    # ------------------------------------------------------------------
    # 3️⃣  Projected Adam optimisation with a smoothed max surrogate
    # --------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1, beta2 = 0.90, 0.999
    beta1_pow, beta2_pow = 1.0, 1.0
    eps = 1e-8

    beta = 5.0                       # temperature parameter (small → smooth)
    beta_max = 5000.0
    beta_factor = 1.003              # slow annealing

    iteration = 0
    # keep the best vector ever seen
    best_local_a = a.copy()
    best_local_score = best_score

    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # a*a and its maximum
        b = _fft_self_convolve(a)
        max_b = b.max()

        # --------------------------------------------------------------
        # Update temperature (increase β → sharper approximation)
        if beta < beta_max:
            beta = min(beta_max, beta * beta_factor)

        # --------------------------------------------------------------
        # Soft‑max weights w_k = exp(β (b_k‑max_b)) / Σ … .
        # The exponent is non‑positive, so overflow cannot happen.
        exp_shift = np.exp(beta * (b - max_b))
        w_sum = exp_shift.sum()
        if w_sum == 0.0:                                 # under‑flow → exact max
            eps_active = 1e-12 * (abs(max_b) + 1.0)
            active = np.flatnonzero(b >= max_b - eps_active)
            w = np.zeros_like(b)
            if active.size > 0:
                w[active] = 1.0 / active.size
        else:
            w = exp_shift / w_sum

        # --------------------------------------------------------------
        # Gradient of the smooth max:
        #   ∇ = a + conv_rev(a, w)   (see analysis for derivation)
        conv_full = _fft_convolve(a[::-1], w)            # length 3·n‑2
        conv_slice = conv_full[n - 1 : n - 1 + n]        # exactly n entries
        grad = a + conv_slice

        # --------------------------------------------------------------
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # simple learning‑rate schedule (decreasing with √t)
        step = 0.30 / (math.sqrt(iteration) + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # Project back onto the simplex (keeps sum = 1 and non‑negativity)
        a = _project_onto_simplex(a, s=1.0)

        # --------------------------------------------------------------
        # Track the best exact score seen so far
        cur_score = 2.0 * n * max_b
        if cur_score < best_local_score - 1e-14:
            best_local_score = cur_score
            best_local_a = a.copy()
            print(
                f"[{time.time():.1f}] n={n} iter={iteration} "
                f"score≈{cur_score:.6f}  β={beta:.1f}"
            )
            # early exit if the target is reached
            if cur_score <= 1.503:
                print("Target score achieved – exiting.")
                return best_local_a.tolist()

        # occasional tiny random perturbation to escape plateaux
        if iteration % 5000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    # ------------------------------------------------------------------
    # Convert the best found vector to a Python list and return.
    # (If evaluate_sequence would raise, we fall back to the exact score we
    #   already computed – it is the same quantity up to the sum‑normalisation.)
    final_seq = best_local_a.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception:
        final_score = best_local_score
    print(
        f"Final best score: {final_score:.6f} (length = {len(final_seq)})"
    )
    return final_seq
