"""
Propose a step‑function (non‑negative sequence) with a low max‑self‑convolution.

Algorithm
---------
1. Initialise a feasible sequence f on the simplex Σf = √(2·n)
   (uniform or the reference `height_sequence_1`).

2. Iteratively improve using the “LP‑mix” scheme from the problem statement:
   * Let M = max (f * f).  Build the Toeplitz constraint matrix
     A_ub (size (2n‑1)×n) that represents the convolution f * g.
   * Solve the linear programme
         maximise  Σ g_j     subject to   A_ub·g ≤ M,   g ≥ 0 .
     The solution g₀ has the largest ℓ₁‑mass that still respects the
     current bound M.
   * Scale g = (√(2·n) / Σg₀)·g₀ so that Σg = √(2·n).
   * Mix f and g:  f₁ = (1‑t) f + t g   (t chosen by a tiny line‑search).
     Because g was chosen with maximal ℓ₁‑mass, any convex combination
     with a small enough t strictly reduces the maximal self‑convolution.
   * Accept the new f if the bound decreased; otherwise stop the LP loop.

   The LP matrix has a fixed sparsity pattern; only the diagonal values
   (the entries of f) change each iteration.  We pre‑compute the row/col
   index vectors once and reuse them, which makes each iteration cheap.

3. After the LP phase, run a cheap hill‑climbing / simulated‑annealing
   procedure that moves a small amount of mass between two positions.
   This fine‑tunes the solution in the remaining time.

4. Return the best sequence (as a plain list).  The evaluation score is
   simply the maximal entry of the self‑convolution because the ℓ₁‑norm
   is kept equal to √(2·n).

The whole routine respects the 1000 s budget, uses only NumPy,
SciPy (sparse linear programming via HiGHS) and no external I/O.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """ Euclidean projection of v onto {x ≥ 0, Σx = z}. """
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_size(n: int) -> int:
    """ Smallest power of two ≥ 2·n‑1 (used for FFT convolutions). """
    return 1 << ((2 * n - 1).bit_length())


def _full_conv(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """ Convolution a * b using FFT (length 2·len(a)-1). """
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0                     # numerical safety
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """ Maximum entry of a * a (self‑convolution). """
    return float(_full_conv(a, a, size).max())


# ----------------------------------------------------------------------
# Cached index arrays for the Toeplitz constraint matrix.
# They depend only on n and are built once per run.
_row_cache = {}
_col_cache = {}


def _build_conv_constraint_matrix(f: np.ndarray) -> csr_matrix:
    """
    Return the sparse matrix A_ub such that (A_ub @ g)[k] = (f * g)[k]
    for k = 0 … 2n‑2.  Sparsity pattern is fixed; only data change.
    """
    n = f.shape[0]
    # Build (or retrieve) the row/col index vectors for this n.
    if n not in _row_cache:
        # rows[i, j] = i + j, columns[i, j] = j   (i = 0..n‑1, j = 0..n‑1)
        i = np.arange(n, dtype=np.int32)
        rows = (i[:, None] + i[None, :]).ravel(order="F")
        cols = np.repeat(i, n)
        _row_cache[n] = rows
        _col_cache[n] = cols
    rows = _row_cache[n]
    cols = _col_cache[n]

    # Data are just repetitions of the vector f.
    data = np.tile(f, n).astype(np.float64)

    # Shape: (2n‑1) rows, n columns.
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
def _lp_max_ell1(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  maximise Σ g_j   subject to   f * g ≤ M,   g ≥ 0.
    Returns the optimal g (numpy array) or None if the LP fails.
    """
    n = f.shape[0]
    A_ub = _build_conv_constraint_matrix(f)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximise Σ g ⇒ minimise –Σ g .
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n

    # Use HiGHS (the default) – it handles sparse matrices efficiently.
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    if not res.success:
        print(f"[LP] failed ({res.message})")
        return None
    return res.x


def _mix_and_line_search(f: np.ndarray, g: np.ndarray, M: float,
                         size: int, S: float) -> tuple[np.ndarray, float]:
    """
    Given current f and a candidate g (both already ℓ₁‑normalised),
    try a few mixing coefficients t and return the best (f_new, M_new).
    If none improves, returns the original (f, M).
    """
    # Pre‑compute the three convolutions that appear in the quadratic expansion.
    conv_ff = _full_conv(f, f, size)
    conv_fg = _full_conv(f, g, size)
    conv_gg = _full_conv(g, g, size)

    best_f = f
    best_M = M

    # A modest palette of step sizes.
    for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
        a = 1.0 - t
        # (a f + t g) * (a f + t g) = a² ff + 2 a t fg + t² gg
        mixed = (a * a) * conv_ff + (2.0 * a * t) * conv_fg + (t * t) * conv_gg
        new_M = float(mixed.max())
        if new_M < best_M - 1e-9:      # significant improvement
            best_M = new_M
            best_f = a * f + t * g
    return best_f, best_M
