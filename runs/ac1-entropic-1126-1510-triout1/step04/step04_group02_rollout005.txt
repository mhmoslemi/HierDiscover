def power_law_distribution(n, p, invert=True):
    i = np.arange(n, dtype=np.float64)
    product = (i + 0.5) * (n - i - 0.5)
    # product may be zero for n small? For i=0: product ~ 0.5*(n-0.5) >0.
    if invert:
        vals = product ** (-p)
    else:
        vals = product ** p
    vals_sum = vals.sum()
    if vals_sum == 0:
        vals = np.full(n, 1.0/n)
    else:
        vals = vals / vals_sum
    return vals

    x = i / n
    vals = (x ** (a-1)) * ((1-x) ** (a-1))

        vals = np.full(n, 1.0/n)

"""
A stochastic multi‑stage optimizer for the non‑negative step‑function problem.

The algorithm proceeds in three phases:

1. **Random shape search** – Generates many candidate probability vectors
   using a few simple parametric families (power‑law, beta, Gaussian,
   mixtures).  Each candidate is projected onto the simplex and evaluated
   with a fast FFT based “max‑self‑convolution” proxy for the official
   `evaluate_sequence` objective.

2. **Exponential‑gradient refinement** – The best few candidates are
   refined with the mirror‑descent / exponentiated‑gradient method used
   in the reference solution, but with a more aggressive schedule and
   tighter integration with the soft‑max surrogate.

3. **Local random‑move hill‑climbing** – A cheap stochastic refinement
   moves a small fraction of probability mass between two coordinates
   whenever it improves the objective.

The whole search respects a 1000 s budget, keeps the best sequence found
and returns it as a plain Python list.  All helper functions are defined
at module level (no closures) and only NumPy and the built‑in `time`
module are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float, invert: bool = True) -> np.ndarray:
    """Symmetric 1/(i·(n‑i))^p or (i·(n‑i))^p distribution."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    if invert:
        vals = prod ** (-p)
    else:
        vals = prod ** p
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) distribution on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    vals = (x ** (a - 1)) * ((1 - x) ** (a - 1))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _gaussian_distribution(n: int, sigma: float) -> np.ndarray:
    """Centered Gaussian on the index set."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """Convex combination (1‑lam)*arcsine + lam*uniform."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n, dtype=np.float64)
    vals = (1.0 - lam) * arcs + lam * uni
    # already normalised, but guard against tiny drift
    return _project_onto_simplex(vals, s=1.0)


# ----------------------------------------------------------------------
def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the (full) self‑convolution a*a using FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum a)^2.
    The vector is projected onto the simplex before evaluation.
    """
    a = _project_onto_simplex(a, s=1.0)
    n = a.shape[0]
    max_conv = _max_self_convolution(a)
    return 2.0 * n * max_conv  # denominator is 1 because sum a == 1


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the (1/α)·log‑sum‑exp(α·b_k) surrogate w.r.t. a.
    Implements the formula g = 2 * conv_rev(w, a_rev).
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser.
    Returns the best sequence seen and its (2·n·max) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    eta0 = 0.8               # initial step‑size constant
    alpha = 1.0              # temperature for soft‑max
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0       # switch to pure sub‑gradient after this
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient
        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)

        # renormalise onto simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑move hill‑climbing on the best sequence.
    Moves a fraction of mass from one index to another if it improves.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the source mass
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = _objective(a_candidate)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score

    except Exception:
        base_n = 8000

    length_factors = (0.5, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 20000
    })
    # ensure a few hand‑picked values are present
    lengths.extend([5000, 8000, 12000, 16000, 20000])
    lengths = sorted({l for l in lengths if l <= 20000})

    # ------------------------------------------------------------------
    # Global best bookkeeping
    best_seq = None
    best_score = float("inf")
    best_candidates = []   # list of (score, sequence) tuples

    # initialise from previously‑found solution if available
    try:
        seq0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if seq0.size:
            seq0 = np.maximum(seq0, 0.0)
            seq0 = _project_onto_simplex(seq0, s=1.0)
            score0 = _objective(seq0)
            print(f"Prev‑solution score={score0:.6f}")
            best_seq = seq0.copy()
            best_score = score0
            best_candidates.append((score0, seq0))
    except Exception:
        pass

    # also seed with a plain arcsine of base length
    arcs_seq = _arcsine_distribution(base_n)
    arcs_score = _objective(arcs_seq)
    print(f"Arcsine (n={base_n}) score={arcs_score:.6f}")
    if arcs_score < best_score:
        best_seq = arcs_seq.copy()
        best_score = arcs_score
    best_candidates.append((arcs_score, arcs_seq))

    # ------------------------------------------------------------------
    # Phase 1 – random parametric search
    families = ("power_invert", "power_direct", "beta", "gauss", "mix")
    search_deadline = deadline - 200.0   # keep ~200 s for refinements
    while time.time() < search_deadline:
        n = int(np.random.choice(lengths))
        fam = np.random.choice(families)

        if fam == "power_invert":
            p = np.random.uniform(0.2, 1.2)
            seq = _power_law_distribution(n, p, invert=True)
        elif fam == "power_direct":
            p = np.random.uniform(0.2, 1.2)
            seq = _power_law_distribution(n, p, invert=False)
        elif fam == "beta":
            a = np.random.uniform(0.2, 5.0)
            seq = _beta_distribution(n, a)
        elif fam == "gauss":
            sigma = np.random.uniform(n / 20.0, n / 2.0)
            seq = _gaussian_distribution(n, sigma)
        elif fam == "mix":
            lam = np.random.uniform(0.0, 1.0)
            seq = _mixture_arcsine_uniform(n, lam)
        else:
            seq = _arcsine_distribution(n)

        seq = _project_onto_simplex(seq, s=1.0)
        score = _objective(seq)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"New global best (rand)  len={n} fam={fam} score={best_score:.6f}")

        # maintain a small pool of promising candidates
        if len(best_candidates) < 12:
            best_candidates.append((score, seq))
        else:
            # replace the worst if the new one is better
            worst_idx = max(range(len(best_candidates)),
                            key=lambda i: best_candidates[i][0])
            if score < best_candidates[worst_idx][0]:
                best_candidates[worst_idx] = (score, seq)

    # ------------------------------------------------------------------
    # Phase 2 – exponential‑gradient refinement of top candidates
    best_candidates.sort(key=lambda x: x[0])
    top_to_refine = [cand[1] for cand in best_candidates[:4]]

    for cand_seq in top_to_refine:
        if time.time() >= deadline - 20.0:
            break
        n = cand_seq.shape[0]
        # give each refine run up to 60 s (or whatever is left)
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(60.0, remaining * 0.6)
        refined_seq, _ = _optimize_exponential(cand_seq, n, slice_deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"Refinement improved to {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # Phase 3 – final random‑move hill climbing
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        refined_seq, _ = _local_refinement(best_seq, n, deadline)
        refined_score = _objective(refined_seq)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"After local refinement best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # fallback safety net
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)

    print(f"Final best score: {best_score:.6f} (length = {best_seq.shape[0]})")
    # return plain Python list as required by the competition harness
    return best_seq.tolist()
