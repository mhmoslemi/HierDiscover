        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

def _intensive_local_search(seq, n, deadline):
    a = seq.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()
    best_a = a.copy()
    # set schedule for temperature of moves: start with step fraction ~0.3 and exponentially decay
    while time.time() < deadline:
        # choose random i,j; if a[i] tiny continue
        i = np.random.randint(n)
        if a[i] < 1e-12: continue
        j = np.random.randint(n-1)
        if j >= i: j+=1
        # step fraction: maybe based on temperature proportional to remaining time
        frac = np.random.rand() * 0.3
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # evaluate
        b = _fft_self_convolve(a_new)
        score = 2.0 * n * b.max()
        # accept if better else with some probability decreasing
        if score < best_score or np.random.rand() < 0.001:
            a = a_new
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
        # optionally jitter
        if np.random.rand() < 0.001:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)
    return best_a, best_score

"""
Advanced search for a step‑function (non‑negative heights) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The optimizer works in three stages:

1. **Length selection** – try several candidate lengths (including the length
   of the previous best solution) with a fast Adam‑based projected gradient
   descent to obtain a rough estimate of the achievable score.

2. **Intensive refinement** – for the most promising length run a long‑running
   Adam optimizer (with a soft‑max approximation of the max‑convolution) and
   a targeted stochastic local‑search that moves mass away from the indices
   contributing most to the current convolution peak.

3. **Final polishing** – a short extra local‑search using the remaining time.

All updates stay on the probability simplex (sum = 1, a_i ≥ 0).  The code
uses FFT‑based convolutions for O(n log n) cost and respects the 1000 s
time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution on {0,…,n‑1} (probability vector)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:

    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:

            if a.size != n:

            a = a / a.sum()

    else:
        a = _arcsine_distribution(n)
    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _spike_sequence(n: int) -> np.ndarray:
    """Two‑spike pattern: half the mass near each end."""
    a = np.zeros(n, dtype=np.float64)
    pos1 = np.random.randint(max(1, n // 10))
    pos2 = np.random.randint(n - max(1, n // 10), n)
    a[pos1] = 0.5
    a[pos2] = 0.5
    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·b_k) w.r.t. a.
    Implements the sub‑gradient of max(b) when α is large.
    """
    n = a.shape[0]

    # ----- soft‑max weights -----
    if alpha > 5000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ----- compute 2 * (w * a_rev) -----
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g

def _optimize_one_length_adam(init_a: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimizer on the simplex for a fixed length n.
    Returns (best_sequence, best_internal_score) where the internal score
    is 2·n·max(self‑convolution) (sum of the vector is 1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    step0 = 0.5               # base step size
    t = 0

    # soft‑max temperature schedule
    alpha = 50.0
    alpha_mul = 1.02
    alpha_max = 20000.0

    # FFT size that can accommodate w (2n‑1) + a_rev (n) –1
    size_fft = 1 << ((3 * n - 2).bit_length())

    while time.time() < deadline:
        t += 1

        # evaluate current objective
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={t} score={best_score:.6f}")

        # gradient of soft‑max approximation of max(b)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter to hop out of plateaus
        if t % 4000 == 0 and np.random.rand() < 0.3:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _targeted_local_search(a: np.ndarray,
                           n: int,
                           deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that repeatedly moves mass away from indices
    heavily contributing to the current convolution peak.
    """
    best_a = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_a).max()
    cur_a = a.copy()
    rng = np.random.default_rng()
    while time.time() < deadline:
        # full convolution and peak location
        b = _fft_self_convolve(cur_a)
        k_max = int(np.argmax(b))

        # compute contributions of each index to this peak
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[idx] = cur_a[idx] * cur_a[k_max - idx]

        total = contrib.sum()
        if total > 0:
            prob = contrib / total
            i = rng.choice(n, p=prob)
        else:
            i = rng.integers(n)

        # pick a distinct destination index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the mass at i to j
        frac = rng.random() * 0.25  # up to 25 %
        delta = cur_a[i] * frac
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # evaluate
        new_score = 2.0 * n * _fft_self_convolve(new_a).max()
        if new_score < best_score:
            best_score = new_score
            best_a = new_a.copy()
            cur_a = new_a
        else:
            # occasional uphill acceptance (simulated‑annealing style)
            if rng.random() < 0.001:
                cur_a = new_a

        # tiny random jitter every few thousand steps
        if rng.random() < 0.001:
            cur_a += rng.normal(scale=1e-6, size=n)
            cur_a = _project_onto_simplex(cur_a, s=1.0)

    return best_a, best_score
