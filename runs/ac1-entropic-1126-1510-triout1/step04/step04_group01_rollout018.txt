"""
Propose a low‑score step‑function.

The routine builds an arcsine‑type distribution (which already has a fairly
small ∥f∗f∥∞) and then repeatedly improves it by solving a linear programme.
For a given distribution f we maximise Σ g_j under the linear constraints
   (f ∗ g)_k ≤ M   (M = max_k (f∗f)_k) .
The solution g has a larger total mass; after normalising it we mix it with f.
A back‑tracking line search guarantees a strict decrease of the peak of the
self‑convolution.  Several sequence lengths are tried within the time budget;
the best sequence found is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Build the matrix A (size (2n‑1)×n) such that A·g = f∗g."""
    n = a.shape[0]
    L = 2 * n - 1
    # index matrix k‑j
    idx = np.arange(L)[:, None] - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    A = np.zeros((L, n), dtype=np.float64)
    A[mask] = a[idx[mask]]
    return A


def _lp_improve(a0: np.ndarray, deadline: float, max_iter: int = 15):
    """
    Improve a distribution a0 by repeatedly solving the LP that maximises Σg
    under (a0∗g)_k ≤ M = max(a0∗a0).  Returns the best vector found and its
    exact evaluation score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current peak of the self‑convolution
        M = _fft_self_convolve(a).max()

        # linear constraints (a ∗ g)_k ≤ M
        A = _build_conv_matrix(a)
        L = A.shape[0]
        c = -np.ones(a.shape[0])                     # maximise Σg  ⇔ minimise –Σg
        res = linprog(c, A_ub=A, b_ub=np.full(L, M),
                      bounds=(0, None), method='highs')
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:                      # cannot improve any further
            break

        g_norm = g / sum_g                            # bring back to Σ=1

        # back‑tracking line search: find a mixing weight that strictly lowers M
        t = 1.0
        improved = False
        while t > 1e-4:
            cand = (1.0 - t) * a + t * g_norm
            cand /= cand.sum()
            M_new = _fft_self_convolve(cand).max()
            if M_new < M - 1e-14:
                a = cand
                improved = True
                break
            t *= 0.5

        if not improved:
            break

        # evaluate exact score
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # early exit if we are already very good
            if best_score <= 1.45:
                break

    return best_a, best_score

    """

    LP‑based improvement described above.  The best sequence encountered is

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # moderate lengths for which the LP is still tractable
    candidate_lengths = [800, 1200, 1600, 2000, 2500]
