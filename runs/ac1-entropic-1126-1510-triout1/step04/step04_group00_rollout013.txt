def _solve_lp_for_g(f):
    n = len(f)
    # Build A_ub, b_ub
    # A_ub shape (2n-1, n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        row_start = i
        row_end = i + n  # exclusive, so indices i to i+n-1, length n
        # row indices: i to i+n-1 inclusive, length n
        rows.extend(np.arange(row_start, row_end))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    M = (np.convolve(f, f)).max()
    # bounds: g >= 0
    bounds = [(0, None)] * n
    c = -np.ones(n)  # minimize -sum(g)
    # Solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback
        return None
    g = res.x
    return g

"""
Hybrid LP‑+‑gradient search for a step‑function minimizing
    score = 2·n·max(a*a) / (sum(a)**2)

The algorithm explores several lengths n (including the length of the
previous best construction), creates a few simple initial probability
vectors (uniform, random, arcsine), and then iterates two complementary
phases:

1. **Linear‑programming phase** – given the current vector f,
   solve the LP
        maximise   Σ g_i
        subject to f * g ≤ M,   g ≥ 0
   where M = max(f*f).  The LP has 2·n‑1 linear constraints and n variables.
   The solution g is renormalised to a probability vector and mixed with f.
   A few LP steps already lower the maximum convolution considerably.

2. **Projected‑gradient phase** – after the LP phase a few hundred (or few
   thousand) iterations of a momentum‑based projected gradient descent are
   applied.  The gradient is the exact sub‑gradient of the max‑operator
   (or a soft‑max surrogate while a temperature parameter α is still small).

Both phases keep the iterate on the simplex by Euclidean projection.
The best vector found across all lengths is finally verified with the
provided `evaluate_sequence` function and returned as a plain list of
non‑negative heights.

The code respects the 1000 s time budget, prints progress information
and falls back to a uniform vector if nothing better is discovered.
"""

import time
import numpy as np
import scipy.sparse
import scipy.optimize


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)) of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _initial_vector(kind: str, n: int) -> np.ndarray:
    """Create a feasible probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    return _project_onto_simplex(a, s=1.0)


def _build_convolution_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the matrix A (size (2n‑1)×n) such that (f * g) = A @ g.
    Each column i of A contains the value f[i] on rows i … i+n‑1.
    """
    n = f.shape[0]
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    data = np.empty(nnz, dtype=np.float64)

    idx = 0
    rng = np.arange(n, dtype=np.int32)
    for i in range(n):
        rows[idx:idx + n] = i + rng
        cols[idx:idx + n] = i
        data[idx:idx + n] = f[i]
        idx += n

    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _solve_lp_for_g(f: np.ndarray, M: float, time_limit: float = None) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   s.t.   f * g ≤ M , g ≥ 0 .
    Returns the optimal g (may be un‑normalised), or None if the solver fails.
    """
    n = f.shape[0]
    A = _build_convolution_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -∑g_i
    bounds = [(0, None)] * n

    options = {}
    if time_limit is not None:
        # the underlying HiGHS solver accepts a time_limit option (seconds)
        options["time_limit"] = time_limit

    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method="highs", options=options)
    if not res.success:
        return None
    return res.x


def _refine_by_gradient(a: np.ndarray,
                        max_iters: int,
                        deadline: float) -> np.ndarray:
    """Projected gradient descent on the max‑convolution objective."""
    n = a.size
    velocity = np.zeros_like(a)
    beta = 0.9
    eta0 = 0.5
    alpha = 5.0                # temperature for the soft‑max surrogate
    alpha_factor = 1.003

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _fft_self_convolve(a)
        cur_max = conv.max()

        if alpha < 500.0:
            # smoothed max (soft‑max) gradient
            exp_term = np.exp(alpha * (conv - cur_max))
            w = exp_term / exp_term.sum()
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]
            alpha = min(alpha * alpha_factor, 500.0)
        else:
            # exact sub‑gradient
            eps = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(conv >= cur_max - eps)
            w = np.zeros_like(conv)
            w[active] = 1.0
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1:n - 1 + n]

            # diagonal correction for terms where i=j (k = 2i)
            even_mask = (active % 2 == 0)
            diag_idx = (active[even_mask] // 2)
            if diag_idx.size > 0:
                grad[diag_idx] -= a[diag_idx]

        velocity = beta * velocity + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(it)
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to escape plateaus
        if it % 2000 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score non‑negative step‑function.
    Returns a plain Python list of heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safe margin for finalisation

    # ----------------------------------------------------------------------
    # Lengths to explore – try a handful around the historic length.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 3000

    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)
            if int(base_n * f) >= 500
        )
    )
    # keep the list reasonably small
    if len(lengths) > 6:
        lengths = lengths[:6]

    # ----------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")
    target_score = 1.503  # we stop early once we beat this

    for n in lengths:
        if time.time() > deadline:
            break

        # allocate a slice of the remaining time to this length
        remaining = max(0.0, deadline - time.time())
        slice_deadline = time.time() + min(remaining * 0.30, 180.0)

        init_kinds = ("prev", "arcsine", "uniform", "random")
        for kind in init_kinds:
            if time.time() > slice_deadline:
                break

            # ----- initialise -------------------------------------------------
            f = _initial_vector(kind, n)
            best_local_f = f.copy()
            conv = _fft_self_convolve(f)
            best_local_M = conv.max()
            best_local_score = 2.0 * n * best_local_M

            # ----- LP phase ---------------------------------------------------
            max_lp_iters = 6
            t_mix = 0.5  # mixing coefficient
            for lp_it in range(max_lp_iters):
                if time.time() > slice_deadline:
                    break

                # Solve the linear programme for the current f
                g = _solve_lp_for_g(f, best_local_M, time_limit=30.0)
                if g is None:
                    break
                # renormalise g to a probability vector
                sum_g = g.sum()
                if sum_g <= 0:
                    break
                g = g / sum_g

                # convex combination
                f = (1.0 - t_mix) * f + t_mix * g
                f = _project_onto_simplex(f, s=1.0)

                # evaluate new max‑convolution
                conv = _fft_self_convolve(f)
                cur_M = conv.max()
                cur_score = 2.0 * n * cur_M

                if cur_M < best_local_M - 1e-15:
                    best_local_M = cur_M
                    best_local_f = f.copy()
                    best_local_score = cur_score
                    print(
                        f"[{time.time():.1f}] n={n} init={kind} LP‑iter={lp_it+1} "
                        f"maxB={best_local_M:.8e} score≈{best_local_score:.6f}"
                    )
                    if cur_score <= target_score:
                        break

                # gradually reduce mixing to avoid large jumps later
                t_mix = max(0.1, t_mix * 0.9)

            # ----- gradient‑descent fine‑tuning -------------------------------
            gd_deadline = min(deadline, slice_deadline)
            f = best_local_f.copy()
            f = _refine_by_gradient(f, max_iters=5000, deadline=gd_deadline)

            # final local evaluation
            conv = _fft_self_convolve(f)
            final_M = conv.max()
            final_score = 2.0 * n * final_M

            # keep the best across all runs
            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = f.copy()
                print(
                    f"*** New global best → n={n} init={kind} "
                    f"score≈{global_best_score:.6f}"
                )
                if global_best_score <= target_score:
                    print("Target reached – exiting early.")
                    return global_best_seq.tolist()

            # also keep best local for this length (used for possible later mixes)
            if final_score < best_local_score:
                best_local_f = f.copy()
                best_local_score = final_score

        # if time is exhausted, move to the next length
        if time.time() > deadline:
            break

    # ----------------------------------------------------------------------
    # Fallback – uniform vector if nothing else succeeded.
    if global_best_seq is None:

            f"(length = {len(global_best_seq)})"
        )
    except Exception as e:
        print(f"Verification failed: {e}")

    return global_best_seq.tolist()
