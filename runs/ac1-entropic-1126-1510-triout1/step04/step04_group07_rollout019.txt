Now implement compute_score:

```
def compute_score(a):
    sum_a = np.sum(a)
    if sum_a == 0: return np.inf
    # compute convolution via FFT
    n = a.shape[0]
    L = 1 << (2*n - 1).bit_length()  # next power of 2
    fft_a = np.fft.rfft(a, L)
    conv = np.fft.irfft(fft_a * fft_a, L)[:2*n-1]
    max_b = conv.max()
    return float(2 * n * max_b / (sum_a * sum_a))
```

Works.

def _conv_fft(a, b):
    n = len(a) + len(b) - 1
    L = 1 << ((n - 1).bit_length())
    fft_a = np.fft.rfft(a, L)
    fft_b = np.fft.rfft(b, L)
    conv = np.fft.irfft(fft_a * fft_b, L)
    return conv[:n]
```

"""
Improved search for a low‑score step function.

The objective of `evaluate_sequence` (with the ℓ¹‑norm fixed to √(2·n))
is simply the maximum value of the linear convolution a*a.
We therefore optimise the probabilities p_i = a_i / Σa on the simplex.
The smooth‑max surrogate

    F_μ(p) = max_k (p*p)[k] + μ·log Σ_k exp((p*p)[k]/μ)

is convex in the lifted variables and has a cheap gradient that can be
computed with FFT‑based convolutions.  A continuation on μ (large → 0)
combined with L‑BFGS‑B gives a very good approximation of the true
hard‑max problem.  After the continuation we polish the incumbent with
a projected sub‑gradient descent on the real max, a cheap random‑block
search and a simple hill‑climbing move‑step.  All heavy convolutions
are performed with FFT, which makes the whole routine fast enough to
handle n≈5000 within the 1000 s budget.

The routine respects the time limit, prints progress messages and
returns the best sequence it found (as a plain Python list of floats).
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real‑FFT (returns length len(a)+len(b)-1)."""
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)
    return conv[:n]


def _score_from_a(a: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float("inf")
    n = a.shape[0]
    conv = _conv_fft(a, a)                     # length 2·n‑1
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


class _SmoothMaxObjectiveFFT:
    """
    Callable for scipy.minimize – returns (smooth‑max value, gradient)
    using FFT based convolutions.
    """

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        """
        Returns (value, gradient) of the smooth‑max surrogate
        for the current free‑parameters y.
        """
        n = self.n
        S = self.S
        mu = self.mu

        # probabilities on the simplex and the actual heights
        p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
        a = S * p                # Σ a_i = S

        # ----- convolution a*a (length 2·n‑1) -----
        conv = _conv_fft(a, a)

        max_c = conv.max()

        if mu > 0.0:
            diff = (conv - max_c) / mu
            diff = np.clip(diff, -700.0, 700.0)   # protect exp overflow
            exp_diff = np.exp(diff)
            Z = exp_diff.sum()
            w = exp_diff / Z
            f_val = max_c + mu * math.log(Z)
        else:   # hard max (μ = 0)
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            f_val = max_c

        # ----- gradient w.r.t. a -----
        # g_a[i] = 2 * Σ_k w_k * a_{k‑i}
        conv_wr = _conv_fft(w, a[::-1])          # length 3·n‑2
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]     # slice length n

        # ----- chain rule for y (soft‑max Jacobian) -----
        inner = np.dot(g_a, p)                   # scalar Σ_j p_j g_a_j
        grad_y = S * p * (g_a - inner)

        return float(f_val), grad_y.astype(np.float64)
