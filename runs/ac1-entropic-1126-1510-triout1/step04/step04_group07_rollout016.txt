import time, math, numpy as np

def _project_to_simplex(v, z):
    """Project onto non-negative simplex of sum z."""
    if z <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  The algorithm starts from the known reference
    (``height_sequence_1``) if available, otherwise from a random Dirichlet
    sample, and then performs a large number of cheap *mass‑transfer* moves
    that update the convolution incrementally (O(n) per move).  The step
    size decays with the remaining time, and occasional Gaussian perturbations
    are performed to escape local minima.  The best sequence found before the
    time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine problem size n and target L1 norm S = sqrt(2 n)
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3500
        have_ref = False

    S = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Build an initial candidate a (L1 = S, non‑negative)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        best_a = a_ref.copy()
        conv = np.convolve(best_a, best_a)
        best_score = conv.max()
        # possibly a random alternative
        raw = rng.gamma(1.0, 1.0, size=n)
        a_rand = raw / raw.sum() * S
        conv_rand = np.convolve(a_rand, a_rand)
        if conv_rand.max() < best_score:
            best_a = a_rand.copy()
            conv = conv_rand
            best_score = conv.max()
    else:
        raw = rng.gamma(1.0, 1.0, size=n)
        best_a = raw / raw.sum() * S
        conv = np.convolve(best_a, best_a)
        best_score = conv.max()

    # ------------------------------------------------------------------
    # Helper to update best when a new candidate is strictly better.
    def _update_if_better(a_vec: np.ndarray, conv_vec: np.ndarray) -> None:
        nonlocal best_a, best_score, conv
        sc = conv_vec.max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_vec.copy()
            conv = conv_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    print(f"[init] n={n}, S={S:.6f}, initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Local mass‑transfer search
    L = 2 * n - 1
    max_step_factor_init = 0.5          # early moves may transfer up to 50% of source mass
    min_step_factor = 0.001
    no_improve_counter = 0
    iteration = 0
    # parameters for occasional Gaussian perturbation
    pert_interval = 5000
    pert_sigma = 0.02 * S

    while time.time() < deadline:
        remaining = deadline - time.time()
        # linear decay of step factor
        ratio = max(0.0, remaining / (budget_s - 0.5))
        step_factor = min_step_factor + (max_step_factor_init - min_step_factor) * ratio

        # pick source index i (weighted by current mass to favour large entries)
        if best_a.sum() > 0:
            prob = best_a / best_a.sum()
            i = rng.choice(n, p=prob)
        else:
            i = rng.integers(n)

        # pick destination j uniformly (different from i)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # ensure there is mass to move
        mass_i = best_a[i]
        if mass_i <= 0.0:
            continue

        delta_max = mass_i * step_factor
        if delta_max <= 0.0:
            continue
        delta = rng.random() * delta_max
        if delta <= 0.0:
            continue

        # tentative update of convolution (in‑place copy)
        conv_tmp = conv.copy()
        # linear part: shift of source/subtraction, shift of dest/addition
        conv_tmp[i:i + n] -= 2.0 * delta * best_a
        conv_tmp[j:j + n] += 2.0 * delta * best_a
        # second‑order corrections
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        conv_tmp[idx2i] += delta * delta
        conv_tmp[idx2j] += delta * delta
        conv_tmp[idxij] -= 2.0 * delta * delta

        # evaluate new max
        new_score = conv_tmp.max()
        iteration += 1

        if new_score < best_score - 1e-12:
            # accept move
            best_a[i] -= delta
            best_a[j] += delta
            conv = conv_tmp
            best_score = new_score
            no_improve_counter = 0
            if iteration % 1000 == 0:
                print(f"[iter {iteration}] improved score = {best_score:.6f}")
        else:
            no_improve_counter += 1

        # occasional global perturbation to jump out of plateaus
        if no_improve_counter >= pert_interval:
            cand = best_a + rng.normal(scale=pert_sigma, size=n)
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s > 0:
                cand = cand * (S / s)
            else:
                cand = np.full(n, S / n)
            conv_cand = np.convolve(cand, cand)
            cand_score = conv_cand.max()
            if cand_score < best_score - 1e-12:
                best_a = cand.copy()
                conv = conv_cand.copy()
                best_score = cand_score
                print(f"[perturb] score improved to {best_score:.6f}")
            no_improve_counter = 0

    # ------------------------------------------------------------------
    # Final sanity check using the provided evaluate_sequence
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] best score = {best_score:.6f} (evaluate_sequence = {final_score:.6f})")
    return best_a.tolist()

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σ x = z}.
    If z ≤ 0 an all‑zero vector is returned.
    """
    if z <= 0.0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]                    # sorted descending
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0

    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w

    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.

    The algorithm proceeds as follows:

    1. Determine the length *n* (from the global ``height_sequence_1`` if
       available, otherwise a default of 3500) and set the target ℓ¹‑norm
       ``S = sqrt(2·n)``.
    2. Initialise a candidate:
         – use the provided reference (projected onto the simplex) if it
           exists,
         – otherwise draw a random Dirichlet sample.

       can be updated in O(n) time using the identity derived from
       ``(a‑δe_i+δe_j)*(a‑δe_i+δe_j)``.
       A move is accepted only if it strictly reduces the maximum of the
       convolution.  The maximal transferable fraction decays linearly with
       the remaining time, allowing large jumps early on and fine
       adjustments later.
    4. If a long stretch of moves yields no improvement, a Gaussian
       perturbation of the current best vector is tried (followed by a
       projection onto the simplex).  This helps to escape local minima.
    5. When the time budget is exhausted a final call to the provided
       ``evaluate_sequence`` checks the result and the best sequence is
       returned.

    The whole procedure respects the 1000 s time budget and prints a
    progress log.
    """

            conv = conv_rand
            best_score = conv.max()
    else:
        raw = rng.gamma(1.0, 1.0, size=n)
        best_a = raw / raw.sum() * S
        conv = np.convolve(best_a, best_a)
        best_score = conv.max()
