"""
Fast gradient‑based search for a low‑score non‑negative step function.

The algorithm works on the simplex (sum = 1) so the objective of
`evaluate_sequence` reduces to ``2·n·max(a*a)``.
It repeats the following for several lengths and random restarts:

1. Initialise a probability vector (arcsine distribution, a
   random Dirichlet, or an interpolated previous solution).

2. Adam optimisation of a smooth‐max surrogate
   ``Lα(a) = (1/α)·log Σ exp(α·(a*a)_k)`` with a slowly increasing
   temperature α.  The gradient is obtained by a weighted
   convolution, and after each step the iterate is projected back onto
   the simplex.

3. Short refinement by sub‑gradient descent on the exact ℓ∞‑objective.
   The sub‑gradient uses an uniform weight over all indices attaining
   the current maximal convolution value.

4. The best candidate seen so far (according to the fast FFT estimate)
   is periodically evaluated with the supplied ``evaluate_sequence`` to
   obtain the true score.

Time is split among a few lengths (≈5000–12000).  The total runtime is
limited to the given budget (default 1000 s).  The returned value is a
plain Python list ready for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length n (already normalised)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative sequence to a new length by linear interpolation."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray,
                         alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

    Returns (grad, cur_max) where cur_max = max_k (a*a)_k.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()

    # soft‑max weights (stable)
    # shift by cur_max to avoid overflow
    shifted = alpha * (b - cur_max)
    # overflow protection – clamp to a safe range
    shifted = np.clip(shifted, -700, 0)          # exp(-700) ≈ 5e-305
    exp_term = np.exp(shifted)
    w_sum = exp_term.sum()
    if w_sum == 0.0:
        # Fallback: uniform weight on the maximal entries
        w = np.zeros_like(b)
        active = np.isclose(b, cur_max, atol=1e-12)
        cnt = np.count_nonzero(active)
        if cnt > 0:
            w[active] = 1.0 / cnt
    else:
        w = exp_term / w_sum

    # gradient = 2·rev(a) * w   (restricted to the n entries of a)
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.25,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max objective.
    Returns the best sequence encountered and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate schedule (diminishing)
        lr = lr0 / np.sqrt(t)

        # update & project onto simplex
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on all indices attaining the current maximum.
    Returns (grad, cur_max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # active set (tiny absolute tolerance)
    tol = max(1e-14, cur_max * 1e-14)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short refinement using the exact max (sub‑gradient descent).
    Runs until the time deadline is reached.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.025 / np.sqrt(it)      # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 400 == 0 and np.random.rand() < 0.09:
            a += np.random.randn(n) * 6e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain Python list ready for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.8   # small safety margin

    # ------------------------------------------------------------------
    # Load a previous construction if present
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        base_exists = True
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_exists = False
        base_n = 8000

    # ------------------------------------------------------------------
    # Choose a handful of lengths (larger lengths often give a tighter constant)
    factors = (0.8, 1.0, 1.2, 1.5, 1.8)
    lengths = sorted({max(500, int(base_n * f)) for f in factors})
    # keep the list short – we only have limited time
    lengths = lengths[:5]

    overall_best_seq = None
    overall_best_score = float("inf")

    # allocate roughly equal time per length
    time_per_length = max(2.0, (deadline - start_time) / max(1, len(lengths)))

    for n in lengths:
        if time.time() >= deadline:
            break
        length_deadline = time.time() + time_per_length

        # ------------------------------------------------------------------
        # Several random restarts for each length
        restarts = 4
        for r in range(restarts):
            if time.time() >= length_deadline:
                break

            # --------------------------------------------------------------
            # initialise a0 on the simplex
            if r == 0:
                a0 = _arcsine_distribution(n)
            elif r == 1 and base_exists:
                a0 = _interpolate_to_length(base_seq, n)
            elif r == 2:
                a0 = np.random.rand(n)
                a0 /= a0.sum()
            else:
                # Mix arcsine and a random Dirichlet
                mix = np.random.rand()
                a_arc = _arcsine_distribution(n)
                a_rand = np.random.rand(n)
                a_rand /= a_rand.sum()
                a0 = mix * a_arc + (1 - mix) * a_rand

            # small jitter before optimisation
            a0 += np.random.randn(n) * 2e-6
            a0 = np.maximum(a0, 0.0)
            a0 = _project_onto_simplex(a0, s=1.0)

            # --------------------------------------------------------------
            # Phase‑1 : Adam on smooth‑max surrogate
            phase1_deadline = min(length_deadline,
                                  time.time() + (length_deadline - time.time()) * 0.58)
            a_opt, peak1 = _adam_optimize(a0, phase1_deadline,
                                          alpha_max=8000.0,
                                          lr0=0.3)

            # --------------------------------------------------------------
            # Phase‑2 : short exact‑max refinement
            phase2_deadline = min(length_deadline,
                                  time.time() + (length_deadline - time.time()) * 0.42)
            a_refined = _refine_max(a_opt, phase2_deadline)

            # --------------------------------------------------------------
            # Fast evaluation (FFT based)
            b = _fft_self_convolve(a_refined)
            cur_max = b.max()
            fast_score = 2.0 * n * cur_max

            if fast_score < overall_best_score - 1e-12:
                overall_best_score = fast_score
                overall_best_seq = a_refined.copy()
                print(f"[{time.time():.1f}] New fast best: len={n} restart={r} "
                      f"score={fast_score:.6f}")

            # --------------------------------------------------------------
            # True black‑box evaluation (cheap compared to the remaining budget)
            try:
                true_score = evaluate_sequence(overall_best_seq.tolist())
                if true_score < overall_best_score - 1e-12:
                    overall_best_score = true_score
                    overall_best_seq = np.asarray(overall_best_seq, dtype=np.float64)
                    print(f"[{time.time():.1f}] New true best: len={n} restart={r} "
                          f"score={true_score:.6f}")
            except Exception as exc:
                print("evaluate_sequence raised:", exc)

            # early exit if we already beat the target
            if overall_best_score <= 1.503:
                break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback uniform used.")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
