"""
Propose a low‑score step‑function.

The algorithm is a single long optimisation run on a relatively large
problem size.  It proceeds in two stages:

1.  Adam optimisation on a smooth soft‑max approximation of the ℓ∞
    self‑convolution peak.  The temperature (α) is slowly increased so
    that the landscape becomes progressively sharper.  This stage
    consumes ≈70 % of the time budget.

2.  A long projected (sub‑)gradient descent on the exact ℓ∞ objective.
    The direction is the convex combination of the gradients of the
    active (max‑attaining) convolution entries (uniform weighting).  The
    step size decays as 1/√t and we perform a simple back‑tracking line
    search when the peak does not improve.  This stage consumes the
    remaining time.

The initial distribution is a discretised arcsine law (the current
state‑of‑the‑art starting point).  All iterates are projected onto the
probability simplex after each update.  The best sequence seen (according
to the supplied ``evaluate_sequence`` black‑box) is returned.

The routine respects the 1000 s time limit, uses only NumPy and the
provided ``evaluate_sequence`` function, and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # Stabilised soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    # Gradient = 2 * rev(a) convolve w  (restricted to n entries)
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max

    Returns the best iterate (as a probability vector) and its exact peak.
    """

    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Learning‑rate schedule (inverse‑sqrt decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep best point w.r.t. the exact (non‑smooth) peak
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 700 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    Compute a sub‑gradient of the exact ℓ∞ objective.
    Returns (grad, cur_max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # active set (indices attaining the maximum up to tiny tolerance)
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray, deadline: float):
    """
    Projected sub‑gradient descent on the true ℓ∞ objective.
    Returns the best iterate encountered.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    # initial step size (tuned empirically)
    base_step = 0.03

    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)

        # decay ∝ 1/√it
        step = base_step / np.sqrt(it)

        # simple back‑tracking: if the peak does not improve, shrink step
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate, s=1.0)
        new_peak = _fft_self_convolve(a_candidate).max()
        if new_peak > cur_max:
            # try a reduced step
            for factor in (0.5, 0.25, 0.125):
                a_candidate = a - step * factor * grad
                a_candidate = _project_onto_simplex(a_candidate, s=1.0)
                new_peak = _fft_self_convolve(a_candidate).max()
                if new_peak <= cur_max:
                    step *= factor
                    break

        a = a_candidate
        cur_max = new_peak

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # occasional random jitter to diversify search
        if it % 600 == 0 and np.random.rand() < 0.09:
            a += np.random.randn(n) * 8e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low score.
    Returns a plain Python list that will be supplied to `evaluate_sequence`.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # keep a small margin

    # ------------------------------------------------------------------
    # Choose a problem size.  Larger n gives a finer discretisation and
    # often a lower bound; we pick a size that still fits comfortably in
    # memory and allows many iterations within the time budget.
    n = 20000
    print(f"[{time.time():.1f}] Starting optimisation with length {n}")

    # initialise with a discretised arcsine law (empirically strong)
    a0 = _arcsine_distribution(n)
    a0 = _project_onto_simplex(a0, s=1.0)

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth max (≈70 % of total time)
    phase1_deadline = time.time() + 0.70 * (deadline - start)
    a_phase1, peak_phase1 = _adam_optimize(a0, phase1_deadline)
    fast_score = 2.0 * n * peak_phase1
    print(f"[{time.time():.1f}] Phase 1 finished: peak={peak_phase1:.6e}, "
          f"fast_score={fast_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent (remaining time)
    a_best = a_phase1
    best_peak = peak_phase1
    if time.time() < deadline:
        a_phase2 = _subgradient_optimize(a_best, deadline)
        best_peak = _fft_self_convolve(a_phase2).max()
        a_best = a_phase2

    # ------------------------------------------------------------------
    # Final evaluation via the black‑box (cheap compared to FFT work)
    try:
        true_score = evaluate_sequence(a_best.tolist())
        print(f"[{time.time():.1f}] Final true score = {true_score:.6f}")
    except Exception as exc:
        # Fallback to the fast estimate if something goes wrong
        true_score = 2.0 * n * best_peak
        print("evaluate_sequence raised an exception:", exc)
        print(f"Falling back to fast estimate: {true_score:.6f}")

    # Return the best sequence found
    return a_best.tolist()
