"""
Propose a low‑score step‑function.

The routine uses a soft‑max parametrisation (so the ℓ¹‑norm is fixed to 1)
and minimises a smooth approximation of the infinity‑norm of the
autoconvolution by Adam with a geometric temperature schedule.
Several random restarts, a large random‑block search, permutation
heuristics (descending, alternating extremes, random shuffles) and a
simple hill‑climbing phase are performed, all respecting the 1000 s
budget.  Convolutions are carried out with the FFT for speed.
The best sequence found is returned (as a plain list of floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, put the whole mass on the largest entry
        w = np.zeros_like(v)
        w[np.argmax(v)] = z
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (outputs a probability vector, Σ = 1)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / np.sum(e)


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (int(np.ceil(np.log2(x))) if x > 1 else 0)


# ----------------------------------------------------------------------
def _linear_conv_fft(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """
    Linear (non‑circular) convolution using the FFT.
    Returns only the first len(a)+len(b)-1 entries.
    """
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    C = A * B
    conv = np.fft.irfft(C, size)
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
def _smooth_max_obj_and_grad(
    y: np.ndarray,
    mu: float,
    n: int,
    size_a: int,
    size_wr: int,
) -> tuple[float, np.ndarray, float]:
    """
    Smooth‑max surrogate value, its gradient w.r.t. the free parameters `y`,
    and the (raw) maximum of the convolution (needed for the true score).
    The parameters `size_a` and `size_wr` are FFT sizes pre‑computed for
    the two convolutions that appear in the gradient.
    """
    # probabilities on the simplex
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = p                    # we keep the ℓ¹‑norm = 1

    # ----- first convolution a * a ------------------------------------
    conv = _linear_conv_fft(a, a, size_a)           # length 2·n‑1
    max_c = conv.max()

    # ----- smooth‑max weight vector w -------------------------------
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:  # hard max
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # ----- gradient w.r.t. the heights a ----------------------------
    a_rev = a[::-1]
    conv_wr = _linear_conv_fft(w, a_rev, size_wr)   # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]          # slice length n

    # ----- chain rule for the free parameters y (soft‑max Jacobian) -
    inner = np.dot(g_a, p)                           # scalar Σ_j p_j g_a_j
    grad_y = p * (g_a - inner)                       # shape (n,)

    return float(f_val), grad_y.astype(np.float64), float(max_c)


# ----------------------------------------------------------------------
def _score_from_a(a: np.ndarray, n: int, size_a: int) -> float:
    """
    Exact evaluate_sequence score (but without the clipping that evaluate_sequence
    performs – our a stays in the valid range).  Returns +∞ if the ℓ¹‑norm is
    too small.
    """
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = _linear_conv_fft(a, a, size_a)
    max_c = conv.max()
    return float(2.0 * n * max_c / (s * s))

    (seconds) and returns the best sequence discovered.

            a[i1], a[i2] = a[i2], a[i1]          # perform swap
            sc = _score_from_a(a, n, size_a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[improve] swap improved to {best_score:.6f}")
            else:
                # revert – we keep the previous best configuration
                a[i1], a[i2] = a[i2], a[i1]

    # ------------------------------------------------------------------
    # final fine‑tuning (smooth‑max with a tiny temperature)
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("[final‑opt] polishing the best candidate")
        # initialise y from the current best (log‑softmax trick)
        eps = 1e-12
        y = np.log(best_a + eps)

        # start temperature again, but much smaller number of steps
        a_cur = _softmax(y)
        conv_cur = _linear_conv_fft(a_cur, a_cur, size_a)
        mu = max(conv_cur.max(), 1.0)

        m = np.zeros_like(y)
        v = np.zeros_like(y)
        t_iter = 0
        while mu > 1e-6 and time.time() < deadline:
            for _ in range(100):
                if time.time() > deadline:
                    break
                t_iter += 1
                f_val, grad, _ = _smooth_max_obj_and_grad(y, mu, n, size_a, size_wr)
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad ** 2)
                m_hat = m / (1.0 - beta1 ** t_iter)
                v_hat = v / (1.0 - beta2 ** t_iter)
                lr = (lr_initial * 0.5) / (1.0 + 0.002 * t_iter)
                y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                a = _softmax(y)
                sc = _score_from_a(a, n, size_a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[improve] final polish improved to {best_score:.6f}")
            mu *= 0.7   # finish quicker

    # ------------------------------------------------------------------
    # final verification with the official evaluate_sequence (clipping etc.)
    final_seq = best_a.tolist()
    official_score = evaluate_sequence(final_seq)
    print(f"[final] official score = {official_score:.6f}")
    return final_seq
