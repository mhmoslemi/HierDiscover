```
def adam_optimize(init_a, n, deadline):
    a = init_a.copy()
    best = a.copy()
    best_val = 2*n * convolve_self_fft(a).max()
    # Adam params
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    iteration = 0
    alpha = 5.0
    alpha_mul = 1.05
    alpha_max = 50000.0
    size_fft = 1 << ((3*n - 2).bit_length())
    while time.time() < deadline and (max_iter is None or iteration < max_iter):
        iteration += 1
        # compute convolution and score
        b = convolve_self_fft(a)
        cur_score = 2 * n * b.max()
        if cur_score < best_val:
            best_val = cur_score
            best = a.copy()
        # gradient
        grad = gradient_via_softmax(a, b, alpha, size_fft)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = 0.1 / (np.sqrt(v_hat) + eps)  # step size scaling
        a = a - step * m_hat
        # project
        a = project_onto_simplex(a, s=1.0)
        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)
    return best, best_val
```

The program searches for a non‑negative step function (a probability vector) that minimises  

\[
\text{score}= 2\,n\;\max_k (a*a)_k
\]

because `evaluate_sequence` is scale‑invariant.  
The algorithm consists of three stages

1. **Parametric warm‑start** – enumerate a few simple families  
   (uniform, arcsine, Beta‑shaped, mixtures with spikes) and keep the
   best one.  This gives a good initial direction without any optimisation.

2. **Projected Adam optimisation** – run Adam on the probability simplex.
   The true objective `max(conv)` is replaced by a soft‑max with an
   exponentially increasing temperature; for large temperature the
   gradient is an exact sub‑gradient of the max‑convolution.
   After each Adam step the vector is projected back onto the simplex.

3. **Stochastic hill climbing** – repeatedly move a random fraction of mass
   from one index to another (and occasionally jitter the whole vector).
   Every successful move strictly reduces the objective, giving a cheap
   final polishing.

All work respects the 1000 s budget; the last stage receives the
remaining time, so the optimiser can run for many thousands of
iterations on the currently best length.  The function `propose_candidate`
conforms to the required signature and returns a plain Python list.

```python
"""
Proposes a non‑negative step function (list of heights) that minimises
`evaluate_sequence`.  The algorithm works in three phases:

1. Fast parametric search over simple families (uniform, arcsine,
   symmetric Beta, mixtures with uniform/spikes) to obtain a good
   starting shape.
2. Projected Adam optimiser on the simplex, using a soft‑max surrogate
   for the max‑convolution and an exponentially increasing temperature.
3. Stochastic hill‑climbing (mass swaps + tiny jitter) for final polishing.

All work is bounded by the 1000 s time budget; the final best sequence
is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.stats import beta as beta_dist  # Beta pdf for the parametric family


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) with respect to a.
    For huge α this is a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # --- soft‑max weights -------------------------------------------------
    if alpha > 2_000:
        # treat every entry that is numerically maximal as active
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()                # for stability
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # --- convolution w * reversed(a) ------------------------------------
    a_rev = a[::-1]                        # reversed vector
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]        # factor 2 comes from derivative of a*a
    return g


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float) -> np.ndarray:
    """Symmetric Beta(a,a) pdf evaluated on mid‑points of n bins."""
    # avoid a too close to zero (pdf would explode)
    a = max(a, 0.05)
    x = (np.arange(n) + 0.5) / n
    # pdf ∝ x^{a-1} (1-x)^{a-1}
    pdf = np.power(x, a - 1.0) * np.power(1.0 - x, a - 1.0)
    # normalise
    pdf_sum = pdf.sum()
    if pdf_sum == 0.0 or not np.isfinite(pdf_sum):
        pdf = np.full(n, 1.0 / n)
    else:
        pdf /= pdf_sum
    return pdf


# ----------------------------------------------------------------------
def _candidate_shapes(n: int, prev_seq: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest list of hand‑crafted candidate probability vectors."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) arcsine (Beta(½,½))
    candidates.append(_arcsine_distribution(n))

    # 3) symmetric Beta(a,a) for a in a modest grid
    for a in np.linspace(0.25, 0.95, 13):
        candidates.append(_beta_distribution(n, a))

    # 4) mixtures uniform + arcsine
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    arcsine = _arcsine_distribution(n)
    for w in np.linspace(0.0, 0.5, 6):
        mix = (1.0 - w) * arcsine + w * uniform
        candidates.append(mix)

    # 5) add tiny spikes at the boundaries to the arcsine shape
    for spike_frac in (0.10, 0.20):
        mix = arcsine.copy()
        mix[0] += spike_frac / 2
        mix[-1] += spike_frac / 2
        mix = _project_onto_simplex(mix, 1.0)
        candidates.append(mix)

    # 6) previous best (if supplied and length matches)
    if prev_seq is not None and prev_seq.size == n:
        candidates.append(prev_seq.copy())

    # 7) a few random Dirichlet draws
    rng = np.random.default_rng()
    for _ in range(4):
        rand = rng.dirichlet(np.ones(n))
        candidates.append(rand)

    # make sure every candidate lies on the simplex
    final = [_project_onto_simplex(c, 1.0) for c in candidates]
    return final


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex.
    Returns the best vector seen and its internal (2·n·max(conv)) score.
    """
    a = init_a.copy()
    best = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 50_000.0

    size_fft = 1 << ((3 * n - 2).bit_length())      # ≥ (2n‑1)+(n)‑1

    it = 0
    while time.time() < deadline:
        it += 1

        # current objective
        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} it={it} score={best_score:.6f}")

        # gradient (soft‑max surrogate)
        grad = _gradient_via_softmax(a, conv, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base learning rate 0.2 works well in practice
        step = 0.2 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # projection back onto the simplex
        a = _project_onto_simplex(a, 1.0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _hill_climb(start_a: np.ndarray,
                n: int,
                deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random fraction of the mass
    from one index to another and keep the move if it improves the score.
    """
    best = start_a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        it += 1
        i, j = rng.choice(n, size=2, replace=False)
        if best[i] < 1e-12:
            continue

        # move up to 30 % of the mass at i to j
        frac = rng.random() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = 2.0 * n * _fft_self_convolve(cand).max()
        if cand_score < best_score:
            best = cand
            best_score = cand_score

        # occasional tiny jitter to escape shallow plateaus
        if it % 5000 == 0:
            jitter = best + rng.normal(scale=1e-6, size=n)
            jitter = _project_onto_simplex(jitter, 1.0)
            jitter_score = 2.0 * n * _fft_self_convolve(jitter).max()
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the grader.
    Returns a list of non‑negative heights (probability vector) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    np.random.seed(seed)
    # small safety margin for the final conversion / printing
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine a sensible length.  Prefer the length of a previously
    # discovered solution (height_sequence_1) but stay below a safe ceiling.
    try:
        base_n = int(len(height_sequence_1))   # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Clamp to a reasonable interval (2 k … 15 k)
    n = max(2000, min(base_n, 15000))

    # ------------------------------------------------------------------
    # Phase 0 – quick parametric warm‑start
    prev_seq = None
    try:
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        pass

    shape_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.20)
    candidates = _candidate_shapes(n, prev_seq)
    best_seq = None
    best_score = float("inf")

    for cand in candidates:
        if time.time() >= shape_deadline:
            break
        # each candidate is already normalised to sum = 1
        scr = evaluate_sequence(cand.tolist())
        if scr < best_score:
            best_score = scr
            best_seq = cand.copy()
            print(f"[shape] New best score {best_score:.6f} (length={n})")

    # If for some reason we have no candidate (should never happen), fall back:
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # Phase 1 – long Adam optimisation on the current best length
    adam_deadline = time.time() + max(0.0, (deadline - time.time()) * 0.60)
    seq_adam, _ = _adam_optimize(best_seq, n, adam_deadline)

    # ------------------------------------------------------------------
    # Phase 2 – final hill‑climbing polishing
    hill_deadline = deadline
    seq_final, final_score = _hill_climb(seq_adam, n, hill_deadline)

    # ------------------------------------------------------------------
    # Final safety fallback (should never be exercised)
    if final_score is None or math.isinf(final_score):
        seq_final = np.full(2000, 1.0 / 2000, dtype=np.float64)
        final_score = evaluate_sequence(seq_final.tolist())

    print(f"Final best score: {final_score:.6f} (length = {len(seq_final)})")
    return seq_final.tolist()
```
