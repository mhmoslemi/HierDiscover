"""
Improved search for low‑score step‑functions.

The evaluation function equals the maximum value of the autoconvolution
(∥a*a∥∞) when the ℓ¹‑norm of a is fixed to √(2·n).  Consequently we can
work with the simpler objective max_k (a*a)[k] and keep the sum
constraint by Euclidean projection onto the simplex.
The solver consists of three stages:

1️⃣  Smooth‑max L‑BFGS (as in the reference) – gives a good
    non‑convex neighbourhood.
2️⃣  Projected sub‑gradient descent on the true (hard) max – many
    inexpensive iterations, adaptively tuning the step‑size.
3️⃣  Fast random‑perturbation hill‑climbing – cheap cheap‑objective
    evaluations (using an FFT‑based convolution) allow thousands of
    trials in the remaining time.

All phases respect the global 1000 s budget.  The best sequence found
is returned as a plain list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be zero, put everything on first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate value and gradient w.r.t. the free parameters y.
    """
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    conv = np.convolve(a, a)                # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:   # mu == 0  → hard max
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # Jacobian of soft‑max (see matrix‑free formulation)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable required by scipy.optimize – returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _max_convolution_fft(a: np.ndarray) -> float:
    """
    Fast (FFT) computation of max_k (a*a)[k].  a is non‑negative,
    length n.  Returns the maximum entry of the full convolution.
    """
    n = a.shape[0]
    size = 2 * n - 1
    # zero‑pad to next power‑of‑two for speed
    m = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(A * A, n=m)[:size]
    # numerical noise may create tiny negative values – ignore them
    return float(conv.max())


# ----------------------------------------------------------------------
def _hard_subgradient(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of  max_k (a*a)[k] with respect to a.
    Returns (g, max_c) where g = ∂max / ∂a.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    w = np.zeros_like(conv, dtype=float)
    if cnt:
        w[mask] = 1.0 / cnt
    # g = 2 * (w * a_rev)
    wr = np.convolve(w, a[::-1])
    g = 2.0 * wr[n - 1:n - 1 + n]
    return g, max_c

def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects the supplied time‑budget (seconds) and
    returns the best sequence it discovers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    safety = 5.0                      # leave a few seconds for cleanup
    deadline = start + budget_s - safety

    # --------------------------------------------------------------
    # problem size (use reference length if available)
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3500
        have_ref = False

    S = math.sqrt(2.0 * n)           # target ℓ¹‑norm (scale‑invariant)

    # --------------------------------------------------------------
    # bookkeeping
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # cheap update using max‑convolution (exact when Σa = S)
    def _maybe_update(a_vec: np.ndarray) -> None:
        nonlocal best_score, best_a
        # compute cheap objective
        cand_score = _max_convolution_fft(a_vec)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_vec.copy()
            print(f"[improve] new best (cheap) = {best_score:.8f}")

    # --------------------------------------------------------------
    # Build a collection of starting points (parameter vectors y for soft‑max)
    init_ys = []

    # uniform
    init_ys.append(np.zeros(n, dtype=np.float64))

    # decreasing exponential (large on the left)
    alpha = rng.uniform(0.001, 0.008) * n / 1000.0
    y_exp = -alpha * np.arange(n, dtype=np.float64)
    init_ys.append(y_exp)

    # increasing exponential (large on the right)
    y_rev = -alpha * np.arange(n - 1, -1, -1, dtype=np.float64)
    init_ys.append(y_rev)

    # random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # reference (if present)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, S)
        eps = 1e-12
        init_ys.append(np.log(a_ref + eps))

    # --------------------------------------------------------------
    # Phase 1 – smooth‑max L‑BFGS with continuation
    mu_factor = 0.8          # slower cooling gives finer finalisation
    mu_min = 1e-7
    maxiter_per_mu = 300
    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] initialising")
        y = y0.copy()
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = float(conv_cur.max())
        mu = max(mu, 1.0)         # safety against a zero start

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            # cool down
            mu *= mu_factor

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the hard max
    if time.time() < deadline:
        print("[sgd] starting projected sub‑gradient descent")
        a = best_a.copy()
        it = 0
        # initial step‑size based on gradient magnitude
        step_mul = 0.5
        while time.time() < deadline and it < 8000:
            g, max_c = _hard_subgradient(a)
            norm_g = np.linalg.norm(g)
            if norm_g == 0:
                break
            # adaptive step size
            eta = step_mul / norm_g
            cand = a - eta * g
            cand = _project_to_simplex(cand, S)
            cand_score = _max_convolution_fft(cand)

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                # success → slightly enlarge step multiplier
                step_mul = min(step_mul * 1.05, 2.0)
                print(f"[sgd] iter={it} improved to {best_score:.8f}")
            else:
                # reject → shrink step multiplier
                step_mul *= 0.7
                # keep current a for next iteration
            it += 1
            if it % 1000 == 0:
                print(f"[sgd] iter={it}, best={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – fast random hill‑climbing (many cheap trials)
    if time.time() < deadline:
        print("[hill] starting random perturbation hill‑climb")
        a = best_a.copy()
        avg = S / n
        sigma = avg * 0.4               # initial perturbation scale
        it = 0
        while time.time() < deadline:
            # generate a Gaussian perturbation (zero‑mean)
            delta = rng.normal(scale=sigma, size=n)
            cand = a + delta
            cand = _project_to_simplex(cand, S)
            cand_score = _max_convolution_fft(cand)

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                sigma = min(sigma * 1.05, avg)   # slowly broaden after success
                print(f"[hill] iter={it} new best {best_score:.8f}")
            else:
                # no improvement – shrink perturbation a bit
                sigma *= 0.995
            it += 1
            if it % 2000 == 0:
                print(f"[hill] iter={it}, best={best_score:.8f}")

    # --------------------------------------------------------------
    # Final safety check – compute the official score once more
    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] official best score = {final_score:.8f} (cheap ≈ {best_score:.8f})")
    return best_a.tolist()
