def _inv_power_distribution(n, gamma):
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-gamma)
    vals = np.maximum(vals, 0.0)
    vals_sum = vals.sum()
    if vals_sum == 0:
        return _arcsine_distribution(n)
    vals /= vals_sum
    return vals
```

def _exp_decay_distribution(n, lam):
    i = np.arange(n)
    left = np.exp(-lam * i)
    right = np.exp(-lam * (n-1 - i))
    vals = left + right
    # if lam large, right may be small for most.
    vals_sum = vals.sum()
    if vals_sum == 0:
        return np.full(n, 1.0/n)
    vals /= vals_sum
    return vals

"""
Search for a non‑negative step function (probability vector) that minimizes
the black‑box ``evaluate_sequence``.  The routine explores several
parametric families (inverse‑power, exponential decay, arcsine, beta,
random), refines each candidate with an exponentiated‑gradient / mirror‑descent
optimizer that uses a soft‑max surrogate for the L∞ norm of the
self‑convolution, and finishes with a short random‑move hill‑climbing
refinement.  It respects a 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _beta_skewed_distribution(n: int) -> np.ndarray:
    """Beta(0.5,0.5)‑skewed distribution – heavy at the ends, similar to arcsine."""
    a = np.random.beta(0.5, 0.5, size=n)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()

    return a

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]

def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at the index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g

def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b_k) w.r.t. a using the soft‑max weights.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()

        w[np.argmax(b)] = 1.0

    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

def _optimize_exponential(a0: np.ndarray,

def _exp_decay_distribution(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·min(i, n‑1‑i))  (symmetric, normalised)."""
