def propose_candidate(seed=42, budget_s=1000):
    """Search for a non‑negative sequence minimizing the evaluate_sequence metric.
    Uses a large-scale Adam optimizer on a smooth max surrogate, starting from
    the arcsine distribution and progressively increasing the smoothing parameter.
    The optimisation runs until the time budget is exhausted, after which the
    best sequence found is returned.
    """
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0

    # Choose length based on remaining time (larger length gives lower bound).
    # We target roughly 20000 elements which is feasible in the time budget.
    n = 20000
    # If we have very little time left (unlikely), fall back to smaller n.
    if deadline - time.time() < 30:
        n = 5000

    # Initialise with arcsine law (good starting point)
    a = _arcsine_distribution(n)
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = 1.0
    # Hyper‑parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha_factor = 1.01
    alpha_max = 5000.0   # large enough to approximate max well

    # Main optimisation loop
    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak:
            best_peak = cur_max
            best_a = a.copy()
            # optional debug print

        # increase smoothing parameter

    except Exception as exc:
        true_score = fast_score
        print(f"evaluate_sequence failed ({exc}), using fast score {fast_score:.6f}")

"""
Propose a low‑score step‑function.

The search uses a large‑scale Adam optimiser on a smooth approximation of the
ℓ∞‑norm of the self‑convolution.  The initial sequence is the arcsine
distribution (or an interpolated previous best) on a relatively large length
(≈20 000).  The temperature α of the soft‑max surrogate is raised slowly up to
a large value, so the optimiser eventually sees the true maximum.  After the
Adam phase a short exact‑max sub‑gradient refinement is performed.  The whole
routine runs until the given time budget (≈1000 s) is exhausted and returns
the best sequence found (normalised to sum 1, which does not affect the final
score).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative sequence to a new length by linear interpolation."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()  # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # gradient = 2 * reverse(a) * w   (restricted to n entries)
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)               # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short refinement using the exact max (sub‑gradient descent).
    Stops when the time budget expires.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter to escape tiny plateaus
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative height sequence with a low evaluate_sequence score.
    The method works on a large length (≈20 000), starts from the arcsine law
    (or an interpolated previous best), and runs Adam on a smooth‑max surrogate
    until the time budget runs out, finishing with an exact‑max refinement.
    The returned list will be fed to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Choose a length that is large enough to give a low theoretical bound.
    # 20000 gives a lower bound ≈1.0 and is still cheap to FFT‑optimise.
    n = 20000
    if deadline - time.time() < 30.0:      # very little time left → use a smaller n
        n = 5000

    # ------------------------------------------------------------------
    # Initialise a on the simplex.
    try:
        # ``height_sequence_1`` may be defined by the harness – use it if present.
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        a = _interpolate_to_length(base_seq / base_seq.sum(), n)
        init_tag = "interpolated previous"
    except Exception:
        a = _arcsine_distribution(n)
        init_tag = "arcsine"

    a = _project_onto_simplex(a, s=1.0)

    # ------------------------------------------------------------------
    # Adam optimiser on the smooth‑max surrogate.
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = 1.0

    # hyper‑parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha_factor = 1.01
    alpha_max = 5000.0

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep best exact peak seen so far
        if cur_max < best_peak:
            best_peak = cur_max
            best_a = a.copy()
            print(f"[{time.time():.1f}] iter={t} new_best_peak={cur_max:.8f}")

        # increase temperature slowly
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # small random jiggle every few thousand iterations
        if t % 2000 == 0 and np.random.rand() < 0.05:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    # ------------------------------------------------------------------
    # Exact‑max refinement (sub‑gradient descent) until the deadline expires.
    if time.time() < deadline:
        a_refined = _refine_max(best_a, deadline)
    else:
        a_refined = best_a

    # ------------------------------------------------------------------
    # Final reporting
    final_peak = _fft_self_convolve(a_refined).max()
    fast_score = 2.0 * n * final_peak
    try:
        true_score = evaluate_sequence(a_refined.tolist())
        print(f"Final true score = {true_score:.6f} (fast approx = {fast_score:.6f}) "
              f"len={n} init={init_tag}")
    except Exception as exc:
        true_score = fast_score
        print(f"evaluate_sequence raised {exc!r}; using fast score {fast_score:.6f}")

    return a_refined.tolist()
