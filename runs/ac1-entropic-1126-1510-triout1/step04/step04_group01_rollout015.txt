def _simulated_annealing(a, deadline):
    n = len(a)
    cur_a = a.copy()
    cur_max = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_max = cur_max

    start = time.time()
    total = deadline - start
    # target temperatures:
    start_temp = cur_max * 0.5
    final_temp = cur_max * 1e-5
    if start_temp <= 0:
        start_temp = 1.0
    if final_temp <= 0:
        final_temp = 1e-8
    cooling_rate = -math.log(final_temp/start_temp) / total if total >0 else 1.0

    iter = 0
    while time.time() < deadline:
        iter += 1
        # Choose two distinct indices
        i = np.random.randint(n)
        if cur_a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # Determine delta: random fraction of source mass, up to 5% maybe
        max_delta = min(cur_a[i], 0.05)  # limit step size
        delta = np.random.rand() * max_delta
        # Propose new a
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # Compute new max
        new_max = _fft_self_convolve(a_new).max()
        # acceptance
        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)
        delta_obj = new_max - cur_max
        accept = False
        if delta_obj < 0:
            accept = True
        else:
            # compute acceptance probability
            prob = math.exp(-delta_obj / temperature) if temperature > 0 else 0.0
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur_a = a_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = cur_a.copy()
        # optional occasional random restart (e.g., 0.001 chance)
        if iter % 10000 == 0 and np.random.rand() < 0.01:
            # reinitialize to best or random
            cur_a = best_a.copy()
            cur_max = best_max
    return best_a, best_max

"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1.  **Broad search** – generate a few candidate distributions for several
    lengths (arcsine, uniform, random, a small grid of Beta‑densities,
    and simple mixtures).  The best candidate according to the fast
    ℓ∞‑convolution surrogate is kept.

2.  **Gradient‑based refinement** – run Adam on a smooth soft‑max
    approximation of the peak, then a short exact‑max sub‑gradient
    descent (the same machinery used in the previous solution).

3.  **Stochastic polishing** – a simulated‑annealing phase that moves
    small amounts of probability mass between positions while accepting
    uphill moves with a temperature schedule.  The result is finally
    evaluated with the black‑box `evaluate_sequence`.

All steps respect the simplex constraint Σaᵢ=1, aᵢ≥0, and the total
running time is limited to ``budget_s`` seconds (default 1000 s).
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) distribution on a uniform grid, normalised to sum 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid overflow for α,β<1
    log_vals = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    vals = np.exp(log_vals)
    # clean up possible infinities
    vals[~np.isfinite(vals)] = 0.0
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.005,
                   alpha_max: float = 2000.0,
                   lr0: float = 0.2,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best probability vector encountered and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        lr = lr0 / np.sqrt(t)

        # update & project back onto the simplex
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to avoid stagnation
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the exact ℓ∞ objective.
    Uses a uniform weight on the active set (indices attaining the current maximum).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short exact‑max refinement using sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.02 / np.sqrt(it)          # diminishing step
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        # occasional jitter
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray, deadline: float):
    """
    Simple simulated‑annealing polishing.
    Moves a small amount of mass between two random positions,
    accepting uphill moves with a temperature schedule.
    Returns the best vector found and its peak convolution.
    """
    n = a0.shape[0]
    cur_a = a0.copy()
    cur_peak = _fft_self_convolve(cur_a).max()
    best_a = cur_a.copy()
    best_peak = cur_peak

    start = time.time()
    total_time = max(deadline - start, 1e-6)

    # temperature schedule (exponential decay)
    start_temp = max(cur_peak, 1e-3) * 0.5
    final_temp = start_temp * 1e-5
    if final_temp <= 0.0:
        final_temp = 1e-9
    cooling_rate = -math.log(final_temp / start_temp) / total_time

    iter_cnt = 0
    while time.time() < deadline:
        iter_cnt += 1
        # pick source index with non‑zero mass
        src = np.random.randint(n)
        if cur_a[src] <= 1e-12:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        # move a random fraction (up to 5 %) of the source mass
        max_delta = min(cur_a[src], 0.05)
        delta = np.random.rand() * max_delta

        # propose new vector
        a_new = cur_a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_peak = _fft_self_convolve(a_new).max()
        delta_obj = new_peak - cur_peak

        # compute temperature
        elapsed = time.time() - start
        temperature = start_temp * math.exp(-cooling_rate * elapsed)

        accept = False
        if delta_obj < 0.0:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-delta_obj / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = a_new
            cur_peak = new_peak
            if cur_peak < best_peak:
                best_peak = cur_peak
                best_a = cur_a.copy()

        # occasional reset to the best found so far (helps when temperature is low)
        if iter_cnt % 20000 == 0 and np.random.rand() < 0.05:
            cur_a = best_a.copy()
            cur_peak = best_peak

    return best_a, best_peak

def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals
