"""
Propose a non‑negative step function with a low self‑convolution peak.

Algorithm
---------
1. Determine the problem size `n`.  The ℓ₁‑norm is fixed to
   `S = sqrt(2*n)` because then the evaluation score equals the peak of the
   self‑convolution.

2. Generate a small pool of diverse initial candidates (uniform,
   Dirichlet draws, power‑law, sinusoidal, triangular and the supplied reference
   if any) and project each onto the simplex Σa = S.

3. For every candidate run a projected Adam optimiser that minimises a *smooth*
   approximation of the ℓ∞‑norm of the convolution.
   The smooth surrogate is the soft‑max
       fλ(a) = (1/λ)·log Σ_k exp(λ·(a*a)_k) .
   Its gradient is obtained via a weighted cross‑correlation:
       ∇fλ = 2·(a_rev ★ w)   where   w_i ∝ exp(λ·conv_i) .
   λ starts modest (≈0.5) and is increased slowly to concentrate the weight on
   the largest convolution entries.

4. Each Adam step is projected back onto the simplex (Euclidean projection).

5. Keep the best sequence (lowest convolution peak) seen across all candidates.

6. With the remaining time perform a simple hill‑climbing on the permutation
   of the values: random swaps are kept when they lower the peak; occasional
   small mass‑shifts are also tried.

The whole search respects the 1000 s budget and returns the best sequence
found.  The sequence is already normalised (Σa = √(2 n)), so the
evaluation score equals the reported peak.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_convolve_fft(a: np.ndarray, b: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a * b using FFT (full length a.size + b.size – 1)."""
    fa = np.fft.rfft(a, n=size_fft)
    fb = np.fft.rfft(b, n=size_fft)
    conv = np.fft.irfft(fa * fb, n=size_fft)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft_self: int) -> float:
    """Exact peak of the self‑convolution a * a."""
    conv = _full_convolve_fft(a, a, size_fft_self)
    return float(conv.max())


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Return a probability vector proportional to exp(lam * conv)."""
    c = lam * conv
    # numerical stability
    cmax = c.max()
    exp_c = np.exp(c - cmax)
    w = exp_c / exp_c.sum()
    return w


# ----------------------------------------------------------------------
def _gradient_via_weights(a: np.ndarray, w: np.ndarray,
                         size_fft_rev: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate:
        ∇ = 2 * (a_rev ★ w)
    where ★ denotes linear convolution and a_rev = a reversed.
    The slice corresponding to indices 0…n‑1 is returned.
    """
    a_rev = a[::-1]
    conv_rev = _full_convolve_fft(a_rev, w, size_fft_rev)
    n = a.shape[0]
    start = n - 1
    grad = 2.0 * conv_rev[start: start + n]
    return grad


# ----------------------------------------------------------------------
def _generate_initial_candidates(n: int, target_sum: float,
                                 have_ref: bool) -> list[np.ndarray]:
    """Return a modest list of diverse starting vectors, already projected."""
    candidates = []

    # reference if available
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        candidates.append(a_ref)

    # uniform baseline
    a_uni = np.ones(n, dtype=float)
    a_uni = _project_to_simplex(a_uni, target_sum)
    candidates.append(a_uni)

    rng = np.random.default_rng(12345)

    # Dirichlet draws with several concentration parameters
    for alpha in (0.02, 0.1, 0.5, 1.0):
        for _ in range(3):
            probs = rng.dirichlet(np.full(n, alpha))
            a = probs * target_sum
            candidates.append(_project_to_simplex(a, target_sum))

    # power‑law decreasing shapes
    idx = np.arange(n)
    for _ in range(5):
        p = rng.uniform(0.2, 3.2)
        a = 1.0 / (idx + 1) ** p
        candidates.append(_project_to_simplex(a, target_sum))

    # exponential decay
    for _ in range(5):
        dec = rng.uniform(0.001, 0.016)
        a = np.exp(-dec * idx)
        candidates.append(_project_to_simplex(a, target_sum))

    # sinusoidal shape (positive)
    a_sin = np.sin(np.pi * (idx + 1) / (n + 1))
    candidates.append(_project_to_simplex(a_sin, target_sum))

    # triangular (increasing) shape
    a_tri = (idx + 1).astype(float)
    candidates.append(_project_to_simplex(a_tri, target_sum))

    # random blockwise constant patterns
    for _ in range(20):
        nb = rng.integers(2, 7)      # number of blocks
        cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = rng.random(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        candidates.append(_project_to_simplex(cand, target_sum))

    return candidates


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation score.
    The function respects a 1000 s time budget and returns the best sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------ problem size ------------------
    try:
        n = len(height_sequence_1)               # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)               # Σa = √(2 n)
    size_fft_self = _fft_size(2 * n - 1)          # for a*a
    size_fft_rev = _fft_size(3 * n - 2)           # for a_rev * w (len = n+L‑1)

    # ------------------ helpers ------------------
    def quick_score(a: np.ndarray) -> float:
        return _max_convolution(a, size_fft_self)

    # Build initial pool
    init_candidates = _generate_initial_candidates(n, target_sum, have_ref)

    print(f"[init] generated {len(init_candidates)} candidates, n = {n}")

    # ------------------ optimisation per candidate ------------------
    global_best_a = None
    global_best_score = float("inf")

    # Adam hyper‑parameters (shared)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # smoothing schedule
    lam_start = 0.5
    lam_max = 20.0
    lam_mul = 1.001

    # iterate over candidates
    for cand_id, a0 in enumerate(init_candidates, 1):
        if time.time() >= deadline:
            break

        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # step‑size scale: about the size of a typical entry
        eta0 = target_sum * 0.5 / a.size

        lam = lam_start

        # decide how many iterations we can afford for this candidate
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 1500))

        print(f"[cand {cand_id}] start optimisation, "
              f"raw score = {quick_score(a):.6f}, max iters = {max_iters}")

        for it in range(1, max_iters + 1):
            # check time
            if time.time() >= deadline:
                break

            # full self‑convolution and current peak
            conv = _full_convolve_fft(a, a, size_fft_self)
            cur_max = float(conv.max())
            if cur_max < global_best_score:
                global_best_score = cur_max
                global_best_a = a.copy()
                print(f"[cand {cand_id}] it {it:,} new global best = {global_best_score:.6f}")

            # weights (soft‑max)
            w = _softmax_weights(conv, lam)

            # gradient w.r.t. a
            g = _gradient_via_weights(a, w, size_fft_rev)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            eta = eta0 / math.sqrt(it)

            a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto the simplex Σa = target_sum
            a = _project_to_simplex(a, target_sum)

            # slowly increase λ
            lam = min(lam * lam_mul, lam_max)

            # occasional progress report
            if it % 20000 == 0:
                print(f"[cand {cand_id}] it {it:,} λ={lam:.3f} "
                      f"peak={cur_max:.6f}")

        # end of candidate loop
        print(f"[cand {cand_id}] finished, best peak seen = {global_best_score:.6f}")

    # ---------------------------------------------------------------
    # Post‑processing: local permutation optimisation (hill‑climbing)
    if global_best_a is None:
        raise RuntimeError("No candidate was produced within the time budget.")

    a = global_best_a.copy()
    best_score = global_best_score

    print("[swap] start hill‑climbing on permutation")
    while time.time() < deadline:
        # try a random swap
        i, j = np.random.choice(n, 2, replace=False)
        a_swapped = a.copy()
        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
        sc = _max_convolution(a_swapped, size_fft_self)
        if sc < best_score:
            best_score = sc
            a = a_swapped
            global_best_a = a.copy()
            global_best_score = sc
            print(f"[swap] improved to {best_score:.6f}")

        # tiny perturbation (move a small amount of mass between two cells)
        if np.random.rand() < 0.003:
            i1, i2 = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.04 * target_sum / n
            if a[i1] - delta > 0:
                a2 = a.copy()
                a2[i1] -= delta
                a2[i2] += delta
                a2 = _project_to_simplex(a2, target_sum)
                sc2 = _max_convolution(a2, size_fft_self)
                if sc2 < best_score:
                    best_score = sc2
                    a = a2
                    global_best_a = a.copy()
                    global_best_score = sc2
                    print(f"[perturb] improved to {best_score:.6f}")

        # break if only a few milliseconds are left
        if time.time() + 0.01 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    # Return a plain Python list (the evaluator expects a list)
    return global_best_a.tolist()
