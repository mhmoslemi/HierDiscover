"""
Propose a step‑function (non‑negative sequence) with a low self‑convolution peak.

Algorithm – high‑level overview
--------------------------------
1. **Problem size** – infer the length `n` from a reference sequence when
   available, otherwise use `n≈3000`.  The L¹‑norm target is
   `S = √(2 n)` because `evaluate_sequence` returns
   `2·n·max(conv)/(∑a)²` and we want it to equal the raw convolution peak.

2. **Broad random sampling** – generate a diverse pool of candidates:
   Uniform, Dirichlet draws (various concentration), power‑law / exponential
   decays, random block‑constant vectors and the reference solution (if any).
   Every candidate is projected onto the simplex `{x≥0, Σx=S}` and its exact
   convolution peak is evaluated with an FFT‑based routine.  The best `K`
   candidates are retained.

3. **Intensive refinement** – each of the stored candidates is improved with
   an Adam‑style sub‑gradient method that directly minimizes the hard
   `ℓ∞`‑norm of the self‑convolution (`max(conv)`).  After each step the
   vector is projected back onto the simplex.  The number of iterations is
   limited only by the remaining time budget.

4. **Mass‑transfer annealing** – starting from the current global best, we
   repeatedly move a small random amount of mass from one index to another.
   The move is accepted if it reduces the peak (or with a Metropolis
   probability that gradually cools).  This simple simulated‑annealing step
   helps to escape plateaus that sub‑gradient descent can get stuck in.

5. **Permutation hill‑climb** – finally we fine‑tune the ordering of the
   values by swapping pairs of positions; a swap is kept only when it lowers
   the peak.  This cheap local search often yields a few extra points of
   improvement.

The whole routine respects a 1000 s wall‑clock limit and returns the best
sequence discovered.  All helper functions are top‑level to satisfy the
execution environment constraints.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a (length = 2 n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution (no clipping)."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)_k   at the current maximal index k_max.
    The gradient is the exact sub‑gradient of the L∞‑norm of the convolution.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        # overlapping part is a[0..k_max] with reversed a[k_max::-1]
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        # k_max ≥ n
        start = k_max - n + 1                # first i that participates
        length = 2 * n - 1 - k_max           # number of terms
        g[start:] = 2.0 * a[:length][::-1]

    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent on the hard max‑convolution.
    Returns the best (lowest peak) sequence and its peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    eta0 = target_sum * 0.5 / a.size  # base step size ~ typical entry size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _mass_transfer_anneal(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    init_temp: float = 1e-3,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style mass‑transfer moves.
    Randomly moves a small amount of mass from i→j and accepts it if it
    improves the peak or with a probability exp(-Δ/T).  Temperature decays
    geometrically.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)
    temperature = init_temp

    n = a.size
    while time.time() < deadline:
        # pick source and destination
        i, j = np.random.choice(n, 2, replace=False)
        if a[i] <= 0:
            continue
        # propose moving a fraction of source mass
        frac = np.random.rand() * 0.05  # up‑to 5 % of source value
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        # projection is not needed – sum is preserved and non‑negativity holds
        cur_score = _max_convolution(a, size_fft)

        accept = False
        if cur_score < best_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(cur_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            # revert move
            a[i] += delta
            a[j] -= delta

        temperature *= decay
        # occasionally reheat a little to stay mobile
        if temperature < 1e-7:
            temperature = init_temp

    return best_a, best_score
