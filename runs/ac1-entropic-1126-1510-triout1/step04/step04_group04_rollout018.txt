def binary_anneal(n,m, max_iters, size_fft):
    # initialize set random
    idx = np.arange(n)
    chosen = np.random.choice(n, m, replace=False)
    x = np.zeros(n, dtype=float)
    x[chosen]=1.0

    conv = _full_conv(x, size_fft)
    cur_max = conv.max()
    best_x = x.copy()
    best_max = cur_max

    not_in = set(np.where(x==0)[0])
    in_set = set(chosen)

    T0 = 1.0
    T_min = 1e-4

    for it in range(1, max_iters+1):
        # temperature schedule linear cool
        T = T0 * (1 - it/max_iters) + T_min * (it/max_iters)

        # propose swap
        p = np.random.choice(list(in_set))
        q = np.random.choice(list(not_in))
        # propose new vector copy
        x_new = x.copy()
        x_new[p] = 0.0; x_new[q] = 1.0
        conv_new = _full_conv(x_new, size_fft)
        new_max = conv_new.max()
        delta = new_max - cur_max
        if delta <= 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            x = x_new
            cur_max = new_max
            # update sets
            in_set.remove(p); in_set.add(q)
            not_in.remove(q); not_in.add(p)
            if new_max < best_max:
                best_max = new_max
                best_x = x.copy()
        # optional early break if best_max == 1? Then optimum.

    return best_x, best_max

def refine_smooth(a0, size_fft, target_sum, deadline, max_iters=200_000, gamma_start=0.5, gamma_end=5.0):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a) (we can use _max_convolution defined).

    # Initialize Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute full conv via FFT using earlier function
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[:2*n-1]
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
        # schedule gamma
        gamma = gamma_start + (gamma_end - gamma_start) * (it / max_iters)
        # compute weights
        max_c = cur_max
        w = np.exp(gamma * (conv - max_c))
        w_sum = w.sum()
        w /= w_sum
        # gradient via conv of w_rev and a
        w_rev = w[::-1]
        fw = np.fft.rfft(w_rev, n=size_fft)
        grad = 2.0 * np.fft.irfft(fw * fa, n=size_fft)[n-1:2*n-1][::-1]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        # simplex projection
        a = _project_to_simplex(a, target_sum)
    return best_a, best_score

"""
Search for a non‑negative step‑function (list of heights) that minimizes the
evaluation score defined by `evaluate_sequence`.  The algorithm works in three
stages:

1. **Broad random sampling** – generate a few hundred candidates from a variety
   of families (Dirichlet draws, power‑law, exponential, block‑wise constants)
   and keep the K best on the simplex Σa = √(2 n).

2. **Smooth L∞ optimisation** – for each of the K best candidates run an
   Adam‑style projected gradient descent on a smooth surrogate of the
   max‑convolution:
        F(a) = (1/γ)·log Σₖ exp(γ·(a*a)ₖ) .
   Gradients are obtained efficiently with FFTs.  The schedule slowly raises γ
   from a small value (≈ 0.5) up to ≈ 5 so the optimisation first moves in a
   smooth landscape and then focuses on the actual peak.

3. **Local permutation hill‑climbing** – a final fast loop swaps two positions
   (which does not change the ℓ₁‑norm) and keeps the swap if it lowers the
   peak.  Random tiny perturbations are occasionally tried to escape plateaus.

The whole routine respects the 1000 s budget and returns the best sequence
found.  All numbers are non‑negative and the ℓ₁‑norm is exactly
√(2 n), so `evaluate_sequence` will simply output the maximal convolution
value.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ `required` (good for FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear self‑convolution a*a using FFT (length = 2·n‑1)."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact maximal value of the self‑convolution of a."""
    conv = _full_conv_fft(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _refine_smooth(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 150_000,
    gamma_start: float = 0.5,
    gamma_end: float = 5.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the smoothed objective
        F(a) = (1/γ) log Σₖ exp(γ·(a*a)ₖ).
    Returns the best vector seen and its true peak (max convolution).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base step size (order of magnitude of a single entry)
    eta0 = target_sum * 0.5 / a.size

    n = a.shape[0]

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- convolution of a with itself ----
        fa = np.fft.rfft(a, n=size_fft)
        conv = np.fft.irfft(fa * fa, n=size_fft)[: 2 * n - 1]
        cur_max = float(conv.max())

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[refine] it {it:,} best max = {best_score:.6f}")

        # ---- compute the smooth gradient ----
        # temperature (γ) grows linearly from start to end
        gamma = gamma_start + (gamma_end - gamma_start) * (it / max_iters)

        # numerically stable soft‑max weights
        max_c = cur_max
        w = np.exp(gamma * (conv - max_c))
        w_sum = w.sum()
        w /= w_sum

        # gradient = Σₖ wₖ ∂(a*a)ₖ/∂a
        # using the convolution trick described in the analysis
        w_rev = w[::-1]
        fw = np.fft.rfft(w_rev, n=size_fft)
        grad_full = np.fft.irfft(fw * fa, n=size_fft)
        # slice corresponds to indices n‑1 … 2n‑2 (inclusive)
        grad_slice = grad_full[n - 1 : 2 * n - 1][::-1]
        grad = 2.0 * grad_slice

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule ~ 1/√t
        eta = eta0 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex Σa = target_sum, a ≥ 0
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Returns a list of non‑negative heights that (hopefully)
    yields the smallest possible evaluation score within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a tiny safety margin

    # --------------------------------------------------------------
    # problem size and target ℓ₁‑norm
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)           # Σa = √(2 n)
    # FFT size large enough for both (a*a) (length 2n‑1) and
    # convolution of a with a weight vector (length 3n‑2)
    size_fft = _fft_size(3 * n)

    # --------------------------------------------------------------
    # helpers
    def push_candidate(score: float, a: np.ndarray, pool: list[tuple[float, np.ndarray]], K: int):
        """Insert (score,a) in the sorted pool of size ≤ K (ascending by score)."""
        if len(pool) < K:
            pool.append((score, a.copy()))
            pool.sort(key=lambda x: x[0])
        else:
            if score < pool[-1][0]:
                pool[-1] = (score, a.copy())
                pool.sort(key=lambda x: x[0])

    # keep the K best raw samples
    K = 6
    best_raw: list[tuple[float, np.ndarray]] = []   # (score, a)

    # --------------------------------------------------------------
    # 1. broad random sampling --------------------------------------
    idx = np.arange(n)

    # reference if present
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        push_candidate(_max_convolution(a_ref, size_fft), a_ref, best_raw, K)
        print(f"[init] reference peak = {_max_convolution(a_ref, size_fft):.6f}")

    # uniform baseline
    a_uni = np.ones(n, dtype=float)
    a_uni = _project_to_simplex(a_uni, target_sum)
    push_candidate(_max_convolution(a_uni, size_fft), a_uni, best_raw, K)
    print(f"[init] uniform peak = {_max_convolution(a_uni, size_fft):.6f}")

    # Dirichlet draws (different concentration parameters)
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(10):
            probs = np.random.dirichlet(np.full(n, alpha))
            a = probs * target_sum
            push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    # Power‑law decays
    for _ in range(30):
        p = np.random.rand() * 2.8 + 0.2      # exponent ∈ [0.2, 3.0]
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    # Exponential decays
    for _ in range(30):
        dec = np.random.rand() * 0.014 + 0.001   # decay ∈ [1e‑3, 1.5e‑2]
        a = np.exp(-dec * idx)
        a = _project_to_simplex(a, target_sum)
        push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    # Random block‑wise constant sequences (2‑6 blocks)
    for _ in range(150):
        nb = np.random.randint(2, 7)            # how many blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05        # small positive values
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        push_candidate(_max_convolution(cand, size_fft), cand, best_raw, K)

    print(f"[sampling] best raw peak = {best_raw[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2. refined smooth optimisation ---------------------------------
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    for rank, (raw_sc, a0) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly equal remaining time to each remaining candidate
        max_iters = int(min(200_000, remaining * 1200))
        print(f"[refine] candidate #{rank+1} (raw peak {raw_sc:.6f}), iters {max_iters}")
        refined_a, refined_sc = _refine_smooth(
            a0, size_fft, target_sum, deadline, max_iters=max_iters,
            gamma_start=0.5, gamma_end=5.0,
        )
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] new best after smooth refinement = {global_best_score:.6f}")

    # --------------------------------------------------------------
    # 3. permutation hill‑climbing (local swaps) --------------------
    print("[local swaps] start")
    a = global_best_a.copy()
    best_sc = global_best_score
    while time.time() < deadline:
        # propose a random swap (keeps the ℓ₁‑norm unchanged)
        i, j = np.random.choice(n, 2, replace=False)
        a_swapped = a.copy()
        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
        sc = _max_convolution(a_swapped, size_fft)
        if sc < best_sc:
            best_sc = sc
            a = a_swapped
            global_best_a = a.copy()
            global_best_score = sc
            print(f"[swap] improved to {best_sc:.6f}")

        # occasional tiny perturbation to escape plateaus
        if np.random.rand() < 0.001:
            delta = (np.random.rand() * 2 - 1) * 0.02 * target_sum
            i2 = np.random.randint(n)
            if a[i2] - delta > 0:
                a2 = a.copy()
                a2[i2] -= delta
                a2 = _project_to_simplex(a2, target_sum)
                sc2 = _max_convolution(a2, size_fft)
                if sc2 < best_sc:
                    best_sc = sc2
                    a = a2
                    global_best_a = a2.copy()
                    global_best_score = sc2
                    print(f"[perturb] improved to {best_sc:.6f}")

        # stop a moment before the deadline
        if time.time() + 0.01 > deadline:
            break

    print(f"[final] best peak = {global_best_score:.6f}")
    return global_best_a.tolist()
