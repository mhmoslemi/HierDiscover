def fft_convolve(a, b):
    size = 1 << ((a.size + b.size - 1)-1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A*B, n=size)
    return conv[:a.size+b.size-1]

def max_subgrad_step(a):
    n = len(a)
    b = self_convolve(a)
    cur_max = b.max()
    # active set tolerance
    tol = max(1e-14, cur_max*1e-12)
    active = (b >= cur_max - tol)
    w = np.zeros_like(b)
    if active.any():
        w[active] = 1.0/active.sum()
    else:
        # no active? improbable.
        return np.zeros_like(a), cur_max
    a_rev = a[::-1]
    conv_rev = fft_convolve(a_rev, w)
    start = n-1
    grad = 2.0 * conv_rev[start:start+n]
    return grad, cur_max

def _adam_optimize(a0, deadline, init_alpha=0.5, alpha_factor=1.01, alpha_max=5000.0, lr0=0.3, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_grad(a, alpha)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # Keep track of best
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # temperature increase
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # Occasionally jitter
        if t % 400 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak

def _local_search(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = objective_score(a)
    n = len(a)
    start = time.time()
    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random direction
        if np.random.rand() < 0.5:
            # move mass from i to j
            max_move = a[i]
            if max_move <= 0:
                continue
            delta = max_move * np.random.rand() * 0.02
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            max_move = a[j]
            if max_move <= 0:
                continue
            delta = max_move * np.random.rand() * 0.02
            a_new = a.copy()
            a_new[i] += delta
            a_new[j] -= delta
        # compute score
        cand_score = objective_score(a_new)
        if cand_score < best_score - 1e-16:
            a = a_new
            best_a = a_new.copy()
            best_score = cand_score
        else:
            # occasional acceptance based on temperature
            elapsed = time.time() - start
            total = deadline - start
            if total <= 0:
                continue
            temperature = max(1e-8, (1 - elapsed / total) * 0.001)
            if temperature > 0 and np.random.rand() < np.exp(-(cand_score - best_score) / temperature):
                a = a_new
    return best_a

"""
Propose a low‑score step‑function.

The search proceeds in three stages for each candidate length:

1.  Initialise the distribution on the simplex (arcsine, uniform, random,
    or the previous best interpolated).
2.  Adam optimisation of a smooth (soft‑max) surrogate of the ℓ∞
    self‑convolution peak.  The temperature α is raised slowly so that
    the optimiser first sees a smooth landscape and finally converges
    to the true max.
3.  Exact ℓ∞ descent: a sub‑gradient step on the true maximum with a
    cheap back‑tracking line‑search.
4.  Fine‑grained stochastic local search that repeatedly moves a tiny
    amount of mass between two random coordinates, accepting only
    improvements (occasional uphill moves for exploration).

The best sequence seen (according to `evaluate_sequence`) is kept.
The whole routine respects a global time budget (default 1000 s) and
returns a plain Python list ready for the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=float)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a non‑negative sequence to a new length by linear interpolation."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    # grad = 2·rev(a) * w   (restricted to the n entries of a)
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    n = len(a)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, cur_max


def _max_subgrad_step(a: np.ndarray):
    """
    Sub‑gradient of the true ℓ∞ objective.
    Uniform weight on all indices attaining the current maximum.
    Returns (grad, cur_max).
    """
    n = len(a)
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) (sum(a) is 1)."""
    n = len(a)
    peak = _fft_self_convolve(a).max()
    return 2.0 * n * peak


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, deadline: float,
                   init_alpha: float = 0.5,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 4000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best iterate (on simplex) and its peak value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_grad(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        lr = lr0 / np.sqrt(t)

        # update & projection
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # tiny random jitter every few hundred iterations
        if t % 400 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Sub‑gradient descent on the true ℓ∞ objective with a simple back‑track.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = objective_score(a)
    it = 0

    while time.time() < deadline:
        grad, cur_max = _max_subgrad_step(a)
        if np.linalg.norm(grad) < 1e-14:
            break

        # base step size, decreasing like 1/√it
        step = 0.07 / np.sqrt(it + 1)

        # try a few back‑tracking reductions
        improved = False
        for _ in range(5):
            cand = a - step * grad
            cand = _project_onto_simplex(cand, s=1.0)
            cand_val = objective_score(cand)
            if cand_val < best_val - 1e-14:
                a = cand
                best_a = cand.copy()
                best_val = cand_val
                improved = True
                break
            step *= 0.5

        if not improved:
            # occasional jitter to escape plateaus
            a += np.random.randn(len(a)) * 5e-7
            a = _project_onto_simplex(a, s=1.0)

        it += 1

    return best_a


# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Stochastic pairwise mass‑transfer search.
    Moves a small fraction of mass between two random coordinates.
    Accepts only improving moves (tiny uphill probability for exploration).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = objective_score(a)
    n = len(a)
    start = time.time()

    while time.time() < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # direction: i→j or j→i
        if np.random.rand() < 0.5:
            # i → j
            max_move = a[i]
            if max_move <= 0:
                continue
            delta = max_move * np.random.rand() * 0.02
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            # j → i
            max_move = a[j]
            if max_move <= 0:
                continue
            delta = max_move * np.random.rand() * 0.02
            a_new = a.copy()
            a_new[i] += delta
            a_new[j] -= delta

        cand_score = objective_score(a_new)
        if cand_score < best_score - 1e-16:
            a = a_new
            best_a = a_new.copy()
            best_score = cand_score
        else:
            # tiny uphill acceptance (simulated annealing)
            elapsed = time.time() - start
            total = deadline - start
            if total > 0:
                temperature = max(1e-8, (1.0 - elapsed / total) * 0.001)
                if temperature > 0 and np.random.rand() < np.exp(-(cand_score - best_score) / temperature):
                    a = a_new

    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function (non‑negative heights) with a low evaluation score.
    The routine explores several lengths and initialisations, performs
    Adam optimisation on a smooth surrogate, exact ℓ∞ sub‑gradient descent,
    and a final stochastic local search.  The best sequence found is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # try to obtain the previous best construction, if available
    try:
        base_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        base_n = len(base_seq)
    except Exception:
        base_seq = None
        base_n = 6000

    # candidate lengths (a modest spread around the historic size)
    length_factors = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]
    lengths = sorted({max(500, int(base_n * f)) for f in length_factors})
    lengths = [l for l in lengths if l <= 10000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # equally split time budget among lengths
    time_per_length = max(5.0, (deadline - start_time) / max(1, len(lengths)))

    for n in lengths:
        if time.time() >= deadline:
            break
        length_deadline = min(deadline, time.time() + time_per_length)

        init_strategies = ["prev", "arcsine", "uniform", "random"]
        for kind in init_strategies:
            if time.time() >= length_deadline:
                break

            # ----- initialise a0 -------------------------------------------------
            if kind == "prev" and base_seq is not None:
                a0 = _interpolate_to_length(base_seq, n)
            elif kind == "arcsine":
                a0 = _arcsine_distribution(n)
            elif kind == "uniform":
                a0 = np.full(n, 1.0 / n, dtype=float)
            elif kind == "random":
                a0 = np.random.rand(n)
                a0 /= a0.sum()
            else:
                a0 = np.full(n, 1.0 / n, dtype=float)

            a0 = _project_onto_simplex(a0, s=1.0)

            # ----- split remaining time for the three optimisation phases -----
            remaining = length_deadline - time.time()
            if remaining <= 0.0:
                break
            phase1_deadline = time.time() + 0.55 * remaining
            a_adam, _ = _adam_optimize(a0, phase1_deadline)

            remaining = length_deadline - time.time()
            if remaining <= 0.0:
                break
            phase2_deadline = time.time() + 0.30 * remaining
            a_sub = _subgradient_descent(a_adam, phase2_deadline)

            remaining = length_deadline - time.time()
            if remaining <= 0.0:
                break
            phase3_deadline = length_deadline
            a_final = _local_search(a_sub, phase3_deadline)

            # ----- evaluate candidate -------------------------------------------
            fast_score = objective_score(a_final)
            if fast_score < overall_best_score - 1e-12:
                overall_best_score = fast_score
                overall_best_seq = a_final.copy()
                print(f"[{time.time():.1f}] len={n} init={kind} fast_score={fast_score:.6f}")

                # true evaluation (should match fast_score, but we call it anyway)
                try:
                    true_score = evaluate_sequence(overall_best_seq.tolist())
                    if true_score < overall_best_score - 1e-12:
                        overall_best_score = true_score
                        overall_best_seq = np.array(overall_best_seq, dtype=float)
                        print(f"*** New true best: len={n} init={kind} score={true_score:.6f}")
                except Exception as exc:
                    print("evaluate_sequence raised:", exc)

                # early exit if target already met
                if overall_best_score <= 1.503:
                    print("Target achieved – stopping early.")
                    break

        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback in the extremely unlikely case nothing was produced
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=float)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback uniform used.")

    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
