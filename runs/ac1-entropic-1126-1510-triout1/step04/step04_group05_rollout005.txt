```python
def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # problem size

    S = math.sqrt(2.0 * n)

    # interaction
    best_a = None
    best_score = float('inf')

    run_idx = 0

        remaining_total = deadline - time.time()
        # allocate per-run time: at least 80 sec, at most 200 sec
        run_time = min(180.0, remaining_total * 0.6)  # allocate majority of remaining for this run
        if run_time < 80.0:

        # initialise start vector
        if run_idx == 0 and have_ref:
            a0 = np.asarray(height_sequence_1, dtype=float)

        # run pipeline
        start_time = time.time()
```

```python
def _l_bfgs_refine(a_start, S, budget, start_time):
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    size = _fft_size(n)

    # initial mu from max conv
    conv_cur = _full_convolution(a_start, size)
    mu = float(conv_cur.max())
    if mu <= 0:
        mu = 1.0

    mu_factor = 0.7
    mu_min = 1e-8
    maxiter_per_mu = 400

    while mu > mu_min and time.time() - start_time < budget:
        # if not enough time left for this mu iteration, break
        time_left = budget - (time.time() - start_time)
        if time_left < 0.5:  # not enough for L-BFGS
            break
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(obj, y, method="L-BFGS-B", jac=True,
                               options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12})
        except Exception as exc:
            print("[L-BFGS error] ", exc)
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L-BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score
```

```python
def _subgradient_opt(a_start, S, budget, start_time):
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # initial step size maybe based on typical magnitude
    # compute gradient at start
    conv = _full_convolution(a, size)
    max_c = conv.max()
    mask = conv == max_c
    cnt = mask.sum()
    w = np.zeros_like(conv)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    # subgradient
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    grad_norm = np.linalg.norm(g_a)
    if grad_norm == 0:
        grad_norm = 1.0
    eta0 = S / n * 5.0 / grad_norm  # scale step relative to gradient magnitude

    it = 0
    no_improve = 0
    while time.time() - start_time < budget and it < 20000:
        # recompute convolution and subgradient each iteration
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # subgradient
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n-1:n-1+n]

        eta = eta0 / math.sqrt(it + 1)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)
        # evaluate; if better update best
        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            # optionally increase step size a bit
        else:
            no_improve += 1

        a = a_new
        it += 1

        # early stop if no improvement for many iterations
        if no_improve >= 5000:
            break

    return best_a, best_score
```

"""
Multi‑restart stochastic optimisation for the “max self‑convolution” problem.

The algorithm works in three stages for each restart:

1. Simulated‑annealing (SA) on the simplex – cheap mass moves that
   directly minimise the true objective (the maximum of the
   self‑convolution, because the ℓ¹‑norm is fixed).

2. Continuation with a smooth‑max surrogate solved by L‑BFGS‑B.
   The surrogate becomes tighter as the smoothing parameter μ is
   gradually decreased.

3. Projected sub‑gradient descent on the hard max (no smoothing)
   with a diminishing step‑size and a large number of iterations.

Several independent restarts (uniform, Dirichlet‑sampled and
perturbations of the current best) are performed inside the total
budget (≈1000 s).  The best sequence found across all restarts is
returned.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for FFT‑based convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Full (linear) convolution a*a using FFT, length 2·n‑1."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0                     # eliminate tiny negatives
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a (full convolution) using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (soft‑max) surrogate f(y) = max_k (a*a)[k] + μ log Σ exp((a*a)[k]−max)/μ,
    together with its gradient w.r.t. y.
    a = S·softmax(y) automatically satisfies Σa = S and a ≥ 0.
    """
    p = _softmax(y)                # Σp = 1
    a = S * p                      # Σa = S

    # exact convolution (still cheap for n≈3k)
    conv = np.convolve(a, a)       # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (see derivation in the background notes)
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)               # scalar = Σ p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Simple SA on the simplex – moves a random fraction of mass between two indices."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    # temperature schedule
    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-7

    it = 0
    while time.time() - start_time < budget:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / budget
        T = max(Tmin, T0 * (1.0 - frac))

        # pick donor i (must have some mass) and receiver j≠i
        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 30 %) of a[i] to j
        delta = np.random.rand() * a[i] * 0.3
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = False
        if diff < 0.0:
            accept = True
        else:
            if T > 0.0 and np.random.rand() < math.exp(-diff / T):
                accept = True

        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Continuation on the smooth‑max surrogate with L‑BFGS‑B."""
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    size = _fft_size(n)

    best_a = a_start.copy()
    best_score = _max_convolution(best_a, size)

    # initial smoothing parameter from the current max‑convolution
    mu = _max_convolution(a_start, size)
    mu = max(mu, 1.0)

    mu_factor = 0.7
    mu_min = 1e-8
    maxiter_per_mu = 400

    while mu > mu_min and time.time() - start_time < budget:
        # time‑check – abort if not enough time for another L‑BFGS run
        if budget - (time.time() - start_time) < 0.5:
            break
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = _max_convolution(a, size)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_opt(a_start: np.ndarray, S: float,
                     budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = _max_convolution(best_a, size)

    # initial step‑size – scaled to the typical magnitude of the gradient
    conv = _full_convolution(a, size)
    max_c = conv.max()
    mask = conv == max_c
    cnt = mask.sum()
    w = np.zeros_like(conv)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    fa_rev = np.fft.rfft(a[::-1], n=size)
    fw = np.fft.rfft(w, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]
    grad_norm = np.linalg.norm(g_a)
    if grad_norm == 0:
        grad_norm = 1.0
    eta0 = (S / n) * 5.0 / grad_norm   # modest initial step

    it = 0
    no_improve = 0
    while time.time() - start_time < budget and it < 20000:
        # full convolution & active set
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # sub‑gradient of the hard max
        fa_rev = np.fft.rfft(a[::-1], n=size)
        fw = np.fft.rfft(w, n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        eta = eta0 / math.sqrt(it + 1)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = _max_convolution(a_new, size)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        if no_improve >= 5000:
            break

    return best_a, best_score

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
