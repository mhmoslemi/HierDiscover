Optimising a non‑negative step function for the self‑convolution peak.

The algorithm consists of three stages:

   cosine, Gaussian mixtures, random piecewise‑linear) is sampled.

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def _adam_refine(a0: np.ndarray,
                 deadline: float,
                 n: int,
                 size_fft: int,
                 target_sum: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the *soft‑max* of the convolution.
    Returns best (a, score) seen during the run.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    eta0 = target_sum * 0.5 / n

    # initialise temperature from the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0

        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights (smooth max) -----
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        # g = 2 * (w * a) convolved with reversed a

        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        T *= 0.9997
        it += 1

            best_score = cur_score
            best_a = a.copy()

        # index (or one of the indices) where the maximum occurs
        k = int(np.argmax(conv))

        # sub‑gradient g_i = 2·a_{k‑i}  (if the index is inside the vector)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # Adam‑style update
        eta = eta0 / (1.0 + it * 1e-5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick to avoid stalls
        if it % 4000 == 0 and it > 0:
            a_new += np.random.randn(n) * target_sum * 0.0008
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises the
    self‑convolution peak.  The function respects a ~1000 s budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)      # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)            # normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # Keep the K best candidates from the exploratory phase
    K = 20
    best_candidates: list[tuple[float, np.ndarray]] = []   # (score, array)

    def consider(sc: float, a: np.ndarray) -> None:
        """Maintain a sorted list of the top‑K candidates (ascending score)."""
        if len(best_candidates) < K:
            best_candidates.append((sc, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        elif sc < best_candidates[-1][0]:
            best_candidates[-1] = (sc, a.copy())
            best_candidates.sort(key=lambda x: x[0])

    idx = np.arange(n)

    # ------------------------------------------------------------------
    # ----- 1. PARAMETRIC EXPLORATION --------------------------------

    def add_candidate(arr: np.ndarray) -> None:
        arr = _project_to_simplex(arr, target_sum)
        sc = quick_score(arr, n, size_fft)
        consider(sc, arr)

    # (a) Uniform
    add_candidate(np.ones(n))

    # (b) Linear decreasing (a_i ∝ (n‑i)^p)
    for _ in range(150):
        p = np.random.rand() * 2.5 + 0.2          # exponent ∈ [0.2, 2.7]
        a = (n - idx) ** p
        add_candidate(a)

    # (c) Inverse‑sqrt (single‑ended)
    for _ in range(150):
        beta = np.random.rand() * 12.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        add_candidate(a)

    # (d) Power‑law
    for _ in range(150):
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 1.8 + 0.2
        a = 1.0 / (idx + beta + 1.0) ** pexp
        add_candidate(a)

    # (e) Exponential decay (single‑ended)
    for _ in range(150):
        decay = np.random.rand() * 0.015 + 0.0002
        a = np.exp(-decay * idx)
        add_candidate(a)

    # (f) Double‑ended exponential (asymmetric)
    for _ in range(150):
        dl = np.random.rand() * 0.015 + 0.0002
        dr = np.random.rand() * 0.015 + 0.0002
        a = np.exp(-dl * idx) + np.exp(-dr * (n - 1 - idx))
        add_candidate(a)

    # (g) Cosine‑based (large at ends, small in the middle)
    for _ in range(150):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        add_candidate(a)

    # (h) Sine‑based (zero at ends, peak in the middle)
    for _ in range(150):
        p = np.random.rand() * 6.0 + 0.5
        a = (np.sin(np.pi * idx / (n - 1)) ** p)
        add_candidate(a)

    # (i) Gaussian mixture (2‑component)
    for _ in range(200):
        mu1 = np.random.rand() * n
        mu2 = np.random.rand() * n
        sigma = np.random.rand() * 0.25 + 0.05
        sigma = sigma * n
        a = (np.exp(-0.5 * ((idx - mu1) / sigma) ** 2) +
             np.exp(-0.5 * ((idx - mu2) / sigma) ** 2))
        add_candidate(a)

    # (j) Random piecewise‑linear (few break‑points)
    for _ in range(200):
        nb = np.random.randint(3, 9)               # 3‑8 break‑points
        xs = np.sort(np.random.choice(n, nb, replace=False))
        ys = np.random.rand(nb) + 0.1
        a = np.interp(idx, xs, ys)
        add_candidate(a)

    # (k) Random block pattern with zero gaps (few large blocks)
    for _ in range(300):
        nb = np.random.randint(2, 6)               # 2‑5 non‑zero blocks
        block_len = max(1, n // (3 * nb))          # keep gaps at least block_len
        start = 0
        cand = np.zeros(n, dtype=float)
        for _ in range(nb):
            if start + block_len > n:
                break
            val = np.random.rand() + 0.1
            cand[start:start + block_len] = val
            start += block_len * 2                # block + zero gap of same size
        add_candidate(cand)

    # (l) Reference sequence, if available
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        add_candidate(a_ref)

    best_candidates.sort(key=lambda x: x[0])
    print(f"[exploration] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # ---------- 2. REFINEMENT (Adam + Exact‑max sub‑gradient) -------
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    for rank, (raw_score, a0) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank + 1} (raw score {raw_score:.6f})")
        remaining = deadline - time.time()
        # allocate a modest number of iterations proportionally to remaining time
        max_iters = int(min(200_000, remaining * 1300))

        # First a few hundred Adam steps on the soft‑max objective
        a_mid, sc_mid = _adam_refine(a0, deadline, n, size_fft,
                                    target_sum, max_iters // 4)

        # Then switch to exact‑max sub‑gradient descent
        a_refined, sc_refined = _subgrad_refine(a_mid, deadline, n,
                                               size_fft, target_sum,
                                               max_iters - max_iters // 4)

        if sc_refined < global_best_score:
            global_best_score = sc_refined
            global_best_a = a_refined.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # ---------- 3. FINAL RANDOM‑BLOCK SWEEP -------------------------

    while time.time() < deadline:
        nb = np.random.randint(2, 6)
        block_len = max(1, n // (3 * nb))
        start = 0
        cand = np.zeros(n, dtype=float)
        for _ in range(nb):
            if start + block_len > n:
                break
            val = np.random.rand() + 0.1
            cand[start:start + block_len] = val
            start += block_len * 2
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand, n, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")

        # stop if only a tiny slice of time is left
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")

    # return as plain Python list (as required by the harness)
    return global_best_a.tolist()
