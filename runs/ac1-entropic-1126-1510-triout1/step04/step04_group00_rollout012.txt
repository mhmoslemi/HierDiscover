        remaining = deadline - time.time()
        if remaining <= 0: break
        slice_deadline = time.time() + min(remaining * 0.5, 200)  # maybe

        For each init_kind in ("prev", "arcsine", "uniform", "random"):

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (probability vector) that minimizes

    score = 2·n·max(a*a) / (sum(a)**2)

using a soft‑max (smooth‑max) Adam optimiser on the probability simplex.
The algorithm works in the log‑softmax (unconstrained) space, anneals a
temperature parameter to turn the smooth max into the true max and keeps the
best sequence seen.  Several lengths and initialisations are explored within
the given time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)) of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _initial_vector(init_kind: str, n: int) -> np.ndarray:
    """Create a feasible probability vector of length n."""
    if init_kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init_kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif init_kind == "arcsine":
        a = _arcsine_distribution(n)
    elif init_kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    # ensure exactly on the simplex
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------

    Search for a low‑score step‑function within the supplied time budget.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0        # small safety margin

    # ------------------------------------------------------------------

    try:

    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)

    best_approx_score = float("inf")

        if remaining <= 0.0:
            break
        slice_deadline = time.time() + min(remaining * 0.35, 180.0)   # ≤180 s per length

        for init_kind in ("prev", "arcsine", "uniform", "random"):
            # ----- initialise -------------------------------------------------
            a0 = _initial_vector(init_kind, n)
            # log‑softmax parametrisation (x can be any real vector)
            x = np.log(a0 + 1e-15)

            # Adam state

            alpha_factor = 1.005
            alpha_max = 5000.0

            best_local_approx = float("inf")
            stagnate_counter = 0

                dot = np.dot(a, grad_a)
