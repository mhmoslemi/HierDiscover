assistantfinal**Solution Overview**

This program searches for a non‑negative step function (a list `a₀,…,a_{n‑1}`) that minimises  

```
score(a) = 2·n·max(a*a) / (sum(a))²                (1)
```

`evaluate_sequence` implements exactly the same formula but also clips
negative values, NaNs, infinities etc.  Because the score is *scale‑invariant*
(`a → λa` leaves (1) unchanged) we can restrict ourselves to probability
vectors `∑a_i = 1`.  The problem then reduces to

```
minimise   2·n·max(a*a)   subject to   a_i ≥ 0 ,  ∑a_i = 1            (2)
```

The algorithm consists of three phases

1. **Initialisation** – start from several well‑known shapes
   (arcsine, uniform, random, two‑spike).  If a good sequence has already
   been found it is up‑sampled to the current length.
2. **Adam optimisation on the simplex** – a smooth approximation of the
   `max` term is used

```
f_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)                (3)
```

   Its gradient is computed by a single FFT, and an Adam step is taken.
   After each step the vector is projected back onto the probability simplex.
   The temperature `α` is slowly increased (soft‑max → true max).
3. **Stochastic local search** – after the gradient phase a cheap
   “mass‑swap” hill‑climber is run for the remaining time.  It repeatedly
   moves a random fraction of the mass from one index to another and keeps
   the move if the max‑convolution improves.

The whole budget (default ≈ 1000 s) is split in two parts:

* 30 % is spent on a few short restarts with different initial shapes,
  each of them runs the Adam optimiser for a few seconds.
* the remaining time refines the current global best (up‑scaled if necessary)
  with a long Adam run followed by an intensive local‑search phase.

All FFTs are performed with `numpy.fft.rfft` which gives `O(n log n)` time.
The code never exceeds the time limit and never writes to the file system.

```python
"""Propose a non‑negative step function with a low `evaluate_sequence` score.

The algorithm is a two‑stage optimiser:
 1. Adam gradient descent on a soft‑max approximation of the max‑convolution,
    with projection onto the probability simplex.  The temperature of the
    soft‑max is annealed to obtain the true sub‑gradient of `max(a*a)`.
 2. A cheap stochastic local‑search (pairwise mass swaps) that is run
    after the gradient phase and also as a final refinement.

Multiple initial shapes (arcsine, uniform, random, two‑spike, previous best)
are tried; the best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Weights w_k = exp(α·b_k) / Σ exp(α·b_j) (or uniform on the active set).
       For very large α we use a mask to avoid overflow."""
    if alpha > 2000.0:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()               # for numerical stability
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum
    return w


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·b_k) with respect to a.
    For large α it is a sub‑gradient of max(b).
    """
    n = a.shape[0]
    w = _softmax_weights(b, alpha)

    # convolution w * a_rev, where a_rev = a[::-1]
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    # the centre part of the convolution gives the gradient
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _adam_optimize_one_length(init_a: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex for the objective 2·n·max(a*a).
    Returns the best sequence encountered and its internal score
    (2·n·max_b) which equals the official score when sum(a)=1.
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # base learning‑rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.05
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for w + a_rev – 1
    it = 0

    while time.time() < deadline:
        it += 1

        # current objective
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient of the smooth max
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project back onto the simplex
        a = _project_onto_simplex(a, s=1.0)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape plateaux
        if it % 2500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic local search: repeatedly transfer a random fraction of mass
    from one index to another and keep the change if max‑convolution improves.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] <= 1e-12:
            continue
        frac = np.random.rand() * 0.30            # up to 30 % of the mass
        delta = best_seq[i] * frac
        a_new = best_seq.copy()
        a_new[i] -= delta
        a_new[j] += delta

        b = _fft_self_convolve(a_new)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_seq = a_new

        # tiny jitter every few thousand steps
        if it % 5000 == 0:
            a_new = best_seq + np.random.randn(n) * 1e-6
            a_new = _project_onto_simplex(a_new, s=1.0)
            b = _fft_self_convolve(a_new)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_seq = a_new

    return best_seq, best_score


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _upscale_distribution(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation of a smaller distribution to a larger length.
    The result is re‑normalised to sum to 1.
    """
    if a_small.size == n_large:
        return a_small.copy()
    x_small = np.arange(a_small.size, dtype=np.float64)
    x_large = np.linspace(0, a_small.size - 1, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large[a_large < 0] = 0.0
    if a_large.sum() == 0.0:
        a_large = np.full(n_large, 1.0 / n_large, dtype=np.float64)
    else:
        a_large /= a_large.sum()
    return a_large

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point called by the grader.
    Searches for a step‑function (list of non‑negative heights) with a low
    `evaluate_sequence` score.
    """
    np.random.seed(seed)
    # keep a small safety margin for final printing / cleanup
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Determine a base length – reuse a previously discovered one if present
    try:
        base_n = len(height_sequence_1)      # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # candidate lengths (we never exceed 30 000 for speed)
    length_factors = (0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(300, int(base_n * f))
        for f in length_factors
        if max(300, int(base_n * f)) <= 30000
    })

    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – quick restarts with diverse initial shapes
    for n in lengths:
        if time.time() >= deadline:
            break
        # allocate roughly 30 % of the remaining time for this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(0.1, remaining * 0.30)

        init_kinds = ("prev", "arcsine", "uniform", "random", "spike")
        for kind in init_kinds:
            if time.time() >= slice_deadline:
                break

            # ------------------------------------------------------------------
            # build initial probability vector of length n
            if kind == "spike":
                a0 = np.zeros(n, dtype=np.float64)
                pos1 = np.random.randint(max(1, n // 10))
                pos2 = np.random.randint(n - max(1, n // 10), n)
                a0[pos1] = 0.5
                a0[pos2] = 0.5
                a0 = _project_onto_simplex(a0, s=1.0)
            elif kind == "prev" and global_best_seq is not None:
                a0 = _upscale_distribution(np.asarray(global_best_seq,
                                                      dtype=np.float64), n)
            elif kind == "arcsine":
                a0 = _arcsine_distribution(n)
            elif kind == "uniform":
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            else:   # "random"
                a0 = np.random.rand(n)
                a0 = a0 / a0.sum()

            # ------------------------------------------------------------------
            # short Adam optimisation
            seq, int_score = _adam_optimize_one_length(a0, n, slice_deadline)

            # ------------------------------------------------------------------
            # very brief local‑search (if any time left)
            time_left = slice_deadline - time.time()
            if time_left > 0.5:
                seq, int_score = _local_search(seq, n,
                                               time.time() + min(time_left, 3.0))

            # ------------------------------------------------------------------
            # evaluate with the official black‑box
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = int_score

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq.copy()
                print(f"*** New global best → len={n} init={kind} "
                      f"score={global_best_score:.6f}")

            # early exit if target already reached
            if global_best_score <= 1.503:
                break
        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 2 – long refinement on the overall best (if time permits)
    if global_best_seq is not None and time.time() < deadline:
        n_best = len(global_best_seq)

        # up‑scale to a slightly larger length if we still have time;
        # larger support often gives a lower constant factor.
        if time.time() + 60 < deadline and n_best * 1.3 <= 30000:
            n_target = int(n_best * 1.3)
            a_up = _upscale_distribution(global_best_seq, n_target)
            # run a medium‑length Adam optimisation
            seq, _ = _adam_optimize_one_length(a_up, n_target,
                                               time.time() + 45.0)
            # final local search on the enlarged sequence
            seq, _ = _local_search(seq, n_target,
                                   time.time() + min(deadline - time.time(),
                                                     50.0))
            # bring it back to the original length (interpolation is linear)
            final_seq = _upscale_distribution(seq, n_best)
        else:
            # stay at the current length and spend the rest refining it
            seq, _ = _adam_optimize_one_length(global_best_seq, n_best,
                                               deadline - 5.0)
            final_seq, _ = _local_search(seq, n_best, deadline)

        try:
            refined_score = evaluate_sequence(final_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = final_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} "
          f"(length = {len(global_best_seq)})")
    return global_best_seq.tolist()
```
