def compute_score(a):
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = conv.max()
    score = 2.0 * len(a) * max_b / (sum_a**2)
    return float(score)

"""
Propose candidate step‑function sequence.

The algorithm builds a collection of simple parametric families
(exponential, power‑law, linear, random sparse) and evaluates each
candidate on the objective 2·n·max(a*a)/(∑a)² (the same value returned
by ``evaluate_sequence`` when the ℓ¹‑norm equals √(2·n)).  The best
candidate is then polished by a short projected‑gradient descent and,
if SciPy is available, a final L‑BFGS‑B optimisation of a smooth‑max
surrogate.  All work is bounded by the supplied time budget
(≈1000 s).
"""

import time
import math
import numpy as np
try:
    import scipy.optimize as opt
except Exception:
    opt = None

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max surrogate value and gradient w.r.t. the free parameters y.
    """
    p = _softmax(y)           # Σ p_i = 1, p_i ≥ 0
    a = S * p                 # Σ a_i = S

    conv = np.convolve(a, a)  # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t a
    conv_wr = np.convolve(w, a[::-1])   # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable interface for scipy.optimize – returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Fast evaluation of the same quantity returned by evaluate_sequence."""
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    ``evaluate_sequence`` score.  The routine respects the given time
    budget (seconds) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Determine problem dimension ------------------------------------------------
    try:
        # global variable `height_sequence_1` may be defined by the evaluation harness
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3500
        have_ref = False

    S = math.sqrt(2.0 * n)      # target ℓ¹‑norm

    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # Helper to update incumbent -------------------------------------------------
    def _maybe_update(a_vec: np.ndarray, tag: str = "") -> None:
        nonlocal best_score, best_a
        sc = _score_of(a_vec)
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve{tag}] new best score = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 1. simple deterministic families
    # ----------------------------------------------------------------------
    # uniform
    a_uniform = np.full(n, S / n)
    _maybe_update(a_uniform, tag="_uniform")

    # linear decreasing (large at the left)
    w_lin = np.arange(n, 0, -1, dtype=float)   # n, n-1, ..., 1
    a_lin = w_lin * (S / w_lin.sum())
    _maybe_update(a_lin, tag="_linDec")

    # linear increasing (large at the right)
    w_inc = np.arange(1, n + 1, dtype=float)
    a_inc = w_inc * (S / w_inc.sum())
    _maybe_update(a_inc, tag="_linInc")

    # reference if supplied
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        _maybe_update(a_ref, tag="_ref")

    # ----------------------------------------------------------------------
    # 2. exponential families (a_i ∝ exp(-α i^β))
    # ----------------------------------------------------------------------
    alphas = np.logspace(-4, -1, num=20)   # from 1e-4 to 1e-1
    betas = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]

    idx = np.arange(n, dtype=float)

    for beta in betas:
        if time.time() > deadline:
            break
        for alpha in alphas:
            vec = np.exp(-alpha * (idx ** beta))
            # avoid all‑zero vector for extreme parameters
            if vec.sum() == 0.0:
                continue
            a = vec * (S / vec.sum())
            _maybe_update(a, tag=f"_exp_b{beta}_a{alpha:.2e}")

    # ----------------------------------------------------------------------
    # 3. power‑law families (a_i ∝ (i+1)^{-p})
    # ----------------------------------------------------------------------
    powers = np.linspace(0.2, 3.0, num=30)
    base = idx + 1.0
    for p in powers:
        if time.time() > deadline:
            break
        vec = base ** (-p)
        a = vec * (S / vec.sum())
        _maybe_update(a, tag=f"_pow_{p:.2f}")

    # ----------------------------------------------------------------------
    # 4. random sparse vectors (K non‑zero equal entries)
    # ----------------------------------------------------------------------
    # we try various K values that are plausible to beat the current record
    K_vals = list(range(80, 120, 5))
    for K in K_vals:
        if time.time() > deadline:
            break
        for _ in range(30):   # several random attempts per K
            positions = rng.choice(n, size=K, replace=False)
            a = np.zeros(n, dtype=float)
            a[positions] = S / K
            _maybe_update(a, tag=f"_sparseK{K}")

    # ----------------------------------------------------------------------
    # 5. random block piecewise‑constant vectors (similar to baseline code)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # allow up to ~3000 block trials, scaled to remaining time
        max_trials = int(min(3000, remaining * 150))
        print(f"[block] performing up to {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(2, 13)          # number of blocks 2 … 12
            # cut positions
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand, tag="_block")

    # ----------------------------------------------------------------------
    # 6. final polishing with projected gradient descent on the hard max
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        a = best_a.copy()
        it = 0
        # base step size (empirically scaled)
        eta0 = S * 0.8 / n
        decay = 6e-5
        while time.time() < deadline and it < 8000:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt
            conv_wr = np.convolve(w, a[::-1])
            g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

            eta = eta0 / (1.0 + decay * it)
            a_new = a - eta * g_a
            a_new = _project_to_simplex(a_new, S)

            sc = _score_of(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"[grad] improved to {best_score:.6f}")
            a = a_new
            it += 1

    # ----------------------------------------------------------------------
    # 7. optional L‑BFGS‑B refinement of the smooth‑max surrogate (if scipy present)
    # ----------------------------------------------------------------------
    if opt is not None and time.time() < deadline:
        # initialise from log of best vector (adding epsilon for safety)
        eps = 1e-12
        y0 = np.log(best_a + eps)

        # initial temperature based on current max‑convolution
        a_cur = best_a
        mu = float(np.convolve(a_cur, a_cur).max())
        mu = max(mu, 1.0)

        # continuation schedule (geometric cooling)
        mu_factor = 0.7
        mu_min = 1e-7
        n_restart = 1

        y = y0.copy()
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": 300, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:
                print(f"[lbfgs] failed: {exc}")
                break
            y = res.x
            a = S * _softmax(y)
            _maybe_update(a, tag="_lbfgs")
            mu *= mu_factor

    # ----------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
