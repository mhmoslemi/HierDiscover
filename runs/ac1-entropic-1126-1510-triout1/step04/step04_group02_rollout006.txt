def gradient_softmax(a, b, alpha, size_fft):
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1
    else:
        w /= w_sum
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

"""
Propose a step‑function minimizing `evaluate_sequence`.

Algorithm
---------

* Several candidate lengths (up to 40 000) are tried.
* For each length an *arcsine* distribution (a≈1/√{i( n‑i)}) is used as a
  symmetric start point on the probability simplex.
* A mirror‑descent (exponentiated‑gradient) optimiser works on the smooth
  surrogate   Fα(a)= (1/α)·log∑exp(α·(a∗a)) .  The temperature α is increased
  geometrically so that the surrogate converges to the true L∞‑norm.
  The gradient is obtained by a single FFT‑based convolution.
* After the main phase a short stochastic “local‑refinement’’ hill‑climbing
  (random mass transfers) is performed.
* The best sequence seen (according to the fast internal objective)
  is finally evaluated with the official `evaluate_sequence` function.
* The whole routine respects a 1000 s wall‑time budget and prints progress.

The construction is symmetric, stays on the simplex and never produces
negative entries, satisfying all checks of `evaluate_sequence`.

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (symmetrical, sum‑to‑one after normalisation)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b) w.r.t. a.
    Implements 2·conv_rev(w, a_rev) where w are the soft‑max weights.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _internal_score(a: np.ndarray) -> float:
    """Fast surrogate of the true score (sum(a)=1 assumed)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _optimize_softmax_exponential(a0: np.ndarray,
                                  n: int,
                                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using the soft‑max surrogate.
    Returns the best sequence (as a ndarray) seen before the deadline
    together with its internal (2·n·max) score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = _internal_score(a)

    # hyper‑parameters
    eta0 = 1.0                # initial step‑size factor
    alpha = 0.5               # start temperature for log‑sum‑exp
    alpha_mul = 1.02
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # for gradient
    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and current internal score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient of the smooth surrogate
        grad = _gradient_softmax(a, b, alpha, size_fft)

        # step‑size (diminishing)
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (mirrored step)
        a = a * np.exp(-eta * grad)

        # keep positivity
        a = np.maximum(a, 0.0)

        # enforce symmetry (the optimum is symmetric)
        a = (a + a[::-1]) / 2.0

        # project back onto the simplex
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a random fraction of mass
    from one index to another and keep any improvement.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = _internal_score(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        if eps <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= eps
        cand[j] += eps
        if cand[i] < 0.0:
            continue
        # evaluate
        cur_score = _internal_score(cand)
        if cur_score < best_score:
            best_score = cur_score
            best = cand.copy()
            a = cand
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 123,
                      budget_s: int = 1000,
                      **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The routine respects a 1000 s time budget and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # try to reuse a previously discovered solution if it exists
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # length candidates (capped at 40 000 for speed / memory)
    length_factors = (0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted({
        max(500, min(40000, int(base_n * f)))
        for f in length_factors
    })
    # add a few absolute values for robustness
    for extra in (5000, 10000, 20000, 30000, 40000):
        if extra >= 500:
            lengths.append(extra)
    lengths = sorted(set(lengths))

    overall_best_seq = None
    overall_best_val = float("inf")
    target_score = 1.503

    # ------------------------------------------------------------------
    for n in lengths:
        # remaining total time
        if time.time() > deadline - 5.0:
            break
        remaining = deadline - time.time()
        # allocate a slice (no more than 120 s) for this length
        slice_time = min(120.0, remaining * 0.35)
        slice_deadline = time.time() + slice_time

        # initialise with the symmetric arcsine distribution
        a0 = _arcsine_distribution(n)
        a0 = a0 / a0.sum()
        a0 = (a0 + a0[::-1]) / 2.0   # enforce exact symmetry

        # main optimisation phase
        seq, internal_score = _optimize_softmax_exponential(a0, n, slice_deadline)

        # evaluate with the official black‑box (more reliable)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = internal_score

        if final_score < overall_best_val:
            overall_best_val = final_score
            overall_best_seq = seq.copy()
            print(
                f"*** New global best → len={n} score={overall_best_val:.6f}"
            )

        if overall_best_val <= target_score:
            break

    # ------------------------------------------------------------------
    # final local refinement (use whatever time is left)
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n = overall_best_seq.shape[0]
        seq_ref, int_score_ref = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq_ref.tolist())
        except Exception:
            final_score = int_score_ref
        if final_score < overall_best_val:
            overall_best_val = final_score
            overall_best_seq = seq_ref.copy()
            print(f"*** After local refinement score={overall_best_val:.6f}")

    # ------------------------------------------------------------------
    # safety net – should never be triggered
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_val = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_val:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
