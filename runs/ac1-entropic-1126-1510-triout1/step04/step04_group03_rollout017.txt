"""
Propose a step‑function (list of non‑negative heights) that minimizes
`evaluate_sequence`.

Algorithm
---------

1. **Coarse parametric search** –  The optimal shape is approximately
   “U‑shaped” (more mass near the ends).  For each candidate length `n`
   we generate several families of distributions:
     * uniform,
     * the discrete arcsine (Beta(½,½)),
     * convex mixtures of the two,
     * power‑law shapes `a_i ∝ ((i+½)(n‑i‑½))^p` with `p∈[‑0.5,0]`.
   All candidates are normalised to sum 1 and evaluated with a fast
   FFT‑based convolution; the best candidate for this length is kept.

2. **Exact sub‑gradient descent** –  Starting from the best coarse candidate
   we minimise the objective `2·n·max(a*a)` on the probability simplex.
   The sub‑gradient of the max‑convolution is obtained analytically
   (indicator of the active indices).  A Nesterov‑style momentum update
   and a diminishing step size `η₀/√t` are used, together with occasional
   tiny random jitter to escape plateaus.  The loop runs until the allocated
   time slice expires.

3. **Local refinement** –  a cheap stochastic “mass‑swap’’ procedure that
   repeatedly moves a random fraction of mass between two positions and
   keeps the move if the max‑convolution improves.

4. **Global selection** –  several lengths (≈ ±20 % around a base length)
   are tried; the best sequence seen over all lengths is finally evaluated
   with the official `evaluate_sequence` and returned.

The whole search respects the 1000 s budget, using only FFT‑based
convolutions (`O(n log n)`) for the internal score and calling the
expensive `evaluate_sequence` only for the final few candidates.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Return the internal objective 2·n·max(a*a) (sum(a)=1)."""
    n = len(a)
    max_b = _fft_self_convolve(a).max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ ((i+0.5)*(n-i-0.5))**p   ,  p∈[-0.5,0] gives a U‑shaped family.
    The vector is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** p
    # Numerical safety – replace any NaN/inf by zero (should not happen)
    vals = np.where(np.isfinite(vals), vals, 0.0)
    total = vals.sum()
    if total == 0.0:
        # fallback to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / total


# ----------------------------------------------------------------------
def _refine_by_subgradient(a0: np.ndarray,
                           deadline: float,
                           verbose_prefix: str = "") -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the simplex for the objective
    2·n·max(a*a).  Returns the best sequence found before `deadline`.
    """
    a = a0.copy()
    n = len(a)
    best_seq = a.copy()
    best_score = _score_of(best_seq)

    velocity = np.zeros_like(a)
    beta = 0.9                 # momentum factor
    eta0 = 0.6                 # initial step‑size scale
    it = 0
    # pre‑allocate FFT size for the gradient (w * a_rev)
    size_fft = 1 << ((3 * n - 2).bit_length())

    while time.time() < deadline:
        it += 1
        # current convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] {verbose_prefix}it={it} score={best_score:.6f}")

        # --- exact sub‑gradient ---
        # indicator of the active (max) indices
        mask = np.isclose(b, cur_max, atol=1e-14)
        w = mask.astype(float)

        # gradient = 2 * (w * a_rev)   (convolution)
        a_rev = a[::-1]
        W = np.fft.rfft(w, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        conv = np.fft.irfft(W * A_rev, n=size_fft)
        start = n - 1
        grad = 2.0 * conv[start:start + n]

        # Nesterov‑style momentum
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny random jitter
        if it % 2500 == 0 and np.random.rand() < 0.25:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

def _local_search(seq: np.ndarray,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random fraction of mass
    from one index to another and keep the move if the max‑convolution
    improves.  Stops when `deadline` is reached.
    """
    a = seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_score = _score_of(best_seq)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] <= 1e-12:
            continue
        frac = np.random.rand() * 0.30                # up to 30 % transfer
        delta = best_seq[i] * frac
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _score_of(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand
        # tiny global jitter occasionally
        if it % 5000 == 0:
            cand = best_seq + np.random.randn(n) * 1e-7
            cand = _project_onto_simplex(cand, s=1.0)
            sc = _score_of(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
    return best_seq, best_score
