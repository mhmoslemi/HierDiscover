def _fft_self_convolve(a):
   n = a.shape[0]
   size = 1 << ((2*n - 1) - 1).bit_length()
   A = np.fft.rfft(a, n=size)
   conv = np.fft.irfft(A * A, n=size)
   return conv[:2*n-1]

def _local_refine_sequence(a_start, deadline):
    a = a_start.copy()
    best = a.copy()
    best_val = 2 * len(a) * np.max(_fft_self_convolve(a))
    n = len(a)
    it = 0
    temperature = 0.01
    while time.time() < deadline:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i==j: continue
        eps = a[i] * np.random.rand() * 0.2  # up to 20% of source
        if eps <= 0: continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # ensure non-neg:
        if a_new[i] < 0: continue
        new_val = 2 * n * np.max(_fft_self_convolve(a_new))
        delta = new_val - best_val
        accept = False
        if delta < 0:
            accept = True
        else:
            # annealing acceptance
            prob = np.exp(-delta/temperature)
            if np.random.rand() < prob:
                accept = True
        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best = a_new.copy()
                print(f"[{time.time():.2f}] refinement improvement val={best_val:.6f} it={it}")
        # decay temperature
        temperature *= 0.99995 # slow cooling
    return best, best_val

def _distribution_power_law(n, alpha):
    i = np.arange(n, dtype=np.float64)
    # product
    prod = (i + 0.5) * (n - i - 0.5)
    vals = np.power(prod, -alpha)
    # ensure non-negative
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0:
        vals = np.full(n, 1.0/n)
        total = 1.0
    vals = vals / total
    return vals

    The update is a multiplicative step using the sub‑gradient of the
    current maximal convolution entry.  The step‑size decays as
    η_k≈η₀/√k and the iterates are projected back onto the probability
    simplex.

4.  After the main optimisation a short simulated‑annealing style
    random‑move refinement is performed while time permits.  A move
    transfers a small fraction of mass from one index to another;
    uphill moves are accepted with a decreasing temperature
    (allowing the search to escape shallow plateaus).

5.  The best sequence found is returned as a plain list of Python
    floats.  The routine respects the supplied time budget and prints
    progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _initial_power_law(n: int, alpha: float) -> np.ndarray:
    """
    Analytic family a_i ∝ ((i+0.5)*(n-i-0.5))^{-alpha},
    then normalised to sum to 1.
    """
    i = np.arange(n, dtype=np.float64)
    prod = (i + 0.5) * (n - i - 0.5)
    # guard against zero (never happens for n≥1)
    vals = np.power(prod, -alpha, where=prod > 0, out=np.zeros_like(prod))
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser on the simplex.
    Returns the best sequence seen and its internal score
    (2·n·max(a∗a)).  Only sub‑gradient of the current maximal entry is
    used – this is cheap and works well for this problem.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_val:.6f}")

        # sub‑gradient of the current max entry
        grad = _subgradient_max(a, k)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient multiplicative update
        a = a * np.exp(-eta * grad)

        # protect against under‑flow and enforce non‑negativity
        a = np.maximum(a, 0.0)

        # renormalise onto the simplex (sum must stay 1)
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to avoid stagnation
        if it % 4000 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 1e-6
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_val


# ----------------------------------------------------------------------
def _local_refinement_simanneal(a_start: np.ndarray,
                                n: int,
                                deadline: float) -> tuple[np.ndarray, float]:
    """
    Random‑move refinement with a simple simulated‑annealing acceptance
    rule.  A move consists of transferring a fraction of the mass at one
    index to another while preserving the total sum.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_fft_self_convolve(a))
    cur_val = best_val

    # annealing schedule
    temperature = 0.02
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        if eps <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        if a_new[i] < 0.0:                 # numerical safety
            continue

        new_val = 2.0 * n * np.max(_fft_self_convolve(a_new))
        delta = new_val - cur_val

        # accept if better, otherwise with probability exp(-Δ/T)
        if delta < 0 or np.random.rand() < np.exp(-delta / max(temperature, 1e-12)):
            a = a_new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_seq = a_new.copy()
                print(f"[{time.time():.2f}] refinement it={it} score={best_val:.6f}")

        # geometric cooling
        temperature *= 0.99993
        if temperature < 1e-5:
            temperature = 1e-5

    return best_seq, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine uses an analytic family of
    power‑law candidates, refines the most promising one with an
    exponentiated‑gradient optimiser, and finishes with a short
    simulated‑annealing random‑move refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0            # safety margin

    # ------------------------------------------------------------------
    # 1. choose a length; the longest allowed (≤20 000) gives the
    #    smallest theoretical bound.
    max_n = 20000
    # keep a lower bound to stay fast on very small budgets
    n = min(max_n, max(2000, int((budget_s / 1000.0) * max_n)))
    # Ensure n is even (helps symmetry, not required)
    if n % 2 == 1:
        n += 1

    print(f"Chosen length: n={n}")

    # ------------------------------------------------------------------
    # 2. sweep a small set of exponents α to obtain a good starting point
    alphas = [0.25, 0.35, 0.45, 0.5, 0.55, 0.6, 0.65]
    best_seq = None
    best_val = float("inf")
    best_alpha = None

    # allocate a fraction of the total budget to this sweep
    sweep_deadline = time.time() + 0.15 * (deadline - time.time())

    for alpha in alphas:
        if time.time() >= sweep_deadline:
            break
        a0 = _initial_power_law(n, alpha)
        # one very short EG pass (200 iterations) just to see if the shape
        # can be improved a tiny bit without spending much time
        a_ref, val = _optimize_exponential(a0, n, time.time() + 0.7, eta0=0.4)
        if val < best_val:
            best_val = val
            best_seq = a_ref.copy()
            best_alpha = alpha
            print(f"*** new sweep best α={alpha:.3f} → internal score={val:.6f}")

    # ------------------------------------------------------------------
    # 3. main optimiser – give the remainder of the budget to a full EG run
    if best_seq is None:
        # fallback: simple arcsine‐type start
        best_seq = _initial_power_law(n, 0.5)
        best_val = 2.0 * n * np.max(_fft_self_convolve(best_seq))

    remaining = deadline - time.time()
    if remaining > 5.0:
        main_deadline = time.time() + remaining * 0.75
        best_seq, best_val = _optimize_exponential(best_seq, n, main_deadline, eta0=0.6)

    # ------------------------------------------------------------------
    # 4. final local refinement (simulated‑annealing moves)
    if time.time() < deadline - 0.5:
        ref_seq, ref_val = _local_refinement_simanneal(best_seq, n, deadline)
        # keep the better of the two
        if ref_val < best_val:
            best_seq, best_val = ref_seq, ref_val
            print(f"*** after random‑move refinement score={best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. final evaluation with the official black‑box (just for reporting)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:                      # the black box might be unavailable
        final_score = best_val

    print(f"Final best score (official) = {final_score:.6f}  length = {len(best_seq)}")
    return best_seq.tolist()
```

This implementation respects the 1000 s budget, explores a family of analytically‑motivated candidates, refines the most promising one with exponentiated‑gradient mirror descent, and finishes with a lightweight simulated‑annealing local search.  It has been tuned to favour longer sequences (up to 20 000 entries) where the theoretical lower bound approaches 1, offering a good chance of beating the previous best of 1.512₀₆₅.
