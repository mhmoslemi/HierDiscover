def _optimize_candidate(a0, size_fft_conv, size_fft_corr, target_sum, deadline, max_iters):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft_conv)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta0 = target_sum * 0.2 / a.size
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv = _full_conv(a, size_fft_conv)
        max_conv = float(conv.max())
        if max_conv < best_score:
            best_score = max_conv
            best_a = a.copy()
        # p schedule
        p = 2 + (30-2) * it / max_iters  # linear increase to 30
        # compute weights
        weights = conv ** (p-1)
        # gradient via correlation
        # compute correlation a with weights
        # Use size_fft_corr = appropriate size
        Fa = np.fft.rfft(a, n=size_fft_corr)
        Fw = np.fft.rfft(weights, n=size_fft_corr)
        grad = 2 * np.fft.irfft(np.conj(Fa) * Fw, n=size_fft_corr)[:a.size]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)
    return best_a, best_score

"""
Propose a step‑function (non‑negative sequence) with a low self‑convolution peak.

Algorithm
---------
1. Choose the problem size `n` (the length of the reference sequence if it
   exists, otherwise a default of 3000).  The ℓ₁‑norm is fixed to
   `S = sqrt(2·n)`.  With this normalisation the evaluation score equals
   the peak of the self‑convolution.

2. **Broad sampling** – generate a diverse pool of raw candidates
   (Dirichlet draws, power‑law / exponential shapes, random block‑wise
   constant vectors and sparse random patterns).  Each candidate is
   projected onto the simplex `{x≥0, Σx=S}` and its exact convolution
   peak (computed with an FFT) is recorded.  The `K` best raw candidates
   are kept.

3. **Weighted‑gradient descent** – for every kept candidate we run an
   Adam‑style optimizer that minimizes the smooth `p`‑norm of the full
   convolution:
        Lₚ(a) = ( Σₖ (a*a)ₖᵖ )¹ᐟᵖ   ,  p → large  ≈ max(a*a) .
   The gradient is obtained analytically via a correlation, which is
   evaluated with FFTs.  `p` is increased from 2 up to 20 during the run,
   allowing the optimizer to first smooth the landscape and then focus on
   the worst peak.  After each step the vector is projected back onto the
   simplex.  The best vector seen during the run is remembered.

4. **Local permutation hill‑climbing** – with the remaining time a simple
   swap‑based hill‑climber is applied to the current global best.  Random
   swaps are accepted only if they lower the peak of the convolution; an
   occasional small random perturbation helps to escape plateaus.

5. Return the best sequence discovered before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (efficient FFT size)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear self‑convolution a * a (length 2·n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution (no clipping)."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, conv: np.ndarray, p: float,
                       size_fft_corr: int) -> np.ndarray:
    """
    Gradient of   ( Σₖ convₖᵖ )¹ᐟᵖ   w.r.t. a.
    It equals 2·corr(a , conv^{p‑1}) where corr denotes the (full) correlation.
    """
    # weight = conv^{p-1}
    weight = conv ** (p - 1.0)

    # FFT of a and weight (both real, use rfft)
    Fa = np.fft.rfft(a, n=size_fft_corr)
    Fw = np.fft.rfft(weight, n=size_fft_corr)

    # correlation a ⋆ weight = irfft(conj(Fa) * Fw)
    corr_full = np.fft.irfft(np.conj(Fa) * Fw, n=size_fft_corr)

    # gradient = 2 * first n entries of correlation
    grad = 2.0 * corr_full[: a.size]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   target_sum: float,
                   size_fft_conv: int,
                   size_fft_corr: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smooth p‑norm of the convolution.
    Returns the best vector (lowest peak) and its peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft_conv)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base step size (order of magnitude of a single entry)
    eta0 = target_sum * 0.2 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution (full) -----
        conv = _full_conv(a, size_fft_conv)          # length 2n-1
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[opt] it {it:,} new best peak = {best_score:.6f}")

        # ----- p schedule (smooth → sharp) -----
        # start at 2, finish at 20
        p = 2.0 + 18.0 * (it - 1) / max_iters

        # ----- weighted gradient -----
        grad = _weighted_gradient(a, conv, p, size_fft_corr)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex (keeps Σa = target_sum) -----
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score

    """
    Search for a non‑negative step‑function minimising the evaluation score.
    Returns the best sequence found within the time budget.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------

        n = len(height_sequence_1)   # type: ignore[name-defined]

    size_fft_conv = _fft_size(2 * n - 1)               # for self‑convolution
    size_fft_corr = _fft_size(3 * n - 2)               # for correlation in gradient

    # --------------------------------------------------------------
    # Helpers
    def quick_score(a: np.ndarray) -> float:
        return _max_convolution(a, size_fft_conv)

    # keep top‑K raw candidates (ascending by peak)
    K = 12
    best_raw: list[tuple[float, np.ndarray]] = []   # (score, a)

    def push_candidate(score: float, a: np.ndarray) -> None:
        if len(best_raw) < K:
            best_raw.append((score, a.copy()))
            best_raw.sort(key=lambda x: x[0])
        else:
            if score < best_raw[-1][0]:
                best_raw[-1] = (score, a.copy())
                best_raw.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 1. Broad random sampling
    idx = np.arange(n)

    # reference (if available)
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        push_candidate(quick_score(a_ref), a_ref)
        print(f"[init] reference peak = {quick_score(a_ref):.6f}")

    # uniform baseline
    a_uni = np.ones(n)
    a_uni = _project_to_simplex(a_uni, target_sum)
    push_candidate(quick_score(a_uni), a_uni)
    print(f"[init] uniform peak = {quick_score(a_uni):.6f}")

    # Dirichlet draws (various concentrations)
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(10):

            a = probs * target_sum
            push_candidate(quick_score(a), a)

    # Power‑law decreasing shapes
    for _ in range(40):
        p = np.random.rand() * 3.0 + 0.2   # exponent ∈ [0.2, 3.2]
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        push_candidate(quick_score(a), a)

    # Exponential decays
    for _ in range(40):
        dec = np.random.rand() * 0.015 + 0.001   # decay ∈ [1e‑3, 1.6e‑2]
        a = np.exp(-dec * idx)
        a = _project_to_simplex(a, target_sum)
        push_candidate(quick_score(a), a)

    # Random block‑wise constant vectors (2‑6 blocks)
    for _ in range(150):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        push_candidate(quick_score(cand), cand)

    # Sparse random patterns (≈5 % non‑zeros)

        mask = np.random.rand(n) < 0.05

        cand[mask] = vals
        cand = _project_to_simplex(cand, target_sum)
        push_candidate(quick_score(cand), cand)

    # --------------------------------------------------------------

    global_best_a = best_raw[0][1].copy()
