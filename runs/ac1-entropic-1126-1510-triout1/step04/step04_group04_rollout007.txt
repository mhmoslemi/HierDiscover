"""
Propose a step‑function (non‑negative sequence) with a low self‑convolution peak.

Algorithm
---------
1. Choose a problem size `n`.  If a reference sequence is available we
   embed it among zeros, otherwise we use a moderately large size (n≈6000)
   – larger than before, which gives the optimizer more degrees of freedom.

2. Build a pool of diverse raw candidates (uniform, power‑law, exponential,
   random block‑wise constant, a few Dirichlet draws).  Each candidate is
   projected onto the simplex {x≥0, Σx = √(2 n)} and its exact peak
   `max(conv)` is recorded.  The best `K` candidates are kept.

3. **Smoothed‑gradient descent** – for each of the top candidates we run
   an Adam‑style optimisation on a *soft‑max* surrogate of the peak:
        fα(a) = log‑sum‑exp( α·(a*a) ).
   The smoothing parameter `α` is increased on a schedule
   (0.1 → … → 300) so the optimizer first sees a smooth landscape and
   later focuses on the true maximum.  After each step we project back to
   the simplex.

4. **Hard‑max refinement** – the best sequence obtained above is refined
   with an Adam‑style sub‑gradient descent that directly minimises the
   true `max(a*a)`.

5. **Local simulated‑annealing** – a cheap hill‑climber that swaps
   positions or moves a tiny amount of mass between two entries.
   Occasionally worse moves are accepted with a temperature that decays
   over time, allowing the search to escape shallow plateaus.

6. Return the best sequence found before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return a power‑of‑2 size ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a using FFT (length = 2 n‑1)."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution (no clipping)."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, size_fft: int, alpha: float):
    """
    Gradient of the smoothed objective
        fα(a) = log‑sum‑exp( α·(a*a) )
    Returns (gradient, exact_max_conv_of_a).
    """
    # (a*a) – exact convolution
    conv = _full_conv(a, size_fft)                     # length L = 2n‑1
    # soft‑max weights
    scaled = conv * alpha
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)            # stabilise
    w = exp_vals / np.sum(exp_vals)                   # Σw = 1

    # gradient = 2 * (a_rev * w) taken at the centre slice
    rev_a = a[::-1]
    L = conv.shape[0]                                 # = 2n‑1
    size_fft2 = _fft_size(len(rev_a) + L - 1)
    Fa = np.fft.rfft(rev_a, n=size_fft2)
    W = np.fft.rfft(w, n=size_fft2)
    conv_rev = np.fft.irfft(Fa * W, n=size_fft2)

    mid = L // 2                                      # = n‑1 (centre)
    grad = 2.0 * conv_rev[mid:mid + a.shape[0]]
    return grad, float(conv.max())


# ----------------------------------------------------------------------
def _smooth_descent(a0: np.ndarray,
                    size_fft: int,
                    target_sum: float,
                    deadline: float,
                    alphas,
                    iters_per_alpha: int):
    """
    Adam‑style descent on the smoothed max‑convolution.
    Returns the best (lowest exact max) sequence seen and its exact max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    eta0 = target_sum * 0.4 / a.size   # an empirical base step size

    for alpha in alphas:
        for _ in range(iters_per_alpha):
            if time.time() >= deadline:
                return best_a, best_score

            grad, cur_max = _smooth_gradient(a, size_fft, alpha)
            if cur_max < best_score:
                best_score = cur_max
                best_a = a.copy()

            t += 1
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            eta = eta0 / math.sqrt(t)
            a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

            # project onto the simplex Σa = target_sum, a ≥ 0
            a = _project_to_simplex(a, target_sum)

    # final best record
    return best_a, best_score


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """Exact sub‑gradient of max_k (a*a)_k at the current worst index k_max."""
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        start = k_max - n + 1
        length = 2 * n - 1 - k_max
        g[start:] = 2.0 * a[:length][::-1]
    return g


# ----------------------------------------------------------------------
def _refine_subgrad(a0: np.ndarray,
                    size_fft: int,
                    target_sum: float,
                    deadline: float,
                    max_iters: int = 200_000):
    """
    Adam‑style sub‑gradient descent on the *hard* max‑convolution.
    Returns the best (lowest peak) sequence and its exact max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[hard] it {it:,} new best = {best_score:.6f}")

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_search_sa(a0: np.ndarray,
                     size_fft: int,
                     target_sum: float,
                     deadline: float):
    """
    Simple simulated‑annealing / hill‑climbing on permutations and
    tiny mass transfers.  Returns the best sequence discovered.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)
    cur_score = best_score
    n = a.shape[0]

    # temperature roughly 10 % of current value; it shrinks slowly
    temp = max(best_score * 0.1, 1e-3)

    it = 0
    while time.time() < deadline:
        it += 1

        # choose move type
        if np.random.rand() < 0.7:               # swap two entries
            i, j = np.random.choice(n, 2, replace=False)
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:                                    # move a tiny amount of mass
            i = np.random.randint(n)
            if a[i] <= 1e-12:
                continue
            eps = min(a[i], np.random.rand() * target_sum * 0.02)
            j = np.random.randint(n)
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps

        new_score = _max_convolution(a_new, size_fft)
        delta = new_score - cur_score

        if delta < 0 or np.exp(-delta / temp) > np.random.rand():
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[SA] it {it:,} improved to {best_score:.6f}")

        # geometric cooling
        temp = max(temp * 0.99995, 1e-6)

        # occasional early‑exit check
        if it % 1000 == 0 and time.time() + 0.01 >= deadline:
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(...).  Runs for at most `budget_s` seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ---------- problem size ----------
    try:
        n_ref = len(height_sequence_1)       # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n_ref = 0
        have_ref = False

    # Use a larger size than before – more degrees of freedom.
    n = max(6000, n_ref)
    target_sum = math.sqrt(2.0 * n)          # Σa = √(2 n)
    size_fft = _fft_size(2 * n - 1)

    # ---------- helpers ----------
    def quick_score(a: np.ndarray) -> float:
        return _max_convolution(a, size_fft)

    # maintain top‑K raw candidates (ascending by score)
    K = 12
    best_raw: list[tuple[float, np.ndarray]] = []   # (score, a)

    def push_candidate(score: float, a: np.ndarray) -> None:
        if len(best_raw) < K:
            best_raw.append((score, a.copy()))
            best_raw.sort(key=lambda x: x[0])
        else:
            if score < best_raw[-1][0]:
                best_raw[-1] = (score, a.copy())
                best_raw.sort(key=lambda x: x[0])

    # ---------- 1. sampling ----------
    idx = np.arange(n)

    # reference (if any) – embed it into a longer zero‑padded vector
    if have_ref:
        a_ref = np.zeros(n, dtype=float)
        start = np.random.randint(0, n - n_ref + 1)
        a_ref[start:start + n_ref] = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        push_candidate(quick_score(a_ref), a_ref)
        print(f"[init] reference embedded score = {quick_score(a_ref):.6f}")

    # uniform baseline
    a_uni = np.ones(n)
    a_uni = _project_to_simplex(a_uni, target_sum)
    push_candidate(quick_score(a_uni), a_uni)
    print(f"[init] uniform score = {quick_score(a_uni):.6f}")

    # Dirichlet draws (a few, low‑dimensional concentration)
    for alpha in [0.02, 0.05, 0.1, 0.5, 1.0]:
        for _ in range(5):
            probs = np.random.dirichlet(np.full(n, alpha))
            a = probs * target_sum
            push_candidate(quick_score(a), a)

    # Power‑law decreasing shapes
    for _ in range(30):
        p = np.random.rand() * 3.0 + 0.2               # exponent ∈ [0.2, 3.2]
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        push_candidate(quick_score(a), a)

    # Exponential decay shapes
    for _ in range(30):
        dec = np.random.rand() * 0.015 + 0.001        # decay ∈ [1e‑3, 1.6e‑2]
        a = np.exp(-dec * idx)
        a = _project_to_simplex(a, target_sum)
        push_candidate(quick_score(a), a)

    # Random block‑wise constant sequences (2‑8 blocks)
    for _ in range(150):
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        push_candidate(quick_score(cand), cand)

    print(f"[sampling] best raw score = {best_raw[0][0]:.6f}")

    # ---------- 2. smoothed‑gradient descent ----------
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    # schedule of smoothing parameters
    alphas = [0.1, 0.5, 1.0, 2.0, 5.0, 10.0, 30.0, 100.0, 300.0]
    iters_per_alpha = 2000

    for rank, (raw_sc, a0) in enumerate(best_raw[:5]):   # only top‑5 for this phase
        if time.time() >= deadline:
            break
        print(f"[smooth start] candidate #{rank+1} raw score {raw_sc:.6f}")
        a_s, sc_s = _smooth_descent(a0, size_fft, target_sum,
                                    deadline, alphas, iters_per_alpha)
        if sc_s < global_best_score:
            global_best_score = sc_s
            global_best_a = a_s.copy()
            print(f"[global] after smooth #{rank+1} = {global_best_score:.6f}")

    # ---------- 3. hard‑max (sub‑gradient) refinement ----------
    if time.time() < deadline:
        print("[hard max refinement] start")
        # allocate as many iterations as time permits (≈ 150 k max)
        remaining_time = deadline - time.time()
        # each iteration costs ≈ 0.0005 s → allow ~200 k if enough time
        max_iters = int(min(300_000, remaining_time * 1800))
        refined_a, refined_sc = _refine_subgrad(global_best_a, size_fft,
                                                target_sum, deadline,
                                                max_iters=max_iters)
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] after hard refinement = {global_best_score:.6f}")

    # ---------- 4. local simulated annealing ----------
    if time.time() < deadline:
        print("[local SA] start")
        a_sa, sc_sa = _local_search_sa(global_best_a, size_fft,
                                       target_sum, deadline)
        if sc_sa < global_best_score:
            global_best_score = sc_sa
            global_best_a = a_sa.copy()
            print(f"[global] after SA = {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    # Return as plain Python list (evaluate_sequence will handle conversion)
    return global_best_a.tolist()
