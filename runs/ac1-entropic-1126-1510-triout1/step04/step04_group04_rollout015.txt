"""
Propose a non‑negative step function with a low self‑convolution peak.

Algorithm
---------
1. Choose a problem size `n` (use the reference sequence if it exists,
   otherwise a moderately large size, e.g. 8000).  The ℓ₁‑norm is forced to
   `S = √(2 n)` because the evaluation score equals the peak of the
   convolution under this normalisation.

2. **Sampling** – generate a diverse pool of raw candidates:
   * uniform baseline,
   * Dirichlet draws with several concentration parameters,
   * power‑law / exponential decays,
   * random block‑wise constant sequences,
   * **sparse Sidon‑like sets** – greedy construction that keeps the
     multiplicity of every sum `i+j` ≤ `max_coll`.  By targeting a set
     size slightly larger than `S` (≈ 1.2 S) we obtain a small weight
     `w = S / |P|` and, thanks to the low collision count, an excellent
     initial peak.

   The best `K` raw candidates (lowest peak) are kept.

3. **Refinement** – each top candidate is polished by a projected Adam
   sub‑gradient descent on the true `max`‑convolution (the sub‑gradient
   is exact for the active index).  The iteration count is adapted to
   the remaining time budget.

4. **Local permutation search** – a simple hill‑climbing swaps two
   positions if it lowers the peak.  Occasionally a small random
   perturbation together with a simplex projection is performed to
   escape plateaus.

5. Return the best sequence found before the 1000 s budget expires.

The implementation is completely self‑contained, uses only NumPy and
Python's standard library and obeys the required function signature.
"""

import math
import time
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for FFT efficiency)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a * a (full length = 2·n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak value of the self‑convolution."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of max_i (a*a)_i at the current worst index k_max."""
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        # overlap of a[0..k_max] with reversed a[k_max::-1]
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        # k_max >= n
        start = k_max - n + 1
        length = 2 * n - 1 - k_max
        g[start:] = 2.0 * a[:length][::-1]

    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the hard max‑convolution.
    Returns the best sequence encountered and its peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # a rough step size proportional to the typical entry size
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[refine] it {it:,} new best = {best_score:.6f}")

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # keep feasibility
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _generate_sparse_sidonic_set(
    n: int,
    desired_m: int,
    max_coll: int = 2,
    max_trials: int = 30,
) -> tuple[np.ndarray, int]:
    """
    Greedy construction of a set of indices `P ⊂ {0,…,n‑1}` such that
    each sum `i+j` appears at most `max_coll` times.
    Returns the array of positions and the maximal sum multiplicity actually attained.
    """
    best_positions = []
    best_max_coll = max_coll

    for _ in range(max_trials):
        positions = []
        sum_counts = {}    # maps sum → current multiplicity
        order = np.random.permutation(n)

        for x in order:
            # check if adding x would violate the bound
            ok = True
            # self‑sum
            s_self = x + x
            if sum_counts.get(s_self, 0) + 1 > max_coll:
                ok = False
            else:
                for y in positions:
                    s = x + y
                    if sum_counts.get(s, 0) + 1 > max_coll:
                        ok = False
                        break
            if not ok:
                continue

            # accept x
            for y in positions:
                s = x + y
                sum_counts[s] = sum_counts.get(s, 0) + 1
            sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
            positions.append(int(x))

            if len(positions) >= desired_m:
                break

        # compute the realised maximal multiplicity
        cur_max = max(sum_counts.values()) if sum_counts else 0
        if len(positions) > len(best_positions) or (
            len(positions) == len(best_positions) and cur_max < best_max_coll
        ):
            best_positions = positions
            best_max_coll = cur_max
            if len(best_positions) >= desired_m and best_max_coll <= max_coll:
                break

    return np.array(best_positions, dtype=int), best_max_coll

            idx1 = np.random.randint(n)
            delta = (np.random.rand() * 2 - 1) * 0.02 * target_sum
            if a[idx1] - delta > 0:
                a2 = a.copy()
                a2[idx1] -= delta
                a2 = _project_to_simplex(a2, target_sum)
                sc2 = _max_convolution(a2, size_fft)
                if sc2 < best_sc:
                    best_sc = sc2
                    a = a2
                    global_best_a = a2.copy()
                    global_best_score = sc2
                    print(f"[perturb] improved to {best_sc:.6f}")

        # early exit if only a few milliseconds are left
        if time.time() + 0.01 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    # final projection (numerical drift)
    final_a = _project_to_simplex(global_best_a, target_sum)
    return final_a.tolist()
