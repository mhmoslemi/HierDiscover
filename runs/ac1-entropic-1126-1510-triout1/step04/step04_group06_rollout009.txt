def _simplex_project(v, target):
    v = np.maximum(v, 0)
    s = v.sum()
    if s == 0:
        v = np.full_like(v, target / v.size)
    else:
        v = v * (target / s)
    return v

def _refine_adam_simple(a0, deadline, n, size_fft, target_sum):
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta0 = target_sum * 0.3 / n  # maybe adjust

    # initial temperature
    conv = _self_convolution(a, size_fft)
    T = max(1e-3, (conv.max() - conv.min()) * 0.3)

    it = 0
    while time.time() < deadline:
        conv = _self_convolution(a, size_fft)
        cur_score = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # compute softmax weights
        shifted = conv - conv.max()
        exp_arg = shifted / max(T, 1e-12)
        np.clip(exp_arg, -50, 0, out=exp_arg)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = _weighted_gradient(a, w, size_fft)

        eta = eta0 / (1.0 + it * 1e-5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # optional kick
        if it % 2000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a += noise
            a = np.maximum(a, 0)
            a = a * (target_sum / a.sum())

        # temperature cooling
        T *= 0.9995
        it += 1
        # small break if it reaches large iteration count, but time condition ensures stop.
    return best_a, best_score

import numpy as np

def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    (w must sum to 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


def _simplex_project(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to non‑negative and renormalise to the simplex sum."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        v = np.full_like(v, target_sum / v.size)
    return v


def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective used during the search (max of self‑convolution)."""
    conv = _self_convolution(a, size_fft)
    return float(conv.max())


def _refine_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max approximation of the max‑convolution.
    Returns the best (lowest‑score) vector observed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # base learning rate (scale ~ target_sum / n)
    eta0 = target_sum * 0.5 / n

    # initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break

        conv = _self_convolution(a, size_fft)
        cur_score = float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights (smooth approx. of max) -----
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        np.clip(exp_arg, -50, 0, out=exp_arg)      # protect from overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        eta = eta0 / (1.0 + it * 1e-5)               # very mild decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex (clip + renormalise) -----
        a = _simplex_project(a, target_sum)

        # occasional random kick to escape plateaus
        if (it % 5000) == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a = a + noise
            a = _simplex_project(a, target_sum)

        # temperature cooling
        T *= 0.9997
        it += 1

    return best_a, best_score

    Search for a non‑negative step‑function that minimises the self‑convolution
    peak.  The function respects a 1000 s time budget and returns the best

    # ------------------------------------------------------------------
    # 1. generate a diverse set of raw candidates
    raw_candidates = []

    def add_candidate(arr: np.ndarray) -> None:
        arr = _simplex_project(arr, target_sum)
        sc = quick_score(arr, n, size_fft)
        raw_candidates.append((sc, arr))

    # (a) uniform
    add_candidate(np.ones(n))

    # (b) beta‑shaped families
    for alpha, beta in [(0.5, 0.5), (1.0, 2.0), (2.0, 2.0), (0.2, 0.7), (3.0, 5.0)]:
        a = (idx + 1) ** alpha * (n - idx) ** beta
        add_candidate(a)

    # (c) inverse‑sqrt (single‑ended)
    for _ in range(80):
        beta = np.random.rand() * 12.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        add_candidate(a)

    # (d) double‑ended inverse‑sqrt
    for _ in range(80):
        beta = np.random.rand() * 12.0
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        add_candidate(a)

    # (e) power‑law
    for _ in range(80):
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 1.8 + 0.2
        a = 1.0 / (idx + beta + 1.0) ** pexp
        add_candidate(a)

    # (f) double exponential (possibly asymmetric)
    for _ in range(80):
        decay_l = np.random.rand() * 0.015 + 0.0002
        decay_r = np.random.rand() * 0.015 + 0.0002
        a = np.exp(-decay_l * idx) + np.exp(-decay_r * (n - 1 - idx))
        add_candidate(a)

    # (g) cosine‑based (large at ends, small in centre)
    for _ in range(80):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        add_candidate(a)

    # (h) sine‑based (zero at ends, peak in centre)
    for _ in range(80):
        p = np.random.rand() * 6.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** p
        add_candidate(a)

    # (i) arcsine‑like shape (a_i ∝ 1/√(i·(n‑i)))
    for _ in range(40):
        eps = 0.5
        a = 1.0 / np.sqrt((idx + eps) * (n - idx - eps))
        add_candidate(a)

    # (j) random block candidates (5‑20 blocks)
    for _ in range(300):
        nb = np.random.randint(5, 21)                 # 5‑20 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        add_candidate(cand)

    # (k) reference sequence if available
    if have_ref:
        add_candidate(np.array(height_sequence_1, dtype=float))

    # keep the K best raw candidates (ascending score)
    K = 12
    raw_candidates.sort(key=lambda x: x[0])
    best_raw = raw_candidates[:K]
    print(f"[exploration] best raw score = {best_raw[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. quick refinement of the top raw candidates
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    for rank, (raw_sc, cand) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        print(f"[quick‑refine] start #{rank+1} (raw {raw_sc:.6f})")
        # give each candidate a modest budget (≈0.1 s per candidate)
        rem = deadline - time.time()
        max_iters = int(min(25_000, rem * 1400))
        refined_a, refined_sc = _refine_adam(cand, deadline, n, size_fft,
                                            target_sum, max_iters)
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] improved to {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. deep optimisation of the current global best until time runs out
    if time.time() < deadline:
        remaining = deadline - time.time()
        max_iters = None                     # run until deadline
        print(f"[deep‑optim] remaining time {remaining:.1f}s, starting long run")
        global_best_a, global_best_score = _refine_adam(
            global_best_a, deadline, n, size_fft, target_sum, max_iters
        )
        print(f"[deep‑optim] final deep score = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. final safety: one last random‑block sweep (tiny budget)
    while time.time() < deadline - 0.05:
        nb = np.random.randint(5, 21)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _simplex_project(cand, target_sum)
        sc = quick_score(cand, n, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final‑sweep] improved to {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
