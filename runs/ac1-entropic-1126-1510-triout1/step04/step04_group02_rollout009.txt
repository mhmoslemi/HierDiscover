    eps = 1e-12
    y = np.log(a0 + eps)  # shape (n,)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    t_iter = 0
    alpha = 0.5  # temperature for softmax
    alpha_mul = 1.05
    alpha_thr = 2000 # might be final stage
    best_score = float('inf')
    best_seq = None

    while time.time() < deadline:
        t_iter += 1
        # compute probabilities a
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        # compute convolution and scores
        b = _fft_self_convolve(a)
        cur_max = b.max()
        # internal score = 2*n*cur_max
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={t_iter} score={best_score:.6f}")

        # gradient
        if alpha > alpha_thr:
            # hard subgradient
            k = int(np.argmax(b))
            grad_a = _subgradient_max(a, k)
        else:
            # softmax gradient
            grad_a = _gradient_softmax(a, b, alpha, size_fft=int(1 << ((3*n-2).bit_length())))

        # compute gradient w.r.t logits y
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)
        lr = 0.5 / (np.sqrt(t_iter) + 1)  # decreasing LR

        y = y - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # update temperature
        if alpha < alpha_thr:
            alpha = min(alpha_thr, alpha * alpha_mul)
        else:
            # after surpassing thr, keep high enough maybe
            alpha = min(alpha_thr * 10, alpha * 1.01)  # slowly increase

        # occasional jitter

"""
Optimisation of a non‑negative step‑function (probability vector) to
minimise `evaluate_sequence`.  The optimiser uses:

* soft‑max (log‑sum‑exp) surrogate for the ℓ∞‑norm of the
  self‑convolution, with a temperature that grows until the surrogate
  becomes essentially the true max,
* Adam (adaptive moment) updates on the log‑odds (mirror‑descent with
  KL‑geometry) and projection onto the probability simplex via the
  soft‑max,
* several initialisers (arcsine, β‑pdf with random parameters, uniform,
  random) and many independent runs at a few different lengths,
* a final stochastic hill‑climbing refinement that moves mass between
  indices.

The routine respects a 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_pdf_distribution(n: int, a: float, b: float) -> np.ndarray:
    """
    Deterministic Beta( a , b ) probability vector on n points.
    The points correspond to the mid‑points (i+0.5)/n in (0,1).
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # protect against overflow/under‑flow by working in log‑space
    log_vals = (a - 1) * np.log(x) + (b - 1) * np.log(1 - x)
    # shift for numeric stability
    log_vals -= np.max(log_vals)
    vals = np.exp(log_vals)
    total = vals.sum()
    if total == 0.0 or np.isinf(total):
        # fallback to arcsine if something went wrong
        return _arcsine_distribution(n)
    return vals / total

def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a probability vector of length n according to ``kind``."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a = a / a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "beta":
        # random β‑pdf parameters
        a_param = np.random.uniform(0.2, 2.0)
        b_param = np.random.uniform(0.2, 2.0)
        a = _beta_pdf_distribution(n, a_param, b_param)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    # exact projection onto the simplex (numerical safety)
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b_k) w.r.t. a using the soft‑max weights.
    Returns a vector of size n.
    """
    # Form soft‑max weights over the convolution entries
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        # Numerical catastrophe – fall back to hard max
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # Convolution of reversed a with w (see derivation in the comment)
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑odds (softmax) parameterisation.
    Returns best sequence seen and its internal (2·n·max) score.
    """
    # initialise logits (log‑odds) – a stable start by taking log of a0
    eps = 1e-12
    y = np.log(a0 + eps)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # soft‑max temperature for the smooth max (starts low, then grows)
    alpha = 0.2
    alpha_mul = 1.03
    alpha_thr_hard = 5000.0          # switch to hard sub‑gradient after this

    size_fft = 1 << ((3 * n - 2).bit_length())

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # ---------- probability vector ----------
        y_shift = y - np.max(y)                     # numerical stability
        a = np.exp(y_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # ---------- objective ----------
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # ---------- gradient ----------
        if alpha > alpha_thr_hard:
            # hard sub‑gradient (exact max)
            k = int(np.argmax(b))
            grad_a = _subgradient_max(a, k)
        else:
            # smooth approximation (soft‑max)
            grad_a = _gradient_softmax(a, b, alpha, size_fft)

        # ---------- chain rule: ∂/∂y ----------
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)   # ∂L/∂y = Jᵀ·∂L/∂a for softmax

        # ---------- Adam update ----------
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # decreasing stepsize, but still fairly aggressive
        lr = 0.6 / (np.sqrt(it) + 1.0)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- temperature schedule ----------
        if alpha < alpha_thr_hard:
            alpha = min(alpha_thr_hard, alpha * alpha_mul)

        # occasional jitter to escape tiny plateaus
        if it % 2500 == 0 and np.random.rand() < 0.15:
            y += np.random.randn(n) * 1e-4

    # best_seq is guaranteed non‑None (at least the start point)
    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that moves a random fraction of mass
    from one index to another.  Accepts only improving moves; it is fast
    and works well as a final polishing step.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move at most 15 % of the source mass
        eps = a[i] * np.random.rand() * 0.15
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = 2.0 * n * np.max(_fft_self_convolve(a_candidate))
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises `evaluate_sequence`.  The routine respects the given time
    budget and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0          # leave a tiny safety margin

    # --------------------------------------------------------------
    # base length – reuse previous solution if present
    try:
        base_n = len(height_sequence_1)                # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # candidate lengths (capped at 20000 for speed)
    factors = (0.6, 0.8, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in factors
        if max(500, int(base_n * f)) <= 20000
    })
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503                     # stop early if we beat the target

    # --------------------------------------------------------------
    for n in lengths:
        remaining = deadline - time.time()
        if remaining <= 5.0:
            break

        # allocate a slice of the remaining budget to this length
        slice_deadline = time.time() + max(0.3, remaining * 0.35)

        # -----------------------------------------------------------------
        # 1) Gradient‑based optimisation from several initial seeds
        for init_kind in ("prev", "arcsine", "beta", "random", "uniform"):
            if time.time() >= slice_deadline:
                break

            a0 = _initial_sequence(n, init_kind)
            seq, _ = _adam_optimize(a0, n, slice_deadline)

            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                # fallback to the internal surrogate score
                final_score = 2.0 * n * np.max(_fft_self_convolve(seq))

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq.copy()
                print(
                    f"*** New global best → len={n} init={init_kind} "
                    f"score={overall_best_score:.6f}"
                )

            if overall_best_score <= target_score:
                break

        if overall_best_score <= target_score:
            break

        # -----------------------------------------------------------------
        # 2) Random Beta‑pdf search (explores the (a,b) parameter space)
        while time.time() < slice_deadline:
            a_param = np.random.uniform(0.1, 2.0)
            b_param = np.random.uniform(0.1, 2.0)
            seq = _beta_pdf_distribution(n, a_param, b_param)

            # cheap surrogate score
            conv_max = np.max(_fft_self_convolve(seq))
            cand_score = 2.0 * n * conv_max
            if cand_score >= overall_best_score - 1e-5:
                continue

            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = cand_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq.copy()
                print(
                    f"*** Beta‑pdf improve → len={n} a={a_param:.3f} b={b_param:.3f} "
                    f"score={overall_best_score:.6f}"
                )
                if overall_best_score <= target_score:
                    break

        if overall_best_score <= target_score:
            break

    # --------------------------------------------------------------
    # final local refinement (use whatever time is left)
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n = len(overall_best_seq)
        seq, _ = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = 2.0 * n * np.max(_fft_self_convolve(seq))

        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq.copy()
            print(f"*** After local refinement score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # safety fallback (should never happen)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
