"""
Propose a non‑negative step‑function (sequence) that minimizes the peak of its
self‑convolution.  The algorithm consists of three phases:

1. **Broad sampling** – generate many random candidates (Dirichlet, power‑law,
   exponential, block‑wise constant, sinusoidal), project each onto the simplex
   Σa = √(2n) and keep the best raw one.

2. **Smooth‑max gradient refinement** – minimise a differentiable surrogate of
   the L∞‑norm, namely  τ·log∑exp(conv/τ).  The gradient is obtained efficiently
   via FFTs, an Adam‑style step is taken, and the iterate is projected back onto
   the simplex.  The temperature τ is annealed during optimisation.

3. **Permutation annealing** – starting from the refined vector, repeatedly
   swap two positions (or perform tiny mass transfers) and accept the move with
   a Metropolis rule.  This explores the combinatorial space of orderings,
   which heavily influences the convolution peak.

The whole search respects a 1000 s budget; if the time expires the best sequence
found so far is returned.  All numbers are kept non‑negative and ≤ 1000, so the
call to ``evaluate_sequence`` will simply return the maximal convolution value.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a * a (full length = 2·n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution (no clipping)."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, tau: float, size_fft: int) -> np.ndarray:
    """
    Gradient of τ·log∑exp(conv/τ) w.r.t. a (τ > 0).
    The expression is 2·(w_rev * a) where w_i ∝ exp(conv_i/τ).
    """
    # convolution a*a
    conv = _full_conv(a, size_fft)                     # length 2n‑1
    # softmax weights
    exponent = np.clip(conv / tau, -500, 500)          # avoid overflow
    w = np.exp(exponent)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w = w / w_sum

    # reversed weights
    w_rev = w[::-1]

    # convolution w_rev * a (full)
    L = w_rev.shape[0]
    size_fft2 = _fft_size(L + a.size - 1)
    Fa = np.fft.rfft(a, n=size_fft2)
    Fw = np.fft.rfft(w_rev, n=size_fft2)
    grad_full = np.fft.irfft(Fa * Fw, n=size_fft2)

    offset = L - 1                     # index corresponding to m = 0
    idx = offset - np.arange(a.size)   # vector of indices offset‑m
    g = 2.0 * grad_full[idx]           # factor 2 as in sub‑gradient definition
    return g


# ----------------------------------------------------------------------
def _refine_smooth(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Annealed smooth‑max gradient descent (Adam‑style) on the max‑convolution.
    Returns the best (lowest peak) sequence found and its exact peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # initial learning‑rate scale (≈ half of the average entry)
    eta0 = target_sum * 0.5 / a.size

    tau = 0.5                 # initial temperature for the smooth‑max
    tau_min = 0.01
    tau_decay = 0.995         # applied every few thousand iterations

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        if it % 2000 == 0:
            tau = max(tau_min, tau * tau_decay)

        # gradient of the smooth approximation
        g = _smooth_gradient(a, tau, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex Σa = target_sum, a ≥ 0
        a = _project_to_simplex(a, target_sum)

        # occasional exact evaluation
        if it % 1000 == 0 or it == max_iters:
            cur_score = _max_convolution(a, size_fft)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[refine] iter {it:,} new best = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
def _random_candidate(n: int, target_sum: float) -> np.ndarray:
    """Generate a random non‑negative vector and project it onto the simplex."""
    idx = np.arange(n)
    method = np.random.choice(
        ["dirichlet", "power", "exp", "blocks", "sin"], p=[0.3, 0.2, 0.2, 0.2, 0.1]
    )
    if method == "dirichlet":
        alpha = np.random.choice([0.02, 0.05, 0.1, 0.2, 0.5, 1.0])
        probs = np.random.dirichlet(np.full(n, alpha))
        a = probs * target_sum
    elif method == "power":
        p = np.random.rand() * 3.0 + 0.2
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
    elif method == "exp":
        dec = np.random.rand() * 0.015 + 0.001
        a = np.exp(-dec * idx)
        a = _project_to_simplex(a, target_sum)
    elif method == "blocks":
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        a = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            a[blk] = v
        a = _project_to_simplex(a, target_sum)
    else:  # sinusoidal shape (always non‑negative after shift)
        period = np.random.randint(20, max(21, n // 2))
        phase = np.random.rand() * 2 * np.pi
        a = np.sin(2 * np.pi * idx / period + phase) + 1.5
        a = np.maximum(a, 0.0)
        a = _project_to_simplex(a, target_sum)
    return a


# ----------------------------------------------------------------------
def _annealing_swap(
    a_start: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the permutation of the entries.
    Swaps are accepted with a Metropolis rule; occasional tiny mass transfers
    are also attempted to escape plateaus.
    """
    n = a_start.shape[0]
    best_a = a_start.copy()
    best_score = _max_convolution(best_a, size_fft)

    cur_a = best_a.copy()
    cur_score = best_score

    # initial temperature (fraction of the current best)
    temp = best_score * 0.1
    it = 0

    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)

        cand = cur_a.copy()
        cand[i], cand[j] = cand[j], cand[i]          # pure permutation change
        cand_score = _max_convolution(cand, size_fft)

        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / temp) if temp > 0 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[swap] improved to {best_score:.6f} (it {it})")

        # geometric cooling
        temp *= 0.99999

        # occasional tiny redistribution move
        if np.random.rand() < 0.001:
            i1, i2 = np.random.choice(n, 2, replace=False)
            delta_mass = (np.random.rand() - 0.5) * 0.02 * target_sum
            if cur_a[i1] + delta_mass > 0 and cur_a[i2] - delta_mass > 0:
                cand2 = cur_a.copy()
                cand2[i1] += delta_mass
                cand2[i2] -= delta_mass
                # re‑project to stay on the simplex (very cheap)
                cand2 = _project_to_simplex(cand2, target_sum)
                cand2_score = _max_convolution(cand2, size_fft)
                if cand2_score < best_score:
                    best_score = cand2_score
                    best_a = cand2.copy()
                    cur_a = cand2
                    cur_score = cand2_score
                    print(f"[perturb] improved to {best_score:.6f}")

        it += 1
        if time.time() + 0.01 >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low self‑convolution peak.
    Returns the best sequence found before the 1000 s budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # Σa = √(2 n)
    size_fft = _fft_size(2 * n - 1)          # enough for full convolution

    # ------------------------------------------------------------------
    # 1. Broad sampling
    best_a = None
    best_score = float("inf")

    def consider(candidate: np.ndarray):
        nonlocal best_a, best_score
        sc = _max_convolution(candidate, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            print(f"[sampling] new best raw score = {best_score:.6f}")

    # reference (if any)
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        consider(a_ref)

    # uniform baseline
    a_uni = np.ones(n)
    a_uni = _project_to_simplex(a_uni, target_sum)
    consider(a_uni)

    # many random draws
    for _ in range(200):
        consider(_random_candidate(n, target_sum))

    print(f"[sampling] best raw score after sampling = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Smooth‑max gradient refinement (majority of the time budget)
    remaining = deadline - time.time()
    # allocate ~70 % of remaining time to refinement
    refine_deadline = time.time() + 0.70 * remaining
    max_iters = int((refine_deadline - time.time()) * 1500)   # heuristic: ~1500 it./s
    max_iters = max(50_000, min(max_iters, 400_000))

    refined_a, refined_score = _refine_smooth(
        best_a, size_fft, target_sum, refine_deadline, max_iters
    )
    if refined_score < best_score:
        best_score = refined_score
        best_a = refined_a.copy()
        print(f"[refine] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Short random restarts (if time permits)
    while time.time() < deadline - 30.0:   # keep ~30 s for permutation search
        cand = _random_candidate(n, target_sum)
        # a very short smooth‑max run
        short_deadline = time.time() + 8.0
        short_iters = int((short_deadline - time.time()) * 1500)
        short_iters = max(5_000, short_iters)
        cand_ref, cand_score = _refine_smooth(
            cand, size_fft, target_sum, short_deadline, short_iters
        )
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand_ref.copy()
            print(f"[restart] new best after short refine = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Permutation optimisation via simulated annealing
    print("[ordering] start simulated‑annealing swaps")
    final_a, final_score = _annealing_swap(
        best_a, size_fft, target_sum, deadline
    )
    if final_score < best_score:
        best_score = final_score
        best_a = final_a.copy()

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
