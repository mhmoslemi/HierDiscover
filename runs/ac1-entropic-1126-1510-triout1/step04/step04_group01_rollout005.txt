"""
Propose a low‑score step‑function.

This implementation improves the previous search by:
  * Enforcing symmetry of the sequence (a_i = a_{n-1-i}) which is
    natural for the self‑convolution problem and reduces the search space.
  * Using a pure‐Adam optimiser on a soft‑max surrogate with an aggressive
    temperature schedule, followed by an exact ℓ∞ sub‑gradient descent.
  * Keeping the iterate on the probability simplex after every update.
  * Trying several initialisations (interpolated previous solution,
    arcsine shape, uniform and random) for a handful of lengths and
    allocating the remaining time adaptively.
The final candidate is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """
    Discrete arcsine distribution (probability vector) of length n.
    """
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT (length 2·len(a)‑1).
    """
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using a real FFT.
    """
    size = 1 << ((a.size + b.size - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _make_symmetric(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n‑1‑i}."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate:
        L_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (gradient, current_maximum_of_convolution).
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()
    # grad = 2·rev(a) * w, restricted to the n entries of a
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)   # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize_symmetric(a0: np.ndarray,
                             deadline: float,
                             init_alpha: float = 1.0,
                             alpha_factor: float = 1.02,
                             alpha_max: float = 5000.0,
                             lr0: float = 0.2,
                             beta1: float = 0.9,
                             beta2: float = 0.999,
                             eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate under symmetry
    and simplex constraints. Returns the best iterate seen and its
    peak convolution value.
    """
    a = _make_symmetric(a0.copy())
    a = _project_onto_simplex(a, s=1.0)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_and_grad(a, alpha)

        # enforce symmetry of the gradient (numerical noise)
        grad = (grad + grad[::-1]) * 0.5

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule
        lr = lr0 / np.sqrt(t)

        # update, enforce symmetry and project
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _make_symmetric(a)
        a = _project_onto_simplex(a, s=1.0)

        # keep best
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter for exploration
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(len(a)) * 1e-6
            a = _make_symmetric(a)
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak

def _subgradient_descent_symmetric(a0: np.ndarray,
                                   deadline: float,
                                   step0: float = 0.02):
    """
    Sub‑gradient descent on the exact ℓ∞ objective, keeping symmetry and
    the simplex constraint. Returns the best iterate seen.
    """
    a = _make_symmetric(a0.copy())
    a = _project_onto_simplex(a, s=1.0)

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0

    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        cur_max = b.max()

        # active set (indices attaining the max, up to a tiny tolerance)
        tol = max(1e-12, cur_max * 1e-12)
        active = np.where(b >= cur_max - tol)[0]
        w = np.zeros_like(b)
        if active.size:
            w[active] = 1.0 / active.size

        # sub‑gradient of the max (uniform over the active set)
        grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1: len(a) - 1 + len(a)]
        grad = (grad + grad[::-1]) * 0.5

        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _make_symmetric(a)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(len(a)) * 5e-7
            a = _make_symmetric(a)
            a = _project_onto_simplex(a, s=1.0)

    return best_a

def _optimise_sequence(a0: np.ndarray, total_deadline: float):
    """
    Runs the two‑phase optimisation (Adam → exact sub‑gradient) within the
    supplied time budget. Returns the refined sequence.
    """
    now = time.time()
    # give roughly 60 % of the remaining budget to the smooth surrogate

    # remaining time for exact refinement
    a_refined = _subgradient_descent_symmetric(a_opt, total_deadline)

    return a_refined

                    overall_best_seq = np.asarray(overall_best_seq, dtype=np.float64)
                    print(
                        f"*** New true best: len={n} init={kind} score={true_score:.6f}"
                    )
            except Exception as exc:
                print("evaluate_sequence raised:", exc)

            # early termination if we already beat the target
            if overall_best_score <= 1.503:
                break

        if overall_best_score <= 1.503:
            break
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # fall‑back (should never happen)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback uniform used.")

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
