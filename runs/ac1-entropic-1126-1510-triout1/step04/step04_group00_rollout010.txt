def build_conv_matrix(f):
    n = f.shape[0]
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i: fill rows i : i+n with f_i
    for i in range(n):
        A[i:i+n, i] = f[i]
    return A

"""
Propose a low‑score step‑function.

Algorithm Summary
-----------------
1. **Initialisation** – start from a few simple distributions
   (arcsine law, uniform, the previous best `height_sequence_1`).

2. **Alternating linear‑program optimisation** – for a fixed length `n`
   keep the vector `f` on the probability simplex (∑f = 1).
   Let `M = max(f*f)`.  Solve the LP  

        maximise   ∑ g_i
        subject to  (f ⊛ g)_k ≤ M   (k = 0 … 2n‑2)
                    g_i ≥ 0

   (`⊛` denotes ordinary convolution).  This linear programme finds a
   non‑negative vector `g` that can be added to `f` without increasing
   the current worst convolution value.  After solving, `g` is rescaled
   to have sum = 1 and mixed into `f` with a decreasing step size.

3. **Optional gradient polishing** – after the LP phase a few thousand
   iterations of a soft‑max (smooth max) gradient descent are applied
   to the same objective `max(f*f)`.  The gradient is obtained analytically
   via FFT‑based convolutions and a momentum term is used.

4. **Multiple lengths & restarts** – the procedure is executed for several
   lengths (≈ 500 … 3500).  The best sequence (lowest `evaluate_sequence`
   value) over all runs is returned.

The whole search respects the global time budget (default 1000 s) and
never returns an empty or invalid sequence.  All numbers are kept
non‑negative and far below the clipping bound 1000, so the evaluation
function does not alter them.

"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = v.astype(np.float64, copy=True)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)) on `n` points, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _initial_vector(init_kind: str, n: int) -> np.ndarray:
    """Create a feasible probability vector of length `n`."""
    if init_kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init_kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif init_kind == "arcsine":
        a = _arcsine_distribution(n)
    elif init_kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                # linear interpolation to required size
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _build_conv_matrix_sparse(f: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix `A` such that  (A @ g)[k] = (f ⊛ g)[k]
    for all k = 0 … 2n‑2.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # column indices: each column i repeats n times
    col_idx = np.repeat(np.arange(n, dtype=np.int64), n)

    # row indices: for column i we use rows i … i+n‑1
    row_idx = (np.arange(n, dtype=np.int64)[:, None] + np.arange(n, dtype=np.int64)).ravel()

    # data: each column i is filled with f[i]
    data = np.repeat(f, n)

    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(rows, n), dtype=np.float64)
    return A


# ----------------------------------------------------------------------
def _solve_lp_for_g(f: np.ndarray, M: float):
    """
    Solve  maximise 1·g   subject to  (f ⊛ g) ≤ M  and g ≥ 0.
    Returns the optimal `g` as a dense NumPy array, or ``None`` on failure.
    """
    n = f.shape[0]
    A = _build_conv_matrix_sparse(f)                # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # right‑hand side
    c = -np.ones(n, dtype=np.float64)                # minimise -∑g  → maximise ∑g
    bounds = [(0, None)] * n

    # Use HiGHS; it accepts sparse matrices directly.
    try:
        res = scipy.optimize.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
    except Exception:  # pragma: no cover
        return None

    if res.success:
        g = np.maximum(res.x, 0.0)   # numerical safety
        return g
    return None


# ----------------------------------------------------------------------
def _lp_alternating_optimize(a0: np.ndarray,
                             n: int,
                             slice_deadline: float):
    """
    Perform the alternating LP refinement described in the problem statement.
    Returns the best vector seen, together with its current `max(a*a)`.
    """
    f = a0.copy()
    best_f = f.copy()
    best_max = _fft_self_convolve(f).max()
    iteration = 0

    # step‑size schedule (decreasing)
    while time.time() < slice_deadline and iteration < 120:
        M = _fft_self_convolve(f).max()
        g = _solve_lp_for_g(f, M)
        if g is None or g.sum() <= 0:
            break

        # rescale to keep the sum equal to 1
        g = g / g.sum()

        # decreasing mixing factor
        t = 0.5 * (0.94 ** iteration)

        f_new = (1.0 - t) * f + t * g
        f_new = _project_onto_simplex(f_new, s=1.0)
        f = f_new

        cur_max = _fft_self_convolve(f).max()
        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_f = f.copy()
            print(
                f"[{time.time():.1f}] n={n} LP‑iter={iteration} max={best_max:.8e}"
            )

        # early stop if we have already hit the target bound
        if 2.0 * n * best_max <= 1.503:
            break

        iteration += 1

    return best_f, best_max


# ----------------------------------------------------------------------
def _gradient_polish(a0: np.ndarray,
                    deadline: float,
                    max_iter: int = 5000):
    """
    A few thousand steps of soft‑max gradient descent (the same
    technique used in the previous state‑of‑the‑art code).  The goal
    is to shave off the last bits after the LP phase.
    """
    f = a0.copy()
    n = f.size
    best_f = f.copy()
    best_max = _fft_self_convolve(f).max()

    velocity = np.zeros_like(f)
    beta = 0.9
    eta0 = 0.3
    alpha = 200.0  # temperature for the smoothed max

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = _fft_self_convolve(f)
        cur_max = conv.max()

        # soft‑max weights over the full convolution result
        exp_term = np.exp(alpha * (conv - cur_max))
        w = exp_term / exp_term.sum()

        # gradient of the weighted max (see derivation in the analysis)
        grad = 2.0 * _fft_convolve(f[::-1], w)[n - 1 : n - 1 + n]

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * grad
        eta = eta0 / np.sqrt(it + 1.0)
        f = f - eta * velocity
        f = _project_onto_simplex(f, s=1.0)

        # track best
        new_max = _fft_self_convolve(f).max()
        if new_max < best_max - 1e-15:
            best_max = new_max
            best_f = f.copy()
            print(
                f"[{time.time():.1f}] grad‑iter={it} max={best_max:.8e}"
            )

        # anneal temperature slowly
        alpha = max(alpha * 0.995, 5.0)

        it += 1

    return best_f, best_max
