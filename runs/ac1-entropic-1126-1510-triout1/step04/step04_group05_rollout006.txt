"""
Propose a non‑negative step‑function (a list of heights) that minimises
the evaluation function defined in the challenge.  The search combines
several complementary phases that fit into the 1000 s budget:

1. **Multiple random restarts of Simulated Annealing (SA)**
   – cheap mass‑transfer moves on the simplex.
   – a few short SA runs from diverse initial points (uniform,
     reference sequence, Dirichlet samples).

2. **Quasi‑Newton refinement (L‑BFGS)**
   – smooth‑max surrogate (log‑sum‑exp) of the convolution max.
   – continuation on the smoothing parameter μ.

3. **Adam‑style optimisation on the smooth‑max**
   – works directly on the soft‑max parameters y (log‑probabilities).
   – adaptive step sizes help escape shallow basins.

4. **Projected sub‑gradient polishing on the true hard max**
   – many cheap iterations to push the final score down.

All helper functions are top‑level, no closures or lambdas are used.
The algorithm respects the simplex constraint Σa = √(2·n) by explicit
projection after each update, and all numbers stay in the allowed range.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Softmax utilities ----------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / np.sum(e)


# ----------------------------------------------------------------------
# FFT based convolution helpers -----------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for all FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a (full discrete convolution) using FFT."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0          # numerical noise cleanup
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a."""
    conv = _full_convolution(a, size)
    return float(conv.max())


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient ---------------------------------
def _smooth_max_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Compute the smooth‑max value (log‑sum‑exp surrogate) and the
    gradient w.r.t. the soft‑max parameters y.
    """
    p = _softmax(y)                # probabilities, Σp = 1
    a = S * p                      # actual heights, Σa = S

    conv = _full_convolution(a, _fft_size(n))
    max_c = conv.max()

    if mu > 0.0:
        # log‑sum‑exp formulation
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # exact max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (derived in the reference solution)
    conv_wr = np.convolve(w, a[::-1])[: 2 * n - 1]
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # chain rule back to y
    inner = np.dot(g_a, p)                     # scalar ⟨p, g_a⟩
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxCallable:
    """Wrapper required by scipy.optimize.minimize (returns f,g)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _smooth_max_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
# Simulated annealing ---------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Very lightweight SA that moves a random fraction of mass from one index
    to another while staying on the simplex.
    """
    n = a_start.shape[0]
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    # temperature schedule – start relatively high, exponential decay
    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-6

    it = 0
    while time.time() - start_time < budget:
        # exponential cooling
        elapsed = time.time() - start_time
        frac = elapsed / budget
        T = max(Tmin, T0 * (1.0 - frac) ** 1.5)

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move up to 40 % of the mass from i to j
        delta = np.random.rand() * a[i] * 0.4
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        if diff < 0.0 or (T > 0.0 and np.random.rand() < math.exp(-diff / T)):
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS continuation ---------------------------------------------------
def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation on the smooth‑max surrogate.  Starts with μ equal to the
    current max‑convolution and reduces μ geometrically.
    """
    n = a_start.shape[0]
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    mu_factor = 0.6
    mu_min = 1e-6
    maxiter_per_mu = 300

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxCallable(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max -----------------------------------------
def _adam_optimize(y0: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate with a geometric cooling of μ.
    Works directly on the log‑probability vector y.
    """
    n = y0.shape[0]
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # initial µ from a uniform start
    a0 = S * _softmax(y)
    mu = float(_max_convolution(a0, _fft_size(n)))
    mu_min = 1e-6
    mu_factor = 0.85
    base_lr = 0.8

    best_y = y.copy()
    best_score = evaluate_sequence((S * _softmax(y)).tolist())

    while time.time() - start_time < budget and mu > mu_min:
        f, grad = _smooth_max_and_grad(y, mu, n, S)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y = y - base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional evaluation (once every 10 iterations)
        if t % 10 == 0:
            a = S * _softmax(y)
            sc = evaluate_sequence(a.tolist())
            if sc < best_score:
                best_score = sc
                best_y = y.copy()
                print(f"[Adam improve] μ={mu:.5e} iter={t} score={sc:.6f}")

        mu = max(mu * mu_factor, mu_min)

    return best_y, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient polishing ---------------------------------------
def _subgradient_polish(a_start: np.ndarray, S: float,
                        budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *hard* max‑convolution.
    Uses a diminishing step size and runs until time expires.
    """
    n = a_start.shape[0]
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    eta0 = S * 0.6 / n
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt:
            w[mask] = 1.0 / cnt

        # sub‑gradient w.r.t. a
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    provided evaluation function.  The routine respects the 1000 s
    time limit and returns the best found sequence as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size ------------------------------------------------------
    try:
        n = len(height_sequence_1)   # use the reference if present
        have_ref = True
    except Exception:
        n = 2000                      # sensible fallback
        have_ref = False

    S = math.sqrt(2.0 * n)           # ℓ¹‑norm that makes the score = max‑conv

    # ------------------------------------------------------------------
    # generate a few diversified starting points ------------------------
    starts = []

    # uniform start
    uniform = np.full(n, S / n, dtype=float)
    starts.append(uniform)

    # reference (if any)
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        starts.append(a_ref)

    # three Dirichlet‑based random starts (more “spiky” than uniform)
    for _ in range(3):
        rnd = np.random.rand(n)
        a_rand = rnd / rnd.sum() * S
        starts.append(a_rand)

    # ------------------------------------------------------------------
    # Phase 1 – short multi‑restart Simulated Annealing -----------------
    best_a = None
    best_score = float("inf")
    time_per_sa = max(5.0, (deadline - time.time()) * 0.25)   # cap to about 25 % of remaining
    for idx, a0 in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[SA start {idx}]   budget {time_per_sa:.1f}s")
        a_sa, sc_sa = _simulated_annealing(a0, S, time_per_sa, time.time())
        if sc_sa < best_score:
            best_score = sc_sa
            best_a = a_sa.copy()
            print(f"[SA overall best] score = {best_score:.6f}")

    if best_a is None:
        # very unlikely, but keep a fallback
        best_a = uniform.copy()
        best_score = evaluate_sequence(best_a.tolist())

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS continuation -------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print(f"[L‑BFGS] refining for {remaining:.1f}s")
        a_lbfgs, sc_lbfgs = _l_bfgs_refine(best_a, S, remaining, time.time())
        if sc_lbfgs < best_score:
            best_score, best_a = sc_lbfgs, a_lbfgs.copy()
            print(f"[L‑BFGS new global best] {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on smooth‑max --------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[Adam] optimisation for {remaining:.1f}s")
        # initialise Adam from the soft‑max parameters of the current best
        p0 = best_a / S
        eps = 1e-12
        y0 = np.log(p0 + eps).astype(np.float64)
        y_opt, sc_adam = _adam_optimize(y0, S, remaining, time.time())
        a_adam = S * _softmax(y_opt)
        sc_adam = evaluate_sequence(a_adam.tolist())
        if sc_adam < best_score:
            best_score, best_a = sc_adam, a_adam.copy()
            print(f"[Adam new global best] {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final hard‑max sub‑gradient polishing -------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        print(f"[Subgrad] polishing for {remaining:.1f}s")
        a_final, sc_final = _subgradient_polish(best_a, S, remaining, time.time())
        if sc_final < best_score:
            best_score, best_a = sc_final, a_final.copy()
            print(f"[Subgrad new global best] {best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity check (should never be needed)
    if best_score == float("inf"):
        best_a = uniform.copy()
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
