"""
propose_candidate
-----------------
Search for a non‑negative step function (list of heights) that minimizes the
self‑convolution peak as measured by ``evaluate_sequence``.

Strategy
--------
1. **Exploration** – create a diverse set of analytic families (double‑ended
   power‑law, asymmetric double‑exponential, inverse‑sqrt, random blocks …).
   Every candidate is scaled to the required ℓ¹‑norm
   ``sqrt(2·n)`` and its exact score (via FFT) is stored.
   The best ``K`` raw candidates are kept.

2. **Heavy Adam refinement** – the current best candidate is refined using a
   soft‑max approximation of the maximum convolution value.  Gradients are
   computed analytically by FFT, an Adam optimiser is used and each iterate is
   projected onto the simplex.  A relatively long budget is allocated to this
   single “promising” candidate.

3. **Sub‑gradient descent** – after the Adam phase we switch to the true
   (hard) max using the indicator of the maximal convolution entries.
   A decreasing step‑size gradient descent with simplex projection further
   reduces the objective.

4. **Hill‑climbing** – finally we perform cheap random transfers of a tiny
   amount of mass between two positions and keep any improvement.  This
   explores the nearby neighbourhood of the current best point.

All operations respect the security checks of ``evaluate_sequence``.
The function returns the best sequence found before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution via FFT (output length = len(u)+len(v)-1)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a  (w must sum to 1).
    Returns a vector of length len(a).
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective without the security‑check overhead."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best point encountered and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam states
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Base stepsize (roughly the average entry magnitude)
    eta0 = target_sum * 0.5 / n

    # Initialise temperature from the current convolution spread
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- smooth‑max weights -----
        shifted = conv - conv.max()               # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)       # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        eta = eta0 / (1.0 + it * 1e-5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- project onto simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random kick -----
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9997               # slow cooling
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the 1000 s budget
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # the normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())  # next power of two ≥ 3n‑2

    # --------------------------------------------------------------
    # Keep the K best raw candidates from the exploratory phase
    K = 12
    best_candidates: list[tuple[float, np.ndarray]] = []   # (score, array)

    def consider(arr: np.ndarray) -> None:
        """Scale arr to target_sum, evaluate, and keep top‑K."""
        arr_sum = arr.sum()
        if arr_sum == 0:
            return
        a = arr * (target_sum / arr_sum)
        sc = quick_score(a, n, size_fft)
        if len(best_candidates) < K:
            best_candidates.append((sc, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if sc < best_candidates[-1][0]:
                best_candidates[-1] = (sc, a.copy())
                best_candidates.sort(key=lambda x: x[0])

    idx = np.arange(n)

    # ---------- 1. PARAMETRIC SHADE SEARCH ----------
    # (a) Uniform
    consider(np.ones(n))

    # (b) Double‑ended power‑law  a_i = (i+1)^‑p + (n-i)^‑p
    for _ in range(200):
        p = np.random.rand() * 2.0 + 0.05          # p ∈ [0.05, 2.05]
        a = 1.0 / ((idx + 1) ** p) + 1.0 / ((n - idx) ** p)
        consider(a)

    # (c) Asymmetric double‑exponential
    for _ in range(200):
        d1 = np.random.rand() * 0.0015 + 0.0001   # left decay
        d2 = np.random.rand() * 0.0015 + 0.0001   # right decay
        a = np.exp(-d1 * idx) + np.exp(-d2 * (n - 1 - idx))
        consider(a)

    # (d) Inverse‑sqrt (single‑ended)
    for _ in range(150):
        beta = np.random.rand() * 6.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        consider(a)

    # (e) Inverse‑sqrt (double‑ended)
    for _ in range(150):
        beta = np.random.rand() * 6.0
        a = (1.0 / np.sqrt(idx + beta + 1.0) +
             1.0 / np.sqrt((n - 1 - idx) + beta + 1.0))
        consider(a)

    # (f) Power‑law left heavy  a_i = (i+1)^‑p
    for _ in range(150):
        p = np.random.rand() * 2.0 + 0.1
        a = 1.0 / ((idx + 1) ** p)
        consider(a)

    # (g) Mixed exponential / inverse‑sqrt
    for _ in range(150):
        w = np.random.rand()
        a = w * np.exp(-0.0004 * idx) + (1 - w) / np.sqrt(idx + 1.0)
        consider(a)

    # (h) Random block candidates (few constant blocks)
    for _ in range(300):
        nb = np.random.randint(2, 8)                # 2‑7 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        consider(cand)

    # (i) Reference sequence, if available
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        consider(a_ref)

    best_candidates.sort(key=lambda x: x[0])
    print(f"[exploration] best raw score = {best_candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2. HEAVY ADAM REFINEMENT ON THE CURRENT BEST CANDIDATE
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    remaining = deadline - time.time()
    if remaining > 60:                               # reserve a decent chunk for refinement
        allowed_iters = int(min(800_000, remaining * 1500))
        refined_a, refined_score = _refine_candidate_adam(
            global_best_a, deadline, n, size_fft, target_sum, allowed_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[adam] improved to {global_best_score:.6f}")

    # --------------------------------------------------------------
    # 3. SUB‑GRADIENT DESCENT USING THE HARD MAX
    def subgradient_descent(a0: np.ndarray, limit: float) -> tuple[np.ndarray, float]:
        a = a0.copy()
        best_a_loc = a.copy()
        best_sc = quick_score(a, n, size_fft)
        it = 0
        while time.time() < limit:
            conv = _self_convolution(a, size_fft)
            m = conv.max()
            mask = (conv >= m - 1e-12)                # indices attaining the max
            w = mask.astype(float)
            if w.sum() == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w.sum()
            grad = _weighted_gradient(a, w, size_fft)

            # step size decays slowly
            eta = target_sum * 0.4 / n / (1.0 + it * 0.001)
            a_new = a - eta * grad
            a_new = _project_to_simplex(a_new, target_sum)
            a = a_new

            cur_sc = quick_score(a, n, size_fft)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a_loc = a.copy()
            it += 1

            # occasional tiny random shuffle to escape plateaus
            if it % 3000 == 0:
                i, j = np.random.choice(n, 2, replace=False)
                delta = (np.random.rand() - 0.5) * target_sum * 0.001
                a_tmp = a.copy()
                a_tmp[i] = max(a_tmp[i] - delta, 0.0)
                a_tmp[j] = max(a_tmp[j] + delta, 0.0)
                a_tmp = _project_to_simplex(a_tmp, target_sum)
                a = a_tmp

        return best_a_loc, best_sc

    remaining = deadline - time.time()
    if remaining > 10:
        sub_a, sub_sc = subgradient_descent(global_best_a, deadline)
        if sub_sc < global_best_score:
            global_best_score = sub_sc
            global_best_a = sub_a.copy()
            print(f"[subgrad] improved to {global_best_score:.6f}")

    # --------------------------------------------------------------
    # 4. LIGHT HILL‑CLIMBING SEARCH
    while time.time() < deadline:
        # pick two positions and move a tiny amount of mass
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * target_sum * 0.005   # up to ±0.5 % of total mass
        cand = global_best_a.copy()
        cand[i] = max(cand[i] - delta, 0.0)
        cand[j] = max(cand[j] + delta, 0.0)
        cand = _project_to_simplex(cand, target_sum)

        sc = quick_score(cand, n, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[hill] improved to {global_best_score:.6f}")

        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
