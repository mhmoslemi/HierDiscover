"""
propose_candidate
=================

Search for a non‑negative step function (list of heights) that minimises
the self‑convolution peak evaluated by ``evaluate_sequence``.

The algorithm consists of three stages:

1. **Exploration** – generate a large variety of analytic families
   (U‑shaped inverse‑power, product‑shape, exponential, cosine, Gaussian,
   random block, etc.), project each candidate onto the simplex
   Σ a_i = √(2 n) and keep the K best raw scores.

2. **Refinement** – each of the best candidates is improved by two
   optimiser phases:
   * an ADAM optimiser that minimises a *soft‑max* approximation of the
     max‑convolution,
   * a projected sub‑gradient descent on the exact max‐convolution.
   The best point seen during the whole refinement is retained.

3. **Local random search** – a short, time‑bounded hill‑climbing phase
   that repeatedly perturbs the current best sequence with a zero‑sum
   random vector and re‑projects onto the simplex.

All operations respect the security checks of ``evaluate_sequence``.
The function returns the best sequence found before the 1000 s budget
expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear self‑convolution a*a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a
    (w must sum to 1).
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective used during the search (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _adam_refine(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    ADAM optimiser on the *soft‑max* approximation of the max‑convolution.
    Returns the best sequence observed and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam momentum buffers
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning rate (scaled with problem size)
    eta0 = target_sum * 0.6 / n

    # initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max approximation of max -----
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-12)
        exp_arg = np.clip(exp_arg, -50, 0)         # avoid overflow
        w = np.exp(exp_arg)
        s = w.sum()
        if s == 0:
            w[:] = 1.0 / w.size
        else:
            w /= s

        # ----- gradient of smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        eta = eta0 / (1.0 + it * 1e-5)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random kick -----
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9997
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _subgrad_refine(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution.
    Returns the best sequence observed and its exact score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # initial step size (chosen heuristically)
    step0 = target_sum * 0.5 / n

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        max_val = conv.max()

        # build weight vector concentrated on all (near‑)max positions
        eps = 1e-12 * max_val
        mask = conv >= max_val - eps
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[mask] = 1.0 / cnt

        grad = _weighted_gradient(a, w, size_fft)

        # decreasing step size
        step = step0 / math.sqrt(it + 1.0)

        a_new = a - step * grad
        a_new = _project_to_simplex(a_new, target_sum)

        cur_score = quick_score(a_new, n, size_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: add a zero‑sum Gaussian perturbation,
    re‑project, keep improvement.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    it = 0
    while time.time() < deadline and it < max_iters:
        # small Gaussian perturbation that keeps the sum zero in expectation
        delta = np.random.randn(n) * target_sum * 0.002
        a_new = a + delta
        a_new = _project_to_simplex(a_new, target_sum)
        cur_score = quick_score(a_new, n, size_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            a = a_new
        else:
            # occasional acceptance of non‑improving move to escape plateaus
            if np.random.rand() < 0.001:
                a = a_new
        it += 1

    return best_a, best_score
