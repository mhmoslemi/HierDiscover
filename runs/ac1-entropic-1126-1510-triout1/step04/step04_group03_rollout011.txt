"""
Propose a non‑negative step function (list of heights) that (hopefully)
improves on the previously found score ~1.5123.
The algorithm is a simple simulated‑annealing / stochastic‑local‑search
loop that works on the *simplex* (sum of coefficients = 1) and uses an
FFT‑based convolution to evaluate the objective
    E(a) = 2·n·max( a * a )               (identical to evaluate_sequence
                                            when sum(a)=1)
The search runs for the allotted time (≈1000 s) and keeps the best
sequence found.  It can start from the previously discovered construction
available as the global variable ``height_sequence_1``; if that variable is
missing a sensible arcsine distribution is used as a seed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _energy(a: np.ndarray) -> float:
    """
    Energy = 2·n·max( a * a ).  The vector a is assumed to sum to 1.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _annealing_step(a: np.ndarray, temperature: float) -> np.ndarray:
    """
    Produce a candidate by moving a random fraction of mass from one index
    to another, keeping the sum invariant.  The move is accepted later
    depending on the Metropolis criterion.
    """
    n = a.shape[0]

    # pick donor and receiver
    i, j = np.random.choice(n, size=2, replace=False)
    # amount to move: a random fraction of the donor's mass
    if a[i] == 0.0:
        return a.copy()
    frac = np.random.rand() * 0.5          # up to 50 %
    delta = a[i] * frac

    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # projection is unnecessary because the move preserves the simplex,
    # but numerical drift may occur; re‑project occasionally.
    if np.random.rand() < 0.01:
        a_new = _project_onto_simplex(a_new, s=1.0)
    return a_new


# ----------------------------------------------------------------------
def _initial_sequence(n: int) -> np.ndarray:
    """
    Return a reasonable starting point for length n.
    Preference is given to the previously discovered sequence
    (global ``height_sequence_1``) if it matches the length,
    otherwise an arcsine‑distributed probability vector is used.
    """
    # try to reuse earlier best if possible
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.size == n:
            a = prev / prev.sum()
            return _project_onto_simplex(a, s=1.0)
    except Exception:
        pass

    # fallback – discrete arcsine distribution (a classic “U‑shaped” prior)
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return _project_onto_simplex(vals, s=1.0)

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Main entry point required by the grader.  The function performs a
    stochastic‑annealing search for a step‑function that minimises
    ``evaluate_sequence``.  It returns the best list of heights found.
    """
    np.random.seed(seed)
    # leave a short safety margin for printing / cleanup
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a few lengths to explore – we stay below 30 000 for speed.
    # The previous run used around 6 500; we will also try a longer one.
    length_candidates = [5000, 8000, 12000, 20000]
    length_candidates = [L for L in length_candidates if L <= 30000]

    best_seq = None
    best_energy = float("inf")

    # ------------------------------------------------------------------
    for n in length_candidates:
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining time to this length
        remaining = deadline - time.time()
        slice_deadline = time.time() + max(0.1, remaining * 0.30)

        # initialise
        a = _initial_sequence(n)

        # initial energy (cheap, no FFT needed if uniform)
        cur_energy = _energy(a)

        # Simulated‑annealing parameters
        T_start = 1.0
        T_end = 1e-4
        decay = 0.9995                # geometric decay per iteration
        T = T_start

        it = 0
        while time.time() < slice_deadline:
            it += 1
            a_new = _annealing_step(a, T)

            new_energy = _energy(a_new)

            # Metropolis acceptance
            if new_energy < cur_energy or np.random.rand() < np.exp((cur_energy - new_energy) / max(T, 1e-12)):
                a, cur_energy = a_new, new_energy

                # keep global best

                    best_energy = cur_energy

        # early exit if we already beat the target
        if best_energy <= 1.503:
            break

    # ------------------------------------------------------------------

        best_energy = evaluate_sequence(best_seq.tolist())

    print(f"Final best score reported: {best_energy:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
