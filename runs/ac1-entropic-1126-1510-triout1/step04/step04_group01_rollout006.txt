"""
Propose a low‑score step‑function.

Algorithm
---------
1. Initialise a probability vector (sum = 1) of a chosen length.
   Several initialisations are tried (previous best, arcsine, uniform, random).

2. Iteratively improve the vector using a cutting‑plane linear‑program (LP)
   sub‑problem:
      • For the current f compute M = max(f∗f).
      • Solve   maximise   Σ g_j    subject to   f∗g ≤ M   (componentwise) , g ≥ 0.
        This LP is linear because f is fixed, only the convolution with the
        unknown g appears linearly.  Only the constraints that are (almost)
        active are kept initially; violated constraints are added iteratively.
      • The optimal g is normalised to a probability vector (sum = 1).

3. Form a convex combination f←(1−λ)f+λg for several λ∈{0.1,…,0.8} and keep the
   combination with the smallest peak of the self‑convolution.
   If no improvement is obtained a tiny Gaussian jitter is added.

4. The loop stops after a fixed number of iterations, when time runs out,
   or when the fast objective 2·n·max(f∗f) falls below the target 1.503.

5. The best sequence found (according to the cheap 2·n·max value) is finally
   evaluated with the provided ``evaluate_sequence`` function.

The LP is solved with ``scipy.optimize.linprog`` (method *highs*).  If SciPy
is unavailable the routine falls back to the previous Adam‑based optimiser.
"""

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _active_indices(conv: np.ndarray, M: float,
                    rel_tol: float = 1e-6, abs_tol: float = 1e-12) -> list:
    """Return indices where convolution is within tolerance of the maximum."""
    delta = max(abs_tol, rel_tol * M)
    idx = np.where(conv >= M - delta)[0]
    if idx.size == 0:
        idx = np.array([np.argmax(conv)], dtype=int)
    return idx.tolist()


# ----------------------------------------------------------------------
def _lp_subproblem(f: np.ndarray, M: float,
                   max_active_iter: int = 5,
                   rel_delta: float = 1e-6,
                   abs_delta: float = 1e-12) -> np.ndarray:
    """
    Solve   maximise Σ g_j   s.t.   f∗g ≤ M , g ≥ 0.
    Only constraints that are (almost) active are kept; violated constraints are
    added iteratively (cutting‑plane).
    Returns a feasible g normalised to sum = 1.
    """
    n = f.shape[0]
    if linprog is None:
        # LP solver not available – return the input unchanged
        return f.copy()

    # start with constraints that are tight for f∗f
    conv_self = _fft_self_convolve(f)
    active = _active_indices(conv_self, M, rel_delta, abs_delta)

    for _ in range(max_active_iter):
        # Build the (len(active) × n) constraint matrix A such that A·g ≤ M
        A = np.zeros((len(active), n), dtype=np.float64)
        for row, k in enumerate(active):
            j0 = max(0, k - (n - 1))
            j1 = min(k, n - 1)
            idx = np.arange(j0, j1 + 1, dtype=int)
            A[row, idx] = f[k - idx]

        # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        b_ub = np.full(A.shape[0], M, dtype=np.float64)

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception:  # pragma: no cover
            return f.copy()

        if not res.success:
            return f.copy()

        g = np.maximum(res.x, 0.0)

        # Verify all (2n‑1) convolution constraints
        conv_fg = _fft_convolve(f, g)
        viol = np.where(conv_fg > M + 1e-9)[0]
        if viol.size == 0:
            # feasible – normalise and return
            s = g.sum()
            if s < 1e-12:
                return f.copy()
            return g / s

        # add a few violated constraints to the active set
        for v in viol[:5]:
            if v not in active:
                active.append(int(v))

    # If we exit the loop without a truly feasible point, just normalise
    s = g.sum()
    if s < 1e-12:
        return f.copy()
    return g / s

                # solve the LP sub‑problem (g is feasible w.r.t. current M)
                g = _lp_subproblem(a, M)

                # try several convex combinations (λ in [0.1,0.8])
                improved = False
                best_local_f = a
                best_local_M = M
                for lam in (0.1, 0.2, 0.4, 0.6, 0.8):
                    cand = (1.0 - lam) * a + lam * g
                    cand = np.clip(cand, 0, None)
                    s_c = cand.sum()
                    if s_c <= 0:
                        continue
                    cand = cand / s_c
                    cand_M = _conv_peak(cand)
                    if cand_M < best_local_M - 1e-14:
                        best_local_f = cand
                        best_local_M = cand_M
                        improved = True

                if not improved:
                    # tiny Gaussian jitter to escape possible stagnation
                    a += np.random.randn(n) * 1e-6
                    a = np.clip(a, 0, None)
                    s_a = a.sum()
                    if s_a > 0:
                        a = a / s_a
                    M = _conv_peak(a)
                else:
                    a = best_local_f
                    M = best_local_M

                fast_score = 2.0 * n * M
                if fast_score < best_score - 1e-14:
                    best_score = fast_score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] len={n} init={kind} iter={outer+1} "
                          f"fast_score={fast_score:.6f}")

                # early termination if the target is already reached
                if best_score <= 1.503:
                    break

            # --------------------------------------------------------------
            # after the inner optimisation evaluate the current best with the
            # black‑box function (cheap compared to the overall budget)
            if best_seq is not None:
                try:
                    true_score = evaluate_sequence(best_seq.tolist())
                    if true_score < best_score:
                        best_score = true_score
                        best_seq = np.asarray(best_seq, dtype=np.float64)
                        print(f"*** New true best: len={n} init={kind} score={true_score:.6f}")
                except Exception as exc:   # pragma: no cover
                    print("evaluate_sequence raised:", exc)

            if best_score <= 1.503:
                break
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # fallback – should never happen but kept for safety
    if best_seq is None:
        n = 1000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * (1.0 / n)   # = 2.0

    # final true evaluation (if not already done)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final true score: {final_score:.6f} (len={len(best_seq)})")
    except Exception:  # pragma: no cover
        final_score = best_score
        print(f"Final fast score used: {final_score:.6f} (len={len(best_seq)})")

    # return plain Python list as required
    return best_seq.tolist()
