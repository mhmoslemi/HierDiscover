"""
Propose a low‑score step‑function.

The algorithm mixes two optimisation phases on the simplex Σ a_i = 1 :

1. **Variance descent** – minimise the variance of the self‑convolution
   (which is equivalent to a smooth surrogate of the L∞‑norm).  The
   gradient of  ∥a∗a‑μ‖² is 4·conv_rev(a, a∗a‑μ) and we use a projected
   Adam update with Euclidean projection onto the probability simplex.

2. **Max‑softmax descent** – once the variance is small we switch to a
   smoothed max (soft‑max) objective.  With weight w = softmax(α·(a∗a‑max))
   the gradient is 2·conv_rev(a, w).  The temperature α grows
   exponentially, turning the surrogate into the true max.

Multiple lengths (derived from any previously‑found sequence) and several
initialisations (previous best, arcsine law, uniform, random) are tried.
Every few thousand iterations the exact black‑box score is evaluated and
the best sequence is kept.  A final short fine‑tuning run (phase 2 only)
is performed on the overall best candidate before returning it.

All updates are projected onto the simplex, guaranteeing non‑negative
entries and a sum of exactly one, which makes the evaluation function
scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)) of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _initial_vector(init_kind: str, n: int) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if init_kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init_kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif init_kind == "arcsine":
        a = _arcsine_distribution(n)
    elif init_kind == "prev":
        # Re‑use a previously discovered construction if available.

            if prev.size != n:
                # Linear interpolation to required length.

                a = np.interp(x_new, x_old, prev)
            else:

        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    # Ensure exact feasibility.
    a = _project_onto_simplex(a, s=1.0)
    return a

    # ------------------------------------------------------------------
    # Final fine‑tuning on the overall best candidate (phase‑2 only).
    if best_seq is not None:
        a = best_seq.copy()
        n = a.size
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        lr0 = 0.02
        alpha = 5.0
        alpha_factor = 1.008
        alpha_max = 3000.0
        for it in range(1, 3001):
            b = _fft_self_convolve(a)
            cur_max = b.max()
            exp_term = np.exp(alpha * (b - cur_max))
            w = exp_term / exp_term.sum()
            grad = 2.0 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
            if alpha < alpha_max:
                alpha = min(alpha_max, alpha * alpha_factor)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            eta = lr0 / np.sqrt(it)
            step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
            a = a - step
            a = _project_onto_simplex(a, s=1.0)

            if it % 500 == 0:
                cur_score = evaluate_sequence(a.tolist())
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] fine‑tune iter={it} score={best_score:.6f}"
                    )
                if best_score <= TARGET:
                    print("Target achieved during fine‑tune.")
                    return best_seq.tolist()

    # ------------------------------------------------------------------
    # Fallback – return a uniform vector if nothing was found.
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback uniform seq score={best_score:.6f}")

    print(f"Final best score: {best_score:.6f} (length = {len(best_seq)})")
    return best_seq.tolist()
