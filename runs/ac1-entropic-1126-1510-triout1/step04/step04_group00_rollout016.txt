    beta2 = 0.999

    alpha_max = 5000.0
    best_seq = None
    best_score = np.inf
    # main loop
    while time.time() < deadline:
        t += 1
        # compute a softmax
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)
        a = exp_u / np.sum(exp_u)
        # self convolution
        b = _fft_self_convolve(a)
        cur_max = np.max(b)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}] n={n} iter={t} score={cur_score:.6f} maxB={cur_max:.8e} alpha={alpha:.2f}")
        # compute w
        # if alpha very large maybe use only max indices? We'll just do softmax always
        exp_term = np.exp(alpha * (b - cur_max))

        # gradient of smooth max:
        conv_rev = _fft_convolve(a[::-1], w)
        grad_a = 2.0 * conv_rev[n-1:n-1 + n]
        # convert to gradient on logits
        grad_u = a * (grad_a - np.dot(a, grad_a))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_u
        v = beta2 * v + (1 - beta2) * (grad_u ** 2)
        m_hat = m / (1 - beta1 ** t)

        # increase temperature

            alpha = min(alpha_max, alpha * alpha_factor)
        # occasional jitter
        if t % 2000 == 0 and np.random.rand() < 0.05:
            u += np.random.randn(n) * 0.1
    return best_seq, best_score

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that
minimises

    score = 2·n·max(a*a) / (sum(a)**2)

The implementation uses a logit (soft‑max) parameterisation
    a_i = exp(u_i) / Σ_j exp(u_j)
and performs Adam optimisation on the logits u.  The loss is a smooth
approximation of the max‑convolution obtained with a temperature‑scaled
soft‑max:

    Lα(a) = (1/α)·log∑_k exp(α·(a*a)_k) .

The temperature α is increased gradually so that the optimisation
first sees a smooth landscape and finally converges to the true
max‑objective.  Convolutions are computed with a real FFT, which makes
each iteration O(n log n).  Several sequence lengths and initialisations
(arc‑sine law, uniform, random, and a previous best construction if
available) are tried within the given time budget.  The best exact
sequence found (according to ``evaluate_sequence``) is returned.
"""

import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

    """Stable softmax."""
    shift = np.max(u)
    exp_u = np.exp(u - shift)
    return exp_u / exp_u.sum()

        grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # ---- map gradient to logit space (entropy mirror) ----------------------------
        # ∂L/∂u_i = a_i·(∂L/∂a_i − ⟨a,∂L/∂a⟩)
        grad_u = a * (grad_a - np.dot(a, grad_a))

        # ---- Adam update ---------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.10  # base learning rate
        u -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- temperature annealing ----------------------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # ---- occasional jitter to escape plateaus ------------------------------------
        if it % 2000 == 0 and np.random.rand() < 0.05:
            u += np.random.randn(n) * 0.1

        # ---- early exit when we are already good enough -------------------------------
        if best_score <= 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point used by the evaluation harness.
    Returns a plain Python list of heights (non‑negative, any length).
    """
    np.random.seed(seed)
    # leave a small margin for finalisation / printing
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Lengths to explore – based on a previously known construction if present.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    factors = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 3.0)
    lengths = sorted(
        set(max(500, int(base_n * f)) for f in factors if int(base_n * f) >= 500)
    )
    # ensure we have at least one length
    if not lengths:
        lengths = [8000]

    overall_best_seq = None          # type: np.ndarray | None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = max(0.0, deadline - time.time())
        if remaining <= 0.0:
            break

        # allocate a slice of the remaining budget to this length
        slice_deadline = time.time() + min(remaining * 0.30, 180.0)  # ≤180 s per n

        for init_kind in ("prev", "arcsine", "uniform", "random"):
            if time.time() >= slice_deadline:
                break

            # ---- initialise ---------------------------------------------------------
            a0 = _initial_vector(init_kind, n)
            u0 = _logits_from_probs(a0)

            # ---- run the optimiser --------------------------------------------------
            best_local_seq, best_local_score = _optimize_logits(u0, n, slice_deadline)

            if best_local_seq is None:
                continue

            # ---- exact evaluation (protect against clipping issues) ------------------
            try:
                final_score = evaluate_sequence(best_local_seq.tolist())
            except Exception:
                # fallback to the smooth‑approximation score
                final_score = best_local_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = best_local_seq.copy()
                print(f"*** New global best → n={n} init={init_kind} "
                      f"score={overall_best_score:.6f}")

                if overall_best_score <= 1.503:
                    print("Target reached – returning early.")
                    return overall_best_seq.tolist()

        # check overall time
        if time.time() >= deadline:
            break
        if overall_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – in the extremely unlikely case we found nothing.
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} "
          f"(length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
