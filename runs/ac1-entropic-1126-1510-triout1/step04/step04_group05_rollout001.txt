def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    try:
        n = len(height_sequence_1)
        a_ref = np.array(height_sequence_1, dtype=float)
    except NameError:
        n = 2000
        a_ref = None

    S = math.sqrt(2.0 * n)

    # If we have reference, use it as start; else uniform.
    if a_ref is not None:
        a0 = _project_to_simplex(a_ref, S)
        # compute block representation from a0: compute mass per block
    else:
        a0 = np.full(n, S/n)

    # Block partition
    m = min(200, max(20, n // 10)) # ensure at least 20 blocks
    block_len = n // m
    remainder = n % m
    block_sizes = np.array([block_len+1 if i < remainder else block_len for i in range(m)], dtype=int)

    # function to convert block masses to full sequence
    def blocks_to_seq(masses):
        # masses length m sum to S
        seq = np.empty(n, dtype=float)
        start = 0
        for i, sz in enumerate(block_sizes):
            end = start + sz
            seq[start:end] = masses[i] / sz
            start = end
        return seq

    # initial masses from a0
    masses0 = np.empty(m, dtype=float)
    start = 0
    for i, sz in enumerate(block_sizes):
        end = start + sz
        masses0[i] = a0[start:end].sum()
        start = end

    # Stage 1: (1+lambda) ES on block masses
    remaining = deadline - time.time()
    stage1_budget = remaining * 0.4 if remaining > 30 else 0
    if stage1_budget > 0:
        best_masses = masses0.copy()
        best_score = evaluate_sequence(blocks_to_seq(best_masses).tolist())
        sigma = S * 0.1  # standard deviation for Gaussian on log domain
        it = 0
        while time.time() < deadline - stage1_budget*0.1: # keep some tail
            # generate lambda offspring
            lam = 30
            parent_log = np.log(best_masses + 1e-12)
            candidates = []
            scores = []
            for _ in range(lam):
                noise = np.random.randn(m) * sigma
                cand_log = parent_log + noise
                cand = np.exp(cand_log)
                # project to simplex sum S
                cand = _project_to_simplex(cand, S)
                seq = blocks_to_seq(cand)
                sc = evaluate_sequence(seq.tolist())
                candidates.append(cand)
                scores.append(sc)
            # include parent
            scores.append(best_score)
            candidates.append(best_masses)
            # pick best
            idx = int(np.argmin(scores))
            best_masses = candidates[idx]
            best_score = scores[idx]
            if it % 20 == 0:
                print(f"[ES iter {it}] best score = {best_score:.6f}")
            # adapt sigma
            if it % 200 == 0 and it > 0:
                # reduce sigma gradually
                sigma *= 0.9
            it += 1
            if time.time() >= deadline - stage1_budget * 0.9:
                break
        # after stage1, we have best_masses
        best_a = blocks_to_seq(best_masses)
        best_score = evaluate_sequence(best_a.tolist())
        print(f"[ES] done best score = {best_score:.6f}")

    # Stage2: L-BFGS refine
    remaining = deadline - time.time()
    if remaining > 10:
        best_a, best_score = _l_bfgs_refine(best_a, S, remaining, time.time())
        print(f"[L‑BFGS] after refine best = {best_score:.6f}")

    # Stage3: final subgradient
    remaining = deadline - time.time()
    if remaining > 5:
        best_a, best_score = _final_subgradient(best_a, S, remaining, time.time())
        print(f"[final] best = {best_score:.6f}")

    return best_a.tolist()

"""
Propose a non‑negative step‑function (list of heights) that minimizes the
evaluation score.  The algorithm works in three stages within a 1000 s
budget:

1. **Coarse global search on a block‑wise simplex** – a simple (1+λ)
   Evolution Strategy (ES) optimises masses of a small number of blocks
   (≈ 200).  Each candidate is projected onto the simplex Σmass = √(2·n)
   and evaluated with a fast FFT based max‑convolution score.

2. **Smooth‑max L‑BFGS continuation** – starting from the best full‑length
   sequence found by the ES, minimise a log‑sum‑exp (smooth‑max)
   surrogate of the max‑convolution using SciPy’s L‑BFGS‑B.  The smoothing
   parameter μ is halved repeatedly (continuation) to approach the true
   max operator.

3. **Projected sub‑gradient polishing** – finally run a few thousand
   projected sub‑gradient steps on the exact (hard) max‑convolution
   objective to clean up any remaining imperfections.

All helper functions are defined at top level, no closures or lambdas
are used.  The routine respects the time budget, falls back to a uniform
sequence if necessary, and returns a list of non‑negative floats.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Softmax (used by the smooth‑max objective) -------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()

# ----------------------------------------------------------------------
# FFT helpers for fast convolution --------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for all FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())

def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0          # clamp tiny negatives caused by round‑off
    return conv

def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())

# ----------------------------------------------------------------------
# Fast score used during the ES (avoids the overhead of evaluate_sequence)
def _fast_score(seq: np.ndarray, n: int, size: int) -> float:
    max_c = _max_convolution(seq, size)
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_a ** 2)

# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS) ---------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max (log‑sum‑exp) surrogate of max_k (a*a)[k] and gradient
    w.r.t. the log‑probability vector y.
    """
    p = _softmax(y)               # Σp = 1, p_i ≥ 0
    a = S * p                     # Σa = S

    conv = np.convolve(a, a)      # length 2·n‑1, exact convolution
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                                             # exact max
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (see derivation in the reference code)
    conv_wr = np.convolve(w, a[::-1])                 # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]            # slice of length n

    # chain rule to y (since a = S·p, p = softmax(y))
    inner = np.dot(g_a, p)                            # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable required by scipy.optimize.minimize – returns (value, grad)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g

# ----------------------------------------------------------------------
# L‑BFGS continuation (smooth‑max) ------------------------------------
def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation scheme on the smooth‑max surrogate.
    Starts from a_start, repeatedly halves the smoothing parameter μ
    while running a limited number of L‑BFGS iterations at each stage.
    """
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    mu_factor = 0.5
    mu_min = 1e-6
    maxiter_per_mu = 250

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score

# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true hard‑max ------------------
def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the max‑convolution
    objective (no smoothing).  A simple decaying step‑size schedule is
    used; the iterate is projected back onto the simplex after each step.
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # step‑size parameters
    eta0 = S * 0.5 / n          # rough magnitude of a typical entry
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 8000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # sub‑gradient of the hard max
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score

# ----------------------------------------------------------------------
# Block‑wise Evolution Strategy (coarse global exploration) ----------
def _block_es_optimize(mass0: np.ndarray, block_sizes: np.ndarray,
                       total_sum: float, time_budget: float,
                       start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple (1+λ) ES on the block‑mass simplex.
    - mass0 : initial block masses (already sum to total_sum)
    - block_sizes : lengths of each block (used only for up‑sampling)
    - total_sum : target Σ masses = √(2·n)
    Returns the best block‑mass vector found and its *fast* score.
    """
    m = mass0.shape[0]
    n = block_sizes.sum()
    fft_sz = _fft_size(n)

    # Helper to up‑sample block masses to a full‑length sequence
    def blocks_to_seq(masses: np.ndarray) -> np.ndarray:
        seq = np.empty(n, dtype=float)
        pos = 0
        for i, sz in enumerate(block_sizes):
            seq[pos : pos + sz] = masses[i] / float(sz)
            pos += sz
        return seq

    # initialise
    best_masses = mass0.copy()
    best_seq = blocks_to_seq(best_masses)
    best_score = _fast_score(best_seq, n, fft_sz)

    # ES hyper‑parameters
    lam = 30                         # number of offspring per generation
    sigma = 0.5                      # standard deviation on log‑scale
    iteration = 0

    while time.time() - start_time < time_budget:
        parent_log = np.log(best_masses + 1e-12)

        # generate λ offspring
        for _ in range(lam):
            noise = np.random.randn(m) * sigma
            child_log = parent_log + noise
            child = np.exp(child_log)
            child = _project_to_simplex(child, total_sum)

            seq = blocks_to_seq(child)
            sc = _fast_score(seq, n, fft_sz)

            if sc < best_score:
                best_score = sc
                best_masses = child
                best_seq = seq
                print(f"[ES improve] iter {iteration:,} score = {best_score:.6f}")

        # slowly shrink the mutation radius
        if iteration % 200 == 0 and iteration > 0:
            sigma *= 0.9

        iteration += 1

        # safety break if we are very close to the deadline
        if time.time() - start_time > time_budget * 0.99:
            break

    return best_masses, best_score
