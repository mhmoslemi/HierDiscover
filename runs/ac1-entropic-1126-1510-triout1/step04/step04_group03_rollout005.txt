"""
propose_candidate
-----------------

Searches for a non‑negative step function (probability vector) that
minimises the black‑box ``evaluate_sequence`` cost

    cost = 2·n·max(conv(a,a)) / (sum(a))²

The sum cancels under positive scaling, therefore we keep the vector on
the probability simplex (sum = 1).  The algorithm combines several ideas
that differ from the reference implementation:

* a family of *Beta* initial shapes (including the arcsine case) is
  explored,
* projected Adam optimisation on the simplex with a soft‑max
  approximation of the “max‑convolution’’,
* explicit symmetrisation after every step (the optimum is known to be
  symmetric),
* a cheap stochastic local search that swaps mass between the largest
  and smallest coefficients,
* aggressive time‑slicing and many random restarts.

The whole search is bounded by the supplied time budget (default 1000 s)
and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, a: float, b: float) -> np.ndarray:
    """
    Symmetric Beta‑type distribution on {0,…,n‑1}.
    a,b > 0 ; the density is proportional to
        (i+0.5)^(a‑1) * (n‑i‑0.5)^(b‑1)
    """
    i = np.arange(n, dtype=np.float64) + 0.5          # shift to avoid zero
    vals = (i ** (a - 1)) * ((n + 0.5 - i) ** (b - 1))
    # guard against infinities when a or b < 1
    vals = np.where(np.isfinite(vals), vals, 0.0)
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= s
    return vals

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) w.r.t. a.
    For very large α this becomes a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # ---- soft‑max (or hard max) weights ----
    if alpha > 2000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()                 # for numerical stability
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ---- convolution w * a_rev ----
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]    # factor 2 from symmetry of convolution
    return g


# ----------------------------------------------------------------------
def _adam_optimize_one_length(init_a: np.ndarray,
                              n: int,
                              deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the simplex, with built‑in
    symmetrisation, to reduce max(self‑convolution).
    Returns the best sequence seen and the associated internal
    score 2·n·max_b (i.e. the exact value of evaluate_sequence when sum=1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                       # initial learning rate constant

    # soft‑max temperature
    alpha = 5.0
    alpha_mul = 1.07
    alpha_max = 20000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # enough for w + a_rev – 1
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    stagnation = 0
    while time.time() < deadline:
        it += 1

        # current objective
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            stagnation = 0
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.8f}")
        else:
            stagnation += 1

        # early stop if nothing improves for a while and temperature is maxed
        if stagnation > 3000 and alpha >= alpha_max:
            break

        # gradient (soft‑max or hard‑max)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)

        # enforce symmetry (optimal solutions are symmetric)
        a = (a + a[::-1]) / 2.0
        a = _project_onto_simplex(a, s=1.0)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random fraction of mass from a
    large coefficient to a small one.  The move is kept if the max‑convolution
    improves.  A few jitter steps are added for robustness.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    no_improve = 0

    while time.time() < deadline and no_improve < 1000:
        # pick a source from the top 10 % and a destination from the bottom 10 %
        top_k = max(1, n // 20)
        src_idx = np.argpartition(best_seq, -top_k)[-top_k:]
        dst_idx = np.argpartition(best_seq, top_k)[:top_k]
        i = np.random.choice(src_idx)
        j = np.random.choice(dst_idx)

        if best_seq[i] <= 1e-12:
            continue

        frac = np.random.rand() * 0.5          # up to 50 % of the source mass
        delta = best_seq[i] * frac

        candidate = best_seq.copy()
        candidate[i] -= delta
        candidate[j] += delta

        score = 2.0 * n * _fft_self_convolve(candidate).max()
        if score < best_score - 1e-12:
            best_seq = candidate
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        # occasional tiny jitter
        if np.random.rand() < 0.005:
            jitter = best_seq + np.random.randn(n) * 1e-6
            jitter = _project_onto_simplex(jitter, s=1.0)
            jitter = (jitter + jitter[::-1]) / 2.0
            jitter = _project_onto_simplex(jitter, s=1.0)
            score = 2.0 * n * _fft_self_convolve(jitter).max()
            if score < best_score:
                best_seq = jitter
                best_score = score
                no_improve = 0

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point.
    Searches for a step function (list of non‑negative heights) with a low
    ``evaluate_sequence`` score.  The algorithm respects the time budget
    (default 1000 s) and returns a plain Python list.
    """
    np.random.seed(seed)
    # leave a small safety margin for final printing / cleanup
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # base length – use previously discovered length if available
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # candidate lengths (capped at 20000 for speed)
    factors = (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)
    lengths = sorted({
        max(512, int(base_n * f))
        for f in factors
        if max(512, int(base_n * f)) <= 20000
    })
    if not lengths:
        lengths = [8192]

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate roughly 25 % of the remaining time to this length
        slice_deadline = time.time() + max(0.1, remaining * 0.25)

        attempts = 0
        while time.time() < slice_deadline and attempts < 30:
            attempts += 1

            # ------------------------------------------------------------------
            # initialise a candidate
            init_kind = np.random.choice(["arcsine", "uniform", "random", "beta"])
            if init_kind == "beta":
                a_param = np.random.uniform(0.2, 2.0)
                b_param = np.random.uniform(0.2, 2.0)
                a0 = _beta_distribution(n, a_param, b_param)
            else:
                a0 = _initial_sequence(n, init_kind)

            # ------------------------------------------------------------------
            # gradient‑based optimisation (Adam)
            seq, int_score = _adam_optimize_one_length(a0, n, slice_deadline)

            # ------------------------------------------------------------------
            # cheap local refinement if any time left
            time_left = slice_deadline - time.time()
            if time_left > 0.5:
                seq, int_score = _local_search(seq, n,
                                               time.time() + min(2.0, time_left))

            # ------------------------------------------------------------------
            # evaluate with the official black‑box
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                # fall back to the internal estimate (exact when sum==1)
                final_score = int_score

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq.copy()
                print(f"*** New global best → len={n} init={init_kind} "
                      f"score={global_best_score:.6f}")

            # early stop if target reached
            if global_best_score <= 1.503:
                break

        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # final refinement on the overall best (if time permits)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _local_search(global_best_seq,
                                       len(global_best_seq),
                                       deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
