"""
Propose a low‑score step function.

The routine builds a probability vector a (non‑negative, ℓ¹‑norm = √(2·n))
by optimising a smooth approximation of the infinity‑norm of the
autoconvolution a*a.  A soft‑max parametrisation y → a is used, and a
geometric temperature schedule (μ) drives the smooth‑max from a very
smooth bound (μ ≈ 10·max) down to the true max (μ → 0).  For each start
point we run a bounded L‑BFGS‑B optimisation, then finish with a
sub‑gradient descent and a cheap random‑perturbation phase.  Several
diverse initialisations (uniform, exponential, linear, random Dirichlet
samples) are explored; the best sequence found before the time budget
expires is returned.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z <= 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the autoconvolution and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained parameters (length n).
    mu : float
        Temperature (≥0).  Large μ gives a very smooth approximation.
    n : int
        Length of the step function.
    S : float
        Required ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : np.ndarray
        Gradient ∂f/∂y (shape (n,)).
    """
    p = _softmax(y)                # Σ p_i = 1
    a = S * p                      # Σ a_i = S

    conv = np.convolve(a, a)       # length 2·n‑1

    max_c = conv.max()
    if mu > 0.0:
        # log‑sum‑exp smoothing
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z                     # weights over antidiagonals
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # Gradient w.r.t. the heights a:
    # g_a[i] = 2 * Σ_k w_k * a_{k-i}
    conv_wr = np.convolve(w, a[::-1])            # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]          # slice of length n

    # Chain rule through the soft‑max
    inner = np.dot(g_a, p)                       # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)

    a_uniform = S * np.full(n, 1.0 / n)
    max_conv_uniform = np.convolve(a_uniform, a_uniform).max()
    mu_vals = []
    mu = max_conv_uniform * 10.0
    while mu > 1e-7:
        mu_vals.append(mu)
        mu *= 0.6
    mu_vals.append(0.0)   # final hard‑max phase

    maxiter_per_mu = 300               # L‑BFGS iterations per temperature
    n_restart = len(init_ys)

    # ------------------------------------------------------------------
    # ---- main continuation loop ----------------------------------------
    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] initialising")
        y = y0.copy()

        # initialise temperature from current max‑convolution (ensures μ is not too small)
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu_start = max(conv_cur.max(), 1.0)    # safeguard
        # replace first entry of mu_vals if needed
        mu_vals[0] = max(mu_vals[0], mu_start)

        for mu in mu_vals:
            if time.time() > deadline:
                break
            obj = _SmoothMaxObjective(mu, n, S)

            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={
                        "maxiter": maxiter_per_mu,
                        "gtol": 1e-12,
                        "disp": False,
                    },
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

        # end of temperature loop

    # ------------------------------------------------------------------
    # ---- stochastic refinement (random block perturbations) ------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        max_trials = int(min(15000, remaining * 300))
        print(f"[perturb] {max_trials} random block trials")
        for _ in range(max_trials):
            # pick a random contiguous block of indices
            block_len = rng.integers(1, min(30, n // 10) + 1)
            start = rng.integers(0, n - block_len + 1)
            idx = slice(start, start + block_len)

            cand = best_a.copy()
            # inject fresh random mass inside the block
            blk = rng.random(block_len)
            blk = _project_to_simplex(blk, S * (block_len / n))
            # replace block while keeping the rest unchanged; afterwards re‑project whole vector
            cand[idx] = blk * (S / block_len)
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # ---- simple monotone heuristics ------------------------------------
    if time.time() < deadline:
        desc = np.sort(best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(desc)

        asc = np.sort(best_a)
        asc = _project_to_simplex(asc, S)
        _maybe_update(asc)

    # ------------------------------------------------------------------
    # ---- final sub‑gradient descent on the hard max (μ = 0) -------------
    if time.time() < deadline:
        a = best_a.copy()
        it = 0
        eta0 = S * 0.7 / n
        decay = 4e-5
        while time.time() < deadline and it < 8000:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt

            conv_wr = np.convolve(w, a[::-1])
            g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

            eta = eta0 / (1.0 + decay * it)
            a_new = a - eta * g_a
            a_new = _project_to_simplex(a_new, S)

            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"[subgrad] improved to {best_score:.6f}")

            a = a_new
            it += 1

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
