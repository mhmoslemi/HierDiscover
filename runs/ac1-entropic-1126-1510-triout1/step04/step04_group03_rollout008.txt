def subgradient_descent(a, n, deadline):
    a = a.copy()
    best = a.copy()
    best_score = 2 * n * _fft_self_convolve(a).max()
    # momentum
    v = np.zeros_like(a)
    it =0
    while time.time() < deadline:
        it+=1
        # compute max convolution
        b = _fft_self_convolve(a)
        max_b = b.max()
        cur_score = 2 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score=cur_score
            best=a.copy()
            print(f'[{time.time():.2f}] n={n} it={it} score={best_score:.6f}')
        # compute active mask
        mask = (b >= max_b - 1e-12).astype(float)
        # compute gradient
        a_rev = a[::-1]
        size_fft = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
        # FFT of mask and a_rev:
        W = np.fft.rfft(mask, n=size_fft)
        A_rev = np.fft.rfft(a_rev, n=size_fft)
        conv = np.fft.irfft(W * A_rev, n=size_fft)
        start = n - 1
        grad = 2.0 * conv[start:start+n]
        # momentum update
        beta = 0.9
        v = beta * v + (1-beta) * grad
        # step size
        eta = 0.5 / np.sqrt(it)   # hyperparameter
        a = a - eta * v
        # project onto simplex
        a = _project_onto_simplex(a, s=1.0)
        # random jitter occasionally:
        if it % 2000 == 0:
            # random small perturbation to escape plateau
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)
    return best, best_score

def local_search(a, n, deadline):
    best = a.copy()
    b = _fft_self_convolve(best)
    best_score = 2 * n * b.max()
    it =0
    while time.time() < deadline:
        it+=1
        i, j = np.random.choice(n, 2, replace=False)
        if best[i] < 1e-12:
            continue
        frac = np.random.rand() * 0.5 # up to 50% move
        delta = best[i] * frac
        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # projection not needed as sum remains 1, non-negative satisfied
        # evaluate
        b = _fft_self_convolve(candidate)
        cand_score = 2 * n * b.max()
        if cand_score < best_score:
            best_score = cand_score
            best = candidate
        # jitter occasionally
        if it % 5000 == 0:
            cand = best + np.random.randn(n) * 1e-6
            cand = _project_onto_simplex(cand, s=1.0)
            b = _fft_self_convolve(cand)
            cand_score = 2 * n * b.max()

    – compute the self‑convolution via FFT,
    – build a mask of the active shifts (those attaining the maximum),
    – the sub‑gradient of the objective w.r.t. a is 2 · (mask * a_rev),
      which is obtained again by FFT.
    – Nesterov‑style momentum and a diminishing step‑size η_t = η0/√t are
      used.
    – after each step we project back onto the probability simplex.
* Periodically add a tiny random jitter to escape plateaus.
* After the gradient phase, a cheap **pairwise mass‑swap local search** is
  performed for the remaining budget: a random fraction of mass is moved
  from one index to another and the move is kept if it reduces the
  objective.
* The best sequence seen (global across restarts) is returned.  The final
  score is printed for reference.

The whole routine respects the 1000‑second budget and never returns an
empty/invalid list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    # fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine probability vector of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a * a)_k  w.r.t. a.
    `b` must be the full convolution a*a (length 2n‑1).
    """
    n = a.shape[0]
    max_b = b.max()
    # binary mask of active shifts (where the max is attained)
    mask = (np.abs(b - max_b) < 1e-12).astype(np.float64)

    # convolution mask * a_rev -> gradient (factor 2 already applied)
    a_rev = a[::-1]
    size_fft = 1 << ((len(mask) + len(a_rev) - 1).bit_length())
    W = np.fft.rfft(mask, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    grad = 2.0 * conv[start:start + n]   # 2·∑_k mask_k a_{k-i}
    return grad


# ----------------------------------------------------------------------
def _gradient_descent(a0: np.ndarray,
                     n: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent (with momentum) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # hyper‑parameters
    eta0 = 0.8               # initial step‑size constant
    beta = 0.9               # momentum factor
    vel = np.zeros_like(a)   # momentum accumulator
    it = 0

    while time.time() < deadline:
        it += 1

        # evaluate current objective
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] len={n} it={it} score={best_score:.6f}")

        # sub‑gradient of the max‑convolution term
        grad = _subgradient(a, b)

        # Nesterov‑style momentum update
        vel = beta * vel + (1.0 - beta) * grad

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # gradient step + simplex projection
        a = a - eta * vel
        a = _project_onto_simplex(a, s=1.0)

        # occasional tiny jitter to help escape flat plateaux
        if it % 2000 == 0 and np.random.rand() < 0.3:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_search(a0: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic pairwise mass‑swap local search.
    Keeps Σa_i = 1 and a_i ≥ 0 by construction.
    """
    best = a0.copy()
    best_score = 2.0 * n * _fft_self_convolve(best).max()
    it = 0

    while time.time() < deadline:
        it += 1
        # pick two distinct positions
        i, j = np.random.choice(n, size=2, replace=False)
        if best[i] <= 1e-12:
            continue

        # move a random fraction (up to 30 %) of mass from i to j
        frac = np.random.rand() * 0.30
        delta = best[i] * frac
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate candidate
        score = 2.0 * n * _fft_self_convolve(cand).max()
        if score < best_score:
            best = cand
            best_score = score

        # tiny jitter every few thousand steps
        if it % 5000 == 0:
            cand = best + np.random.randn(n) * 1e-6
            cand = _project_onto_simplex(cand, s=1.0)
            score = 2.0 * n * _fft_self_convolve(cand).max()
            if score < best_score:
                best = cand
                best_score = score

    return best, best_score
