def _grad_descent(a0, size_fft, target_sum, p_values, max_iters_per_phase, deadline):
    a = a0.copy()
    best_a = a.copy()
    # compute best score initial
    conv = _full_conv(a, size_fft) # returns full conv
    best_score = 2 * len(a) * conv.max() / (target_sum**2)
    # maintain best across all phases
    step_counter = 0
    t = 0 # iteration index
    # Adam states
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999

    # base eta
    eta0 = target_sum * 0.5 / a.size

    for p in p_values:
        # define number of steps for this p
        max_iters = max_iters_per_phase
        for it in range(max_iters):

            t += 1

            max_scaled = scaled.max()

            exp_scaled = np.exp(scaled - max_scaled)
            w = exp_scaled / exp_scaled.sum()   # shape m
            # Compute gradient via correlation
            W_fft = np.fft.rfft(w, n=size_fft)
            corr = np.fft.irfft(np.conj(W_fft) * A_fft, n=size_fft)  # length size_fft

            idx = (a.size - 1) - np.arange(a.size)   # n-1 ... 0
            grad = 2.0 * p * corr[idx]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)

            a = np.maximum(a, 0.0)
            total = a.sum()
            if total > 0:
                a *= target_sum / total
            else:
                a[:] = target_sum / a.size

            # evaluate

"""
propose_candidate
-----------------

Search for a non‑negative step function (a list of heights) that minimizes the
evaluation score defined by ``evaluate_sequence``.

Algorithm
~~~~~~~~~
1. **Problem size** – infer the length *n* from the provided reference
   ``height_sequence_1`` (fallback to 3000) and fix the ℓ₁‑norm to
   ``S = sqrt(2·n)``.  The scaling does not affect the score but keeps the
   values comfortably below the 1000‑clip.

2. **Broad sampling** – generate a small pool of diverse candidates
   (reference, uniform, Dirichlet draws, power‑law and exponential shapes).
   Keep the best few (by the true peak of the self‑convolution).

3. **Smooth max‑convolution descent** – for each seed run an Adam‑style
   gradient descent on a *smooth* surrogate of the ℓ∞‑norm:
   ```
       L_p(a) = log Σ_k exp(p·(a*a)_k)          (p grows during the run)
   ```
   The gradient of ``L_p`` is a weighted correlation between the current
   self‑convolution and the sequence itself and can be obtained with two
   FFTs per iteration.  After each update we project back onto the simplex
   (non‑negative, sum = S).  A schedule of increasing ``p`` yields a
   progressively tighter approximation to the true max.

4. **Permutation hill‑climbing** – with the best shape found, repeatedly
   swap two positions; keep a swap that strictly reduces the peak
   convolution.  Occasionally a tiny random perturbation (mass transfer)
   is tried to escape plateaus.

5. Return the best sequence before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Smallest power‑of‑2 ≥ required (good for FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length = 2 n‑1)."""
    A = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(A * A, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _scale_to_sum(a: np.ndarray, target: float) -> np.ndarray:
    """Project onto {x≥0, Σx = target} by simple clipping + scaling."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a *= target / s
    else:                         # degenerate case – fall back to uniform
        a[:] = target / a.size
    return a


# ----------------------------------------------------------------------
def _smooth_max_descent(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    p_schedule: list[int],
    iters_per_p: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style descent on the smooth max‑convolution surrogate.
    Returns the best sequence (lowest true peak) seen.
    """
    n = a0.size
    a = a0.copy()
    best_a = a.copy()
    # initial score
    conv = _full_conv(a, size_fft)
    best_score = 2.0 * n * conv.max() / (target_sum ** 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    # base learning rate (order of magnitude of a single entry)
    eta0 = target_sum * 0.5 / n
    t = 0

    for p in p_schedule:
        for _ in range(iters_per_p):
            if time.time() >= deadline:
                return best_a, best_score

            t += 1

            # ---- compute self‑convolution ----
            A_fft = np.fft.rfft(a, n=size_fft)
            conv_full = np.fft.irfft(A_fft * A_fft, n=size_fft)[: 2 * n - 1]

            # ---- softmax weights for the smooth max ----
            scaled = p * conv_full
            max_scaled = scaled.max()
            # stable softmax
            exp_scaled = np.exp(scaled - max_scaled)
            w = exp_scaled / exp_scaled.sum()          # shape (2n‑1)

            # ---- gradient of log‑sum‑exp ----
            # correlation w * a  (we need Σ_i w_i a_{i‑j})
            # can be obtained via convolution with the reversed a
            a_rev = a[::-1]
            W_fft = np.fft.rfft(w, n=size_fft)
            A_rev_fft = np.fft.rfft(a_rev, n=size_fft)
            corr = np.fft.irfft(W_fft * A_rev_fft, n=size_fft)

            # slice indices n‑1 … n‑1+n‑1  (length n)
            grad = 2.0 * p * corr[n - 1 : n - 1 + n]

            # ---- Adam update ----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            eta = eta0 / math.sqrt(t)
            a -= eta * m_hat / (np.sqrt(v_hat) + eps)

            # ---- projection onto the simplex ----
            a = _scale_to_sum(a, target_sum)

            # ---- keep the best (true) score ----
            conv = _full_conv(a, size_fft)
            cur_score = 2.0 * n * conv.max() / (target_sum ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                if t % 5000 == 0:
                    print(f"[descent] iter {t:,}, p={p}, new best {best_score:.6f}")

    return best_a, best_score

    return global_best_a.tolist()
