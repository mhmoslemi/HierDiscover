def refine_candidate(a0, deadline, n, size_fft, target_sum):

    a = a0.copy()
    best = a.copy()
    best_score = quick_score(a, n, size_fft)
    # optimization state
    m = np.zeros_like(a)
    beta = 0.9
    epsilon = 1e-8
    it = 0
    T = None
    while time.time() < deadline:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (target_sum**2)   # but target_sum = sqrt(2n) => cur_score = conv.max().
        # update best
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # Determine gradient:
        max_val = conv.max()
        # compute indices close to max within tolerance maybe 1e-12
        idxs = np.where(conv >= max_val - 1e-12)[0]
        if len(idxs) > 0 and (T is None or T < 1e-5):
            # exact subgradient
            w = np.zeros_like(conv)
            w[idxs] = 1.0 / len(idxs)
            grad = _weighted_gradient(a, w, size_fft)
        else:
            # softmax gradient
            if T is None:
                # init T from range
                T = max(1e-4, (conv.max() - conv.min()) * 0.2)
            shifted = conv - max_val
            exp_arg = shifted / max(T, 1e-8)
            exp_arg = np.clip(exp_arg, -50, 0)
            w = np.exp(exp_arg)
            w /= w.sum()
            grad = _weighted_gradient(a, w, size_fft)
            # cooling
            T *= 0.9995
        # update with momentum
        m = beta * m + (1 - beta) * grad
        # step size schedule
        eta = target_sum * 0.2 / (np.sqrt(it+1) * n)   # adjust
        a_new = a - eta * m
        # projection onto simplex
        a_new = _project_to_simplex(a_new, target_sum)
        # random kick occasionally
        if it % 4000 == 0 and it > 0:
            a_new += np.random.randn(n) * target_sum * 0.0005
            a_new = _project_to_simplex(a_new, target_sum)
        a = a_new
        it += 1
        # break if many iterations approx 200k
        if it >= 200000:
            break
    return best, best_score

"""
Simple but aggressive search for a step‑function that minimises the
self‑convolution peak used by ``evaluate_sequence``.

Strategy
--------
1.  Choose a relatively large problem size (n≈5400) – the FFT size stays
    at 16384, so the cost per iteration is still modest.
2.  Generate a very diverse set of seed vectors:
    * uniform, Dirichlet draws,
    * power‑law / inverse‑sqrt families (single‑ and double‑ended),
    * exponential decays (single & double),
    * cosine / sine shapes with random exponents,
    * centred Gaussian bumps,
    * triangular / beta‑shaped forms,
    * random block‑constant vectors.
    Every seed is projected onto the simplex
    {a≥0, Σa = √(2 n)} and its exact score (no security checks) is stored.
3.  Keep the K best raw seeds (K≈20) and improve each of them with a
    projected Adam optimiser that minimises a smooth approximation of the
    max‑convolution.  The temperature T is slowly cooled; once T falls
    below 1e‑6 the algorithm switches to the exact sub‑gradient of the
    max (weights are 1 on the active indices).  Random “kicks’’ are added
    every few thousand steps to escape plateaus.
4.  After the refinement stage, perform a fast final sweep:
    * jitter the current global best with tiny Gaussian noise,
    * try a few more random‑block candidates.
    The best vector seen is returned.
5.  The whole routine respects the 1000 s time budget – if the budget is
    exhausted the best vector found so far is output (the scorer will
    return ∞ only for invalid inputs, which we never produce).

All helper functions are defined at top level, no closures or file I/O are
used.  The code works with the already‑imported ``evaluate_sequence``.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    (w must sum to 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = len(w)+len(a)-1 = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact score used during the search (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑based minimisation of max(a*a) with a cooling temperature.
    For large T we use a soft‑max weight; once T is tiny we switch to
    the exact sub‑gradient of the max (uniform weight over the active
    indices).  The iterate is projected onto the simplex after every step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning rate (scale‑free)
    eta0 = target_sum * 0.5 / n

    # initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (target_sum ** 2)   # = conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        max_val = conv.max()

        # ---------- choose weighting ----------
        if T > 1e-6:                     # smooth approximation
            shifted = conv - max_val                # ≤ 0
            exp_arg = shifted / T
            exp_arg = np.clip(exp_arg, -50, 0)       # avoid overflow
            w = np.exp(exp_arg)
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:                           # exact sub‑gradient of the max
            idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]
            w = np.zeros_like(conv)
            w[idxs] = 1.0 / len(idxs)

        # ---------- gradient ----------
        grad = _weighted_gradient(a, w, size_fft)

        # ---------- Adam update ----------
        eta = eta0 / np.sqrt(it + 1)                     # sqrt‑decay
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection ----------
        a_new = _project_to_simplex(a_new, target_sum)

        # ---------- occasional random kick ----------
        if it % 4000 == 0 and it > 0:
            a_new += np.random.randn(n) * target_sum * 0.001
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9995                      # gentle cooling
        it += 1

    return best_a, best_score

    # -------------------- 1. PARAMETRIC SHAPE SEARCH --------------------
    # (a) uniform

        add_candidate(a)

    # (f) exponential decay – double (asymmetric)

        decay_r = np.random.rand() * 0.025 + 0.0005
        a = np.exp(-decay_l * idx) + np.exp(-decay_r * (n - 1 - idx))
        add_candidate(a)

    # (g) cosine‑based (large at ends, small in the middle)
    for _ in range(200):
        p = np.random.rand() * 8.0 + 0.2
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        add_candidate(a)

    # (h) sine‑based (zero at ends, peak in the middle)
    for _ in range(200):
        p = np.random.rand() * 8.0 + 0.2
