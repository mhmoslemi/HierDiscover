def iterative_lp_refine(a, S, deadline):
    best_a = a.copy()
    best_score = max_convolution(a) (peak)
    while time remaining > something:
        conv = full_conv(a)
        M = conv.max()
        # identify active constraints
        active_idx = np.where(conv >= M * 0.99)[0]  # maybe threshold 0.99
        # limit number of constraints to maybe max_active = 30
        if len(active_idx) > max_active:
            # take top entries (largest conv)
            order = np.argsort(-conv)[:max_active]
            active_idx = order
        # build A matrix size (len(active_idx), n)
        K = len(active_idx)
        A = np.zeros((K, n))
        for row_idx, k in enumerate(active_idx):
            start = max(0, k - (n-1))
            end = min(k, n-1)
            j_range = np.arange(start, end+1)
            A[row_idx, j_range] = a[k - j_range]
        # Solve LP: maximize sum(g) subject to A g <= M, g>=0.
        # SciPy's linprog by default minimize; set c = -1.
        res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(K, M), bounds=[(0, None)]*n, method='highs', options={'disp': False})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g == 0:
            break
        # rescale g to have sum S
        g = g * (S / sum_g)
        # update a via convex combination
        t = 0.4  # step size
        a = (1 - t) * a + t * g
        # compute new score
        new_score = max_convolution(a)
        if new_score < best_score - 1e-6:
            best_score = new_score
            best_a = a.copy()
            print(f"[lp] new best score = {best_score:.6f}")
        # if improvement small break
        if new_score >= best_score * (1 - 1e-4):
            # maybe break if not improving
            pass
        # else continue
    return best_a, best_score

def refine_via_lp(a: np.ndarray, target_sum: float, deadline: float) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)
    # Setup parameters
    # active set threshold
    act_thresh = 0.99  # fraction of max to consider active
    max_active = 30
    t = 0.4
    while time.time() < deadline:
        conv = _full_conv(best_a, size_fft)   # maybe need conv for current a?
        M = conv.max()
        # Determine active indices
        active = np.where(conv >= M * act_thresh)[0]
        if active.size > max_active:
            # pick top
            idx_sorted = np.argsort(-conv)[:max_active]
            active = idx_sorted
        K = active.size
        if K == 0:
            break
        # Build A_ub
        A = np.zeros((K, n), dtype=np.float64)
        for row_i, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            # j indices from start to end inclusive
            j = np.arange(start, end + 1, dtype=int)
            A[row_i, j] = best_a[k - j]
        # Solve LP: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        # Using high accuracy but fast method
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(K, M), bounds=bounds,
                                    method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            # fallback: break
            break
        g = res.x
        # Scale to target sum
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g * (target_sum / sum_g)
        # Update a
        a_new = (1 - t) * best_a + t * g
        # compute new score
        new_score = _max_convolution(a_new, size_fft)
        if new_score < best_score - 1e-7:
            best_score = new_score
            best_a = a_new
            print(f"[lp] improved to {best_score:.6f}")
        else:
            # Not improving, maybe reduce t
            t = max(t * 0.5, 0.01)
        # check if improvement on M is small
        if M - new_score < 1e-8:
            break
    return best_a, best_score

def propose_candidate(seed=42, budget_s=1000):

    # ---------- Swap hill climbing ----------

    a = global_best_a.copy()

    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)

        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]

            a = a_swapped

                    print(f"[perturb] improved to {best_sc:.6f}")

        # break near deadline
        if time.time() + 0.01 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()

"""
Propose a non‑negative step‑function (list of heights) that minimises the
evaluation function defined in the challenge.

Algorithm
---------
1. Determine the problem size n (use the provided reference sequence if it
   exists, otherwise n≈3000).  The ℓ₁‑norm is fixed to S = √(2 n), which makes the
   evaluation equal to the peak of the self‑convolution.

2. **Broad sampling** – generate a diverse pool of raw candidates using
   Dirichlet draws, power‑law / exponential shapes, block‑wise constant
   patterns, sine / parabola shapes and random sparse “spike” patterns.
   Keep the K best (lowest convolution peak).

3. **Linear‑programming refinement** – for the top few raw candidates,
   iteratively improve the peak by solving a small LP:
      maximise  Σ g  subject to (a * g)[k] ≤ M  for the currently most
      active convolution indices k,   g ≥ 0.
   The solution g is rescaled to the target ℓ₁‑norm and mixed with a.
   This reproduces the “active‑set LP” idea from the reference paper and
   quickly drives the peak down.

4. **Projected Adam sub‑gradient descent** – run a long Adam‑style descent
   on the hard L∞‑objective (gradient only at the worst index).  After each
   step we project back onto the simplex {x ≥ 0, Σ x = S}.

5. **Permutation hill‑climbing** – with the remaining time repeatedly swap
   two positions (cost‑free for the ℓ₁‑norm) and keep the swap if it lowers
   the peak.  Occasionally a small random perturbation of the mass is
   performed to escape plateaus.

All stages respect the global 1000 s budget; the best sequence found is
returned as a plain Python list.
"""

import math
import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for FFT efficiency)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear self‑convolution a * a (full length = 2 n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of maxₖ (a * a)ₖ at the worst index k_max."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if k_max < n:
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        start = k_max - n + 1
        length = 2 * n - 1 - k_max
        g[start:] = 2.0 * a[:length][::-1]
    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam‑style projected sub‑gradient descent on the hard max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    eta0 = target_sum * 0.5 / a.size  # base step size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[subgrad] it {it:,} new best = {best_score:.6f}")

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # Simplex projection
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _lp_refine(a0: np.ndarray, target_sum: float,
               size_fft: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Small‑LP active‑set refinement.

    Repeatedly:
        * compute the current convolution and its maximum M
        * select a handful of indices where the convolution is ≥ 0.99 M
        * solve   max Σ g   s.t. (a * g)[k] ≤ M for the selected k, g ≥ 0
        * rescale g to have ℓ₁‑norm = target_sum
        * mix a ← (1‑t)·a + t·g   (t≈0.4)

    Stops when the deadline is reached or no progress is made.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_score = _max_convolution(a, size_fft)

    # LP parameters
    t = 0.4
    act_thresh = 0.99          # which entries belong to the active set
    max_active = 30            # limit on number of constraints per LP

    while time.time() < deadline:
        conv = _full_conv(a, size_fft)
        M = conv.max()
        # active indices (worst spots)
        active = np.where(conv >= M * act_thresh)[0]
        if active.size > max_active:
            # keep the top‑max entries only
            active = np.argsort(-conv)[:max_active]

        K = active.size
        if K == 0:
            break

        # Build the K×n constraint matrix A g ≤ M
        A = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            j = np.arange(start, end + 1, dtype=int)
            A[row, j] = a[k - j]

        # Linear program: maximise Σg  ⇔ minimise -Σg
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        # Solve with HiGHS (fast for moderate size)
        res = optimize.linprog(
            c,
            A_ub=A,
            b_ub=np.full(K, M, dtype=np.float64),
            bounds=bounds,
            method="highs",
        )
        if not res.success:
            # abort LP refinement if the solver fails
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # Rescale to the target ℓ₁‑norm
        g = g * (target_sum / sum_g)

        # Convex combination
        a_new = (1 - t) * a + t * g
        new_score = _max_convolution(a_new, size_fft)

        if new_score < best_score - 1e-7:
            best_score = new_score
            a = a_new
            print(f"[LP] improved score = {best_score:.6f}")
        else:
            # If we stall, shrink the mixing factor
            t = max(t * 0.5, 0.01)
            if t < 1e-4:
                break

    return a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a low‑peak self‑convolution step‑function."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------

    try:

    # uniform baseline
    a_uni = np.ones(n, dtype=float)
    a_uni = _project_to_simplex(a_uni, target_sum)
    push_candidate(quick_score(a_uni), a_uni)
    print(f"[init] uniform score = {quick_score(a_uni):.6f}")

    # Dirichlet draws (various concentrations)
    for alpha in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:

    for _ in range(60):

        a = _project_to_simplex(a, target_sum)

    # Power‑law shapes
    for _ in range(40):

        push_candidate(quick_score(a), a)

    # Exponential decays
    for _ in range(40):

        a = _project_to_simplex(a, target_sum)

    # Sine‑type shapes (non‑negative half‑sinusoid)
    for _ in range(15):

        a = _project_to_simplex(a, target_sum)
        push_candidate(quick_score(a), a)

    # Parabolic (bell) shape
    for _ in range(15):

        if time.time() + 0.01 > deadline:

    print(f"[final] best score = {global_best_score:.6f}")
