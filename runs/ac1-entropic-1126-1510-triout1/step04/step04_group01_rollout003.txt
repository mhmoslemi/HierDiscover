def _triangular_distribution(n):
    mid = n//2
    vals = np.arange(1, mid+2)
    if n%2==0:
        left = vals
        right = vals[::-1]
    else:
        left = np.arange(1, mid+2)
        right = np.arange(1, mid+1)[::-1]
    arr = np.concatenate([left, right])[:n]
    arr = arr.astype(float)
    arr /= arr.sum()
    return arr

"""
Propose a low‑score step‑function.

Algorithm
---------
1.  Generate a set of candidate probability vectors for each chosen length.
    Candidates include:
        - arcsine distribution (analytic optimal‑shape guess)
        - uniform distribution
        - symmetric beta‑type distributions with various shape parameters
        - a simple triangular distribution
        - a few random draws
        - an interpolation of the best sequence found so far (progressive
          refinement across lengths).

2.  Score every candidate with a fast exact evaluation
        score = 2·n·max(a*a)   (the sequence is normalised to sum = 1).

3.  Pick the best few candidates and improve them by
    * Adam optimisation of a smooth‑max surrogate of the ℓ∞‑norm of the
      convolution (soft‑max temperature annealed to a large value); all
      iterates are projected onto the simplex.
    * A short sub‑gradient descent directly on the true ℓ∞‑objective.

4.  Evaluate the refined candidate with the provided `evaluate_sequence`
    (identical to the fast score because of the normalisation) and keep the
    best overall sequence.

5.  The procedure runs for the allotted wall‑clock time (default 1000 s),
    allocating the remaining budget roughly equally over a handful of
    lengths (starting from the previously known construction and then
    increasing the length).  Early stopping occurs once the target
    score ≤ 1.503 is reached.

All helper functions are defined at module level; no closures or lambdas are
used.  The code works with only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) distribution discretised on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = (i + 0.5) ** (p - 1.0) * (n - i - 0.5) ** (p - 1.0)
    # Guard against under/overflow for extreme p
    vals = np.where(np.isfinite(vals), vals, 0.0)
    s = vals.sum()
    if s == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= s
    return vals


# ----------------------------------------------------------------------
def _triangular_distribution(n: int) -> np.ndarray:
    """Simple symmetric triangular distribution on n points."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=np.float64)
        right = left[:-1][::-1]
    vals = np.concatenate([left, right])
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)                     # full convolution a*a
    cur_max = b.max()
    # soft‑max weights (stable)
    shifted = b - cur_max
    exp_term = np.exp(alpha * shifted)
    w_sum = exp_term.sum()
    w = exp_term / w_sum                           # weight vector, length 2n‑1
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)             # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]       # central part
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 1.0,
                   alpha_factor: float = 1.008,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence encountered and its (exact) peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate schedule
        lr = lr0 / math.sqrt(t)

        # update & project onto simplex
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep best exact peak seen so far
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny noise for escaping flat regions
        if t % 500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 8e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the true ℓ∞ objective.
    Uniform weight on all indices attaining the current maximum.
    Returns (grad, cur_max).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # active set tolerance
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Short refinement using exact ℓ∞ sub‑gradient descent.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.04 / math.sqrt(it)          # slowly decreasing step size
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.09:
            a += np.random.randn(n) * 6e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """
    Exact objective for a normalised probability vector (sum=1):
    score = 2·n·max(a*a).
    """
    n = a.shape[0]
    max_b = _fft_self_convolve(a).max()
    return 2.0 * n * max_b

        # ------ generate diverse candidates for this length -----------------
        candidates = []
