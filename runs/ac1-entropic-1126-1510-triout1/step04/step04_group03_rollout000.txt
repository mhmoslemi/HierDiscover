"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1.  Determine a baseline length (the length of the previously found
    sequence, otherwise a default ~6000) and generate a set of target
    lengths up to 30 000.
2.  For each length ``n`` we create a diverse collection of initial
    probability vectors:
        * the previous best up‑sampled to the new length,
        * arcsine‑like distributions with a tunable offset ``δ``,
        * mixtures of an arcsine shape and the uniform distribution,
        * a few random / uniform candidates.
3.  Each candidate is refined by an Adam‑style projected optimisation
    on the probability simplex.  The loss is the soft‑max approximation
    of the convolution maximum; the temperature (``α``) is increased
    slowly so the optimiser first sees a smooth landscape and later
    concentrates on the true max‑operator.
4.  After the gradient step we run a cheap stochastic local‑search
    that moves a small fraction of mass between two random indices,
    accepting only moves that improve the score.
5.  The best sequence found for the current length is evaluated with the
    official ``evaluate_sequence`` function.  The global best across all
    lengths is kept.
6.  When the global time budget (≈1000 s) is almost exhausted a final
    refinement (local search) is performed on the overall best vector.

The implementation uses only NumPy and respects the required
API – the grader will call ``propose_candidate`` and expects a plain
list of floats as the return value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ_k exp(α·b_k) w.r.t. a.
    For large α this is a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # ---- soft‑max weights (or uniform over maximal entries) ----
    if alpha > 2000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ---- convolution w * a_rev (a_rev = a reversed) ----
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the simplex to minimise the
    max‑convolution.  The function returns the best vector found and the
    internal score 2·n·max(b) (i.e. the exact evaluate_sequence value when
    the sum of coefficients equals 1).
    """
    a = init_a.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # soft‑max temperature schedule
    alpha = 5.0
    alpha_mul = 1.04
    alpha_max = 15000.0

    # FFT size sufficient for w (2n‑1) and a_rev (n) convolution
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1
        # current convolution and score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} cur_score={best_score:.6f}")

        # gradient of the smoothed max
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # diminishing step size
        eta = 0.6 / np.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search that repeatedly transfers a random
    fraction of mass from one index to another and keeps the move if it
    improves the max‑convolution.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] < 1e-12:
            continue
        # move up to 30 % of the mass from i to j
        frac = np.random.rand() * 0.30
        delta = best_seq[i] * frac
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        score = 2.0 * n * _fft_self_convolve(cand).max()
        if score < best_score:
            best_score = score
            best_seq = cand

        # occasional jitter to escape flat regions
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-6
            cand = best_seq + jitter
            cand = _project_onto_simplex(cand, s=1.0)
            score = 2.0 * n * _fft_self_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_seq = cand
    return best_seq, best_score


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int, delta: float = 0.5) -> np.ndarray:
    """
    A discrete arcsine‑like distribution with an adjustable offset ``delta``.
    ``delta`` controls how strongly the density diverges near the edges.
    """
    i = np.arange(n, dtype=np.float64)
    den = np.sqrt((i + delta) * (n - 1 - i + delta))
    vals = 1.0 / den
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float = 0.5) -> np.ndarray:
    """
    Discrete symmetric Beta(α,α) distribution on n points.
    """
    # centre points at half‑integers to avoid zero at the edges when α<1
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _mix(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a + (1‑w)·b, renormalised to sum 1."""
    mixed = w * a + (1.0 - w) * b
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
def _resize_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    resized = np.interp(new_x, old_x, a)
    resized /= resized.sum()
    return resized

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq_opt.copy()
                print(f"*** New global best → len={n} score={global_best_score:.6f}")

            # early exit if we already beat the target
            if global_best_score <= 1.503:
                break

        if global_best_score <= 1.503:
            break

    # ----------------------------------------------------------------------
    # final refinement on the overall best (if any time is left)
    if global_best_seq is not None and time.time() < deadline:
        refined, _ = _local_search(global_best_seq,
                                   len(global_best_seq),
                                   deadline)
        try:
            refined_score = evaluate_sequence(refined.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # fallback (should never trigger)
    if global_best_seq is None:
        n = 4000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    # return a plain Python list
    return global_best_seq.tolist()
