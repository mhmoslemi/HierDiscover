"""optimisation of step‑function heights.

The algorithm works in two stages.
* Stage 1 – a fast soft‑max (smoothed max) Adam optimisation on a modest
  size (2000–4000 entries).  The variables are stored as log‑probabilities,
  which guarantees non‑negativity and exact simplex feasibility after the
  soft‑max.
* Stage 2 – the refined solution is interpolated to a larger length and
  continued with the same Adam optimiser, but the temperature α is
  increased until the true max is used (α → ∞).  A diagonal correction is
  applied so that the gradient of the exact max is accurate.

Both stages keep the best sequence seen and return the one with the lowest
score according to the supplied `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σ x = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete version of the Beta(½,½) law, already normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate(old: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    if old.size == new_len:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_len)
    new = np.interp(x_new, x_old, old)
    new = _project_onto_simplex(new, s=1.0)
    return new


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    time_budget: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the (soft) max of the self‑convolution.
    Returns the best sequence seen and the corresponding score
    (2·n·max_b because the sum is kept equal to 1).
    """
    # log‑probabilities (softmax guarantees feasibility)
    z = np.log(a0 + 1e-30)

    m = np.zeros_like(z)          # first moment
    v = np.zeros_like(z)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.15                     # base learning rate for Adam

    # temperature for the smoothed max
    alpha = 0.5
    alpha_factor = 1.0015
    alpha_max = 4000.0

    best_seq = a0.copy()
    best_b = _fft_self_convolve(a0)
    best_score = 2.0 * n * best_b.max()

    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        it += 1

        # ---- obtain a from logits ----
        z_max = np.max(z)
        a = np.exp(z - z_max)          # shift for stability
        a = a / a.sum()                # simplex

        # ---- convolution and (soft) max weighting ----
        b = _fft_self_convolve(a)
        cur_max = b.max()

        if alpha < 2000.0:
            # smoothed max: weights ≈ exp(-α·(cur_max-b))
            w = np.exp(alpha * (b - cur_max))
        else:
            # exact max – indicator of active indices
            eps_tol = 1e-12 * (abs(cur_max) + 1.0)
            w = np.zeros_like(b)
            active = np.where(b >= cur_max - eps_tol)[0]
            w[active] = 1.0

        # ---- gradient of the (soft) max w.r.t. a ----
        conv_rev = _fft_convolve(a[::-1], w)                # length 3n‑2
        part = conv_rev[n - 1 : n - 1 + n]                  # size n

        # diagonal correction (because each a_i a_i term is counted twice)
        idx = np.arange(n)
        diag_mask = (2 * idx) < w.size
        diag_corr = np.zeros_like(part)
        diag_corr[diag_mask] = w[2 * idx[diag_mask]] * a[diag_mask]
        grad_a = 2.0 * part - diag_corr                     # ∂max/∂a

        # ---- gradient w.r.t. logits (softmax) ----
        mu = np.dot(a, grad_a)                              # <a,grad_a>
        grad_z = a * (grad_a - mu)                          # chain rule

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # re‑center logits to avoid overflow
        z -= np.max(z)

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # ---- keep the best exact score seen ----
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] n={n} it={it} "
                f"alpha={alpha:.1f} score={best_score:.6f}"
            )

        # occasional random jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.1:
            z += np.random.randn(n) * 1e-3

        # early exit if we have already passed the target
        if best_score <= 1.503:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step‑function within the given time budget.
    Returns a plain Python list of non‑negative heights.
    """
    np.random.seed(seed)
    # keep a tiny safety margin for the surrounding harness
    overall_deadline = time.time() + budget_s - 2.0

    # lengths are grown geometrically – start small, finish with a larger vector
    lengths = [2000, 4000, 8000, 16000]

    overall_best_seq = None  # type: np.ndarray | None
    overall_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = max(0.0, overall_deadline - time.time())
        if remaining <= 0.0:
            break

        # allocate at most 30 % of the remaining budget to this length
        slice_budget = min(remaining * 0.30, 250.0)  # cap to keep a safety margin

        # initial vector: either the up‑sampled best so far or arcsine
        if overall_best_seq is not None:
            a0 = _interpolate(overall_best_seq, n)
        else:
            a0 = _arcsine_distribution(n)

        # run the Adam optimiser on this length
        best_seq_local, best_score_local = _adam_optimize(a0, n, slice_budget)

        # keep the global best (use the exact evaluator for the final check)
        try:
            exact = evaluate_sequence(best_seq_local.tolist())
        except Exception:
            exact = best_score_local  # fallback
        if exact < overall_best_score:
            overall_best_score = exact
            overall_best_seq = best_seq_local.copy()
            print(
                f"*** New global best → n={n} score={overall_best_score:.6f}"
            )

        # early stopping if the target has been reached
        if overall_best_score <= 1.503:
            print("Target reached – exiting early.")
            return overall_best_seq.tolist()

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if overall_best_seq is None:
        n_fallback = 1000
        overall_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    final_score = evaluate_sequence(overall_best_seq.tolist())
    print(
        f"Final best score: {final_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
