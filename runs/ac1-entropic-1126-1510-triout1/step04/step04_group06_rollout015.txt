def _lp_improve(f, target_sum, max_iter, deadline, size_fft):
    n = len(f)
    best_f = f.copy()
    best_score = quick_score(f, n, size_fft)
    M = ( _self_convolution(f, size_fft) ).max()
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # build A matrix
        offsets = -np.arange(n)
        data = [np.full(n, f[i]) for i in range(n)]
        A = diags(data, offsets, shape=(2*n-1, n), format='csr')
        b = np.full(2*n-1, M)
        # solve LP
        res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g == 0:
            break
        g = g0 * (target_sum / sum_g)
        # compute convs needed for mixing t search:
        conv_ff = _self_convolution(f, size_fft)
        conv_fg = _conv_fft(f, g, size_fft)
        conv_gg = _self_convolution(g, size_fft)
        # best t search
        t_candidates = np.linspace(0, 1, 21)
        best_local_score = float('inf')
        best_t = 0.0
        for t in t_candidates:
            conv = (1-t)*(1-t)*conv_ff + 2*t*(1-t)*conv_fg + t*t*conv_gg
            score = 2*n*conv.max() / (target_sum**2)  # since sum = target_sum fixed
            if score < best_local_score:
                best_local_score = score
                best_t = t
        # refine around best_t with finer grid if necessary
        # (optional)
        # update f
        f_new = (1 - best_t) * f + best_t * g
        # compute new M
        conv_new = _self_convolution(f_new, size_fft)
        M_new = conv_new.max()
        # if not decreasing M significantly break
        if M_new >= M * (1 - 1e-6):
            # maybe accept if improved score but not M?
            # break
            break
        f = f_new
        M = M_new
        # update best if improved
        if best_local_score < best_score:
            best_f = f.copy()
            best_score = best_local_score
            print(f"[LP iter {it+1}] improved score {best_score:.6f}")
    return best_f, best_score

"""
Propose a step‑function (list of non‑negative heights) that minimises the
self‑convolution peak used by `evaluate_sequence`.

Algorithm
---------
1. **Parametric exploration** – generate a very large set of analytic
   candidates, focusing on the double‑ended power‑law family
   a_i ∝ (i+β)^‑p + (n‑1‑i+β)^‑p, which is known to give low max‑convolution.
   Each candidate is simply scaled to the required ℓ₁‑norm
   (√(2 n)) and scored with an exact FFT based routine.
   The best raw candidate is kept.

2. **Long Adam‑based refinement** – the best candidate is further
   optimised by a smoothed‑max (soft‑max) descent.  The temperature is
   slowly annealed, Adam parameters are tuned for the problem size and a
   projection onto the simplex keeps the sum fixed.

3. **Linear‑programming bound tightening** – from the refined sequence we
   solve a linear programme that finds a second sequence `g` of maximal
   ℓ₁‑norm under the constraint (f∗g)(k) ≤ max(f∗f).  Mixing `f` and `g`
   with an optimal convex coefficient (searched on a small grid) provably
   does not increase the peak.  A few LP‑mix iterations are performed.

4. **Final random‑block sweep** – the remaining time is spent generating
   cheap random block candidates and keeping any that improve the score.

All steps respect the security checks in `evaluate_sequence`.  The function
returns the best sequence found before the 1000 s budget expires.
"""

import math
import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective (without the security wrapper)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    (w must sum to 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)  # length = 3·n‑2
    start = len(w) - len(a)               # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Smooth‑max Adam descent.  Returns best a seen and its raw score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning rate (scaled to problem size)
    eta0 = target_sum * 0.5 / n

    # initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights (smooth approximation of max) -----
        shifted = conv - conv.max()                # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)         # prevent overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam step -----
        eta = eta0 / (1.0 + it * 1e-5)               # gentle decay
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9997                               # slower cooling
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _lp_improve(
    f: np.ndarray,
    target_sum: float,
    size_fft: int,
    deadline: float,
    max_iter: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑based mixing steps (the “LP tightening” described
    in the paper).  Returns an improved sequence and its score.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = quick_score(f, n, size_fft)

    # current bound M = max(f*f)
    conv_ff = _self_convolution(f, size_fft)
    M = conv_ff.max()

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # Build convolution matrix A (shape (2n‑1, n))
        offsets = -np.arange(n, dtype=int)
        data = [np.full(n, f[i]) for i in range(n)]
        A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csr')

        # Solve: maximise Σ g_j  subject to A g ≤ M, g ≥ 0
        c = -np.ones(n)                     # minimise –sum(g)
        bounds = [(0, None)] * n
        b = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – stop trying further improvements
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 == 0:
            break

        # Scale g to the required ℓ₁‑norm
        g = g0 * (target_sum / sum_g0)

        # Pre‑compute convolutions needed for mixing
        conv_fg = _conv_fft(f, g, size_fft)
        conv_gg = _self_convolution(g, size_fft)

        # Search for the best mixing coefficient t ∈ [0,1]
        t_grid = np.linspace(0.0, 1.0, 21)
        best_local_score = float('inf')
        best_t = 0.0
        for t in t_grid:
            # (1‑t)² ff + 2t(1‑t) fg + t² gg
            conv_mix = ((1 - t) ** 2) * conv_ff + (2 * t * (1 - t)) * conv_fg + (t ** 2) * conv_gg
            score = 2.0 * n * conv_mix.max() / (target_sum ** 2)   # sum fixed = target_sum
            if score < best_local_score:
                best_local_score = score
                best_t = t

        # Optional finer search around best_t
        if best_t not in (0.0, 1.0):
            fine_grid = np.linspace(max(0.0, best_t - 0.05),
                                    min(1.0, best_t + 0.05), 11)
            for t in fine_grid:
                conv_mix = ((1 - t) ** 2) * conv_ff + (2 * t * (1 - t)) * conv_fg + (t ** 2) * conv_gg
                score = 2.0 * n * conv_mix.max() / (target_sum ** 2)
                if score < best_local_score:
                    best_local_score = score
                    best_t = t

        # Apply the mixing
        f = (1 - best_t) * f + best_t * g

        # Update quantities for next iteration
        conv_ff = _self_convolution(f, size_fft)
        M = conv_ff.max()

        if best_local_score < best_score - 1e-8:
            best_f = f.copy()
            best_score = best_local_score
            print(f"[LP iter {it+1}] improved score = {best_score:.6f}")

        # if improvement becomes tiny we stop
        if best_local_score >= best_score * (1 - 1e-6):
            break

    return best_f, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that
    minimises the self‑convolution peak defined in `evaluate_sequence`.
    The function respects a 1000 s time budget and returns the best
    sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)                 # normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())     # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # Keep the best raw candidate
    best_score = float('inf')
    best_a = None

    def consider(arr: np.ndarray) -> None:
        """Scale arr to the target ℓ₁‑norm, score it and keep if best."""
        nonlocal best_score, best_a
        if arr.sum() == 0:
            return
        arr = arr.astype(np.float64, copy=False)
        arr = arr * (target_sum / arr.sum())
        sc = quick_score(arr, n, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = arr.copy()

    # ------------------------------------------------------------------
    # ---- 1. PARAMETRIC SHAPE SEARCH ---------------------------------
    idx = np.arange(n)

    # (a) Uniform
    consider(np.ones(n))

    # (b) Inverse‑sqrt (single‑ended)
    for _ in range(300):
        beta = np.random.rand() * 12.0
        a = 1.0 / np.sqrt(idx + beta + 1.0)
        consider(a)

    # (c) Double‑ended inverse‑sqrt
    for _ in range(300):
        beta = np.random.rand() * 12.0
        a = 1.0 / np.sqrt(idx + beta + 1.0) + 1.0 / np.sqrt((n - 1 - idx) + beta + 1.0)
        consider(a)

    # (d) Power‑law (single‑ended)
    for _ in range(300):
        beta = np.random.rand() * 12.0
        pexp = np.random.rand() * 1.8 + 0.2
        a = 1.0 / (idx + beta + 1.0) ** pexp
        consider(a)

    # (e) Double‑ended power‑law (the most promising family)
    for _ in range(2000):
        beta = np.random.rand() * 12.0
        p = np.random.rand() * 1.8 + 0.2          # exponent ∈ [0.2, 2.0]
        a = (idx + beta + 1.0) ** (-p) + ((n - 1 - idx) + beta + 1.0) ** (-p)
        consider(a)

    # (f) Exponential decay (double‑ended, possibly asymmetric)
    for _ in range(500):
        decay_l = np.random.rand() * 0.015 + 0.0002
        decay_r = np.random.rand() * 0.015 + 0.0002
        a = np.exp(-decay_l * idx) + np.exp(-decay_r * (n - 1 - idx))
        consider(a)

    # (g) Cosine‑based (large at ends, small in the middle)
    for _ in range(200):
        p = np.random.rand() * 6.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** p
        consider(a)

    # (h) Sine‑based (zero at ends, peak in the middle)
    for _ in range(200):
        p = np.random.rand() * 6.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** p
        consider(a)

    # (i) Gaussian‑like bump (centered)
    for _ in range(200):
        sigma = (np.random.rand() * 0.25 + 0.05) * n
        centre = (n - 1) / 2.0 + (np.random.rand() - 0.5) * n * 0.1
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        consider(a)

    # (j) Triangular (symmetric peak at centre)
    for _ in range(120):
        peak = np.random.rand() * 3.0 + 0.5
        a = np.minimum(idx + 1, n - idx) ** peak
        consider(a)

    # (k) Random block candidates (few large constant blocks)
    for _ in range(600):
        nb = np.random.randint(2, 8)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        consider(cand)

    # (l) Reference sequence, if available
    if have_ref:
        consider(np.array(height_sequence_1, dtype=float))   # type: ignore[name-defined]

    print(f"[exploration] best raw score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 2. REFINEMENT (Adam‑based smoothed max descent) ------------
    if best_a is None:
        # fallback – uniform with correct norm
        best_a = np.ones(n, dtype=float) * (target_sum / n)

    refined_a, refined_score = _refine_candidate_adam(
        best_a, deadline, n, size_fft, target_sum, max_iters=10**9
    )
    best_a = refined_a
    best_score = refined_score
    print(f"[refine] after Adam best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 3. LP‑based mixing (bound tightening) -----------------------
    if time.time() < deadline - 2.0:
        lp_a, lp_score = _lp_improve(best_a, target_sum, size_fft, deadline, max_iter=5)
        if lp_score < best_score:
            best_a = lp_a
            best_score = lp_score
            print(f"[LP] after mixing best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # ---- 4. FINAL RANDOM BLOCK SWEEP ---------------------------------
    while time.time() < deadline:
        nb = np.random.randint(2, 9)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        consider(cand)
        # Break early if budget almost exhausted
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
