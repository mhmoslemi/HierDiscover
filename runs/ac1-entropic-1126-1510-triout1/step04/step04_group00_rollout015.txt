"""
propose_candidate
-----------------

A two‑phase search for a low‑max self‑convolution step‑function.

Phase‑1  : cheap parametric sweep.
           We try discrete “Beta‑type’’ families
           a_i ∝ ((i+0.5)(n‑i‑0.5))^{-β}  (β≈0.5 is the arcsine law).
           For several lengths (up to ≈8000) the best few candidates are kept.

Phase‑2  : refined optimisation.
           Each retained candidate is refined with a projected Adam optimiser.
           The optimiser uses a soft‑max surrogate of the ∞‑norm at first and
           switches to the exact sub‑gradient once the temperature is large.
           All updates are projected onto the probability simplex
           (non‑negative, sum = 1).

The whole routine stops after the allotted time (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)), already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, beta: float) -> np.ndarray:
    """
    A 1‑parameter family a_i ∝ ((i+0.5)(n‑i‑0.5))^{-β},
    normalised to sum‑1.
    """
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-beta)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _score_of_seq(a: np.ndarray) -> float:
    """Return the evaluation score for a (assumed to sum to 1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _refine_sequence(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector using projected Adam with a soft‑max
    surrogate of max(b). Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.shape[0]

    # keep the best encountered vector
    best_a = a.copy()
    best_score = _score_of_seq(best_a)

    # Adam buffers
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # temperature for the soft‑max approximation of max(b)
    alpha = 5.0
    alpha_factor = 1.0015
    alpha_max = 2000.0

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution and max
        b = _fft_self_convolve(a)
        cur_max = b.max()

        # record improvement
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
            print(
                f"[refine] n={n} iter={iteration} max={cur_max:.8e} "
                f"score={best_score:.6f}"
            )
            # early exit if target already reached
            if best_score <= 1.503:
                break

        # ----- gradient (soft‑max vs exact) -----
        if alpha < 500.0:
            # smoothed max using soft‑max weights
            exp_term = np.exp(alpha * (b - cur_max))
            w = exp_term / exp_term.sum()
            active = None  # not used in this branch
        else:
            # exact active set
            eps_act = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(b >= cur_max - eps_act)
            w = np.zeros_like(b)
            w[active] = 1.0

        # gradient of the (soft) max
        conv_rev = _fft_convolve(a[::-1], w)          # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]      # size n

        # diagonal correction for exact case (avoid double counting a_i)
        if alpha >= 500.0 and active is not None:
            even_mask = (active % 2 == 0)
            diag_idx = active[even_mask] // 2
            if diag_idx.size > 0:
                grad[diag_idx] -= a[diag_idx]

        # ----- Adam update with projection onto simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # base step‑size (slowly decreasing)
        eta = 0.15 / (np.sqrt(iteration) + 1.0)

        step = eta * m_hat / (np.sqrt(v_hat) + eps_adam)
        a = _project_onto_simplex(a - step, s=1.0)

        # increase temperature (sharpen the surrogate)
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to break plateaus
        if iteration % 1000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_score

    # Typical useful lengths (the longer the better, but we keep it manageable)
    lengths = [800, 1200, 2000, 3200, 4800, 6400, 8000]
    betas = np.linspace(0.30, 0.70, 21)   # 0.020 steps, centred around 0.5

    # keep the best few candidates (score, vector)
    candidate_pool: list[tuple[float, np.ndarray]] = []

    for n in lengths:
        # ensure we still have time
        if time.time() > deadline:
            break
        for beta in betas:
            a = _beta_distribution(n, beta)
            # tiny safety projection (handles rounding)
            a = _project_onto_simplex(a, s=1.0)
            sc = _score_of_seq(a)
            if sc < 1.8:   # very loose filter, keeps only promising ones
                candidate_pool.append((sc, a))
                # keep the pool small (top 15)
                candidate_pool = sorted(candidate_pool, key=lambda x: x[0])[:15]
            # early‑stop inner loop if we run out of time
            if time.time() > deadline:
                break
        # also check the pure arcsine law for this length
        if time.time() < deadline:
            a_arc = _arcsine_distribution(n)
            a_arc = _project_onto_simplex(a_arc, s=1.0)
            sc_arc = _score_of_seq(a_arc)
            candidate_pool.append((sc_arc, a_arc))
            candidate_pool = sorted(candidate_pool, key=lambda x: x[0])[:15]

    # Try to reuse the previously discovered construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = _project_onto_simplex(prev, s=1.0)
            candidate_pool.append((_score_of_seq(prev), prev))
            candidate_pool = sorted(candidate_pool, key=lambda x: x[0])[:15]
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) Refine the most promising candidates with the Adam optimiser
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # sort candidates by their raw score (ascending)
    candidate_pool.sort(key=lambda x: x[0])

    # limit the number of refinements to what time permits (max 8 refinements)
    max_refinements = min(8, len(candidate_pool))

    for idx in range(max_refinements):
        if time.time() > deadline:
            break
        init_score, init_vec = candidate_pool[idx]
        # allocate a slice of the remaining time for this refinement
        remaining = deadline - time.time()
        # give at most 0.25 of the remaining budget to a single run
        slice_deadline = time.time() + min(remaining * 0.25, 250.0)

        print(
            f"[refine‑start] candidate #{idx+1}/{max_refinements} "
            f"len={len(init_vec)} init_score={init_score:.6f}"
        )
        refined_vec, refined_score = _refine_sequence(init_vec, slice_deadline)

        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_vec.copy()
            print(
                f"*** New global best (refined) → len={len(best_seq)} "
                f"score={best_score:.6f}"
            )
            if best_score <= 1.503:
                print("Target reached – terminating early.")
                break

    # ------------------------------------------------------------------
    # 3) Fallback – if nothing succeeded, return a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = _score_of_seq(best_seq)

    print(
        f"Final best score: {best_score:.6f} (length = {len(best_seq)})"
    )
    return best_seq.tolist()
