def _smooth_optimize(a0, target_sum, size_fft, deadline, max_iters=200_000, alpha_start=0.1, alpha_end=30.0):

   a = a0.copy()
   best_a = a.copy()
   best_score = _max_convolution(a, size_fft)

   m = np.zeros_like(a)
   v = np.zeros_like(a)
   beta1=0.9
   beta2=0.999
   eps=1e-8
   eta0 = target_sum * 0.1

   for it in range(1, max_iters+1):
        if time.time() >= deadline: break

        # compute conv
        conv = _full_conv(a, size_fft)
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[smooth] it {it:,} new best = {best_score:.6f}")

        # schedule alpha
        # linearly increase
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, (it-1)/ (max_iters/2)) # maybe after half iterations reach end?
        # compute w (softmax)
        maxc = conv.max()
        # compute exponentials in stable way
        exp_vals = np.exp(alpha * (conv - maxc))
        sum_exp = exp_vals.sum()
        if sum_exp == 0:
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = exp_vals / sum_exp

        # gradient: weighted sum of subgradients
        # compute w_rev
        w_rev = w[::-1]
        # compute convolution a * w_rev
        size2 = _fft_size(len(a) + len(w_rev) - 1)
        fa = np.fft.rfft(a, n=size2)
        fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(fa * fw, n=size2)

        idx_offset = len(w_rev) - 1  # 2n-2
        grads = 2.0 * conv2[idx_offset - np.arange(len(a))]

        # Adam update
        m = beta1 * m + (1 - beta1) * grads
        v = beta2 * v + (1 - beta2) * (grads ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = eta0 / np.sqrt(it)  # decaying

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_to_simplex(a, target_sum)

   return best_a, best_score

"""
Propose a non‑negative step‑function (list of heights) that minimises the
evaluation score 2 n·max(conv)/(∑a)².
The algorithm consists of three stages:

1️⃣  **Initial sampling** – build a small pool of promising raw candidates
    (reference, uniform, Dirichlet draws, power‑law / exponential shapes,
    random block‑wise constants).  Each candidate is projected onto the simplex
    {a≥0, Σa = √(2 n)} and its peak convolution is measured.

2️⃣  **Continuation gradient descent** – for each promising candidate we run a
    projected Adam optimiser on a *smooth* approximation of the max‑convolution.
    The smooth objective is the log‑sum‑exp of the convolution entries; by
    slowly increasing the temperature parameter α we obtain a deterministic
    annealing that can escape shallow local minima.  The gradient is computed
    efficiently with two FFT‑convolutions per iteration.

3️⃣  **Local permutation hill‑climbing** – starting from the best refined
    sequence we perform random swaps (and occasional small perturbations) while
    keeping the best peak found.  This fine‑tunes the ordering of the values.

All updates respect the ℓ₁‑norm constraint via Euclidean projection onto the
simplex.  The routine stops before the supplied 1000 s budget and returns the
best sequence discovered.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Exact linear convolution a * a (length = 2 n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Peak of the self‑convolution (no clipping)."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _smooth_optimize(
    a0: np.ndarray,
    target_sum: float,
    size_fft: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha_start: float = 0.05,
    alpha_end: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth (log‑sum‑exp) approximation of the
    max‑convolution.  Returns the best sequence seen and its peak convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base step size (empirically chosen)
    base_lr = 0.05

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # current convolution and its peak
        conv = _full_conv(a, size_fft)
        cur_max = float(conv.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[smooth] it {it:,} new best = {best_score:.6f}")

        # temperature schedule (linear increase)
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, (it - 1) / max_iters)

        # softmax weights for the smooth objective
        maxc = conv.max()
        exp_vals = np.exp(alpha * (conv - maxc))
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth objective:
        #   grad = Σ_k w_k * ∂conv_k/∂a  = 2 * (a ** w_rev)_(2n-2‑i)
        w_rev = w[::-1]
        size2 = _fft_size(len(a) + len(w_rev) - 1)
        fa = np.fft.rfft(a, n=size2)
        fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(fa * fw, n=size2)
        idx_offset = len(w_rev) - 1               # = 2n‑2
        grad = 2.0 * conv2[idx_offset - np.arange(len(a))]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / math.sqrt(it)               # decay like 1/√t
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep a on the simplex
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑peak self‑convolution sequence within the given time budget.
    Returns a list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------- problem size ---------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]

    target_sum = math.sqrt(2.0 * n)          # Σa = √(2 n)

    # --------------------- helpers ---------------------
    def peak(a: np.ndarray) -> float:
        return _max_convolution(a, size_fft)

    # --------------------- 1. broad sampling ---------------------
    K = 15                                   # keep the best K raw candidates

    def push_candidate(score: float, a: np.ndarray) -> None:
        if len(best_raw) < K:
            best_raw.append((score, a.copy()))
            best_raw.sort(key=lambda x: x[0])
        else:
            if score < best_raw[-1][0]:
                best_raw[-1] = (score, a.copy())
                best_raw.sort(key=lambda x: x[0])

    idx = np.arange(n)

    # reference (if present)
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        push_candidate(peak(a_ref), a_ref)
        print(f"[init] reference peak = {peak(a_ref):.6f}")

    # uniform baseline
    a_uni = np.ones(n, dtype=float)
    a_uni = _project_to_simplex(a_uni, target_sum)
    push_candidate(peak(a_uni), a_uni)
    print(f"[init] uniform peak = {peak(a_uni):.6f}")

    # Dirichlet draws with various concentration parameters
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(8):

    for _ in range(25):

        a = np.exp(-dec * idx)
        a = _project_to_simplex(a, target_sum)
        push_candidate(peak(a), a)

    # Random block‑wise constant sequences (2‑6 blocks)
    for _ in range(120):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        push_candidate(peak(cand), cand)

    print(f"[sampling] best raw peak = {best_raw[0][0]:.6f}")

    # --------------------- 2. continuation gradient refinement ---------------------
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    for rank, (raw_score, a_start) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        # allocate a modest budget per candidate (seconds)
        remaining = deadline - time.time()
        iters = int(min(150_000, remaining * 2500))
        print(f"[refine] candidate #{rank+1} raw peak {raw_score:.6f} → {iters} iters")
        refined_a, refined_score = _smooth_optimize(
            a_start,
            target_sum,
            size_fft,
            deadline,

            global_best_score = refined_score
            global_best_a = refined_a.copy()

    print("[local swaps] start")
    a = global_best_a.copy()
    cur_score = global_best_score
    swap_attempts = 0

    while time.time() < deadline:
        swap_attempts += 1
        i, j = np.random.choice(n, 2, replace=False)
        a_swapped = a.copy()
        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
        sc = peak(a_swapped)
        if sc < cur_score:
            cur_score = sc
            a = a_swapped
            if sc < global_best_score:
                global_best_score = sc
                global_best_a = a.copy()
                print(f"[swap] improved to {global_best_score:.6f}")

        # occasional tiny perturbation to escape plateaus
