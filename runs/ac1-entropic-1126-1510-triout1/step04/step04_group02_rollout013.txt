def _fft_self_convolve(a):
    n = a.shape[0]
    # Use next power of 2 >= 2*n-1
    size = 1 << ((2*n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[:2*n-1]

def _optimize_pgda(a0, n, deadline):
    a = a0.copy()
    best_a = a.copy()
    conv = _fft_self_convolve(a)
    best_score = 2.0 * n * conv.max()
    # hyperparameters
    eta0 = 0.5
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 10000.0
    size_fft = 1 << ((3 * n - 2).bit_length())
    it = 0
    momentum = np.zeros_like(a)
    beta = 0.9
    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] pgd n={n} it={it} score={best_score:.6f}")

        # update alpha
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # gradient
        if it % 10 == 0 and alpha > 5000:  # Use subgradient when alpha large
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, conv, alpha, size_fft)

        # step
        eta = eta0 / np.sqrt(it)
        momentum = beta * momentum + grad
        a = a - eta * momentum
        # ensure nonnegative
        a = np.maximum(a, 0)
        # project onto simplex
        sum_a = a.sum()
        if sum_a <= 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a
    return best_a, best_score

"""
search for a non‑negative step‑function (probability vector) that
minimises `evaluate_sequence`.

The optimiser consists of three stages:

1. **Projected (additive) gradient descent** on the simplex.
   The objective is a smoothed max (soft‑max) of the self‑convolution.
   A momentum term accelerates convergence; we anneal the soft‑max
   temperature α and the step size η.

2. **Targeted local refinement** – repeatedly move a fraction of mass
   from the index that contributes most to the current maximal
   convolution entry to an index that does not, accepting only
   improvements.

3. **Final random‑move hill‑climbing** (a lightweight simulated‑annealing
   style search) to escape tiny plateaus.

All stages respect the global time‑budget (default 1000 s) and the best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]               # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_skewed_distribution(n: int) -> np.ndarray:
    """Beta(0.5,0.5)‑skewed distribution – heavy at the ends."""
    a = np.random.beta(0.5, 0.5, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = _arcsine_distribution(n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "beta":
        a = _beta_skewed_distribution(n)
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    a = np.maximum(a, 0.0)
    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (aperiodic) convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())          # power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max entry of a*a at index k."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·conv) w.r.t. a (soft‑max surrogate of max).
    Returns a vector of length len(a).
    """
    # soft‑max weights
    shift = conv - conv.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:                     # numerical safety
        w = np.zeros_like(w)
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # convolution of w with the reversed a
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv_g = np.fft.irfft(W * A_rev, n=size_fft)

    start = a.shape[0] - 1
    g = 2.0 * conv_g[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_pgda(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent (additive) with momentum.
    Returns the best sequence seen and its *internal* score 2·n·max.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # hyper‑parameters
    eta0 = 0.6                     # initial step‑size factor
    beta = 0.9                     # momentum coefficient
    alpha = 1.0                    # start temperature for soft‑max
    alpha_mul = 1.02
    alpha_max = 12000.0
    size_fft = 1 << ((3 * n - 2).bit_length())
    grad_mom = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        conv = _fft_self_convolve(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] pgda n={n} it={it} score={best_score:.6f}")

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # gradient (smooth max ↦ sub‑gradient later)
        if alpha > 3000.0:                         # switch to cheap sub‑gradient
            k = int(np.argmax(conv))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, conv, alpha, size_fft)

        # momentum update
        grad_mom = beta * grad_mom + grad

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # additive update + projection onto the simplex
        a = a - eta * grad_mom
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

    return best_seq, best_score


# ----------------------------------------------------------------------
def _targeted_refinement(a_start: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climbing: move mass from the heaviest contributor
    to the least‑used index of the current maximal convolution entry.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        k = int(np.argmax(conv))               # index of the current peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        support = np.arange(i_start, i_end + 1)

        # pick the index inside the support with the largest mass
        i = support[np.argmax(a[support])]

        # pick an index outside the support with the smallest mass
        outside = np.concatenate([np.arange(0, i_start), np.arange(i_end + 1, n)])
        if outside.size == 0:
            break
        j = outside[np.argmin(a[outside])]

        # transfer up to 20 % of a[i] (random amount)
        eps = a[i] * np.random.rand() * 0.2
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue

        cur_score = 2.0 * n * _fft_self_convolve(a_candidate).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] targeted improve it={it} score={best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Very small random‑move hill‑climbing (as in the previous code)."""
    a = a_start.copy()
    best_seq = a.copy()
