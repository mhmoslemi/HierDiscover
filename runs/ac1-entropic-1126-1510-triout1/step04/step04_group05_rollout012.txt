def _exp_gradient_descent(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    n = len(a_start)
    size = _fft_size(n)
    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    eta0 = 1.0 # initial step size factor
    eps = 1e-12
    it = 0
    while time.time() - start_time < budget:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        # compute w for active entries
        mask = conv == max_c   # could use tolerance for floating point?
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        # gradient w.r.t a
        # compute convolution of w and reversed a via FFT
        w_f = np.fft.rfft(w, n=size)
        a_rev_f = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(w_f * a_rev_f, n=size)[:2*n-1]
        g_a = 2.0 * conv_wr[n-1:n-1+n]
        # update step size
        eta = eta0 / math.sqrt(it + 1)
        # exponentiated update: a_new = a * exp(-eta * g_a)
        # Clip exponent to avoid overflow
        exp_arg = -eta * g_a
        # clip exp_arg to [-50, 50]
        exp_arg = np.clip(exp_arg, -50, 50)
        a = a * np.exp(exp_arg)
        # renormalize sum
        sum_a = np.sum(a)
        if sum_a <= 0:
            a = np.full_like(a, S / n)
            sum_a = S
        a = a * (S / sum_a)
        # evaluate
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[expGD improve] iter {it:,} score = {best_score:.6f}")
        it += 1
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative sequence) that minimizes the
evaluation score defined by `evaluate_sequence`.  The algorithm consists
of three phases:

1.  Simple initialisations (uniform, arcsine‑shaped, random Dirichlet,
    and optionally the supplied reference sequence).  The best of these
    seeds is selected.

2.  Exponentiated‑gradient (mirror‑descent) on the true hard‑max
    objective.  The update is a multiplicative step
        a_i ← a_i * exp(‑η·∇_i)
    followed by a Euclidean projection onto the simplex
    (a simple rescaling because the update keeps all entries non‑negative).
    The step‑size η decays as 1/√t.  This phase is fast and typically
    reduces the objective far below the previous 1.514 bound.

3.  A final refinement using the smooth‑max L‑BFGS continuation
    (as in the reference implementation) and a short projected‑subgradient
    polish.

All helper functions are top‑level; no closures or lambdas are used.
The total running time respects the 1000 s budget (a small safety margin
is subtracted from the deadline).  Progress is printed to stdout.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (soft‑max surrogate) and its gradient w.r.t. y.
    """
    p = _softmax(y)          # Σp = 1, p_i ≥ 0
    a = S * p                # Σa = S

    conv = np.convolve(a, a)      # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. heights a
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)                     # scalar = Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for all FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    # numerical noise may produce tiny negatives – clamp to zero
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


# ----------------------------------------------------------------------
def _exp_gradient_descent(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using an exponentiated‑gradient step.
    The objective is the hard max of the self‑convolution (i.e. the
    evaluation score up to the constant factor 2·n).
    """
    n = len(a_start)
    size = _fft_size(n)

    a = np.maximum(a_start, 1e-12)          # avoid exact zeros
    sum_a = a.sum()
    a = a * (S / sum_a)                     # enforce Σa = S

    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[expGD] initial score = {best_score:.6f}")

    eta0 = 5.0                               # initial step‑size factor
    it = 0
    while time.time() - start_time < budget:
        conv = _full_convolution(a, size)
        max_c = conv.max()

        # active set (tolerance to capture all nearly‑maximal entries)
        tol = 1e-12
        mask = conv >= max_c - tol
        cnt = np.count_nonzero(mask)

        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # gradient of the hard max w.r.t. a
        w_f = np.fft.rfft(w, n=size)
        a_rev_f = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(w_f * a_rev_f, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        # step‑size schedule (1/√t)
        eta = eta0 / math.sqrt(it + 1)

        # exponentiated update with clipping to avoid overflow
        exp_arg = -eta * g_a
        exp_arg = np.clip(exp_arg, -50.0, 50.0)
        a = a * np.exp(exp_arg)

        # rescale to the simplex (all entries stay non‑negative)
        sum_a = a.sum()
        if sum_a <= 0.0:
            a = np.full(n, S / n, dtype=float)
            sum_a = S
        a = a * (S / sum_a)

        # evaluate
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[expGD improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation on the smooth‑max surrogate (as in the reference
    implementation) but with a longer schedule and deeper inner L‑BFGS loops.
    """
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    mu_factor = 0.5
    mu_min = 1e-6
    maxiter_per_mu = 300

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


# ----------------------------------------------------------------------
def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true hard max (no smoothing).
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    eta0 = S * 0.5 / n            # roughly the average entry magnitude
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 5000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # gradient of the hard max
        w_f = np.fft.rfft(w, n=size)
        a_rev_f = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(w_f * a_rev_f, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score

        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 2000
        have_ref = False

    S = math.sqrt(2.0 * n)            # ℓ¹‑norm that makes the score = max‑conv

    # ------------------------------------------------------------------
    # Build a few diverse initial candidates
    candidates = []

    # uniform
    candidates.append(np.full(n, S / n, dtype=float))

    # arcsine‑like (large mass at the two ends, small in the centre)
    idx = np.arange(n, dtype=float)
    a_arc = 1.0 / np.sqrt((idx + 1) * (n - idx))
    p_arc = a_arc / a_arc.sum()
    candidates.append(S * p_arc)

    # random Dirichlet
    p_rand = np.random.dirichlet(np.ones(n))
    candidates.append(S * p_rand)

    # optional reference sequence
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        candidates.append(a_ref)

    # ------------------------------------------------------------------
    # Pick the best of the simple seeds
    best_a = None
    best_score = float("inf")
    for a in candidates:
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent (mirror descent)
    remaining = deadline - time.time()
    if remaining > 20.0:
        egd_budget = remaining * 0.55
        print(f"[expGD] running for {egd_budget:.1f}s")
        best_a, best_score = _exp_gradient_descent(best_a, S, egd_budget, time.time())
        print(f"[expGD] best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max L‑BFGS refinement
    remaining = deadline - time.time()
    if remaining > 10.0:
        print(f"[L‑BFGS] refining for {remaining:.1f}s")
        best_a, best_score = _l_bfgs_refine(best_a, S, remaining, time.time())
        print(f"[L‑BFGS] best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hard‑max sub‑gradient polish
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[final] polishing for {remaining:.1f}s")
        best_a, best_score = _final_subgradient(best_a, S, remaining, time.time())
        print(f"[final]  best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    if best_score == float("inf") or best_a is None:
        best_a = np.full(n, S / n, dtype=float)
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
