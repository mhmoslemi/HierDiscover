"""
Heuristic optimisation for minimizing the maximum of the self‑convolution
of a non‑negative step‑function under the ℓ¹‑norm constraint.
The search consists of three phases:
1. Simulated annealing on the simplex (mass moves).
2. Multi‑restart L‑BFGS on a smooth max (log‑sum‑exp) surrogate with a
   continuation schedule.
3. Projected sub‑gradient descent on the true hard‑max objective.
The target length is at least 8000 points (or twice the reference length
if a reference sequence is supplied). All iterates are projected onto the
simplex {x≥0, Σx = √(2n)} so that the evaluation score equals the maximal
convolution entry.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ---------- basic linear‑algebra helpers ---------------------------------

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for FFT‑based convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return the (non‑circular) convolution a*a of length 2·len(a)-1 via FFT."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a (using FFT)."""
    conv = _full_convolution(a, size)
    return float(conv.max())


def _interpolate_sequence(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    n0 = len(old_seq)
    if n0 == 0:
        return np.zeros(new_n)
    if n0 == 1:
        return np.full(new_n, old_seq[0])
    old_x = np.arange(n0)
    new_x = np.linspace(0, n0 - 1, new_n)
    return np.interp(new_x, old_x, old_seq)


# ---------- simulated annealing -----------------------------------------

def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Simple SA on the simplex: move a random amount of mass between two indices.
    The score is the maximal entry of the self‑convolution (equal to the true
    evaluate_sequence value when ∑a = S).
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    T0 = max(0.1, curr_max * 0.5)
    Tmin = 1e-6
    it = 0

    while time.time() - start_time < budget:
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * a[i] * 0.3
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = False
        if diff < 0.0:
            accept = True
        else:
            if T > 0.0 and np.random.rand() < math.exp(-diff / T):
                accept = True

        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score


# ---------- smooth‑max objective and L‑BFGS ------------------------------

def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Returns the smoothed max (log‑sum‑exp) of the self‑convolution and its
    gradient with respect to the unconstrained variables y.
    """
    p = _softmax(y)               # probability vector, Σp = 1
    a = S * p                     # enforce Σa = S

    size = _fft_size(n)
    conv = _full_convolution(a, size)   # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # exact max, distribute weight uniformly over all maximisers
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a: g_a = 2 * (w * a_rev) evaluated at indices n‑1 … n‑1+n‑1
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # chain rule back to y
    inner = np.dot(g_a, p)                # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable compatible with scipy.optimize – returns (value, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation on the smooth‑max surrogate.
    Starts from a_start, runs L‑BFGS for a decreasing sequence of smoothing
    parameters and keeps the best feasible point (evaluated with the true
    objective).
    """
    n = a_start.shape[0]
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    # initial smoothing parameter based on current max convolution
    size = _fft_size(n)
    conv_cur = _full_convolution(a_start, size)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    mu_factor = 0.7
    mu_min = 1e-5
    maxiter_per_mu = 300

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


def _multi_l_bfgs(a_start: np.ndarray, S: float,
                  budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """

"""
Heuristic optimisation for minimising the maximum of the self‑convolution
of a non‑negative step‑function under the ℓ¹‑norm constraint.

The search consists of three phases:

1. Simulated annealing on the simplex (random mass moves).
2. Multi‑restart L‑BFGS on a smoothed‑max surrogate (log‑sum‑exp) with a
   continuation schedule.
3. Projected sub‑gradient descent on the true hard‑max objective.

The sequence length is at least 8000 points (or twice the length of the
provided reference sequence, if any).  All iterates are projected onto the
simplex {x ≥ 0, Σx = √(2 n)} so that the evaluation score equals the maximal
convolution entry.  The routine respects the given time budget and returns
the best sequence it discovered.
"""

import time
import math
import numpy as np
import scipy.optimize as opt


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (for FFT‑based convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


def _interpolate_sequence(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    n0 = len(old_seq)
    if n0 == 0:
        return np.zeros(new_n)
    if n0 == 1:
        return np.full(new_n, old_seq[0])
    old_x = np.arange(n0)
    new_x = np.linspace(0, n0 - 1, new_n)
    return np.interp(new_x, old_x, old_seq)


# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Simple SA on the simplex – minimise max‑convolution."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    T0 = max(0.1, curr_max * 0.5)
    Tmin = 1e-6
    it = 0

    while time.time() - start_time < budget:
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * a[i] * 0.3
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = False
        if diff < 0.0:
            accept = True
        else:
            if T > 0.0 and np.random.rand() < math.exp(-diff / T):
                accept = True

        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float) -> tuple[float, np.ndarray]:
    """Smooth‑max (log‑sum‑exp) of the self‑convolution and its gradient."""
    p = _softmax(y)               # Σp = 1, p ≥ 0
    a = S * p                     # Σa = S

    size = _fft_size(n)
    conv = _full_convolution(a, size)   # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)                # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Wrapper for scipy.optimize – returns (value, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Continuation on the smooth‑max surrogate using L‑BFGS‑B."""
    n = a_start.shape[0]
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    size = _fft_size(n)
    mu = float(_full_convolution(a_start, size).max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())

    mu_factor = 0.7
    mu_min = 1e-5
    maxiter_per_mu = 300

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


def _multi_l_bfgs(a_start: np.ndarray, S: float,
                  budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Base L‑BFGS refinement plus a few cheap random‑perturbation restarts.

        best_a = S * np.ones(n) / n
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
