"""
Propose a low‑score step‑function.

The method starts from the best known construction (``height_sequence_1``)
when it is present, otherwise from a few generic seeds.  For each seed we
run a two‑phase optimisation:

1.  Adam minimisation of a smooth log‑sum‑exp surrogate of the peak of the
    self‑convolution.  The temperature (α) is increased aggressively up to a
    very large value (α_max=10000) so that the surrogate becomes almost the
    true ℓ∞‑objective.  Hyper‑parameters (learning‑rate, β‑moments) are chosen
    for fast convergence and many iterations are performed until the allocated
    time expires.

2.  An extended sub‑gradient descent on the exact maximum (the true ℓ∞‑objective)
    that continues as long as time permits.  The step size decays as 1/√t.

The whole budget (default 1000 s) is split between a handful of initial
strategies, giving the optimiser ample iterations to squeeze the peak lower.
The best sequence encountered (according to ``evaluate_sequence``) is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linear interpolation) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soft‑max weights (numerically stable)
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    # gradient = 2·rev(a) * w   (restricted to the n entries of a)
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 0.5,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 10000.0,
                   lr0: float = 0.3,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence encountered and its peak convolution value.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size schedule (slow decay)
        lr = lr0 / np.sqrt(t)

        # update & project onto the simplex
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        # keep the best exact peak seen so far
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional tiny jitter to escape plateaus
        if t % 500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 8e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the true ℓ∞‑objective.
    Uses a uniform weight on all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # active set (allow a tiny absolute tolerance)
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float, step0: float = 0.02) -> np.ndarray:
    """
    Extended sub‑gradient descent on the exact ℓ∞‑objective.
    The step size decays as step0/√t.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    t = 0
    while time.time() < deadline:
        t += 1
        grad, cur_max = _max_subgradient_step(a)
        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        # occasional jitter
        if t % 400 == 0 and np.random.rand() < 0.08:
            a += np.random.randn(n) * 6e-7
            a = _project_onto_simplex(a, s=1.0)

    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns a plain Python list ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Try to obtain the historic construction (if it exists)
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
    except Exception:
        base_seq = None

    # ------------------------------------------------------------------
    # Choose lengths to explore
    if base_seq is not None:
        lengths = [len(base_seq)]
    else:
        # generic lengths when no prior construction is available
        lengths = [2000, 4000, 6000, 8000]

    # Remove duplicates and keep reasonable sizes
    lengths = sorted({max(500, n) for n in lengths})

    # Global best tracking
    best_seq = None          # numpy array
    best_score = float("inf")

    # Split the total budget roughly equally across lengths
    time_per_length = max(2.0, (deadline - start_time) / len(lengths))

    for n in lengths:
        length_deadline = min(deadline, time.time() + time_per_length)
        if time.time() >= deadline:
            break

        # Initialisation strategies
        init_strategies = ["prev", "arcsine", "uniform", "random"]
        for kind in init_strategies:
            if time.time() >= length_deadline:
                break

            # ------------------------------------------------------------------
            # Initialise a0 on the simplex
            if kind == "prev" and base_seq is not None:
                a0 = _interpolate_to_length(base_seq, n)
            elif kind == "arcsine":
                a0 = _arcsine_distribution(n)
            elif kind == "uniform":
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            elif kind == "random":
                a0 = np.random.rand(n)
                a0 /= a0.sum()
            else:
                a0 = np.full(n, 1.0 / n, dtype=np.float64)

            a0 = _project_onto_simplex(a0, s=1.0)

            # ------------------------------------------------------------------
            # Phase 1 – Adam on the smooth‑max surrogate
            phase1_deadline = min(length_deadline,
                                  time.time() + (length_deadline - time.time()) * 0.55)
            a_opt, _ = _adam_optimize(a0, phase1_deadline,
                                      init_alpha=0.5,
                                      alpha_factor=1.01,
                                      alpha_max=10000.0,
                                      lr0=0.3,
                                      beta1=0.9,
                                      beta2=0.999,
                                      eps=1e-8)

            # ------------------------------------------------------------------
            # Phase 2 – Prolonged exact max refinement
            phase2_deadline = min(length_deadline,
                                  time.time() + (length_deadline - time.time()) * 0.45)
            a_refined = _refine_max(a_opt, phase2_deadline, step0=0.02)

            # ------------------------------------------------------------------
            # Fast (approximate) score based on the peak of the convolution
            b = _fft_self_convolve(a_refined)
            fast_max = b.max()
            fast_score = 2.0 * n * fast_max

            if fast_score < best_score:
                best_score = fast_score
                best_seq = a_refined.copy()
                print(f"[{time.time():.1f}] len={n} init={kind} fast_score={fast_score:.6f}")

            # ------------------------------------------------------------------
            # Exact evaluation (the black‑box is cheap compared to the search)
            try:
                true_score = evaluate_sequence(best_seq.tolist())
                if true_score < best_score:
                    best_score = true_score
                    best_seq = best_seq.copy()
                    print(f"*** New true best: len={n} init={kind} score={true_score:.6f}")
            except Exception as exc:
                print("evaluate_sequence raised:", exc)

            # Early exit if target already reached
            if best_score <= 1.503:
                break

        if best_score <= 1.503 or time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but ensures validity
    if best_seq is None:
        n_fallback = 1000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback uniform used.")

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # Return a plain Python list as required by the harness
    return best_seq.tolist()
