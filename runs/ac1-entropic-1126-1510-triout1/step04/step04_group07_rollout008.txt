"""
Low‑score step‑function search.

The algorithm consists of three phases:

1.  Smooth‑max continuation – we optimise a soft‑max parametrisation of the
    step heights with L‑BFGS‑B while gradually cooling the smooth‑max
    temperature (μ).  This gives a good starting point.

2.  Linear‑program refinement – for the current sequence `a` we solve a
    linear programme that maximises the attainable ℓ¹‑mass under the
    current convolution bound `M = max(a*a)`.  After scaling the optimiser
    `g` to the target ℓ¹‑norm we mix it with `a`.  Repeating a few times
    usually pushes the peak of the autoconvolution down.

3.  Cheap large‑scale random block search, a monotone‑sorting heuristic
    and a short projected‑gradient descent on the true hard‑max objective.

All steps respect a user supplied time budget (default 1000 s) and keep
the best feasible sequence found.  The functions are written without
closures or lambdas, use only NumPy/SciPy and can be called directly as
`propose_candidate()`.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
from scipy.sparse import diags
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – put the whole mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate value and gradient w.r.t. the free parameters y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained logits (length n).
    mu : float
        Positive temperature (larger ⇒ smoother).
    n : int
        Length of the step‑function.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : np.ndarray
        Gradient ∂f/∂y (shape (n,)).
    """
    # probabilities on the simplex and actual heights
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    # full autoconvolution a*a  (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weight vector w_k ∝ exp((conv_k - max)/mu)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        # clip to avoid overflow in exp
        diff = np.clip(diff, -700.0, 700.0)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                                   # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a
    # g_a[i] = 2 * Σ_k w_k * a_{k‑i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _solve_lp_for_g(a: np.ndarray, M: float, S: float):
    """
    Solve the linear programme:

        maximise   Σ_j g_j
        s.t.       (a * g)[k] ≤ M   for all k
                   g_j ≥ 0

    Returns the optimiser `g` (or None on failure).  The matrix A_ub is built
    as a sparse Toeplitz matrix with O(n²) non‑zeros.
    """
    n = a.shape[0]
    # Build the convolution matrix A where A[k, j] = a_j if 0 ≤ k‑j < n else 0
    # This is a Toeplitz matrix with n diagonals equal to the vector a.
    A = diags([a] * n, offsets=np.arange(n), shape=(2 * n - 1, n), format="csr")
    b = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=float)  # tiny slack

    c = -np.ones(n, dtype=float)  # minimise -Σ g_j  → maximise Σ g_j
    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception as exc:   # pragma: no cover
        print(f"[lp error] {exc}")
        return None

    if not res.success:
        # Occasionally the LP is infeasible due to numerical drift.
        # In that case we simply return None.
        return None

    return np.asarray(res.x, dtype=float)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects the given time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        # the global `height_sequence_1` may already contain a good candidate
        a_ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = a_ref.shape[0]
        have_ref = True
    except Exception:
        n = 3500
        a_ref = None
        have_ref = False

    S = math.sqrt(2.0 * n)               # target ℓ¹‑norm (normalisation)

    # bookkeeping
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # --------------------------------------------------------------
    def _maybe_update(a_vec: np.ndarray) -> None:
        """Evaluate `a_vec` via the public function and keep the incumbent."""
        nonlocal best_score, best_a
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # ---------- 1.  Smooth‑max continuation (L‑BFGS) -----------------
    init_ys = []

    # 0 – uniform (log‑probability zero)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – decreasing exponential
    alpha = rng.uniform(0.001, 0.008) * n / 1000.0
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # 2 – increasing exponential (mirror of the previous)
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 3 – random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 4 – reference if available (log‑probability of a projected reference)
    if have_ref:
        a_ref_proj = _project_to_simplex(a_ref, S)
        eps = 1e-12
        init_ys.append(np.log(a_ref_proj + eps))

    maxiter_per_mu = 400            # more inner iterations than the baseline
    mu_factor = 0.85                # slower cooling → finer continuation
    mu_min = 1e-8

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{len(init_ys)}] start optimisation")
        y = y0.copy()
        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = max(float(conv_cur.max()), 1.0)  # safeguard against zero start

        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu, "gtol": 1e-12, "disp": False},
                )
            except Exception as exc:          # pragma: no cover
                print(f"[lbfgs error] {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # --------------------------------------------------------------
    # ---------- 2.  Linear‑program refinement -----------------------
    # Use the current best_a as the starting point for the refinement.
    # If we never found a candidate (unlikely), fall back to uniform.
    if best_score == float("inf"):
        best_a = np.full(n, S / n, dtype=float)
        best_score = evaluate_sequence(best_a.tolist())

    a = best_a.copy()
    lp_iters = 0
    max_lp_iters = 12          # a modest number to stay within the budget
    t_mix = 0.5                # mixing coefficient for (a ← (1‑t)·a + t·g)

    while lp_iters < max_lp_iters and time.time() < deadline:
        conv = np.convolve(a, a)
        M = float(conv.max())
        g = _solve_lp_for_g(a, M, S)
        if g is None:
            break

        # Scale g to the target ℓ¹‑norm (if it already exceeds we project)
        if g.sum() > S:
            g = _project_to_simplex(g, S)
        else:
            # if the LP cannot deliver enough mass we fill the deficit uniformly
            deficit = S - g.sum()
            g = g + deficit / n

        # Mix with the current iterate and re‑project onto the simplex
        a_new = (1.0 - t_mix) * a + t_mix * g
        a_new = _project_to_simplex(a_new, S)

        _maybe_update(a_new)

        # prepare for next iteration
        a = a_new
        lp_iters += 1
        print(f"[lp] iteration {lp_iters}, best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # ---------- 3.  Random block search (cheap) --------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        max_trials = int(min(8000, remaining * 180))  # a few thousand trials
        print(f"[block] performing up to {max_trials} block trials")
        for _ in range(max_trials):
            nb = rng.integers(2, 13)                     # number of blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # --------------------------------------------------------------
    # ---------- 4.  Monotone‑sorting heuristic ---------------------
    if time.time() < deadline:
        desc = np.sort(best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(desc)

        asc = np.sort(best_a)
        asc = _project_to_simplex(asc, S)
        _maybe_update(asc)

    # --------------------------------------------------------------
    # ---------- 5.  Final projected‑gradient descent (hard max) -----
    if time.time() < deadline:
        a = best_a.copy()
        it = 0
        eta0 = S * 0.6 / n          # initial step size (empirically good)
        decay = 5e-5
        while time.time() < deadline and it < 5000:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt

            # gradient of the hard‑max objective
            conv_wr = np.convolve(w, a[::-1])
            g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

            eta = eta0 / (1.0 + decay * it)
            a_new = a - eta * g_a
            a_new = _project_to_simplex(a_new, S)

            sc = evaluate_sequence(a_new.tolist())
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"[grad] improved to {best_score:.6f}")

            a = a_new
            it += 1

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
