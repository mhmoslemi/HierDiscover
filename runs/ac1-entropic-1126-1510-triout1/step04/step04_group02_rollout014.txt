    a_i ∝ 1 / sqrt((i+½)(n‑i‑½))

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    # enforce exact simplex constraint (tiny numerical drift)
    vals = _project_onto_simplex(vals, s=1.0)
    return vals

    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()

    B = A * A
    conv = np.fft.irfft(B, n=size)

        final_score = evaluate_sequence(seq)
    except Exception:   # the evaluator is external – ignore any failure
        final_score = None
    print(f"[{time.time():.2f}] Proposed sequence – length={n}, score≈{final_score}")

    return seq
