def solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    n = len(a)
    # Build constraint matrix A (shape (2n-1, n)) for convolution a*b <= M.
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # for i in [i_min, i_max], column j = k - i
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
    # Upper bounds vector ub = np.full(2*n - 1, M)
    c = -np.ones(n)  # minimization of -sum b = maximize sum b
    bounds = [(0, None) for _ in range(n)]
    res = opt.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    return res.x
```

def _lp_refine(a_start: np.ndarray, S: float, budget: float, start_time: float) -> Tuple[np.ndarray, float]:
    n = len(a_start)
    size = _fft_size(n)
    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    # constraints matrix: we can precompute rows, cols, data for a (since a changes each iteration)
    # Actually A depends on a each iteration, thus need rebuild.
    # We'll loop with time constraint.
    it = 0
    while time.time() - start_time < budget and it < 10:
        M = _max_convolution(a, size)
        # Build LP matrix for current a
        rows = []
        cols = []
        data = []
        # To speed up, we can use vectorized approach:
        # For each i, j vector
        for i in range(n):
            # For column j = 0..n-1, row = i + j
            rows.extend(range(i, i + n))
            cols.extend(range(n))
            data.extend([a[i]] * n)
        A = csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))
        # Solve LP
        c = -np.ones(n)  # maximize sum b
        b_ub = np.full(2*n - 1, M, dtype=float)
        bounds = [(0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print("[LP] zero sum b")
            break
        # Scale to sum S
        g = b * (S / sum_b)
        # line search
        best_t = 0.0
        best_M = M
        # sample t's
        for t in np.linspace(0, 1, 11):
            f_new = (1 - t) * a + t * g
            M_f = _max_convolution(f_new, size)
            if M_f < best_M:
                best_M = M_f
                best_t = t
        # update a
        a = (1 - best_t) * a + best_t * g
        # evaluate score
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP improve] iter {it} t={best_t:.3f} score={best_score:.6f}")
        it += 1
        # optional break if no improvement
        if best_t == 0:
            # no better t found
            break
    return best_a, best_score

def _project_to_simplex(v, z):
    # Ensure v is 1d index.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # If all zeros, put all mass on first component.
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step sequence that minimizes the evaluation
function.  The search combines four stages:

1.  A quick simulated‑annealing (SA) on the simplex to get a decent
    starting point (mass moves between coordinates).
2.  An LP‑based refinement: for the current sequence `a` with
    max‑convolution `M`, solve a linear program that maximises the
    ℓ¹‑mass of a new sequence `b` subject to `a * b ≤ M`.  After scaling
    `b` to the required ℓ¹‑norm we line‑search a convex combination
    `a ← (1‑t) a + t b` that reduces the true max‑convolution.
    This step is cheap (a few LP solves) and often pushes the bound
    below the best SA result.
3.  Smooth‑max continuation using L‑BFGS‑B (the “μ‑continuation”
    technique from the reference) to further reduce the maximum.
4.  A final projected sub‑gradient descent on the hard max‑operator
    for polishing.

All phases respect the time budget (default 1000 s).  The code works
with any sequence length – if a global `height_sequence_1` is present
it is used as the initial point, otherwise a uniform start is used.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1]
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (for FFT‑based convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0               # remove tiny negatives
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    return float(_full_convolution(a, size).max())


# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float):
    """Simple SA on the simplex – moves a random fraction of mass."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-6
    it = 0
    while time.time() - start_time < budget:
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * a[i] * 0.3
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = diff < 0.0 or (T > 0.0 and np.random.rand() < math.exp(-diff / T))
        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
def _lp_refine(a_start: np.ndarray, S: float,
               budget: float, start_time: float):
    """LP‑based refinement (see description)."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # Pre‑compute index arrays for the (a*b) constraint matrix.
    i_arr = np.repeat(np.arange(n, dtype=np.int32), n)
    j_arr = np.tile(np.arange(n, dtype=np.int32), n)
    rows = i_arr + j_arr                     # length n²
    cols = j_arr

    max_iters = 8
    it = 0
    while time.time() - start_time < budget and it < max_iters:
        M = _max_convolution(a, size)

        # Build the constraint matrix A such that (A @ b) = a * b.
        data = a[i_arr]                       # a[i] repeated n times each
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Linear programme: maximise sum(b)  ⇔  minimise -sum(b)
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        try:
            res = opt.linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                              method="highs", options={"presolve": True})
        except Exception as exc:
            print(f"[LP] linprog exception: {exc}")
            break

        if not res.success:
            print(f"[LP] linprog failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print("[LP] zero sum – abort")
            break

        # Scale to the required ℓ¹‑norm.
        g = b * (S / sum_b)

        # Simple line–search on t ∈ [0,1].
        best_t = 0.0
        best_M = M
        for t in np.linspace(0.0, 1.0, 11):
            f_new = (1.0 - t) * a + t * g
            M_f = _max_convolution(f_new, size)
            if M_f < best_M:
                best_M = M_f
                best_t = t

        a = (1.0 - best_t) * a + best_t * g
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP improve] iter {it} t={best_t:.3f} score={best_score:.6f}")

        # If no movement helped, stop early.
        if best_t == 0.0:
            break
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """Smooth‑max surrogate and its gradient w.r.t. the log‑probabilities y."""
    p = _softmax(y)                 # Σp = 1, p ≥ 0
    a = S * p                       # Σa = S

    # hard max (via np.convolve) – cheap for n ≈ few thousand
    conv = np.convolve(a, a)        # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = max(1, np.count_nonzero(mask))
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient wrt a (see derivation in the reference)
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float):
    """Continuation on the smooth‑max surrogate (μ‑schedule)."""
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    mu = max(mu, 1.0)

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())

    mu_factor = 0.5
    mu_min = 1e-6
    maxiter_per_mu = 250

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(obj, y, method="L-BFGS-B",
                               jac=True,
                               options={"maxiter": maxiter_per_mu,
                                        "gtol": 1e-12,
                                        "disp": False})
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor
    return best_a, best_score


# ----------------------------------------------------------------------
def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float):
    """Projected sub‑gradient descent on the true hard max."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    eta0 = S * 0.5 / n          # typical magnitude of a coordinate
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 5000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt:
            w[mask] = 1.0 / cnt

        # sub‑gradient of max‑convolution
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score

    """Search for a low‑score step‑function within the given time‑budget."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the host

    # ------------------------------------------------------------------
    # problem size
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        a0 = np.asarray(height_sequence_1, dtype=float)
        have_ref = True
    except Exception:
        n = 2000
        a0 = np.full(n, 1.0, dtype=float)
        have_ref = False

    S = math.sqrt(2.0 * n)                 # ℓ¹‑norm that makes the score = max‑conv

    if have_ref:
        a0 = _project_to_simplex(a0, S)
    else:

    best_a = a0.copy()

    # ------------------------------------------------------------------

        best_score = evaluate_sequence(best_a.tolist())
        print(f"[SA] best score = {best_score:.6f}")

    remaining = deadline - time.time()

    if remaining > 15.0:

        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
