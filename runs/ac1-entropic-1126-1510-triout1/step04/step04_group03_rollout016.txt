def _gd_optimize(a_init, n, deadline):
    a = a_init.copy()
    best_a = a.copy()
    b = _fft_self_convolve(a)
    best_score = compute_score(a)
    # hyperparameters
    eta0 = 0.5
    beta_mom = 0.9
    velocity = np.zeros_like(a)
    alpha = 5.0
    alpha_factor = 1.03
    size_fft = 1 << ((3*n - 2).bit_length())
    it = 0
    while time.time()<deadline:
        it +=1
        b = _fft_self_convolve(a)
        cur_score = 2 * n * b.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"GD n={n} iter={it} score={best_score:.6f}")
        # gradient
        grad = _gradient_via_softmax(a,b,alpha,size_fft) # compute grad
        # momentum update
        velocity = beta_mom*velocity + (1-beta_mom)*grad
        # step size
        eta = eta0 / np.sqrt(it)
        a = a - eta*velocity
        a = _project_onto_simplex(a, s=1.0)
        # anneal temperature
        if alpha < 50000:
            alpha = min(50000, alpha*alpha_factor)
        # random jitter occasionally
        if it%2000==0 and np.random.rand()<0.3:
            a += np.random.randn(n)*1e-5
            a = _project_onto_simplex(a, s=1.0)
    return best_a, best_score

def _local_search_beef(a, n, deadline):
    best = a.copy()
    best_score = compute_score(best)
    while time.time()<deadline:
        # choose random donor index i with a_i > 1e-12
        i = np.random.choice(np.where(best>1e-12)[0])
        # choose random receiver j != i
        j = np.random.choice(n-1)
        if j>=i: j+=1
        # move fraction up to 0.5 of donor
        frac = np.random.rand()*0.5
        delta = best[i]*frac
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        # compute score
        # we could compute convolution incrementally but easier to compute full.
        new_score = compute_score(new)
        if new_score < best_score:
            best = new
            best_score = new_score
        # occasionally do jitter
        if np.random.rand() < 0.001:
            new = best + np.random.randn(n)*1e-6
            new = _project_onto_simplex(new, s=1.0)
            new_score = compute_score(new)
            if new_score < best_score:
                best = new
                best_score = new_score
    return best, best_score

        # allocate time slice
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        slice_deadline = time.time() + max(0.5, remaining * 0.25) # allocate quarter time each len.

        # candidate init_params

1.  A small family of parametrised probability vectors is generated.
    • Beta(α,α) densities discretised on {0,…,n‑1} (α∈[0.3,1.0]).
    • The discrete arcsine distribution (α=0.5) and a uniform baseline.
    • The previously found best sequence (if the global variable
      `height_sequence_1` exists) is also used as a restart.

2.  For each candidate length `n` (several lengths up to 30 000 are tried)
    a projected gradient descent on the simplex is run.
    * The objective is `2·n·max(conv(a,a))` (the same value returned by
      `evaluate_sequence` for a vector normalised to sum = 1).
    * The gradient of a smooth approximation of `max` is computed via a
      soft‑max (log‑sum‑exp) with an annealing temperature `α`.
    * Momentum (Nesterov‑style) and a diminishing step‑size `η₀/√t`
      are employed.
    * Very occasional random jitter helps escape plateaus.

3.  After the gradient stage a cheap stochastic local‑search is applied:
    repeatedly move a random fraction of mass from one index to another
    and keep the move if it improves the objective.

4.  The best sequence over all lengths and initial shapes is kept.
    If time permits a final refinement (local search) is performed on the
    global best.

The whole procedure respects the 1000 s wall‑clock budget, prints
progress information and finally returns the best list of heights found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Discrete Beta(α,α) density on n points, normalised to sum = 1."""
    # mid‑point discretisation to avoid pathological zero values
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    log_vals = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    vals = np.exp(log_vals)
    # protect against overflow when α ≪ 1
    if not np.isfinite(vals).all():
        vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    total = vals.sum()
    if total == 0.0:
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / total
    return vals


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)) on n points."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a computed via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power‑of‑two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log Σ exp(α·b_k) w.r.t. a.
    For very large α this becomes a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # ---- soft‑max weights (or exact arg‑max) ----
    if alpha > 20000:
        # treat all entries attaining the maximum as active
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()          # improve numerical stability
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ---- convolution w * reverse(a) ----
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector a (sum(a)=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _optimize_one_length(a0: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected (soft‑max) gradient descent on the simplex for a fixed length.
    Returns the best sequence seen and its internal (unnormalised) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = _objective(a)

    # hyper‑parameters
    beta = 0.85                # momentum coefficient
    eta0 = 0.7                 # initial step‑size factor
    velocity = np.zeros_like(a)

    alpha = 5.0                # start temperature for log‑sum‑exp
    alpha_mul = 1.08
    alpha_max = 50000.0

    size_fft = 1 << ((3 * n - 2).bit_length())   # safe size for w + a_rev

    it = 0
    while time.time() < deadline:
        it += 1

        # ----- current objective ------------------------------------------------
        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # ----- convolution and gradient -----------------------------------------
        b = _fft_self_convolve(a)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # ----- momentum / Nesterov style update ---------------------------------
        velocity = beta * velocity + (1.0 - beta) * grad

        # ----- diminishing step size --------------------------------------------
        eta = eta0 / np.sqrt(it)

        # ----- gradient step + projection onto simplex ---------------------------
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # ----- anneal temperature ------------------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # ----- occasional tiny random jitter ------------------------------------
        if it % 3000 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search.
    Randomly moves a fraction of mass from one coordinate to another,
    keeping the change if the objective improves.
    """
    best = a.copy()
    best_score = _objective(best)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        # pick a donor index with non‑zero mass
        donor_candidates = np.where(best > 1e-12)[0]
        if donor_candidates.size == 0:
            break
        i = rng.choice(donor_candidates)
        # pick a distinct receiver
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 30 %) of the donor's mass
        frac = rng.random() * 0.30
        delta = best[i] * frac
        a_new = best.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective(a_new)
        if new_score < best_score:
            best = a_new
            best_score = new_score

        # tiny jitter every few thousand iterations
        if it % 5000 == 0:
            jitter = best + rng.standard_normal(n) * 1e-6
            jitter = _project_onto_simplex(jitter, s=1.0)
            jitter_score = _objective(jitter)
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score

    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Main entry point for the grader.
    Searches for a step‑function (list of non‑negative heights) with a low
    ``evaluate_sequence`` score.  The function respects the 1000 s wall‑clock
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    # keep a small safety margin for final printing
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # base length – reuse a previously discovered sequence if available
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # candidate lengths (capped at 30 000 for speed)
    length_factors = (0.7, 0.85, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(300, int(base_n * f))
        for f in length_factors
        if max(300, int(base_n * f)) <= 30000
    })

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # allocate roughly 20 % of remaining time to this length
        slice_deadline = time.time() + max(0.2, remaining * 0.20)

        # ---------- initialise with a variety of shapes -----------------
        init_params = []

        # previous best (if compatible)
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size == n:
                init_params.append(("prev", _project_onto_simplex(prev, s=1.0)))
        except Exception:
            pass

        # beta‑family (α from 0.30 to 1.00)
        for a in (0.30, 0.40, 0.50, 0.60, 0.80, 1.00):
            init_params.append((f"beta_{a:.2f}", _beta_distribution(n, a)))

        # uniform and arcsine for completeness
        init_params.append(("uniform", np.full(n, 1.0 / n, dtype=np.float64)))
        init_params.append(("arcsine", _arcsine_distribution(n)))

        # two‑spike pattern (helps escaping special plateaux)
        a_spike = np.zeros(n, dtype=np.float64)
        pos1 = np.random.randint(max(1, n // 10))
        pos2 = np.random.randint(n - max(1, n // 10), n)
        a_spike[pos1] = 0.5
        a_spike[pos2] = 0.5
        init_params.append(("spike", _project_onto_simplex(a_spike, s=1.0)))

        # ------------------------------------------------------------------
        for kind, a0 in init_params:
            if time.time() >= slice_deadline:
                break

            # gradient‑based optimisation
            seq, _ = _optimize_one_length(a0, n, slice_deadline)

            # short local‑search phase (if any time left)
            time_left = slice_deadline - time.time()
            if time_left > 1.0:
                seq, _ = _local_search(seq, n, time.time() + min(time_left, 10.0))

            # evaluate with the official black‑box (more reliable)
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = _objective(seq)   # fallback to internal value

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq.copy()
                print(f"*** New global best → len={n} init={kind} "
                      f"score={global_best_score:.6f}")

            if global_best_score <= 1.503:
                break

        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # final refinement on the overall best (if time permits)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _local_search(global_best_seq,
                                       len(global_best_seq),
                                       deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # fallback – should never be needed
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
