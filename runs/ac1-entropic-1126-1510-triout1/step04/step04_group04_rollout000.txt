def _smooth_descent(a0, size_fft, size_fft2, target_sum, deadline, max_iters=200_000, init_alpha=1.0, alpha_factor=1.5, alpha_increase_iters=5000):
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    alpha = init_alpha

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # compute conv and smooth weights
        conv_full = _full_conv(a, size_fft)
        # compute smooth gradient
        # stable softmax
        max_c = conv_full.max()
        shifted = conv_full - max_c
        exp_alpha = np.exp(alpha * shifted)
        w = exp_alpha / exp_alpha.sum()
        # compute convolution w * a_rev
        a_rev = a[::-1]
        # compute convolution via FFT:
        wf = np.fft.rfft(w, n=size_fft2)
        ar = np.fft.rfft(a_rev, n=size_fft2)
        grad_full = np.fft.irfft(wf * ar, n=size_fft2)
        # slice:
        grad = 2 * grad_full[(len(a)-1):(len(a)-1)+len(a)]

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = target_sum * 0.5 / a.size / np.sqrt(it)  # similar base step
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection to simplex
        a = _project_to_simplex(a, target_sum)

        # compute actual peak
        cur_score = _max_convolution(a, size_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # maybe print
            if it % 2000 == 0:
                print(f"[smooth_descent] it {it} new best {best_score:.6f}")

        # update alpha schedule:
        if it % alpha_increase_iters == 0:
            alpha *= alpha_factor
            # limit alpha maybe to 100?
            if alpha > 200:
                alpha = 200

    return best_a, best_score

"""
propose_candidate
=================

Search for a non‑negative step‑function (vector) that minimises the
self‑convolution peak  – i.e. the evaluation score returned by
``evaluate_sequence``.  

The algorithm combines several ideas:

1. **Broad random sampling** – generate a pool of diverse candidates
   (Dirichlet draws, power‑law, exponential, block‑wise constant, and the
   reference if available).  All candidates are projected onto the
   ℓ₁‑simplex Σa = √(2 n) and the best *K* raw candidates are kept.

2. **Smooth max‑convolution descent** – minimise a smooth approximation of
   the max‑convolution using a soft‑max (log‑sum‑exp) with an increasing
   temperature parameter.  The gradient is obtained analytically via FFT
   (convolution of the soft‑max weights with the reversed vector).  Adam
   with Euclidean projection onto the simplex drives the optimisation.

3. **Hard max‑subgradient polishing** – once the smooth phase is finished
   a few thousand steps of the original max‑subgradient method are applied
   to fine‑tune the worst index.

4. **Local permutation / mass‑transfer hill climbing** – random swaps of
   positions and small transfers of mass between entries are performed
   while keeping the ℓ₁‑norm fixed.

The whole search respects the 1000 s time budget; the function returns the
best sequence found within that limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Smallest power‑of‑2 ≥ required."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of x and y using FFT, returns length len(x)+len(y)-1."""
    fx = np.fft.rfft(x, n=size_fft)
    fy = np.fft.rfft(y, n=size_fft)
    conv = np.fft.irfft(fx * fy, n=size_fft)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2 n‑1)."""
    return _fft_convolve(a, a, size_fft)


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of a*a (no clipping)."""
    return float(_full_conv(a, size_fft).max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)_k at the current worst index k_max."""
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        start = k_max - n + 1
        length = 2 * n - 1 - k_max
        g[start:] = 2.0 * a[:length][::-1]

    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Hard max‑subgradient (Adam) descent with simplex projection.
    Returns the best sequence seen and its max‑convolution value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())

        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[hard_subgrad] it {it:,} new best = {best_score:.6f}")

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _smooth_descent(
    a0: np.ndarray,
    size_fft: int,
    size_fft2: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 150_000,
    init_alpha: float = 1.0,
    alpha_factor: float = 2.0,
    alpha_step: int = 3_000,
) -> tuple[np.ndarray, float]:
    """
    Adam descent on the smooth approximation
        f(a) = (1/α) log Σ_k exp(α·(a*a)_k)
    The gradient is obtained via FFT.  α is increased geometrically every
    ``alpha_step`` iterations to approach the true max.
    Returns the best vector seen and its *hard* max‑convolution.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    eta0 = target_sum * 0.5 / a.size
    alpha = init_alpha

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convolution and soft‑max weights -----
        conv_full = _full_conv(a, size_fft)               # length 2n‑1
        # stable soft‑max
        shift = conv_full.max()
        exp_alpha = np.exp(alpha * (conv_full - shift))
        w = exp_alpha / exp_alpha.sum()                  # length 2n‑1, sums to 1

        # ----- gradient via convolution w * a_rev -----
        a_rev = a[::-1]
        grad_full = _fft_convolve(w, a_rev, size_fft2)   # length 3n‑2
        grad = 2.0 * grad_full[(n - 1) : (n - 1) + n]    # slice of length n

        # ----- Adam update -----
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = _project_to_simplex(a, target_sum)

        # ----- keep best hard value -----
        cur_score = _max_convolution(a, size_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[smooth] it {it:,} new best (hard) = {best_score:.6f}")

        # ----- increase α gradually -----
        if it % alpha_step == 0:
            alpha = min(alpha * alpha_factor, 200.0)

    return best_a, best_score


# ----------------------------------------------------------------------

    Search for a step‑function with a low self‑convolution peak.
    Respects the 1000 s time budget and returns the best sequence found.
    """

    try:

    # ------------------- utility -------------------

        """Peak of a*a (the quantity we actually minimise)."""
        return _max_convolution(a, size_fft)

    # keep top‑K raw candidates

    def add_candidate(sc: float, a: np.ndarray) -> None:
        if len(best_raw) < K:
            best_raw.append((sc, a.copy()))
            best_raw.sort(key=lambda x: x[0])
        else:
            if sc < best_raw[-1][0]:
                best_raw[-1] = (sc, a.copy())
                best_raw.sort(key=lambda x: x[0])

    # ------------------- 1. broad sampling -------------------
    idx = np.arange(n)

    # reference (if any)
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]

        add_candidate(quick_score(a_ref), a_ref)
        print(f"[init] reference score = {quick_score(a_ref):.6f}")

    # uniform baseline
    a_uni = np.ones(n, dtype=float)
    a_uni = _project_to_simplex(a_uni, target_sum)
    add_candidate(quick_score(a_uni), a_uni)
    print(f"[init] uniform score = {quick_score(a_uni):.6f}")

    # Dirichlet draws (various concentrations)
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(12):

            a = probs * target_sum
            add_candidate(quick_score(a), a)

    # Power‑law decreasing shapes
    for _ in range(40):
        p = np.random.rand() * 3.0 + 0.2          # exponent ∈ [0.2, 3.2]
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        add_candidate(quick_score(a), a)

    # Exponential decay shapes
    for _ in range(40):
        dec = np.random.rand() * 0.015 + 0.001   # decay ∈ [1e‑3, 1.6e‑2]
        a = np.exp(-dec * idx)
        a = _project_to_simplex(a, target_sum)
        add_candidate(quick_score(a), a)

    # Random block‑wise constant sequences (2‑6 blocks)
    for _ in range(180):
        nb = np.random.randint(2, 7)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)

            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        add_candidate(quick_score(cand), cand)

    print(f"[sampling] best raw score = {best_raw[0][0]:.6f}")

    # ------------------- 2. smooth‑max descent -------------------
    global_best_a = best_raw[0][1].copy()

    for rank, (raw_sc, a0) in enumerate(best_raw):
