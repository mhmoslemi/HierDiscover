def _lp_step(a, M):
    n = len(a)
    # Build matrix A of shape (2*n-1, n)
    rows = []
    cols = []
    data = []
    for i in range(n):
        # column i contributes a[i] to rows i .. i+n-1
        rows.extend(range(i, i+n))
        cols.extend([i]*n)
        data.extend([a[i]]*n)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ g <= M
    c = -np.ones(n)  # maximize sum -> minimize -sum
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

def _lp_step(a, M):
    n = len(a)
    # build constraint matrix A_ub (2n-1) x n
    # each column i has a[i] in rows i..i+n-1
    rows = np.arange(2*n-1).repeat(0) # placeholder

The routine first searches a one‑parameter family of symmetric beta

    _SCIPY_AVAILABLE = True

    _SCIPY_AVAILABLE = False


def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric beta(α,α) law on {0,…,n‑1} (discrete, ℓ₁‑normalised)."""
    i = np.arange(n, dtype=np.float64)
    # exponent = α‑1; values are (i+½)^{α‑1}·(n‑i‑½)^{α‑1}
    vals = ((i + 0.5) * (n - i - 0.5)) ** (alpha - 1.0)
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total == 0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= total
    return vals


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the smooth max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k) .
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # stable soft‑max weights
    exp_term = np.exp(alpha * (b - cur_max))
    w_sum = exp_term.sum()
    w = exp_term / w_sum
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)          # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    init_alpha: float = 0.5,
    alpha_factor: float = 1.003,
    alpha_max: float = 5000.0,
    lr0: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best sequence encountered.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()

        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter
        if t % 500 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(a.shape[0]) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


def _max_subgradient_step(a: np.ndarray):
    """
    One sub‑gradient step for the true ℓ∞ objective.
    Uses a uniform weight over all indices attaining the current maximum.
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size > 0:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """Short sub‑gradient refinement of the exact ℓ∞ objective."""
    a = a0.copy()
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.01 / np.sqrt(it)  # diminishing step size
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        if cur_max < best_peak - 1e-14:
            best_peak = cur_max
            best_a = a.copy()
        if it % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(a.shape[0]) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


def _lp_step(a: np.ndarray, M: float):
    """
    Solve the linear program
        maximise   Σ g_i
        subject to (a * g)_k ≤ M   for all k,
                   g_i ≥ 0 .
    Returns the optimal g (or None on failure).
    """
    if not _SCIPY_AVAILABLE:
        return None
    n = len(a)
    # Build sparse matrix A with shape (2n‑1, n)
    # Column i contains a[i] on rows i … i+n‑1
    A = _sp.lil_matrix((2 * n - 1, n), dtype=np.float64)
    for i, ai in enumerate(a):
        A[i : i + n, i] = ai
    A = A.tocsr()
    c = -np.ones(n, dtype=np.float64)          # minimise –Σ g_i  == maximise Σ g_i
    b = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = _linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.
    Returns a plain Python list that will be fed to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # keep a safety margin

    # ----------------------------------------------------------------------

        base_n = 4000

    n = max(800, min(8000, base_n))
    print(f"[{time.time():.1f}] chosen length = {n}")

    # ----------------------------------------------------------------------
    # 1️⃣  QUICK SEARCH IN A ONE‑PARAMETER FAMILY (beta(α,α))
    alphas = np.linspace(0.35, 1.75, 22)  # a fairly dense grid

    best_score = float("inf")

        cand = _project_onto_simplex(cand, s=1.0)
        peak = _fft_self_convolve(cand).max()
        score = 2.0 * n * peak
        if score < best_score:
            best_score = score
            best_alpha = a
    print(f"[{time.time():.1f}] best α from grid = {best_alpha:.4f}, score≈{best_score:.6f}")

    # Initialise from the best beta‑law; blend with the historic solution if available
    a0 = _beta_distribution(n, best_alpha)
    a0 = _project_onto_simplex(a0, s=1.0)
    if base_seq is not None:
        a_prev = _interpolate_to_length(base_seq, n)
        a0 = 0.5 * a0 + 0.5 * a_prev
        a0 = _project_onto_simplex(a0, s=1.0)

    # ----------------------------------------------------------------------
    # 2️⃣  ADAM OPTIMISATION OF THE SMOOTH MAX SURROGATE
    phase1_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.75)
    a_opt, _ = _adam_optimize(a0, phase1_deadline)

    # ----------------------------------------------------------------------
    # 3️⃣  OPTIONAL LINEAR‑PROGRAMMING “FRANK‑WOLFE” IMPROVEMENT
    if _SCIPY_AVAILABLE and n <= 2500:

        for lp_iter in range(5):
            if time.time() >= deadline - 0.2:
                break
            M = _fft_self_convolve(a_opt).max()
            g = _lp_step(a_opt, M)
            if g is None:

            s = g.sum()
            if s <= 1.0 + 1e-8:
                break
            g = g / s                     # bring to the simplex
            a_opt = 0.5 * a_opt + 0.5 * g  # convex blend
            a_opt = _project_onto_simplex(a_opt, s=1.0)

    # ----------------------------------------------------------------------
    # 4️⃣  EXACT‑MAX SUB‑GRADIENT REFINEMENT
    a_final = _refine_max(a_opt, deadline)

    # ----------------------------------------------------------------------
    # Return the best sequence (as plain Python list)
    final_list = a_final.tolist()
    try:
        # final sanity check – will be called by the harness anyway
        final_score = evaluate_sequence(final_list)
        print(f"Final evaluated score = {final_score:.6f}")
    except Exception as exc:
        print("evaluate_sequence raised during final check:", exc)

    return final_list
