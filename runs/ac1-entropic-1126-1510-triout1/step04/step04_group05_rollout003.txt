"""
Sequence optimizer for the self‑convolution bound.

The algorithm proceeds in three phases (all time‑bounded):
1. Simulated annealing on the simplex – cheap random mass swaps that
   directly minimise the maximal convolution value (the true objective).
2. Continuation on a smooth log‑sum‑exp surrogate of the max, solved
   by L‑BFGS‑B with a long cooling schedule (many µ‑levels and more
   inner iterations than the reference implementation).
3. A long entropic‑gradient (Adam on logits) refinement on the *hard*
   max (sub‑gradient) using the exact active‑set gradient.
   This stage runs as many iterations as the remaining budget permits.

All convolutions are performed via FFT for O(n log n) speed.
The sequence is always kept on the L¹‑simplex Σa = sqrt(2·n) (scale‑invariant
objective).  The best sequence found is returned as a plain list of floats.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (for FFT convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear convolution a * a (length 2·n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0          # numerical noise
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a."""
    conv = _full_convolution(a, size)
    return float(conv.max())


def _subgrad_max_convolution(a: np.ndarray, conv: np.ndarray, size: int) -> np.ndarray:
    """Sub‑gradient of max_k (a*a)[k] w.r.t. a (ordered‑pair version)."""
    max_c = conv.max()
    # treat all entries within 1e‑12 of the maximum as active
    mask = np.isclose(conv, max_c, atol=1e-12)
    cnt = mask.sum()
    w = np.zeros_like(conv, dtype=float)
    if cnt > 0:
        w[mask] = 1.0 / cnt
    # convolution w * a_rev (ordered‑pair version)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * len(a) - 1]
    g_a = 2.0 * conv_wr[len(a) - 1 : len(a) - 1 + len(a)]
    return g_a


# ----------------------------------------------------------------------
def _compute_smooth_max(y: np.ndarray, mu: float, n: int, S: float):
    """Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. logits y."""
    p = _softmax(y)              # Σp = 1, p_i ≥ 0
    a = S * p                    # Σa = S
    conv = np.convolve(a, a)     # length 2·n‑1 (numpy uses O(n²) but n is modest here)
    max_c = conv.max()
    if mu > 0.0:
        # log‑sum‑exp around max_c (stable)
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z                      # soft max weights
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt:
            w[mask] = 1.0 / cnt
        f_val = max_c
    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]
    # chain rule to y
    inner = np.dot(g_a, p)               # scalar = Σ p_j g_a_j
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_smooth_max(y, self.mu, self.n, self.S)
        return f, g

def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Very cheap SA on the simplex – move a random fraction of mass."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max

    # temperature schedule
    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-6
    it = 0
    while time.time() - start_time < budget:
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = np.random.rand() * a[i] * 0.3   # up to 30 % of source mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = False
        if diff < 0.0:
            accept = True
        elif T > 0.0 and np.random.rand() < math.exp(-diff / T):
            accept = True

        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1
    return best_a, best_score


def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """Continuation on the smooth‑max surrogate with a finer µ‑schedule."""
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    # initialise µ from current max‑convolution
    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[L‑BFGS] start score = {best_score:.6f}")

    mu_factor = 0.7          # slower cooling than the reference
    mu_min = 1e-8
    maxiter_per_mu = 500    # a bit more work per µ

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


def _exp_gradient_refine(y_start: np.ndarray, S: float,
                        budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam on the logits y (a = S·softmax(y)) using the exact hard‑max sub‑gradient.
    Runs as many iterations as the time budget permits.
    """
    n = y_start.shape[0]
    size = _fft_size(n)

    y = y_start.copy()
    best_y = y.copy()
    p = _softmax(y)
    best_a = S * p
    best_score = evaluate_sequence(best_a.tolist())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.2          # base learning rate

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    while time.time() - start_time < budget:
        t += 1
        p = _softmax(y)
        a = S * p

        # full convolution (via FFT, faster than np.convolve for n>~500)
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt:
            w[mask] = 1.0 / cnt

        # sub‑gradient w.r.t. a
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a_rev, n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        inner = np.dot(p, g_a)            # scalar = Σ p_j g_a_j
        grad_y = S * p * (g_a - inner)    # shape (n,)

        # Adam update on y
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # periodic checking (every 20 iterations)
        if t % 20 == 0:
            a_cand = S * _softmax(y)
            sc = evaluate_sequence(a_cand.tolist())
            if sc < best_score:
                best_score = sc
                best_y = y.copy()
                print(f"[exp‑grad improve] iter {t:,} score = {best_score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            break

    best_a = S * _softmax(best_y)
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point.  Searches for a low‑score step‑function within the
    allotted time and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000          # a reasonable default if no reference is available
        have_ref = False

    S = math.sqrt(2.0 * n)            # ℓ¹‑norm; ensures evaluate ≈ max‑conv

    # --------------------------------------------------------------
    # Initial sequence
    if have_ref:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = _project_to_simplex(a0, S)
    else:
        a0 = np.full(n, S / n, dtype=float)   # uniform start

    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init]   score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Simulated annealing
    remaining = deadline - time.time()
    if remaining > 30.0:
        sa_time = remaining * 0.55
        print(f"[SA] running for {sa_time:.1f}s")
        best_a, best_score = _simulated_annealing(best_a, S, sa_time, time.time())
        print(f"[SA]    best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – L‑BFGS on smooth max
    remaining = deadline - time.time()
    if remaining > 15.0:
        print(f"[L‑BFGS] refining for {remaining:.1f}s")
        best_a, best_score = _l_bfgs_refine(best_a, S, remaining, time.time())
        print(f"[L‑BFGS] best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Entropic Adam (logits) on the true max
    remaining = deadline - time.time()
    if remaining > 5.0:
        # initialise logits from the best a we have
        p0 = best_a / S
        eps = 1e-12
        y0 = np.log(p0 + eps).astype(np.float64)
        print(f"[exp‑grad] polishing for {remaining:.1f}s")
        best_a, best_score = _exp_gradient_refine(y0, S, remaining, time.time())
        print(f"[exp‑grad] best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Safety fallback (should never happen)
    if best_score == float("inf"):
        best_a = S * np.ones(n) / n
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
