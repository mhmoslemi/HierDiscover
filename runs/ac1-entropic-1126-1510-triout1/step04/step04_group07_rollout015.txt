"""
Optimise a non‑negative step‑function (list of heights) for the
evaluate_sequence() score.

The algorithm works in three phases:

1.  Smooth‑max optimisation.
    ‑ Initialise a set of free parameters `y` (soft‑max logits) from
      several diverse guesses (uniform, exponential, reversed‑exponential,
      random Gaussian, and the provided reference if any).
    ‑ Minimise the smooth‑max surrogate
          f_μ(y) = max(conv) + μ·log Σ exp((conv‑max)/μ)
      using Adam‑accelerated gradient descent.
    ‑ The temperature μ is slowly reduced (geometric cooling) until it
      reaches a tiny value, so the surrogate approaches the true
      hard‑max objective.

2.  Hard‑max projected Adam.
    ‑ Starting from the best point of phase‑1, perform projected Adam
      directly on the true objective (max of the auto‑convolution).
    ‑ The gradient of max(conv) w.r.t. the heights is obtained analytically.
    ‑ After each step the vector is projected onto the simplex
      {a≥0, Σa = √(2n)}.

3.  Random local perturbations.
    ‑ With the remaining time, repeatedly perturb a handful of entries,
      re‑project onto the simplex and keep the improvement.
    ‑ This cheap stochastic search often discovers tiny extra gains.

The whole routine respects the supplied time budget (default 1000 s) and
returns the best sequence found.  The implementation uses only NumPy and
SciPy and avoids any closures or lambdas, as required.
"""

import time
import math
import numpy as np
import scipy.linalg  # only for the LP variant (not used here, keep import safe)


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(
    y: np.ndarray, mu: float, n: int, S: float
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max surrogate value and gradient w.r.t. the free parameters y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained parameters (length n).
    mu : float
        Temperature (>0) – larger ⇒ smoother.
    n : int
        Length of the step‑function.
    S : float
        Desired ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : np.ndarray
        Gradient ∂f/∂y (shape (n,)).
    """
    # probabilities on the simplex and actual heights
    p = _softmax(y)          # Σ p_i = 1, p_i ≥ 0
    a = S * p                # Σ a_i = S

    # full convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weight vector w_k ∝ exp((conv_k - max)/μ)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # guard overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:  # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a
    # g_a[i] = 2 * Σ_k w_k * a_{k‑i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]      # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
def _hard_max_grad(a: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Gradient of the true (hard) max‑convolution objective.
    Returns (max_convolution, gradient wrt a).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = np.count_nonzero(mask)
    w = np.zeros_like(conv, dtype=float)
    w[mask] = 1.0 / cnt
    grad_a = 2.0 * np.convolve(w, a[::-1])[n - 1 : n - 1 + n]
    return float(max_c), grad_a.astype(np.float64)


# ----------------------------------------------------------------------
def _score_from_a(a: np.ndarray) -> float:
    """Exact score as used by evaluate_sequence (without clamping)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_c / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluate_sequence() score.  The routine respects the given time budget
    (seconds) and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size
    try:
        # the reference from the previous run, if it exists
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        # a comfortable length; larger length gives more freedom
        n = 3400
        have_ref = False

    S = math.sqrt(2.0 * n)               # target ℓ¹‑norm (makes score = max conv)

    # bookkeeping
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # --------------------------------------------------------------
    def _maybe_update(a_vec: np.ndarray) -> None:
        """Update global incumbent if the new sequence is better."""
        nonlocal best_score, best_a
        sc = _score_from_a(a_vec)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # build a collection of initial free‑parameter vectors y
    init_ys = []

    # 0 – uniform (y = 0)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – decreasing exponential (large on the left)
    alpha = rng.uniform(0.001, 0.008) * n / 1000.0
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # 2 – increasing exponential (large on the right)
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 3 – random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 4 – reference if available
    if have_ref:
        a_ref = np.asarray(height_sequence_1, dtype=float)
        a_ref = _project_to_simplex(a_ref, S)
        eps = 1e-12
        init_ys.append(np.log(a_ref + eps))

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    mu_factor = 0.85          # slower cooling than the baseline
    mu_min = 1e-7
    steps_per_mu = 30         # after this many updates we cool μ

    # --------------------------------------------------------------
    # loop over all initial guesses
    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{len(init_ys)}] initialising")
        y = y0.copy()
        m = np.zeros_like(y)
        v = np.zeros_like(y)

        # initial temperature: max‑convolution of the current point
        a_cur = S * _softmax(y)
        mu = max(1.0, np.max(np.convolve(a_cur, a_cur)))
        step_counter = 1

        # -------- smooth‑max phase (Adam on y) --------------------
        while mu > mu_min and time.time() < deadline:
            f_val, grad_y = _compute_f_and_grad(y, mu, n, S)

            # Adam update on y (unconstrained)
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** step_counter)
            v_hat = v / (1.0 - beta2 ** step_counter)

            # learning‑rate schedule – slightly decaying
            lr = 0.12 / (1.0 + 0.001 * step_counter)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # project via soft‑max, evaluate
            a = S * _softmax(y)
            _maybe_update(a)

            # cooling
            if step_counter % steps_per_mu == 0:
                mu *= mu_factor
                mu = max(mu, mu_min)

            step_counter += 1

        # -------- hard‑max phase (projected Adam on a) ----------
        # start from the best a obtained in the smooth phase
        if best_a.sum() == 0:          # no improvement yet
            a = S * _softmax(y)
        else:
            a = best_a.copy()
        m_a = np.zeros_like(a)
        v_a = np.zeros_like(a)

        hard_step = 1
        # a simple schedule: 5000 iterations or until time runs out
        while time.time() < deadline and hard_step <= 5000:
            max_c, grad_a = _hard_max_grad(a)

            # Adam step on a (in the ambient space)
            m_a = beta1 * m_a + (1.0 - beta1) * grad_a
            v_a = beta2 * v_a + (1.0 - beta2) * (grad_a ** 2)
            mhat_a = m_a / (1.0 - beta1 ** hard_step)
            vhat_a = v_a / (1.0 - beta2 ** hard_step)

            # learning‑rate decays with √t
            lr_a = (0.6 * S / n) / (1.0 + 5e-5 * hard_step)
            a_new = a - lr_a * mhat_a / (np.sqrt(vhat_a) + eps_adam)

            # project onto the simplex (keeps Σa = S and a≥0)
            a = _project_to_simplex(a_new, S)

            _maybe_update(a)
            hard_step += 1

    # --------------------------------------------------------------
    # random local perturbations – cheap cheap‑search
    if time.time() < deadline:
        print("[random search] entering cheap perturbation phase")
        while time.time() < deadline:
            # pick a random block size (2‑8 indices)
            k = rng.integers(2, 9)
            idx = rng.choice(n, size=k, replace=False)

            # generate a zero‑sum random shift
            delta = rng.normal(scale=S * 0.001, size=k)
            delta -= delta.mean()                     # keep total sum unchanged

            a_candidate = best_a.copy()
            a_candidate[idx] += delta
            a_candidate = _project_to_simplex(a_candidate, S)

            _maybe_update(a_candidate)

    # --------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
