   constraints via Euclidean projection.  Temperature is cooled very
   slowly and occasional random kicks help escape plateaus.

3. **Random‑block sweep** – remaining time is spent generating cheap
   random‑block candidates; any improvement updates the global best.

All heavy work is timed; the function returns the best sequence found
before the 1000 s deadline.
"""

import math
import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (full length)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def quick_score(a: np.ndarray, n: int, fft_len_self: int) -> float:
    """Fast evaluation of the objective (no security checks)."""
    conv = _self_convolution(a, fft_len_self)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def _weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    (w must be non‑negative and sum to 1.)
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len_grad)
    fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv = np.fft.irfft(fw * fa_rev, n=fft_len_grad)
    start = len(w) - len(a)                     # = n‑1
    grad = 2.0 * conv[start : start + len(a)]
    return grad


def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    fft_len_self: int,
    fft_len_grad: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimisation of a smooth max‑convolution objective."""
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, fft_len_self)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base learning rate scales with target sum / n
    eta0 = target_sum * 0.4 / n

    # initialise temperature from current convolution spread
    conv = _self_convolution(a, fft_len_self)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, fft_len_self)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights (smooth approx of max) -----
        shift = conv - conv.max()                # ≤ 0
        exp_arg = shift / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)       # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smoothed objective -----
        grad = _weighted_gradient(a, w, fft_len_grad)

        # ----- Adam update -----
        eta = eta0 / (1.0 + it * 1e-5)            # mild decay
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random kick -----
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9997               # cool temperature slowly
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# families of analytic shapes -----------------------------------------

def _family_beta(params: np.ndarray, idx: np.ndarray, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^α * (n-i)^β."""
    α, β = params
    return (idx + 1) ** α * (n - idx) ** β


def _family_double_exp(params: np.ndarray, idx: np.ndarray, n: int) -> np.ndarray:
    """a_i = exp(-ℓ·i) + exp(-r·(n‑1‑i))."""
    ℓ, r = params
    return np.exp(-ℓ * idx) + np.exp(-r * (n - 1 - idx))


def _family_power(params: np.ndarray, idx: np.ndarray, n: int) -> np.ndarray:
    """a_i = 1/(i+β)^p + 1/((n‑1‑i)+β)^p."""
    β, p = params
    return 1.0 / (idx + β + 1.0) ** p + 1.0 / ((n - 1 - idx) + β + 1.0) ** p


def _family_gaussian(params: np.ndarray, idx: np.ndarray, n: int) -> np.ndarray:
    """a_i = exp(-0.5·((i‑c)/σ)^2)."""
    σ, c = params
    return np.exp(-0.5 * ((idx - c) / σ) ** 2)


def _random_block_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Few constant blocks (2‑7) with random heights."""
    nb = rng.integers(2, 8)                 # number of blocks
    cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
    blocks = np.split(np.arange(n), cuts)
    vals = rng.random(nb) + 0.1
    cand = np.empty(n, dtype=float)
    for blk, v in zip(blocks, vals):
        cand[blk] = v
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The function respects a 1000 s time budget
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)          # normalisation used in the paper
    fft_len_self = _next_pow2(2 * n - 1)     # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)     # for weighted gradient

    idx = np.arange(n)

    # ------------------------------------------------------------------
    # Storage for the best K raw candidates
    K = 12
    best_raw: list[tuple[float, np.ndarray]] = []   # (score, array)

    def _push_candidate(arr: np.ndarray) -> None:
        """Project, score and keep top‑K raw candidates."""
        arr = _project_to_simplex(arr, target_sum)
        sc = quick_score(arr, n, fft_len_self)
        if len(best_raw) < K:
            best_raw.append((sc, arr.copy()))
            best_raw.sort(key=lambda x: x[0])
        elif sc < best_raw[-1][0]:
            best_raw[-1] = (sc, arr.copy())
            best_raw.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 1. PARAMETRIC EXPLORATION ----------------------------------------
    # Uniform baseline
    _push_candidate(np.ones(n))

    # ---- beta‑shaped ---------------------------------------------------
    bounds_beta = [(0.1, 5.0), (0.1, 5.0)]
    def beta_score(p):
        cand = _family_beta(p, idx, n)
        cand = _project_to_simplex(cand, target_sum)
        return quick_score(cand, n, fft_len_self)
    res_beta = differential_evolution(beta_score, bounds_beta,
                                      maxiter=25, popsize=12,
                                      polish=False, seed=rng.integers(0, 2**32))
    _push_candidate(_family_beta(res_beta.x, idx, n))

    # ---- double exponential --------------------------------------------
    bounds_de = [(0.0002, 0.02), (0.0002, 0.02)]
    def de_score(p):
        cand = _family_double_exp(p, idx, n)
        cand = _project_to_simplex(cand, target_sum)
        return quick_score(cand, n, fft_len_self)
    res_de = differential_evolution(de_score, bounds_de,
                                    maxiter=25, popsize=12,
                                    polish=False, seed=rng.integers(0, 2**32))
    _push_candidate(_family_double_exp(res_de.x, idx, n))

    # ---- power‑law -----------------------------------------------------
    bounds_pow = [(0.0, 12.0), (0.2, 3.0)]   # β shift, exponent p
    def pow_score(p):
        cand = _family_power(p, idx, n)
        cand = _project_to_simplex(cand, target_sum)
        return quick_score(cand, n, fft_len_self)
    res_pow = differential_evolution(pow_score, bounds_pow,
                                     maxiter=25, popsize=12,
                                     polish=False, seed=rng.integers(0, 2**32))
    _push_candidate(_family_power(res_pow.x, idx, n))

    # ---- Gaussian -------------------------------------------------------
    sigma_min = 0.05 * n
    sigma_max = 0.30 * n
    centre_min = 0.0
    centre_max = n - 1
    bounds_gauss = [(sigma_min, sigma_max), (centre_min, centre_max)]
    def gauss_score(p):
        cand = _family_gaussian(p, idx, n)
        cand = _project_to_simplex(cand, target_sum)
        return quick_score(cand, n, fft_len_self)
    res_gauss = differential_evolution(gauss_score, bounds_gauss,
                                       maxiter=25, popsize=12,
                                       polish=False, seed=rng.integers(0, 2**32))
    _push_candidate(_family_gaussian(res_gauss.x, idx, n))

    # ---- random block candidates (quick cheap sweep) --------------------
    for _ in range(200):
        cand = _random_block_candidate(n, rng)
        _push_candidate(cand)

    # ---- reference sequence if available --------------------------------
    if have_ref:
        cand_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        _push_candidate(cand_ref)

    print(f"[exploration] best raw score = {best_raw[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. REFINEMENT (Adam descent) ---------------------------------------
    # Keep the globally best sequence seen so far
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    # Refine only the top few raw candidates (balance time vs quality)
    for rank, (raw_sc, a0) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        max_iters = int(min(250_000, remaining * 1500))
        print(f"[refine] candidate #{rank + 1} (raw {raw_sc:.6f}) – max_iters {max_iters}")
        a_refined, sc_refined = _refine_candidate_adam(
            a0, deadline, n, fft_len_self, fft_len_grad,
            target_sum, max_iters=max_iters
        )
        if sc_refined < global_best_score:
            global_best_score = sc_refined
            global_best_a = a_refined.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK SWEEP -----------------------------------------
    while time.time() < deadline:
        cand = _random_block_candidate(n, rng)
        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand, n, fft_len_self)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")
        if time.time() + 0.02 > deadline:
            break

    # ------------------------------------------------------------------
    # 4. OPTIONAL POST‑SMOOTHING (low‑pass) ------------------------------
    # A very light moving‑average can sometimes lower the peak.
    kernel = np.array([0.25, 0.5, 0.25])
    smoothed = np.convolve(global_best_a, kernel, mode='same')
    smoothed = _project_to_simplex(smoothed, target_sum)
    sm_sc = quick_score(smoothed, n, fft_len_self)
    if sm_sc < global_best_score:
        global_best_score = sm_sc
        global_best_a = smoothed
        print(f"[smoothing] improved to {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
