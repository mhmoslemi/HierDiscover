"""
Propose a non‑negative step function (list of heights) that minimizes the
evaluation score 2·n·max(conv)/(sum a)² (equivalently the peak of the
self‑convolution).  The method combines:

* a very broad random sampling of many families (Dirichlet, power‑law,
  exponential, block‑wise, random‑phase “flat‑spectrum”),
* a simple Adam‑style sub‑gradient descent on the hard L∞‑objective for
  the K best raw candidates,
* a long simulated‑annealing phase that moves mass between entries and
  swaps values, always keeping the ℓ₁‑norm fixed.

All operations respect the 1000 s time budget and the simplex constraint
Σ a = √(2 n).  The returned list can be fed directly to ``evaluate_sequence``.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("target sum must be non‑negative")
    n = v.shape[0]
    # sort v decreasingly
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero, put everything on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (good for FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a (length = 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Peak (maximum value) of the self‑convolution of a."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)_k   at the current maximal index ``k_max``.
    It returns a vector g with the property that g·δ is a sub‑gradient
    direction for a perturbation δ.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        # overlap a[0..k_max] with reversed a[k_max::-1]
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        # k_max ≥ n
        start = k_max - n + 1                     # first index that participates
        length = 2 * n - 1 - k_max                # size of the overlapping window
        g[start:start + length] = 2.0 * a[:length][::-1]
    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent on the hard max‑convolution.
    Returns the best sequence seen (lowest peak) and its peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base step size (order of magnitude of one entry)
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full convolution and current worst index
        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[refine] it {it:,} new best = {best_score:.6f}")

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / math.sqrt(it)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # keep a on the simplex (sum = target_sum, non‑negative)
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _random_phase_sequence(n: int, size_fft: int) -> np.ndarray:
    """
    Generate a non‑negative sequence whose DFT has (approximately) flat magnitude.
    Random unit‑modulus phases are drawn, an inverse FFT is taken,
    and the absolute value is kept.  The result is projected onto the simplex.
    """
    # generate random phases for the positive frequencies (including DC)
    phases = np.exp(1j * 2.0 * np.pi * np.random.rand(size_fft // 2 + 1))

    # enforce Hermitian symmetry to obtain a real sequence
    X = np.zeros(size_fft, dtype=complex)
    X[0] = 0.0                    # DC can be zero, it does not harm
    X[1:size_fft // 2] = phases[1:size_fft // 2]
    if size_fft % 2 == 0:
        X[size_fft // 2] = phases[size_fft // 2].real   # Nyquist must be real
    X[size_fft // 2 + 1:] = np.conj(X[1:size_fft // 2][::-1])

    seq = np.fft.ifft(X).real
    seq = np.abs(seq[:n])        # keep the first n entries, enforce non‑negativity
    return seq

        vals = np.random.rand(nb) + 0.01
        a = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            a[blk] = v
        add_to_pool(a)

    # Random “flat‑spectrum” sequences generated via random phases
    for _ in range(150):
        a = _random_phase_sequence(n, size_fft)
        add_to_pool(a)

    # Keep the K best raw candidates (small K keeps refinement cheap)
    K = 30
    raw_pool.sort(key=lambda x: x[0])
    top_raw = raw_pool[:K]
    print(f"[sampling] best raw score = {top_raw[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2. Refinement of the best raw candidates (Adam sub‑gradient)
    # ------------------------------------------------------------------
    best_a = top_raw[0][1].copy()
    best_score = top_raw[0][0]

    for rank, (raw_sc, a0) in enumerate(top_raw):
        if time.time() >= deadline:
            break
        print(f"[refine] candidate #{rank+1}/{len(top_raw)} start score {raw_sc:.6f}")
        # allocate a modest budget per candidate; we simply rely on the global deadline
        remaining_time = deadline - time.time()
        max_iters = int(min(300_000, remaining_time * 1500))
        refined_a, refined_sc = _refine_subgrad(
            a0, size_fft, target_sum, deadline, max_iters
        )
        if refined_sc < best_score:
            best_score = refined_sc
            best_a = refined_a.copy()
            print(f"[global] after refinement new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Long simulated‑annealing phase on the current best candidate
    # ------------------------------------------------------------------
    print("[annealing] start")
    a_cur = best_a.copy()
    cur_score = best_score

    # annealing schedule
    start_temp = cur_score * 0.08   # a few percent of the current peak
    end_temp = cur_score * 0.001
    total_time = deadline - time.time()

    # Pre‑calculate a few constants
    epsilon = 1e-12
    n_float = float(n)

    while time.time() < deadline:
        # progress fraction for temperature decay
        elapsed = time.time() - start_time
        frac = min(elapsed / (budget_s - 0.5), 1.0)
        temp = start_temp * (1.0 - frac) + end_temp * frac

        # decide between a swap or a mass transfer
        if np.random.rand() < 0.55:
            # ----- swap two positions (no change of ℓ₁ norm) -----
            i, j = np.random.choice(n, 2, replace=False)
            a_new = a_cur.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            # ----- move a small amount of mass from i to j -----
            # pick a source with non‑zero mass
            src_candidates = np.where(a_cur > epsilon)[0]
            if src_candidates.size == 0:
                continue
            i = np.random.choice(src_candidates)
            j = np.random.choice(n)
            if i == j:
                continue
            max_transfer = min(a_cur[i], 1000.0 - a_cur[j])
            if max_transfer <= 0:
                continue
            # transfer a random fraction (up to ~30 % of the source entry)
            delta = max_transfer * np.random.rand() * 0.3
            a_new = a_cur.copy()
            a_new[i] -= delta
            a_new[j] += delta

        # evaluate the new candidate
        sc_new = _max_convolution(a_new, size_fft)

        # Metropolis acceptance
        if sc_new < cur_score or np.random.rand() < math.exp(-(sc_new - cur_score) / (temp + epsilon)):
            a_cur = a_new
            cur_score = sc_new
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_cur.copy()
                print(f"[anneal] improved to {best_score:.6f}")

        # occasional status line
        if int(elapsed) % 30 == 0:
            print(f"[anneal] time {elapsed:.0f}s best {best_score:.6f}")

    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
