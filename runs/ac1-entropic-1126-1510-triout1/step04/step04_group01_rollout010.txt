def _local_search(a_init, deadline):
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    b = _fft_self_convolve(a)
    best_max = b.max()
    best_score = 2 * n * best_max
    rng = np.random.default_rng()
    iter = 0
    # schedule of step size factor: start high, reduce
    while time.time() < deadline:
        iter += 1
        # decreasing step factor: from 0.5 to 0.001 over iterations
        factor = max(0.001, 0.5 * (deadline - time.time()) / (deadline - start_time))
        # propose move:
        i = rng.integers(n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        delta = min(a[i], factor * a[i] * rng.random())
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # maybe ensure nonnegativity small tolerance
        # no need to renormalize sum exactly near 1, but small error.
        # compute new max
        b_new = _fft_self_convolve(a_new)
        new_max = b_new.max()
        new_score = 2 * n * new_max
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_max = new_max
            best_a = a_new
            a = a_new
            # optional recompute b for next iteration
        else:
            # maybe sometimes accept random with probability decreasing?
            pass

    return best_a

        # choose indices
        i = rng.integers(n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # delta = fraction of a[i]
        max_delta = a[i]
        # choose delta proportion up to factor
        # factor decays: start 0.5 decreasing to 0.001
        remaining = deadline - time.time()
        total = deadline - start_time
        factor = max(0.001, 0.5 * remaining / total)

            best_score = score
            best_a = a_new.copy()
            a = a_new  # accept move
            best_max = max_b
        # else maybe accept occasionally?
    return best_a

"""
Propose a low‑score step function.

The search proceeds in three stages for each candidate length:

1. **Adam on a smooth soft‑max surrogate** – a fast first‑order method that
   approximately minimises the ℓ∞ norm of the self‑convolution while staying
   on the probability simplex.

2. **Exact sub‑gradient descent** – a short projected sub‑gradient descent on
   the true maximum (uniform weight on the active set).  A larger number of
   iterations than the historic code is used.

3. **Greedy random local refinement** – repeatedly moves a tiny amount of mass
   from one coordinate to another and accepts the move only if the exact peak
   of the self‑convolution decreases.  This simple “hill‑climbing’’ phase
   often squeezes out the last bits of improvement.

The routine tries several lengths (including interpolations of the previous
best construction) and a handful of initialisations (uniform, arcsine,
Beta‑shaped, random).  The best sequence found (according to the black‑box
`evaluate_sequence`) is returned.  The whole search respects a user‑supplied
budget (default 1000 s) and never exceeds it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=s}."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) distribution on n points (α>0)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n  # stay away from 0 and 1
    # avoid overflow for very small α by clipping
    eps = np.finfo(float).eps
    x = np.clip(x, eps, 1 - eps)
    vals = np.power(x, alpha - 1) * np.power(1 - x, alpha - 1)
    if np.any(np.isnan(vals)) or np.all(vals == 0):
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _interpolate_to_length(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize (linearly interpolate) a non‑negative sequence to a new length."""
    old_n = len(old_seq)
    if old_n == new_n:
        return old_seq.astype(float).copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_vals = np.interp(x_new, x_old, old_seq)
    new_vals = np.maximum(0.0, new_vals)
    s = new_vals.sum()
    if s == 0:
        new_vals = np.full(new_n, 1.0 / new_n)
    else:
        new_vals /= s
    return new_vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length 2·len(a)‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, alpha: float):
    """
    Gradient of the soft‑max surrogate
          Lα(a) = (1/α)·log Σ_k exp(α·(a*a)_k)
    Returns (grad, cur_max) where grad has the same shape as a.
    """
    b = _fft_self_convolve(a)
    cur_max = b.max()
    # soften the max – numerically stable
    exp_term = np.exp(alpha * (b - cur_max))
    w = exp_term / exp_term.sum()                     # soft‑max weights
    a_rev = a[::-1]
    conv_rev = _fft_convolve(a_rev, w)               # length = 3n‑2
    grad = 2.0 * conv_rev[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, cur_max


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   init_alpha: float = 0.5,
                   alpha_factor: float = 1.01,
                   alpha_max: float = 5000.0,
                   lr0: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam optimisation of the soft‑max surrogate.
    Returns (best_a, best_peak) where best_peak is the true max of a*a.
    """
    a = a0.copy()
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    alpha = init_alpha

    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()

    while time.time() < deadline:
        t += 1
        grad, cur_max = _smooth_max_gradient(a, alpha)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # increase temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # occasional jitter to avoid stagnation
        if t % 400 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 1e-6
            a = _project_onto_simplex(a, s=1.0)

    return best_a, best_peak


# ----------------------------------------------------------------------
def _max_subgradient_step(a: np.ndarray):
    """Compute a sub‑gradient of the true ℓ∞ objective."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    tol = max(1e-12, cur_max * 1e-12)
    active = np.where(b >= cur_max - tol)[0]
    w = np.zeros_like(b)
    if active.size:
        w[active] = 1.0 / active.size
    a_rev = a[::-1]
    grad = 2.0 * _fft_convolve(a_rev, w)[n - 1 : n - 1 + n]
    return grad, cur_max


# ----------------------------------------------------------------------
def _refine_max(a0: np.ndarray, deadline: float) -> np.ndarray:
    """
    Projected sub‑gradient descent on the exact ℓ∞ objective.
    The step size decays as 0.03/√t (a bit larger than the historic 0.02).
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_peak = _fft_self_convolve(a).max()
    it = 0
    while time.time() < deadline:
        it += 1
        grad, cur_max = _max_subgradient_step(a)
        step = 0.03 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)

        if cur_max < best_peak - 1e-15:
            best_peak = cur_max
            best_a = a.copy()

        # occasional jitter
        if it % 500 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-7
            a = _project_onto_simplex(a, s=1.0)
    return best_a


# ----------------------------------------------------------------------
def _random_local_search(a_init: np.ndarray, deadline: float) -> np.ndarray:
    """
    Simple greedy random walk on the simplex.
    At each iteration a tiny amount of mass is moved from one coordinate to another.
    The move is kept only if the exact ℓ∞ ‑value improves.
    """
    a = a_init.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_max = _fft_self_convolve(a).max()
    best_score = 2.0 * n * best_max
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        # pick a source with some mass
        i = rng.integers(n)
        if a[i] <= 1e-14:
            continue
        # pick a different target
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # fraction of source mass to move – decays with remaining time
        remaining = deadline - time.time()
        total = deadline - start
        frac = max(0.001, 0.4 * remaining / total)   # start aggressive, fade out
        delta = a[i] * frac * rng.random()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_max = _fft_self_convolve(a_new).max()
        new_score = 2.0 * n * new_max

        if new_score < best_score - 1e-15:
            best_score = new_score
            best_max = new_max
            best_a = a_new.copy()
            a = a_new      # accept the move
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.
    The routine respects the provided time budget (default 1000 s)
    and returns a plain Python list suitable for `evaluate_sequence`.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # base sequence (if a previous solution is available)
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
    except Exception:
        base_seq = None

    base_n = len(base_seq) if base_seq is not None else 8000

    # explore a handful of lengths (including a larger one for extra smoothing)
    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in (0.5, 0.8, 1.0, 1.2, 1.4, 2.0)
            if int(base_n * f) >= 500
        )
    )
    if not lengths:
        lengths = [8000]

    overall_best_seq = None
    overall_best_score = float("inf")

    # split the global budget roughly equally among the lengths
    time_per_length = max(2.0, (deadline - start_time) / len(lengths))

    for n in lengths:
        length_deadline = min(deadline, time.time() + time_per_length)
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        init_strategies = [
            "prev", "arcsine", "beta0.5", "beta1.0", "beta2.0", "uniform", "random"
        ]

        for kind in init_strategies:
            if time.time() >= length_deadline:
                break

            # --------------------------------------------------------------
            # initialise a0 on the simplex
            if kind == "prev" and base_seq is not None:
                a0 = _interpolate_to_length(base_seq, n)
            elif kind == "arcsine":
                a0 = _arcsine_distribution(n)
            elif kind.startswith("beta"):
                try:
                    alpha = float(kind[4:])
                except Exception:
                    alpha = 0.5
                a0 = _beta_distribution(n, alpha)
            elif kind == "uniform":
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            elif kind == "random":
                a0 = np.random.rand(n)
                a0 /= a0.sum()
            else:  # fallback
                a0 = np.full(n, 1.0 / n, dtype=np.float64)

            a0 = _project_onto_simplex(a0, s=1.0)

            # --------------------------------------------------------------
            # Phase‑1 : Adam on the smooth surrogate (≈60 % of the per‑length budget)
            phase1_deadline = min(
                length_deadline,
                time.time() + 0.6 * (length_deadline - time.time()),
            )
            a_adam, _ = _adam_optimize(a0, phase1_deadline)

            # --------------------------------------------------------------
            # Phase‑2 : Exact sub‑gradient descent (≈30 % of the budget)
            phase2_deadline = min(
                length_deadline,
                time.time() + 0.3 * (length_deadline - time.time()),
            )
            a_sub = _refine_max(a_adam, phase2_deadline)

            # --------------------------------------------------------------
            # Phase‑3 : Greedy random local search (remaining time)
            a_loc = _random_local_search(a_sub, length_deadline)

            # --------------------------------------------------------------
            # fast estimate (exact max but cheap compared to the black‑box)
            cur_max = _fft_self_convolve(a_loc).max()
            fast_score = 2.0 * n * cur_max

            if fast_score < overall_best_score - 1e-12:
                overall_best_score = fast_score
                overall_best_seq = a_loc.copy()
                print(
                    f"[{time.time():.1f}] len={n} init={kind} fast_score={fast_score:.6f}"
                )
                # get the true black‑box score (the call is cheap)
                try:
                    true_score = evaluate_sequence(overall_best_seq.tolist())
                    if true_score < overall_best_score:
                        overall_best_score = true_score
                        overall_best_seq = np.asarray(overall_best_seq, dtype=np.float64)
                        print(
                            f"*** New true best: len={n} init={kind} score={true_score:.6f}"
                        )
                except Exception as exc:
                    print("evaluate_sequence raised:", exc)

            # early exit if the target is already met
            if overall_best_score <= 1.503:
                break
        if overall_best_score <= 1.503 or time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # fallback – should rarely happen
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())
        print("Fallback uniform used.")

    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # return a plain Python list as required by the harness
    return overall_best_seq.tolist()
