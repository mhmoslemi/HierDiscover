"""
Propose a non‑negative step‑function (list of heights) that minimizes the
self‑convolution peak evaluated by ``evaluate_sequence``.

Algorithm
---------
1. Determine the problem size ``n`` (use the reference sequence
   ``height_sequence_1`` if it exists, otherwise fall back to 3000).
   All candidates are projected onto the simplex
   Σa = √(2 n) – this does **not** affect the final score because
   ``evaluate_sequence`` is homogeneous of degree 0.

2. **Broad random sampling** – generate a large pool of raw candidates
   from several families (Dirichlet, power‑law, exponential, block‑wise
   constant, logistic).  Each candidate is projected to the simplex and
   its exact convolution peak is measured (via an FFT).  The best ``K``
   candidates (lowest peaks) are kept.

3. **Adam‑style sub‑gradient refinement** – for each of the top candidates
   run a projected Adam optimiser on the *hard* max‑convolution (sub‑gradient
   taken at the current worst shift).  The best sequence seen during the run
   is retained.  The number of iterations is chosen adaptively from the
   remaining time budget.

4. **Permutation annealing** – with the current global best we optimise the
   ordering of the values (the multiset is fixed).  Simulated annealing with
   random swaps explores the permutation space; swaps that lower the peak are
   always accepted, uphill moves are accepted with a temperature‑dependent
   probability.  The temperature is exponentially cooled.

5. Return the best sequence found before the 1000 s budget expires.
"""

# ----------------------------------------------------------------------

import time

def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a (length = 2 n‑1) using FFT."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k at the current maximal index k_max.
    Returns a vector g such that  g·δa  is a sub‑gradient of the max‑norm.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        # overlapping part a[0..k_max] with reversed a[k_max::-1]
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        # k_max ≥ n
        start = k_max - n + 1                 # first index that participates
        length = 2 * n - 1 - k_max            # number of terms in overlap
        g[start:] = 2.0 * a[:length][::-1]

    return g


def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent on the hard max‑convolution.
    Returns the best sequence seen and its peak value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Base step size (order of magnitude of one entry)
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # Full convolution and current worst index
        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[refine] it {it:,} new best = {best_score:.6f}")

        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # Learning‑rate schedule ~ 1/√t
        eta = eta0 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # Projection back onto the simplex
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


def _anneal_permutation(
    a: np.ndarray,
    size_fft: int,
    deadline: float,
    max_swaps: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing on the permutation of ``a``.
    Only swaps are performed (the ℓ₁‑norm stays unchanged).
    Returns the best permutation seen and its peak value.
    """
    n = a.shape[0]
    cur = a.copy()
    cur_score = _max_convolution(cur, size_fft)

    best = cur.copy()
    best_score = cur_score

    # Temperature schedule (exponential cooling)
    T_start = cur_score * 0.05 + 1e-3      # a small positive offset
    T_end = 1e-5

    swaps_done = 0
    start_time = time.time()

    while time.time() < deadline and swaps_done < max_swaps:
        # Linearised progress for temperature
        progress = swaps_done / max_swaps
        T = T_start * (T_end / T_start) ** progress

        i, j = np.random.choice(n, 2, replace=False)
        # perform swap
        cur[i], cur[j] = cur[j], cur[i]

        new_score = _max_convolution(cur, size_fft)

        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # metropolis criterion
            if np.random.rand() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
                # occasional progress report
                if swaps_done % 20000 == 0:
                    print(f"[anneal] swap {swaps_done:,} new best = {best_score:.6f}")
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]

        swaps_done += 1

    return best, best_score

        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        push_candidate(quick_score(cand), cand)

    print(f"[sampling] best raw peak = {best_raw[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 3. Refinement of the top candidates
    # ------------------------------------------------------------------
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    # We will only refine the top `R` candidates (to keep time reasonable)
    R = min(12, len(best_raw))
    for rank in range(R):
        if time.time() >= deadline:
            break
        raw_sc, a0 = best_raw[rank]
        print(f"[refine] candidate #{rank+1} (raw peak {raw_sc:.6f})")

        # Time left for this refinement
        remaining = deadline - time.time()
        if remaining < 5.0:
            break

        # Heuristic: allocate up to 120 k iterations, but cap by remaining time
        max_iters = int(min(150_000, remaining * 1800))

        refined_a, refined_sc = _refine_subgrad(
            a0, size_fft, target_sum, deadline, max_iters
        )
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Permutation annealing (ordering optimisation)
    # ------------------------------------------------------------------
    print("[anneal] start permutation optimisation")
    a_perm, perm_score = _anneal_permutation(
        global_best_a, size_fft, deadline, max_swaps=300_000
    )
    if perm_score < global_best_score:
        global_best_score = perm_score
        global_best_a = a_perm.copy()
        print(f"[global] new best after annealing = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence as a plain Python list
    # ------------------------------------------------------------------
    print(f"[final] best peak = {global_best_score:.6f}")
    return global_best_a.tolist()
