def _optimize_adagrad(a0, n, deadline):
    a = a0.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()
    # parameters
    eta0 = 1.0
    alpha = 1.0
    alpha_mul = 1.03
    alpha_max = 20000.0
    size_fft = 1 << ((3*n-2).bit_length())
    acc = np.zeros_like(a)  # sum of squared grads
    eps = 1e-12
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv and softmax gradient
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # compute softmax weights
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.zeros_like(w)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

        # gradient
        # compute conv of w (length 2n-1) with a_rev (len n)
        conv = _gradient_softmax(a, b, alpha, size_fft) # but that internally recomputes w; we can compute directly

        # [ revise: _gradient_softmax uses w internally; we can call it simply similarly]
        # Already _gradient_softmax uses b & alpha; we pass them.

        grad = _gradient_softmax(a, b, alpha, size_fft)

        # accumulate squares
        acc += grad**2
        # compute per-coordinate step
        step = eta0 / np.sqrt(acc + eps)
        # multiplicative update
        a = a * np.exp(-step * grad)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0/n)

            a /= s

        # occasional jitter

            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

def _expand_distribution(a, new_n):
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new
```

"""
propose_candidate
=================

A standalone optimiser that searches for a non‑negative step function
(sequence of heights) minimising the black‑box `evaluate_sequence`.

The algorithm combines several ideas that go beyond the reference
implementation:

* **AdaGrad‑style mirror descent** on the probability simplex – each
  coordinate gets its own diminishing step‑size, which typically yields
  faster convergence than a global 1/√t schedule.

* **Deterministic annealing** – the smooth log‑sum‑exp surrogate for the
  ℓ∞‑norm is used for *all* iterations.  The temperature `α` starts
  small (≈1) and is multiplied by a factor > 1 each step until a large
  upper bound (≈2·10⁴).  This gradually turns the soft‑max into an
  (almost) exact max, but the gradient stays well‑behaved throughout.

* **Multi‑scale search** – a good shape for a short distribution is
  interpolated to longer lengths, allowing the optimiser to start each
  new length with a sensible initial guess.

* **Multiple initialisations** – arcsine, beta‑skewed, uniform,
  Dirichlet‑sampled and the interpolated best‑so‑far vectors are tried
  for every length.

* **Final random‑move hill‑climbing** – after the main optimisation a
  cheap local search moves a random fraction of mass from one index to
  another and accepts improvements.

The whole routine respects a `budget_s` time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length n (probability vector)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_skewed_distribution(n: int) -> np.ndarray:
    """Beta(0.5,0.5)-skewed distribution (heavy on the ends)."""
    a = np.random.beta(0.5, 0.5, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return _arcsine_distribution(n)
    return a / s


# ----------------------------------------------------------------------
def _expand_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    n = len(a)
    if new_n == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b_k) w.r.t. a using the soft‑max weights.
    a : probability vector (length n)
    b : self‑convolution a*a (length 2n‑1)
    alpha : temperature
    size_fft : FFT size ≥ 3n‑2
    returns gradient of shape (n, ).
    """
    # soft‑max weights
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        # all under‑flowed – fall back to hard max
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    # convolution of w with reversed a
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g

def _optimize_adagrad(a0: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    AdaGrad‑style exponentiated‑gradient optimiser on the simplex.
    Returns the best sequence seen and its internal (2·n·max) score.
    """
    a = a0.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = float(2 * n * b.max())

    # hyper‑parameters
    eta0 = 0.8               # base learning‑rate
    alpha = 1.0              # start temperature
    alpha_mul = 1.03
    alpha_max = 20000.0
    size_fft = 1 << ((3 * n - 2).bit_length())
    eps = 1e-12
    g_acc = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # current convolution and score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} best={best_score:.6f}")

        # gradient of the smoothed max
        grad = _gradient_softmax(a, b, alpha, size_fft)

        # AdaGrad accumulation and per‑coordinate step
        g_acc += grad * grad
        step = eta0 / np.sqrt(g_acc + eps)

        # mirror‑descent (exponentiated) update
        a = a * np.exp(-step * grad)

        # numerical safety
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter to escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.12:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_seq, best_score

    """

    best_seq = a.copy()
