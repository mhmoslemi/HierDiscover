def max_subgradient(a, size_fft):
    conv = _self_convolution(a, size_fft)  # length 2n-1
    M = conv.max()
    # active set: all k where conv[k] == M (or close)
    # We'll take indices where conv >= M - 1e-12 * M maybe
    eps = 1e-12 * max(1.0, M)
    active = np.where(conv >= M - eps)[0]
    # compute gradient: average over active indices
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active:
        # compute contribution as described:
        j_start = max(0, k - (n-1))
        j_end = min(k, n-1)
        js = np.arange(j_start, j_end+1)
        grad[js] += 2 * a[k - js]
    grad = grad / len(active)
    return grad, M
```

    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)

"""
Propose a non‑negative step function (list of heights) that minimizes the
self‑convolution peak evaluated by `evaluate_sequence`.

Algorithm
---------
1. **Broad random/analytic sampling** – generate many candidates from a
   diverse set of analytic families (inverse‑sqrt, power‑law,
   exponential, cosine, Gaussian, triangular, beta‑shaped) and from
   pure Dirichlet/blocked random draws.  Each candidate is projected onto
   the simplex Σa = √(2 n) and evaluated with an exact FFT convolution.
   The $K$ best raw candidates are remembered.

2. **Targeted sub‑gradient descent** – for each remembered candidate we
   minimise the *maximum* of the autocorrelation (the only term that
   matters because the $L^1$ norm is kept constant).  The gradient of the
   active peak(s) is computed analytically (∇_a (a*a)_k = 2·a_{k‑j}),
   a momentum‑Adam style step is taken and the iterate is projected back
   onto the simplex.  The step size decays with the iteration count and
   occasional random “kicks’’ are injected to escape plateaus.

3. **Final stochastic block sweep** – with whatever time remains we
   randomly perturb the current best sequence using a few constant‑height
   blocks; any improvement is kept.

All operations respect the security checks of `evaluate_sequence` and
the whole search stops before the 1000 s budget expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective (without security padding)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _max_subgradient(a: np.ndarray, size_fft: int):
    """
    Return (grad, M) where M = max(a * a) and grad is the sub‑gradient of M.
    The gradient is averaged over all active indices where the max is attained.
    """
    n = a.shape[0]
    conv = _self_convolution(a, size_fft)
    M = conv.max()
    # active set – a tiny tolerance combats floating‑point ties
    tol = max(1e-12, M * 1e-12)
    active = np.where(conv >= M - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] += 2.0 * a[k - js]
    grad /= max(1, len(active))
    return grad, M


# ----------------------------------------------------------------------
def _refine_subgradient(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Perform sub‑gradient descent on max(a*a) with simplex projection.
    Returns the best point seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _self_convolution(a, size_fft)
    best_M = conv.max()
    best_score = 2.0 * n * best_M / (target_sum ** 2)

    # momentum variables
    m = np.zeros_like(a)
    beta = 0.9
    # base learning rate (scale‑invariant)
    eta0 = target_sum * 0.6 / n

    it = 0
    while time.time() < deadline and it < max_iters:
        grad, M = _max_subgradient(a, size_fft)

        # current score (sum is fixed → denominator constant)
        cur_score = 2.0 * n * M / (target_sum ** 2)

        # Adam‑style momentum update (we only need the first moment)
        m = beta * m + (1.0 - beta) * grad
        # step size decays slowly
        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * m
        a_new = _project_to_simplex(a_new, target_sum)

        # evaluate new point
        conv_new = _self_convolution(a_new, size_fft)
        M_new = conv_new.max()
        new_score = 2.0 * n * M_new / (target_sum ** 2)

        # accept if improvement; occasional non‑improving acceptance for escape
        if new_score < cur_score or np.random.rand() < 0.001:
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                best_M = M_new

        # periodic diagnostics
        if it % 5000 == 0:
            print(
                f"[refine] iter {it:6d} cur {cur_score:.6f} best {best_score:.6f}"
            )

        # occasional random kick to avoid stagnation
        if it % 20000 == 0 and it > 0:
            kick = np.random.randn(n) * target_sum * 0.002
            a = _project_to_simplex(a + kick, target_sum)

        it += 1

    return best_a, best_score

        # tiny safety break
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
