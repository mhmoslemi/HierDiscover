def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2

    try:
        base_n = len(height_sequence_1)
    except NameError:
        base_n = 8000

    max_len = 30000

    length_factors = [0.5,0.75,1.0,1.25,1.5,2.0]
    lengths = sorted(set(min(max_len, max(500, int(base_n * f))) for f in length_factors))

    # also include larger predetermined lengths if less than max
    for extra in [20000,25000]:
        if extra <= max_len:
            lengths.append(extra)
    lengths = sorted(set(lengths))

    global_best_seq = None
    global_best_score = float('inf')

    # storage of candidates for later refinement: list of (seq,score,length)
    candidate_pool = []

    for n in lengths:
        if time.time() >= deadline:
            break
        # compute a few candidate sequences
        # define list of shape kinds
        shape_params = []
        # arcsine is beta with p=0.5
        shape_params.append(('arcsine', 0.5))
        for p in [0.3, 0.4, 0.5, 0.6, 0.8, 1.0, 1.2]:
            shape_params.append((f'beta_{p:.2f}', p))
        shape_params.append(('uniform', None))
        shape_params.append(('random', None))

        for kind, param in shape_params:
            if time.time() >= deadline:
                break

            # generate seq
            if kind == 'uniform':
                a = np.full(n, 1.0/n, dtype=np.float64)
            elif kind == 'random':
                a = np.random.rand(n)
                a = a / a.sum()
            else:  # beta style
                a = _beta_distribution(n, param)
                # for arcsine param=0.5, same as arcsine
            # Ensure positivity and sum=1.
            a = _project_onto_simplex(a, s=1.0)

            # fast score
            score = fast_score(a)

            # Store for later refinement if good
            if score < global_best_score:
                global_best_seq = a.copy()
                global_best_score = score
                print(f"[init] n={n} kind={kind} score={score:.6f}")

            # Keep a candidate if within some factor of best
            # add to pool for potential gradient refinement
            if score < global_best_score * 1.2:
                candidate_pool.append((a.copy(), score, n))

            # Also limit pool size
            if len(candidate_pool) > 200:
                # keep only top 200 by score
                candidate_pool.sort(key=lambda x: x[1])
                candidate_pool = candidate_pool[:200]

    # Now refine top candidates using gradient descent
    # Sort candidate_pool by score
    candidate_pool.sort(key=lambda x: x[1])
    # limit number of refined candidates to maybe 30
    top_to_refine = candidate_pool[:30]

    for seq, init_score, n in top_to_refine:
        if time.time() >= deadline:
            break
        # allocate small time slice
        slice_deadline = time.time() + min(10.0, deadline - time.time() - 5.0)
        # gradient descent
        seq_opt, opt_score_internal = _optimize_one_length(seq, n, slice_deadline)
        # local search
        seq_opt, opt_score_internal = _local_search(seq_opt, n,
                                                    time.time() + min(5.0, deadline - time.time()))
        # final score using fast.
        opt_score = fast_score(seq_opt)
        if opt_score < global_best_score:
            global_best_seq = seq_opt.copy()
            global_best_score = opt_score
            print(f"[refine] n={n} improved score={opt_score:.6f}")

        # optionally evaluate via official evaluate_sequence if time left
        if time.time() < deadline - 2:
            try:
                final_score = evaluate_sequence(seq_opt.tolist())
                if final_score < global_best_score:
                    global_best_seq = seq_opt.copy()
                    global_best_score = final_score
                    print(f"[official] n={n} final score={final_score:.6f}")
            except Exception:
                pass

        if global_best_score <= 1.503:
            break

    # final global refinement on best sequence (if time)
    if global_best_seq is not None and time.time() < deadline - 2:
        seq_refined, _ = _local_search(global_best_seq, len(global_best_seq), deadline)
        try:
            final_score = evaluate_sequence(seq_refined.tolist())
            if final_score < global_best_score:
                global_best_seq = seq_refined
                global_best_score = final_score
                print(f"[final refine] best score={global_best_score:.6f}")
        except Exception:
            pass

    # fallback
    if global_best_seq is None:
        n = 5000
        global_best_seq = np.full(n, 1.0/n, dtype=np.float64)
        try:
            global_best_score = evaluate_sequence(global_best_seq.tolist())
        except Exception:
            global_best_score = fast_score(global_best_seq)

    # ensure final list
    return global_best_seq.tolist()

"""
Propose a non‑negative step function (list of heights) minimizing
`evaluate_sequence`.  

The algorithm works in three stages:

1. **Broad exploration** – a set of candidate lengths is chosen (around the
   previous best length and a few larger values).  For each length a suite of
   parametrised families (Beta / arcsine shapes, truncated geometric shapes,
   uniform and random) is generated.  Each candidate is scored instantly by a
   fast FFT‑based computation of `2·n·max(self‑convolution)`.

2. **Refinement** – the best few candidates are fed to a projected gradient
   descent on the probability simplex.  The gradient is obtained via a
   soft‑max approximation of the max‑convolution, with Nesterov momentum,
   annealed temperature and occasional jitter.  After the gradient phase a
   cheap stochastic local‑search (pairwise mass swaps) further reduces the
   objective.

3. **Final check** – the best sequence found is evaluated with the official
   `evaluate_sequence` function and, if time permits, a final local‑search
   refinement is performed.

All heavy convolution work uses FFT (O(n log n)), allowing the search to
run comfortably within the 1000 s budget.  The code contains no closures,
only top‑level helpers, and respects the security checks of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Fast proxy for evaluate_sequence: 2·n·max(self‑convolution)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _beta_distribution(n: int, p: float) -> np.ndarray:
    """Discrete symmetric Beta(p,p) shape, p>0.  Returns a probability vector."""
    i = np.arange(n, dtype=np.float64) + 0.5
    # exponent (p-1) may be negative; start at 0.5 avoids division by zero
    left = i ** (p - 1.0)
    right = (n - i) ** (p - 1.0)
    a = left * right
    # avoid NaNs if p is very small – replace infinities by large numbers
    a[np.isinf(a)] = np.finfo(np.float64).max
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def _geometric_distribution(n: int, r: float) -> np.ndarray:
    """Truncated geometric distribution on {0,…,n-1} with ratio r∈(0,1)."""
    i = np.arange(n, dtype=np.float64)
    # unnormalised probabilities
    a = r ** i
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(b) w.r.t. a.
    For very large alpha this approaches a sub‑gradient of max(b).
    """
    n = a.shape[0]

    # ----- soft‑max weights -----
    if alpha > 2000:
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ----- convolution w * rev(a) -----
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g

def _optimize_one_length(init_a: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the simplex attempting to minimise max
    self‑convolution.  Returns the best sequence discovered and its fast score
    (2·n·max_b).
    """
    a = init_a.copy()
    best_seq = a.copy()
    best_score = _fast_score(a)

    # hyper‑parameters
    beta = 0.8                     # momentum factor
    eta0 = 0.9                     # initial step‑size constant
    velocity = np.zeros_like(a)
    alpha = 5.0                    # starting temperature for soft‑max
    alpha_mul = 1.06
    alpha_max = 20000.0
    size_fft = 1 << ((3 * n - 2).bit_length())   # size for gradient conv
    it = 0

    while time.time() < deadline:
        it += 1

        # current score
        cur_score = _fast_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[grad] n={n} it={it} score={best_score:.6f}")

        # compute convolution for gradient
        b = _fft_self_convolve(a)

        # gradient (soft‑max or sub‑gradient)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Nesterov‑style momentum
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter
        if it % 2500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score

def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: move a random fraction of mass from one
    index to another and keep the change if the max‑convolution improves.
    """
    best_seq = a.copy()
    best_score = _fast_score(best_seq)
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_seq[i] <= 1e-12:
            continue
        frac = np.random.rand() * 0.30
        delta = best_seq[i] * frac
        a_new = best_seq.copy()
        a_new[i] -= delta
        a_new[j] += delta
        score = _fast_score(a_new)
        if score < best_score:
            best_seq = a_new
            best_score = score

        # tiny jitter every few thousand iterations
        if it % 5000 == 0:
            a_new = best_seq + np.random.randn(n) * 1e-6
            a_new = _project_onto_simplex(a_new, s=1.0)
            score = _fast_score(a_new)
            if score < best_score:
                best_seq = a_new
                best_score = score

    return best_seq, best_score

            # store for the next phase
            # we will refine them later if we have time
            if sc < global_best_score * 1.2:
                # attach length information for later use
                pass  # nothing extra needed – we will simply keep the best later

    # ------------------------------------------------------------------
    # 2) Refine the globally best candidate (if any) with gradient + local search
    if global_best_seq is not None and time.time() < deadline:
        n_best = len(global_best_seq)
        # allocate a sizeable slice for refinement
        slice_deadline = time.time() + max(5.0, (deadline - time.time()) * 0.5)

        # gradient descent
        refined_seq, _ = _optimize_one_length(global_best_seq, n_best, slice_deadline)

        # local search (remaining time)
        if time.time() < deadline:
            refined_seq, _ = _local_search(refined_seq, n_best, deadline)

        # evaluate with fast proxy
        refined_score = _fast_score(refined_seq)
        if refined_score < global_best_score:
            global_best_seq = refined_seq.copy()
            global_best_score = refined_score
            print(f"[refine] length={n_best} improved fast score={refined_score:.6f}")

        # final official evaluation (if time permits)
        if time.time() < deadline - 1.0:
            try:
                final_score = evaluate_sequence(refined_seq.tolist())
                if final_score < global_best_score:
                    global_best_seq = refined_seq.copy()
                    global_best_score = final_score
                    print(f"[official] final score={final_score:.6f}")
                else:
                    # keep the fast proxy score for reporting
                    print(f"[official] official score={final_score:.6f} (no improvement)")
            except Exception:
                # fallback to fast score if the black‑box crashes
                print("[official] evaluation failed – using fast proxy")
                pass

    # ------------------------------------------------------------------
    # 3) Safety fallback
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = _fast_score(global_best_seq)
        print(f"[fallback] length={n} score={global_best_score:.6f}")

    print(f"Best fast proxy score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    return global_best_seq.tolist()
