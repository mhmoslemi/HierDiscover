"""
Proposes a non‑negative step‑function (list of heights) that minimizes the
self‑convolution peak evaluated by ``evaluate_sequence``.

Algorithm
---------
1. **Exploratory sampling** – a large pool of analytic families
   (inverse‑sqrt, power‑law, exponential, cosine, beta‑shaped,
   Gaussian, triangular, random block) is sampled.  Every sample is
   projected onto the simplex {a_i ≥ 0, Σ a_i = √(2 n)} and scored with an
   exact FFT‑based convolution.  The best ``K`` candidates are retained.

2. **Gradient‑based refinement** – each retained candidate is refined with a
   smoothed‑max Adam optimiser.  The hard maximum in the objective is
   approximated by a soft‑max with a temperature that is slowly cooled.
   Once the temperature is tiny the optimiser switches to the exact
   sub‑gradient (uniform weight on the active max indices).  After each
   update the vector is projected back onto the simplex.  The best sequence
   observed during the run is kept.

3. **Local perturbation sweep** – with the remaining budget a cheap hill‑climb
   moves a small amount of mass between random indices; improvements are
   accepted immediately.  This fine‑tunes the best sequence found in stage 2.

All heavy work respects the 1000 s time budget; the function returns the
best sequence before the deadline.  No file or network access is used.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    (w must sum to 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact score used during the search (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using a smoothed‑max Adam optimiser.
    Returns the best vector observed and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Base learning rate – scales with target sum / n
    eta0 = target_sum * 0.5 / n

    # Initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- weight computation (soft‑max → sub‑gradient) -----
        if T > 1e-4:
            shifted = conv - conv.max()                # ≤ 0
            exp_arg = shifted / T
            exp_arg = np.clip(exp_arg, -50, 0)         # avoid overflow
            w = np.exp(exp_arg)
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)
        else:
            # exact sub‑gradient: uniform weight on all indices attaining the max
            mask = conv >= conv.max() - 1e-12
            cnt = mask.sum()
            w = np.where(mask, 1.0 / cnt, 0.0)

        # ----- gradient of (soft‑max) objective -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        eta = eta0 / (1.0 + it * 1e-5)               # simple decay
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto simplex -----
        a_new = _project_to_simplex(a_new, target_sum)

        # ----- occasional random kick to escape plateaus -----
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9997                               # very slow cooling
        it += 1

    return best_a, best_score

    # ----------------------------------------------------------------
    # ---------- 3. LOCAL PERTURBATION SWEEP -------------------------
    # Very cheap hill‑climb that moves a tiny amount of mass between random indices.
    eps = target_sum * 0.001   # typical step size
    while time.time() < deadline:
        # propose a small perturbation
        i = np.random.randint(0, n)
        if global_best_a[i] < eps:
            continue
        j = np.random.randint(0, n)
        if i == j:
            continue
        new_a = global_best_a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # projection not needed – sum unchanged and we kept non‑negativity at i
        sc = quick_score(new_a, n, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = new_a
            print(f"[local] improved to {global_best_score:.6f}")
        # early exit when time is about to run out
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
