"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1. **Generate a pool of structured candidates** on several lengths.
   Candidates are taken from simple parametric families
   (uniform, arcsine, power‑law U‑shapes, symmetric Beta distributions
   and mixtures).  The convolution is evaluated with an FFT‑based
   routine, and the 3‑best candidates per length are kept.

2. **Refine each retained candidate** by projected Adam optimisation.
   The objective is  f(a)=2·n·max(a*a) (the denominator is 1 because we
   keep  Σa_i=1).  A smooth‑max surrogate (soft‑max) is used while the
   temperature grows, then the exact sub‑gradient of the max‑operator
   is employed.  After each Adam step the iterate is projected onto the
   probability simplex.

3. **Return the globally best sequence** found within the time budget.
   All numbers are ≤1000 and ≥0, so the black‑box `evaluate_sequence`
   will see exactly the same value as our fast estimator.

The whole search respects the 1000 s limit, uses only NumPy and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (β(0.5,0.5)), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _power_distribution(n: int, gamma: float) -> np.ndarray:
    """U‑shaped distribution a_i ∝ ((i+0.5)(n‑i‑0.5))^{‑γ}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    denom = (i * (n - i)) ** gamma
    a = 1.0 / denom
    a = _project_onto_simplex(a)
    return a


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) law on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid 0⁰ issues
    if np.isclose(alpha, 1.0):
        a = np.full(n, 1.0 / n)
    else:
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a = _project_onto_simplex(a)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _score_from_vector(a: np.ndarray) -> float:
    """Exact objective value for a (Σa=1)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # soft‑max temperature (for a smoothed max)
    temp = 1.0
    temp_factor = 1.02
    temp_max = 5000.0

    best_a = a.copy()
    best_score = _score_from_vector(a)

    it = 0
    while time.time() < time_limit:
        it += 1
        # --- current convolution ------------------------------------------------
        b = _fft_self_convolve(a)
        cur_max = b.max()

        # --- gradient -----------------------------------------------------------
        if temp < 300.0:
            # smoothed max (soft‑max)
            exp_term = np.exp(temp * (b - cur_max))
            w = exp_term / exp_term.sum()
        else:
            # exact sub‑gradient (uniform weight on all active entries)
            eps_act = 1e-12 * (abs(cur_max) + 1.0)
            active = np.flatnonzero(b >= cur_max - eps_act)
            if active.size == 0:
                w = np.zeros_like(b)
                w[np.argmax(b)] = 1.0
            else:
                w = np.zeros_like(b)
                w[active] = 1.0 / active.size
        # gradient = 2 * conv_rev (where conv_rev = a_rev * w)
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # --- Adam update --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate schedule (decreasing with √it)
        lr = 0.2 / (np.sqrt(it) + 1.0)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a)

        # --- temperature annealing ------------------------------------------------
        if temp < temp_max:
            temp = min(temp_max, temp * temp_factor)

        # --- keep best -----------------------------------------------------------
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # small console feedback
            print(
                f"[opt] n={n} it={it} temp={temp:.1f} "
                f"score≈{best_score:.6f}"
            )
            if best_score <= target:
                break

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.1:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    evaluate_sequence() score.  The routine mixes a quick parametric
    sweep with a projected Adam optimiser and respects the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # tiny safety margin

    overall_best_seq: np.ndarray | None = None
    overall_best_score = float("inf")
    target_score = 1.503

    # base length (use previous construction if available)
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # lengths to explore – keep them moderate for speed
    lengths = sorted(
        set(
            max(500, int(base_n * f))
            for f in (0.5, 0.75, 1.0, 1.25, 1.5, 2.0)
            if int(base_n * f) >= 500
        )
    )
    if not lengths:
        lengths = [8000]

    # ------------------------------------------------------------------
    for n in lengths:
        # ----------- generate a rich set of initial candidates -----------------
        candidates: list[tuple[str, np.ndarray]] = []

        # uniform
        candidates.append(("uniform", np.full(n, 1.0 / n)))

        # arcsine
        candidates.append(("arcsine", _arcsine_distribution(n)))

        # power‑law family (U‑shaped)
        for gamma in np.linspace(0.0, 1.5, 13):
            candidates.append((f"power_{gamma:.2f}", _power_distribution(n, gamma)))

        # symmetric Beta family
        for alpha in np.linspace(0.2, 1.0, 9):
            candidates.append((f"beta_{alpha:.2f}", _beta_distribution(n, alpha)))

        # mixtures of arcsine and uniform
        for lam in np.linspace(0.0, 1.0, 6):
            mixed = lam * np.full(n, 1.0 / n) + (1.0 - lam) * _arcsine_distribution(n)
            mixed = _project_onto_simplex(mixed)
            candidates.append((f"mix_{lam:.2f}", mixed))

        # a few random Dirichlet draws
        for k in range(3):
            rnd = np.random.rand(n)
            rnd = _project_onto_simplex(rnd)
            candidates.append((f"rand_{k}", rnd))

        # ------------------------------------------------------------------
        # evaluate quickly (our own fast estimator) and keep the best few
        scored: list[tuple[float, str, np.ndarray]] = []
        for name, vec in candidates:
            # safety: clip to allowed range before scoring
            if vec.max() > 1000.0:
                vec = np.clip(vec, 0.0, 1000.0)
                vec = _project_onto_simplex(vec)
            sc = _score_from_vector(vec)
            scored.append((sc, name, vec))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = vec.copy()
                print(f"[init] n={n} {name:12s} → score≈{sc:.6f}")

        # keep the top 3 candidates for refinement
        scored.sort(key=lambda x: x[0])
        top_k = 3
        top_candidates = scored[:top_k]

        # ------------------------------------------------------------------
        # Refine each top candidate with Adam optimisation
        for cur_score, cur_name, cur_vec in top_candidates:
            # allocate a slice of the remaining budget
            now = time.time()
            if now >= deadline:
                break
            remaining = deadline - now
            slice_deadline = now + min(remaining * 0.5, 180.0)  # at most 3 min

            refined, refined_score = _adam_optimize(
                cur_vec, n, slice_deadline, target_score
            )

            # sanity check with the official evaluator (may be redundant)
            try:
                final_score = evaluate_sequence(refined.tolist())
            except Exception:
                final_score = refined_score

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = refined.copy()
                print(
                    f"[refine] n={n} {cur_name:12s} → score={final_score:.6f}"
                )

            if overall_best_score <= target_score:
                print("Target achieved – returning best sequence.")
                return overall_best_seq.tolist()

        # ------------------------------------------------------------------
        if time.time() >= deadline:
            break
        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but we return a uniform vector.
    if overall_best_seq is None:
        fallback_n = 2000
        overall_best_seq = np.full(fallback_n, 1.0 / fallback_n)
        overall_best_score = _score_from_vector(overall_best_seq)

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length={len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
