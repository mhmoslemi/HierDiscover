"""
Propose a step‑function (non‑negative vector) with a low
evaluate_sequence() score.

Algorithm
---------
1.  Determine a problem size n (use the reference `height_sequence_1`
    if it exists, otherwise default to 3400).  The ℓ¹‑norm is fixed to
    S = sqrt(2·n) so that the evaluate_sequence() score equals
    max(conv(a,a)).

2.  Generate several diverse initial vectors (uniform, exponential,
    reversed, random Gaussian, reference).  Each is projected onto the
    simplex Σ a_i = S.

3.  For each start we run a short *sub‑gradient descent* on the
    hard‑max objective.
    *   The sub‑gradient of max(conv) is computed via the active
        convolution entries.
    *   A decreasing step‑size η = η₀ / sqrt(it+1) is used,
        η₀ = 0.5·S / n.
    *   After each step the iterate is projected back onto the simplex.
    *   Whenever a new best score is found it is recorded.

4.  After the gradient phase we perform a *Frank‑Wolfe‑like refinement*.
    For the current vector a we:
    *   Compute its convolution and take the K (default 200) largest
        entries – these are the only constraints that can become active.
    *   Build a small linear programme

            maximise   Σ b_j
            subject to  (a * b)[k] ≤ M   for the selected k,
                        b_j ≥ 0,

        where M = max(conv(a,a)).
        The programme has n variables and K constraints, solved with
        ``scipy.optimize.linprog`` (HiGHS).
    *   Scale the LP solution to the required ℓ¹‑norm (b ← b·S/Σb) and
        take a convex combination a ← (1‑t)·a + t·b (t starts at 0.3 and
        is halved if the combination does not improve the score).
    *   The loop is repeated until no improvement is observed or time
        runs out (maximum 25 LP iterations per start).

5.  A cheap *random‑block* search is finally applied to the best vector
    found: the index set is partitioned into a random number of blocks,
    each block gets a random constant height and the whole vector is
    projected onto the simplex.

6.  The best feasible vector encountered (converted to a list) is
    returned.  All intermediate results are printed for progress monitoring.

The whole routine respects a user‑supplied time budget (default 1000 s)
and never exceeds it.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> float:
    """Return max of the auto‑convolution a*a."""
    return np.convolve(a, a).max()


# ----------------------------------------------------------------------
def _subgradient_step(a: np.ndarray, S: float, eta: float) -> np.ndarray:
    """
    Perform a single projected sub‑gradient step for max(conv(a,a)).
    Returns the new feasible vector.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    # active set (allow a tiny tolerance)
    eps = 1e-12
    mask = conv >= max_c - eps
    cnt = mask.sum()
    w = np.zeros_like(conv)
    if cnt > 0:
        w[mask] = 1.0 / cnt

    # gradient wrt a: g_i = 2 * Σ_k w_k * a_{i-(k-n+1)}
    conv_wr = np.convolve(w, a[::-1])
    g = 2.0 * conv_wr[n - 1:n - 1 + n]

    a_new = a - eta * g
    a_proj = _project_to_simplex(a_new, S)
    return a_proj


# ----------------------------------------------------------------------
def _lp_refine(a: np.ndarray, S: float, M: float,
               K: int = 200, t: float = 0.3) -> np.ndarray:
    """
    One Frank‑Wolfe refinement step based on an LP with only the K
    largest convolution constraints.
    Returns a new vector (convex combination of a and the LP solution)
    or the original a if the LP fails.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)

    # pick indices of the K largest entries (or all if less)
    if K >= conv.size:
        idx = np.arange(conv.size)
    else:
        idx = np.argpartition(conv, -K)[-K:]

    K_sel = idx.shape[0]

    # Build a dense constraint matrix of shape (K_sel, n)
    A = np.zeros((K_sel, n), dtype=float)
    for row, k in enumerate(idx):
        # rows correspond to convolution index k
        # for column j (b_j) we need a_{k-j} if 0 ≤ k-j < n
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        # indices of b_j that appear
        cols = np.arange(j_start, j_end + 1, dtype=int)
        A[row, cols] = a[k - cols]

    # Linear programme: maximise Σ b_j  <=> minimise -Σ b_j
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None) for _ in range(n)]
    b_ub = np.full(K_sel, M, dtype=float)

    # Solve with HiGHS (fast for moderate size)
    with np.errstate(all='ignore'):
        res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30.0})

    if not res.success or res.x is None:
        return a

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a

    # scale to satisfy the ℓ¹‑norm exactly
    scale = S / sum_b
    g = b * scale

    # convex combination (both a and g already sum to S)
    a_new = (1.0 - t) * a + t * g
    # due to numerical noise the sum may drift a little – re‑project
    a_new = _project_to_simplex(a_new, S)
    return a_new
