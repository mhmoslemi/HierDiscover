"""
Propose a non‑negative step function (list of heights) that minimizes the
self‑convolution peak used by ``evaluate_sequence``.

Algorithm
---------
1. **Exploration** – generate a diverse pool of candidates from several
   analytic families (double‑ended power‑law, double‑ended exponential,
   cosine‑based, random block, etc.).  Each raw candidate is projected onto
   the simplex Σa = √(2 n) and its exact score (via FFT) is computed.
   The best ``K`` candidates are retained.

2. **Refinement** – each retained candidate is improved with a
   smooth‑max Adam optimiser.  The objective is approximated by a
   soft‑max over the convolution entries; its gradient is obtained by a
   single extra FFT per iteration.  The optimiser uses a larger initial
   learning rate and temperature than the reference code, occasional
   random kicks, and a final few sub‑gradient steps that directly target
   the current maximal convolution index.

3. **Local random‑swap sweep** – the current global best is further
   polished by tiny mass transfers between random positions; improvements
   are kept immediately.

All helper functions are defined at the top level (no closures) and the
search respects the 1000 s time budget.  The returned list is ready for
``evaluate_sequence``.

import math
import time

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using real FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a (w must sum to 1).
    Uses convolution with the reversed vector of a.
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact score used during the search (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

        idx = np.arange(n)
        mask = (k_max - idx >= 0) & (k_max - idx < n)
        g = np.where(mask, 2.0 * a[k_max - idx], 0.0)

        step = target_sum * 0.05 / (1.0 + sub_it)
        a = a - step * g
        a = _project_to_simplex(a, target_sum)

        sub_it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_improve(a_start: np.ndarray,
                   deadline: float,
                   n: int,
                   size_fft: int,
                   target_sum: float) -> tuple[np.ndarray, float]:
    """
    Tiny random‑mass‑swap hill‑climbing around ``a_start``.
    Returns the best sequence seen and its score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    while time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # transfer a random fraction of a[i] to a[j]
        frac = np.random.rand() * 0.4          # up to 40 % of a[i]
        delta = a[i] * frac
        a[i] -= delta
        a[j] += delta

        sc = quick_score(a, n, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

        # occasional tiny Gaussian perturbation to escape flat plateaus
        if np.random.rand() < 0.001:
            a = best_a + np.random.randn(n) * target_sum * 0.0005
            a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the self‑convolution peak.  The function respects the 1000 s budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)               # normalisation used in the paper
    size_fft = 1 << ((3 * n - 2).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # Keep the K best raw candidates from the exploratory phase
    K = 10
    best_candidates: list[tuple[float, np.ndarray]] = []   # (score, array)

    def _consider(score: float, a: np.ndarray) -> None:
        """Maintain a sorted list of the top‑K candidates (ascending score)."""
        if len(best_candidates) < K:
            best_candidates.append((score, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if score < best_candidates[-1][0]:
                best_candidates[-1] = (score, a.copy())
                best_candidates.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # ------------ 1. PARAMETRIC SHAPE SEARCH -------------------------

    idx = np.arange(n, dtype=float)

    def _add_candidate(arr: np.ndarray) -> None:
        arr = _project_to_simplex(arr, target_sum)
        sc = quick_score(arr, n, size_fft)
        _consider(sc, arr)

    # (a) Uniform
    _add_candidate(np.ones(n))

    # (b) Double‑ended power‑law (i+β)^‑p + (n‑i‑1+β)^‑p
    for _ in range(250):
        beta = np.random.rand() * 20.0
        p = np.random.rand() * 1.8 + 0.2          # p ∈ [0.2, 2.0]
        a = 1.0 / (idx + beta + 1.0) ** p + 1.0 / (n - 1 - idx + beta + 1.0) ** p
        _add_candidate(a)

    # (c) Double‑ended exponential
    for _ in range(200):
        decay_l = np.random.rand() * 0.02 + 0.0001
        decay_r = np.random.rand() * 0.02 + 0.0001
        a = np.exp(-decay_l * idx) + np.exp(-decay_r * (n - 1 - idx))
        _add_candidate(a)

    # (d) Cosine‑based, large at ends, small middle
    for _ in range(150):
        power = np.random.rand() * 5.0 + 0.5
        a = np.abs(np.cos(np.pi * idx / (n - 1))) ** power
        _add_candidate(a)

    # (e) Sine‑based, zero at ends, peak in middle (as a control)
    for _ in range(80):
        power = np.random.rand() * 5.0 + 0.5
        a = np.sin(np.pi * idx / (n - 1)) ** power
        _add_candidate(a)

    # (f) Gaussian bump (centered) – usually not good but kept for completeness
    for _ in range(80):
        sigma = (np.random.rand() * 0.25 + 0.05) * n
        centre = (n - 1) / 2.0 + (np.random.rand() - 0.5) * n * 0.1
        a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        _add_candidate(a)

    # (g) Random block candidates (few constant blocks)
    for _ in range(400):
        nb = np.random.randint(2, 9)                     # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1                  # ensure positivity
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        _add_candidate(cand)

    # (h) Reference sequence if available
    if have_ref:
        ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        _add_candidate(ref)

    best_candidates.sort(key=lambda x: x[0])
    print(f"[exploration] best raw score = {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # ------------ 2. REFINEMENT (Adam + sub‑gradient) ---------------

    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    # we will devote roughly 70 % of the remaining budget to refinement
    remaining = deadline - time.time()
    refine_budget = remaining * 0.70
    num_refine = min(5, len(best_candidates))   # refine at most 5 candidates

    # allocate time equally among the selected candidates
    time_per_candidate = refine_budget / max(1, num_refine)

    for rank in range(num_refine):
        raw_score, a0 = best_candidates[rank]
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank + 1} (raw score {raw_score:.6f})")
        cand_deadline = time.time() + time_per_candidate
        # generous iteration budget – limited by remaining time
        max_iters = int(min(500_000, (cand_deadline - time.time()) * 1500))
        refined_a, refined_score = _refine_candidate_adam(
            a0, cand_deadline, n, size_fft, target_sum, max_iters
        )
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # ------------ 3. FINAL LOCAL RANDOM‑SWAP IMPROVEMENT -----------

    if time.time() < deadline:
        print("[final local sweep] starting")
        final_a, final_score = _local_improve(
            global_best_a, deadline, n, size_fft, target_sum
        )
        if final_score < global_best_score:
            global_best_score = final_score
            global_best_a = final_a.copy()
            print(f"[global] new best after local sweep = {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    # Return a plain Python list as required by the scorer
    return global_best_a.tolist()
