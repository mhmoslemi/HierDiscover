"""
Propose a near‑optimal non‑negative step‑function (probability vector).

Algorithm
---------
* Work on the probability simplex Σ a_i = 1 (scaling does not affect the
  score).
* For a few candidate lengths (derived from the previously found best
  construction) generate several initial vectors:
    - uniform, arcsine law, random, and a few Beta‑distribution shapes.
* Each initial vector is refined by a projected **Adam** optimiser.
  The objective is   f(a) = max_k (a*a)_k   (the denominator of the score
  is constant because Σa = 1).
* Gradient computation:
    - While a “temperature’’ α is modest we use a smooth soft‑max
      approximation of the max, which yields a well‑behaved gradient.
    - After α exceeds a threshold we switch to the exact sub‑gradient,
      obtained from the active convolution entries (the same technique as
      the reference implementation).
* After every update the iterate is projected back onto the simplex.
  Occasional jitter is added to escape plateaus.
* The best sequence found (according to the black‑box `evaluate_sequence`)
  is kept globally.  The search stops early if the target score 1.503 is
  reached or the time budget (≈1000 s) expires.

The use of Adam (adaptive moment estimation) together with an annealed
smooth‑max surrogate gives a more stable and faster convergence than the
fixed‑step momentum scheme used previously, allowing us to push the
score below the target within the allotted time.

import time
import numpy as np
from scipy.special import beta as beta_func   # for Beta pdf

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(½,½)) of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    lengths = sorted(
        {
            max(500, int(base_n * f))
            for f in (0.7, 0.85, 1.0, 1.15, 1.3)
        }
    )

    best_seq = None          # type: np.ndarray | None
    best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        # allocate a generous portion of the remaining budget to this length
        remaining = max(0.0, deadline - time.time())
        if remaining < 30.0:
            break
        slice_deadline = time.time() + min(remaining * 0.85, 800.0)

        # ------------------------------------------------------------------
        # Build a collection of initial feasible vectors.
        initial_vectors = []

        # deterministic starters
        initial_vectors.append(_arcsine_distribution(n))
        initial_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))
        initial_vectors.append(np.random.rand(n))

        # a few random Beta‑law shapes (α,β in (0.2, 2.0))
        for _ in range(5):
            alpha = np.random.uniform(0.2, 2.0)
            beta = np.random.uniform(0.2, 2.0)
            # midpoint discretisation to avoid 0/1 singularities
            i = np.arange(n, dtype=np.float64) + 0.5
            x = i / n
            vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
            # replace possible NaNs (should not happen with the midpoint grid)
            vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
            if vals.sum() == 0.0:
                vals = np.full(n, 1.0 / n)
            else:
                vals /= vals.sum()
            initial_vectors.append(vals)

        # project all to the simplex (ensures exact feasibility)
        initial_vectors = [_project_onto_simplex(v, 1.0) for v in initial_vectors]

        # ------------------------------------------------------------------
        for init_id, a in enumerate(initial_vectors):
            # keep the best local candidate seen during optimisation
            best_local_seq = a.copy()
            b = _fft_self_convolve(a)
            best_local_max = b.max()
            best_local_score = 2.0 * n * best_local_max

            # Adam moments
            m = np.zeros_like(a)
            v_exp = np.zeros_like(a)

            # hyper‑parameters
            lr = 0.05               # base learning rate
            beta1 = 0.9
            beta2 = 0.999
            eps = 1e-8

            # temperature schedule for the smoothed‑max surrogate
            alpha = 5.0
            alpha_factor = 1.002
            alpha_max = 2000.0

            it = 0
            while time.time() < slice_deadline:
                it += 1

                # ----- statistics -----
                b = _fft_self_convolve(a)
                cur_max = b.max()

                # update best local record
                if cur_max < best_local_max - 1e-14:
                    best_local_max = cur_max
                    best_local_seq = a.copy()
                    best_local_score = 2.0 * n * cur_max
                    print(
                        f"[{time.time():.1f}] n={n} init={init_id} it={it} "
                        f"maxB={cur_max:.8e} score≈{best_local_score:.6f}"
                    )

                # ----- gradient (soft‑max or exact) -----
                if alpha < 500.0:
                    # smooth approximation of max
                    exp_term = np.exp(alpha * (b - cur_max))
                    w = exp_term / exp_term.sum()
                    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
                    # raise temperature gradually
                    if alpha < alpha_max:
                        alpha = min(alpha_max, alpha * alpha_factor)
                else:
                    # exact sub‑gradient from the active set
                    eps_act = 1e-12 * (abs(cur_max) + 1.0)
                    active = np.flatnonzero(b >= cur_max - eps_act)
                    w = np.zeros_like(b)
                    w[active] = 1.0
                    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
                    # correct the double‑count on the diagonal (i=j)
                    even_mask = (active % 2 == 0)
                    diag_idx = (active[even_mask] // 2)
                    if diag_idx.size > 0:
                        grad[diag_idx] -= a[diag_idx]

                # ----- Adam update -----
                m = beta1 * m + (1.0 - beta1) * grad
                v_exp = beta2 * v_exp + (1.0 - beta2) * (grad ** 2)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v_exp / (1.0 - beta2 ** it)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                a = a - step
                a = _project_onto_simplex(a, 1.0)

                # occasional jitter to escape flat regions
                if (it % 2000 == 0) and (np.random.rand() < 0.1):
                    a += np.random.randn(n) * 1e-6
                    a = _project_onto_simplex(a, 1.0)

                # early exit if we have already crossed the target bound
                if best_local_score <= 1.502:
                    break

            # ----- evaluate the best candidate of this run -----
            try:
                final_score = evaluate_sequence(best_local_seq.tolist())
            except Exception:
                final_score = best_local_score

            if final_score < best_score:
                best_score = final_score
                best_seq = best_local_seq.copy()
                print(
                    f"*** New global best → n={n} init={init_id} "
                    f"score={best_score:.6f}"
                )

            # stop if target achieved
            if best_score <= 1.503:
                print("Target score reached – exiting early.")
                return best_seq.tolist()

            # respect the slice deadline
            if time.time() >= slice_deadline:
                break

        # overall time check
        if time.time() >= deadline:
            break
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but we return a uniform vector just in case.
    if best_seq is None:
        n_fallback = 1000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback uniform sequence used.")

    print(
        f"Final best score: {best_score:.6f} (length = {len(best_seq)})"
    )
    return best_seq.tolist()
