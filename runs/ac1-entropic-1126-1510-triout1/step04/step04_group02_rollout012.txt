def _adam_optimize(a0, n, deadline):
    # convert to logits
    eps = 1e-12
    z = np.log(a0 + eps)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1=0.9
    beta2=0.999
    eps_adam=1e-8
    lr=0.1?
    best_score = 2*n * np.max(_fft_self_convolve(a0))
    best_a = a0.copy()
    # schedule for temperature
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0
    size_fft = 1 << ((3*n - 2).bit_length())
    t = 0
    while time.time() < deadline:
        t+=1
        a = np.exp(z)
        a_sum = a.sum()
        a = a / a_sum
        
        b = _fft_self_convolve(a)
        maxb = b.max()
        cur_score = 2*n*maxb
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        
        if alpha > alpha_thr:
            # subgradient
            k = int(np.argmax(b))
            grad_base = _subgradient_max(a, k)
        else:
            grad_base = _gradient_softmax(a, b, alpha, size_fft)
        grad_a = 2 * n * grad_base
        
        # gradient w.r.t logits
        dot = np.dot(a, grad_a)
        g_z = a * (grad_a - dot)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_z
        v = beta2 * v + (1 - beta2) * (g_z ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # optional jitter
        if t % 5000 == 0 and np.random.rand() < 0.1:
            z += np.random.randn(n) * 1e-4
        
        # temperature schedule
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative height vector) that minimizes
`evaluate_sequence`.  The algorithm works in three phases:

1. **Coarse scan** – evaluate several candidate lengths (including the
   length of a previous solution) and several simple families
   (arcsine, symmetric Beta(α,α)).  The best family/length is kept.

2. **Fine optimisation** – starting from the best candidate, run a
   mirror‑descent style optimiser on the probability simplex.  We
   optimise the logits of a soft‑max representation with an Adam‑like
   update, using a smooth soft‑max surrogate for the ‖·‖∞‑norm of the
   self‑convolution and switching to a hard sub‑gradient later.
   The optimiser respects the remaining time budget.

3. **Local refinement** – a cheap random‑move hill‑climber that moves a
   tiny amount of mass between two coordinates while the objective
   improves.

All helper functions are defined at module level; the function
`propose_candidate` respects the 1000 s budget and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine (Beta(0.5,0.5)) distribution of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_alpha_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on n points.
    For α=0.5 this coincides with the arcsine law.
    """
    # centre points of the bins
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid overflow/underflow for extreme α
    with np.errstate(divide='ignore', invalid='ignore'):
        vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    # replace NaNs (caused by 0**neg) by 0
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    total = vals.sum()
    if total == 0.0:
        # fallback to arcsine
        vals = _arcsine_distribution(n)
    else:
        vals /= total
    return vals


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index k (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate (1/α)·log∑exp(α·b) w.r.t. a.
    Uses convolution in the Fourier domain for speed.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _score_of_distribution(a: np.ndarray) -> float:
    """Internal (2·n·max) score for a probability vector a (sum=1)."""
    n = len(a)
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimiser on the logits of a soft‑max representation.
    Returns the best distribution (sum=1) found and its internal score.
    """
    # initialise logits safely
    eps = 1e-12
    z = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.08                     # learning‑rate, works well for a wide range of n

    # temperature schedule for soft‑max surrogate
    alpha = 1.0
    alpha_mul = 1.02
    alpha_max = 15000.0
    alpha_thr = 2000.0

    size_fft = 1 << ((3 * n - 2).bit_length())

    best_a = a0.copy()
    best_score = _score_of_distribution(best_a)

    it = 0
    while time.time() < deadline:
        it += 1

        # obtain probability vector from logits
        a = np.exp(z)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a_sum

        # convolution and current score
        b = _fft_self_convolve(a)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient of the surrogate objective
        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad_base = _subgradient_max(a, k)
        else:
            grad_base = _gradient_softmax(a, b, alpha, size_fft)

        # full gradient w.r.t. a (objective = 2·n·max)
        grad_a = 2.0 * n * grad_base

        # chain rule to obtain gradient w.r.t. logits z
        dot = np.dot(a, grad_a)
        g_z = a * (grad_a - dot)

        # Adam update (no bias‑correction – simpler & fast)
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z ** 2)
        z = z - lr * m / (np.sqrt(v) + eps_adam)

        # temperature annealing
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to escape flat regions
        if it % 5000 == 0 and np.random.rand() < 0.13:
            z += np.random.randn(n) * 1e-5

    return best_a, best_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple random‑move hill‑climbing on the best distribution.
    Moves up to 10 % of the source mass from one index to another.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = _score_of_distribution(a)

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a small fraction of the mass at i towards j
        eps = a[i] * np.random.rand() * 0.1
        if eps <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        if a_candidate[i] < 0.0:
            continue
        cur_score = _score_of_distribution(a_candidate)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] local improve it={it} score={best_score:.6f}")

    return best_a, best_score
