def _adam_optimize(a0, n, deadline):
    # start with logits x = log(a0) (or safe).
    x = np.log(a0 + 1e-12)  # to avoid log(0)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1  # base learning rate
    t = 0
    best_seq = None
    best_score = float('inf')
    # temperature schedule
    alpha = 1.0
    alpha_mul = 1.03
    alpha_max = 1e5

    size_fft = 1 << ((3 * n - 2) - 1).bit_length()

    while time.time() < deadline:
        t += 1
        # compute a
        max_x = np.max(x)
        exps = np.exp(x - max_x)  # stable
        a = exps / np.sum(exps)

        # compute convolution
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # print

        # compute gradient w.r.t a
        if alpha > 2000.0:  # use hard max subgradient
            # find indices where b is within small epsilon of max
            idxs = np.where(b >= cur_max - 1e-12)[0]
            # sum subgradients for each index
            grad = np.zeros_like(a)
            for k in idxs:
                grad += _subgradient_max(a, k)
        else:
            # softmax gradient
            # b may be length 2n-1
            # compute softmax weights
            shift = b - b.max()
            w = np.exp(alpha * shift)
            w_sum = w.sum()
            w = w / w_sum
            # compute grad via convolution of w and reversed a
            a_rev = a[::-1]
            W = np.fft.rfft(w, n=size_fft)
            A_rev = np.fft.rfft(a_rev, n=size_fft)
            conv = np.fft.irfft(W * A_rev, n=size_fft)
            # extract slice
            start = n - 1
            grad = 2.0 * conv[start:start + n]

        # map gradient to logits
        dot_ag = np.dot(a, grad)
        grad_x = a * (grad - dot_ag)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        x += step

        # temperature increase
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional jitter
        if t % 2000 == 0 and np.random.rand() < 0.1:
            x += np.random.randn(n) * 1e-4

    return best_seq, best_score

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _beta_skewed_distribution(n: int) -> np.ndarray:
    """Beta(0.5,0.5)‑skewed distribution – heavy at the ends."""
    a = np.random.beta(0.5, 0.5, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return _arcsine_distribution(n)
    return a / s


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create a probability vector of length *n*."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "beta":
        a = _beta_skewed_distribution(n)
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "prev":
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if a.size != n:
                raise ValueError
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        except Exception:
            a = _arcsine_distribution(n)
    else:
        a = _arcsine_distribution(n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(a∗a) at index *k* (i.e. 2·a_{k‑i} for each i)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray,
                      b: np.ndarray,
                      alpha: float,
                      size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b) w.r.t. a using the soft‑max weights.
    Returns a vector of length len(a).
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    grad = 2.0 * conv[start:start + a.shape[0]]
    return grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex (logits → soft‑max) using a
    temperature‑controlled soft‑max surrogate for the ℓ∞‑norm of the
    self‑convolution.  Returns the best sequence seen and its *raw*
    2·n·max(b) score (without the denominator because the sum is 1).
    """
    # logits – any positive vector works as a starting point
    x = np.log(a0 + 1e-12)

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2                     # base learning rate for logits

    # temperature schedule for the soft‑max surrogate
    alpha = 1.0
    alpha_mul = 1.03
    alpha_max = 1e5
    switch_alpha = 2_000.0

    # size needed for the gradient‑via‑fft (w has length 2n‑1, a_rev length n)
    size_fft = 1 << ((3 * n - 2) - 1).bit_length()

    best_seq = None
    best_raw_score = float("inf")
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- current point on simplex -------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # stable soft‑max
        a = exp_shift / exp_shift.sum()

        # ---- convolution and raw score -------------------------------------------------
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_raw = 2.0 * n * cur_max
        if cur_raw < best_raw_score - 1e-14:
            best_raw_score = cur_raw
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} raw={best_raw_score:.6f}")

        # ---- gradient w.r.t. a ---------------------------------------------------------
        if alpha < switch_alpha:
            # soft‑max surrogate (smooth)
            grad_a = _softmax_gradient(a, b, alpha, size_fft)
        else:
            # hard max – use sub‑gradient (may be many ties, average them)
            idxs = np.where(b >= cur_max - 1e-12)[0]
            grad_a = np.zeros_like(a)
            for k in idxs:
                grad_a += _subgradient_max(a, k)

        # ---- map to logits gradient (soft‑max Jacobian) --------------------------------
        dot_ag = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot_ag)

        # ---- Adam update ---------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        x += step

        # ---- temperature annealing ------------------------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # ---- occasional random jitter to escape plateaus -------------------------------
        if it % 3000 == 0 and np.random.rand() < 0.15:
            x += np.random.randn(n) * 1e-4

    return best_seq, best_raw_score


# ----------------------------------------------------------------------
def _local_refinement(a_start: np.ndarray,
                      n: int,
                      deadline: float) -> tuple[np.ndarray, float]:
    """Tiny random‑move hill‑climbing around *a_start*."""
    a = a_start.copy()
    best_seq = a.copy()
    best_raw = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    while time.time() < deadline:
        it += 1
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = a[i] * np.random.rand() * 0.12   # move up to 12 % of source mass
        if eps <= 0.0:
            continue
        a_cand = a.copy()
        a_cand[i] -= eps
        a_cand[j] += eps
        if a_cand[i] < 0.0:
            continue
        cur_raw = 2.0 * n * np.max(_fft_self_convolve(a_cand))
        if cur_raw < best_raw:
            best_raw = cur_raw
            best_seq = a_cand.copy()
            a = a_cand
            print(f"[{time.time():.2f}] local improve it={it} raw={best_raw:.6f}")

    return best_seq, best_raw


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s budget and returns
    the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0        # safety margin

    # ------------------------------------------------------------------
    # Base length – reuse previous solution if available
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 8000

    # Candidate lengths (allow a bit larger values – still cheap FFT)
    length_factors = (0.5, 0.75, 1.0, 1.5, 2.0, 3.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 100_000
    })
    if not lengths:
        lengths = [base_n]

    overall_best_seq = None
    overall_best_score = float("inf")
    target_score = 1.503                       # stop early if reached

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = deadline - time.time()
        if remaining <= 8.0:
            break
        # give each length a fixed share of the remaining budget
        slice_deadline = time.time() + min(remaining * 0.25, 40.0)

        for init_kind in ("prev", "arcsine", "beta", "random", "uniform"):
            if time.time() >= slice_deadline:
                break

            a0 = _initial_sequence(n, init_kind)

            seq, raw_score = _adam_optimize(a0, n, slice_deadline)

            # official evaluation (more robust)
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                final_score = raw_score / (seq.sum() ** 2)      # fallback

            if final_score < overall_best_score:
                overall_best_score = final_score
                overall_best_seq = seq.copy()
                print(
                    f"*** New global best → len={n} init={init_kind} "
                    f"score={overall_best_score:.6f}"
                )
            if overall_best_score <= target_score:
                break
        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # Final local refinement (if any time left)
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n = len(overall_best_seq)
        seq, raw = _local_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = raw / (seq.sum() ** 2)
        if final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq.copy()
            print(f"*** After local refinement score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # safety net – should never be triggered
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(
        f"Final best score: {overall_best_score:.6f} "
        f"(length = {len(overall_best_seq)})"
    )
    return overall_best_seq.tolist()
