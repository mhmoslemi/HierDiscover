"""
Optimisation of a non‑negative step function for the self‑convolution peak.

The algorithm proceeds in three stages:

1. **Rich parametric search** – generate thousands of candidate shapes
   (inverse‑sqrt, power‑law, exponential, Gaussian, cosine, beta‑shaped,
   random block, mixed Gaussian, etc.).  Each candidate is projected onto the
   simplex {a≥0, Σa = √(2 n)} and its exact score (the quantity returned by
   `evaluate_sequence`) is computed with an FFT‑based convolution.  The best
   K candidates are retained.

2. **Refinement** – for each of the retained candidates a two‑phase
   optimiser is applied:
   * *Adam* on a soft‑max approximation of the max‑convolution, with a slow
     temperature decay and projection onto the simplex.
   * *Mirror‑descent* (exponential update) on the same smoothed objective.
   The best point ever seen is kept.

3. **Final local polishing** – a fast random‑block sweep followed by a tiny
   hill‑climbing move (transfer a random ε from one coordinate to another)
   is performed under the remaining time budget.

All operations respect the security checks of `evaluate_sequence`.  The
function returns the best list of heights found before the 1000 s budget
expires.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution of u and v using FFT (size_fft is a power of two)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t a.
    (w must be non‑negative and Σ w = 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = len(w)+len(a)-1
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


def _quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective (without security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

def _adam_refine(a0: np.ndarray,
                 deadline: float,
                 n: int,
                 size_fft: int,
                 target_sum: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a soft‑max approximation of the peak.
    Returns the best point seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # learning‑rate scale
    eta0 = target_sum * 0.4 / n

    # initialise temperature
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.6)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ----- soft‑max weights -----
        shifted = conv - conv.max()          # ≤ 0
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)   # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient -----
        grad = _weighted_gradient(a, w, size_fft)

        # ----- Adam update -----
        eta = eta0 / (1.0 + it * 1e-5)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection -----
        a_new = _project_to_simplex(a_new, target_sum)

        # occasional random kick
        if it % 4000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= 0.9999       # slower cooling
        it += 1

    return best_a, best_score


def _mirror_refine(a0: np.ndarray,
                   deadline: float,
                   n: int,
                   size_fft: int,
                   target_sum: float,
                   max_iters: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimiser (exponential update) on the same smoothed loss.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _quick_score(a, n, size_fft)

    T = 1.0   # temperature for the soft‑max
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _self_convolution(a, size_fft)
        shifted = conv - conv.max()
        exp_arg = shifted / max(T, 1e-8)
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = _weighted_gradient(a, w, size_fft)

        # diminishing step size
        eta = (target_sum * 0.6) / (n * math.sqrt(it + 1))

        a = a * np.exp(-eta * grad)        # mirror step (KL geometry)
        s = a.sum()
        if s > 0:
            a = a / s * target_sum         # project onto simplex (exact)
        else:
            a = np.full_like(a, target_sum / n)

        cur_score = _quick_score(a, n, size_fft)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        T *= 0.99995
    return best_a, best_score


def _local_hill_climb(a: np.ndarray,
                      current_score: float,
                      deadline: float,
                      n: int,
                      size_fft: int,
                      target_sum: float) -> tuple[np.ndarray, float]:
    """
    Tiny hill‑climbing: move a random ε from one coordinate to another.
    Accept only improvements.
    """
    best_a = a.copy()
    best_score = current_score
    while time.time() < deadline - 0.02:
        i, j = np.random.choice(n, size=2, replace=False)
        # max ε we can transfer while staying in [0,1000]
        eps_max = min(best_a[i], 1000.0 - best_a[j])
        if eps_max <= 1e-12:
            continue
        eps = np.random.rand() * eps_max
        cand = best_a.copy()
        cand[i] -= eps
        cand[j] += eps
        # no need to project – constraints already respected
        sc = _quick_score(cand, n, size_fft)
        if sc < best_score:
            best_score = sc
            best_a = cand
        # optional early break when time is short
    return best_a, best_score

    # 2. REFINEMENT
    global_best_a = best_candidates[0][1].copy()
    global_best_score = best_candidates[0][0]

    for rank, (raw_score, a0) in enumerate(best_candidates):
        if time.time() >= deadline:
            break
        print(f"[refine start] candidate #{rank+1} (raw score {raw_score:.6f})")
        remaining = deadline - time.time()
        max_iters = int(min(200_000, remaining * 1300))

        # Adam phase
        a_adam, score_adam = _adam_refine(a0, deadline, n, size_fft,
                                          target_sum, max_iters=max_iters)
        if score_adam < global_best_score:
            global_best_score = score_adam
            global_best_a = a_adam.copy()
            print(f"[global] new best after Adam = {global_best_score:.6f}")

        # Mirror‑descent phase (shorter)
        if time.time() < deadline:
            remaining = deadline - time.time()
            max_iters_md = int(min(80_000, remaining * 800))
            a_md, score_md = _mirror_refine(a_adam, deadline, n, size_fft,
                                            target_sum, max_iters=max_iters_md)
            if score_md < global_best_score:
                global_best_score = score_md
                global_best_a = a_md.copy()
                print(f"[global] new best after Mirror = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. FINAL RANDOM BLOCK SWEEP
    while time.time() < deadline:
        nb = np.random.randint(2, 8)
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.1
        cand = np.empty(n, dtype=float)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        sc = _quick_score(cand, n, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")

        # break if very little time left
        if time.time() + 0.02 > deadline:
            break

    # ------------------------------------------------------------------
    # 4. LOCAL HILL‑CLIMB POLISHING
    if time.time() < deadline - 0.2:
        global_best_a, global_best_score = _local_hill_climb(
            global_best_a, global_best_score, deadline, n, size_fft, target_sum
        )
        print(f"[hill climb] best score after polishing = {global_best_score:.6f}")

    print(f"[final] best score = {global_best_score:.6f}")
    # Convert to Python list of floats as required by the interface
    return global_best_a.tolist()
