def _hard_subgradient(y, n, S):
    p = _softmax(y) # probabilities sum 1
    a = S * p
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = mask.sum()
    w = np.zeros_like(conv)
    w[mask] = 1.0/cnt
    conv_wr = np.convolve(w, a[::-1]) # length 3n-2
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return grad_y, max_c

def adam_optimize(y, max_iters=5000):
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = S * 0.5 / n
    for it in range(max_iters):
        if time.time() > deadline: break
        # compute gradient and max conv
        grad_y, max_c = _hard_subgradient(y, n, S)
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        lr = lr0 / (1 + 1e-5 * it)  # decaying
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = S * _softmax(y)
        sc = _score(a)
        if sc < best_score: best_score = sc; best_a = a.copy()

"""
Propose a low‑score step‑function.

The algorithm works in three phases:

1.  **Smooth‑max L‑BFGS** –  a temperature‑continuation scheme
    minimises a smooth approximation of the infinity‑norm of the
    self‑convolution.  Several diversified initialisations (uniform,
    exponentials, classic window functions, random Gaussian) are used
    and each is cooled down to a very small temperature.

2.  **Hard‑max Adam descent** –  the true objective (maximal convolution
    value) is differentiable almost everywhere.  Using the sub‑gradient
    of the hard max a few thousand Adam steps are taken, staying on the
    simplex via the soft‑max parametrisation.

3.  **Random‑block polishing & monotone heuristics** –  after the
    gradient phase a cheap random‑block search and a sorting heuristic
    are applied to squeeze out any remaining improvement.

All operations respect the supplied time budget (default 1000 s) and
the best sequence found is returned as a list of floats.  The code
contains no closures, lambdas or file I/O and runs with only numpy,
scipy and the already‑imported ``evaluate_sequence`` function.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _smooth_obj(y: np.ndarray, mu: float, n: int, S: float):
    """
    Value and gradient of the smooth‑max surrogate:
        fμ(y) = max(conv) + μ·log Σ_k exp((conv_k‑max)/μ)
    where conv = a*a and a = S·softmax(y).
    """
    p = _softmax(y)                 # Σp = 1, p ≥ 0
    a = S * p                       # Σa = S
    conv = np.convolve(a, a)        # length 2·n‑1
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # overflow guard
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                                           # exact max
        mask = conv == max_c
        cnt = mask.sum()
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a (sub‑gradient of weighted sum)
    conv_wr = np.convolve(w, a[::-1])               # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]          # slice of length n

    inner = np.dot(g_a, p)                          # scalar
    grad_y = S * p * (g_a - inner)                  # chain rule

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _smooth_obj(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _hard_subgradient(y: np.ndarray, n: int, S: float):
    """
    Sub‑gradient of the true hard max (μ = 0) and the associated
    maximal convolution value.
    Returns (grad_y, max_conv).
    """
    p = _softmax(y)
    a = S * p
    conv = np.convolve(a, a)
    max_c = conv.max()
    mask = conv == max_c
    cnt = mask.sum()
    w = np.zeros_like(conv, dtype=float)
    w[mask] = 1.0 / cnt
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return grad_y, max_c


# ----------------------------------------------------------------------
def _score_of(a: np.ndarray) -> float:
    """Fast evaluation of the official score (identical to evaluate_sequence)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    s = a.sum()
    # protect against division by zero – the algorithm never generates s = 0
    return float(2.0 * n * max_c / (s * s))
