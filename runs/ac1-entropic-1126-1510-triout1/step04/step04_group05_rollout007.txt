"""
Improved optimizer for the step‑function self‑convolution problem.

Algorithm:
1. Initialise a feasible sequence (uniform / triangular / Dirichlet / reference).
2. Perform an extensive smooth‑max optimisation with L‑BFGS‑B.
   The smoothing parameter μ is reduced geometrically (factor 0.85) and a
   large number of inner iterations (500) are allowed, which gives the
   optimiser much more freedom than the previous baseline.
3. Refine the result by solving a linear programme
   maximise Σ b_j  subject to  a * b ≤ M ,  b ≥ 0.
   The optimal b is scaled back to the required ‖b‖₁ = S and replaces a.
   Repeating the LP a few times pushes the maximum of the self‑convolution
   lower than gradient‑based methods alone.
4. A short projected sub‑gradient descent polish finishes the search.

Multiple random restarts are performed while time permits; the best sequence
found is returned.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x >= 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (soft‑max) objective value and its gradient w.r.t. y.
    The underlying variable a = S * softmax(y) satisfies Σ a = S and a ≥ 0.
    """
    p = _softmax(y)                # probability vector, Σ p = 1
    a = S * p                      # actual heights

    # full self‑convolution (length 2·n‑1)
    conv = np.convolve(a, a)
    max_c = conv.max()

    if mu > 0.0:
        # smooth max via log‑sum‑exp
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # exact max (mu==0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient of the weighted sum w·(a*a) w.r.t. a
    # conv_wr = w * a_rev  (convolution)
    a_rev = a[::-1]
    conv_wr = np.convolve(w, a_rev)          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]   # keep the n components that align with a

    # chain rule from a to y
    inner = np.dot(g_a, p)                   # scalar = Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for FFT‑based convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


# ----------------------------------------------------------------------
def _l_bfgs_optimize(a_start: np.ndarray, S: float,
                     budget: float, start_time: float,
                     maxiter_per_mu: int = 500,
                     mu_factor: float = 0.85,
                     mu_min: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Smooth‑max optimisation with L‑BFGS‑B.
    The smoothing parameter μ is decreased geometrically.
    """
    n = len(a_start)
    # initialise y = log(p) where p = a_start / S
    eps = 1e-12
    p0 = a_start / S
    y = np.log(p0 + eps).astype(np.float64)

    # initial μ based on current max convolution
    cur_conv = np.convolve(a_start, a_start)
    mu = max(1.0, float(cur_conv.max()))

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    it_mu = 0

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor
        it_mu += 1

    return best_a, best_score

def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true hard max (no smoothing).
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # step‑size parameters
    eta0 = S * 0.5 / n            # rough magnitude of a_i
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 5000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # gradient of the hard max
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _lp_refine(a_start: np.ndarray, S: float,
               budget: float, start_time: float,
               max_iters: int = 10) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    For the current a we solve:
        maximise Σ b_j
        s.t.   a * b ≤ M ,  b ≥ 0
    where M = max_k (a*a)_k.
    The optimal b is scaled back to ℓ¹‑norm S and replaces a.
    The process is repeated a few times or until time runs out.
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    M = _max_convolution(a, size)

    it = 0
    while it < max_iters and time.time() - start_time < budget:
        f = a
        # Build sparse matrix A with entries A[k, j] = f[k-j] (if 0≤k-j<n)
        # Equivalent to a collection of shifted diagonals.
        diags = [f] * n
        offsets = list(range(n))                       # 0 .. n‑1
        A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
        d = np.full(2 * n - 1, M, dtype=float)

        # maximise Σ b  → minimise –Σ b
        c = -np.ones(n, dtype=float)

        res = sp.optimize.linprog(c, A_ub=A, b_ub=d,
                                 bounds=(0, None),
                                 method="highs",
                                 options={"presolve": True, "time_limit": budget - (time.time() - start_time)})
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b0 = res.x
        sum_b = b0.sum()
        if sum_b <= 0:
            print("[LP] zero solution, abort")
            break

        scale = S / sum_b
        g = b0 * scale

        # Evaluate new candidate
        sc = evaluate_sequence(g.tolist())
        M_new = _max_convolution(g, size)

        if sc < best_score:
            best_score = sc
            best_a = g.copy()
            print(f"[LP improve] iter {it} score = {best_score:.6f}")

        # Prepare for next iteration
        a = g
        M = M_new
        it += 1

    return best_a, best_score

def _initialize_sequence(kind: str, n: int, S: float, ref: np.ndarray | None,
                         rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial feasible sequence of length n.
    Supported kinds: 'ref', 'uniform', 'triangular', 'exp', 'random'.
    """
    if kind == "ref" and ref is not None:
        a = ref.copy()
        a = _project_to_simplex(a, S)
        return a
    if kind == "uniform":
        return np.full(n, S / n, dtype=float)
    if kind == "triangular":
        t = np.arange(n)
        a = np.minimum(t + 1, n - t)
        a = a.astype(float)
        a = S * a / a.sum()
        return a
    if kind == "exp":
        decay = 0.003
        a = np.exp(-decay * np.arange(n))
        a = S * a / a.sum()
        return a
    # default: random Dirichlet (Gamma(1) variables)
    gamma = rng.gamma(shape=1.0, scale=1.0, size=n)
    a = S * gamma / gamma.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:

    Search for a low‑score step‑function within the allotted time.
    Returns the best sequence found (list of non‑negative floats).
    """
    np.random.seed(seed)

"""
Improved optimizer for the step‑function self‑convolution problem.

Algorithm:
1. Initialise a feasible sequence (uniform / triangular / exponential /
   Dirichlet / reference).
2. Perform an extensive smooth‑max optimisation with L‑BFGS‑B.  The smoothing
   parameter μ is decreased geometrically (factor 0.85) and a large number of
   inner iterations (600) are allowed, giving the optimiser far more freedom
   than the baseline.
3. Refine the result by solving a linear programme

        maximise Σ b_j
        s.t.   a * b ≤ M ,  b ≥ 0 ,

   where a is the current sequence, M = max (a*a) and * denotes discrete
   convolution.  The optimal b is scaled back to the required ℓ¹‑norm and
   replaces a.  Repeating the LP a few times pushes the maximum of the
   self‑convolution lower than gradient‑based methods alone.
4. A short projected sub‑gradient descent polish finishes the search.

Multiple random restarts are performed while time permits; the best sequence
found is returned.
"""

import time
import math
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
from typing import Optional

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum z must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (soft‑max) objective value and its gradient w.r.t. y.
    Underlying variable a = S * softmax(y) satisfies Σ a = S, a ≥ 0.
    """
    p = _softmax(y)                # probability vector, Σ p = 1
    a = S * p                      # actual heights

    conv = np.convolve(a, a)       # full self‑convolution, length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient of weighted sum w·(a*a) w.r.t. a
    a_rev = a[::-1]
    conv_wr = np.convolve(w, a_rev)          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]   # central n components

    # chain rule from a to y
    inner = np.dot(g_a, p)                   # Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        f, g = _compute_f_and_grad(y, self.mu, self.n, self.S)
        return f, g


# ----------------------------------------------------------------------
def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for FFT‑based convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


# ----------------------------------------------------------------------
def _l_bfgs_optimize(a_start: np.ndarray, S: float,
                     budget: float, start_time: float,
                     maxiter_per_mu: int = 600,
                     mu_factor: float = 0.85,
                     mu_min: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Smooth‑max optimisation with L‑BFGS‑B.
    The smoothing parameter μ is decreased geometrically.
    """
    n = len(a_start)
    eps = 1e-12
    p0 = a_start / S
    y = np.log(p0 + eps).astype(np.float64)

    # initialise μ from the current max‑convolution
    cur_conv = np.convolve(a_start, a_start)
    mu = max(1.0, float(cur_conv.max()))

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())
    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


# ----------------------------------------------------------------------
def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true hard max (no smoothing).
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    eta0 = S * 0.5 / n          # initial step magnitude
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 5000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # gradient of the hard max
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _lp_refine(a_start: np.ndarray, S: float,
               budget: float, start_time: float,
               max_iters: int = 6) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.

    For the current a we solve:
        maximise Σ b_j
        s.t.   a * b ≤ M ,   b ≥ 0
    where M = max (a*a).  The optimal b is scaled to ℓ¹‑norm S and
    becomes the new a.  The process is repeated a few times.
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    M = _max_convolution(a, size)

    it = 0
    while it < max_iters and time.time() - start_time < budget:
        f = a
        # Build sparse matrix A with entries A[k, j] = f[k-j]  (if 0≤k-j<n)
        diags = [f] * n
        offsets = list(range(n))                       # 0 … n‑1
        A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
        d = np.full(2 * n - 1, M, dtype=float)

        # maximise Σ b  → minimise –Σ b
        c = -np.ones(n, dtype=float)

        res = opt.linprog(c, A_ub=A, b_ub=d,
                         bounds=(0, None),
                         method="highs",
                         options={"presolve": True})
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b0 = res.x
        sum_b = b0.sum()
        if sum_b <= 0:
            print("[LP] zero solution, abort")
            break

        scale = S / sum_b
        g = b0 * scale

        sc = evaluate_sequence(g.tolist())
        M_new = _max_convolution(g, size)

        if sc < best_score:
            best_score = sc
            best_a = g.copy()
            print(f"[LP improve] iter {it} score = {best_score:.6f}")

        a = g
        M = M_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
def _initialize_sequence(kind: str,
                         n: int,
                         S: float,
                         ref: Optional[np.ndarray],
                         rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial feasible sequence.
    Kinds: 'ref', 'uniform', 'triangular', 'exp', 'random'.
    """
    if kind == "ref" and ref is not None:
        a = ref.copy()
        a = _project_to_simplex(a, S)
        return a
    if kind == "uniform":
        return np.full(n, S / n, dtype=float)
    if kind == "triangular":
        t = np.arange(n)
        a = np.minimum(t + 1, n - t)
        a = S * a.astype(float) / a.sum()
        return a
    if kind == "exp":
        decay = 0.003
        a = np.exp(-decay * np.arange(n))
        a = S * a / a.sum()
        return a
    # random Dirichlet (Gamma(1) variables)
    gamma = rng.gamma(shape=1.0, scale=1.0, size=n)
    a = S * gamma / gamma.sum()
    return a
