"""
Propose a non‑negative step function that minimises the self‑convolution peak.

Algorithm
---------
1. **Exploration** – generate a large and diverse set of analytic families
   (uniform, inverse‑sqrt, power‑law, exponential, sinusoidal, Gaussian,
   triangular, beta‑shaped, arcsine, random block, and random convex
   combinations of these).  Each candidate is projected onto the simplex
   {a ≥ 0, Σa = √(2 n)} and evaluated with an exact FFT‑based score.
   The best `K` raw candidates are retained.

2. **Heavy refinement** – take the single best raw candidate and run a
   long Adam‑based optimisation on a smooth approximation of the max
   (soft‑max with a slowly cooled temperature).  The optimisation works
   directly in the original space with Euclidean projection onto the
   simplex after every step.  The temperature decays very slowly and a
   small random kick is added every few thousand iterations to escape
   plateaus.

3. **Final sweep** – while any time is left, perform a fast random‑block
   search (few constant blocks) and keep any improvement.

The function respects the 1000 s budget and returns the best sequence found.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of u and v using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    (w must sum to 1.)
    """
    a_rev = a[::-1]
    conv = _conv_fft(w, a_rev, size_fft)          # length = 3·n‑2
    start = len(w) - len(a)                       # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact score used during the search (no security checks)."""
    conv = _self_convolution(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _refine_candidate_adam(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of a smooth max‑convolution surrogate.
    Returns the best sequence observed and its (exact) score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = quick_score(a, n, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Base learning rate (scaled to problem size)
    eta0 = target_sum * 0.5 / n

    # Initialise temperature from the spread of the first convolution
    conv = _self_convolution(a, size_fft)
    T = max(1e-4, (conv.max() - conv.min()) * 0.5)

    # Cool slowly – this factor proved useful in experiments
    cooling = 0.9999

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---------- current evaluation ----------
        conv = _self_convolution(a, size_fft)
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---------- soft‑max weights ----------
        shifted = conv - conv.max()          # ≤ 0
        exp_arg = shifted / max(T, 1e-12)
        exp_arg = np.clip(exp_arg, -50, 0)   # avoid overflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient ----------
        grad = _weighted_gradient(a, w, size_fft)

        # ---------- Adam step ----------
        eta = eta0 / (1.0 + it * 1e-5)        # simple decay
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- projection ----------
        a_new = _project_to_simplex(a_new, target_sum)

        # ---------- occasional random kick ----------
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a_new = a_new + noise
            a_new = _project_to_simplex(a_new, target_sum)

        a = a_new
        T *= cooling
        it += 1

    return best_a, best_score

        nb = np.random.randint(2, 9)                     # 2‑8 blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        cand = np.empty(n, dtype=float)
        start = 0
        vals = np.random.rand(nb) + 0.1
        for i, cut in enumerate(cuts):
            cand[start:cut] = vals[i]
            start = cut
        cand[start:] = vals[-1]

        cand = _project_to_simplex(cand, target_sum)
        sc = quick_score(cand, n, size_fft)
        if sc < global_best_score:
            global_best_score = sc
            global_best_a = cand.copy()
            print(f"[final block] improved to {global_best_score:.6f}")

        # break early if only a tiny sliver of the budget remains
        if time.time() + 0.02 > deadline:
            break

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
