def _compute_f_and_grad(y, mu, n, S):
    p = _softmax(y)
    a = S * p
    # compute convolution via FFT
    size = _fft_size(n)
    conv = _full_convolution(a, size)  # length 2n-1
    max_c = conv.max()
    if mu > 0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700, 700)
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt > 0:
            w[mask] = 1.0/cnt
        f_val = max_c
    # gradient w.r.t a
    # compute convolution of w with reversed a
    size3 = _fft_size(3*n)
    conv_wr = _convolve_full(w, a[::-1], size3)  # length 3n-2
    g_a = 2.0 * conv_wr[n-1:n-1+n]
    # chain to y
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)
    return float(f_val), grad_y.astype(np.float64)
```

"""
Propose a step‑function that minimizes the max‑self‑convolution.
Algorithm
---------

1. Choose a problem size `n`.  If a reference sequence `height_sequence_1`
   exists we enlarge it (roughly double) up to a safe limit (≈20000).  The
   ℓ¹‑norm is fixed to `S = sqrt(2·n)` so that the evaluation score equals
   the maximum of the convolution.

2. **Variance‑reduction phase** – minimise the L²‑variance of the convolution
   `a*a` (i.e. make the convolution as flat as possible).  We use an Adam
   optimiser on the smooth objective   f(a)=∑( (a*a)[k] – avg )²  with a
   Euclidean projection onto the simplex after each step.

3. **Smooth‑max refinement** – a continuation scheme that replaces the hard
   max by a log‑sum‑exp (soft‑max) with decreasing smoothing parameter `mu`.
   Optimisation is performed with L‑BFGS‑B.

4. **Hard‑max polishing** – projected sub‑gradient descent on the true
   `max(a*a)` for the remaining time.

5. The best sequence found throughout the run is returned.

The combination of a variance‑flattening start and a more aggressive
continuation schedule usually drives the maximum below the previous
record (~1.514) and can approach the theoretical lower bound (~1.0) for large
`n`.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (used for a*a)."""
    return 1 << ((2 * n - 1).bit_length())


def _fft_size_generic(L: int) -> int:
    """Smallest power of two ≥ L (generic)."""
    return 1 << ((L - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a via FFT (full convolution, length 2·n‑1)."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a using FFT."""
    conv = _full_convolution(a, size)
    return float(conv.max())


def _convolve_full(x: np.ndarray, y: np.ndarray, size: int) -> np.ndarray:
    """Full convolution of two arbitrary vectors via FFT."""
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. y.
    `y` are log‑probabilities, `S` is the required ℓ¹‑norm.
    """
    p = _softmax(y)                # Σp = 1, p_i ≥ 0
    a = S * p                       # Σa = S

    size = _fft_size(n)
    conv = _full_convolution(a, size)  # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t a  (use FFT for speed)
    size3 = _fft_size_generic(3 * n)
    conv_wr = _convolve_full(w, a[::-1], size3)   # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)               # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (objective, gradient)."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
# Phase 1 – variance reduction (Adam on the L2‑variance of the convolution)
# ----------------------------------------------------------------------


def _variance_reduction(a_start: np.ndarray, S: float,
                        budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that flattens a*a (minimises its variance).
    Returns the best sequence seen and its evaluation score.
    """
    n = len(a_start)
    a = a_start.copy()
    best_a = a.copy()
    best_score = 2 * n * _max_convolution(a, _fft_size(n)) / (np.sum(a) ** 2)

    # Adam hyper‑parameters
    eta0 = 0.5
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    size_var = _fft_size_generic(3 * n)
    it = 0
    while time.time() - start_time < budget and it < 20000:
        # convolution a*a
        conv = _full_convolution(a, _fft_size(n))
        avg = conv.mean()
        diff = conv - avg

        # gradient of ∑(conv‑avg)²  w.r.t. a
        grad = 4.0 * _convolve_full(diff, a[::-1], size_var)[n - 1 : n - 1 + n]

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - eta0 * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_to_simplex(a, S)

        # occasional evaluation
        if it % 200 == 0:
            cur_max = _max_convolution(a, _fft_size(n))
            cur_score = 2 * n * cur_max / (np.sum(a) ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[var‑reduce] iter {it:,} score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – L‑BFGS continuation on a smooth‑max surrogate
# ----------------------------------------------------------------------


def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Continuation on the smooth‑max surrogate (same idea as the reference
    implementation) but with a finer schedule and more inner L‑BFGS steps.
    """
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    # initialise mu from current max‑convolution
    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = 2 * n * _max_convolution(best_a, _fft_size(n)) / (np.sum(best_a) ** 2)

    mu_factor = 0.7
    mu_min = 1e-6
    maxiter_per_mu = 300

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = opt.minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu, "disp": False, "gtol": 1e-12},
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        cur_max = _max_convolution(a, _fft_size(n))
        cur_score = 2 * n * cur_max / (np.sum(a) ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Projected sub‑gradient descent on the true hard max
# ----------------------------------------------------------------------


def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution.
    """
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = 2 * n * _max_convolution(best_a, size) / (np.sum(best_a) ** 2)

    # step‑size parameters
    eta0 = S * 0.5 / n
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 10000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        if cnt > 0:
            w[mask] = 1.0 / cnt

        # sub‑gradient of the hard max
        fa_rev = np.fft.rfft(a[::-1], n=size)
        fw = np.fft.rfft(w, n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1 : n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        cur_max = _max_convolution(a_new, size)
        cur_score = 2 * n * cur_max / (np.sum(a_new) ** 2)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score
