"""
Improved step‑function search.

The aim is to minimise  2 n·max(a⋆a) / (∑a)²  with a_i ≥ 0.
The algorithm consists of three stages

1. **Smoothed max optimisation** –  minimise a smooth
   approximation of the ∞‑norm of the auto‑convolution using
   L‑BFGS‑B on the unconstrained parameters `y` (soft‑max
   representation on the ℓ¹‑simplex).  The continuation uses a
   slower cooling schedule (μ←0.85·μ) and more inner iterations
   than the baseline.

2. **Coarse random‑block search** –  generate many piece‑wise
   constant candidates (2–12 blocks), project them onto the
   simplex and keep any improvement.

3. **Greedy local mass‑transfer** – repeatedly move a small
   amount of mass from a heavy index to a light one and keep the
   move if the evaluate_sequence score decreases.  This cheap
   hill‑climbing can exploit the remaining budget to push the
   bound below the previous record (≈1.5148).

All stages respect the 1000 s wall‑time limit; the routine returns
the best sequence found.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int,
                       S: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (soft‑max) surrogate value and gradient w.r.t. free
    parameters y (the unconstrained logits of a soft‑max on the simplex).
    """
    p = _softmax(y)               # Σ p_i = 1, p_i ≥ 0
    a = S * p                     # Σ a_i = S

    conv = np.convolve(a, a)      # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:                         # hard max (μ = 0)
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. the heights a
    conv_wr = np.convolve(w, a[::-1])
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    inner = np.dot(g_a, p)               # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)       # chain rule through soft‑max

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable required by scipy.optimize – returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence()
    score.  The routine respects the given time budget and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5      # a small safety margin

    # ------------------------------------------------------------------
    # problem size ----------------------------------------------------
    try:
        # a reference from the previous run (if present)
        ref = np.asarray(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        n = ref.shape[0]
        have_ref = True
    except Exception:
        # sensible default when no reference is available
        n = 3500
        have_ref = False

    S = math.sqrt(2.0 * n)                         # target ℓ¹‑norm

    # ------------------------------------------------------------------
    # bookkeeping ------------------------------------------------------
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    def _maybe_update(a_vec: np.ndarray) -> None:
        """Evaluate and keep the best sequence seen so far."""
        nonlocal best_score, best_a
        sc = evaluate_sequence(a_vec.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            print(f"[improve] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Smooth‑max optimisation (multiple restarts) -------------------
    init_ys = []

    # uniform start (y = 0)
    init_ys.append(np.zeros(n, dtype=np.float64))

    # decreasing exponential
    alpha = rng.uniform(0.001, 0.01) * n / 1000.0
    y_exp = -alpha * np.arange(n, dtype=np.float64)
    init_ys.append(y_exp)

    # increasing exponential (mirror of the above)
    y_rev = -alpha * np.arange(n - 1, -1, -1, dtype=np.float64)
    init_ys.append(y_rev)

    # random Gaussian
    init_ys.append(rng.standard_normal(n) * 0.4)

    # random uniform
    init_ys.append(rng.uniform(-1.0, 1.0, size=n))

    # reference (if we have one)
    if have_ref:
        a_ref = _project_to_simplex(ref, S)
        eps = 1e-12
        init_ys.append(np.log(a_ref + eps))

    mu_factor = 0.85          # slower cooling than the baseline
    mu_min = 1e-8
    maxiter_per_mu = 500     # more inner L‑BFGS steps per temperature

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{len(init_ys)}] initialising")
        y = y0.copy()

        a_cur = S * _softmax(y)
        conv_cur = np.convolve(a_cur, a_cur)
        mu = max(conv_cur.max(), 1.0)   # safeguard against a zero start

        # continuation loop
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)

            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False},
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor

    # ------------------------------------------------------------------
    # 2) Coarse block‑wise random search --------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        max_trials = int(min(20000, remaining * 200))
        print(f"[block] performing up to {max_trials} random block trials")
        for _ in range(max_trials):
            nb = rng.integers(2, 13)                # 2…12 blocks
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # 3) Monotone‑sorting heuristic (helps flatten the convolution) -----
    if time.time() < deadline:
        desc = np.sort(best_a)[::-1]
        desc = _project_to_simplex(desc, S)
        _maybe_update(desc)

        asc = np.sort(best_a)
        asc = _project_to_simplex(asc, S)
        _maybe_update(asc)

    # ------------------------------------------------------------------
    # 4) Greedy local mass‑transfer (hill‑climbing) ----------------------
    if time.time() < deadline:
        cur_a = best_a.copy()
        cur_score = best_score
        # pre‑compute a list of “heavy” indices (top 5 % by value)
        heavy_frac = 0.05
        light_frac = 0.20

        while time.time() < deadline:
            # periodically refresh the heavy / light index pools
            if rng.random() < 0.01:
                sorted_idx = np.argsort(-cur_a)
                n_heavy = max(1, int(heavy_frac * n))
                n_light = max(1, int(light_frac * n))
                heavy_idx = sorted_idx[:n_heavy]
                light_idx = sorted_idx[-n_light:]
            # source = a fairly large entry, destination = a small one
            i = rng.choice(heavy_idx)
            if rng.random() < 0.7:
                j = rng.choice(light_idx)
            else:
                j = rng.integers(0, n)
                while j == i:
                    j = rng.integers(0, n)

            if cur_a[i] <= 0.0:
                continue
            # move at most 30 % of the source mass
            max_delta = cur_a[i] * 0.3
            delta = max_delta * rng.random()
            if delta <= 0.0:
                continue

            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta

            # maintain feasibility (non‑negativity + sum)
            if cand[i] < 0.0:
                continue

            sc = evaluate_sequence(cand.tolist())
            if sc < cur_score:
                cur_a = cand
                cur_score = sc
                _maybe_update(cand)          # updates global best as well

    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f}")
    return best_a.tolist()
