def self_convolution_fft(a, size_fft):
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2*len(a)-1]

Note: we can compute self convolution via zero padding as above.

def weighted_gradient_fft(a, w, size_fft):
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size_fft)
    fa_rev = np.fft.rfft(a_rev, n=size_fft)
    # conv = irfft(fw * fa_rev, n=size_fft)
    conv = np.fft.irfft(fw * fa_rev, n=size_fft)
    # conv length = len(w)+len(a)-1 = 3n-2
    start = len(w) - len(a)  # = n-1
    grad = 2.0 * conv[start : start + len(a)]
    return grad

"""
Propose a non‑negative step function that minimizes the self‑convolution peak.

The algorithm consists of three phases:
1. Exploration – generate a diverse pool of candidates from analytic families,
   random block patterns and (if available) the reference sequence.
   Each candidate is projected onto the simplex Σa = √(2 n) and the best
   K are kept.
2. Refinement – each retained candidate is improved by a projected
   sub‑gradient descent on the exact max‑convolution objective.
   The sub‑gradient is the average of the gradients of all active
   (max‑attaining) convolution entries.  A diminishing step size and a
   simple momentum term are used; projection onto the simplex guarantees
   feasibility.
3. Final sweep – a quick pass of random block candidates is performed on any
   remaining time budget, updating the global best when a lower score is
   found.

All operations respect the security checks of ``evaluate_sequence``.
The function returns the best sequence discovered before the 1000 s budget
expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – put the whole mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow_2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    # keep only the linear part
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient_fft(a: np.ndarray, w: np.ndarray, size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    (w must sum to 1; all entries non‑negative.)
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size_fft)
    fa_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(fw * fa_rev, n=size_fft)
    # conv length = len(w) + len(a) - 1 = 3·n‑2
    start = len(w) - len(a)  # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact score used during the search (no security checks)."""
    conv = _self_convolution_fft(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    n: int,
    size_fft: int,
    target_sum: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for the max‑convolution objective.
    Returns the best sequence observed and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _quick_score(a, n, size_fft)

    # Simple momentum term (helps accelerate in flat regions)
    momentum = np.zeros_like(a)

    # Initial step – scaled to the problem size
    step = target_sum * 0.6 / n
    decay = 0.9995

    # Tolerance for detecting active constraints
    rel_atol = 1e-12

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _self_convolution_fft(a, size_fft)
        cur_max = conv.max()

        # Identify all indices attaining (approximately) the maximum.
        atol = max(rel_atol * cur_max, 1e-12)
        active = np.where(conv >= cur_max - atol)[0]

        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size

        grad = _weighted_gradient_fft(a, w, size_fft)

        # Normalise gradient to keep step size roughly comparable across
        # different scales.
        grad_norm = np.linalg.norm(grad) + 1e-12
        # Momentum update
        momentum = 0.9 * momentum + grad / grad_norm

        a_new = a - step * momentum
        a_new = _project_to_simplex(a_new, target_sum)

        # Evaluate
        cand_score = _quick_score(a_new, n, size_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_new.copy()

        a = a_new
        step *= decay

        # Occasional random perturbation to escape shallow plateaus
        if it % 5000 == 0 and it > 0:
            perturb = np.random.randn(n) * target_sum * 0.001
            a = _project_to_simplex(a + perturb, target_sum)

    return best_a, best_score

"""
Propose a non‑negative step function (list of floats) that minimises the
self‑convolution peak defined in ``evaluate_sequence``.

The algorithm proceeds in three stages:

1. **Exploration** – a large, diverse pool of candidates is generated from
   analytic families (inverse‑sqrt, power‑law, exponentials, trigonometric
   windows, Gaussian bumps, beta‑shaped, etc.), from random block patterns,
   and from the provided reference sequence (if it exists).  Each candidate
   is projected onto the simplex Σa = √(2 n) and then scored with an exact
   FFT‑based convolution.  The best *K* candidates are retained.

2. **Refinement** – every retained candidate is improved by a projected
   sub‑gradient descent on the true max‑convolution objective.
   The sub‑gradient is the average of the gradients of all convolution
   entries attaining the current maximum.  A simple momentum term together
   with a geometrically decaying step size is used; after each step the
   iterate is projected back onto the simplex.  The descent stops when the
   overall time budget is exhausted.

3. **Final sweep** – with any remaining time a quick pass of random block
   candidates is performed, updating the global best whenever a lower score
   is observed.

All operations respect the security checks of ``evaluate_sequence``.
The function returns the best sequence discovered before the 1000 s budget
expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow_2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_fft(u: np.ndarray, v: np.ndarray, size_fft: int) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (full length)."""
    fu = np.fft.rfft(u, n=size_fft)
    fv = np.fft.rfft(v, n=size_fft)
    conv = np.fft.irfft(fu * fv, n=size_fft)
    return conv[: len(u) + len(v) - 1]


# ----------------------------------------------------------------------
def _self_convolution_fft(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a."""
    return _conv_fft(a, a, size_fft)


# ----------------------------------------------------------------------
def _weighted_gradient_fft(a: np.ndarray, w: np.ndarray,
                          size_fft: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a   (w must sum to 1).
    Returns a vector of length len(a).
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=size_fft)
    fa_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(fw * fa_rev, n=size_fft)
    start = len(w) - len(a)          # = n‑1
    grad = 2.0 * conv[start:start + len(a)]
    return grad


# ----------------------------------------------------------------------
def _quick_score(a: np.ndarray, n: int, size_fft: int) -> float:
    """Exact objective value (no security checks)."""
    conv = _self_convolution_fft(a, size_fft)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _subgradient_descent(a0: np.ndarray,
                         deadline: float,
                         n: int,
                         size_fft: int,
                         target_sum: float,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the max‑convolution objective.

    Returns the best sequence seen and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _quick_score(a, n, size_fft)

    momentum = np.zeros_like(a)
    step = target_sum * 0.6 / n            # initial step size
    decay = 0.9995                         # geometric decay
    rel_atol = 1e-12                       # tolerance for active set

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _self_convolution_fft(a, size_fft)
        cur_max = conv.max()
        atol = max(rel_atol * cur_max, 1e-12)
        active = np.where(conv >= cur_max - atol)[0]

        w = np.zeros_like(conv)
        if active.size:
            w[active] = 1.0 / active.size

        grad = _weighted_gradient_fft(a, w, size_fft)
        grad_norm = np.linalg.norm(grad) + 1e-12
        momentum = 0.9 * momentum + grad / grad_norm

        a_new = a - step * momentum
        a_new = _project_to_simplex(a_new, target_sum)

        cand_score = _quick_score(a_new, n, size_fft)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_new.copy()

        a = a_new
        step *= decay

        # occasional random kick to avoid stagnation
        if it % 5000 == 0 and it > 0:
            noise = np.random.randn(n) * target_sum * 0.001
            a = _project_to_simplex(a + noise, target_sum)

    return best_a, best_score
