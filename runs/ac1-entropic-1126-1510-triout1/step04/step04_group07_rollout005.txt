The score is 2·n·max(a∗a)/(∑a)².  It is scale‑invariant, so we
normalise a to have ℓ¹‑norm S=√(2·n) and minimise the infinity‑norm of
the autoconvolution.  The search consists of four stages

1.  Several warm‑starts (uniform, exponentials, random Gaussian,
    reference if present) are optimised with a smooth‐max surrogate.
    The surrogate  f_μ(y) = max_k (a∗a)[k] + μ·log∑_k exp((a∗a)[k]/μ)
    is minimised by L‑BFGS‑B on the unconstrained logits y (soft‑max
    parametrisation).  The temperature μ is cooled geometrically.

2.  A projected sub‑gradient descent on the true hard max (μ=0) with a
    back‑tracking line‑search further reduces the maximal convolution
    entry.

3.  A cheap random‑block search (piecewise constant candidates) explores
    the large‑scale landscape.

4.  Finally a simple hill‑climbing phase moves tiny amounts of mass
    between two coordinates; every improvement is kept.

All stages respect a global time‑budget (default 1000 s).  The routine
returns the best sequence discovered; the returned values are already
inside the allowed range, so `evaluate_sequence` can be called safely.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = z} (z≥0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / np.sum(e)


# ----------------------------------------------------------------------
def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max value and gradient w.r.t. logits y.

    Returns (value, grad_y).  The gradient is exact for the chosen µ.
    """
    p = _softmax(y)          # Σ p_i = 1 , p_i ≥ 0
    a = S * p                # ℓ¹‑norm = S

    conv = np.convolve(a, a)               # length 2·n‑1
    max_c = conv.max()

    # smooth‑max weighting
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # safety
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight over all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient wrt a  (g_a = 2 * (w * a)_reversed )
    conv_wr = np.convolve(w, a[::-1])                # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]            # slice of length n

    # chain rule through the soft‑max
    inner = np.dot(g_a, p)                # scalar
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (value, gradient)."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _score_from_a(a: np.ndarray, n: int) -> float:
    """
    Fast evaluation of the official score, assuming all entries are already
    within the allowed interval.  Uses the exact formula from evaluate_sequence.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv_max = np.max(np.convolve(a, a))
    return float(2.0 * n * conv_max / (s * s))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """Search for a step‑function with a low evaluate_sequence score."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size ----------------------------------------------------
    try:
        # height_sequence_1 is supplied by the runner if a reference exists
        a_ref = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = a_ref.shape[0]
        have_ref = True
    except Exception:
        # no reference – use a reasonably large size (the previous best used ~3500)
        n = 3500
        have_ref = False

    S = math.sqrt(2.0 * n)          # target ℓ¹‑norm  (scale for probability simplex)

    # ------------------------------------------------------------------
    best_a = np.zeros(n, dtype=float)
    best_score = float("inf")

    # ------------------------------------------------------------------
    def _maybe_update(a_vec: np.ndarray) -> None:
        """Update incumbent using the fast score; final check with evaluate_sequence."""
        nonlocal best_score, best_a
        sc = _score_from_a(a_vec, n)
        if sc < best_score:
            best_score = sc
            best_a = a_vec.copy()
            # optional call to the official evaluator for sanity (cheap)
            try:
                official = evaluate_sequence(a_vec.tolist())
                # keep the official value if it is better (they should match)
                if official < best_score:
                    best_score = official
                    best_a = a_vec.copy()
            except Exception:
                pass
            print(f"[improve] new best (fast) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # initialise a collection of free‑parameter vectors y (logits)
    init_ys = []

    # 0 – uniform
    init_ys.append(np.zeros(n, dtype=np.float64))

    # 1 – decreasing exponential (large on the left)
    alpha = rng.uniform(0.001, 0.008) * n / 1000.0
    init_ys.append(-alpha * np.arange(n, dtype=np.float64))

    # 2 – increasing exponential (large on the right)
    init_ys.append(-alpha * np.arange(n - 1, -1, -1, dtype=np.float64))

    # 3 – random Gaussian logits
    init_ys.append(rng.standard_normal(n) * 0.5)

    # 4 – reference if present (log of the reference after projection)
    if have_ref:
        a_proj = _project_to_simplex(a_ref, S)
        eps = 1e-12
        init_ys.append(np.log(a_proj + eps))

    # ------------------------------------------------------------------
    # continuation parameters (smooth‑max)
    mu_factor = 0.6                 # slower cooling – gives more accurate solutions
    mu_min = 1e-9
    maxiter_per_mu = 400            # more inner L‑BFGS iterations per temperature

    n_restart = len(init_ys)

    for rid, y0 in enumerate(init_ys):
        if time.time() > deadline:
            break
        print(f"[restart {rid + 1}/{n_restart}] initialising")
        y = y0.copy()

        # initial temperature derived from the current max‑convolution
        a_cur = S * _softmax(y)
        mu = float(np.max(np.convolve(a_cur, a_cur)))
        mu = max(mu, 1.0)            # avoid a zero start

        # continuation loop ------------------------------------------------
        while mu > mu_min and time.time() < deadline:
            obj = _SmoothMaxObjective(mu, n, S)
            try:
                res = opt.minimize(
                    obj,
                    y,
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": maxiter_per_mu,
                             "gtol": 1e-12,
                             "disp": False}
                )
            except Exception as exc:
                print(f"[error] L‑BFGS failed: {exc}")
                break

            y = res.x
            a = S * _softmax(y)
            _maybe_update(a)

            mu *= mu_factor   # geometric cooling

    # ------------------------------------------------------------------
    # random block search – large number of cheap trials
    remaining = deadline - time.time()
    if remaining > 2.0:
        max_trials = int(min(10000, remaining * 180))
        print(f"[block] {max_trials} random block trials")
        for _ in range(max_trials):
            # pick a random number of blocks (2‑12)
            nb = rng.integers(2, 13)
            cuts = np.sort(rng.choice(n - 1, nb - 1, replace=False) + 1)
            blocks = np.split(np.arange(n), cuts)
            vals = rng.random(nb)          # raw block values
            cand = np.empty(n, dtype=float)
            for blk, v in zip(blocks, vals):
                cand[blk] = v
            cand = _project_to_simplex(cand, S)
            _maybe_update(cand)

    # ------------------------------------------------------------------
    # monotone sorting heuristic (often helps)
    if time.time() < deadline:
        sorted_desc = np.sort(best_a)[::-1]
        sorted_desc = _project_to_simplex(sorted_desc, S)
        _maybe_update(sorted_desc)

        sorted_asc = np.sort(best_a)
        sorted_asc = _project_to_simplex(sorted_asc, S)
        _maybe_update(sorted_asc)

    # ------------------------------------------------------------------
    # hard‑max projected sub‑gradient descent with back‑tracking line‑search
    if time.time() < deadline:
        a = best_a.copy()
        it = 0
        eta0 = S * 0.7 / n         # initial step scale
        while time.time() < deadline and it < 4000:
            conv = np.convolve(a, a)
            max_c = conv.max()
            mask = conv == max_c
            cnt = np.count_nonzero(mask)
            w = np.zeros_like(conv, dtype=float)
            w[mask] = 1.0 / cnt

            # sub‑gradient wrt a
            conv_wr = np.convolve(w, a[::-1])
            g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

            # simple Armijo back‑tracking
            eta = eta0 / (1.0 + it * 1e-4)
            for _ in range(10):
                a_new = a - eta * g_a
                a_new = _project_to_simplex(a_new, S)
                new_score = _score_from_a(a_new, n)
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                    print(f"[grad] iteration {it} improved to {best_score:.6f}")
                    a = a_new
                    break
                eta *= 0.5
            else:
                # no improvement – reduce step size globally
                eta0 *= 0.5
            it += 1

    # ------------------------------------------------------------------
    # final local hill‑climbing (mass transfers between two coordinates)
    if time.time() < deadline:
        a = best_a.copy()
        moves = 0
        while time.time() < deadline and moves < 8000:
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] == 0.0:
                moves += 1
                continue
            # move a random fraction of the mass at i to j
            frac = rng.uniform(0.0, 0.3)   # up to 30 % of a[i]
            delta = frac * a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # projection is not needed because the sum stays unchanged
            # but we enforce non‑negativity (already satisfied)
            cand_score = _score_from_a(cand, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                print(f"[hill] move {moves} improved to {best_score:.6f}")
            moves += 1

    # ------------------------------------------------------------------
    # final verification with the official evaluator (once)
    try:
        official = evaluate_sequence(best_a.tolist())
        if official < best_score:
            best_score = official
    except Exception:
        pass

    print(f"[final] best score = {best_score:.6f} (n={n})")
    return best_a.tolist()
