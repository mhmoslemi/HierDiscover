"""
Propose a candidate step‑function (non‑negative heights) that minimises
`evaluate_sequence`.  The algorithm combines three stages:

1.  Initialise a probability vector on the simplex using a discrete arcsine
    distribution (good for low ∞‑norm of the self‑convolution) and a few
    random restarts.
2.  Run a mirror‑descent / exponentiated‑gradient optimiser with a
    temperature‑annealed soft‑max surrogate for the max‑convolution.
    The step‑size decays as 1/√t and the temperature grows until a pure
    sub‑gradient step is used.
3.  Perform a directed local refinement: at each iteration move a small
    amount of mass from the index that contributes most to the current
    maximal convolution entry to an index that does **not** contribute,
    accepting the move only if it lowers the objective.

Time is split among a handful of problem sizes (up to 20 000 entries);
the best sequence found over all sizes is returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()

    u = np.sort(v)[::-1]                     # descending order
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution of length n (sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Return a probability vector of length n."""
    if kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:                       # fallback
        a = _arcsine_distribution(n)
    return _project_onto_simplex(a, s=1.0)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _subgradient_max(a: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of max(b) at the arg‑max index k (b = a*a)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx = np.arange(i_start, i_end + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
def _gradient_softmax(a: np.ndarray,
                     b: np.ndarray,
                     alpha: float,
                     size_fft: int) -> np.ndarray:
    """
    Gradient of (1/α)·log∑exp(α·b) w.r.t. a using soft‑max weights.
    """
    shift = b - b.max()
    w = np.exp(alpha * shift)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.zeros_like(w)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum

    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)
    start = a.shape[0] - 1
    g = 2.0 * conv[start:start + a.shape[0]]
    return g


# ----------------------------------------------------------------------
def _optimize_exponential(a0: np.ndarray,
                          n: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent / exponentiated‑gradient optimiser.
    Returns the best sequence seen and its internal score
    (2·n·max(conv)).
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(a).max()

    # hyper‑parameters
    eta0 = 1.0                # initial step‑size
    alpha = 0.5               # start with a gentle soft‑max
    alpha_mul = 1.03
    alpha_max = 5000.0
    alpha_thr = 2000.0        # after this switch to exact sub‑gradient
    size_fft = 1 << ((3 * n - 2).bit_length())

    it = 0
    while time.time() < deadline:
        it += 1

        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient
        if alpha > alpha_thr:
            k = int(np.argmax(b))
            grad = _subgradient_max(a, k)
        else:
            grad = _gradient_softmax(a, b, alpha, size_fft)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0.0)

        # re‑project onto simplex
        s = a.sum()
        if s <= 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny random jitter to avoid stagnation
        if it % 2000 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-5
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_seq, best_score

def _directed_refinement(a_start: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climbing: move mass from the index that contributes
    most to the current maximal convolution entry to an index that does not
    contribute, accepting only improvements.
    """
    a = a_start.copy()
    best_seq = a.copy()
    best_score = 2.0 * n * np.max(_fft_self_convolve(a))

    it = 0
    eps_factor = 0.1          # maximal fraction of source mass to move
    while time.time() < deadline:
        it += 1
        b = _fft_self_convolve(a)
        k = int(np.argmax(b))

        # indices that actually participate in b_k
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        idx = np.arange(i_low, i_high + 1)               # i such that a[i] appears
        contrib = a[k - idx]                               # a_{k-i}

        # source: index with largest contribution (and non‑zero mass)
        src_candidates = idx[contrib > 0.0]
        if src_candidates.size == 0:
            # fallback to random move
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            eps = a[i] * np.random.rand() * eps_factor
            if eps <= 0.0:
                continue
            a_candidate = a.copy()
            a_candidate[i] -= eps
            a_candidate[j] += eps
            if a_candidate[i] < 0.0:
                continue
        else:
            src = src_candidates[np.argmax(contrib[src_candidates - i_low])]
            # destination: any index that does *not* contribute to b_k
            non_contrib = np.setdiff1d(np.arange(n), idx)
            if non_contrib.size > 0:
                dst = np.random.choice(non_contrib)
            else:
                # everything contributes → pick the minimal contribution index
                dst = idx[np.argmin(contrib)]
                if dst == src:
                    dst = (src + 1) % n
            eps = a[src] * np.random.rand() * eps_factor
            if eps <= 0.0:
                continue
            a_candidate = a.copy()
            a_candidate[src] -= eps
            a_candidate[dst] += eps
            if a_candidate[src] < -1e-12:
                continue

        # evaluate candidate
        cur_score = 2.0 * n * np.max(_fft_self_convolve(a_candidate))

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_seq = a_candidate.copy()
            a = a_candidate
            print(f"[{time.time():.2f}] dir_refine it={it} score={best_score:.6f}")

        # occasional random jitter to avoid tiny local traps
        if it % 3000 == 0 and np.random.rand() < 0.07:
            a += np.random.randn(n) * 5e-6
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_seq, best_score

                overall_best_seq = seq.copy()
                print(
                    f"*** New global best → len={n} init={init_kind} "
                    f"score={overall_best_score:.6f}"
                )
            if overall_best_score <= target_score:
                break
        if overall_best_score <= target_score:
            break

    # ------------------------------------------------------------------
    # final refinement on the best candidate (if any time left)
    if overall_best_seq is not None and time.time() < deadline - 1.0:
        n = len(overall_best_seq)
        seq, _ = _directed_refinement(overall_best_seq, n, deadline)
        try:
            final_score = evaluate_sequence(seq.tolist())
        except Exception:
            final_score = None
        if final_score is not None and final_score < overall_best_score:
            overall_best_score = final_score
            overall_best_seq = seq.copy()
            print(f"*** After directed refinement score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # safety fallback (should never be needed)
    if overall_best_seq is None:
        n = 1000
        overall_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        overall_best_score = evaluate_sequence(overall_best_seq.tolist())

    print(f"Final best score: {overall_best_score:.6f} (length = {len(overall_best_seq)})")
    return overall_best_seq.tolist()
