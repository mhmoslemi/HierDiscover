* Momentum (β) and a diminishing stepsize η = η0 / sqrt(t) help
  convergence.
* Every few thousand iterations a tiny random jitter is added to escape
  plateaus.
* After the gradient phase a very cheap pairwise‐mass‑swap local
  search is performed for the remaining budget.
* The best sequence found over all tried lengths is returned.

The code uses only `numpy`, `time` and the already‑imported
`evaluate_sequence`.  All helper functions are top‑level and contain no
closures, complying with the execution rules.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                 # descending sort
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine distribution (probability vector) of length n."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _initial_sequence(n: int, kind: str) -> np.ndarray:
    """Create an initial probability vector of length n."""
    if kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif kind == "arcsine":
        a = _arcsine_distribution(n)
    elif kind == "spike":
        a = np.zeros(n, dtype=np.float64)
        # two spikes placed randomly but not too close to the borders
        pos1 = np.random.randint(max(1, n // 10))
        pos2 = np.random.randint(n - max(1, n // 10), n)
        a[pos1] = 0.5
        a[pos2] = 0.5
    else:          # fallback
        a = _arcsine_distribution(n)

    a = _project_onto_simplex(a, s=1.0)
    return a


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _gradient_via_softmax(a: np.ndarray,
                         b: np.ndarray,
                         alpha: float,
                         size_fft: int) -> np.ndarray:
    """
    Sub‑gradient of max(b) w.r.t. a using a soft‑max approximation.
    For very large `alpha` the soft‑max collapses to a uniform weight over
    the (near) maximal entries.
    """
    n = a.shape[0]

    # ----- soft‑max weights over the convolution entries -----
    if alpha > 2000.0:
        # treat all entries within a tiny tolerance as active
        max_b = b.max()
        mask = np.abs(b - max_b) < 1e-12
        w = mask.astype(float)
        cnt = w.sum()
        if cnt > 0:
            w /= cnt
        else:
            w[np.argmax(b)] = 1.0
    else:
        shift = b - b.max()            # numerical stability
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.zeros_like(w)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum

    # ----- gradient = 2 * (w * a_rev)  (convolution) -----
    a_rev = a[::-1]
    W = np.fft.rfft(w, n=size_fft)
    A_rev = np.fft.rfft(a_rev, n=size_fft)
    conv = np.fft.irfft(W * A_rev, n=size_fft)

    # the relevant slice corresponds to indices n‑1 … 2·n‑2
    start = n - 1
    g = 2.0 * conv[start:start + n]
    return g


# ----------------------------------------------------------------------
def _optimize_one_length(init_a: np.ndarray,
                         n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected (soft‑max) gradient descent on the simplex to minimise
    `2·n·max(conv(a,a))`.  Returns the best sequence discovered and its
    internal score (the quantity before final normalisation).
    """
    a = init_a.copy()
    best_seq = a.copy()
    b = _fft_self_convolve(a)
    best_score = 2.0 * n * b.max()

    # hyper‑parameters
    beta = 0.8           # momentum factor
    eta0 = 0.8           # initial step‑size constant
    velocity = np.zeros_like(a)
    alpha = 5.0          # start temperature for the soft‑max
    alpha_mul = 1.06
    alpha_max = 20000.0
    size_fft = 1 << ((3 * n - 2).bit_length())   # for w * a_rev
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution & score
        b = _fft_self_convolve(a)
        cur_max = b.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} it={it} score={best_score:.6f}")

        # gradient (soft‑max or sub‑gradient)
        grad = _gradient_via_softmax(a, b, alpha, size_fft)

        # Nesterov‑style momentum
        velocity = beta * velocity + (1.0 - beta) * grad

        # diminishing step size
        eta = eta0 / np.sqrt(it)

        # gradient step + projection onto the simplex
        a = a - eta * velocity
        a = _project_onto_simplex(a, s=1.0)

        # anneal temperature
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # occasional tiny jitter to help escape plateaus
        if it % 2500 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-5
            a = _project_onto_simplex(a, s=1.0)

    return best_seq, best_score


# ----------------------------------------------------------------------
def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: repeatedly move a random
    fraction of mass between two positions and keep the move if it
    reduces the max‑convolution.
    """
    best_seq = a.copy()
    best_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)

        if best_seq[i] <= 1e-12:
            continue

        frac = np.random.rand() * 0.30               # up to 30 % transfer
        delta = best_seq[i] * frac
        a_new = best_seq.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # evaluate
        b = _fft_self_convolve(a_new)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_seq = a_new

        # tiny jitter every few thousand iterations
        if it % 5000 == 0:
            a_new = best_seq + np.random.randn(n) * 1e-6
            a_new = _project_onto_simplex(a_new, s=1.0)
            b = _fft_self_convolve(a_new)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_seq = a_new

    return best_seq, best_score

    """
    Main entry point invoked by the grader.
    Searches for a step‑function with a low `evaluate_sequence` score.
    """
    np.random.seed(seed)
    # give ourselves a small safety margin
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # try to reuse a previously discovered sequence if it exists
    try:
        base_n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        base_n = 7000                               # a reasonable default

    length_factors = (0.9, 1.0, 1.2, 1.5, 2.0)
    lengths = sorted({
        max(500, int(base_n * f))
        for f in length_factors
        if max(500, int(base_n * f)) <= 30000    # keep the convolution cheap
    })

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    for n in lengths:
        remaining = deadline - time.time()
        if remaining <= 0:
            break

        # allocate a slice of the remaining budget to this length
        slice_deadline = time.time() + max(0.1, remaining * 0.30)

        # different initial shapes increase diversification
        for init_kind in ("prev", "arcsine", "uniform", "random", "spike"):
            if time.time() >= slice_deadline:
                break

            # ------------------------------------------------------------------
            # initialise
            if init_kind == "prev":
                try:
                    a0 = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
                    if a0.size != n:
                        raise ValueError
                    a0 = a0 / a0.sum()
                except Exception:
                    a0 = _initial_sequence(n, "arcsine")
            else:
                a0 = _initial_sequence(n, init_kind)

            # ------------------------------------------------------------------
            # gradient‑based optimisation
            seq, _ = _optimize_one_length(a0, n, slice_deadline)

            # ------------------------------------------------------------------
            # local‑search refinement (if we have a few seconds left)
            time_left = slice_deadline - time.time()
            if time_left > 0.5:
                seq, _ = _local_search(seq, n,
                                       time.time() + min(time_left, 5.0))

            # ------------------------------------------------------------------
            # final evaluation with the official black‑box
            try:
                final_score = evaluate_sequence(seq.tolist())
            except Exception:
                # fallback to the internal score (identical up to scaling)
                b = _fft_self_convolve(seq)
                final_score = 2.0 * n * b.max()

            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = seq.copy()
                print(f"*** New global best → len={n} init={init_kind} "
                      f"score={global_best_score:.6f}")

            # early exit if we reach the target bound
            if global_best_score <= 1.503:
                break
        if global_best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # final refinement on the overall best (if time permits)
    if global_best_seq is not None and time.time() < deadline:
        refined_seq, _ = _local_search(global_best_seq,
                                       len(global_best_seq),
                                       deadline)
        try:
            refined_score = evaluate_sequence(refined_seq.tolist())
            if refined_score < global_best_score:
                global_best_seq = refined_seq
                global_best_score = refined_score
                print(f"*** Refined global best → score={global_best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # fallback – should never be needed but guarantees a return value
    if global_best_seq is None:
        n = 2000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_score = evaluate_sequence(global_best_seq.tolist())
        print(f"Fallback sequence (len={n}) score={global_best_score:.6f}")

    print(f"Final best score: {global_best_score:.6f} (length = {len(global_best_seq)})")
    # Convert to plain Python list as required by the grader
    return global_best_seq.tolist()
