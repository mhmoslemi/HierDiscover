def _exp_gradient_descent(a, deadline, max_iter=50000):
    n = len(a)
    t = 0
    eta0 = 0.5
    while time.time() < deadline and t < max_iter:
        t += 1
        # compute convolution, max index
        b = np.convolve(a, a)
        max_b = b.max()
        k = np.argmax(b)
        # compute gradient g
        i = np.arange(n)
        g = np.zeros(n)
        offset = k - i
        mask = (offset >= 0) & (offset < n)
        g[mask] += a[offset[mask]]
        offset2 = i - k
        mask2 = (offset2 >= 0) & (offset2 < n)
        g[mask2] += a[offset2[mask2]]
        # update
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)
        # renorm to sum 1
        sum_a = a.sum()
        if sum_a <= 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a
        # optional jitter
        if t % 2000 == 0:
            # add small random noise
            a += np.random.randn(n) * 1e-6
            a = np.clip(a, 0, None)
            a = a / a.sum()
    return a

"""Propose a step‑function (list of non‑negative heights) with a low
evaluate_sequence() score.

The algorithm works in three stages:

1. **Random parametric initialization** – generate many candidates from
   symmetric Beta(α,β) distributions (uniform, arcsine and intermediate
   shapes).  The parameters α,β are sampled log‑uniformly so that both
   “flat” and “U‑shaped” profiles are explored.

2. **Exponentiated‑gradient descent on the simplex** – for a fixed length
   `n` the objective is 2·n·max_k (a*a)[k].  The gradient of the active
   term (the index *k* attaining the maximum) is easy to compute:
   `∂/∂a_i (a*a)[k] = a_{k-i} + a_{i-k}` where the indices are inside the
   support.  The update `a_i ← a_i·exp(‑η·g_i)` followed by re‑normalisation
   stays on the probability simplex and directly drives the current
   maximum down.  A diminishing step‑size `η = η0/√t` and occasional tiny
   random jitter avoid stagnation.

3. **Cheap local‑search** – after gradient descent a stochastic pairwise
   mass‑swap refinement (move a random fraction of mass from a high‑gradient
   index to a low‑gradient one) further reduces the maximum convolution.

The routine runs several restarts on different lengths (up to ~8000,
well within the time budget) and keeps the best sequence found.  All
operations use FFT‑based convolutions for O(n log n) speed, and the final
sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, sum(x)=s}. """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - s
    rho_arr = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) distribution on n points:
        a_i ∝ (i+0.5)^{α‑1}·(n‑i‑0.5)^{β‑1}
    The result is normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    # clip to the hard limit used by evaluate_sequence
    if a.max() > 1000.0:
        a = a * (1000.0 / a.max())
    return a


# ----------------------------------------------------------------------
def _random_beta_sequence(n: int) -> np.ndarray:
    """Sample a random symmetric or asymmetric Beta distribution."""
    # log‑uniform sampling of α,β in [0.2, 5]
    log_a = np.random.uniform(np.log(0.2), np.log(5.0))
    log_b = np.random.uniform(np.log(0.2), np.log(5.0))
    alpha = np.exp(log_a)
    beta = np.exp(log_b)
    return _beta_distribution(n, alpha, beta)


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using real FFT (returns length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    B = A * A
    conv = np.fft.irfft(B, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _conv_max_idx(a: np.ndarray) -> tuple[float, int]:
    """
    Return the maximal value of a*a and the index where it occurs.
    Uses FFT when the length is moderate.
    """
    n = a.shape[0]
    if n <= 1024:               # direct convolution is cheap for very small n
        b = np.convolve(a, a)
    else:
        b = _fft_self_convolve(a)
    idx = int(np.argmax(b))
    return float(b[idx]), idx

def _exp_gradient_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex to minimise
    max_k (a*a)[k].  Returns the best vector seen and its internal score
    2·n·max_b (equivalent to evaluate_sequence after normalisation).
    """
    n = a.shape[0]
    best_a = a.copy()
    max_b, _ = _conv_max_idx(a)
    best_score = 2.0 * n * max_b

    t = 0
    eta0 = 0.5                      # initial step‑size denominator

    while time.time() < deadline:
        t += 1
        max_b, k = _conv_max_idx(a)

        # gradient of the active convolution term
        i = np.arange(n)
        g = np.zeros_like(a)

        # contribution from a_{k‑i}
        off1 = k - i
        mask1 = (off1 >= 0) & (off1 < n)
        g[mask1] += a[off1[mask1]]

        # contribution from a_{i‑k}
        off2 = i - k
        mask2 = (off2 >= 0) & (off2 < n)
        g[mask2] += a[off2[mask2]]

        # multiplicative update (exponentiated gradient)
        eta = eta0 / np.sqrt(t)
        a = a * np.exp(-eta * g)

        # projection onto the simplex = renormalisation (a stays non‑negative)
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        # keep the best vector seen
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

            a += np.random.randn(n) * 1e-5
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best_a, best_score

def _pairwise_swap_local_search(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic local search: repeatedly move a random fraction
    of mass from one index to another and keep the move if the max‑convolution
    improves.  Stops when the deadline is reached.
    """
    n = a.shape[0]
    best_a = a.copy()
    max_b, _ = _conv_max_idx(best_a)
    best_score = 2.0 * n * max_b
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = np.random.choice(n, size=2, replace=False)
        if best_a[i] <= 1e-12:
            continue
        frac = np.random.rand() * 0.30          # transfer up to 30 %
        delta = best_a[i] * frac
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        max_b, _ = _conv_max_idx(a_new)
        cur_score = 2.0 * n * max_b
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new

            cur_score = 2.0 * n * max_b
