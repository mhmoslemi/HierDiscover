def anneal_permutation(a, size_fft, deadline, initial_temp=1.0):
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)
    cur_a = a.copy()
    cur_score = best_score
    temp = initial_temp
    while time.time() < deadline:
        i, j = np.random.randint(len(a), size=2)
        # swap
        cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        new_score = _max_convolution(cur_a, size_fft)
        delta = new_score - cur_score
        # acceptance
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                # print improvement.
        else:
            # revert swap
            cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        # update temperature gradually
        temp *= 0.9999
        # maybe occasional resetting
    return best_a, best_score

"""
Propose a non‑negative step function with a low self‑convolution peak.

Algorithm
---------
1. **Problem size** – use the length of the provided reference sequence
   (``height_sequence_1``) if available, otherwise fall back to n≈3000.
   The ℓ₁‑norm is fixed to √(2 n) (this makes the evaluation score equal to
   the convolution peak, eliminating the scaling factor).

2. **Diverse sampling** – generate a pool of candidates from many families:
   uniform, Dirichlet, power‑law, exponential, block‑wise constant,
   binary spikes, random‑phase (flat Fourier magnitude) and saw‑tooth shapes.
   Each candidate is projected onto the simplex Σa = √(2 n) and the best K
   (K≈20) by convolution peak are retained.

3. **Projected Adam sub‑gradient descent** – for every retained candidate we
   run a short Adam‑style optimisation on the *hard* max‑convolution
   (sub‑gradient taken at the current worst index).  After each step the
   vector is projected back onto the simplex.  The best refinement of each
   start is kept and the overall best sequence is tracked.

4. **Simulated‑annealing permutation search** – the values of the best
   refined sequence are permuted.  Pair swaps are accepted if they improve
   the peak, otherwise with probability exp(−Δ/temp).  The temperature is
   slowly decayed; the search runs for the remaining time budget.

5. Return the best sequence discovered before the 1000 s deadline.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if cond.any():
        rho = np.where(cond)[0][-1] + 1
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0.0)
    else:
        # all entries will be zero; put the entire mass on the first coordinate
        w = np.zeros_like(v)
        w[0] = z
    return w


# ----------------------------------------------------------------------
def _fft_size(required: int) -> int:
    """Return the smallest power‑of‑2 ≥ required (for efficient FFT)."""
    return 1 << ((required - 1).bit_length())


# ----------------------------------------------------------------------
def _full_conv(a: np.ndarray, size_fft: int) -> np.ndarray:
    """Full linear convolution a * a using FFT (length = 2 n‑1)."""
    fa = np.fft.rfft(a, n=size_fft)
    conv = np.fft.irfft(fa * fa, n=size_fft)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Exact peak of the self‑convolution."""
    conv = _full_conv(a, size_fft)
    return float(conv.max())


# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a*a)_k   at the current maximal index k_max.
    This is the exact sub‑gradient of the L∞‑norm of the convolution.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k_max < n:
        # overlapping part is a[0..k_max] with reversed a[k_max::-1]
        g[: k_max + 1] = 2.0 * a[k_max :: -1]
    else:
        # k_max ≥ n
        start = k_max - n + 1                # first i that participates
        length = 2 * n - 1 - k_max           # number of terms
        g[start:] = 2.0 * a[:length][::-1]

    return g


# ----------------------------------------------------------------------
def _refine_subgrad(
    a0: np.ndarray,
    size_fft: int,
    target_sum: float,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style sub‑gradient descent on the hard max‑convolution.
    Returns the best (lowest peak) sequence and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution(a, size_fft)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # base step size (order of magnitude of one entry)
    eta0 = target_sum * 0.5 / a.size

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv_full = _full_conv(a, size_fft)
        cur_max = float(conv_full.max())
        if cur_max < best_score:
            best_score = cur_max
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[refine] iter {it:,} new best = {best_score:.6f}")

        # sub‑gradient at worst index
        k_max = int(conv_full.argmax())
        g = _subgradient_max_conv(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decays like 1/√t
        eta = eta0 / math.sqrt(it)

        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection
        a = _project_to_simplex(a, target_sum)

    return best_a, best_score


# ----------------------------------------------------------------------
def _binary_initial(n: int, target_sum: float) -> np.ndarray:
    """Create a sparse binary‑spike vector (equal mass on a few random positions)."""
    # choose fraction of occupied positions uniformly in [0.5 %, 5 %]
    frac = np.random.uniform(0.005, 0.05)
    k = max(1, int(frac * n))
    idx = np.random.choice(n, k, replace=False)
    val = target_sum / k
    a = np.zeros(n)
    a[idx] = val
    return a

def _random_phase_initial(n: int, target_sum: float) -> np.ndarray:
    """
    Generate a vector whose Fourier magnitude is (approximately) flat.
    The zero‑frequency term is set to the required ℓ₁‑norm, the remaining
    frequencies receive random amplitudes and phases.
    """
    size = _fft_size(2 * n - 1)          # length for FFT (power of two)
    half = size // 2 + 1                # length of rfft output

    # start with zeroed spectrum
    F = np.zeros(half, dtype=np.complex128)

    # zero‑frequency = sum of the sequence
    F[0] = target_sum

    # random amplitudes for the remaining frequencies
    # magnitude proportional to target_sum, phase uniform in [0, 2π)
    mags = target_sum * np.random.rand(half - 1)
    phases = 2.0 * np.pi * np.random.rand(half - 1)
    F[1:] = mags * np.exp(1j * phases)

    # inverse real FFT, keep first n samples
    a = np.fft.irfft(F, n=size)[:n]

    # shift to non‑negative and project onto the simplex
    a = a - a.min() + 1e-12
    a = _project_to_simplex(a, target_sum)
    return a

def _sawtooth_initial(n: int, target_sum: float) -> np.ndarray:

    period = np.random.randint(10, 200)          # random period length
    a = (np.arange(n) % period) / period

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimizes the evaluation score.
    The function respects a 1000 s time budget and returns the best sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ---------- problem size ----------
    try:
        n = len(height_sequence_1)                 # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 3000
        have_ref = False

    target_sum = math.sqrt(2.0 * n)                # Σ a = √(2 n)
    size_fft = _fft_size(2 * n - 1)                # FFT length for full convolution

    # ---------- helper ----------
    def push_candidate(score: float, a: np.ndarray, pool: list[tuple[float, np.ndarray]], K: int) -> None:
        """Maintain a pool of the K best candidates (ascending by score)."""
        if len(pool) < K:
            pool.append((score, a.copy()))
            pool.sort(key=lambda x: x[0])
        else:
            if score < pool[-1][0]:
                pool[-1] = (score, a.copy())
                pool.sort(key=lambda x: x[0])

    # ---------- sampling ----------
    K = 20                                          # size of the raw‑candidate pool
    best_raw: list[tuple[float, np.ndarray]] = []   # (score, a)

    # uniform baseline
    a_uni = np.ones(n)
    a_uni = _project_to_simplex(a_uni, target_sum)
    push_candidate(_max_convolution(a_uni, size_fft), a_uni, best_raw, K)

    # reference (if available)
    if have_ref:
        a_ref = np.array(height_sequence_1, dtype=float)   # type: ignore[name-defined]
        a_ref = _project_to_simplex(a_ref, target_sum)
        push_candidate(_max_convolution(a_ref, size_fft), a_ref, best_raw, K)

    idx = np.arange(n)

    # Dirichlet draws (various concentrations)
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]:
        for _ in range(8):
            probs = np.random.dirichlet(np.full(n, alpha))
            a = probs * target_sum
            push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    # Power‑law decreasing shapes
    for _ in range(25):
        p = np.random.rand() * 3.0 + 0.2          # exponent ∈ [0.2, 3.2]
        a = 1.0 / (idx + 1) ** p
        a = _project_to_simplex(a, target_sum)
        push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    # Exponential decay shapes
    for _ in range(25):
        dec = np.random.rand() * 0.015 + 0.001    # decay ∈ [1e‑3, 1.6e‑2]
        a = np.exp(-dec * idx)
        a = _project_to_simplex(a, target_sum)
        push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    # Random block‑wise constant sequences
    for _ in range(120):
        nb = np.random.randint(2, 7)               # number of blocks
        cuts = np.sort(np.random.choice(n - 1, nb - 1, replace=False) + 1)
        blocks = np.split(idx, cuts)
        vals = np.random.rand(nb) + 0.05
        cand = np.empty(n)
        for blk, v in zip(blocks, vals):
            cand[blk] = v
        cand = _project_to_simplex(cand, target_sum)
        push_candidate(_max_convolution(cand, size_fft), cand, best_raw, K)

    # Binary‑spike candidates
    for _ in range(30):
        a = _binary_initial(n, target_sum)
        push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    # Random‑phase candidates
    for _ in range(30):
        a = _random_phase_initial(n, target_sum)
        push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    # Saw‑tooth candidates
    for _ in range(30):
        a = _sawtooth_initial(n, target_sum)
        push_candidate(_max_convolution(a, size_fft), a, best_raw, K)

    print(f"[sampling] retained top {K} raw candidates; best score = {best_raw[0][0]:.6f}")

    # ---------- refinement ----------
    # the pool is already sorted (ascending)
    global_best_a = best_raw[0][1].copy()
    global_best_score = best_raw[0][0]

    for rank, (raw_score, a0) in enumerate(best_raw):
        if time.time() >= deadline:
            break
        # allocate a modest number of Adam iterations depending on remaining time
        remaining = deadline - time.time()
        remaining_cands = len(best_raw) - rank
        max_iters = int(min(150_000, remaining * 2000 / max(1, remaining_cands)))
        print(f"[refine] candidate #{rank+1} (raw {raw_score:.6f}), max_iters={max_iters}")
        refined_a, refined_score = _refine_subgrad(a0, size_fft, target_sum, deadline, max_iters)
        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best_a = refined_a.copy()
            print(f"[global] new best after refinement = {global_best_score:.6f}")

    # ---------- permutation annealing ----------
    print("[annealing] start")
    cur_a = global_best_a.copy()
    cur_score = global_best_score
    best_sc = global_best_score
    best_a = cur_a.copy()

    # temperature initialised as a small fraction of the current peak
    temp = best_sc * 0.1 + 1e-3

    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)

        # propose swap
        cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        new_sc = _max_convolution(cur_a, size_fft)
        delta = new_sc - cur_score

        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-delta / max(temp, 1e-12)):
                accept = True

        if accept:
            cur_score = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = cur_a.copy()
                global_best_a = best_a.copy()
                global_best_score = best_sc
                print(f"[anneal] improved to {best_sc:.6f}")
        else:
            # revert swap
            cur_a[i], cur_a[j] = cur_a[j], cur_a[i]

        # geometric temperature decay
        temp *= 0.99995
        if temp < 1e-6:
            temp = best_sc * 0.1 + 1e-3

        # occasional random perturbation to escape long plateaus
        if np.random.rand() < 0.001:
            np.random.shuffle(cur_a)
            cur_score = _max_convolution(cur_a, size_fft)

    print(f"[final] best score = {global_best_score:.6f}")
    return global_best_a.tolist()
