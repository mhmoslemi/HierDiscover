def _lp_refine(a_start: np.ndarray, S: float, budget: float, start_time: float) -> tuple[np.ndarray, float]:
    n = len(a_start)
    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    eps = 1e-12
    it = 0
    while time.time() - start_time < budget:
        # Compute conv
        size = _fft_size(n)
        conv = _full_convolution(a, size)
        M = conv.max()
        # Determine active constraints
        tol = M * 1e-3  # 0.1% tolerance maybe
        active_idx = np.where(conv >= M - tol)[0]
        if len(active_idx) == 0:
            break
        # Build sparse matrix for active constraints
        rows = []
        cols = []
        data = []
        for idx_row, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            idxs = np.arange(i_start, i_end + 1)
            rows.append(np.full(len(idxs), idx_row, dtype=int))
            cols.append(idxs)
            data.append(a[i_start:i_end+1])
        if not rows:
            break
        row_arr = np.concatenate(rows)
        col_arr = np.concatenate(cols)
        data_arr = np.concatenate(data)
        from scipy.sparse import coo_matrix
        A_active = coo_matrix((data_arr, (row_arr, col_arr)), shape=(len(active_idx), n)).tocsc()
        # Setup LP: maximize sum(b) => minimize - sum(b)
        c = -np.ones(n)
        b_ub = np.full(len(active_idx), M)
        # Bounds for variables
        a_max = np.max(a)
        if a_max < eps:
            a_max = eps
        ub = M / a_max  # safe bound
        bounds = [(0.0, ub) for _ in range(n)]
        # Solve
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A_active, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # maybe due to insufficient constraints; break
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        # scaling factor r = S / sum_b, note sum_b >= S ideally
        r = S / sum_b
        if r > 1:
            # scaling up may violate constraints; reduce maybe
            r = 1.0
        g = b * r
        # step size t maybe dynamic based on iteration
        t = 0.5
        # ensure convex combination sum stays S (should)
        a_new = (1.0 - t) * a + t * g
        # projection to simplex to correct rounding errors
        a_new = _project_to_simplex(a_new, S)
        # Compute new max conv
        conv_new = _full_convolution(a_new, size)
        M_new = conv_new.max()
        # Accept if improvement
        if M_new < M - 1e-6:
            a = a_new
            score = float(M_new)  # since sum = S -> score M_new
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[LP improve] iter {it} M={M_new:.6f} score={score:.6f}")
        else:

"""
Hybrid optimizer for the step‑function minimisation problem.

Algorithm
---------
1. Initialise a feasible sequence (uniform or from `height_sequence_1`).
2. Quick simulated‑annealing (SA) on the simplex – cheap random mass moves.
3. Iterative linear‑program refinement:
   * Build only the “active” convolution constraints where the current
     max‑convolution is tight.
   * Solve a linear program that maximises the ℓ¹‑mass while keeping all
     active constraints ≤ current maximum.
   * Scale the LP solution back to the required ℓ¹‑norm and move the
     sequence towards it (convex combination).  Accept the move only if
     the maximum convolution strictly decreases.
   * Repeat until time runs out – this is a Frank‑Wolfe style scheme that
     provably reduces the ℓ∞‑norm of the auto‑convolution.
4. Finish with a smoothed‑max L‑BFGS refinement (as in the reference
   implementation) and a short projected sub‑gradient polish.
5. Return the best sequence found.

All helpers are top‑level functions; no closures or lambdas are used.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import diags, coo_matrix


# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_size(n: int) -> int:
    """Smallest power of two ≥ 2·n‑1 (for FFT‑based convolutions)."""
    return 1 << ((2 * n - 1).bit_length())


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a*a (length 2·n‑1) using FFT; result is non‑negative."""
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * len(a) - 1]
    conv[conv < 0] = 0.0
    return conv


def _max_convolution(a: np.ndarray, size: int) -> float:
    """Maximum entry of a*a."""
    return float(_full_convolution(a, size).max())


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


def _compute_f_and_grad(y: np.ndarray, mu: float, n: int, S: float):
    """
    Smooth‑max (soft‑max) objective and gradient w.r.t. the log‑probability
    vector y (the optimiser works on the simplex via soft‑max).
    """
    p = _softmax(y)          # probabilities, Σp = 1
    a = S * p                # actual heights, Σa = S

    conv = np.convolve(a, a)          # length 2·n‑1
    max_c = conv.max()

    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # avoid overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. a
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

    # chain rule to y
    inner = np.dot(g_a, p)
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


class _SmoothMaxObjective:
    """Callable for scipy.optimize – returns (obj, grad) for L‑BFGS."""
    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_f_and_grad(y, self.mu, self.n, self.S)


# ----------------------------------------------------------------------
def _simulated_annealing(a_start: np.ndarray, S: float,
                         budget: float, start_time: float):
    """Simple SA that moves a random fraction of mass between two indices."""
    n = len(a_start)
    size = _fft_size(n)
    a = a_start.copy()
    curr_max = _max_convolution(a, size)
    best_a = a.copy()
    best_score = curr_max
    T0 = max(0.2, curr_max * 0.5)
    Tmin = 1e-6
    it = 0

    while time.time() - start_time < budget:
        frac = (time.time() - start_time) / budget
        T = max(Tmin, T0 * (1.0 - frac))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = np.random.rand() * a[i] * 0.3
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_max = _max_convolution(a_new, size)
        diff = new_max - curr_max

        accept = diff < 0.0 or (T > 0.0 and np.random.rand() < math.exp(-diff / T))
        if accept:
            a = a_new
            curr_max = new_max
            if new_max < best_score:
                best_score = new_max
                best_a = a.copy()
                print(f"[SA improve] iter {it:,} score = {best_score:.6f}")

        it += 1
    return best_a, best_score

def _lp_refine(a_start: np.ndarray, S: float,
               budget: float, start_time: float):
    """
    Frank‑Wolfe‑style refinement.
    Solve a linear programme on the currently active convolution constraints,
    move a convex combination towards the LP solution, and accept only
    improvements of the true max‑convolution.
    """
    n = len(a_start)
    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    it = 0
    eps = 1e-12

    while time.time() - start_time < budget:
        size = _fft_size(n)
        conv = _full_convolution(a, size)
        M = conv.max()

        # active constraints: values within 0.1 % of the maximum
        tol = M * 1e-3
        active_idx = np.where(conv >= M - tol)[0]
        if len(active_idx) == 0:
            break

        # Build sparse matrix with only active rows
        row_inds = []
        col_inds = []
        data = []
        for r, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            cols = np.arange(i_start, i_end + 1, dtype=int)
            rows = np.full(cols.shape, r, dtype=int)
            row_inds.append(rows)
            col_inds.append(cols)
            data.append(a[i_start:i_end + 1])
        row_arr = np.concatenate(row_inds)
        col_arr = np.concatenate(col_inds)
        data_arr = np.concatenate(data)
        A_active = coo_matrix((data_arr, (row_arr, col_arr)),
                              shape=(len(active_idx), n)).tocsc()

        # Linear programme: maximise Σb  ⇔ minimise -Σb
        c = -np.ones(n)
        b_ub = np.full(len(active_idx), M)

        # simple uniform upper bound to avoid unboundedness
        a_max = max(a.max(), eps)
        ub = M / a_max
        bounds = [(0.0, ub) for _ in range(n)]

        res = linprog(c, A_ub=A_active, b_ub=b_ub,
                      bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {it} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Scale back to the required ℓ¹‑norm.
        r = S / sum_b          # r ≤ 1 in practice
        g = b * r

        # convex combination (t = 0.5 works well)
        t = 0.5
        a_candidate = (1.0 - t) * a + t * g
        a_candidate = _project_to_simplex(a_candidate, S)

        conv_new = _full_convolution(a_candidate, size)
        M_new = conv_new.max()

        if M_new < M - 1e-7:   # strict improvement
            a = a_candidate
            score = float(M_new)   # sum = S ⇒ evaluate = M_new
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[LP improve] iter {it:,} M = {M_new:.6f}")
        else:
            # try smaller step sizes
            improved = False
            t_small = t
            while t_small > 1e-3:
                t_small *= 0.5
                a_try = (1.0 - t_small) * a + t_small * g
                a_try = _project_to_simplex(a_try, S)
                M_try = _full_convolution(a_try, size).max()
                if M_try < M - 1e-7:
                    a = a_try
                    score = float(M_try)
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                        print(f"[LP improve] iter {it:,} (t={t_small:.4f}) M = {M_try:.6f}")
                    improved = True
                    break
            if not improved:
                # no improvement – stop early
                print(f"[LP] iteration {it} no improvement, terminating LP phase.")
                break

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
def _l_bfgs_refine(a_start: np.ndarray, S: float,
                   budget: float, start_time: float):
    """Continuation on the smooth‑max surrogate (more iterations than before)."""
    n = len(a_start)
    p0 = a_start / S
    eps = 1e-12
    y = np.log(p0 + eps).astype(np.float64)

    conv_cur = np.convolve(a_start, a_start)
    mu = float(conv_cur.max())
    if mu <= 0.0:
        mu = 1.0

    best_a = a_start.copy()
    best_score = evaluate_sequence(best_a.tolist())

    mu_factor = 0.7
    mu_min = 1e-7
    maxiter_per_mu = 500

    while mu > mu_min and time.time() - start_time < budget:
        obj = _SmoothMaxObjective(mu, n, S)
        try:
            res = minimize(
                obj,
                y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_mu,
                         "disp": False,
                         "gtol": 1e-12}
            )
        except Exception as exc:
            print(f"[L‑BFGS error] {exc}")
            break

        y = res.x
        a = S * _softmax(y)
        sc = evaluate_sequence(a.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS improve] new best = {best_score:.6f}")

        mu *= mu_factor

    return best_a, best_score

def _final_subgradient(a_start: np.ndarray, S: float,
                       budget: float, start_time: float):
    """Projected sub‑gradient descent on the true hard max."""
    n = len(a_start)
    size = _fft_size(n)

    a = a_start.copy()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    eta0 = S * 0.5 / n
    decay = 1e-4
    it = 0

    while time.time() - start_time < budget and it < 5000:
        conv = _full_convolution(a, size)
        max_c = conv.max()
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv)
        if cnt:
            w[mask] = 1.0 / cnt

        # sub‑gradient of the hard max
        fw = np.fft.rfft(w, n=size)
        fa_rev = np.fft.rfft(a[::-1], n=size)
        conv_wr = np.fft.irfft(fw * fa_rev, n=size)[: 2 * n - 1]
        g_a = 2.0 * conv_wr[n - 1:n - 1 + n]

        eta = eta0 / (1.0 + decay * it)
        a_new = a - eta * g_a
        a_new = _project_to_simplex(a_new, S)

        sc = evaluate_sequence(a_new.tolist())
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            print(f"[subgrad improve] iter {it:,} score = {best_score:.6f}")

        a = a_new
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score step‑function within the time budget.
    Returns the best sequence (list of non‑negative floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    try:
        n = len(height_sequence_1)        # type: ignore[name-defined]
        have_ref = True
    except Exception:
        n = 2000
        have_ref = False

    S = math.sqrt(2.0 * n)                # ℓ¹‑norm that makes the score = max‑conv

    # initialise ---------------------------------------------------------
    if have_ref:
        a0 = np.asarray(height_sequence_1, dtype=float)
        a0 = _project_to_simplex(a0, S)
    else:
        a0 = np.full(n, S / n, dtype=float)   # uniform start

    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    print(f"[init]   score = {best_score:.6f}")

    # 1) quick simulated‑annealing --------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        sa_time = remaining * 0.15
        print(f"[SA] running for {sa_time:.1f}s")
        best_a, best_score = _simulated_annealing(best_a, S, sa_time, time.time())
        print(f"[SA]    best = {best_score:.6f}")

    # 2) LP‑based active‑set refinement ---------------------------------
    remaining = deadline - time.time()
    if remaining > 50.0:
        print(f"[LP] refining for {remaining:.1f}s")
        best_a, best_score = _lp_refine(best_a, S, remaining, time.time())
        print(f"[LP] best = {best_score:.6f}")

    # 3) smooth‑max L‑BFGS ---------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        print(f"[L‑BFGS] refining for {remaining:.1f}s")
        best_a, best_score = _l_bfgs_refine(best_a, S, remaining, time.time())
        print(f"[L‑BFGS] best = {best_score:.6f}")

    # 4) final sub‑gradient polish --------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print(f"[final] polishing for {remaining:.1f}s")
        best_a, best_score = _final_subgradient(best_a, S, remaining, time.time())
        print(f"[final]  best = {best_score:.6f}")

    # fallback -----------------------------------------------------------
    if best_score == float("inf"):
        best_a = S * np.ones(n) / n
        best_score = evaluate_sequence(best_a.tolist())
        print("[fallback] uniform sequence used")

    print(f"[result] final best score = {best_score:.6f}")
    return best_a.tolist()
