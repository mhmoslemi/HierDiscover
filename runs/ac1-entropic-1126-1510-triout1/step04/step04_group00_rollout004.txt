def exponentiated_update(a, grad, eta):
    # grad may be large; shift by max to avoid overflow
    shift = np.max(grad)
    exp_factor = np.exp(-eta * (grad - shift))
    a = a * exp_factor
    a = a / np.sum(a)
    return a
```

"""
Propose a low‑score step‑function.

The search works in two stages.

1. **Candidate generation** – a handful of analytically‑motivated
   distributions (Beta / arcsine / Dirichlet) are generated for several
   lengths (≈ 2000 – 12000).  Their scores are evaluated cheaply using an
   FFT based convolution, and the few best candidates are kept.

2. **Iterative refinement** – each kept candidate is refined with a
   projected multiplicative‑gradient (exponentiated‑gradient) scheme.
   The surrogate loss is a smoothed max (soft‑max) of the convolution,
   with a temperature that grows from a smooth regime to the true max.
   Momentum is added to accelerate convergence and occasional jitter
   helps to escape plateaus.  The iterate is always kept on the simplex
   (non‑negative, sum = 1) by renormalisation.

The algorithm runs for the allotted time (default 1000 s) and returns
the best sequence found (as a plain Python list).  It can start from the
previously discovered construction `height_sequence_1` if it is present.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Avoid NaNs at the borders when α or β < 1
    eps = 1e-15
    a = np.clip(x, eps, 1 - eps) ** (alpha - 1.0)
    b = np.clip(1.0 - x, eps, 1 - eps) ** (beta - 1.0)
    vals = a * b
    vals_sum = vals.sum()
    if vals_sum == 0.0:
        # fallback to uniform
        vals = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        vals = vals / vals_sum
    return vals


# ----------------------------------------------------------------------
def _score_from_conv(n: int, conv: np.ndarray) -> float:
    """Score assuming sum(a)=1, i.e. 2·n·max(conv)."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """Compute the evaluate_sequence score for a normalised vector."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return _score_from_conv(n, conv)


# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate of max(conv).
    Returns a vector of length n.
    """
    n = a.shape[0]
    cur_max = conv.max()
    # use stable soft‑max: subtract cur_max to avoid overflow
    exp_term = np.exp(alpha * (conv - cur_max))
    w = exp_term / exp_term.sum()
    # convolution of reversed a with w, then pick the central n entries
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _exact_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the true max, using a uniform weight over all
    (approximately) active indices.
    """
    n = a.shape[0]
    cur_max = conv.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - eps)
    if active.size == 0:
        # fall back to a single max index
        active = np.array([np.argmax(conv)], dtype=np.int64)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for entries where k = 2*i (i.e. self‑product)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad


# ----------------------------------------------------------------------
def _exponential_update(a: np.ndarray, grad: np.ndarray, eta: float,
                       momentum: np.ndarray, beta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Multiplicative (exponentiated) update with simple momentum.
    Returns the new a and updated momentum.
    """
    # Momentum on the raw gradient
    momentum = beta * momentum + (1.0 - beta) * grad
    # Scale for numerical stability
    shift = np.max(momentum)
    factor = np.exp(-eta * (momentum - shift))
    a = a * factor
    s = a.sum()
    if s <= 0.0 or np.isnan(s):
        # protection against collapse – fall back to uniform
        n = a.shape[0]
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a, momentum


# ----------------------------------------------------------------------
def _initial_vector(init_kind: str, n: int) -> np.ndarray:
    """
    Produce an initial feasible probability vector.
    If ``init_kind == "prev"`` we try to reuse the global
    `height_sequence_1` (if it exists); otherwise we fall back to an arcsine law.
    """
    if init_kind == "uniform":
        a = np.full(n, 1.0 / n, dtype=np.float64)
    elif init_kind == "random":
        a = np.random.rand(n)
        a /= a.sum()
    elif init_kind == "arcsine":
        a = _beta_distribution(n, 0.5, 0.5)  # discrete arcsine = Beta(½,½)
    elif init_kind == "prev":
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
            if prev.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev)
            else:
                a = prev.copy()
        except Exception:
            a = _beta_distribution(n, 0.5, 0.5)
    else:
        a = _beta_distribution(n, 0.5, 0.5)

    a = _project_onto_simplex(a, s=1.0)
    return a
