"""
Propose a low–score step function.

The algorithm works in three stages:
  1.  Multi‑restart smooth‑max optimisation.  The true objective
     (max of the auto‑convolution) is approximated by a smooth
     surrogate f_μ (max + μ·log‑sum‑exp).  For a series of
     decreasing temperatures μ we minimise f_μ with L‑BFGS‑B.
     Several diverse initial points (uniform, exponential,
     Dirichlet‑sparse, random Gaussian, and a projected reference
     if available) are explored.
  2.  Adaptive sub‑gradient descent with back‑tracking line search.
     Starting from the best candidate of stage 1 we repeatedly
     move opposite to a sub‑gradient of the hard max objective,
     project back onto the ℓ¹‑simplex and accept the move only if
     the exact score improves.
  3.  A cheap large‑scale random‑block perturbation followed by a
     monotone‑sorting heuristic.  All moves respect the non‑negativity
     and ℓ¹‑norm constraints enforced by the simplex projection.

The routine respects a user‑supplied time budget (default 1000 s) and
returns the best height sequence found.  No closures or lambdas are
used and only NumPy / SciPy are required.
"""

import time
import math
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, Σx = z} (z ≥ 0)."""
    if z == 0.0:
        return np.zeros_like(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = z
        return w
    rho = np.where(cond)[0][-1] + 1
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (probability vector)."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()


# ----------------------------------------------------------------------
def _compute_smooth_obj_and_grad(y: np.ndarray, mu: float,
                                 n: int, S: float) -> tuple[float, np.ndarray]:
    """
    Value and gradient of the smooth‑max surrogate:
        f_μ(y) = max(conv) + μ·log Σ exp((conv - max)/μ)

    Parameters
    ----------
    y : ndarray, shape (n,)
        Unconstrained parameters (log‑odds of the simplex).
    mu : float
        Smoothing temperature (>0).  μ = 0 corresponds to the hard max.
    n : int
        Length of the step function.
    S : float
        Target ℓ¹‑norm (√(2·n)).

    Returns
    -------
    f_val : float
        Smooth‑max value.
    grad_y : ndarray, shape (n,)
        Gradient ∂f/∂y.
    """
    # probabilities on the simplex, actual heights
    p = _softmax(y)                     # Σ p_i = 1, p_i ≥ 0
    a = S * p                           # Σ a_i = S

    # full (linear) convolution a*a (length 2·n‑1)
    conv = np.convolve(a, a)

    # smooth‑max weights w_k ∝ exp((conv_k - max)/μ)
    max_c = conv.max()
    if mu > 0.0:
        diff = (conv - max_c) / mu
        diff = np.clip(diff, -700.0, 700.0)   # guard overflow
        exp_diff = np.exp(diff)
        Z = exp_diff.sum()
        w = exp_diff / Z
        f_val = max_c + mu * math.log(Z)
    else:
        # hard max – uniform weight on all active indices
        mask = conv == max_c
        cnt = np.count_nonzero(mask)
        w = np.zeros_like(conv, dtype=float)
        w[mask] = 1.0 / cnt
        f_val = max_c

    # gradient w.r.t. heights a:
    #   g_a[i] = 2 * Σ_k w_k * a_{k-i}
    conv_wr = np.convolve(w, a[::-1])          # length 3·n‑2
    g_a = 2.0 * conv_wr[n - 1:n - 1 + n]       # slice length n

    # chain rule for y (soft‑max Jacobian)
    inner = np.dot(g_a, p)                     # scalar Σ_j p_j g_a_j
    grad_y = S * p * (g_a - inner)

    return float(f_val), grad_y.astype(np.float64)


# ----------------------------------------------------------------------
class _SmoothMaxObjective:
    """Callable that returns (value, gradient) for scipy.optimize."""

    __slots__ = ("mu", "n", "S")

    def __init__(self, mu: float, n: int, S: float):
        self.mu = mu
        self.n = n
        self.S = S

    def __call__(self, y: np.ndarray):
        return _compute_smooth_obj_and_grad(y, self.mu, self.n, self.S)
