"""
propose_candidate
-----------------

Search for a non‑negative step function (probability vector) that minimises

        score = 2·n·max(a*a) / (sum(a)**2)

The algorithm works on the probability simplex (sum=1) because the
evaluation is homogeneous.  It combines three ideas:

1.  **Soft‑max (smooth max) descent** – a smoothed maximum
    (log‑sum‑exp) provides a differentiable surrogate.
    An exponentiated‑gradient (mirror) update keeps the vector on the
    simplex without an explicit projection.

2.  **Exact max sub‑gradient descent** – once the temperature is large
    enough the true max is used.  Adam (first‑ and second‑moment
    adaptation) together with an Euclidean projection onto the simplex
    refines the solution.

3.  **Local random refinements** – after the gradient phases a short
    random walk (pairwise mass transfers) is performed to escape tiny
    plateaus.

Multiple sequence lengths and initialisations are tried within the
overall time budget; the best sequence found (according to the supplied

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, v.size + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete arcsine law (Beta(0.5,0.5)) of length n, already normalised."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals

def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())   # next power of two
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

    """
    n = a0.size
