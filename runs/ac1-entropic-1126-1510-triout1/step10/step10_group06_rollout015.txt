"""
Propose a step‑function (non‑negative vector) with a low value of
evaluate_sequence().  The search consists of four stages:

1. **Broad analytic scan** – many classical families (uniform,
   arcsine, triangular, exponential, beta, sine‑power, Gaussian) with
   generous parameter ranges are evaluated for several lengths (1500‑5000).
   The best candidate is kept.

2. **Heavy continuous refinement** – the current best vector is optimised
   by a long sequence of EG → PGD → Adam updates (all on the simplex).
   The schedules are extended compared with the reference implementation.

3. **Mass‑transfer hill‑climber** – an O(n) incremental convolution
   update moves a tiny amount of mass from the index with the largest
   (sub‑)gradient to the index with the smallest.  The move is accepted
   only if the maximum self‑convolution decreases.  This stage runs for
   as long as time permits.

4. **Random restarts** – while time remains, fresh Dirichlet draws are
   optimised with a shortened version of the pipeline above; any
   improvement updates the global best.

All steps respect the 1000 s wall‑clock limit; intermediate progress is
printed.  The returned list sums to 1, so it can be fed directly to
evaluate_sequence().
"""

import time
import random

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _smooth_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Smooth approximation of the max‑weight vector."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv: np.ndarray, eps_rel: float = 1e-12) -> np.ndarray:
    """Binary sub‑gradient of max (conv)."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _exponential_gradient_step(a: np.ndarray, lr: float, alpha: float) -> np.ndarray:
    """One EG step (mirror descent) on the simplex."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a: np.ndarray, eta: float, w: np.ndarray) -> np.ndarray:
    """PGD step on the (sub‑)gradient defined by the active set w."""
    n = a.shape[0]
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits: np.ndarray, m: np.ndarray, v: np.ndarray,
               t: int, step: np.ndarray) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Adam update on unconstrained logits."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step
    v = beta2 * v + (1.0 - beta2) * (step ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)
    return logits, m, v


def _apply_move_incremental(a: np.ndarray, conv: np.ndarray,
                           src: int, dst: int, delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Move a tiny amount of mass from src to dst and update the full convolution
    using the exact O(n) formula (no FFT).  Both a and conv are modified in‑place.
    """
    n = a.shape[0]
    d = delta
    # linear part of the convolution update (uses the *old* a)
    conv[src:src + n] -= 2.0 * d * a
    conv[dst:dst + n] += 2.0 * d * a

    # quadratic correction terms
    conv[2 * src] += d * d
    conv[2 * dst] += d * d
    conv[src + dst] -= 2.0 * d * d

    # move the mass
    a[src] -= d
    a[dst] += d
    return a, conv

# Analytic families – all return vectors summing to 1
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # guard against p=0 producing NaNs
    if p <= 0:
        w = np.full_like(x, 1.0 / n)
    else:
        w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
        if w.sum() == 0:
            w = np.full_like(w, 1.0 / n)
        else:
            w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) for a normalized to sum=1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _refine_vector(a: np.ndarray, n: int, deadline: float) -> np.ndarray:
    """
    Extended continuous refinement: EG → PGD → Adam.
    The schedules are longer than in the reference code.
    """
    # -------------------- Exponential‑gradient phase --------------------
    a_eg = a.copy()
    steps_eg = 3000
    a0, a1 = 1.0, 3000.0          # smoothing parameter range
    lr0, lr1 = 0.18, 0.0018       # learning‑rate range
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        alpha = a0 * (a1 / a0) ** t
        lr = lr0 * (lr1 / lr0) ** t
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)

    # ---------------------- Projected‑gradient phase ----------------------
    a_pg = a_eg.copy()
    steps_pg = 2000
    eta0, eta1 = 0.30, 0.003      # decreasing step size
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = eta0 * (eta1 / eta0) ** (it / steps_pg)
        a_pg = _projected_gradient_step(a_pg, eta, w)

    # --------------------------- Adam phase ----------------------------
    a_ad = a_pg.copy()
    logits = np.log(a_ad + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    steps_adam = 2500
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_ad = np.exp(logits)
        a_ad /= a_ad.sum()
        conv = _fft_self_convolve(a_ad)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        grad_a = 2.0 * _fft_convolve(a_ad[::-1], w)[n - 1 : n - 1 + n]
        # gradient w.r.t. logits under softmax parametrisation
        grad_l = (grad_a - np.dot(a_ad, grad_a)) * a_ad
        logits, m, v = _adam_step(logits, m, v, it, grad_l)

    a_final = np.exp(logits)
    a_final /= a_final.sum()
    return a_final


def _hillclimb(a: np.ndarray, n: int, deadline: float,
               max_iters: int | None = None) -> np.ndarray:
    """
    Mass‑transfer hill‑climber using an exact O(n) convolution update.
    It repeatedly moves a small chunk of probability mass from the index
    with the largest sub‑gradient to the one with the smallest.
    """
    eps = 1e-12
    a = a.copy()
    conv = _fft_self_convolve(a)
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline - 0.1:
        if max_iters is not None and it >= max_iters:
            break
        # sub‑gradient (active set) for the current max
        w = _subgradient_weights(conv, eps_rel=1e-12)
        # gradient of the max‑convolution w.r.t. a
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            dst = (src + 1) % n

        # choose a reasonable step size
        max_delta = min(a[src] * 0.5, 0.06)   # up to 6 % of the source mass
        if max_delta < eps:
            break
        delta = max_delta * random.random()
        a_new = a.copy()
        conv_new = conv.copy()
        a_new, conv_new = _apply_move_incremental(a_new, conv_new, src, dst, delta)

        new_max = conv_new.max()
        if new_max < conv.max() - 1e-14:
            a, conv = a_new, conv_new
            score = 2.0 * n * new_max
            if score < best_score - 1e-13:
                best_a, best_score = a.copy(), score
                print(f"[{time.time():.1f}] hill it={it}  new best={score:.6f}")
        else:
            # occasional random jitter to avoid stagnation
            if random.random() < 0.001:
                jitter = np.random.randn(n) * 1e-5
                a = a + jitter
                a = np.maximum(a, eps)
                a /= a.sum()
                conv = _fft_self_convolve(a)
        it += 1
    return best_a


def _generate_candidates_for_n(n: int) -> list[np.ndarray]:
    """Return a list of candidate vectors for a given length n."""
    eps = 1e-12
    cand = []

    # classic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    for lam in (0.004, 0.008, 0.012, 0.02):
        cand.append(_distribution_exponential(n, lam=lam))

    # beta – very U‑shaped values are important
    for p in (0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.8, 1.2, 2.0):
        cand.append(_distribution_beta_sym(n, p))

    # sine‑power – powers < 1 concentrate near the edges
    for pw in (0.2, 0.3, 0.5, 0.7, 1.0, 2.0):
        cand.append(_distribution_sine_power(n, pw))

    # Gaussian – a few widths
    for sigma in (0.15, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # random Dirichlet draws with low concentration (sparse vectors)
    for alpha in (0.05, 0.1, 0.2, 0.5):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # ensure non‑negativity and normalisation
    out = []
    for v in cand:
        v = np.maximum(v, eps)
        v /= v.sum()
        out.append(v)
    return out


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted time.
    """
    # ------------------------------------------------------------------
    # initialisation
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    eps = 1e-12

    best_seq: np.ndarray | None = None
    best_score = float("inf")
    best_n: int | None = None

    # ------------------------------------------------------------------
    # 1. Analytic scan (including very U‑shaped parameters)
    scan_lengths = [1500, 2000, 2500, 3000, 3500, 4000, 4500, 5000]
    for n in scan_lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_n(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            sc = _objective(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate the global variable ``height_sequence_1`` if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback – uniform if everything else failed
    if best_seq is None:
        best_n = 2500
        best_seq = _distribution_uniform(best_n)
        best_score = _objective(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Heavy continuous refinement of the current best vector
    best_seq = _refine_vector(best_seq, best_n, deadline)
    best_score = _objective(best_seq)
    print(f"[{time.time():.1f}] after refinement score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Long hill‑climber on the refined vector (as many iterations as time allows)
    best_seq = _hillclimb(best_seq, best_n, deadline)
    best_score = _objective(best_seq)
    print(f"[{time.time():.1f}] after hill‑climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Random restarts with a shortened pipeline while time permits
    restart_lengths = [1800, 2000, 2500, 3000, 3500, 4000, 4500, 5000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        a0 = np.random.dirichlet(np.full(n, 0.1))   # low concentration → sparsity
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()

        # shortened refinement (fewer EG/PGD/Adam steps)
        a0 = _refine_vector(a0, n, deadline)  # re‑uses the same routine; it will simply stop early
        a0 = _hillclimb(a0, n, deadline, max_iters=8000)

        sc = _objective(a0)
        if sc < best_score - 1e-13:
            best_seq, best_score, best_n = a0.copy(), sc, n
            print(f"[{time.time():.1f}] restart n={n}  new best={sc:.6f}")

    # ------------------------------------------------------------------
    final_score = _objective(best_seq)
    print(
        f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})"
    )
    return best_seq.tolist()
