"""
Search for a non‑negative step function that minimises
`evaluate_sequence`.  The algorithm builds on the previous
convex‑gradient approach but adds a large random‑permutation
stage and a simulated‑annealing hill‑climber.  The idea is to
flatten the self‑convolution more aggressively:  a random
ordering of a good analytic distribution often yields a lower
maximum of the convolution than the deterministic orderings,
and the annealer can escape the remaining local minima.

The routine respects a time budget (≈ 1000 s) and returns the
best vector found (already ℓ₁‑normalised, i.e. ∑a = 1).  The
evaluation is done with the cheap surrogate
   s(a) = 2·n·max(conv(a,a))          (∑a = 1)
which equals the true `evaluate_sequence` value.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Simplex projection
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (real FFT, zero‑padded to next power of two)
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n−1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate score for (∑a = 1): 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution functional (used by the exponential‑gradient)
def _subgrad(a: np.ndarray):
    """Exact sub‑gradient of max(conv(a,a)).
    Returns (gradient, current_max, active_indices)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    rev = a[::-1]
    size = 1 << ((rev.size + w.size - 1) - 1).bit_length()
    A = rfft(rev, n=size)
    B = rfft(w, n=size)
    conv_rev = irfft(A * B, n=size)[: rev.size + w.size - 1]

    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (i=j)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (the most effective on the simplex)
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 2_000_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑type gradient descent
def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 600_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

# Simple SGD on the sub‑gradient
def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 500_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / math.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

# Analytic families (used as seeds)
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.2) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Helper to add a candidate (normalise, score)
def _add_candidate(arr: np.ndarray, candidates: list):
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return
    a = a / s
    cand_score = _max_convolution_score(a)
    candidates.append((a, cand_score))

# Random mass transfer (used by hill‑climbing)
def _random_mass_transfer(a: np.ndarray, epsilon: float = 0.02) -> np.ndarray:
    n = a.shape[0]
    src = np.random.randint(n)
    dst = np.random.randint(n)
    while dst == src or a[src] < 1e-12:
        src = np.random.randint(n)
        dst = np.random.randint(n)
    delta = min(a[src] * epsilon, 1.0 - a[dst])
    if delta <= 0:
        return a
    b = a.copy()
    b[src] -= delta
    b[dst] += delta
    return b


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (swap or mass‑transfer moves)
def _anneal(a0: np.ndarray, deadline: float,
            init_temp: float = 0.03,
            cooling: float = 0.9995,
            min_temp: float = 1e-5,
            max_iters: int = 1_500_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best = a.copy()
    best_score = _max_convolution_score(best)
    cur_score = best_score
    temp = init_temp
    it = 0
    n = a.shape[0]

    while time.time() < deadline and it < max_iters:
        it += 1
        # decide move type
        if np.random.rand() < 0.5:
            # simple swap of two entries
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # random mass transfer (small epsilon)
            cand = _random_mass_transfer(a, epsilon=0.015)

        cand_score = _max_convolution_score(cand)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            delta = cand_score - cur_score
            accept = np.random.rand() < math.exp(-delta / max(temp, 1e-12))

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

        # geometric cooling
        temp = max(temp * cooling, min_temp)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point used by the competition harness.
    Generates a candidate step‑function within the time budget
    and returns it as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  Prefer the length of a previously
    # discovered best vector, otherwise fall back to a default.
    try:
        base_n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        base_n = 5000

    # we keep the length modest (≈ 3–6 k) – large enough to give freedom
    # but small enough for the many FFT calls we will perform.
    n = max(800, min(12000, int(base_n * 1.0)))
    print(f"[{time.time():.1f}] Working length = {n}")

    # ------------------------------------------------------------------
    # 1️⃣  Broad sampling – analytic families + many random permutations
    candidates: list[tuple[np.ndarray, float]] = []

    # deterministic families (a subset of the previous script – enough to give good seeds)
    families = [
        _arcsine_distribution,
        _triangular_distribution,
        _exponential_distribution,
        _gaussian_distribution,
        _cauchy_distribution,
    ]

    for fam in families:
        if time.time() >= deadline:
            break
        a = fam(n)
        _add_candidate(a, candidates)

        # a few deterministic reorderings (still useful)
        desc = np.sort(a)[::-1]
        # alternate extremes
        left, right = 0, n - 1
        alt = np.empty_like(desc)
        toggle = True
        for v in desc:
            if toggle:
                alt[left] = v
                left += 1
            else:
                alt[right] = v
                right -= 1
            toggle = not toggle
        _add_candidate(alt, candidates)

    # random Dirichlet draws – give very diverse shapes
    for alpha in (0.2, 0.5, 1.0, 2.0):
        if time.time() >= deadline:
            break
        for _ in range(12):
            a = _random_dirichlet_alpha(n, alpha)
            _add_candidate(a, candidates)

    # symmetric beta family sweep (coarse grid)
    for p in np.arange(0.05, 1.55, 0.15):
        if time.time() >= deadline:
            break
        a = _beta_symmetric_distribution(n, p)
        _add_candidate(a, candidates)

    # 1️⃣  Random permutations of a good base distribution (arcsine)
    base_arc = _arcsine_distribution(n)
    for _ in range(250):
        if time.time() >= deadline:
            break
        perm = np.random.permutation(base_arc)
        _add_candidate(perm, candidates)

    # ------------------------------------------------------------------
    # 2️⃣  Inject previous best (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev, candidates)
            print(f"Injected previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the best few seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(12, len(candidates))
    seeds = [candidates[i][0] for i in range(top_k)]
    best_score = candidates[0][1]
    best_seq = candidates[0][0].copy()
    print(f"Generated {len(candidates)} seeds, best seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Intensive refinement – a long exponentiated‑gradient run on the current best
    # Use most of the remaining budget for this single promising vector.
    if time.time() < deadline - 100:
        seq, sc = _exp_gradient_descent(best_seq, deadline,
                                        max_iters=4_000_000, eta0=0.9)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"Exp‑Grad refinement -> {best_score:.6f}")

    # Adam and a little SGD pass (short, just cleanup)
    if time.time() < deadline - 60:
        seq, sc = _adam_descent(best_seq, deadline,
                                max_iters=800_000, step0=0.22)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"Adam refinement -> {best_score:.6f}")

    if time.time() < deadline - 40:
        seq, sc = _sgd_subgrad(best_seq, deadline,
                               max_iters=600_000, step0=0.45)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"SGD refinement -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Large‑scale random‑permutation search around the current best.
    # We keep the multiset of values (so the ℓ₁‑norm stays 1) but shuffle them.
    if time.time() < deadline - 30:
        vals = best_seq.copy()
        for _ in range(1500):
            if time.time() >= deadline - 20:
                break
            perm = np.random.permutation(vals)
            sc = _max_convolution_score(perm)
            if sc < best_score:
                best_score = sc
                best_seq = perm.copy()
                print(f"[{time.time():.1f}] Random perm improvement -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Simulated‑annealing hill‑climber (swap + tiny mass moves)
    if time.time() < deadline - 10:
        seq, sc = _anneal(best_seq, deadline,
                          init_temp=0.025, cooling=0.9997,
                          max_iters=1_200_000)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"Annealing improvement -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final tiny Frank‑Wolfe‑style polishing (pairwise transfer)
    if time.time() < deadline - 4:
        # Very short FW run; we reuse the same code as in the original script
        a = best_seq.copy()
        cur_score = _max_convolution_score(a)
        best_local = a.copy()
        best_local_score = cur_score
        it = 0
        while time.time() < deadline and it < 250_000:
            it += 1
            grad, _, _ = _subgrad(a)
            i_src = int(np.argmax(grad))
            i_dst = int(np.argmin(grad))
            if i_src == i_dst:
                break
            delta = min(a[i_src], 1.0 - a[i_dst])
            if delta <= 0:
                break
            step = 0.72 * delta
            a[i_src] -= step
            a[i_dst] += step
            cur_score = _max_convolution_score(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
        if best_local_score < best_score:
            best_score, best_seq = best_local_score, best_local.copy()
            print(f"Frank‑Wolfe polishing -> {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # Convert to plain Python list (evaluate_sequence will re‑normalise anyway)
    return best_seq.tolist()
