"""
Simplex‑constrained search for a low‑score step‑function.

Algorithm
---------

1. **Broad analytic scan** – several families (uniform,
   arcsine, triangular, exponential, sine‑power, Gaussian,
   symmetric Beta, power‑edge, power‑abs, V‑shape,
   Dirichlet draws) are generated for a range of lengths
   (1500‑6000).  The fast score 2·n·max(a*a) (with Σa=1) is
   evaluated via FFT; the best candidate is kept.

2. **Linear‑program refinement** – given the current vector `f`,
   we solve the LP
        maximise   Σ b_j
        subject to (f * b)_k ≤ M   for all k
                   b_j ≥ 0
   where `M` is the current maximal self‑convolution entry.
   The optimal `b` is normalised to unit sum and mixed with `f`
   (convex combination).  A short schedule of mixing factors
   successively reduces the max‑convolution bound.

3. **Smooth‑max mirror descent** – we minimise a smooth approximation
   of the maximum convolution using exponentiated‑gradient updates.
   The smoothing parameter `α` grows from 0.1 to ~200, the learning
   rate decays, and the iterate is projected onto the probability
   simplex after each step.

4. **Final hill‑climber** – a cheap greedy mass‑transfer repeatedly
   removes a little mass from the pair that contributes most to the
   current maximal convolution entry and places it at a random site.
   Occasional jitter prevents stagnation.

All steps respect a global 1000 s wall‑clock budget.
The returned list contains non‑negative floats (tiny `eps` added)
and sums to 1, which is sufficient because `evaluate_sequence`
is scale‑invariant.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]

def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a vector already normalised to Σa = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)

        right = left[:-1][::-1]
    else:

    vals /= vals.sum()
    return vals

    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals

    """Symmetric Beta(p,p) on a regular grid."""

        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w

    vals /= vals.sum()
    return vals

            if sc < best_score:
                best_seq, best_n, best_score = prev.copy(), prev.size, sc
                print(f"[{time.time():.1f}] previous global best score={sc:.6f}")
    except Exception:
        pass

    # ----------
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Helper for a single LP refinement step
    def _lp_refine(f_vec: np.ndarray, mix: float) -> np.ndarray:
        """Perform one LP optimisation and mix the result with f_vec."""
        n_loc = f_vec.shape[0]
        conv = _fft_self_convolve(f_vec)
        M_loc = conv.max()

        # Toeplitz matrix that maps b -> f * b
        col = np.concatenate([f_vec, np.zeros(n_loc - 1)])
        row = np.concatenate([f_vec[:1], np.zeros(n_loc - 1)])
        A = toeplitz(col, row)                 # shape (2n‑1, n)

        b_ub = np.full(A.shape[0], M_loc)
        c = -np.ones(n_loc)                    # maximise Σb ↔ minimise −Σb
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n_loc,
                          method='highs')
        except Exception as e:
            print(f"LP exception: {e}")
            return f_vec

        if not res.success or res.x is None:
            return f_vec

        b = res.x
        if b.sum() <= 0:
            return f_vec
        g = b / b.sum()                         # normalise to Σ=1

        new_f = (1.0 - mix) * f_vec + mix * g
        new_f = _project_onto_simplex(new_f, 1.0)
        new_f = np.maximum(new_f, eps)
        new_f /= new_f.sum()
        return new_f

    # ----------------------------------------------------------------------
    # 3. LP iterative refinement
    a = best_seq.copy()
    n = a.shape[0]
    mix_schedule = [0.5, 0.4, 0.3, 0.2, 0.1]
    for idx, mix in enumerate(mix_schedule):
        if time.time() >= deadline:
            break
        a = _lp_refine(a, mix)
        sc = _max_convolution_score(a)
        if sc < best_score:
            best_seq, best_score, best_n = a.copy(), sc, n
            print(f"[{time.time():.1f}] LP iter {idx}  score={sc:.6f}")

    # ----------------------------------------------------------------------
    # 4. Smooth‑max mirror descent (more aggressive schedule)
    a = best_seq.copy()
    n = a.shape[0]
    total_steps = 15000
    for it in range(total_steps):
        if time.time() >= deadline:
            break
        frac = it / total_steps

        # smoothing parameter grows from 0.1 to ~200
        alpha = 0.1 * (2000.0 ** frac)           # final ≈ 200
        # learning rate decays from ~3 to ~0.02
        lr = 3.0 * (1.0 - frac) + 0.02

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % 500 == 0) or (it == total_steps - 1):
            sc = _max_convolution_score(a)
            if sc < best_score - 1e-12:
                best_seq, best_score = a.copy(), sc
                print(f"[{time.time():.1f}] mirror it={it}  best={sc:.6f}")

    # ----------------------------------------------------------------------
    # 5. Final cheap hill‑climber
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq, best_score = a.copy(), sc
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if hill_iters % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
