"""
propose_candidate
-----------------
Search for a non‑negative step‑function (list of floats) that minimises
`evaluate_sequence`.  The algorithm consists of:

1. Broad exploration – generate many analytic families and random Dirichlet
   vectors for several lengths, keep the best few (the *elite* list).
2. For each elite start perform a chain of refinements
   * smooth‑max mirror‑descent (exponentiated‑gradient on a smoothed max),
   * true sub‑gradient descent with Adam‑style momentum,
   * hill‑climbing with mass transfers and support pruning,
   * short simulated‑annealing pass that makes random mass moves.
3. Keep the overall best vector found, re‑normalise to sum = 1 and return it.

All helper functions are top‑level, no closures or lambda expressions are
used.  The routine respects a user‑supplied time budget (default 1000 s)
by checking the deadline after each major loop.

# Basic utilities -------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a vector with sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the uniform grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotonically decreasing a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

# Candidate generator ----------------------------------------------------
def _generate_candidates_for_length(n):
    """Return a list of vectors (sum = 1) for length n."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws
    for alpha in (0.2, 0.5, 1.0, 2.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # Sparse random start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Shuffled versions of a few bases
    for base in (cand[0], cand[1], cand[4]):
        shuf = base.copy()
        np.random.shuffle(shuf)
        cand.append(shuf)

    return cand

# Optimisation primitives ------------------------------------------------
def _smooth_max_mirror_descent(a, deadline):
    """Exponentiated‑gradient (mirror) descent on the smooth‑max of the conv."""
    n = a.shape[0]
    eps = 1e-12

    # (α0, α1, steps, η0, η1)
    schedule = [
        (1.0, 100.0, 2500, 0.20, 0.02),
        (100.0, 1000.0, 2500, 0.02, 0.002),
        (1000.0, 5000.0, 3000, 0.002, 0.0002),
        (5000.0, 50000.0, 4000, 0.0002, 2e-05),
    ]

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smooth‑max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a, deadline):
    """True sub‑gradient descent with Adam‑style momentum & back‑tracking."""
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a, deadline):
    """Hill climbing with tiny mass transfers and support pruning."""
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- mass‑transfer step -----
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            max_delta = min(a[source] * 0.5, 0.02)
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 200 == 0:
                        print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ----- occasional pruning -----
        if it % 1500 == 0:
            for keep_frac in (0.99, 0.97, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ----- jitter to keep alive -----
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc


def _simulated_annealing(a, cur_score, deadline):
    """Simple SA that moves a tiny amount of mass each iteration."""
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = cur_score
    cur_a = a.copy()
    cur_sc = cur_score

    start_sa = time.time()
    T0 = 0.02 * best_sc  # initial temperature (relative)

    while time.time() < deadline:
        elapsed = time.time() - start_sa
        total = deadline - start_sa
        if total <= 0:
            break
        # linear cooling
        T = T0 * max(0.0, 1.0 - elapsed / total)

        # propose a move
        i = random.randrange(n)
        if cur_a[i] < eps:
            continue
        j = random.randrange(n - 1)
        if j >= i:
            j += 1
        max_delta = min(cur_a[i] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()
        new_sc = _max_convolution_score(a_new)

        if new_sc < cur_sc:
            cur_a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a_new.copy()
                print(f"[{time.time():.1f}] SA new best {best_sc:.6f}")
        else:
            if T > 0:
                accept_prob = math.exp(-(new_sc - cur_sc) / T)
                if random.random() < accept_prob:
                    cur_a = a_new
                    cur_sc = new_sc

    return best_a, best_sc

# Main orchestrator ------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad exploration – keep an elite list (size ≤ 8)
    lengths = [2000, 3500, 5000, 8000, 12000, 18000]  # spread
    elite = []   # (score, vector)

    for n in lengths:
        if time.time() >= deadline:
            break
        cand = _generate_candidates_for_length(n)
        for vec in cand:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain elite (top 8)
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate previously discovered result if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case we have no candidate
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from elite pool (≤4 restarts)
    restarts = min(4, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) start score {elite[idx][0]:.6f}")

        # smooth‑max mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # projected sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # hill‑climbing + pruning
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        # short simulated annealing pass (use remaining time)
        if time.time() < deadline:
            a_mid, sc_mid = _simulated_annealing(a_mid, sc_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity checks and return
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
