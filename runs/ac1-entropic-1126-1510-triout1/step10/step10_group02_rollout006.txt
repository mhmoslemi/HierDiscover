"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  The search consists of three phases:

1. **Broad randomized exploration** – for several lengths we generate many
   analytic base families (uniform, arcsine, beta, power‑law, Gaussian,
   cosine‑window, exponential) and, for each, a large number of random
   permutations, Dirichlet draws and sparse vectors.  The top‑K candidates
   (by the exact objective 2·n·max (conv)) are kept as an elite pool.

2. **Continuous refinement** – each elite vector is refined by
   *smooth‑max mirror descent* (exponentiated‑gradient on a log‑sum‑exp
   surrogate of the maximum) followed by *projected sub‑gradient*
   polishing using an Adam‑style update and back‑tracking line‑search.

3. **Final polishing** – a cheap hill‑climber moves tiny amounts of mass
   from the pair that currently attains the maximum convolution to random
   locations, and occasional pruning of near‑zero entries is performed.

All steps respect a 1000 s wall‑clock budget (a 2 s safety margin).  The
best sequence found is returned as a plain Python list.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.ifft(A * A, n=size).real
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.ifft(A * B, n=size).real
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):

        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _smooth_max_mirror_descent(a, deadline):
    """Exponentiated‑gradient descent on a smooth‑max surrogate."""
    n = a.shape[0]
    eps = 1e-12

    # Schedule: (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 100.0, 2500, 0.18, 0.018),
        (100.0, 1000.0, 2500, 0.018, 0.0018),
        (1000.0, 5000.0, 3000, 0.0018, 0.00018),
        (5000.0, 50000.0, 4000, 0.00018, 1.8e-05),
    ]

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # softmax weights (numerically stable)
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of smooth‑max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if (it % 500) == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a, deadline):
    """True sub‑gradient descent with Adam‑style momentum & back‑tracking."""
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.5
    iteration = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        iteration += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** iteration)

        # back‑tracking line search
        step = step0
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _max_convolution_score(cand)
            if cand_sc < best_sc - 1e-14 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-14:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={iteration} new best {cand_sc:.6f}")

        # occasional jitter to escape plateaus
        if iteration % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a, deadline):
    """Final polishing: tiny mass transfers + support pruning."""
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- mass‑transfer hill climbing ----
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find the pair (i,j) achieving the current max
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            max_delta = min(a[source] * 0.5, 0.02)
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-14:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 200 == 0:
                        print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ---- occasional pruning of tiny entries ----
        if it % 1500 == 0:
            for keep_frac in (0.99, 0.97, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-14:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ---- jitter to keep the search alive ----
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc

    return elite


def _generate_base_families(n):
    """Yield a collection of base (sum‑1) vectors for a given length."""
    # deterministic analytic families
    yield _distribution_uniform(n)
    yield _distribution_arcsine(n)
    yield _distribution_triangular(n)
    yield _distribution_exponential(n, lam=0.003)
    yield _distribution_exponential(n, lam=0.02)

    # symmetric Beta with many exponents
    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        yield _distribution_beta_sym(n, p)

    # sine‑power family
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        yield _distribution_sine_power(n, pw)

    # cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        yield _distribution_cosine_window(n, coeff)

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        yield _distribution_gaussian(n, sigma)

    # power‑law (monotone) and reversed version
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        yield pl
        yield pl[::-1]


def _explore_length(n, elite, deadline,
                    perm_per_base=50,
                    dirichlet_draws=80,
                    sparse_draws=10):
    """Explore a single length n and update the elite pool."""
    eps = 1e-12

    # 1) deterministic families + many permutations
    for base in _generate_base_families(n):
        if time.time() >= deadline:
            return elite
        base = np.maximum(base, eps)
        base /= base.sum()
        sc = _max_convolution_score(base)
        elite = _maybe_update_elite(elite, sc, base)

        # permutations
        for _ in range(perm_per_base):
            if time.time() >= deadline:
                return elite
            perm = np.random.permutation(base)
            perm = np.maximum(perm, eps)
            perm /= perm.sum()
            sc = _max_convolution_score(perm)
            elite = _maybe_update_elite(elite, sc, perm)

    # 2) random Dirichlet draws with varying concentration
    for _ in range(dirichlet_draws):
        if time.time() >= deadline:
            return elite
        # concentration in log‑space between 1e‑2 and 10
        alpha = 10 ** np.random.uniform(-2, 1)
        vec = np.random.dirichlet(np.full(n, alpha))
        vec = np.maximum(vec, eps)
        vec /= vec.sum()
        sc = _max_convolution_score(vec)
        elite = _maybe_update_elite(elite, sc, vec)

    # 3) sparse random vectors
    for _ in range(sparse_draws):
        if time.time() >= deadline:
            return elite
        k = max(2, int(np.random.uniform(2, max(5, n // 300))))
        sparse = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, k, replace=False)
        sparse[idx] = np.random.rand(k) + 0.01
        sparse /= sparse.sum()
        sc = _max_convolution_score(sparse)
        elite = _maybe_update_elite(elite, sc, sparse)

    return elite

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0  # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – broad randomized exploration (elite pool)
    elite = []                     # entries = (score, vector)
    lengths = [2048, 4096, 8192, 12288, 16384]   # diverse sizes

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")
        elite = _explore_length(
            n,
            elite,
            deadline,
            perm_per_base=80,          # many permutations per family
            dirichlet_draws=120,
            sparse_draws=15,
        )

    # ------------------------------------------------------------------
    # Record the absolute best seen so far
    if elite:
        best_score, best_seq = elite[0]
        best_n = best_seq.shape[0]
        print(f"[{time.time():.1f}] Global best after exploration {best_score:.6f} (len={best_n})")

    # ------------------------------------------------------------------
    # Incorporate any previously known solution (height_sequence_1)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            elite = _maybe_update_elite(elite, sc, prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Phase 2 – continuous refinement of the elite pool
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) init score {elite[idx][0]:.6f}")

        # smooth‑max mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # projected sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # final hill‑climber + pruning
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-14:
            best_score, best_seq = sc_mid, a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final sanity check and return
    if best_seq is None:
        # Fallback: uniform distribution of a reasonable size
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score {best_score:.6f}")

    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
