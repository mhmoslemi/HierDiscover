def _product_power_distribution(n, p):
    i = np.arange(n, dtype=np.float64)
    vals = ( (i + 0.5) * (n - i - 0.5) ) ** (-p)
    # ensure non-negative
    vals = np.clip(vals, 0, None)
    s = vals.sum()
    if s == 0:
        vals = np.full(n, 1.0/n)
    else:
        vals /= s
    return vals

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1 (returns 2·n·max)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of max‑convolution on the simplex.
    Returns (grad, cur_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for even active indices (a_i² terms)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _product_power_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+0.5)*(n-i-0.5))^{-p}, p>0 (generalises arcsine, p=0.5)."""
    i = np.arange(n, dtype=np.float64)
    vals = ((i + 0.5) * (n - i - 0.5)) ** (-p)
    vals = np.clip(vals, 0.0, None)
    s = vals.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return vals / s


def _power_arcsine_distribution(n: int, r: float) -> np.ndarray:
    """Raise the classic arcsine density to power r (r>0) and renormalise."""
    base = _arcsine_distribution(n)
    vals = np.power(base, r)
    s = vals.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return vals / s


def _step_end_distribution(n: int, k: int) -> np.ndarray:
    """
    Put equal mass on the first k and the last k positions,
    remaining mass uniformly on the interior.
    """
    if k * 2 >= n:
        k = n // 2 - 1
    interior = n - 2 * k
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    if interior > 0:
        a[k:-k] = 1.0
    a /= a.sum()
    return a


def _mixture_arcsine_uniform(n: int, lam: float) -> np.ndarray:
    """λ * arcsine + (1-λ) * uniform, λ∈[0,1]."""
    arcs = _arcsine_distribution(n)
    uni = np.full(n, 1.0 / n)
    a = lam * arcs + (1.0 - lam) * uni
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Re‑ordering helpers
def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    """Place the largest, second‑largest, … alternately at the left‑most
    and right‑most positions, spreading large masses."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left = 0
    right = n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    """Put the biggest value at the centre and then fill alternatingly
    to the left and to the right, moving outwards."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []
    mid = (n - 1) // 2
    positions.append(mid)
    for d in range(1, n):
        left = mid - d
        if left >= 0:
            positions.append(left)
        right = mid + d
        if right < n:
            positions.append(right)
    for i, pos in enumerate(positions[:n]):
        out[pos] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    """Spread large coefficients by recursively filling the midpoint of
    the remaining interval (binary‑tree layout)."""
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)

    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out


def order_random(desc_vals: np.ndarray) -> np.ndarray:
    """A random permutation of the descending values."""
    a = desc_vals.copy()
    np.random.shuffle(a)
    return a


# ----------------------------------------------------------------------
# Descent / hill‑climbing primitives
def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 1_000_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated‑gradient) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 500_000,
                  step0: float = 0.25,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 400_000,
                 step0: float = 0.6) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score, best_a = cur_score, a.copy()
    return best_a, best_score


def _swap_hill_climb(a: np.ndarray, deadline: float,
                    max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Greedy random swap of two coordinates (accept only improving moves)."""
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        cand = best.copy()
        cand[i], cand[j] = cand[j], cand[i]   # simple swap
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best, best_score = cand, sc
    return best, best_score


def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Iterated pairwise mass transfer (Frank‑Wolfe style)."""
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score, best = cur_score, a.copy()
        # occasional jitter
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
    return best, best_score


# ----------------------------------------------------------------------
def _add_candidate(arr: np.ndarray, cand_list: list):
    """Normalise, evaluate surrogate and store."""
    a = np.asarray(arr, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return
    a = a / s
    cand_list.append((a, _max_convolution_score(a)))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises the evaluation
    function.  The routine respects a ~1000 s budget and returns the best
    vector discovered (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Base length – try to continue from the previous best if available
    try:
        base_n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        base_n = 5000

    # ------------------------------------------------------------------
    # Lengths to explore (moderate, staying well under the time limit)
    lengths = set()
    for factor in (0.6, 0.8, 1.0, 1.2, 1.5):
        n = int(base_n * factor)
        if 800 <= n <= 15000:
            lengths.add(n)
    # a few random lengths for diversity
    for _ in range(4):
        n = np.random.randint(800, min(15000, base_n * 2) + 1)
        lengths.add(n)
    lengths = sorted(lengths)

    # ------------------------------------------------------------------
    # Parameter grids for the new families
    prod_p_grid = [0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.2, 1.5]            # product‑power exponent
    arcsine_r_grid = [0.8, 1.0, 1.2, 1.5, 2.0]                       # power on classic arcsine
    step_k_grid = [int(0.03 * n) for n in lengths]                # number of end‑points
    mix_lambda_grid = [0.0, 0.25, 0.5, 0.75, 1.0]                  # mixture arcsine–uniform

    # ------------------------------------------------------------------
    # Store all candidates (vector, surrogate_score)
    candidates: list[tuple[np.ndarray, float]] = []

    # ------------------------------------------------------------------
    # Helper to generate a set of reorderings for a given vector
    def _store_all_orders(v: np.ndarray):
        """Store original and three deterministic permutations."""
        _add_candidate(v, candidates)
        desc = np.sort(v)[::-1]
        _add_candidate(order_alternate_extremes(desc), candidates)
        _add_candidate(order_center_outwards(desc), candidates)
        _add_candidate(order_binary_tree(desc), candidates)
        # a random ordering occasionally helps
        _add_candidate(order_random(desc), candidates)

    # ------------------------------------------------------------------
    # 1. Broad sampling of analytic families
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- classic arcsine (baseline)
        a = _arcsine_distribution(n)
        _store_all_orders(a)

        # ---- product‑power family (new!)
        for p in prod_p_grid:
            a = _product_power_distribution(n, p)
            _store_all_orders(a)

        # ---- arcsine raised to a power (new!)
        for r in arcsine_r_grid:
            a = _power_arcsine_distribution(n, r)
            _store_all_orders(a)

        # ---- mixture arcsine / uniform
        for lam in mix_lambda_grid:
            a = _mixture_arcsine_uniform(n, lam)
            _store_all_orders(a)

        # ---- simple step‑ends shapes
        k = max(1, int(0.03 * n))
        a = _step_end_distribution(n, k)
        _store_all_orders(a)

        # ---- random Dirichlet draws for extra diversity
        for alpha in (0.2, 0.5, 1.0, 2.0):
            a = np.random.gamma(alpha, 1.0, size=n)
            a /= a.sum()
            _store_all_orders(a)

    # ------------------------------------------------------------------
    # Inject the previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev, candidates)
            print(f"Injected previous best, surrogate={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the best seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(15, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Seed best surrogate = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # ------------------------------------------------------------------
    # 2. Parameter‑wise fine search on the best length/family
    #    We try to improve the best candidate by a cheap 1‑D optimisation
    #    over its main shape parameter if it belongs to a known family.
    if time.time() < deadline - 30:
        # Detect if best belongs to product‑power family (by checking
        # similarity to the analytic form).  We simply re‑search over p.
        n_best = best_seq.shape[0]
        # 1‑D search over p ∈ [0.4, 2.0]
        p_low, p_high = 0.4, 2.0
        for _ in range(6):  # 6 refinement rounds ≈ 5⁶ points total - cheap
            pts = np.linspace(p_low, p_high, num=5)
            scores = []
            for p in pts:
                cand = _product_power_distribution(n_best, p)
                scores.append(_max_convolution_score(cand))
            idx = int(np.argmin(scores))
            best_p = pts[idx]
            best_score = scores[idx]
            # narrow interval around the best point
            delta = (p_high - p_low) / 4
            p_low = max(0.1, best_p - delta)
            p_high = best_p + delta
        # adopt the refined distribution (with ordering that gave smallest score)
        refined = _product_power_distribution(n_best, best_p)
        # try the three deterministic reorderings again
        for ord_func in (lambda x: x,
                         order_alternate_extremes,
                         order_center_outwards,
                         order_binary_tree):
            cand = ord_func(refined)
            sc = _max_convolution_score(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"Param‑search improvement (p={best_p:.4f}) -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Intensive refinement of the top seeds (exp‑grad → Adam → SGD)
    #    The remaining time is divided roughly equally among the seeds.
    remaining = max(0.0, deadline - time.time())
    seeds_left = len(top_candidates)
    for idx, (seed_seq, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # Give each seed about equal share of the remaining budget
        per_seed_deadline = time.time() + remaining / max(seeds_left, 1) - 1.0
        remaining = max(0.0, deadline - time.time())
        seeds_left -= 1

        # ----- exponentiated gradient (most effective)
        seq, sc = _exp_gradient_descent(seed_seq, per_seed_deadline,
                                        max_iters=2_000_000,
                                        eta0=0.9)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[ExpGrad seed {idx+1}] -> {best_score:.6f}")

        # ----- Adam
        if time.time() < per_seed_deadline:
            seq, sc = _adam_descent(best_seq, per_seed_deadline,
                                    max_iters=500_000,
                                    step0=0.25)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[Adam seed {idx+1}]    -> {best_score:.6f}")

        # ----- SGD
        if time.time() < per_seed_deadline:
            seq, sc = _sgd_subgrad(best_seq, per_seed_deadline,
                                   max_iters=400_000,
                                   step0=0.6)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[SGD seed {idx+1}]    -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local combinatorial polishing (swap hill‑climb)
    if time.time() < deadline - 30:
        seq, sc = _swap_hill_climb(best_seq, deadline,
                                   max_iters=300_000)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Swap hill‑climb improvement -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Frank‑Wolfe style pairwise mass transfer
    if time.time() < deadline - 15:
        seq, sc = _frank_wolfe_opt(best_seq, deadline,
                                   max_iters=400_000)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Frank‑Wolfe polishing -> {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final surrogate score = {best_score:.6f} (length={len(best_seq)})")
    # Return as plain Python list (evaluate_sequence will renormalise if needed)
    return best_seq.tolist()
