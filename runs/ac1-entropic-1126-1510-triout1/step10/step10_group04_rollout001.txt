"""
Optimise a non‑negative step‑function to minimise
  2·n·max(a*a) / (∑a)²

The algorithm works in three phases:

1️⃣  Broad scan – generate many analytic/sampled families on several
    lengths (2000–10000).  The cheap FFT based score is used to keep the
    best candidate.

2️⃣  Adam‑based mirror descent on the *smooth‑max* of the self‑convolution.
    The gradient of  maxₖ(a∗a)ₖ is approximated by a soft‑max weighting.
    Multiplicative (exponentiated) updates keep the vector non‑negative;
    after each step we project onto the probability simplex.

3️⃣  A short simulated‑annealing refinement and a final tiny hill‑climber.
    All steps respect a global wall‑clock of ``budget_s`` seconds.

The returned list satisfies the security checks of ``evaluate_sequence``.
"""

import random
import math

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = s}.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using an FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Cheap evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """Edge‑heavy family w_i ∝ ((i+0.5)*(n‑i‑0.5))^{-α}."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    """Linear increase from centre to the edges, tiny epsilon avoids zeros."""
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    """Symmetric power law |i‑centre|^{β} (β>0 = edge‑heavy)."""
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Symmetric Dirichlet draw (tiny α → very edge‑heavy)."""
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full


def _random_powered_uniform(n: int, power: float) -> np.ndarray:
    """Draw u∈[0,1]ⁿ, raise to “power” and normalise (produces many shapes)."""
    u = np.random.rand(n) ** power
    u = np.maximum(u, 1e-12)
    u /= u.sum()
    return u

        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find pair (i,j) with maximal product a_i * a_j that contributes to k
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            sc = 2.0 * n * M
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if hill_iters % 5000 == 0:
            # tiny random jitter to escape flat regions
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter at hill iter {hill_iters}")

    # ------------------------------------------------------------------
    # 6. (optional) up‑sample and re‑evaluate – sometimes a small up‑sampling
    #    reduces the max by spreading mass more finely.
    try:
        up = np.repeat(best_seq, 2)           # double length, keep total mass
        up = np.maximum(up, eps)
        up /= up.sum()
        sc_up = _max_convolution_score(up)
        if sc_up < best_score:
            best_seq = up
            best_score = sc_up
            best_n = up.shape[0]
            print(f"[{time.time():.1f}] up‑sampled improvement  score={sc_up:.6f}")
    except Exception:
        pass

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
