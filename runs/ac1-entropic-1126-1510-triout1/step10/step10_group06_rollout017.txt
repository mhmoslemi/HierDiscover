def refine_candidate(a, n, deadline):
    a = np.maximum(a, 1e-12)
    a = a / a.sum()
    best_local = a.copy()
    best_local_score = _max_convolution_score(a)
    # EG
    steps_eg = 4000
    a_eg = a.copy()
    for it in range(steps_eg):
        if time.time() >= deadline: break
        t = it/steps_eg
        alpha = 1.0 * (5000.0/1.0)**t  # start 1 to 5000
        lr = 0.2 * (0.001/0.2)**t    # start .2 to .001
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)
        # evaluate
        if it % 200 == 0 or it == steps_eg - 1:
            sc = _max_convolution_score(a_eg)
            if sc < best_local_score:
                best_local = a_eg.copy()
                best_local_score = sc
    # PGD
    steps_pg = 2000
    a_pg = best_local.copy()
    for it in range(steps_pg):
        if time.time() >= deadline: break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = 0.3 * (0.005/0.3)**(it/steps_pg)  # schedule
        a_pg = _projected_gradient_step(a_pg, eta, w)
        if it % 200 == 0 or it == steps_pg - 1:
            sc = _max_convolution_score(a_pg)
            if sc < best_local_score:
                best_local = a_pg.copy()
                best_local_score = sc
    # Hillclimber
    a_hc = best_local.copy()
    conv = _fft_self_convolve(a_hc)
    hc_iters = 20000
    for it in range(hc_iters):
        if time.time() >= deadline: break
        a_hc, conv, improved = _hillclimber_move(a_hc, conv)
        if improved:
            sc = 2.0 * n * conv.max()

                best_local = a_hc.copy()

            # random jitter
            jitter = np.random.randn(n) * 1e-6
            a_hc += jitter
            a_hc = np.maximum(a_hc, 1e-12)
            a_hc = a_hc/a_hc.sum()
            conv = _fft_self_convolve(a_hc)

"""
Multi‑restart optimizer for the step‑function problem.

The algorithm repeatedly draws a candidate distribution (either from
analytic families or a Dirichlet draw), normalises it to sum = 1 and then
refines it with a cascade of updates:

1. Exponential‑gradient descent on a smooth approximation of the
   max‑convolution (soft‑max with a temperature that is gradually
   lowered).
2. Projected gradient descent on the true sub‑gradient of the max.
3. Adam optimisation on the logits (i.e. the unconstrained soft‑max
   parametrisation) – this naturally enforces the simplex constraint.
4. A cheap hill‑climber that moves a tiny amount of mass from the
   largest‑gradient index to the smallest‑gradient index.

Larger sequence lengths give a lower theoretical bound (≈ 1) because the
objective is 2·n·max(conv) and max(conv) ≥ 1/(2n−1).  We therefore try
lengths up to 20000 and keep the best sequence found before the time
budget (1000 s) elapses.
"""

import time
import math
import random

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """2·n·max(a*a) assuming Σa = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with Σ = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _smooth_weights(conv, alpha):
    """Soft‑max weighting of conv (smooth max)."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    w /= s
    return w


def _subgradient_weights(conv, eps_rel=1e-12):
    """Uniform weights over entries that are (almost) maximal."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    w /= cnt
    return w

def _exponential_gradient_step(a, lr, alpha):
    """Multiplicative EG update (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """Additive PGD step on the sub‑gradient of the max‑convolution."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, grad):
    """Adam update for logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)   # avoid overflow
    return logits, m, v


def _hillclimber_move(a, conv, eps=1e-12):
    """Move a tiny chunk of mass from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    return (a_new, new_conv, True) if improved else (a, conv, False)


# ----------------------------------------------------------------------
def _refine_candidate(a0, n, deadline):
    """
    Run a full optimisation pipeline on a candidate of length n.
    Returns (best_vector, best_score).
    The pipeline consists of EG → PGD → Adam → hill‑climber.
    """
    eps = 1e-12
    a = np.maximum(a0, eps)
    a /= a.sum()

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # ---- Exponential‑gradient phase ---------------------------------
    steps_eg = 4000
    a_eg = a.copy()
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        alpha = 1.0 * (5000.0 / 1.0) ** t          # 1 → 5000
        lr = 0.2 * (0.001 / 0.2) ** t               # 0.2 → 0.001
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)
        if it % 300 == 0 or it == steps_eg - 1:
            sc = _max_convolution_score(a_eg)
            if sc < best_sc:
                best_a, best_sc = a_eg.copy(), sc
                print(f"[{time.time():.1f}] EG it={it}  best={sc:.6f}")

    # ---- Projected‑gradient phase ----------------------------------
    steps_pg = 2000
    a_pg = best_a.copy()
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = 0.30 * (0.005 / 0.30) ** (it / steps_pg)   # 0.30 → 0.005
        a_pg = _projected_gradient_step(a_pg, eta, w)
        if it % 200 == 0 or it == steps_pg - 1:
            sc = _max_convolution_score(a_pg)
            if sc < best_sc:
                best_a, best_sc = a_pg.copy(), sc
                print(f"[{time.time():.1f}] PG it={it}  best={sc:.6f}")

    # ---- Adam on logits --------------------------------------------
    logits = np.log(a_pg + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    steps_adam = 3000
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_ad = np.exp(logits)
        a_ad /= a_ad.sum()
        conv = _fft_self_convolve(a_ad)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        # gradient w.r.t. a
        grad_a = 2.0 * _fft_convolve(a_ad[::-1], w)[n - 1 : n - 1 + n]
        # gradient w.r.t. logits (soft‑max trick)
        grad_l = (grad_a - np.dot(a_ad, grad_a)) * a_ad
        logits, m, v = _adam_step(logits, m, v, it, grad_l)
        if it % 400 == 0 or it == steps_adam:
            a_ad = np.exp(logits)
            a_ad /= a_ad.sum()
            sc = _max_convolution_score(a_ad)
            if sc < best_sc:
                best_a, best_sc = a_ad.copy(), sc
                print(f"[{time.time():.1f}] Adam it={it}  best={sc:.6f}")

    # ---- Simple hill‑climber ----------------------------------------
    a_hc = best_a.copy()
    conv = _fft_self_convolve(a_hc)
    hc_iters = 20000
    for it in range(hc_iters):
        if time.time() >= deadline:
            break
        a_hc, conv, improved = _hillclimber_move(a_hc, conv, eps=eps)
        if improved:
            sc = 2.0 * n * conv.max()
            if sc < best_sc:
                best_a, best_sc = a_hc.copy(), sc
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] hill it={it}  best={sc:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5
            a_hc += jitter
            a_hc = np.maximum(a_hc, eps)
            a_hc /= a_hc.sum()
            conv = _fft_self_convolve(a_hc)

    return best_a, best_sc


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing ``evaluate_sequence``.
    Returns the best sequence found within the allocated time.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Try to import a previously discovered sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq, best_score, best_n = prev.copy(), sc, prev.size
                print(f"[{time.time():.1f}] loaded previous best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Basic analytic families (quick scan)
    init_lengths = [2000, 3000, 5000, 8000, 10000]
    for n in init_lengths:
        if time.time() >= deadline:
            break
        candidates = [
            _distribution_uniform(n),
            _distribution_triangular(n),
            _distribution_exponential(n, lam=0.005),
            _distribution_exponential(n, lam=0.02),
            _distribution_beta_sym(n, p=0.5),
            _distribution_beta_sym(n, p=1.5),
            _distribution_sine_power(n, power=0.7),
            _distribution_sine_power(n, power=2.5),
            _distribution_gaussian(n, sigma=0.28),
        ]
        # a few random Dirichlet draws
        for alpha in (0.3, 0.6, 1.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq, best_score, best_n = vec.copy(), sc, n
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # If nothing was found (unlikely) fall back to uniform
    if best_seq is None:
        best_n = 2500
        best_seq = _distribution_uniform(best_n)
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main multi‑restart loop
    restart_lengths = [2000, 3000, 5000, 8000, 10000, 15000, 20000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        # initialise a new Dirichlet vector (alpha = 0.5 favours sparsity)
        a0 = np.random.dirichlet(np.full(n, 0.5))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        a_opt, sc_opt = _refine_candidate(a0, n, deadline)
        if sc_opt < best_score:
            best_seq, best_score, best_n = a_opt.copy(), sc_opt, n
            print(f"[{time.time():.1f}] restart n={n}  new best={sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Final interpolation to a slightly larger size (often reduces the constant)
    if best_seq is not None:
        for factor in (1.1, 1.25, 1.5):
            n_new = int(best_n * factor)
            if n_new <= best_n or time.time() >= deadline - 5.0:
                continue
            x_old = np.linspace(0.0, 1.0, best_n)
            x_new = np.linspace(0.0, 1.0, n_new)
            vec = np.interp(x_new, x_old, best_seq)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            a_opt, sc_opt = _refine_candidate(vec, n_new, deadline)
            if sc_opt < best_score:
                best_seq, best_score, best_n = a_opt.copy(), sc_opt, n_new
                print(f"[{time.time():.1f}] interp n={n_new}  best={sc_opt:.6f}")

    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
