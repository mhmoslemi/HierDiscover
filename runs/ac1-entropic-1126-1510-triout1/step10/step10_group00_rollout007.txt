def _smooth_obj_and_grad(a: np.ndarray, mu: float) -> tuple[float, np.ndarray]:
    n = a.shape[0]
    # compute convolution a*a
    conv = _fft_self_convolve(a)  # shape 2n-1
    # compute stable softmax
    scaled = conv / mu
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_scaled)
    # objective
    obj = mu * (max_scaled + np.log(sum_exp))
    # weights
    w = exp_scaled / sum_exp   # shape 2n-1
    # gradient: 2 * sum_i a_i * w_{i+j}
    w_rev = w[::-1]
    # convolution a (n) with w_rev (2n-1) full length = 3n-2

    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_rev = np.fft.irfft(A * W, n=size)[: 3 * n - 2]
    grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]
    return obj, grad

"""
Propose a step‑function (non‑negative vector) that minimizes
evaluate_sequence.

Algorithm
---------
1. Scan a small set of analytic families (uniform, arcsine,
   edge‑weighted with a dense grid of exponents) for a few lengths.
   Keep the best few candidates.
2. Refine each candidate with a *smooth* Adam optimiser.
   We minimise the smooth approximation
        f_μ(a) = μ·log Σ_k exp( (a∗a)_k / μ )
   of the max‑convolution objective.  The gradient is obtained by a
   single FFT‑based convolution, then the iterate is projected onto the
   probability simplex.
   μ is slowly decreased from 0.2 to 0.01 during optimisation.
3. After the Adam phase a cheap greedy pairwise mass‑transfer walk
   (as in the reference code) is performed to polish the best
   vector.
4. The best vector found (as a plain Python list) is returned.
   All numbers are non‑negative; the evaluation function will apply the
   required normalisation itself.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1 - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv_full = np.fft.irfft(A * A, n=size)
    return conv_full[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return float(2.0 * n * b.max())


# ----------------------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(a: np.ndarray, mu: float):
    """
    Returns (f_μ(a), ∇f_μ(a)) where
        f_μ(a) = μ·log Σ_k exp( (a∗a)_k / μ )
    and the gradient is computed via a single FFT convolution.
    """
    n = a.shape[0]

    # ----- convolution a*a (full) -----
    conv = _fft_self_convolve(a)          # length 2n‑1

    # ----- smooth max (log‑sum‑exp) -----
    scaled = conv / mu
    max_scaled = np.max(scaled)
    exp_shifted = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_shifted)
    obj = mu * (max_scaled + np.log(sum_exp))

    # ----- softmax weights -----
    w = exp_shifted / sum_exp               # length 2n‑1, sum(w)=1

    # ----- gradient = 2·∑_i a_i w_{i+j} -----
    w_rev = w[::-1]                         # reverse for convolution
    size = 1 << ((3 * n - 2 - 1).bit_length())   # enough for a (n) + w_rev (2n‑1)
    A = np.fft.rfft(a, n=size)
    W = np.fft.rfft(w_rev, n=size)
    conv_rev = np.fft.irfft(A * W, n=size)[: 3 * n - 2]   # length 3n‑2
    grad = 2.0 * conv_rev[2 * n - 2 - np.arange(n)]

    return float(obj), grad


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   deadline: float,
                   best_seq: np.ndarray,
                   best_score: float,
                   lr: float = 0.1,
                   mu_start: float = 0.2,
                   mu_end: float = 0.01,
                   eval_every: int = 150) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth objective.
    Returns the best sequence (on the true max‑convolution objective)
    discovered within the time budget.
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    start = time.time()
    total_time = deadline - start

    while time.time() < deadline:
        t += 1
        # linearly decay μ from mu_start to mu_end
        elapsed = time.time() - start
        mu = max(mu_end, mu_start * (1.0 - elapsed / total_time))

        # smooth objective + gradient
        _, grad = _smooth_obj_and_grad(a, mu)

        # Adam update (minimisation → move opposite to gradient)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # project back onto the probability simplex
        a = _project_onto_simplex(a)

        # occasional exact evaluation
        if t % eval_every == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t} new global best {best_score:.6f}")

        # safety break (very large number of iterations would be wasteful)
        if t >= 300000:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Cheap random mass‑transfer walk that keeps improving moves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function that minimises evaluate_sequence().
    Returns a Python list of non‑negative heights.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 2.0                     # a tiny safety margin

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 1.  Scan a compact pool of analytic families for several lengths.
    lengths = [4000, 8000, 16000, 32000]                  # growing resolutions
    # dense grid for the exponent p (U‑shaped family)
    p_grid = np.concatenate(
        [
            np.logspace(-2, -0.5, 6),    # 0.01 … 0.316
            np.logspace(-0.5, 0.5, 12),  # 0.316 … 3.16 (covers arcsine at p≈0.5)
        ]
    )
    lam_grid = np.linspace(0.0, 1.0, 11)                 # mixture with uniform

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        # Uniform (baseline)
        uniform = _distribution_uniform(n)
        sc = _objective(uniform)
        top_candidates.append((sc, uniform.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = uniform.copy()
            print(f"[{time.time():.1f}] new global best (uniform) len={n} score={sc:.6f}")

        # Arcsine – known good shape
        arcsine = _distribution_arcsine(n)
        sc = _objective(arcsine)
        top_candidates.append((sc, arcsine.copy()))
        if sc < overall_best_score:
            overall_best_score = sc
            overall_best_seq = arcsine.copy()
            print(f"[{time.time():.1f}] new global best (arcsine) len={n} score={sc:.6f}")

        # Edge‑weighted family for many p values
        for p in p_grid:
            if time.time() >= deadline:
                break
            edge = _distribution_edge_weighted(n, p)
            sc = _objective(edge)
            top_candidates.append((sc, edge.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = edge.copy()
                print(f"[{time.time():.1f}] new global best (edge p={p:.4g}) len={n} score={sc:.6f}")

            # Mix edge‑weighted with uniform – cheap extra degree of freedom
            for lam in lam_grid:
                if time.time() >= deadline:
                    break
                cand = _mix_with_uniform(edge, lam)
                sc = _objective(cand)
                top_candidates.append((sc, cand.copy()))
                if sc < overall_best_score:
                    overall_best_score = sc
                    overall_best_seq = cand.copy()
                    print(f"[{time.time():.1f}] new global best (edge p={p:.4g},λ={lam:.2f}) len={n} score={sc:.6f}")

        # A few random Dirichlet draws (different concentration)
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.05, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best (Dirichlet) len={n} score={sc:.6f}")

        # keep only a modest heap of the best seeds seen so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  Incorporate a previously discovered construction if it exists.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)    # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback – should never happen
    if overall_best_seq is None:
        n = 4000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Refine the top few seeds with Adam on the smooth objective.
    remaining = deadline - time.time()
    if remaining > 20.0 and top_candidates:
        # take the five most promising candidates
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] starting Adam refinements ({len(seeds)} seeds)")

        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _adam_optimize(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr=0.12,
                mu_start=0.2,
                mu_end=0.01,
                eval_every=120,
            )
            if sc_opt < overall_best_score:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(f"[{time.time():.1f}] after Adam seed {idx} new global best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final greedy local refinement (pairwise swaps)
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy pairwise swaps")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after swap refinement new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # evaluate_sequence will clip/normalise, we simply return the raw list.
    return overall_best_seq.tolist()
