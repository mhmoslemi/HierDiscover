def refine_vector(a, deadline):
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(best_a)
    # exponential gradient
    steps_eg = 5000
    a_eg = a.copy()
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        # schedule: alpha from 0.1 to 5000 geometrically
        alpha = 0.1 * (5000/0.1)**t
        lr = 0.2 * (0.001/0.2)**t
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)
        # compute score periodically
        if it % 200 == 0 or it == steps_eg-1:
            sc = _max_convolution_score(a_eg)
            if sc < best_score - 1e-13:
                best_score, best_a = sc, a_eg.copy()
                print(f"[{time.time():.1f}] EG it={it} score={sc:.6f}")
    # projected gradient
    a_pg = a_eg.copy()
    steps_pg = 3000
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        # step size schedule: start 0.3 to 0.001
        eta = 0.3 * (0.001/0.3)**(it/steps_pg)
        a_pg = _projected_gradient_step(a_pg, eta, w)
        if it % 200 == 0 or it == steps_pg-1:
            sc = _max_convolution_score(a_pg)
            if sc < best_score - 1e-13:
                best_score, best_a = sc, a_pg.copy()
                print(f"[{time.time():.1f}] PG it={it} score={sc:.6f}")
    # Adam on logits
    a_ad = a_pg.copy()
    logits = np.log(a_ad + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    steps_adam = 4000
    for it in range(1, steps_adam+1):
        if time.time() >= deadline:
            break
        a_cur = np.exp(logits)
        a_cur /= a_cur.sum()

        grad_a = 2.0 * _fft_convolve(a_cur[::-1], w)[n-1:n-1+n]

            if sc < best_score - 1e-13:
                best_score, best_a = sc, a_cur.copy()
                print(f"[{time.time():.1f}] Adam it={it} score={sc:.6f}")
    # Hillclimber
    a_hc = best_a.copy()

    hc_iters = 50000

            break

        if improved:
            sc = 2.0 * n * conv.max()

                best_score, best_a = sc, a_hc.copy()
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] HC it={it} score={sc:.6f}")
        # jitter to escape

            a_hc += jitter
            a_hc = np.maximum(a_hc, 1e-12)
            a_hc /= a_hc.sum()
            conv = _fft_self_convolve(a_hc)
    return best_a, best_score

"""
A heuristic optimizer for the step‑function bound.

The search proceeds in several stages:

1. **Analytic families** – a broad catalogue of parameterised
   probability vectors (uniform, arcsine, triangular, exponential,
   symmetric Beta, sine‑power, Gaussian) is evaluated for several
   lengths (up to 10 000).  The best few candidates are kept.

2. **Mixture generation** – convex combinations of the top candidates
   are created (random Dirichlet weights) and evaluated; this quickly
   explores the neighbourhood of promising constructions.

3. **Interpolation** – the current best vector is linearly interpolated
   to a few larger lengths, providing a refined starting point.

4. **Iterative refinement** – for a given length the vector is improved
   by a pipeline:
   * Exponential‑gradient updates with a smooth‑max approximation.
   * Projected‑gradient descent with a sub‑gradient of the max‑convolution.
   * Adam optimisation on the log‑it (soft‑max) parametrisation.
   * A cheap hill‑climber that moves a tiny chunk of mass from the
     largest‑gradient index to the smallest‑gradient index, interspersed
     with small random jitter to escape plateaus.

   The pipeline stops when the time budget is exhausted.

5. **Multiple restarts** – while time permits fresh random Dirichlet
   vectors of various lengths are refined with a shortened pipeline; any
   improvement updates the global best.

All vectors are kept on the simplex (non‑negative, sum = 1), because the
objective `2·n·max(a*a)/(sum a)^2` is invariant to scaling.  The code
only uses NumPy and runs comfortably within the 1000 s wall‑clock limit.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) for very small p
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_weights(conv, alpha):
    """Soft‑max weighting for a smooth approximation of max."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """(sub)gradient weights for the true max – uniform on near‑active indices."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _exponential_gradient_step(a, lr, alpha):
    """One EG iteration (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1: len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """PGD step on the (sub)gradient defined by active_w."""
    n = a.shape[0]
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[n - 1: n - 1 + n]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, step):
    """Adam update for logits (soft‑max parametrisation)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * step
    v = beta2 * v + (1.0 - beta2) * (step ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)          # avoid overflow
    return logits, m, v


def _hillclimber_move(a, conv, eps=1e-12):
    """Transfer a tiny chunk of mass from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1: n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.04)          # a slightly larger move than before
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False

def _refine_vector(a, deadline):
    """
    Apply EG → PGD → Adam → hill‑climber to a vector `a`.
    Returns the best vector found and its score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = _max_convolution_score(best_a)

    # ---------- Exponential‑gradient phase ----------
    steps_eg = 5000
    a_eg = a.copy()
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        # α grows from 0.1 to 5 000 (geometric), lr from 0.2 to 0.001
        alpha = 0.1 * (5000.0 / 0.1) ** t
        lr = 0.2 * (0.001 / 0.2) ** t
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)
        if it % 250 == 0 or it == steps_eg - 1:
            sc = _max_convolution_score(a_eg)
            if sc < best_score - 1e-13:
                best_score, best_a = sc, a_eg.copy()
                print(f"[{time.time():.1f}] EG it={it}  score={sc:.6f}")

    # ---------- Projected‑gradient phase ----------
    steps_pg = 3000
    a_pg = a_eg.copy()
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = 0.3 * (0.001 / 0.3) ** (it / steps_pg)
        a_pg = _projected_gradient_step(a_pg, eta, w)
        if it % 200 == 0 or it == steps_pg - 1:
            sc = _max_convolution_score(a_pg)
            if sc < best_score - 1e-13:
                best_score, best_a = sc, a_pg.copy()
                print(f"[{time.time():.1f}] PG it={it}  score={sc:.6f}")

    # ---------- Adam on logits ----------
    steps_adam = 4000
    logits = np.log(a_pg + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_cur = np.exp(logits)
        a_cur /= a_cur.sum()
        conv = _fft_self_convolve(a_cur)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        grad_a = 2.0 * _fft_convolve(a_cur[::-1], w)[n - 1: n - 1 + n]
        # gradient w.r.t. logits under the soft‑max map
        grad_logits = (grad_a - np.dot(a_cur, grad_a)) * a_cur
        logits, m, v = _adam_step(logits, m, v, it, grad_logits)
        if it % 250 == 0 or it == steps_adam:
            a_cur = np.exp(logits)
            a_cur /= a_cur.sum()
            sc = _max_convolution_score(a_cur)
            if sc < best_score - 1e-13:
                best_score, best_a = sc, a_cur.copy()
                print(f"[{time.time():.1f}] Adam it={it}  score={sc:.6f}")

    # ---------- Hill‑climber ----------
    a_hc = best_a.copy()
    conv = _fft_self_convolve(a_hc)
    hc_iters = 60000
    for it in range(hc_iters):
        if time.time() >= deadline:
            break
        a_hc, conv, improved = _hillclimber_move(a_hc, conv, eps=1e-12)
        if improved:
            sc = 2.0 * n * conv.max()
            if sc < best_score - 1e-13:
                best_score, best_a = sc, a_hc.copy()
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] HC it={it}  score={sc:.6f}")
        # occasional jitter
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5
            a_hc += jitter
            a_hc = np.maximum(a_hc, 1e-12)
            a_hc /= a_hc.sum()
            conv = _fft_self_convolve(a_hc)

    return best_a, best_score

        for p in (0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 2.0, 4.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.2, 0.5, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_sine_power(n, pw))

        for sigma in (0.15, 0.25, 0.35, 0.45):
            candidates.append(_distribution_gaussian(n, sigma))

        # random draws with various concentrations
        for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
            candidates.append(np.random.dirichlet(np.full(n, alpha)))

        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            _maybe_update(vec)

    # ------------------------------------------------------------------
    # 2. Incorporate the previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            _maybe_update(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Random convex mixtures of the current elite
    if len(top_k) >= 2:
        for _ in range(30):
            if time.time() >= deadline:
                break
            k = random.randint(2, min(4, len(top_k)))
            chosen = random.sample(top_k, k)
            weights = np.random.dirichlet(np.ones(k))
            mix = np.sum([w * v for w, (_, v) in zip(weights, chosen)], axis=0)
            mix = np.maximum(mix, eps)
            mix /= mix.sum()
            _maybe_update(mix)

    # ------------------------------------------------------------------
    # 4. Interpolate the current best to a few larger lengths
    extra_lengths = [9000, 12000, 15000]
    if best_seq is not None:
        for n in extra_lengths:
            if time.time() >= deadline:
                break
            x_old = np.linspace(0.0, 1.0, best_seq.size)
            x_new = np.linspace(0.0, 1.0, n)
            vec = np.interp(x_new, x_old, best_seq)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            _maybe_update(vec)

    # ------------------------------------------------------------------
    # 5. Refine the current best vector
    if best_seq is None:
        # extremely unlikely, but guard against the empty case
        best_seq = _distribution_uniform(3000)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        best_n = best_seq.size

    best_seq, best_score = _refine_vector(best_seq, deadline)
    best_n = best_seq.shape[0]

    # ------------------------------------------------------------------
    # 6. Multiple random restarts while time permits
    restart_lengths = [2000, 3000, 4000, 6000, 8000, 10000, 12000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        a0 = np.random.dirichlet(np.full(n, 0.6))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        a_ref, sc_ref = _refine_vector(a0, deadline)
        if sc_ref < best_score - 1e-13:
            best_seq, best_score = a_ref, sc_ref
            best_n = n
            print(f"[{time.time():.1f}] restart n={n}  new best={best_score:.6f}")

    print(f"Final best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
