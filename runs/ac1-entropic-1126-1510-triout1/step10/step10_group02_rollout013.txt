        a /= a.sum()
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # compute w = exp(alpha*(conv-maxc)), normalized
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()

            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient w.r.t a

        grad_a = 2.0 * conv_rev[n-1:n-1+n]
        # convert to gradient w.r.t b via softmax rule
        dot = np.dot(a, grad_a)
        grad_b = a * (grad_a - dot)
        # objective value: smooth max
        smooth_val = maxc + (1.0/alpha) * np.log(w_sum)  # because we used shift
        # correct: smooth_val = (1/alpha) * log (Σ exp(alpha conv)).

        # so Σ exp(alpha conv) = w_sum * exp(alpha maxc)
        # => log Σ exp(alpha conv) = log w_sum + alpha maxc
        # => f = (1/alpha) * (log w_sum + alpha maxc) = maxc + (1/alpha)*log w_sum
        # So smooth_val computed fine.
        return smooth_val, grad_b.astype(np.float64)

    b_opt = b0

            break

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is exactly the value that evaluate_sequence would return.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _softmax(b: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a vector that sums to 1)."""
    maxb = np.max(b)
    expb = np.exp(b - maxb)
    return expb / expb.sum()


def _smooth_max_func(b: np.ndarray, alpha: float, n: int) -> float:
    """Smooth‑max surrogate fα(b) = (1/α)·log Σ exp(α·conv)."""
    a = _softmax(b)
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0.0:  # extremely unlikely, but keep safe
        w_sum = 1.0
    # f = maxc + (1/α)·log Σ exp(α(conv-maxc)) = maxc + (1/α)·log(w_sum)
    return float(maxc + (1.0 / alpha) * np.log(w_sum))


def _smooth_max_grad(b: np.ndarray, alpha: float, n: int) -> np.ndarray:
    """Gradient of the smooth‑max surrogate w.r.t. the softmax parameters."""
    a = _softmax(b)
    conv = _fft_self_convolve(a)
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    # gradient w.r.t. a (see the mirror‑descent code)
    conv_rev = _fft_convolve(a[::-1], w)
    grad_a = 2.0 * conv_rev[n - 1 : n - 1 + n]
    # chain‑rule for softmax parametrisation
    dot = np.dot(a, grad_a)
    grad_b = a * (grad_a - dot)
    return grad_b.astype(np.float64)


def _lbfgs_smooth_max(a0: np.ndarray, deadline: float):
    """
    Optimise a smooth‑max approximation of the max‑convolution using
    L‑BFGS‑B on the softmax parametrisation.  A schedule of increasing
    α values is used; the best vector (in terms of the true objective)
    encountered during the schedule is returned.
    """
    n = a0.shape[0]
    eps = 1e-12
    b_opt = np.log(np.maximum(a0, eps))

    best_a = a0.copy()
    best_sc = _max_convolution_score(best_a)

    # Geometric schedule – more aggressive towards the end
    alphas = [1.0, 10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]

    for alpha in alphas:
        if time.time() >= deadline:
            break
        # L‑BFGS‑B – a few hundred iterations are plenty per α
        res = scipy.optimize.minimize(
            fun=_smooth_max_func,
            x0=b_opt,
            args=(alpha, n),
            jac=_smooth_max_grad,
            method="L-BFGS-B",
            options={"maxiter": 250, "ftol": 1e-12, "gtol": 1e-12},
        )
        b_opt = res.x
        a_candidate = _softmax(b_opt)
        sc_candidate = _max_convolution_score(a_candidate)
        if sc_candidate < best_sc:
            best_sc = sc_candidate
            best_a = a_candidate.copy()
            print(f"[{time.time():.1f}] L‑BFGS α={alpha:.0f} new best {sc_candidate:.6f}")

    return best_a, best_sc

def _smooth_max_mirror_descent(a, deadline):
    """Legacy mirror‑descent (kept as fallback if SciPy not available)."""
    n = a.shape[0]
    eps = 1e-12

    schedule = [
        (1.0, 100.0, 2500, 0.20, 0.02),
        (100.0, 1000.0, 2500, 0.02, 0.002),
        (1000.0, 5000.0, 3000, 0.002, 0.0002),
        (5000.0, 50000.0, 4000, 0.0002, 2e-05),
    ]

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 500 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a, deadline):
    """True sub‑gradient descent with Adam‑style momentum & back‑tracking."""
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a, deadline):

    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- hill climbing step (mass transfer) -----------------------
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j

def _generate_candidates_for_length(n):
    """Return a list of candidate vectors (sum = 1) for a given length."""
    cand = []

    # Analytic families (already normalised)
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta
    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Random Dirichlet draws
    for alpha in (0.2, 0.5, 1.0, 2.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Sparse random start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Random shuffles of a few base families (add asymmetry)
    for base in (cand[0], cand[1], cand[4]):
        shuf = base.copy()
        np.random.shuffle(shuf)
        cand.append(shuf)

    return cand

                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case no candidate was found
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(4, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) "
              f"init score {elite[idx][0]:.6f}")

        # ---- Smooth‑max optimisation (L‑BFGS if available) ------------
        if _SCIPY_AVAILABLE:
            a_mid, sc_mid = _lbfgs_smooth_max(a0, deadline)
            print(f"[{time.time():.1f}] L‑BFGS finished, score {sc_mid:.6f}")
        else:
            a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)
            print(f"[{time.time():.1f}] Mirror descent finished, score {sc_mid:.6f}")

        # ---- Projected sub‑gradient polishing ------------------------
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # ---- Final hill‑climber & pruning ----------------------------
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity checks and return
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
