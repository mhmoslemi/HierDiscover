"""
Hybrid optimisation for the step‑function attack (new version).

The previous implementation already found a very good sequence (≈1.50996).
To push the bound further we add a *random‑phase* family of seeds.
A sequence whose discrete Fourier transform has (almost) constant magnitude
produces an almost flat self‑convolution, which is the ideal situation
(maximum ≈ average).  By sampling many random phase spectra, shifting the
result to be non‑negative and normalising, we obtain a diverse pool of
promising candidates.

The algorithm proceeds in three stages:

1. **Seeding** – for several lengths we evaluate a large collection of
   analytic families (uniform, arcsine, beta, power‑law, …) **and**
   a handful of random‑phase seeds.  The best few seeds are kept.

2. **Multimax sub‑gradient descent** – each retained seed is refined by
   mirror‑descent on the non‑smooth objective `max(conv(a,a))`.  The
   gradient is the average of the sub‑gradients over all active convolution
   indices (the “full active set”).  A simple decaying step size
   `η_t = lr0/√t` is used, together with projection onto the probability
   simplex.

3. **Local polishing** – a cheap greedy pairwise‑mass‑transfer walk
   (``swap``) is applied until the time budget expires.

All operations respect the global deadline (`budget_s` seconds).  The
function returns the best sequence found as a plain list of floats,
ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a))/sum(a)^2 (sum forced to 1)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_conv / (s * s)

# Analytic families -------------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _distribution_random_phase(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random (real) sequence whose DFT has (almost) unit magnitude.
    The DC component is fixed to 1 so that the raw sum equals 1.
    Afterwards the vector is shifted to be non‑negative and normalised.
    """
    # rfft length
    m = n // 2 + 1
    H = np.empty(m, dtype=np.complex128)

    # DC term (real) – we set it to 1 to guarantee a non‑zero sum.
    H[0] = 1.0

    if m > 2:
        phases = rng.uniform(0.0, 2.0 * np.pi, size=m - 2)
        H[1:-1] = np.exp(1j * phases)

    if n % 2 == 0:
        # Nyquist frequency must be real; we simply keep it at +1.
        H[-1] = 1.0

    a = np.fft.irfft(H, n=n)
    # Shift to non‑negative.
    a -= a.min()
    # Avoid a completely zero vector (can happen for very tiny n).
    a += 1e-12
    a /= a.sum()
    return a


def _initial_candidates_extended(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Broad pool of seeds for a given length, including random‑phase ones."""
    cand = []

    # Analytic families (same as before)
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))

    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        cand.append(_distribution_beta_sym(n, p))

    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        cand.append(_distribution_beta_asym(n, alpha, beta))

    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        cand.append(_distribution_power_increasing(n, p))

    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        cand.append(_distribution_exponential_increasing(n, lam))

    for block_frac in (0.05, 0.10):
        cand.append(_distribution_two_blocks(n, block_frac))

    # Dirichlet draws
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng, alpha=0.3 + 0.7 * rng.random()))

    cand.append(_distribution_random_sparse(n, rng, density=0.01))

    # ---- NEW: random‑phase seeds ----
    for _ in range(10):          # modest number – cheap to generate
        cand.append(_distribution_random_phase(n, rng))

    return cand

def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient 2·a⋆w where w is a weight vector (length 2n‑1) that sums to 1.
    Implemented via FFT‑based cross‑correlation.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    size = 1 << ((n + w.size - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    grad = 2.0 * conv[start:start + n]
    return grad


# ----------------------------------------------------------------------
def _subgradient_multimax_descent(a: np.ndarray,
                                  deadline: float,
                                  best_seq: np.ndarray,
                                  best_score: float,
                                  lr0: float = 0.7,
                                  max_iter: int = 40000,
                                  eval_interval: int = 500) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the non‑smooth objective.
    The sub‑gradient is the average over all active convolution indices.
    """
    n = a.shape[0]
    t = 0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # uniform weight over the active set (within tolerance)
        mask = conv >= max_val - 1e-14
        cnt = mask.sum()
        if cnt == 0:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = mask.astype(np.float64) / cnt

        grad = _gradient_from_weights(a, w)

        # decaying step size
        eta = lr0 / math.sqrt(t)

        a = a * np.exp(-eta * grad)        # exponentiated‑gradient step
        a = _project_onto_simplex(a)       # keep on probability simplex

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk that keeps a move only if the true objective improves."""
    n = a.shape[0]
    while time.time() < deadline:
        src_idx = np.where(a > 1e-12)[0]
        if src_idx.size == 0:
            break
        i = np.random.choice(src_idx)

        # pick a distinct target index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)

        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.3g} new best {best_score:.6f}")

    return best_seq, best_score

        # Pick a few elite seeds for intensive refinement
        elite = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]

        for seed_seq in elite:
            if time.time() >= deadline:

            seq_out, score_out = _subgradient_multimax_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=1.0,
                max_iter=50000,
                eval_interval=800,
            )
            if score_out < overall_best_score:

                print(f"[{time.time():.1f}] after multimax new best {overall_best_score:.6f}")

    if time.time() < deadline:

        )

    # ------------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    return overall_best_seq.tolist()
