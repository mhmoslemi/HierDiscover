"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  The search proceeds in three stages:

1.  Analytic families – uniform, arcsine and a flexible “edge‑weighted”
    family w[i] ∝ ((i+½)*(n‑i‑½))^(−p).  The optimal exponent `p` is
    found by a cheap golden‑section line‑search for each length `n`.  The
    best few candidates (including mixtures with the uniform vector) are
    retained.

2.  Mirror‑descent on the exact max‑convolution objective.  The vector is
    kept symmetric (a[i]==a[n‑1‑i]) and projected onto the probability
    simplex after every exponentiated‑gradient step.  A longer schedule
    (up to the remaining time budget) is used for the strongest seeds.

3.  A symmetry‑preserving greedy mass‑transfer walk (pairwise swaps) to
    polish the final solution.

All operations respect the 1000 s wall‑clock budget.  The returned
list has sum≈1, so `evaluate_sequence` computes `2·n·max(conv(a,a))`
directly.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT (size = next power of 2)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 2).bit_length())  # smallest power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        f(a) = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+½)*(n‑i‑½))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * uniform + (1.0 - lam) * base
    return mixed / mixed.sum()

def _golden_section_edge(n: int, p_lo: float = 1e-3, p_hi: float = 5.0,
                         max_iter: int = 30) -> tuple[float, np.ndarray, float]:
    """
    Golden‑section search for the exponent p that minimises the objective
    of the edge‑weighted family for a fixed length n.
    Returns (best_score, best_vector, best_p).
    """
    phi = (1 + 5 ** 0.5) / 2
    invphi = 1 / phi
    a, b = p_lo, p_hi
    # initial interior points
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    vc = _objective(_distribution_edge_weighted(n, c))
    vd = _objective(_distribution_edge_weighted(n, d))
    for _ in range(max_iter):
        if vc < vd:
            b, vd = d, vc
            d = c
            c = b - invphi * (b - a)
            vc = _objective(_distribution_edge_weighted(n, c))
        else:
            a, vc = c, vd
            c = d
            d = a + invphi * (b - a)
            vd = _objective(_distribution_edge_weighted(n, d))
        if time.time() >= _search_deadline:
            break
    # pick the better of the final interior points
    if vc < vd:
        best_p = c
    else:
        best_p = d
    best_vec = _distribution_edge_weighted(n, best_p)
    best_score = _objective(best_vec)
    return best_score, best_vec, best_p


# ----------------------------------------------------------------------
def _subgradient_descent_sym(a: np.ndarray,
                             deadline: float,
                             best_seq: np.ndarray,
                             best_score: float,
                             lr0: float = 0.6,
                             eval_interval: int = 150) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror) descent on the exact max‑conv objective.
    The iterate is forced to be symmetric after each step."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # sub‑gradient from the maximal index
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]
        # enforce symmetry of the gradient
        grad = (grad + grad[::-1]) * 0.5

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # keep symmetry and renormalise to the simplex
        a = (a + a[::-1]) * 0.5
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad iter={t} new global best {best_score:.6f}")
    return best_seq, best_score


# ----------------------------------------------------------------------
def _pairwise_swap_sym(a: np.ndarray,
                       deadline: float,
                       best_seq: np.ndarray,
                       best_score: float,
                       max_frac: float = 0.35,
                       max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """Random symmetric mass transfers; keep move only if objective improves."""
    n = a.shape[0]
    left = np.arange(n // 2)            # indices that have distinct mirrors
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        # pick a source index with some mass
        src_candidates = left[a[left] > 1e-14]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # pick a distinct destination index
        j = np.random.choice(left)
        if j == i:
            continue

        # decide how much to move (same amount from both symmetric copies)
        delta = a[i] * np.random.rand() * max_frac
        if delta < 1e-15:
            continue

        new_a = a.copy()
        # remove from i and its mirror
        new_a[i] -= delta
        new_a[n - 1 - i] -= delta
        # add to j and its mirror
        new_a[j] += delta
        new_a[n - 1 - j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Search for a non‑negative step‑function that minimises evaluate_sequence."""
    rng = np.random.default_rng(seed)
    start = time.time()
    # global deadline – a few seconds are kept as safety margin
    global _search_deadline
    _search_deadline = start + budget_s - 2.0

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # --------------------------------------------------------------
    # 1. Analytic families + 1‑D optimisation over the edge exponent
    lengths = [2000, 3000, 4000, 6000, 8000, 12000, 16000]
    lam_grid = np.linspace(0.0, 1.0, 11)      # finer mixing with uniform
    for n in lengths:
        if time.time() >= _search_deadline:
            break
        print(f"[{time.time():.1f}] Scanning length {n}")

        # uniform
        uni = _distribution_uniform(n)
        sc = _objective(uni)
        top_candidates.append((sc, uni.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, uni.copy()
            print(f"[{time.time():.1f}] New global best (uniform) len={n} score={sc:.6f}")

        # arcsine
        arcs = _distribution_arcsine(n)
        sc = _objective(arcs)
        top_candidates.append((sc, arcs.copy()))
        if sc < overall_best_score:
            overall_best_score, overall_best_seq = sc, arcs.copy()
            print(f"[{time.time():.1f}] New global best (arcsine) len={n} score={sc:.6f}")

        # edge‑weighted – golden‑section search for p
        best_sc_edge, best_vec_edge, best_p = _golden_section_edge(n)
        top_candidates.append((best_sc_edge, best_vec_edge.copy()))
        if best_sc_edge < overall_best_score:
            overall_best_score, overall_best_seq = best_sc_edge, best_vec_edge.copy()
            print(
                f"[{time.time():.1f}] New global best (edge) len={n} p={best_p:.4f} score={best_sc_edge:.6f}"
            )
        # mixes of the best edge vector with uniform
        for lam in lam_grid:
            cand = _mix_with_uniform(best_vec_edge, lam)
            sc = _objective(cand)
            top_candidates.append((sc, cand.copy()))
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, cand.copy()
                print(
                    f"[{time.time():.1f}] New global best (edge+U) len={n} lam={lam:.2f} score={sc:.6f}"
                )

        # keep only a modest number of top seeds overall
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # --------------------------------------------------------------
    # 2.  Include a previously discovered construction if present
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score, overall_best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3.  Fallback – should never be needed
    if overall_best_seq is None:
        n = 3000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Pairwise convex combinations of the best seeds (cheap)
    seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:8]]
    for i in range(len(seeds)):
        for j in range(i + 1, len(seeds)):
            a, b = seeds[i], seeds[j]
            for lam in np.linspace(0.0, 1.0, 9):
                cand = lam * a + (1.0 - lam) * b
                # enforce symmetry and normalisation
                cand = (cand + cand[::-1]) * 0.5
                cand /= cand.sum()
                sc = _objective(cand)
                if sc < overall_best_score - 1e-14:
                    overall_best_score, overall_best_seq = sc, cand.copy()
                    print(
                        f"[{time.time():.1f}] convex mix i={i},j={j},lam={lam:.2f} new best {sc:.6f}"
                    )
                top_candidates.append((sc, cand.copy()))
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:30]

    # --------------------------------------------------------------
    # 5.  Mirror‑descent on a few top seeds (most time consuming)
    remaining = _search_deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        print(f"[{time.time():.1f}] Starting subgradient descent on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= _search_deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _subgradient_descent_sym(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.6,
                eval_interval=150,
            )
            if sc_opt < overall_best_score - 1e-14:
                overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
                print(
                    f"[{time.time():.1f}] after subgrad seed {idx} new global best {overall_best_score:.6f}"
                )

    # --------------------------------------------------------------
    # 6.  Final greedy local refinement (symmetric swaps)
    if time.time() < _search_deadline:
        print(f"[{time.time():.1f}] Final symmetric local refinements")
        a_opt, sc_opt = _pairwise_swap_sym(
            overall_best_seq.copy(),
            _search_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            max_frac=0.4,
            max_trials=None,
        )
        if sc_opt < overall_best_score - 1e-14:
            overall_best_score, overall_best_seq = sc_opt, a_opt.copy()
            print(f"[{time.time():.1f}] after final swaps new best {overall_best_score:.6f}")

    # --------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # Return a plain Python list – evaluate_sequence will handle the rest.
    return overall_best_seq.tolist()
