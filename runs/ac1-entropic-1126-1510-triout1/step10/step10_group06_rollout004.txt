"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a))          (a is normalised to sum = 1)

The code follows a multi‑stage stochastic optimisation:

1.  Scan a rich collection of analytic families (Beta, asymmetric Beta,
    power‑decay, arcsine, triangular, exponential, Gaussian, sine‑power)
    for several lengths up to 8000 and keep the best vector.

2.  Refine the current best vector using three optimiser phases
    – exponential‑gradient on a smooth‐max surrogate,
    – projected sub‑gradient descent on the true max‑convolution,
    – Adam on the logits (softmax parametrisation).

3.  Finish each refinement with a very aggressive hill‑climber that moves
    tiny mass from the index with the largest gradient to the index with
    the smallest gradient and occasionally accepts uphill moves (simulated
    annealing).

4.  While time permits, start many random Dirichlet vectors of varying
    lengths and run a shortened version of the refinement pipeline;
    the global best is always updated.

The whole search respects a 1000 s wall‑clock budget and prints progress.
The returned list is ready for ``evaluate_sequence``.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a):
    """Objective 2·n·max(a*a) for a vector already normalised to sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – every function returns a probability vector (sum=1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (q - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_decay(n, p=1.0):
    """Monotonically decreasing power‑law: a_i ∝ (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    vals = (n - i) ** p
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _smooth_weights(conv, alpha):
    """Weights approximating the smooth max of conv (sum to 1)."""
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """Uniform weights over entries that are (almost) maximal."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


# ----------------------------------------------------------------------
def _exponential_gradient_step(a, lr, alpha):
    """One step of EG on the smooth‑max surrogate."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """PGD step on the true max‑convolution sub‑gradient."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    a = np.maximum(a, 1e-14)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, grad):
    """Adam update for logits (softmax parametrisation)."""
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    # Prevent overflow/underflow
    logits = np.clip(logits, -30, 30)
    return logits, m, v


def _hillclimber_move(a, conv, eps=1e-14):
    """Move a tiny chunk of mass from the index with largest gradient to the one with smallest."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    # Larger moves are occasionally useful
    max_delta = min(a[src] * 0.5, 0.05)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    return (a_new, new_conv, True) if improved else (a, conv, False)

def _refine_candidate(a, n, deadline,
                     steps_eg=3000, steps_pg=1500,
                     steps_adam=2000, hill_iters=25000):
    """Full refinement pipeline for a single candidate."""
    best_local = a.copy()
    best_score = _max_convolution_score(best_local)

    # ---------- exponential‑gradient ----------
    a_eg = a.copy()
    a0_alpha, a1_alpha = 0.5, 2000.0
    lr0, lr1 = 0.12, 0.0012
    for it in range(steps_eg):
        if time.time() >= deadline:
            break
        t = it / steps_eg
        alpha = a0_alpha * (a1_alpha / a0_alpha) ** t
        lr = lr0 * (lr1 / lr0) ** t
        a_eg = _exponential_gradient_step(a_eg, lr, alpha)
        if it % 400 == 0 or it == steps_eg - 1:
            sc = _max_convolution_score(a_eg)
            if sc < best_score - 1e-13:
                best_local, best_score = a_eg.copy(), sc
                print(f"[{time.time():.1f}] EG it={it}  best={sc:.6f}")

    # ---------- projected‑gradient ----------
    a_pg = a_eg.copy()
    eta0, eta1 = 0.28, 0.0028
    for it in range(steps_pg):
        if time.time() >= deadline:
            break
        conv = _fft_self_convolve(a_pg)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        eta = eta0 * (eta1 / eta0) ** (it / steps_pg)
        a_pg = _projected_gradient_step(a_pg, eta, w)
        if it % 300 == 0 or it == steps_pg - 1:
            sc = _max_convolution_score(a_pg)
            if sc < best_score - 1e-13:
                best_local, best_score = a_pg.copy(), sc
                print(f"[{time.time():.1f}] PG it={it}  best={sc:.6f}")

    # ---------- Adam on logits ----------
    logits = np.log(a_pg + 1e-14)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    for it in range(1, steps_adam + 1):
        if time.time() >= deadline:
            break
        a_adm = np.exp(logits)
        a_adm /= a_adm.sum()
        conv = _fft_self_convolve(a_adm)
        w = _subgradient_weights(conv, eps_rel=1e-12)
        grad_a = 2.0 * _fft_convolve(a_adm[::-1], w)[n - 1 : n - 1 + n]
        # gradient w.r.t. logits (softmax)
        grad_logits = (grad_a - np.dot(a_adm, grad_a)) * a_adm
        logits, m, v = _adam_step(logits, m, v, it, grad_logits)
        if it % 400 == 0 or it == steps_adam:
            a_adm = np.exp(logits)
            a_adm /= a_adm.sum()
            sc = _max_convolution_score(a_adm)
            if sc < best_score - 1e-13:
                best_local, best_score = a_adm.copy(), sc
                print(f"[{time.time():.1f}] Adam it={it}  best={sc:.6f}")

    # ---------- hill‑climber (with occasional uphill acceptance) ----------
    a_hc = best_local.copy()
    conv = _fft_self_convolve(a_hc)
    temperature = 1e-4  # tiny probability of accepting uphill moves
    for it in range(hill_iters):
        if time.time() >= deadline:
            break
        a_new, conv_new, improved = _hillclimber_move(a_hc, conv, eps=1e-14)
        if improved:
            a_hc, conv = a_new, conv_new
            sc = 2.0 * n * conv.max()
            if sc < best_score - 1e-13:
                best_local, best_score = a_hc.copy(), sc
                if it % 500 == 0:
                    print(f"[{time.time():.1f}] hill it={it}  best={sc:.6f}")
        else:

            if random.random() < temperature:
                # random jitter
                jitter = np.random.randn(n) * 5e-5
                a_hc = np.maximum(a_hc + jitter, 1e-14)
                a_hc /= a_hc.sum()
                conv = _fft_self_convolve(a_hc)
        if it % 5000 == 0 and it > 0:
            # reset temperature slowly
            temperature *= 0.9

    return best_local, best_score
