"""
Improved step‑function optimizer for ``evaluate_sequence``.

The algorithm works in three phases:

1. **Broad generation** – for a set of lengths we create many analytic
   families (uniform, arcsine, Beta(p,q), power‑law, windows, etc.) and a
   few random Dirichlet draws.  The best few (≤10) are kept in an *elite*
   pool.

2. **Continuous optimisation** – each elite vector is refined by
   (a) a long mirror‑descent on a smooth‑max approximation of the
       objective, (b) a projected Adam‑style sub‑gradient descent on the
       true max‑convolution, and (c) a lightweight hill‑climber that moves
       tiny mass from the most offending pair to random locations and
       periodically prunes negligible entries.

3. **Final polishing** – while time permits we perform a simple
   simulated‑annealing style random walk (mass transfers) that accepts
   only improvements.

All vectors are kept on the probability simplex (sum = 1) and
non‑negative, which guarantees scale‑invariance of the score.
The routine respects a user‑specified time budget (default 1000 s) and
returns the best sequence found as a plain ``list[float]``.
"""

# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    rho = rho[-1] if rho.size else -1
    theta = cssv[rho] / (rho + 1) if rho >= 0 else 0.0
    w = np.maximum(v - theta, 0.0)
    return w

# Fast convolution utilities --------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors via real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective value for a probability vector (sum=1):
    2·n·max(a*a)
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a *probability* vector, i.e. sum=1) ---
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[::-1] if n % 2 == 0 else left[:-1][::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the grid (i+0.5)/n."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # use log‑space to avoid under/overflow
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, p: float, q: float) -> np.ndarray:
    """Beta(p,q) on the grid (i+0.5)/n, p,q>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n: int, coeff: float = 0.5) -> np.ndarray:
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n: int, p: float) -> np.ndarray:
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

# Smooth‑max mirror descent (longer schedule) --------------------------
def _smooth_max_mirror_descent(
    a: np.ndarray, deadline: float, total_steps: int = 30000
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (KL‑mirror) descent on a smooth‑max
    approximation of the max‑convolution.
    """
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # α grows from 1 to 1e5, learning rate decays from 0.2 to 1e-5
    alpha0, alpha1 = 1.0, 1e5
    lr0, lr1 = 0.2, 1e-5

    for it in range(total_steps):
        if time.time() >= deadline:
            break
        t = it / total_steps
        alpha = alpha0 * (alpha1 / alpha0) ** t
        lr = lr0 * (lr1 / lr0) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # softmax weights on the convolution values
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the smooth‑max; note the convolution with reversed a
        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        # Exponentiated‑gradient (KL) update
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if it % 2000 == 0 or it == total_steps - 1:
            sc = _max_convolution_score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Mirror it={it}/{total_steps} new best {sc:.6f}")

    return best_a, best_sc

# Projected sub‑gradient (Adam‑style) -----------------------------------
def _projected_subgradient(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    True sub‑gradient descent with Adam momentum and an adaptive back‑tracking
    line search.
    """
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # tolerance for active set (relative)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # sub‑gradient of max‑conv = sum of gradients of active entries
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            a_cand = a - step * v_hat
            a_cand = _project_onto_simplex(a_cand, 1.0)
            a_cand = np.maximum(a_cand, eps)
            a_cand /= a_cand.sum()
            cand_sc = _max_convolution_score(a_cand)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_cand
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter to escape flat plateaus
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        # safety break
        if time.time() + 0.5 >= deadline:
            break

    return best_a, best_sc

# Hill‑climber + support pruning ----------------------------------------
def _hill_climb_prune(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Light‑weight hill‑climber that repeatedly moves a tiny amount of mass
    from the pair (i,j) currently responsible for the maximal convolution
    to a randomly chosen index.  Every ~1500 iterations we try to prune the
    smallest entries.
    """
    n = a.shape[0]
    eps = 1e-12
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- identify the most offending pair --------------------------------
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # ---- move a tiny slice of probability away from the larger of the two --
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            # limit delta to keep enough mass in source and stay numerically stable
            max_delta = min(a[source] * 0.5, 0.02)
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-12:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 200 == 0:
                        print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ---- periodic pruning of tiny entries ---------------------------------
        if it % 1500 == 0:
            for keep_frac in (0.99, 0.97, 0.95):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]          # length may shrink
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ---- occasional jitter -------------------------------------------------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Random small‑mass walk (final polishing) -------------------------------
def _random_walk_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    While time permits we perform tiny random transfers (mass from i→j)
    and keep any improvement.  This is a cheap simulated‑annealing style
    refinement that often squeezes a few extra digits.
    """
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)
    it = 0

    while time.time() < deadline:
        it += 1
        i, j = random.sample(range(n), 2)
        max_delta = min(best_a[i] * 0.4, 0.01)
        delta = max_delta * random.random()
        if delta < eps:
            continue
        a2 = best_a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        sc = _max_convolution_score(a2)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a2
            # print only occasionally to reduce output clutter
            if it % 500 == 0:
                print(f"[{time.time():.1f}] Random walk it={it} new best {sc:.6f}")

    return best_a, best_sc

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimising ``evaluate_sequence``.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # leave a safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None
    elite = []          # (score, vector) sorted ascending, size ≤ 10

    # ------------------------------------------------------------------
    # 1. Broad exploration over a range of lengths
    lengths = [2000, 3500, 6000, 10000, 18000, 30000]   # diverse spread
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")

            # maintain elite list (size ≤10)
            if len(elite) < 10:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Insert any previously discovered sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # may be undefined
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 10:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (should never be needed)
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(6, len(elite))          # a handful of diverse starts
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {idx+1}/{restarts} (len={n0}) score {elite[idx][0]:.6f}")

        # --- smooth‑max mirror descent (more iterations) -----------------
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline, total_steps=35000)

        # --- projected sub‑gradient polishing ---------------------------
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # --- hill‑climber + pruning -----------------------------------
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        # --- final random walk refinement --------------------------------
        a_mid, sc_mid = _random_walk_refine(a_mid, deadline)

        # --- record any improvement ------------------------------------
        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity – ensure non‑negative, normalised, finite entries ----
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
