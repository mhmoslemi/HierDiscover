Algorithm
---------

   edge‑heavy power, power‑law, Dirichlet draws).  All vectors are

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast upper bound: 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals /= vals.sum()
    return vals


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _random_symmetric_dirichlet(n: int, alpha: float) -> np.ndarray:
    half = (n + 1) // 2
    half_vec = np.random.dirichlet(np.full(half, alpha))
    if n % 2 == 0:
        full = np.concatenate([half_vec, half_vec[::-1]])
    else:
        full = np.concatenate([half_vec, half_vec[-2::-1]])
    full /= full.sum()
    return full

    for it in range(max_sa):
        if time.time() >= deadline:
            break
        # pick a source proportional to its mass (more mass ⇒ higher chance)
        src = np.random.choice(n, p=cur)
        tgt = random.randint(0, n - 1)
        if tgt == src:
            continue
        delta = min(cur[src] * 0.25, 0.015) * random.random()
        if delta < eps:
            continue
        cand = cur.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.maximum(cand, eps)
        cand /= cand.sum()
        sc = _max_convolution_score(cand)
        dE = sc - cur_score_fast
        if dE < 0 or random.random() < math.exp(-dE / T):
            cur = cand
            cur_score_fast = sc
            if sc < best_score_fast - 1e-12:
                best_seq = cand.copy()
                best_score_fast = sc
                print(f"[{time.time():.1f}] SA it={it}  fast_score={sc:.6f}")
        T = max(T * decay, T_min)

    # ------------------------------------------------------------------
    # 4. Final cheap hill‑climber (greedy mass transfers)
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1
        # locate index of current maximum
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # find the pair (i,j) that contributes most to this max
        best_i, best_j = i_low, k - i_low
        max_prod = a[best_i] * a[best_j]
        for i in range(i_low, i_high + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue
        max_delta = min(a[source] * 0.5, 0.01)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_snapshot = a.copy()
        a[source] -= delta
        a[target] += delta
        a = np.maximum(a, eps)
        a /= a.sum()
        conv = _fft_self_convolve(a)
        new_M = conv.max()
        if new_M < M - 1e-12:
            M = new_M
            fast_sc = 2.0 * n * M
            if fast_sc < best_score_fast - 1e-12:
                best_seq = a.copy()
                best_score_fast = fast_sc
                print(f"[{time.time():.1f}] hill it={hill_iters}  fast_score={fast_sc:.6f}")
        else:
            a = a_snapshot
            conv = _fft_self_convolve(a)

        if hill_iters % 5000 == 0:
            jitter = np.random.randn(n) * 5e-6
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator (optional but cheap)
    final_score = float("inf")
    try:
        final_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        pass
    print(f"Final best fast score ≈ {best_score_fast:.6f} "
          f"(official evaluate_sequence ≈ {final_score:.6f}, length={best_seq.shape[0]})")
    return best_seq.tolist()
