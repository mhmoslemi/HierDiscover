def _lp_improve(a, deadline, max_iter=15, t0=0.5, epsilon=0.05):
    best = a.copy()
    best_score = _objective(a)
    iteration = 0
    while time.time() < deadline and iteration < max_iter:
        iteration += 1
        conv = _fft_self_convolve(a)
        M = conv.max()
        # select active constraints
        thresh = M * (1 - epsilon)
        active_inds = np.where(conv >= thresh)[0]
        # Build sparse matrix
        rows, cols, vals = [], [], []
        for row, k in enumerate(active_inds):
            i_start = max(0, k - (len(a)-1))
            i_end = min(len(a)-1, k)
            length = i_end - i_start + 1
            rows.extend([row] * length)
            cols.extend(range(i_start, i_end+1))
            vals.extend(a[i_start:i_end+1])
        A_ub = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(len(active_inds), len(a)))
        b_ub = np.full(len(active_inds), M)
        res = scipy.optimize.linprog(c=-np.ones(len(a)), A_ub=A_ub, b_ub=b_ub,
                                    bounds=[(0, None)]*len(a), method='highs',
                                    options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            break
        x = res.x
        if x.sum() == 0:
            break
        g = x / x.sum()
        # mixing factor: t = min(t0, (best_score - new_score)/best_score?) not needed.
        t = t0 * (0.9**(iteration-1))  # decreasing t
        a = (1 - t) * a + t * g
        curr_score = _objective(a)
        if curr_score < best_score - 1e-12:
            best_score = curr_score

            # if no improvement, maybe break

    return best, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes evaluate_sequence.

Algorithm
---------
1.  Scan a rich pool of analytic families (uniform, arcsine, symmetric beta,
    edge‑weighted U‑shape, mixtures with uniform, random Dirichlet, sparse)
    for several lengths and keep the top few candidates (by the exact objective).

2.  For each top candidate run a few iterations of an *active‑set linear program*:
    -  Let a be the current distribution (∑a=1).
    -  Compute its self‑convolution c = a*a and M = max(c).
    -  Keep only those convolution indices where c ≥ (1‑ε)·M (ε≈0.02).
    -  Build the constraint matrix  C where (C x)_k = (a * x)_k for the selected
       indices k.  Solve the LP
            max   1ᵀx
            s.t.  C x ≤ M ,  x ≥ 0 .
       The solution x is the most massive distribution that still respects the
       current maximal convolution bound.
    -  Normalise x to a probability vector g = x/∑x and mix:
            a ← (1‑t)·a + t·g    (t starts at 0.5 and decays).
       If the objective improves, keep the new a.

    The LP is solved with SciPy’s HiGHS‐based `linprog` using a sparse matrix,
    so even for n≈3000 the step is fast.

3.  Refine the best vectors with exponentiated‑gradient (mirror) sub‑gradient
    descent on the true max‑convolution objective (as in the reference code).

4.  Finish with a cheap greedy pairwise mass‑transfer walk that moves a little
    mass from one coordinate to another whenever it strictly reduces the
    objective.

All vectors are kept normalised (∑a = 1).  The objective
    O(a) = 2·n·max(conv(a,a))
is invariant under scaling, matching `evaluate_sequence` when the sum is 1.

The whole routine respects the 1000 s budget and prints progress.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using real FFT (fast and accurate)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    # Numerical noise can make max slightly negative; clamp.
    max_conv = np.maximum(conv.max(), 0.0)
    return 2.0 * n * max_conv

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

# Sub‑gradient descent (mirror / exponentiated gradient) -------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.5,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # sub‑gradient from the maximal index
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] subgrad it={t} new global best {best_score:.6f}")
        if t >= 200000:
            break
    return best_seq, best_score

# Greedy pairwise mass transfer ---------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        # choose a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

# Active‑set linear‑program improvement -------------------------------
def _lp_improve(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 12,
    eps: float = 0.02,
    t0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Given a distribution a (∑a=1), perform up to max_iter LP steps.
    Each step solves:
        max 1ᵀx
        s.t. (a * x)_k ≤ M   for all k with conv_a(k) ≥ (1‑eps)·M,
             x ≥ 0.
    Returns the best vector found and its objective.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _objective(a)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        M = conv.max()
        # active constraints: keep top values
        thresh = M * (1.0 - eps)
        active_idx = np.where(conv >= thresh)[0]
        if active_idx.size == 0:
            active_idx = np.array([conv.argmax()], dtype=int)

        # Build sparse matrix C where (C x)_r = (a * x)_{active_idx[r]}
        rows, cols, data = [], [], []
        for row, k in enumerate(active_idx):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            length = i_end - i_start + 1
            rows.extend([row] * length)
            cols.extend(range(i_start, i_end + 1))
            data.extend(a[i_start:i_end + 1])
        C = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
        b = np.full(len(active_idx), M, dtype=np.float64)

        # Solve LP: max sum x  ⇔  min -sum x
        res = scipy.optimize.linprog(
            c=-np.ones(n),
            A_ub=C,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # LP failed – abort further LP steps
            print(f"[{time.time():.1f}] LP iteration {it} failed ({res.message})")
            break

        x = res.x
        s = x.sum()
        if s < 1e-12:
            break
        g = x / s

        # mixing factor decays geometrically
        t = t0 * (0.8 ** (it - 1))
        a = (1.0 - t) * a + t * g

        cur_score = _objective(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.time():.1f}] LP iter {it} new best {best_score:.6f}")

        # stop early if we already beat the target
        if best_score < 1.503:
            break

    return best, best_score
