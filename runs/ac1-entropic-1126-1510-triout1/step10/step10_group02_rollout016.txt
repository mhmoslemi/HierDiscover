def _smooth_max_adam(a_init, deadline, max_iters=20000, alpha_start=0.1, alpha_end=1e6, lr_start=0.5, lr_end=1e-4):
    a = a_init.copy()
    n = a.shape[0]
    x = np.log(np.maximum(a, 1e-12))  # log variables
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    best_a = a.copy()
    best_sc = _max_convolution_score(a)
    start_iter = 0
    while time.time() < deadline and start_iter < max_iters:
        # compute a from x
        a = np.exp(x - np.max(x))  # for numerical stability
        a /= a.sum()
        # conv:
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # compute soft weights:
        # For numeric stability, compute exponent of α*(conv-maxc)
        # When α large, only max index gets weight ~1.
        # So implement:
        # w = np.exp(alpha * (conv - maxc))
        # but may underflow.
        alpha = alpha_start * (alpha_end / alpha_start) ** (start_iter / max_iters)
        # compute exponent:
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # compute grad wrt a:
        grad_a = 2 * _fft_convolve(a[::-1], w)[n-1: n-1+n]
        # gradient w.r.t x (logit):
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)
        # Adam update:
        start_iter += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** start_iter)
        v_hat = v / (1 - beta2 ** start_iter)
        lr = lr_start * (lr_end / lr_start) ** (start_iter / max_iters)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        if start_iter % 200 == 0:

            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] SmoothAdam iter={start_iter} new best {sc:.6f}")
    return best_a, best_sc

def _projected_subgradient_adam(a, deadline, max_iters=20000):
    n = a.shape[0]
    a = np.maximum(a, 0)
    a /= a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_sc = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        tol = 1e-12 * (1 + abs(maxc))
        # active indices:
        active = np.where(conv >= maxc - tol)[0]
        # weight vector w = zeros, set w[active]=1
        # gradient = 2*conv_rev[n-1:n-1+n]
        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2 * _fft_convolve(a[::-1], w)[n-1:n-1+n]
        # Adam update:
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        lr = 0.1 / np.sqrt(it) # simple decaying lr
        # compute candidate a_new:
        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex:
        a_candidate = _project_onto_simplex(a_candidate, 1.0)
        # enforce small epsilon:
        eps_small = 1e-12
        a_candidate = np.maximum(a_candidate, eps_small)
        a_candidate /= a_candidate.sum()
        # Evaluate:
        sc = _max_convolution_score(a_candidate)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_candidate.copy()
            print(f"[{time.time():.1f}] SubGradAdam it={it} new best {sc:.6f}")
        a = a_candidate
        # optional jitter: every so often add small random noise
        if it % 2500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps_small)
            a /= a.sum()
        # time check break:
        if time.time() + 0.5 > deadline:
            break
    return best_a, best_sc

def _hill_climb_prune(a, deadline):
    n = a.shape[0]
    eps_small = 1e-12
    a = np.maximum(a, eps_small)
    a /= a.sum()
    best_a = a.copy()
    best_sc = _max_convolution_score(a)
    best_M = best_sc / (2.0 * n)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute conv
        conv = _fft_self_convolve(a)
        M = conv.max()
        # find index k of max
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # find (i,j) maximizing product a_i * a_j with i+j=k
        # we can sample a few points for speed:
        best_pair = None
        max_prod = -1.0
        # iterate over i in range(i_start,i_end+1):
        for i in range(i_start, i_end+1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_pair = (i, j)
        i, j = best_pair
        source = i if a[i] >= a[j] else j
        # try several random targets:
        for _ in range(3):
            target = random.randint(0, n-1)
            if target == source:
                continue
            max_delta = min(a[source] * 0.5, 0.02)  # maybe adapt
            delta = max_delta * random.random()
            if delta <= eps_small:
                continue
            a2 = a.copy()
            a2[source] -= delta
            a2[target] += delta
            a2 = np.maximum(a2, eps_small)
            a2 /= a2.sum()
            new_M = _fft_self_convolve(a2).max()
            if new_M < M - 1e-12:
                a = a2
                M = new_M
                if 2 * n * M < best_sc - 1e-12:
                    best_sc = 2 * n * M
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")
                break  # early accept
        # periodic pruning
        if it % 1500 == 0:
            # prune entries less than threshold relative to max.
            threshold = max(a) * 1e-4
            if threshold > 0:
                mask = a >= threshold
                if mask.sum() >= 2 and mask.sum() < n:
                    a = a * mask
                    a /= a.sum()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = _max_convolution_score(a)
                    print(f"[{time.time():.1f}] Prune it={it} new length {n} score {best_sc:.6f}")
        # jitter
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps_small)
            a /= a.sum()
            # recompute best if improved
            sc = _max_convolution_score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Jitter it={it} new best {sc:.6f}")
        if time.time() + 0.5 > deadline:
            break
    return best_a, best_sc

"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  The algorithm proceeds in five phases:

1. **Broad exploration** – generate a diverse set of analytic families
   (uniform, arcsine, triangular, exponential, symmetric/asymmetric
   Beta, sine‑power, cosine‑window, Gaussian, power‑law, two‑block,
   random Dirichlet).  Keep the best few candidates (low score).

2. **Smooth‑max Adam descent** – minimise a smooth approximation of the
   max‑convolution using the log‑softmax parametrisation
   `a = softmax(x)`.  The temperature (α) is increased from 1 to 1e6,
   while Adam’s learning‑rate decays.  This strongly reduces the
   worst convolution coefficient.

3. **Projected sub‑gradient polishing** – after the smoothing stage we
   switch to the true nonsmooth objective.  The active constraints are
   identified and Adam‑style updates on the simplex are performed
   with a back‑tracking line‑search.

4. **Support‑pruning & hill‑climbing** – tiny entries are repeatedly
   dropped (renormalising each time).  A cheap hill‑climber moves a
   small amount of mass from the pair that currently attains the
   maximum convolution towards random locations, accepting any
   improvement.

5. **Final sanity** – enforce non‑negativity, normalise and return the
   best sequence found before the 1000 s budget expires.

The whole search respects the time budget (a 2 s safety margin) and
prints intermediate scores so the progress can be inspected.
"""

# Helper utilities -------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective 2·n·max(a*a) for a vector already normalised to sum = 1.
    This is exactly the value that evaluate_sequence would return.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families – each returns a vector with sum = 1 ----------------
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n, block_ratio=0.3):
    """Two uniform blocks separated by a gap."""
    block_len = int(block_ratio * n)
    gap_len = n - 2 * block_len
    if block_len == 0:
        return _distribution_uniform(n)
    vec = np.zeros(n, dtype=np.float64)
    vec[:block_len] = 1.0 / block_len
    vec[-block_len:] = 1.0 / block_len
    vec /= vec.sum()
    return vec

# Local optimisation routines --------------------------------------------
def _smooth_max_adam(a_init, deadline,
                     max_iters=15000,
                     alpha_start=0.1,
                     alpha_end=1e6,
                     lr_start=0.5,
                     lr_end=1e-4):
    """Adam on the smooth‑max approximation of max‑convolution."""
    a = a_init.copy()
    n = a.shape[0]
    eps = 1e-12

    # work in log‑space to guarantee positivity
    x = np.log(np.maximum(a, eps))
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    epsilon_adam = 1e-8

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        # re‑normalise via softmax
        x_shift = x - np.max(x)
        a = np.exp(x_shift)
        a /= a.sum()

        conv = _fft_self_convolve(a)
        maxc = conv.max()

        # smooth max weights
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max_iters)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # convert to gradient w.r.t. x (log‑softmax)
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # Adam update on x
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr_start * (lr_end / lr_start) ** (it / max_iters)

        x -= lr * m_hat / (np.sqrt(v_hat) + epsilon_adam)

        if it % 500 == 0:
            sc = _max_convolution_score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] SmoothAdam it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient_adam(a_init, deadline,
                                max_iters=20000,
                                lr_init=0.2):
    """Adam on the true (nonsmooth) max‑convolution objective."""
    a = a_init.copy()
    n = a.shape[0]
    eps = 1e-12

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    epsilon_adam = 1e-8

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        maxc = conv.max()
        tol = 1e-12 * (abs(maxc) + 1.0)
        active = np.where(conv >= maxc - tol)[0]

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam step on the simplex (we will project back)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr_init / math.sqrt(it + 1)

        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + epsilon_adam)
        a_candidate = _project_onto_simplex(a_candidate, 1.0)
        a_candidate = np.maximum(a_candidate, eps)
        a_candidate /= a_candidate.sum()

        sc = _max_convolution_score(a_candidate)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_candidate.copy()
            print(f"[{time.time():.1f}] SubGradAdam it={it} new best {sc:.6f}")

        a = a_candidate

        # occasional jitter to avoid plateaus
        if it % 2500 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a_init, deadline):
    """Mass‑transfer hill‑climber with periodic pruning of tiny entries."""
    a = a_init.copy()
    n = a.shape[0]
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)
    it = 0

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        M = conv.max()
        k = int(conv.argmax())

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) giving the largest product a_i a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j

        # try a few random targets and accept the first improvement
        for _ in range(3):
            target = random.randint(0, n - 1)
            if target == source:
                continue
            max_delta = min(a[source] * 0.5, 0.02)
            delta = max_delta * random.random()
            if delta <= eps:
                continue
            a2 = a.copy()
            a2[source] -= delta
            a2[target] += delta
            a2 = np.maximum(a2, eps)
            a2 /= a2.sum()
            new_M = _fft_self_convolve(a2).max()
            if new_M < M - 1e-12:
                a = a2
                M = new_M
                if 2.0 * n * M < best_sc - 1e-12:
                    best_sc = 2.0 * n * M
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")
                break

        # ----- periodic pruning -----
        if it % 1500 == 0:
            # keep the largest entries (≥ 0.1 % of the max entry)
            threshold = max(a) * 1e-4
            keep = a >= threshold
            if keep.sum() >= 2 and keep.sum() < n:
                a = a * keep
                a /= a.sum()
                n = a.shape[0]
                best_a = a.copy()
                best_sc = _max_convolution_score(best_a)
                print(f"[{time.time():.1f}] Prune it={it} new length {n} score {best_sc:.6f}")

        # ----- occasional jitter -----
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            sc = _max_convolution_score(a)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Jitter it={it} new best {sc:.6f}")

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc

# Candidate generator ----------------------------------------------------
def _generate_candidates_for_length(n):
    """Return a list of candidate vectors (sum = 1) for a given length."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta
    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta (a few random pairs)
    for (p, q) in ((0.5, 2.0), (2.0, 0.5), (0.8, 1.2), (1.2, 0.8)):
        cand.append(_distribution_beta_asym(n, p, q))

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (monotone decreasing) and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])

    # Two‑block uniform
    for ratio in (0.2, 0.3, 0.4):
        cand.append(_distribution_two_blocks(n, ratio))

    # Sparse random start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Random Dirichlet draws (different concentration)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Random shuffles of a few base families (add asymmetry)
    for base in (cand[0], cand[1], cand[4]):
        shuf = base.copy()
        np.random.shuffle(shuf)
        cand.append(shuf)

    return cand

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function minimizing `evaluate_sequence`.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None
    elite = []  # (score, vector)

    # ------------------------------------------------------------------
    # 1. Broad exploration – keep top‑8 candidates overall
    lengths = [2000, 3500, 5000, 8000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, 1e-12)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                best_n = n
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")
            # store elite
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2. Incorporate a previously found construction if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 1e-12)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                best_n = prev.size
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (unlikely to be needed)
    if best_seq is None:
        best_n = 4000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Restart {idx + 1}/{restarts} start score {elite[idx][0]:.6f}")

        # Smooth‑max Adam descent
        a_mid, sc_mid = _smooth_max_adam(a0, deadline)

        # Projected sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient_adam(a_mid, deadline)

        # Final hill‑climber & pruning
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            best_n = a_mid.shape[0]
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity check / normalisation
    best_seq = np.maximum(best_seq, 1e-12)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
