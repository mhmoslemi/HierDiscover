"""
Simple yet aggressive search for a low‑score step‑function.

The algorithm proceeds in three phases:

1. **Broad analytic sweep** – a few well‑known families (arcsine,
   beta, edge‑power, exponential, mixtures, Dirichlet draws) are
   sampled on several lengths (powers of two up to ≈20000) with a
   dense grid of parameters (e.g. β∈[0.01,2] in steps of 0.01,
   edge‑power s∈[0.10,0.65] in steps of 0.05, offsets for the arcsine,
   etc.).  The best ≈10 candidates (according to the surrogate
   2·n·max(conv)) are kept.

2. **Heavy local optimisation** – each seed is refined by
   *projected Adam* and *exponentiated‑gradient* descent with a very
   generous iteration budget (the loops stop automatically when the
   global time limit is reached).  The best point seen during the
   descent is retained.

3. **Polishing** – a Frank‑Wolfe‑style mass‑transfer step with a
   small line‑search (multiple fractions of the maximal transferable
   mass) and occasional random jitter is applied to the current best
   solution until the deadline expires.

All vectors are kept on the probability simplex (sum = 1, all
components ≥0); the final list is returned as ordinary Python floats.
`evaluate_sequence` will later rescale and validate the answer.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate: 2·n·max(conv(a,a)) for Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families (all return vectors that already sum to 1)

def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _arcsine_offset_distribution(n: int, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5 + offset) * (n - i - 0.5 + offset))
    vals[np.isnan(vals)] = 0.0
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i+1))^{‑s}, i=1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    total = x.sum()
    if total == 0:
        return np.full_like(x, 1.0 / n)
    return x / total


def _random_block_distribution(n: int, blocks: int = 2,
                              min_frac: float = 0.01,
                              max_frac: float = 0.25) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    mass_per_block = 1.0 / blocks
    for _ in range(blocks):
        blk_len = max(1, int(np.random.uniform(min_frac, max_frac) * n))
        start = np.random.randint(0, n - blk_len + 1)
        a[start:start + blk_len] = mass_per_block / blk_len
    a /= a.sum()
    return a


def _mixture_beta_distribution(n: int, params: np.ndarray) -> np.ndarray:
    """params = [w, a1, b1, a2, b2] where w∈[0,1] and a*,b*>0."""
    w, a1, b1, a2, b2 = params
    comp1 = _beta_asymmetric_distribution(n, a1, b1)
    comp2 = _beta_asymmetric_distribution(n, a2, b2)
    mix = w * comp1 + (1.0 - w) * comp2
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Runs for at most ``budget_s`` seconds and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # -------------------- choose lengths ---------------------------------
    # Prefer powers of two (FFT friendly).  Keep them modest so that many
    # optimisation steps are affordable.
    try:
        base_n = len(height_sequence_1)  # type: ignore[name-defined]
    except Exception:
        base_n = 4000
    # Ensure the base is within the allowed range.
    base_n = max(1500, min(base_n, 25000))

    lengths = sorted(set([
        2048, 4096, 8192, 16384,
        max(1500, int(base_n * 0.8)),
        max(1500, int(base_n)),
        min(25000, int(base_n * 1.2))
    ]))

    print(f"[[{time.time():.1f}]] using lengths {lengths}")

    # -------------------- candidate generation ---------------------------
    top_k = 12                      # keep this many seeds for refinement
    top_candidates: list[tuple[float, np.ndarray]] = []  # (score, vector)

    def _maybe_add(vec: np.ndarray):
        """Normalize and possibly insert a vector into the top‑k list."""
        if vec.ndim != 1:
            return
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s <= 0:
            return
        vec = vec / s
        score = _max_convolution_score(vec)
        if len(top_candidates) < top_k:
            top_candidates.append((score, vec))
            top_candidates.sort(key=lambda x: x[0])
        elif score < top_candidates[-1][0]:
            top_candidates[-1] = (score, vec)
            top_candidates.sort(key=lambda x: x[0])

    # parameter grids (slightly richer than the baseline)
    beta_ps = np.arange(0.01, 2.01, 0.01)                 # symmetric beta
    edge_ps = np.arange(0.10, 0.66, 0.05)                 # edge‑power exponents
    lam_vals = [0.0002, 0.0005, 0.001, 0.002, 0.003,
                0.005, 0.008, 0.01, 0.02, 0.05]
    dirichlet_alphas = [0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0]
    offsets = [0.0, 0.5, 1.0, 2.0, 3.0, 5.0]               # for arcsine offset
    mix_params = []                                        # cache for random mixtures
    for _ in range(30):
        w = np.random.rand()
        a1 = 10 ** np.random.uniform(-2, 1)   # between 0.01 and 10
        b1 = 10 ** np.random.uniform(-2, 1)
        a2 = 10 ** np.random.uniform(-2, 1)
        b2 = 10 ** np.random.uniform(-2, 1)
        mix_params.append(np.array([w, a1, b1, a2, b2]))

    # iterate over lengths and families
    for n in lengths:
        if time.time() >= deadline:
            break

        # deterministic families -------------------------------------------------
        _maybe_add(_arcsine_distribution(n))
        for off in offsets:
            _maybe_add(_arcsine_offset_distribution(n, off))

        _maybe_add(_triangular_distribution(n))

        for lam in lam_vals:
            _maybe_add(_exponential_distribution(n, lam=lam))

        for p in beta_ps:
            _maybe_add(_beta_symmetric_distribution(n, p))

        # a few asymmetric beta combos (hand‑picked + random)
        asym_grid = [(0.2, 0.8), (0.3, 0.7), (0.5, 0.5), (0.8, 0.2),
                     (0.1, 1.0), (1.0, 0.1), (0.4, 0.8), (0.8, 0.4)]
        for a_alpha, a_beta in asym_grid:
            _maybe_add(_beta_asymmetric_distribution(n, a_alpha, a_beta))

        for s in edge_ps:
            _maybe_add(_edge_power_distribution(n, s))

        # power‑law (monotone) – very coarse grid
        for p in [0.3, 0.5, 0.8, 1.0, 1.5]:
            _maybe_add(_power_law_distribution(n, p))

        # random Dirichlet draws
        for alpha in dirichlet_alphas:
            _maybe_add(_random_dirichlet_alpha(n, alpha))

        # random block constructions (2‑4 blocks)
        for blocks in (2, 3, 4):
            for _ in range(2):
                _maybe_add(_random_block_distribution(n, blocks=blocks))

        # random mixtures of two beta distributions
        for params in mix_params:
            _maybe_add(_mixture_beta_distribution(n, params))

    # ----------------------------------------------------- previous best -----
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            _maybe_add(prev)
            for _ in range(6):
                jitter = np.random.randn(prev.size) * 5e-5
                noisy = _project_onto_simplex(prev + jitter, s=1.0)
                _maybe_add(noisy)
            print(f"[{time.time():.1f}] added previous best and jittered variants")
    except Exception:
        pass

    # ----------------------------------------------------- refinement ----------
    print(f"Collected {len(top_candidates)} top seeds (keep {top_k})")
    # ensure we have at least one seed
    if not top_candidates:
        # fall back to uniform
        uniform = np.full(2000, 1.0 / 2000)
        top_candidates.append((_max_convolution_score(uniform), uniform))

    best_score, best_seq = top_candidates[0][0], top_candidates[0][1].copy()
    print(f"Initial best surrogate = {best_score:.6f}")

    # ------------------------------------------------------------------
    # helper: Adam on the simplex (projected)
    def _adam_descent(a0: np.ndarray, deadline_inner: float,
                      step0: float = 0.30,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      max_iters: int = 3_000_000):
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        best_a = a.copy()
        best_sc = _max_convolution_score(a)
        t = 0
        while time.time() < deadline_inner and t < max_iters:
            t += 1
            grad, _, _ = _subgrad(a)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            eta = step0 / (np.sqrt(v_hat) + 1e-8)
            a = a - eta * m_hat
            a = _project_onto_simplex(a, s=1.0)

            sc = _max_convolution_score(a)
            if sc < best_sc - 1e-15:
                best_sc, best_a = sc, a.copy()
        return best_a, best_sc

    # ------------------------------------------------------------------
    # helper: exponentiated gradient descent (KL‑mirror)
    def _exp_gradient_descent(a0: np.ndarray, deadline_inner: float,
                              eta0: float = 0.9,
                              max_iters: int = 2_000_000):
        a = a0.copy()
        best_a = a.copy()
        best_sc = _max_convolution_score(a)
        it = 0
        while time.time() < deadline_inner and it < max_iters:
            it += 1
            grad, _, _ = _subgrad(a)
            eta = eta0 / np.sqrt(it)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s == 0:
                a = np.full_like(a, 1.0 / a.size)
            else:
                a = a / s
            sc = _max_convolution_score(a)
            if sc < best_sc - 1e-15:
                best_sc, best_a = sc, a.copy()
        return best_a, best_sc

    # ------------------------------------------------------------------
    # refinement of each seed
    for idx, (seed_sc, seed_vec) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        n = seed_vec.shape[0]
        # allocate a generous portion of the remaining budget per seed
        per_seed_deadline = time.time() + (deadline - time.time()) * 0.45
        seq_adam, sc_adam = _adam_descent(seed_vec, per_seed_deadline,
                                          max_iters=2_500_000)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} Adam → {best_score:.6f}")

        seq_eg, sc_eg = _exp_gradient_descent(seed_vec, per_seed_deadline,
                                              max_iters=2_000_000)
        if sc_eg < best_score - 1e-14:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{len(top_candidates)} ExpGrad → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Frank‑Wolfe style polishing with a richer set of transfer fractions
    a = best_seq.copy()
    cur_score = best_score
    fw_iter = 0
    fractions = (1.0, 0.9, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05, 0.025, 0.0125)

    while time.time() < deadline and fw_iter < 300_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break
        improved = False
        for f in fractions:
            delta = max_transfer * f
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter={fw_iter} new best {best_score:.6f}")

        # occasional random jitter to shake out tiny plateaus
        if fw_iter % 7500 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 4e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Final simulated‑annealing style jitter (longer schedule)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.035
    temp_end = 0.0012
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 350_000:
        anneal_iter += 1
        # linear schedule based on remaining fraction of the budget
        fraction = max(0.0, (deadline - time.time()) / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_sc = _max_convolution_score(a_new)

        if new_sc < cur_score - 1e-15:
            a = a_new
            cur_score = new_sc
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter={anneal_iter} new best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_sc - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_sc

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # convert to plain Python list (the external evaluate_sequence will handle scaling)
    return best_seq.tolist()
