"""
Propose a very low score step function.

The idea is simple but effective:
1️⃣  Generate a large pool of analytic candidates for several
    sequence lengths (up to the allowed 25 000).  The pool contains
    many variants of the “edge‑power’’ family (w[i] ∝ (i·(n‑i+1))⁻ˢ)
    because theory predicts that the optimum is close to a power‑law on
    the edges (the classic arcsine distribution corresponds to s=0.5).
    A fine grid around 0.5 together with a few other classic families
    (beta, sin‑power, exponential, power‑law, random Dirichlet) is
    explored.

2️⃣  All candidates are evaluated with the cheap surrogate
    ``_max_convolution_score`` (exact for Σa=1).  The best 10 seeds are
    kept.

3️⃣  Each seed is refined by a *projected sub‑gradient method* on the
    simplex.  The step size follows the classic  C/√t schedule,
    which is well suited for the non‑smooth max‑convolution objective.
    The routine runs until the time budget is exhausted (or a fixed
    iteration budget is hit) and constantly remembers the best point.

4️⃣  As a final polish a short simulated‑annealing pass performs tiny
    mass‑transfer moves; this can escape very shallow plateaus.

The whole search respects the 1000 s deadline and returns a plain
Python list of non‑negative floats – `evaluate_sequence` will take
care of the final scaling/validation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact objective when Σa = 1 : 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Sub‑gradient of max(conv(a,a)) under Σa = 1.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0

    # gradient of the (non‑smooth) max‑convolution:
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j terms (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    """w_i ∝ (i·(n‑i+1))^{-s}, i = 1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_asymmetric_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (alpha - 1) * np.log(x) + (beta - 1) * np.log(1.0 - x)
    logw = np.where(np.isfinite(logw), logw, -np.inf)
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _sin_power_distribution(n: int, q: float) -> np.ndarray:
    """w_i ∝ sin(π·(i+0.5)/n)^q   (mass concentrated in the centre)."""
    i = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (i + 0.5) / n) ** q
    if vals.sum() == 0:
        vals = np.full_like(vals, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def _simple_subgrad_descent(a0: np.ndarray, deadline: float,
                           step0: float = 0.8,
                           max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Plain projected sub‑gradient descent with step ∝ 1/√t.
    Returns the best point seen and its surrogate score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(t)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur = _max_convolution_score(a)
        if cur < best_score:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _simulated_annealing_refine(a0: np.ndarray, deadline: float,
                               start_temp: float = 0.04,
                               end_temp: float = 0.001,
                               max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Very light “mass‑transfer’’ annealing: move a random δ from i→j.
    Accepts improvements always; worsening moves with Metropolis rule.
    """
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best_a = a.copy()
    best_score = cur_score
    it = 0
    total_time = deadline - time.time()
    while time.time() < deadline and it < max_iters:
        it += 1
        # linear cooling
        fraction = max(0.0, (deadline - time.time()) / total_time)
        temp = end_temp + (start_temp - end_temp) * fraction

        i = np.random.randint(a.size)
        j = np.random.randint(a.size - 1)
        if j >= i:
            j += 1

        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * temp

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        new_score = _max_convolution_score(a_new)

        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score
    return best_a, best_score

        fin_seq, fin_sc = _simulated_annealing_refine(best_seq, anneal_deadline,
                                                      start_temp=0.03, end_temp=0.001,
                                                      max_iters=250_000)
        if fin_sc < best_score - 1e-12:
            best_score, best_seq = fin_sc, fin_seq.copy()
            print(f"[{time.time():.1f}] annealing gave new best {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Finish: best surrogate score {best_score:.6f} (length={len(best_seq)})")
    # ``evaluate_sequence`` will handle the final scaling, so we return the raw simplex.
    return best_seq.tolist()
