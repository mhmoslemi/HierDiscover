def _simulated_annealing(a, deadline, max_iter=2000):

   n = a.shape[0]
   conv = _fft_self_convolve(a)
   M = conv.max()
   best_a = a.copy()
   best_M = M
   it = 0
   while time.time() < deadline and it < max_iter:
       it+=1
       # propose transfer
       # Choose source weighted by a
       source = np.random.choice(n, p=a/a.sum())
       target = np.random.randint(0, n)
       if target == source: continue
       # delta = random fraction of a[source] bounded
       delta = a[source] * np.random.rand() * 0.1 # say up to 10% of source
       if delta < 1e-12: continue
       a2 = a.copy()
       a2[source] -= delta
       a2[target] += delta
       # maintain non-negativity
       a2[source] = max(a2[source], 0.0) # but we maintain by delta bound

       conv2 = _fft_self_convolve(a2)
       M2 = conv2.max()
       # acceptance
       if M2 < M or np.random.rand() < np.exp((M - M2) / max(1e-12, T)):
           a = a2
           conv = conv2
           M = M2
           if M2 < best_M:
               best_M = M2
               best_a = a.copy()
               # print improvement
               if it % 100 == 0:
                   sc = 2.0 * n * M2
                   print(f"[{time.time():.1f}] SA it={it} new best {sc:.6f}")
       # update temperature
       T = T0 * (T_end / T0) ** (it / max_iter)
   # return best found after SA
   return best_a, 2.0*n*best_M

   occasionally accepting uphill moves to escape local minima.

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of vector v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Return the objective value 2·n·max(a*a) for a vector already
    normalised to sum = 1.  This is exactly the value that
    `evaluate_sequence` would return (up to the scaling by sum²,
    which is 1 here).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n, p, q):
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _generate_candidates_for_length(n):
    """Create a list of candidate vectors (each sums to 1) for a given length."""
    cand = []

    # Analytic families
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))
    cand.append(_distribution_exponential(n, lam=0.003))
    cand.append(_distribution_exponential(n, lam=0.02))

    # Symmetric Beta
    for p in (0.02, 0.04, 0.06, 0.08, 0.12, 0.15, 0.2,
              0.25, 0.3, 0.4, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        cand.append(_distribution_beta_sym(n, p))

    # Asymmetric Beta – a few skewed shapes
    for p, q in ((0.2, 2.0), (2.0, 0.2), (0.5, 3.0), (3.0, 0.5)):
        cand.append(_distribution_beta_asym(n, p, q))

    # Sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0):
        cand.append(_distribution_sine_power(n, pw))

    # Cosine window
    for coeff in (0.3, 0.5, 0.7, 0.9):

    # Gaussian
    for sigma in (0.12, 0.18, 0.25, 0.35, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law decreasing and its reverse
    for p in (1.2, 1.5, 2.0, 2.5, 3.0):
        pl = _distribution_powerlaw(n, p)

        cand.append(pl[::-1])

    # Random shuffled versions of a few base families (add asymmetry)
    for base in (cand[0], cand[1], cand[4]):
        shuf = base.copy()
        np.random.shuffle(shuf)
        cand.append(shuf)

def _smooth_max_mirror_descent_extended(a, deadline):
    """Exponentiated‑gradient mirror descent on the smooth‑max approximation."""
    n = a.shape[0]
    eps = 1e-12

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [

        (100.0, 5_000.0, 3000, 0.02, 0.002),
        (5_000.0, 50_000.0, 3000, 0.002, 0.0002),
        (50_000.0, 500_000.0, 3000, 0.0002, 2e-05),

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smooth‑max objective
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

            a = a * np.exp(-lr * grad)
            a = _project_onto_simplex(a, 1.0)
            a = np.maximum(a, eps)
            a /= a.sum()

            if it % 600 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _simulated_annealing(a, deadline, max_iter=3000):
    """Simple temperature‑controlled random mass‑transfer walk."""
    n = a.shape[0]
    eps = 1e-12

    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M

    # temperature schedule (relative to current max)
    T0 = max(M * 0.05, 1e-9)          # start fairly hot
    T_end = max(M * 1e-7, 1e-12)     # end very cold

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # geometric temperature decay
        T = T0 * (T_end / T0) ** (it / max_iter)

        # pick a source proportional to its mass
        source = np.random.choice(n, p=a)   # a already sums to 1
        target = np.random.randint(0, n)
        if target == source:
            continue

        # propose moving a modest fraction of the source mass
        delta = a[source] * 0.1 * np.random.rand()
        if delta < eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta

        # Ensure non‑negativity (tiny negative corrections)
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()

        conv2 = _fft_self_convolve(a2)
        M2 = conv2.max()
        delta_M = M2 - M

        accept = False
        if delta_M <= 0:
            accept = True
        else:
            prob = math.exp(-delta_M / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            a = a2
            conv = conv2
            M = M2
            if M2 < best_M:
                best_M = M2
                best_a = a2.copy()
                if it % 500 == 0:
                    sc = 2.0 * n * best_M
                    print(f"[{time.time():.1f}] SA it={it} new best {sc:.6f}")

    return best_a, 2.0 * n * best_M


def _projected_subgradient(a, deadline):
    """Adam‑style sub‑gradient descent with back‑tracking."""
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

def _hill_climb_prune(a, deadline):

        k = int(conv.argmax())

        max_prod = a[best_i] * a[best_j]

            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
