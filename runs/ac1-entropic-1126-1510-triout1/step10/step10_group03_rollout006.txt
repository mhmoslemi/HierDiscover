"""
Search for a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The search works under a fixed time‑budget (default
1000 s) and proceeds in three stages:

1. **Broad exploration** – for several sequence lengths (up to 1.6·10⁵)
   a variety of analytic families are instantiated:
   * arcsine (the continuous Beta(½,½))
   * symmetric Beta(p,p)   with p ∈ [0.10,0.50]
   * asymmetric Beta(p, q) (p varies, q fixed = 0.5)
   * edge‑biased power‑law   wᵢ ∝ ((i+½)(n‑i‑½))^{‑α}+c   with α∈[0.25,0.5]
   * a few Dirichlet draws with the same edge‑bias.
   The best *few* candidates are remembered.

2. **Smooth‑max mirror descent** – starting from the current best
   candidate a is updated by an exponentiated‑gradient step that
   minimises a soft‑max approximation of the max‑convolution.
   The temperature (α) grows geometrically while the learning‑rate
   shrinks, giving a smooth transition from exploration to precise
   optimisation.

3. **Projected Adam‑style sub‑gradient descent** on the true max‑convolution,
   followed by a tiny mass‑transfer hill‑climber that moves probability
   mass from the largest contributing pair to the smallest entry.
   Both steps respect the simplex constraint.

All helper routines are pure NumPy and avoid any filesystem or network I/O.
The function returns the best sequence (as a plain Python list) found
before the deadline.
"""

import time

import math

# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1,
    returns the exact value produced by `evaluate_sequence`
    (i.e. 2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) when p < 1
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)  # shift for numerical stability
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_power(n: int, alpha: float, c: float = 0.0) -> np.ndarray:
    """
    Deterministic edge‑biased distribution:
        w_i ∝ ((i+½)(n‑i‑½))^{-α} + c .
    """
    idx = np.arange(n, dtype=np.float64)
    w = ((idx + 0.5) * (n - idx - 0.5)) ** (-alpha) + c
    w = np.maximum(w, 0.0)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w_sum
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Random Dirichlet draw with concentration concentrated at the edges:
    concentration ∝ ((i+½)(n‑i‑½))^{-beta_exp} + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec

def _smooth_max_mirror_step(a: np.ndarray, alpha: float, lr: float, eps: float = 1e-12):
    """
    One exponentiated‑gradient (mirror‑descent) step on the smooth‑max
    objective  f_α(a) = (1/α)·log Σ exp(α·conv_k).
    Returns the updated probability vector.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    m = conv.max()
    # soft‑max weights
    w = np.exp(alpha * (conv - m))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient of the smooth‑max (chain rule)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    # exponentiated‑gradient update on the simplex (mirror w.r.t. KL)
    a_new = a * np.exp(-lr * grad)
    a_new = _project_onto_simplex(a_new, 1.0)
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    return a_new, conv.max()


def _adam_subgrad_step(a: np.ndarray,
                       vel: np.ndarray,
                       step0: float,
                       beta1: float,
                       beta2: float,
                       it: int,
                       eps: float = 1e-12):
    """
    One Adam‑style projected sub‑gradient step on the true max‑convolution.
    Returns the updated vector, the new velocity and the new maximal
    convolution value.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    cur_max = conv.max()
    # active set of indices where the maximum is attained (within tolerance)
    tol = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(conv >= cur_max - tol)

    w = np.zeros_like(conv)
    w[active] = 1.0
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

    # Adam momentum update (bias‑corrected)
    vel = beta1 * vel + (1.0 - beta1) * grad
    v_hat = vel / (1.0 - beta1 ** it)

    # back‑tracking line search
    step = step0
    while True:
        a_cand = a - step * v_hat
        a_cand = _project_onto_simplex(a_cand, 1.0)
        a_cand = np.maximum(a_cand, eps)
        a_cand /= a_cand.sum()
        cand_score = _max_convolution_score(a_cand)
        if cand_score <= _max_convolution_score(a) or step < 1e-9:
            break
        step *= 0.5

    return a_cand, vel, conv.max()


def _mass_transfer_hill_climb(a: np.ndarray,
                             max_iters: int,
                             eps: float = 1e-12):
    """
    Simple hill‑climber that repeatedly moves a tiny amount of probability
    mass from the index that contributes most to the current peak towards
    the smallest entry.  Stops after max_iters or when time runs out.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    for it in range(max_iters):
        k = int(conv.argmax())                # location of the peak
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # Find the pair (i,j) giving the largest product a_i * a_j for this k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # Choose source (the larger of the two) and a tiny target (global min)
        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            # pick second‑smallest if they coincide
            target = np.argsort(a)[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
        # occasional jitter to avoid stagnation
        if (it + 1) % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
    return a

    top_candidates = top_candidates[:K]

    # ------------------------------------------------------------------
    # 2️⃣  Smooth‑max mirror descent on the overall best candidate
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of moderate size
        n0 = 20000
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)

    a = best_seq.copy()
    n = a.shape[0]

    # Decide on iteration budget based on remaining time
    remaining = deadline - time.time()
    est_per_it = 0.004 if n > 50000 else 0.002  # rough estimate
    max_iters = max(0, int(remaining / est_per_it))
    total_iters = max(2000, min(20000, max_iters))

    print(f"[{time.time():.1f}] Mirror‑descent phase: n={n}, iters={total_iters}")

    alpha_start, alpha_end = 0.5, 2e5           # smooth‑max temperature schedule
    lr_start, lr_end = 0.3, 0.001              # learning‑rate schedule

    for it in range(total_iters):
        if time.time() >= deadline:
            break
        t = it / total_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        a, cur_max = _smooth_max_mirror_step(a, alpha, lr, eps)

        # occasional exact score check
        if (it + 1) % max(1, total_iters // 10) == 0 or it == total_iters - 1:
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_seq, best_score = a.copy(), cur_score
                print(f"[{time.time():.1f}] mirror it={it}/{total_iters} new best {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Projected Adam‑style sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.6
    sub_it = 0

    while time.time() < deadline - 0.5:
        sub_it += 1
        a, velocity, cur_max = _adam_subgrad_step(a, velocity, step0, beta1, beta2, sub_it, eps)
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_seq, best_score = a.copy(), cur_score
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cur_score:.6f}")

        # occasional jitter to escape shallow plateaus
        if sub_it % 800 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    # ------------------------------------------------------------------
    # 4️⃣  Tiny mass‑transfer hill‑climber for final polishing
    # ------------------------------------------------------------------
    a = best_seq.copy()
    max_hill_iters = 5000
    a = _mass_transfer_hill_climb(a, max_hill_iters, eps)
    final_score = _max_convolution_score(a)
    if final_score < best_score:
        best_seq, best_score = a.copy(), final_score
        print(f"[{time.time():.1f}] final hill‑climber improved score to {final_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found
    # ------------------------------------------------------------------
    print(f"[{time.time():.1f}] Finished. Best score ≈ {best_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
