def _smoothed_subgradient_descent(a, deadline, best_seq, best_score, lr0=0.6, max_iter=20000, beta0=0.5, beta_mul=1.001, beta_max=50.0, eval_interval=200):

    n = len(a)
    t = 0
    beta = beta0
    while t < max_iter and time.time() < deadline:
        t += 1

        # compute conv
        conv = _fft_self_convolve(a)  # length 2n-1
        max_val = conv.max()

        # compute weights w via softmax
        # subtract max to avoid overflow
        expo = np.exp(beta * (conv - max_val))
        w = expo / expo.sum()

        # gradient
        grad = _gradient_from_weights(a, w)

        # schedule step size
        eta = lr0 / math.sqrt(t)

        # update a using exponentiated gradient
        a = a * np.exp(-eta * grad)

        # simplex projection
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # update beta (annealing)
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # evaluate sometimes
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] smoothed it={t} beta={beta:.2f} new best {best_score:.6f}")

    return best_seq, best_score

The algorithm works in three stages:

1. **Seed generation** – a diverse pool of analytic and random
   probability vectors for several lengths (up to 64 000).  Each
   candidate is normalised to sum‑1 and its exact objective
   `2·n·max(conv(a,a))` (equivalent to `evaluate_sequence` for sum=1)
   is recorded.

2. **Smoothed sub‑gradient descent** – for the best few seeds we run a
   mirror‑descent optimisation on the simplex.  The sub‑gradient is
   taken as `2·a⋆w` where `w` is a *soft‑max* weighting of the
   convolution entries, gradually annealed (β grows) so that the
   gradient approaches the true active‑set sub‑gradient.  We use an
   exponentiated‑gradient update and a Euclidean projection onto the
   probability simplex.

3. **Local polishing** – after the continuous phase we apply a cheap
   greedy pairwise‑mass‑transfer walk that accepts any move that lowers
   the true objective.

The whole search respects a global time budget (`budget_s` seconds,
default 1000).  The best sequence found is returned as a plain list of
floats, ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# -------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# -------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(conv(a,a))
    The factor 2·n is the same as `evaluate_sequence` after the
    normalisation performed therein.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


# -------------------------------------------------------------
# Analytic families of probability distributions (all sum to 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_beta_asym(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-15
    w = np.exp((alpha - 1) * np.log(np.clip(x, eps, 1.0)) +
               (beta - 1) * np.log(np.clip(1.0 - x, eps, 1.0)))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_increasing(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** p
    vals /= vals.sum()
    return vals


def _distribution_exponential_increasing(n: int, lam: float = 0.005) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(lam * i)
    vals /= vals.sum()
    return vals


def _distribution_random_sparse(n: int, rng: np.random.Generator,
                               density: float = 0.01) -> np.ndarray:
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _distribution_two_blocks(n: int, block_frac: float = 0.07) -> np.ndarray:
    k = max(1, int(block_frac * n))
    a = np.zeros(n, dtype=np.float64)
    a[:k] = 1.0
    a[-k:] = 1.0
    a /= a.sum()
    return a


def _distribution_dirichlet(n: int, rng: np.random.Generator,
                           alpha: float = 0.5) -> np.ndarray:
    vals = rng.gamma(alpha, size=n)
    if vals.sum() == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create an assorted list of probability vectors for length n."""
    cand = [
        _distribution_uniform(n),
        _distribution_arcsine(n),
        *_distribution_beta_sym_variants(n),
        *_distribution_beta_asym_variants(n, rng),
        *_distribution_power_variants(n),
        *_distribution_exponential_variants(n),
        _distribution_two_blocks(n, block_frac=0.05),
        _distribution_two_blocks(n, block_frac=0.10),
        _distribution_random_sparse(n, rng, density=0.01),
    ]
    # a few Dirichlet draws
    for _ in range(3):
        cand.append(_distribution_dirichlet(n, rng,
                                            alpha=0.3 + 0.7 * rng.random()))
    return cand


def _distribution_beta_sym_variants(n: int) -> list[np.ndarray]:
    out = []
    for p in (0.05, 0.1, 0.2, 0.3, 0.4, 0.6, 0.8, 1.0):
        out.append(_distribution_beta_sym(n, p))
    return out


def _distribution_beta_asym_variants(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    out = []
    for _ in range(5):
        alpha = 10 ** rng.uniform(-2, 1)
        beta = 10 ** rng.uniform(-2, 1)
        out.append(_distribution_beta_asym(n, alpha, beta))
    return out


def _distribution_power_variants(n: int) -> list[np.ndarray]:
    out = []
    for p in (0.5, 1.0, 2.0, 3.0, 4.0):
        out.append(_distribution_power_increasing(n, p))
    return out


def _distribution_exponential_variants(n: int) -> list[np.ndarray]:
    out = []
    for lam in (0.001, 0.005, 0.01, 0.02, 0.05):
        out.append(_distribution_exponential_increasing(n, lam))
    return out


# -------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the sub‑gradient 2·a⋆w where w has length 2·n‑1.
    The operation is performed via FFT cross‑correlation.
    """
    n = a.shape[0]
    m = w.shape[0]                # = 2·n‑1
    a_rev = a[::-1]
    size = 1 << ((n + m - 1) - 1).bit_length()
    W = np.fft.rfft(w, n=size)
    A_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(W * A_rev, n=size)
    start = n - 1
    return 2.0 * conv[start:start + n]


# -------------------------------------------------------------
def _smoothed_subgradient_descent(a: np.ndarray,
                                  deadline: float,
                                  best_seq: np.ndarray,
                                  best_score: float,
                                  lr0: float = 0.6,
                                  max_iter: int = 20000,
                                  beta0: float = 0.5,
                                  beta_mul: float = 1.001,
                                  beta_max: float = 50.0,
                                  eval_interval: int = 200) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using a soft‑max weighting of the
    convolution entries.  The temperature β grows (annealing) so that the
    gradient approaches the true active‑set sub‑gradient.
    """
    n = a.shape[0]
    t = 0
    beta = beta0
    while t < max_iter and time.time() < deadline:
        t += 1

        conv = _fft_self_convolve(a)
        max_val = conv.max()

        # soft‑max weights (numerically stable)
        exp_term = np.exp(beta * (conv - max_val))
        w = exp_term / exp_term.sum()

        grad = _gradient_from_weights(a, w)

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # project back onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # anneal β
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-13:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] it={t} β={beta:.2f} new best {best_score:.6f}")

    return best_seq, best_score


# -------------------------------------------------------------
def _pairwise_swap_local(a: np.ndarray,
                         deadline: float,
                         best_seq: np.ndarray,
                         best_score: float) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer walk that accepts any improving move."""
    n = a.shape[0]
    while time.time() < deadline:
        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.30
        if delta < 1e-12:
            continue
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        cur = _objective(new_a)
        if cur < best_score - 1e-13:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(f"[{time.time():.1f}] swap {i}->{j} δ={delta:.3g} new best {best_score:.6f}")
    return best_seq, best_score


# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point: search for a non‑negative step‑function that
    minimizes `evaluate_sequence`.  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # -----------------------------------------------------------------
    # 1. Choose lengths – we go up to 64 000 (still cheap for FFT)
    lengths = [2000, 4000, 8000, 16000, 32000, 64000]

    overall_best_seq = None
    overall_best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # -----------------------------------------------------------------
    # 2. Scan analytic / random seeds for each length
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] scanning length {n}")

        for cand in _initial_candidates(n, rng):
            if time.time() >= deadline:
                break
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            sc = _objective(cand)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = cand.copy()
                print(f"[{time.time():.1f}] **** new global best len={n} score={sc:.6f}")
            top_candidates.append((sc, cand.copy()))

        # keep only a few elite seeds across all lengths
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:12]

    # -----------------------------------------------------------------
    # 3. Incorporate a previously found sequence if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s == 0.0:
                prev = _distribution_uniform(prev.shape[0])
                s = prev.sum()
            prev = prev / s
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best improved: {sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # -----------------------------------------------------------------
    # 4. Ensure we have a fallback
    if overall_best_seq is None:
        n = 2000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # -----------------------------------------------------------------
    # 5. Intensive continuous optimisation on the elite seeds
    remaining = deadline - time.time()
    if remaining > 60:
        # pick at most 5 best seeds for the heavy phase
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        for idx, seed_seq in enumerate(seeds, 1):
            if time.time() >= deadline:
                break
            # allocate roughly equal remaining time to each seed
            per_seed_deadline = time.time() + (deadline - time.time()) / max(1, len(seeds) - idx + 1)
            print(f"[{time.time():.1f}] starting sub‑gradient descent on seed {idx}/{len(seeds)} (len={len(seed_seq)})")
            seq_out, score_out = _smoothed_subgradient_descent(
                seed_seq.copy(),
                per_seed_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                lr0=0.7,
                max_iter=30000,
                beta0=0.5,
                beta_mul=1.001,
                beta_max=60.0,
                eval_interval=400,
            )
            if score_out < overall_best_score:
                overall_best_score = score_out
                overall_best_seq = seq_out.copy()
                print(f"[{time.time():.1f}] after descent new best {overall_best_score:.6f}")

    # -----------------------------------------------------------------
    # 6. Final polishing with greedy local swaps
    if time.time() < deadline:
        print(f"[{time.time():.1f}] polishing with local swaps (len={len(overall_best_seq)})")
        overall_best_seq, overall_best_score = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )

    # -----------------------------------------------------------------
    print(f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})")
    # Return a plain Python list as required by the evaluator.
    return overall_best_seq.tolist()
