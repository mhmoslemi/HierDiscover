"""
Propose a low‑score non‑negative step function for ``evaluate_sequence``.

The search builds a modest set of candidate lengths, then for each length
optimises a few 1‑parameter families (exponential, power‑law,
edge‑power) by golden‑section.  The best few candidates are locally
refined with projected Adam and exponentiated‑gradient descent, followed
by a Frank‑Wolfe style mass‑transfer and a final simulated‑annealing /
swap phase.  All operations respect the simplex (non‑negative, sum = 1)
so the final list can be returned directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of ``v`` onto the simplex {x≥0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Surrogate objective for Σa = 1: 2·n·max(conv(a,a))."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """Sub‑gradient of the max‑convolution (used by Frank‑Wolfe)."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    w = np.zeros_like(b)
    w[active] = 1.0
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # diagonal correction for i=j contributions (when k even)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active


# ----------------------------------------------------------------------
def _adam_descent(a0: np.ndarray, deadline: float,
                  step0: float = 0.30,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  max_iters: int = 800_000):
    """Projected Adam on the simplex (returns best point seen)."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score


def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          eta0: float = 0.9,
                          max_iters: int = 600_000):
    """Exponentiated‑gradient (mirror) descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s

        cur = _max_convolution_score(a)
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
    return best_a, best_score

def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet point with concentration ``alpha``."""
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

# Analytic families ----------------------------------------------------
def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    """a_i ∝ exp(-lam·|i - centre|)."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _power_law_distribution(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (i starts at 0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _edge_power_distribution(n: int, s: float) -> np.ndarray:
    """a_i ∝ (i·(n‑i+1))^{-s}, i = 1,…,n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i * (n - i + 1)) ** (-s)
    vals /= vals.sum()
    return vals


def _arcsine_distribution(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine law."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _golden_section(f, lo: float, hi: float, max_iter: int = 30,
                    tol: float = 1e-8):
    """Simple golden‑section search for a unimodal scalar function."""
    phi = (1 + 5 ** 0.5) / 2
    inv = 1 / phi
    a, b = lo, hi
    for _ in range(max_iter):
        c = b - inv * (b - a)
        d = a + inv * (b - a)
        fc = f(c)
        fd = f(d)
        if fc < fd:
            b = d
        else:
            a = c
        if b - a < tol:
            break
    return (a + b) / 2

    def obj(lam: float) -> float:
        a = _exponential_distribution(n, lam)
        return _max_convolution_score(a)

    lam_opt = _golden_section(obj, 1e-4, 0.2, max_iter=28)
    a_opt = _exponential_distribution(n, lam_opt)
    return a_opt, _max_convolution_score(a_opt)


def _best_powerlaw_for_n(n: int) -> tuple[np.ndarray, float]:
    """Find p ∈ [0.1, 2.5] that minimises the surrogate."""
    def obj(p: float) -> float:
        a = _power_law_distribution(n, p)
        return _max_convolution_score(a)

    p_opt = _golden_section(obj, 0.1, 2.5, max_iter=28)
    a_opt = _power_law_distribution(n, p_opt)
    return a_opt, _max_convolution_score(a_opt)


def _best_edge_power_for_n(n: int) -> tuple[np.ndarray, float]:

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main entry point for the competition.

    Generates a collection of candidates, refines a handful of the
    best with first‑order methods and a final simulated‑annealing
    polishing, and returns the best list of floats found before the
    time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose a set of lengths.  We deliberately include a relatively
    # large length (≈20000) because the constant 2·n·max(conv) tends to
    # improve slowly with n for good shapes.
    base_n = 6000
    try:
        base_n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        pass
    base_n = max(3000, min(base_n, 18000))

    lengths = sorted({
        int(base_n * f) for f in (0.7, 1.0, 1.3, 1.6, 2.0)
        if 1500 <= int(base_n * f) <= 25000
    })
    # ensure we also test a very large size (if time permits)
    lengths.append(min(25000, base_n * 2))
    lengths = sorted(set(lengths))

    print(f"[{time.time():.1f}] Exploring lengths: {lengths}")

    # --------------------------------------------------------------
    # Gather analytic candidates (including a fine‑grained param search)
    candidates: list[tuple[np.ndarray, float]] = []

    for n in lengths:
        if time.time() >= deadline:
            break

        # a few deterministic families (arcsine, uniform‐like)
        a = _arcsine_distribution(n)
        candidates.append((a, _max_convolution_score(a)))

        # exponential family – golden‑section optimisation
        a_exp, sc_exp = _best_exponential_for_n(n)
        candidates.append((a_exp, sc_exp))

        # power‑law family
        a_pow, sc_pow = _best_powerlaw_for_n(n)
        candidates.append((a_pow, sc_pow))

        # edge‑power (generalised arcsine)
        a_edge, sc_edge = _best_edge_power_for_n(n)
        candidates.append((a_edge, sc_edge))

        # random Dirichlet draws – give extra diversity
        for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
            a = _random_dirichlet_alpha(n, alpha)
            candidates.append((a, _max_convolution_score(a)))

    # If a previous best exists, use it and some jittered copies.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = prev / prev.sum()
            candidates.append((prev, _max_convolution_score(prev)))
            for _ in range(4):
                jitter = np.random.randn(prev.size) * 4e-5
                noisy = _project_onto_simplex(prev + jitter, s=1.0)
                candidates.append((noisy, _max_convolution_score(noisy)))
            print(f"[{time.time():.1f}] added previous best (+jitter)")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Keep the best few seeds for heavy refinement
    candidates.sort(key=lambda x: x[1])
    top_k = min(30, len(candidates))
    seeds = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, retaining top {top_k}")

    best_seq = seeds[0][0].copy()
    best_score = seeds[0][1]
    print(f"Initial best surrogate = {best_score:.6f}")

    # --------------------------------------------------------------
    # Refine each seed – allocate remaining time fairly.
    for idx, (seed_seq, _) in enumerate(seeds):
        if time.time() >= deadline:
            break
        n = seed_seq.shape[0]

        # Adam descent
        seq_adam, sc_adam = _adam_descent(seed_seq, deadline,
                                          max_iters=500_000)
        if sc_adam < best_score - 1e-14:
            best_score, best_seq = sc_adam, seq_adam.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} Adam → {best_score:.6f}")

        # Exponentiated gradient
        seq_eg, sc_eg = _exp_gradient_descent(seed_seq, deadline,
                                              max_iters=400_000)
        if sc_eg < best_score - 1e-14:
            best_score, best_seq = sc_eg, seq_eg.copy()
            print(f"[{time.time():.1f}] seed {idx+1}/{top_k} EG → {best_score:.6f}")

    # --------------------------------------------------------------
    # Frank‑Wolfe style mass‑transfer polishing
    a = best_seq.copy()
    cur_score = best_score
    fractions = (1.0, 0.7, 0.4, 0.2, 0.1)
    fw_iter = 0
    while time.time() < deadline and fw_iter < 150_000:
        fw_iter += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))   # move mass *away* from large gradient
        i_dst = int(np.argmin(grad))   # move mass *towards* small gradient
        if i_src == i_dst:
            break
        max_transfer = min(a[i_src], 1.0 - a[i_dst])
        if max_transfer < 1e-15:
            break
        improved = False
        for f in fractions:
            delta = max_transfer * f
            a_new = a.copy()
            a_new[i_src] -= delta
            a_new[i_dst] += delta
            sc = _max_convolution_score(a_new)
            if sc < cur_score - 1e-13:
                a = a_new
                cur_score = sc
                improved = True
                break
        if improved and cur_score < best_score - 1e-14:
            best_score, best_seq = cur_score, a.copy()
            print(f"[{time.time():.1f}] FW iter {fw_iter} new best {best_score:.6f}")

        # occasional random jitter to avoid tiny plateaus
        if fw_iter % 6000 == 0 and np.random.rand() < 0.12:
            jitter = np.random.randn(a.size) * 5e-5
            a = _project_onto_simplex(a + jitter, s=1.0)
            cur_score = _max_convolution_score(a)

    # --------------------------------------------------------------
    # Simulated‑annealing / swap phase (final push)
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 0.025
    temp_end = 0.001
    anneal_iter = 0
    while time.time() < deadline and anneal_iter < 250_000:
        anneal_iter += 1
        # linear temperature schedule (based on remaining fraction)
        fraction = max(0.0, (deadline - time.time()) / (budget_s - 0.5))
        temp = temp_end + (temp_start - temp_end) * fraction

        move_type = np.random.rand()
        if move_type < 0.7:          # transfer move (most common)
            i = np.random.randint(a.size)
            j = np.random.randint(a.size - 1)
            if j >= i:
                j += 1
            max_delta = min(a[i], 1.0 - a[j])
            if max_delta <= 0:
                continue
            delta = max_delta * np.random.rand() * temp
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:                        # swap move (occasionally)
            i, j = np.random.choice(a.size, 2, replace=False)
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]

        new_score = _max_convolution_score(a_new)

        if new_score < cur_score - 1e-15:
            a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-14:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{time.time():.1f}] Anneal iter {anneal_iter} best {best_score:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                a = a_new
                cur_score = new_score

    print(f"Final surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # The external ``evaluate_sequence`` will handle any final scaling.
    return best_seq.tolist()
