"""
Iterative LP‑+‑gradient optimiser for the step‑function problem.

The core idea is to treat the current distribution `a` as fixed and
solve a linear programme that maximises the total mass of a new
distribution `b` while guaranteeing that the convolution `a*b` never
exceeds the current worst‑case value `M = max(a*a)`.  After normalising
`b` we mix it with `a` (convex combination) – this never raises the
maximum self‑convolution because each of the three terms
   a*a , a*b , b*b
is bounded by `M`.  Repeating the LP step (typically 10‑30 times) drives
`M` down.  Finally a cheap hill‑climber moves a tiny amount of mass
from indices with large gradient to those with small gradient, providing
fine‑grained improvement.

The optimiser works with normalised vectors (sum = 1), because the
objective `evaluate_sequence` is invariant under scaling.  Several
analytic families (uniform, arcsine, beta, sine‑power, Gaussian) and a
few random Dirichlet draws are used as starting points; the best result
found within the time budget is returned.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a normalised vector (sum = 1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n: int, lam: float = 0.01) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n: int, power: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred at (n‑1)/2."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def _subgradient_weights(conv: np.ndarray, eps_rel: float = 1e-12) -> np.ndarray:
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _hillclimber_move(a: np.ndarray, conv: np.ndarray, eps: float = 1e-12):
    """Move a tiny chunk from a large‑gradient index to a small‑gradient one."""
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False


def _hillclimber_opt(a: np.ndarray, deadline: float, max_iters: int = 30000):
    """Runs a cheap hill‑climber for a while (stops on time‑out)."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    it = 0
    while it < max_iters and time.time() < deadline:
        a, conv, improved = _hillclimber_move(a, conv, eps=1e-12)
        if improved and (it % 500 == 0):
            # occasional progress report
            print(f"[{time.time():.1f}] hillclimb it={it}  max={conv.max():.6f}")
        if it % 5000 == 0 and it > 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            conv = _fft_self_convolve(a)
        it += 1
    return a

def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix M such that M @ b = a * b (full linear convolution).
    """
    n = a.shape[0]
    m = 2 * n - 1
    M = np.zeros((m, n), dtype=np.float64)
    for i in range(n):
        M[i:i + n, i] = a
    return M


def _lp_refine(a: np.ndarray, n: int, deadline: float,
               max_lp_iters: int = 20, t_start: float = 0.5) -> np.ndarray:
    """
    Repeatedly solve the linear programme:
        maximise   Σ b_j
        subject to b ≥ 0,  a * b ≤ M
    where M = max(a * a).  After each solve we normalise `b`,
    mix it into `a` with step‑size `t` (which decays) and repeat.
    Returns the refined vector (still sum = 1).
    """
    eps = 1e-12
    a = np.maximum(a, eps)
    a = a / a.sum()
    best_score = _objective(a)
    t = t_start

    for it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        conv = _fft_self_convolve(a)
        M = conv.max()
        # build convolution matrix for the current a
        A = _build_conv_matrix(a)

        # Linear programme
        b = cp.Variable(n, nonneg=True)
        constraints = [A @ b <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                       feastol=1e-9, abstol=1e-9, reltol=1e-9, verbose=False)
        except Exception:
            # solver failure – abort refinement
            break

        if b.value is None:
            break

        b_val = np.asarray(b.value).flatten()
        sum_b = b_val.sum()
        if sum_b <= eps:
            break

        # normalise b to have unit mass
        g = b_val / sum_b

        # convex mix (guaranteed not to increase the max‑convolution)
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, eps)
        a = a / a.sum()

        # decay step size a little
        t = max(t * 0.9, 0.02)

        # check for improvement
        sc = _objective(a)
        if sc + 1e-12 < best_score:
            best_score = sc
            print(f"[{time.time():.1f}] LP it={it}  n={n}  score={sc:.6f}")

    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Uses a sequence of analytic initialisations, LP‑driven refinement and a
    final hill‑climber.  The best vector found before the deadline is returned.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1.  Assemble a collection of promising initial vectors
    init_lengths = [1800, 2000, 2500, 3000, 3500, 4000]
    init_families = []

    for n in init_lengths:
        init_families.append(_distribution_uniform(n))
        init_families.append(_distribution_arcsine(n))
        init_families.append(_distribution_triangular(n))
        init_families.append(_distribution_exponential(n, lam=0.008))
        init_families.append(_distribution_exponential(n, lam=0.02))
        for p in (0.3, 0.6, 1.0, 2.0, 4.0):
            init_families.append(_distribution_beta_sym(n, p))
        for pw in (0.5, 1.0, 2.0, 4.0):
            init_families.append(_distribution_sine_power(n, pw))
        for sigma in (0.2, 0.35, 0.5):
            init_families.append(_distribution_gaussian(n, sigma))
        # a few random Dirichlet draws
        for alpha in (0.2, 0.5, 1.0, 2.0):
            init_families.append(np.random.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # 2.  If a previous global best exists, include it
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            init_families.append(prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Process each candidate: LP refinement + hill‑climber
    for vec in init_families:
        if time.time() >= deadline:
            break
        n = vec.shape[0]
        # normalise and guard against zero entries
        a = np.maximum(vec, 1e-12)
        a = a / a.sum()

        # LP‑driven refinement
        a = _lp_refine(a, n, deadline, max_lp_iters=25, t_start=0.5)

        # final hill‑climber (lightweight)
        a = _hillclimber_opt(a, deadline, max_iters=20000)

        sc = _objective(a)
        if sc < best_score - 1e-12:
            best_seq = a.copy()
            best_score = sc
            best_n = n
            print(f"[{time.time():.1f}] new global best n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4.  If nothing succeeded (unlikely), fall back to a uniform vector
    if best_seq is None:
        n = 2500
        best_seq = _distribution_uniform(n)
        best_score = _objective(best_seq)
        best_n = n
        print(f"[{time.time():.1f}] fallback uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5.  Optional short random‑restart phase (while time allows)
    restart_lengths = [1800, 2000, 2500, 3000, 3500, 4000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        a = np.random.dirichlet(np.full(n, 0.5))
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
        a = _lp_refine(a, n, deadline, max_lp_iters=12, t_start=0.3)
        a = _hillclimber_opt(a, deadline, max_iters=10000)
        sc = _objective(a)
        if sc < best_score - 1e-12:
            best_seq = a.copy()
            best_score = sc
            best_n = n
            print(f"[{time.time():.1f}] restart improvement n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    # Return plain Python list as required by the competition harness
    return best_seq.tolist()
