"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  The search is split into four phases:

1️⃣  A wide parametric sweep (Beta, power‑law, exponential, Dirichlet,
    uniform, arcsine) over several lengths.  The best few candidates are
    kept.

2️⃣  Multi‑restart mirror‑descent (exponentiated‑gradient on a smooth‑max
    approximation of the convolution) for each retained candidate.

3️⃣  Adam‑style projected sub‑gradient descent on the true max‑convolution.

4️⃣  Simple mass‑transfer hill climbing to polish the result.

All steps respect the time budget (default 1000 s) and the constraints
of `evaluate_sequence`.  The returned list is a plain Python list.
"""

import time
import math
import random
import heapq
import numpy as np

# Utility functions (no closures)

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1 returns the exact
    `evaluate_sequence` value (2·n·max(a*a)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if w.sum() == 0 or not np.isfinite(w).all():
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_power_law(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-alpha) + (n-i)^(-alpha) (symmetric power‑law)."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** (-alpha) + (n - i) ** (-alpha)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def _distribution_exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i) + exp(-lam * (n-1-i)) (symmetric exponential)."""
    i = np.arange(n, dtype=np.float64)
    w = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """Dirichlet draw concentrating mass at the edges."""
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


def _distribution_random_dirichlet(n: int, concentration: float = 1.0) -> np.ndarray:
    """Plain Dirichlet with equal concentration."""
    return np.random.dirichlet(np.full(n, concentration))

def _smooth_max_mirror_descent(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) on a soft‑max approximation of
    the max‑convolution.  Returns the best vector seen during the run.
    """
    a = a0.copy()
    n = a.shape[0]

    # geometric schedules (t∈[0,1])
    alpha_start, alpha_end = 1.0, 15000.0
    lr_start, lr_end = 0.3, 0.001

    best_local = a.copy()
    best_score = _max_convolution_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        t = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        conv_rev = _fft_convolve(a[::-1], w)
        grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if (it % max(1, max_iters // 10) == 0) or (it == max_iters - 1):
            cur = _max_convolution_score(a)
            if cur < best_score - 1e-12:
                best_score, best_local = cur, a.copy()

    return best_local


def _adam_subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    eps: float = 1e-12,
) -> np.ndarray:
    """
    Projected Adam‑style descent on the *true* max‑convolution.
    Stops when time runs out.
    """
    a = a0.copy()
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    step0 = 0.5
    it = 0
    best = a.copy()
    best_score = _max_convolution_score(a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam moment estimates (bias‑corrected)
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        step = step0
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _max_convolution_score(cand)
            if cand_sc < best_score - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_score - 1e-12:
            best_score, best = cand_sc, a.copy()

        # occasional jitter to escape shallow plateaus
        if it % 1000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best


def _hill_climb_polish(a0: np.ndarray, deadline: float, eps: float = 1e-12) -> np.ndarray:
    """
    Very simple mass‑transfer hill‑climbing: move a small amount of mass
    from the element contributing the largest product to the smallest
    element, accepting only strict improvements of the max‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    it = 0

    while time.time() < deadline:
        it += 1
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) causing the largest product for this peak
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod, best_i, best_j = prod, i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = np.argmin(a)
        if target == source:
            # pick the second‑smallest if clash
            srt = np.argsort(a)
            target = srt[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-12:
            a, conv, M = a2, _fft_self_convolve(a2), new_M

        # occasional jitter
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return a


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative list) minimising `evaluate_sequence`.
    The routine respects the time‑budget and returns the best vector found.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # a small safety margin
    eps = 1e-12

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep (keep top‑K candidates)
    K = 5
    # heap with entries (-score, n, vector) – we keep the *smallest* scores,
    # i.e. the most negative keys are the worst.
    heap: list[tuple[float, int, np.ndarray]] = []

    def push_candidate(vec: np.ndarray, n: int):
        sc = _max_convolution_score(vec)
        if len(heap) < K:
            heapq.heappush(heap, (-sc, n, vec.copy()))
        else:
            if sc < -heap[0][0]:
                heapq.heapreplace(heap, (-sc, n, vec.copy()))

    # Lengths to explore – a mix of small, medium and large sizes.
    lengths = [1500, 2500, 3000, 3500, 4000, 5000, 8000, 12000, 20000]
    # Parameter grids
    p_grid = np.concatenate([np.linspace(0.05, 0.95, 91), np.linspace(1.0, 2.0, 11)])
    q_grid = np.linspace(0.05, 2.0, 41)               # for asymmetric Beta
    alpha_grid = np.linspace(0.0, 2.0, 21)           # power‑law exponent
    lam_grid = np.logspace(-5, -1, 15)              # exponential decay
    edge_beta_exps = (0.4, 0.6, 0.8)

    for n in lengths:
        if time.time() >= deadline:
            break

        # uniform (baseline)
        push_candidate(_distribution_uniform(n), n)

        # arcsine (Beta(0.5,0.5))
        push_candidate(_distribution_arcsine(n), n)

        # symmetric Beta(p,p)
        for p in p_grid:
            if time.time() >= deadline:
                break
            push_candidate(_distribution_beta_sym(n, p), n)

        # asymmetric Beta(p,q) – a modest grid for coverage
        for p in [0.1, 0.3, 0.5, 0.8, 1.2]:
            for q in [0.1, 0.3, 0.5, 0.8, 1.2]:
                if time.time() >= deadline:
                    break
                push_candidate(_distribution_beta(n, p, q), n)

        # power‑law family
        for a in alpha_grid:
            if time.time() >= deadline:
                break
            push_candidate(_distribution_power_law(n, a), n)

        # exponential family
        for lam in lam_grid:
            if time.time() >= deadline:
                break
            push_candidate(_distribution_exponential(n, lam), n)

        # edge‑biased Dirichlet draws
        for bex in edge_beta_exps:
            if time.time() >= deadline:
                break
            push_candidate(_distribution_edge_dirichlet(n, bex), n)

        # a few random Dirichlet draws
        for _ in range(3):
            if time.time() >= deadline:
                break
            push_candidate(_distribution_random_dirichlet(n, concentration=1.0), n)

    # ------------------------------------------------------------------
    # Phase 2 – multi‑restart mirror descent on the retained candidates
    # Convert heap into a list sorted by score (ascending)
    candidates = sorted([(-sc, n, vec) for (sc, n, vec) in heap], key=lambda x: x[0])

    best_vec = None
    best_score = float("inf")
    best_n = None

    for base_score, n, vec in candidates:
        if time.time() >= deadline:
            break

        # Determine how many iterations we can allocate to this candidate
        remaining = deadline - time.time()
        est_per_iter = 0.004            # empirical estimate (seconds)
        max_iters = max(2000, int(min(15000, remaining / est_per_iter / len(candidates))))

        # Mirror descent from the current vector
        refined = _smooth_max_mirror_descent(vec, max_iters, deadline, eps=eps)

        sc = _max_convolution_score(refined)
        if sc < best_score:
            best_score, best_vec, best_n = sc, refined.copy(), n

        # One additional random restart for the same length (if time permits)
        if time.time() < deadline - 2.0:
            rand_init = _distribution_random_dirichlet(n, concentration=0.5)
            rand_refined = _smooth_max_mirror_descent(rand_init, max_iters // 2, deadline, eps=eps)
            sc2 = _max_convolution_score(rand_refined)
            if sc2 < best_score:
                best_score, best_vec, best_n = sc2, rand_refined.copy(), n

    # ------------------------------------------------------------------
    # Phase 3 – Adam‑style sub‑gradient descent on the best candidate
    if best_vec is not None:
        sub_deadline = deadline - 2.0
        if time.time() < sub_deadline:
            best_vec = _adam_subgradient_descent(best_vec, sub_deadline, eps=eps)
            best_score = _max_convolution_score(best_vec)

    # ------------------------------------------------------------------
    # Phase 4 – simple hill‑climbing polish
    if best_vec is not None:
        hill_deadline = deadline - 1.0
        if time.time() < hill_deadline:
            best_vec = _hill_climb_polish(best_vec, hill_deadline, eps=eps)
            best_score = _max_convolution_score(best_vec)

    # ------------------------------------------------------------------
    # Fallback – should never happen but guarantees a return value.
    if best_vec is None:
        n = 3000
        best_vec = _distribution_uniform(n)
        best_score = _max_convolution_score(best_vec)
        best_n = n

    final_score = _max_convolution_score(best_vec)
    print(f"Final best score ≈ {final_score:.6f} (length={best_vec.shape[0]})")
    return best_vec.tolist()
