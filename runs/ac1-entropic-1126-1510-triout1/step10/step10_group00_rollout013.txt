def _smooth_gradient_descent(a, deadline, best_seq, best_score,
                             beta_start=5.0, beta_end=5000.0,
                             lr0=0.5, eval_interval=100):
    n = a.shape[0]
    iteration = 0
    # optionally precompute number of steps estimate? We'll just loop.
    while time.time() < deadline:
        iteration += 1

        # compute convolution a * a
        conv = _fft_self_convolve(a)  # size 2n-1

        # annealing factor beta
        t_frac = iteration / 200000.0  # arbitrary large horizon
        beta = beta_start * ((beta_end / beta_start) ** min(1.0, t_frac))

        # compute softmax weights
        # shift by max to avoid overflow
        maxc = conv.max()
        expw = np.exp(beta * (conv - maxc))
        Z = expw.sum()
        if Z == 0.0:
            # fallback to concentrate on max index
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = expw / Z

        # compute gradient: g = 2 * correlation(a, w)
        a_rev = a[::-1]
        # full convolution of a_rev and w via FFT
        size = 1 << ((len(a_rev) + len(w) - 1) - 1).bit_length()
        Ar = np.fft.rfft(a_rev, n=size)
        Wr = np.fft.rfft(w, n=size)
        conv_aw = np.fft.irfft(Ar * Wr, n=size)
        # extract middle n entries
        start = len(a) - 1
        grad = 2.0 * conv_aw[start:start + n]

        # step size schedule
        eta = lr0 / math.sqrt(iteration)

        # exponentiated gradient descent
        a = a * np.exp(-eta * grad)

        # projection onto simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

                best_seq = a.copy()

"""
Optimisation of a non‑negative step function to minimise
evaluate_sequence.  The algorithm proceeds in three phases:
1.  A quick sweep of analytic families (uniform, arcsine, β‑,
    edge‑weighted, mixtures) for a few lengths; the best candidates
    are kept.
2.  For the top seeds a smooth‑max (log‑sum‑exp) relaxation of the
    max‑convolution objective is minimised by exponentiated‑gradient
    descent with an annealing schedule on the soft‑max temperature.
    The gradient is obtained efficiently with FFT‑based correlations.
3.  A cheap greedy pairwise mass‑transfer walk (the “swap” step)
    polishes the best vector found.
All updates keep the vector on the probability simplex.
The whole search respects the supplied time budget (default 1000 s).
"""

import math

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT of appropriate length."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective value 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base

) -> tuple[np.ndarray, float]:
    """Exponential‑gradient descent on a log‑sum‑exp relaxation of max(conv)."""
    n = a.shape[0]
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # 1) self‑convolution
        conv = _fft_self_convolve(a)          # length 2n‑1

        # 2) annealed temperature

            w = np.zeros_like(conv)

        else:

        grad = 2.0 * conv_aw[start:start + n]

        eta = lr0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s == 0.0 or np.isnan(s):

                print(

                )

    return best_seq, best_score

def _pairwise_swap_local(

"""
Optimisation of a non‑negative step function to minimise
evaluate_sequence.  The algorithm proceeds in three phases:

1. **Analytic sweep** – a quick scan of several families (uniform,
   arcsine, symmetric beta, edge‑weighted, mixtures with uniform) for a
   handful of lengths.  The best few candidates are kept.

2. **Smooth‑max gradient descent** – for the top seeds a log‑sum‑exp
   relaxation of the max‑convolution objective is minimised by an
   exponentiated‑gradient (mirror) update.  The temperature (β) is
   annealed from a small to a large value, turning the smooth loss into
   the true max.  Gradients are obtained by an FFT‑based correlation,
   keeping each iteration O(n log n).

3. **Greedy pairwise refinement** – a cheap random mass‑transfer walk
   (pairwise swaps) finishes the search, keeping any improvement.

All updates keep the vector on the probability simplex, and the total
runtime respects the supplied budget (default 1000 s).  The returned
list is ready for `evaluate_sequence`.  The implementation uses only
`numpy`, `math` and the standard library.
"""

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution x*y using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n) - 1).bit_length()
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:n]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()

# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals

    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base
