1️⃣  **Broad parametric sweep** – generate candidates from several
    analytic families (uniform, arcsine, symmetric & asymmetric Beta,

    lengths.  Keep the best (the objective is 2 n · max(a*a) because we
    normalise ∑a=1).

2️⃣  **Adam optimisation on the exact max‑convolution** – we parameterise
    the distribution as a soft‑max of an unconstrained vector
    ``theta`` (a_i = softmax(theta)_i).  The sub‑gradient of the true
    max‑convolution is obtained from the active set of indices attaining
    the maximum.  Adam with bias‑correction updates ``theta``; the
    resulting distribution respects non‑negativity and the sum‑to‑one
    constraint automatically.  This runs as long as the time‑budget
    permits.

    best vector obtained above, repeatedly move a small amount of mass
    from a relatively large entry to a random entry.  Moves that
    increase the objective are occasionally accepted according to a
    temperature schedule, allowing the algorithm to escape shallow

def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    e = np.exp(x - np.max(x))
    return e / e.sum()


def _max_convolution_score(a: np.ndarray) -> float:
    """
    For a vector already normalised to sum = 1 returns
    2·n·max(a*a) – exactly the value produced by `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()


def _grad_max_convolution(a: np.ndarray):
    """
    Return (grad, max_val) where grad = ∂max(a*a)/∂a (sub‑gradient) and
    max_val = max_k (a*a)_k.  The gradient follows from the active set
    of indices attaining the maximum.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    tol = 1e-12 * (abs(max_val) + 1.0)
    active = np.flatnonzero(conv >= max_val - tol)
    w = np.zeros_like(conv)
    w[active] = 1.0
    # grad_i = 2 * sum_j a_j * w_{i+j}
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    return grad, max_val

def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density for numerical stability
    logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1) * np.log(x) + (q - 1) * np.log(1 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        return np.full_like(w, 1.0 / n)
    w /= w.sum()
    return w


def _distribution_edge_dirichlet(n: int, beta_exp: float) -> np.ndarray:
    """
    Dirichlet draw with concentration concentrated at the edges:
    c_i ∝ ((i+0.5)*(n-i-0.5))^{-beta_exp} + 0.1 .
    """
    idx = np.arange(n, dtype=np.float64)
    conc = ((idx + 0.5) * (n - idx - 0.5)) ** (-beta_exp) + 0.1
    vec = np.random.dirichlet(conc)
    return vec


def _distribution_tent(n: int) -> np.ndarray:
    """Symmetric triangular (tent) distribution."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    vals = 1.0 - np.abs(i - center) / center
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals


def _distribution_cosine(n: int) -> np.ndarray:
    """Cosine‑shaped distribution on [0,1]."""
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 - np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 1e-12)
    vals /= vals.sum()
    return vals

        a = _softmax(theta)

        grad_a, cur_max = _grad_max_convolution(a)
        # Sub‑gradient w.r.t. theta (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_theta = a * (grad_a - dot)

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad_theta
        v = beta2 * v + (1.0 - beta2) * (grad_theta ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr / (np.sqrt(v_hat) + eps_adam)

        theta = theta - step * m_hat

        # Occasionally evaluate the true objective and keep the best.
        if (t % max(1, max_iters // 20) == 0) or (t == max_iters):
            cur_score = 2.0 * n * cur_max
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] Adam it={t}/{max_iters} new best {best_score:.6f}")

        # Early‑exit if we are out of time
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4️⃣  Simulated‑annealing mass‑transfer polishing
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    temperature = 1e-3
    decay = 0.9995
    iter_sa = 0

    # Pre‑compute cumulative distribution for source selection (probabilistic)
    while time.time() < deadline:
        iter_sa += 1
        # Choose a source index proportional to its mass (larger entries donate)
        src = np.searchsorted(np.cumsum(a), random.random())
        # Choose a target uniformly (could be the same; retry)
        tgt = random.randrange(n)
        if src == tgt:
            continue

        # Propose a small transfer; magnitude scales with source mass
        max_delta = min(a[src] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta

        # Numerical safety
        a_candidate = np.maximum(a_candidate, eps)
        a_candidate /= a_candidate.sum()

        cand_score = _max_convolution_score(a_candidate)

        # Metropolis acceptance
        if cand_score < cur_score or random.random() < math.exp(-(cand_score - cur_score) / temperature):
            a = a_candidate
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time():.1f}] SA it={iter_sa} new best {best_score:.6f}")

        # Cool down
        temperature *= decay
        # Prevent temperature from vanishing completely
        if temperature < 1e-9:
            temperature = 1e-9

        # Optional early break if temperature is tiny and no improvement for long
        if iter_sa % 5000 == 0 and temperature < 1e-7:
            # a quick check: try a random jitter
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            cur_score = _max_convolution_score(a)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain list)
    # ------------------------------------------------------------------
    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"[{time.time():.1f}] Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
