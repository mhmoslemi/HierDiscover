"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm consists of three stages:

1. **Rich seeding** – families of analytic shapes (arcsine, beta,
   Gaussian, exponential, power‑transforms) are generated for several
   lengths.  For each base shape we augment the two outermost coefficients
   with a tunable amount of mass (parameter λ) and we also form convex
   combinations of two shapes.  All candidates are ℓ₁‑normalised and
   evaluated with the fast surrogate 2·n·max(conv).

2. **Intensive local optimisation** – the best seeds are refined by a
   long run of exponentiated‑gradient descent (mirror descent on the
   simplex) followed by Adam and a plain projected sub‑gradient method.
   The loops stop automatically when the global deadline (≈ 1000 s)
   is reached.

3. **Global polishing** – a random hill‑climber (mass transfer) and a
   Frank‑Wolfe style pairwise transfer are applied to the current best
   vector.  All steps respect the supplied time budget.

The code uses only `numpy` and the provided `evaluate_sequence` (via the
surrogate).  It returns a list of floats ready for `evaluate_sequence`.
"""

import time

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a (length 2·n‑1) via a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = rfft(a, n=size)
    conv = irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using a real FFT."""
    size = 1 << ((a.size + b.size - 1) - 1).bit_length()
    A = rfft(a, n=size)
    B = rfft(b, n=size)
    conv = irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
def _max_convolution_score(a: np.ndarray) -> float:
    """Exact surrogate for `evaluate_sequence` when Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


def _subgrad(a: np.ndarray):
    """
    Exact sub‑gradient of the max‑convolution functional on the simplex.
    Returns (gradient, current_max, active_indices).
    """
    n = a.shape[0]
    b = _fft_self_convolve(a)
    cur_max = b.max()
    eps = 1e-12 * (abs(cur_max) + 1.0)
    active = np.flatnonzero(b >= cur_max - eps)

    # weight for each active convolution index
    w = np.zeros_like(b)
    w[active] = 1.0

    # convolution of reversed a with w gives raw gradient
    conv_rev = _fft_convolve(a[::-1], w)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]

    # correct diagonal terms for even active indices (a_i² contributions)
    even_mask = (active % 2 == 0)
    diag_idx = (active[even_mask] // 2)
    if diag_idx.size > 0:
        grad[diag_idx] -= a[diag_idx]

    return grad, cur_max, active

# Analytic families ----------------------------------------------------
def _arcsine_distribution(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _triangular_distribution(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals = vals.astype(np.float64)
    vals /= vals.sum()
    return vals


def _exponential_distribution(n: int, lam: float = 0.005) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _gaussian_distribution(n: int, sigma_factor: float = 0.15) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    w /= w.sum()
    return w


def _cauchy_distribution(n: int, scale_factor: float = 0.15) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    scale = scale_factor * n
    w = 1.0 / (1.0 + ((idx - centre) / scale) ** 2)
    w /= w.sum()
    return w


def _beta_symmetric_distribution(n: int, p: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Unsymmetric Beta(α,β) on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    with np.errstate(divide='ignore', invalid='ignore'):
        w = np.exp((alpha - 1) * np.log(x) + (beta - 1) * np.log(1 - x))
    if np.isnan(w).any() or np.isinf(w).any() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _random_dirichlet_alpha(n: int, alpha: float) -> np.ndarray:
    """Draw a random point on the simplex from Dirichlet(α,…,α)."""
    if alpha <= 0:
        raise ValueError("alpha must be positive")
    x = np.random.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def order_alternate_extremes(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    left, right = 0, n - 1
    toggle = True
    for v in desc_vals:
        if toggle:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
        toggle = not toggle
    return out


def order_center_outwards(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    mid = (n - 1) // 2
    pos = [mid]
    for d in range(1, n):
        l = mid - d
        if l >= 0:
            pos.append(l)
        r = mid + d
        if r < n:
            pos.append(r)
    for i, p in enumerate(pos[:n]):
        out[p] = desc_vals[i]
    return out


def order_binary_tree(desc_vals: np.ndarray) -> np.ndarray:
    n = desc_vals.shape[0]
    out = np.empty(n, dtype=desc_vals.dtype)
    positions = []

    def recurse(l: int, r: int):
        if l > r:
            return
        m = (l + r) // 2
        positions.append(m)
        recurse(l, m - 1)
        recurse(m + 1, r)

    recurse(0, n - 1)
    out[positions] = desc_vals
    return out

def _exp_gradient_descent(a0: np.ndarray, deadline: float,
                          max_iters: int = 1_000_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Mirror descent (multiplicative) on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / s
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _adam_descent(a0: np.ndarray, deadline: float,
                  max_iters: int = 500_000,
                  step0: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Adam optimiser with simplex projection."""
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        grad, _, _ = _subgrad(a)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        eta = step0 / (np.sqrt(v_hat) + 1e-8)
        a = a - eta * m_hat
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def _sgd_subgrad(a0: np.ndarray, deadline: float,
                 max_iters: int = 500_000,
                 step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Plain projected sub‑gradient descent."""
    a = a0.copy()
    best_a = a.copy()
    best_score = _max_convolution_score(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = _project_onto_simplex(a, s=1.0)
        cur_score = _max_convolution_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def _random_mass_transfer(a: np.ndarray, epsilon: float = 0.02) -> np.ndarray:
    n = a.shape[0]
    src = np.random.randint(n)
    dst = np.random.randint(n)
    while dst == src or a[src] < 1e-12:
        src = np.random.randint(n)
        dst = np.random.randint(n)
    delta = min(a[src] * epsilon, 1.0 - a[dst])
    if delta <= 0:
        return a
    b = a.copy()
    b[src] -= delta
    b[dst] += delta
    return b


def _hill_climb(a: np.ndarray, deadline: float,
                max_iters: int = 300_000, epsilon: float = 0.02) -> tuple[np.ndarray, float]:
    best = a.copy()
    best_score = _max_convolution_score(best)
    n = best.size
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best[i] < 1e-12:
            continue
        delta = min(best[i] * epsilon, 1.0 - best[j])
        if delta <= 0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = _max_convolution_score(cand)
        if sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


# ----------------------------------------------------------------------
# Frank‑Wolfe style polishing -------------------------------------------
def _frank_wolfe_opt(a0: np.ndarray, deadline: float,
                    max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    cur_score = _max_convolution_score(a)
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        grad, _, _ = _subgrad(a)
        i_src = int(np.argmax(grad))
        i_dst = int(np.argmin(grad))
        if i_src == i_dst:
            break
        delta = min(a[i_src], 1.0 - a[i_dst])
        if delta <= 0:
            break
        step = 0.7 * delta
        a[i_src] -= step
        a[i_dst] += step
        cur_score = _max_convolution_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        if it % 4000 == 0 and np.random.rand() < 0.1:
            a = _project_onto_simplex(a + np.random.randn(a.size) * 1e-5, s=1.0)
            cur_score = _max_convolution_score(a)
    return best, best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget
    (≈ 1000 s) and returns the best vector discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose a working length.  4096 is a nice power‑of‑two for FFTs.
    # We also explore a few neighbouring lengths later (see seeding loop).
    base_len = 4096

    # ------------------------------------------------------------------
    # Helper to add a candidate safely
    candidates: list[tuple[np.ndarray, float]] = []

    def _add_candidate(arr: np.ndarray):
        a = np.asarray(arr, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        cand_score = _max_convolution_score(a)
        candidates.append((a, cand_score))

    # ------------------------------------------------------------------
    # 1. Broad seeding – analytic families + edge‑mass augmentation
    lengths = set([base_len, int(base_len * 0.8), int(base_len * 1.2),
                   int(base_len * 0.6), int(base_len * 1.5)])
    # keep them reasonable
    lengths = sorted([n for n in lengths if 800 <= n <= 12000])

    for n in lengths:
        if time.time() >= deadline:
            break

        # basic families ------------------------------------------------
        fams = []

        fams.append(_arcsine_distribution(n))
        fams.append(_triangular_distribution(n))
        fams.append(_exponential_distribution(n, lam=0.005))
        fams.append(_gaussian_distribution(n, sigma_factor=0.15))
        fams.append(_cauchy_distribution(n, scale_factor=0.15))

        # symmetric beta with a few representative p-values
        for p in (0.25, 0.5, 0.75, 1.0, 1.5):
            fams.append(_beta_symmetric_distribution(n, p))

        # power‑transformed arcsine
        base_arc = _arcsine_distribution(n)
        for pwr in (0.8, 1.2, 1.5):
            a = np.power(base_arc, pwr)
            a /= a.sum()
            fams.append(a)

        # unsymmetric beta grid (coarse)
        for alpha in (0.5, 1.0, 2.0):
            for beta in (0.5, 1.0, 2.0):
                fams.append(_beta_distribution(n, alpha, beta))

        # --- edge‑mass augmentation ------------------------------------
        for f in fams:
            # raw family
            _add_candidate(f)

            # try λ‑augmentation on the two outer coefficients
            for lam in np.linspace(0.0, 0.25, 13):  # 0, 0.02, …, 0.24
                if lam * 2 > 1.0:
                    continue
                a = (1.0 - 2.0 * lam) * f
                a[0] += lam
                a[-1] += lam
                # sum stays 1 by construction
                _add_candidate(a)

        # ----------------------------------------------------------------
        # convex combinations of two families (coarse grid)
        for i in range(len(fams)):
            for j in range(i + 1, len(fams)):
                for alpha in np.linspace(0.0, 1.0, 6):  # 0,0.2,…,1
                    a = alpha * fams[i] + (1.0 - alpha) * fams[j]
                    a = a / a.sum()
                    _add_candidate(a)

        # ----------------------------------------------------------------
        # Random Dirichlet draws (moderate amount)
        for _ in range(80):
            a = _random_dirichlet_alpha(n, np.random.uniform(0.2, 2.0))
            _add_candidate(a)

    # ------------------------------------------------------------------
    # 2. Inject previously known best (if any)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        prev = np.maximum(prev, 0.0)
        if prev.sum() > 0:
            prev = prev / prev.sum()
            _add_candidate(prev)
            print(f"Added previous best, score={candidates[-1][1]:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Keep the top seeds
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top_candidates = candidates[:top_k]
    print(f"Generated {len(candidates)} candidates, keeping top {top_k}.")
    print(f"Best seed surrogate = {top_candidates[0][1]:.6f}")

    best_seq = top_candidates[0][0].copy()
    best_score = top_candidates[0][1]

    # ------------------------------------------------------------------
    # 3. Intensive refinement – run the three descent methods in turn.
    #    Each routine respects `deadline`, therefore the total time is
    #    automatically split.
    for idx, (seed_vec, _) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # exponentiated‑gradient (most effective for simplex problems)
        seq, sc = _exp_gradient_descent(seed_vec, deadline,
                                        max_iters=4_000_000,
                                        eta0=0.8)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"[{idx+1}/{len(top_candidates)}] ExpGrad improved → {best_score:.6f}")

        # Adam
        if time.time() < deadline:
            seq, sc = _adam_descent(best_seq, deadline,
                                    max_iters=800_000,
                                    step0=0.2)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{idx+1}/{len(top_candidates)}] Adam improved   → {best_score:.6f}")

        # SGD
        if time.time() < deadline:
            seq, sc = _sgd_subgrad(best_seq, deadline,
                                   max_iters=600_000,
                                   step0=0.5)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[{idx+1}/{len(top_candidates)}] SGD improved    → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Global stochastic polishing ------------------------------------
    if time.time() < deadline - 30:
        seq, sc = _hill_climb(best_seq, deadline,
                              max_iters=400_000,
                              epsilon=0.025)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Hill‑climb improvement → {best_score:.6f}")

    if time.time() < deadline - 20:
        seq, sc = _frank_wolfe_opt(best_seq, deadline,
                                   max_iters=400_000)
        if sc < best_score:
            best_seq, best_score = seq.copy(), sc
            print(f"Frank‑Wolfe polishing → {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"Final best surrogate score: {best_score:.6f} (length={len(best_seq)})")
    # The surrogate assumes ℓ₁‑normalisation; `evaluate_sequence` is
    # scale‑invariant, so we may return the normalised vector directly.
    return best_seq.tolist()
