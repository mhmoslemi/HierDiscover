def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    best_n = None

    # family scanning
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000]
    for n in lengths:
        if time.time() >= deadline: break
        families = []
        families.append(_distribution_uniform(n))
        families.append(_distribution_arcsine(n))
        families.append(_distribution_triangular(n))
        families.append(_distribution_exponential(n, lam=0.005))
        families.append(_distribution_exponential(n, lam=0.02))
        for p in [0.3, 0.6, 1.0, 2.0, 4.0, 8.0]:
            families.append(_distribution_beta_sym(n, p))
        for pw in [0.2,0.5,1.0,2.0,4.0,6.0]:
            families.append(_distribution_sine_power(n, pw))
        for sigma in [0.15, 0.25,0.4]:
            families.append(_distribution_gaussian(n, sigma))
        # random draws
        for alpha in [0.2, 0.5, 1.0, 2.0]:
            for _ in range(2):
                families.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate
        for vec in families:
            if time.time() >= deadline: break
            vec = np.maximum(vec, EPS)
            vec /= vec.sum()
            sc = _score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} score={sc:.6f}")

    # if not found fallback uniform
    if best_seq is None:
        n=3000
        best_seq = _distribution_uniform(n)
        best_n=n
        best_score=_score(best_seq)

    # refinement of best_seq
    best_seq, best_score = _refine(best_seq, best_n, best_seq, best_score, deadline)

"""
Propose a step‑function sequence that minimizes
    2·n·max(conv(a,a)) / (sum(a)^2)

The search consists of three stages:

1. **Broad scan** – Several analytic families (uniform, arcsine, triangular,
   exponential, beta, sine‑power, Gaussian) are sampled for a handful of
   lengths together with random Dirichlet draws.  The best vector found
   (normalised to sum = 1) is taken as a seed.

2. **Intensive refinement** – The seed is optimised with a pipeline:
   * Exponential‑gradient descent on a smooth‐max surrogate (α increases,
     learning‑rate decays).
   * Projected‑gradient descent on the (sub‑)gradient of the true max‑convolution.
   * Adam optimisation on the logits of a soft‑max parametrisation.
   * A simple hill‑climber that transfers a tiny mass from the worst to the best
     index (with periodic jitter).
   All updates enforce non‑negativity, ℓ₁‑normalisation and optional symmetry.

3. **Random restarts** – While time permits new random Dirichlet starts are
   generated (different lengths) and refined with a shortened version of the
   pipeline.  The globally best sequence found is returned.

The implementation uses FFT‑based convolutions for O(n log n) per gradient
evaluation and respects the 1000 s wall‑clock budget.  No closures or lambdas
are used, and all helpers are defined at the top level.
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x):
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a):
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a, b):
    """Full linear convolution of two real vectors."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _score(vec):
    """Objective value for a vector already normalised to sum = 1."""
    n = vec.shape[0]
    conv = _fft_self_convolve(vec)
    return 2.0 * n * conv.max()


def _smooth_weights(conv, alpha):
    """
    Soft‑max over convolution entries.
    Returns a probability vector that approximates the smooth max.
    """
    maxc = conv.max()
    w = np.exp(alpha * (conv - maxc))
    s = w.sum()
    if s == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _subgradient_weights(conv, eps_rel=1e-12):
    """Weights for a (sub)gradient of the max‑convolution."""
    m = conv.max()
    w = np.zeros_like(conv)
    w[conv >= m - eps_rel * (abs(m) + 1.0)] = 1.0
    cnt = w.sum()
    if cnt == 0:
        return w
    return w / cnt


def _exponential_gradient_step(a, lr, alpha):
    """One EG step (a ≥ 0, Σa = 1)."""
    conv = _fft_self_convolve(a)
    w = _smooth_weights(conv, alpha)
    grad = 2.0 * _fft_convolve(a[::-1], w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a * np.exp(-lr * grad)
    a = _project_onto_simplex(a, 1.0)
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()
    return a


def _projected_gradient_step(a, eta, active_w):
    """PGD step on the (sub)gradient defined by active_w."""
    grad = 2.0 * _fft_convolve(a[::-1], active_w)[len(a) - 1 : len(a) - 1 + len(a)]
    a = a - eta * grad
    a = _project_onto_simplex(a, 1.0)
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()
    return a


def _adam_step(logits, m, v, t, grad):
    """Adam update on logits (unconstrained)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = beta1 * m + (1.0 - beta1) * grad
    v = beta2 * v + (1.0 - beta2) * (grad ** 2)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    lr = 0.02 / (np.sqrt(v_hat) + eps)
    logits = logits - lr * m_hat
    logits = np.clip(logits, -30, 30)
    return logits, m, v


def _hillclimber_move(a, conv, eps=1e-12):
    """
    Move a tiny chunk of mass from an index that contributes most to the active
    max entry to one that contributes least.
    """
    n = a.shape[0]
    w = _subgradient_weights(conv, eps_rel=eps)
    grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]
    src = int(np.argmax(grad))
    dst = int(np.argmin(grad))
    if src == dst:
        dst = random.randint(0, n - 1)
    max_delta = min(a[src] * 0.5, 0.02)
    delta = max_delta * random.random()
    if delta < eps:
        return a, conv, False
    a_new = a.copy()
    a_new[src] -= delta
    a_new[dst] += delta
    a_new = np.maximum(a_new, eps)
    a_new /= a_new.sum()
    new_conv = _fft_self_convolve(a_new)
    improved = new_conv.max() < conv.max() - 1e-13
    if improved:
        return a_new, new_conv, True
    else:
        return a, conv, False

# Analytic families – all return a probability vector (sum = 1)
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.01):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1 - x))
    if w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found within the allotted time.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # 1. Scan a variety of analytic families and random Dirichlet draws.
    scan_lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000]
    for n in scan_lengths:

        candidates.append(_distribution_uniform(n))
        candidates.append(_distribution_arcsine(n))
        candidates.append(_distribution_triangular(n))
        candidates.append(_distribution_exponential(n, lam=0.005))
        candidates.append(_distribution_exponential(n, lam=0.02))

        for p in (0.3, 0.6, 1.0, 2.0, 4.0, 8.0):
            candidates.append(_distribution_beta_sym(n, p))

        for pw in (0.2, 0.5, 1.0, 2.0, 4.0, 6.0):
            candidates.append(_distribution_sine_power(n, pw))

        for sigma in (0.15, 0.25, 0.4):
            candidates.append(_distribution_gaussian(n, sigma))

        # random Dirichlet draws (promote sparsity with low concentration)
        for alpha in (0.2, 0.5, 1.0, 2.0):
            for _ in range(2):
                candidates.append(np.random.dirichlet(np.full(n, alpha)))

        # evaluate all candidates
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. If a previously found global solution exists, try it.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global best {sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback uniform if nothing was found.
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_score = _score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine the current best vector (full optimisation pipeline).
    best_seq, best_score = _refine(best_seq, best_n, deadline, short=False)

    # ------------------------------------------------------------------
    # 5. Multiple random restarts while time permits (shorter refinements).
    restart_lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000, 6000]
    while time.time() < deadline - 5.0:
        n = random.choice(restart_lengths)
        a0 = np.random.dirichlet(np.full(n, 0.5))
        a0 = np.maximum(a0, eps)
        a0 /= a0.sum()
        cand_seq, cand_sc = _refine(a0, n, deadline, short=True)
        if cand_sc < best_score:
            best_seq, best_score, best_n = cand_seq.copy(), cand_sc, n
            print(f"[{time.time():.1f}] restart n={n} new best {cand_sc:.6f}")

    # ------------------------------------------------------------------
    final_score = _score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
