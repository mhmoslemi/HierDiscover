#!/usr/bin/env python
"""
search for a non‑negative step function that minimises
`evaluate_sequence`.  The pipeline is:

1.  Generate a diverse pool of candidate distributions from analytic
    families (uniform, arcsine, symmetric β, edge‑weighted) and from
    random Dirichlet / sparse vectors for several lengths.
2.  Keep the best few seeds (simplex‑projected, sum = 1).
3.  Refine each seed with **smooth‑max gradient descent**:
       – a softmax over the self‑convolution values provides a
         differentiable approximation of the max;
       – the gradient is obtained by a single FFT‑based cross‑correlation;
       – exponentiated‑gradient (mirror descent) keeps the iterates on
         the probability simplex.
    The smoothing parameter α is slowly increased while the stepsize
    η decays as 1/√t.
4.  A cheap greedy pairwise‑mass‑transfer walk (the “swap” routine)
    polishes the best vector found.

The whole search respects the 1000 s time budget; intermediate
progress and current best scores are printed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of a and b (both 1‑D real) using a real FFT."""
    n = a.shape[0]
    m = b.shape[0]
    size = 1 << ((n + m - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: n + m - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with Σa = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density to avoid overflow for large p
    logw = (p - 1) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(logw - np.max(logw))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Gradient of the smoothed max (soft‑max) --------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights that sum to 1, concentrated on large conv entries."""
    max_val = conv.max()
    # shift for numerical stability
    exp_vals = np.exp(alpha * (conv - max_val))
    s = exp_vals.sum()
    if s == 0.0:                     # should never happen, but be safe
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


def _grad_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇ ( Σ_k w_k conv_k(a) ) where conv_k = (a*a)_k.
    Equivalent to 2 * cross‑correlation of a with w.
    """
    n = a.shape[0]
    w_rev = w[::-1]
    conv = _fft_convolve(a, w_rev)            # length = 3n‑2
    offset = 2 * n - 2                         # = len(w)‑1
    grad = 2.0 * conv[offset - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Smooth‑max mirror descent -----------------------------------------------
def _smooth_max_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    alpha_start: float = 5.0,
    alpha_inc: float = 0.5,
    eval_interval: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient minimisation of the smooth max of the
    self‑convolution.  The smoothing parameter α is increased
    slowly to tighten the approximation.
    """
    t = 0
    alpha = alpha_start
    n = a.shape[0]

    while time.time() < deadline:
        t += 1

        # current convolution and smooth weights
        conv = _fft_self_convolve(a)
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth objective
        grad = _grad_from_weights(a, w)

        # stepsize schedule (mirror‑descent on the simplex)
        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # simplex projection (ensures Σa = 1, a_i ≥ 0)
        a = _project_onto_simplex(a)

        # occasionally evaluate the true objective
        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(f"[{time.time():.1f}] it={t} α={alpha:.2f} new global best {best_score:.6f}")

        # increase smoothing slowly (helps escaping plateaus)
        if t % 500 == 0:
            alpha = min(alpha + alpha_inc, 100.0)

        # safety break
        if t >= 200_000:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer -------------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a little mass; keep the move only if the objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score
