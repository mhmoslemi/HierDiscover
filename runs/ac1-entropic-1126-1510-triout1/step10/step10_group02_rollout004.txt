"""
Propose a non‑negative step function (list of floats) that minimises the
objective defined by ``evaluate_sequence``.  

The algorithm consists of three phases:

1. **Broad analytic sweep** – For a handful of lengths we evaluate a
   large collection of parametric families (uniform, arcsine, beta,
   exponential, power‑law, sine‑power, cosine window, Gaussian,
   triangular).  For each family a modest grid of parameters is tried.
   The best few (≈10) candidates are kept in an elite list.

2. **Continuous refinement** – Each elite vector is optimised by three
   successive descent stages:
      * *Smooth‑max mirror descent* – a soft‑max approximation of the
        ``max`` is minimised by exponentiated‑gradient updates while the
        temperature α is increased from 1 to 1e6.  This yields a fast
        global‑search phase.
      * *Projected sub‑gradient polishing* – the true sub‑gradient of the
        max‑convolution is used together with an Adam‑style momentum and
        back‑tracking line search.
      * *Targeted hill‑climber* – mass is transferred from the pair that
        contributes most to the current maximal convolution index to a
        location that has the smallest impact on that index; occasional
        pruning of tiny entries and jitter keep the search alive.

3. **Final sanity & return** – the best vector found is projected onto
   the simplex, clipped to be non‑negative and normalised, and returned
   as a plain Python list.

All heavy work respects the 1000 s time budget (2 s safety margin)
and prints intermediate scores.
"""

# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) via real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _objective(a: np.ndarray) -> float:
    """
    Returns the value exactly produced by evaluate_sequence for a vector
    that already satisfies sum(a)=1 (i.e. 2·n·max(conv)).
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # protect against p<1 where log may hit -inf at 0/1
    with np.errstate(divide='ignore', invalid='ignore'):
        logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _add_to_elite(elite, score, vec, max_size=12):
    """Maintain a sorted elite list of (score, vector) of bounded size."""
    if len(elite) < max_size:
        elite.append((score, vec.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if score < elite[-1][0]:
            elite[-1] = (score, vec.copy())
            elite.sort(key=lambda x: x[0])
    return elite


def _evaluate_and_store(vec, elite, best):
    """Normalise ``vec`` safely, evaluate and possibly update elite / best."""
    eps = 1e-12
    vec = np.maximum(vec, eps)
    vec /= vec.sum()
    sc = _objective(vec)
    # update global best
    if sc < best[0]:
        best = (sc, vec.copy())
        print(f"[{time.time():.1f}] New global best {sc:.6f}")
    # update elite
    elite = _add_to_elite(elite, sc, vec)
    return elite, best


def _generate_candidates_for_length(n, elite, best, deadline):
    """Create candidates for a given length and update elite / best."""
    # basic families
    families = [
        _distribution_uniform,
        _distribution_arcsine,
        _distribution_triangular,
    ]

    # families with parameters (list of parameter values)
    param_families = [
        (_distribution_exponential, [0.001, 0.003, 0.01, 0.03]),
        (_distribution_beta_sym, [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 1.5, 2.0, 3.0]),
        (_distribution_sine_power, [0.5, 1.0, 2.0, 4.0, 8.0]),
        (_distribution_cosine_window, [0.3, 0.5, 0.7, 0.9]),
        (_distribution_gaussian, [0.10, 0.15, 0.20, 0.30]),
        (_distribution_powerlaw, [1.0, 1.5, 2.0, 2.5, 3.0]),
    ]

    # unconditional families
    for f in families:
        if time.time() >= deadline:
            return elite, best
        vec = f(n)
        elite, best = _evaluate_and_store(vec, elite, best)

    # families with parameters
    for f, plist in param_families:
        for p in plist:
            if time.time() >= deadline:
                return elite, best
            vec = f(n, p)
            elite, best = _evaluate_and_store(vec, elite, best)

    # reversed power‑law (asymmetric)
    for p in [1.5, 2.0, 2.5]:
        if time.time() >= deadline:
            return elite, best
        base = _distribution_powerlaw(n, p)
        vec = base[::-1]
        elite, best = _evaluate_and_store(vec, elite, best)

    # random Dirichlet draws (different concentration)
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        if time.time() >= deadline:
            return elite, best
        vec = np.random.dirichlet(np.full(n, alpha))
        elite, best = _evaluate_and_store(vec, elite, best)

    # sparse start (few non‑zeros)
    k = max(2, n // 300)
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    elite, best = _evaluate_and_store(sparse, elite, best)

    # shuffled versions of a few bases to add asymmetry
    for base in (elite[0][1], elite[1][1] if len(elite) > 1 else elite[0][1]):
        if time.time() >= deadline:
            return elite, best
        shuf = base.copy()
        np.random.shuffle(shuf)
        elite, best = _evaluate_and_store(shuf, elite, best)

    return elite, best

def _smooth_max_mirror_descent(a0, deadline):
    """Exponentiated‑gradient (mirror) descent on a smooth‑max of the convolution."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    total_steps = 20000
    alpha_start, alpha_end = 1.0, 1e6
    lr_start, lr_end = 0.5, 0.0005

    best_a = a.copy()
    best_sc = _objective(best_a)

    for it in range(total_steps):
        if time.time() >= deadline:
            break
        # schedule in log‑space
        t = it / total_steps
        alpha = np.exp(np.log(alpha_start) + (np.log(alpha_end) - np.log(alpha_start)) * t)
        lr = np.exp(np.log(lr_start) + (np.log(lr_end) - np.log(lr_start)) * t)

        conv = _fft_self_convolve(a)
        maxc = conv.max()
        # soft‑max weights
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (via convolution)
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # multiplicative update + projection
        a = a * np.exp(-lr * grad)
        a = _project_onto_simplex(a, 1.0)
        a = np.maximum(a, eps)
        a /= a.sum()

        if it % 500 == 0:
            sc = _objective(a)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_a = a.copy()
                print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a0, deadline):
    """Exact sub‑gradient descent with Adam‑style momentum and back‑tracking."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0

    best_a = a.copy()
    best_sc = _objective(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1:n - 1 + n]

        # Adam‑like update
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line search
        step = step0
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _objective(cand)
            if cand_sc < best_sc - 1e-14 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_sc - 1e-14:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter to escape plateaus
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

    return best_a, best_sc


def _hill_climb_prune(a0, deadline):
    """Targeted hill‑climbing + support pruning."""
    a = a0.copy()
    n = a.shape[0]
    eps = 1e-12

    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- mass‑transfer step ---------------------------------------
        k = int(conv.argmax())                # current worst index
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # find the pair (i,j) with the largest product contributing to k
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        # pick source = larger of the two entries
        source = best_i if a[best_i] >= a[best_j] else best_j

        # choose a target that least harms the current max index
        # impact on k from adding δ to t is δ * a[k‑t] (if in range)
        idx = np.arange(n)
        mask = (k - idx >= 0) & (k - idx < n)
        impact = np.where(mask, a[k - idx], 0.0)
        # avoid picking the source itself
        impact[source] = np.inf
        target = int(np.argmin(impact))

        # propose a modest transfer
        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta > eps:
            a2 = a.copy()
            a2[source] -= delta
            a2[target] += delta
            a2 = np.maximum(a2, eps)
            a2 /= a2.sum()
            new_M = _fft_self_convolve(a2).max()
            if new_M < M - 1e-14:
                a = a2
                conv = _fft_self_convolve(a)
                M = new_M
                best_a = a.copy()
                best_sc = 2.0 * n * M
                print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ---- periodic pruning of tiny entries -----------------------
        if it % 1500 == 0:
            # keep the largest fraction of entries
            for keep_frac in (0.99, 0.97, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx_keep = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx_keep] = a[idx_keep]
                a_pruned /= a_pruned.sum()
                sc_pruned = _objective(a_pruned)
                if sc_pruned < best_sc - 1e-14:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ---- jitter to keep the search alive -------------------------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered as a plain Python list.
    """
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 2.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    elite = []      # list of (score, vector) – sorted ascending

    # ------------------------------------------------------------------
    # 1. Broad analytic sweep (different lengths)
    lengths = [1000, 2000, 4000, 8000, 16000]
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[{time.time():.1f}] Exploring length {n}")
        elite, (best_score, best_seq) = _generate_candidates_for_length(
            n, elite, (best_score, best_seq), deadline)

    # ------------------------------------------------------------------
    # 2. If a previous solution exists, add it to the elite pool
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _objective(prev)
            if sc < best_score:
                best_score, best_seq = sc, prev.copy()
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            elite = _add_to_elite(elite, sc, prev)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback in the unlikely case we have nothing
    if best_seq is None:
        n = 4000
        best_seq = _distribution_uniform(n)
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)
        elite = _add_to_elite(elite, best_score, best_seq)
        print(f"[{time.time():.1f}] Fallback uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(5, len(elite))
    for idx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[idx][1].copy()
        print(f"[{time.time():.1f}] Restart {idx + 1}/{restarts} (len={a0.shape[0]}) score {elite[idx][0]:.6f}")

        # smooth‑max mirror descent
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # projected sub‑gradient polishing
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # final hill‑climber + pruning
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-14:
            best_score, best_seq = sc_mid, a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity (project onto simplex, clip, renormalise)
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _objective(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
