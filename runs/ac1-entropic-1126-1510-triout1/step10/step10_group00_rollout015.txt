assistantfinal**Solution Overview**

The task is to return a non‑negative vector `a` (the heights of a step function) that minimises  

\[
F(a)=\frac{2\,n\;\max(a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the full self‑convolution of `a`.  
The value of `F` does **not** change under a uniform scaling of `a`, therefore we may restrict ourselves to vectors that lie on the probability simplex (`∑a_i = 1`).  

The programme below builds a diverse pool of analytic candidates (uniform, arcsine, β‑ and edge‑weighted families, Dirichlet samples) for several lengths, keeps the most promising ones and then refines them with a *smooth* gradient‑descent that works on the exact objective.  

The key ideas are

* **Smoothing the max** – instead of a sub‑gradient that only looks at the single maximal entry of the convolution we use a soft‑max

  \[
  \phi_\beta(b)=\frac{1}{\beta}\log\!\Big(\sum_k e^{\beta b_k}\Big)
  \]

  whose gradient is a weighted sum of the gradients of all convolution entries.  
  For large `β` the smoothed objective approaches the true `max`.  

* **Fast gradient computation** – both the convolution and the gradient are obtained by FFTs, i.e. in `O(n log n)` time.

* **Multiplicative (exponentiated) updates** – they automatically keep the vector non‑negative; after each step we project back onto the simplex.

* **Annealing on `β`** – we start with a gentle smoothing (`β≈2`) and increase it (`β≈30`). This gives large, stable steps at the beginning and a fine‑grained optimisation at the end.

* **Final local polishing** – a greedy pairwise mass‑transfer walk (the same routine used previously) finishes the search.

All the work respects the 1000 s budget; the code constantly checks the remaining time and stops early if needed.

"""
propose_candidate:  search for a non‑negative step‑function that minimises
evaluate_sequence.  The search consists of

1.  building a rich pool of analytic seeds (uniform, arcsine, symmetric β,
    edge‑weighted, Dirichlet, sparse) for several lengths;
2.  keeping the best few seeds;
3.  refining each seed with a *smooth* gradient descent that uses a
    soft‑max approximation of the max‑convolution, FFT‑based gradients,
    exponentiated‑gradient updates and an annealing schedule for the
    smoothing parameter β;
4.  an optional short Euclidean sub‑gradient phase;
5.  a cheap greedy pairwise‑swap local optimisation.

The routine is fully self‑contained, uses only numpy, limits itself to the
time budget and returns a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    n = a.shape[0]
    b = _fft_self_convolve(a)
    return 2.0 * n * b.max()


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Smooth (soft‑max) gradient descent ------------------------------------
def _smooth_gradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    betas: list[float],
    lr0: float = 0.5,
    eval_interval: int = 250,
) -> tuple[np.ndarray, float]:
    """
    Optimises the exact objective using a soft‑max surrogate.
    Updates are multiplicative (exponentiated‑gradient) and each step is
    projected onto the simplex.
    The list `betas` is traversed from small to large: this implements
    an annealing schedule that first makes big moves and finally focuses
    on the true max‑convolution.
    """
    n = a.shape[0]
    size = 1 << ((3 * n - 2) - 1).bit_length()      # ≥ 3n‑2 for convolution & correlation
    t = 0

    for beta in betas:
        # early exit if time is gone
        if time.time() >= deadline:
            break
        while time.time() < deadline:
            t += 1

            # FFT of the current vector (reused for conv & gradient)
            A = np.fft.rfft(a, n=size)

            # ----- self‑convolution -------------------------------------------------
            conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]

            # ----- soft‑max weights -------------------------------------------------
            max_val = conv.max()
            # exp(beta*(conv-max)) ≤ 1 → no overflow
            w = np.exp(beta * (conv - max_val))
            w_sum = w.sum()
            if w_sum == 0.0:          # numerically impossible, but safe guard
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient of the smoothed objective ------------------------------
            # grad = 2 * (w ⋆ a)   (cross‑correlation, valid mode)
            # correlation can be computed via FFT of w reversed
            W_rev = np.fft.rfft(w[::-1], n=size)
            grad_full = np.fft.irfft(A * W_rev, n=size)          # length = size
            # extract the “valid” part and reverse to obtain w ⋆ a
            segment = grad_full[n - 1 : 2 * n - 1]               # length n, corresponds to indices (n‑1)…(2n‑2)
            grad = 2.0 * segment[::-1]                          # now grad[l] = 2 Σ_i w_{i+l} a_i

            # ----- multiplicative (exponentiated‑gradient) update ------------------
            eta = lr0 / math.sqrt(t)
            a = a * np.exp(-eta * grad)

            # ----- projection onto the simplex ------------------------------------
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a = a / s

            # ----- periodic exact evaluation ---------------------------------------
            if t % eval_interval == 0:
                cur = _objective(a)
                if cur < best_score - 1e-14:
                    best_score = cur
                    best_seq = a.copy()
                    print(
                        f"[{time.time():.1f}] smooth β={beta:.2f} it={t} new best {best_score:.6f}"
                    )
            # safety break: limit number of iterations per β
            if t >= 30000:
                break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Classic sub‑gradient descent (used only as a final touch) ------------
def _subgradient_descent(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.3,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution objective."""
    n = a.shape[0]
    t = 0
    while time.time() < deadline:
        t += 1
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # sub‑gradient from the maximal index
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]

        eta = lr0 / math.sqrt(t)
        a = a * np.exp(-eta * grad)

        # re‑normalise
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / s

        if t % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] subgrad it={t} new best {best_score:.6f}"
                )
        if t >= 150000:
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass transfer -----------------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a small amount of mass; keep move only if objective improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = np.random.choice(src_candidates)

        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score

                    cand = _mix_with_uniform(base, lam)
                    sc = _objective(cand)
                    top_candidates.append((sc, cand.copy()))
                    if sc < overall_best_score:
                        overall_best_score = sc
                        overall_best_seq = cand.copy()
                        print(
                            f"[{time.time():.1f}] new global best len={n} p={p:.3g} lam={lam:.3f} score={sc:.6f}"
                        )

        # ---- random Dirichlet seeds ----------------------------------------
        for _ in range(8):
            if time.time() >= deadline:
                break
            alpha = rng.uniform(0.01, 2.0)
            a = rng.dirichlet(np.full(n, alpha))
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best Dirichlet len={n} score={sc:.6f}")

        # ---- sparse random seeds --------------------------------------------
        for _ in range(5):
            if time.time() >= deadline:
                break
            a = _distribution_random_sparse(n, rng, density=0.01)
            sc = _objective(a)
            top_candidates.append((sc, a.copy()))
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = a.copy()
                print(f"[{time.time():.1f}] new global best sparse len={n} score={sc:.6f}")

        # keep only the most promising seeds so far
        top_candidates.sort(key=lambda x: x[0])
        top_candidates = top_candidates[:30]

    # ------------------------------------------------------------------
    # 2.  If a previous solution exists, inject it.
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1:
            prev = prev / prev.sum()
            sc = _objective(prev)
            if sc < overall_best_score:
                overall_best_score = sc
                overall_best_seq = prev.copy()
                print(f"[{time.time():.1f}] previous best score={sc:.6f}")
            top_candidates.append((sc, prev.copy()))
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3.  Fallback (unlikely) ------------------------------------------------
    if overall_best_seq is None:
        n = 5000
        overall_best_seq = _distribution_uniform(n)
        overall_best_score = _objective(overall_best_seq)
        print(f"[{time.time():.1f}] fallback uniform score={overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4.  Smooth gradient descent on a few top seeds
    remaining = deadline - time.time()
    if remaining > 30.0 and top_candidates:
        seeds = [seq for _, seq in sorted(top_candidates, key=lambda x: x[0])[:5]]
        per_seed = remaining / max(1, len(seeds))
        betas_schedule = [2.0, 5.0, 10.0, 20.0, 30.0]   # annealing

        print(f"[{time.time():.1f}] smooth optimisation on {len(seeds)} seeds")
        for idx, seed_seq in enumerate(seeds):
            if time.time() >= deadline:
                break
            sub_deadline = time.time() + per_seed
            a_opt, sc_opt = _smooth_gradient_descent(
                seed_seq.copy(),
                sub_deadline,
                best_seq=overall_best_seq,
                best_score=overall_best_score,
                betas=betas_schedule,
                lr0=0.6,
                eval_interval=200,
            )
            if sc_opt < overall_best_score:
                overall_best_score = sc_opt
                overall_best_seq = a_opt.copy()
                print(
                    f"[{time.time():.1f}] after smooth seed {idx} new global best {overall_best_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 5.  Optional short Euclidean sub‑gradient phase (fine tuning)
    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.3
        a_opt, sc_opt = _subgradient_descent(
            overall_best_seq.copy(),
            sub_deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
            lr0=0.4,
            eval_interval=150,
        )
        if sc_opt < overall_best_score:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] post‑subgrad new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    # 6.  Final greedy local refinement
    if time.time() < deadline:
        print(f"[{time.time():.1f}] final greedy swaps")
        a_opt, sc_opt = _pairwise_swap_local(
            overall_best_seq.copy(),
            deadline,
            best_seq=overall_best_seq,
            best_score=overall_best_score,
        )
        if sc_opt < overall_best_score:
            overall_best_score = sc_opt
            overall_best_seq = a_opt.copy()
            print(f"[{time.time():.1f}] after swaps new best {overall_best_score:.6f}")

    # ------------------------------------------------------------------
    print(
        f"Final best score: {overall_best_score:.6f} (length={len(overall_best_seq)})"
    )
    # return plain list – evaluate_sequence will handle any further clipping/normalisation
    return overall_best_seq.tolist()
