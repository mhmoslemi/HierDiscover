"""
Propose a non‑negative step function (list of floats) that minimises
`evaluate_sequence`.  The search consists of:

1. **Broad exploration** – generate many analytically‑parameterised
   families (uniform, arcsine, triangular, exponential, beta,
   sine‑power, cosine‑window, Gaussian, power‑law) together with many
   random Dirichlet draws for a handful of lengths.  The best few
   candidates (lowest objective) are kept.

2. **Exponential‑gradient (smooth‑max) mirror descent** – starting from
   each elite candidate we run a softened max‑convolution minimiser.
   The temperature α is increased in several stages up to 2·10⁶ while
   the learning‑rate decays, letting the method first explore the
   simplex and later focus on the true maximum.

3. **Projected sub‑gradient polishing** – true max‑convolution
   sub‑gradients are used together with Adam‑style momentum and a
   back‑tracking line‑search.  This stage runs until the time budget is
   exhausted.

4. **Hill‑climbing + support pruning** – tiny‑mass transfers are
   performed from the pair that currently attains the maximum
   convolution value; periodic pruning discards negligible entries.
   Small random jitter keeps the dynamics alive.

All phases respect a 1000 s wall‑clock budget (a few seconds safety
margin).  The best vector found is returned as a plain Python list.
"""

# Helper utilities -------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·n‑1) using real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Objective value 2·n·max(a*a) for a vector already normalised to sum = 1.
    This equals the return value of `evaluate_sequence`.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return float(2.0 * n * conv.max())

# Analytic families – each returns a vector with sum = 1                --
def _distribution_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n):
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n):
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_exponential(n, lam=0.005):
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n, p):
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Guard against p<1 causing overflow of log(0)
    with np.errstate(divide='ignore', invalid='ignore'):
        logw = (p - 1) * np.log(x) + (p - 1) * np.log(1 - x)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_sine_power(n, power):
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n) ** power
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_cosine_window(n, coeff=0.5):
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    vals = coeff - coeff * np.cos(2.0 * np.pi * i / (n - 1))
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_gaussian(n, sigma):
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-0.5 * ((x - centre) / (sigma * n)) ** 2)
    vals = np.maximum(vals, 0)
    vals /= vals.sum()
    return vals


def _distribution_powerlaw(n, p):
    """Monotone decreasing power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals /= vals.sum()
    return vals

def _generate_candidates_for_length(n):

    Build a list of candidate vectors (already normalised to sum = 1)
    for a given length `n`.  The list is deliberately moderate in size
    to keep the broad‑exploration phase fast.
    """
    cand = []

    # analytic families with a range of parameters
    cand.append(_distribution_uniform(n))
    cand.append(_distribution_arcsine(n))
    cand.append(_distribution_triangular(n))

    # exponential (various decay rates)
    for lam in (0.001, 0.003, 0.01, 0.03, 0.08):
        cand.append(_distribution_exponential(n, lam=lam))

    # symmetric beta (p spanning several decades)
    for p in (0.01, 0.02, 0.04, 0.07, 0.12, 0.2, 0.35, 0.5,
              0.8, 1.0, 1.5, 2.0, 3.0, 5.0, 8.0):
        cand.append(_distribution_beta_sym(n, p))

    # sine‑power
    for pw in (0.5, 1.0, 2.0, 4.0, 8.0, 12.0, 20.0, 30.0):
        cand.append(_distribution_sine_power(n, pw))

    # cosine window
    for coeff in (0.2, 0.35, 0.5, 0.65, 0.8, 0.95):
        cand.append(_distribution_cosine_window(n, coeff))

    # Gaussian
    for sigma in (0.08, 0.12, 0.18, 0.24, 0.32, 0.45):
        cand.append(_distribution_gaussian(n, sigma))

    # Power‑law (both decreasing and its reverse)
    for p in (0.6, 0.8, 1.2, 1.5, 2.0, 3.0, 4.0):
        pl = _distribution_powerlaw(n, p)
        cand.append(pl)
        cand.append(pl[::-1])               # reversed version

    # Random Dirichlet draws (various concentration)
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        vec = np.random.dirichlet(np.full(n, alpha))
        cand.append(vec)

    # Sparse random start (few non‑zeros, controlled sparsity)
    k = max(2, n // 250)                     # about 0.4 % non‑zeros
    sparse = np.zeros(n, dtype=np.float64)
    idx = np.random.choice(n, k, replace=False)
    sparse[idx] = np.random.rand(k)
    sparse /= sparse.sum()
    cand.append(sparse)

    # Random shuffles of a few base families (asymmetry)
    bases = [cand[0], cand[1], cand[4]]
    for base in bases:
        shuf = base.copy()
        np.random.shuffle(shuf)
        cand.append(shuf)

    return cand

# Local optimisation routines --------------------------------------------
def _smooth_max_mirror_descent(a, deadline):
    """
    Exponentiated‑gradient (mirror) descent on the smooth‑max version
    of the objective.  Temperature α is raised in several stages up to
    2·10⁶ while the learning rate decays.
    """
    eps = 1e-12
    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    # (α_start, α_end, steps, lr_start, lr_end)
    schedule = [
        (1.0, 200.0, 2000, 0.5, 0.05),
        (200.0, 2000.0, 2000, 0.05, 0.005),
        (2000.0, 20000.0, 3000, 0.005, 0.0005),
        (20000.0, 200000.0, 3000, 0.0005, 5e-05),
        (200000.0, 2000000.0, 4000, 5e-05, 5e-06),
        (2000000.0, 20000000.0, 4000, 5e-06, 5e-07),
    ]

    for (alpha0, alpha1, steps, lr0, lr1) in schedule:
        if time.time() >= deadline:
            break
        for it in range(steps):
            if time.time() >= deadline:
                break
            t = it / steps
            alpha = alpha0 * (alpha1 / alpha0) ** t
            lr = lr0 * (lr1 / lr0) ** t

            conv = _fft_self_convolve(a)
            maxc = conv.max()
            # soft‑max weights
            w = np.exp(alpha * (conv - maxc))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smooth‑max
            conv_rev = _fft_convolve(a[::-1], w)
            grad = 2.0 * conv_rev[len(a) - 1: len(a) - 1 + len(a)]

            a = a * np.exp(-lr * grad)          # mirror‑step
            a = _project_onto_simplex(a, 1.0)    # stay on simplex
            a = np.maximum(a, eps)              # guard against zero
            a /= a.sum()

            if it % 1000 == 0:
                sc = _max_convolution_score(a)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_a = a.copy()
                    print(f"[{time.time():.1f}] Mirror it={it} new best {sc:.6f}")

    return best_a, best_sc


def _projected_subgradient(a, deadline):
    """
    True sub‑gradient descent with Adam‑style momentum and a simple
    back‑tracking line‑search.
    """
    eps = 1e-12
    n = a.shape[0]
    velocity = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    step0 = 0.5
    it = 0

    best_a = a.copy()
    best_sc = _max_convolution_score(best_a)

    while time.time() < deadline:
        it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (within numerical tolerance)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        w = np.zeros_like(conv)
        w[active] = 1.0
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam‑style momentum
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** it)

        # back‑tracking line‑search
        step = step0
        while True:
            a_candidate = a - step * v_hat
            a_candidate = _project_onto_simplex(a_candidate, 1.0)
            a_candidate = np.maximum(a_candidate, eps)
            a_candidate /= a_candidate.sum()
            cand_sc = _max_convolution_score(a_candidate)
            if cand_sc < best_sc - 1e-12 or step < 1e-8:
                break
            step *= 0.5

        a = a_candidate
        if cand_sc < best_sc - 1e-12:
            best_sc = cand_sc
            best_a = a.copy()
            print(f"[{time.time():.1f}] Subgrad it={it} new best {cand_sc:.6f}")

        # occasional jitter to avoid stagnation
        if it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    return best_a, best_sc


def _hill_climb_prune(a, deadline):
    """
    Final polishing:
      * move a tiny amount of mass from the entry that participates
        in the maximal convolution product to a random location,
      * periodically prune negligible entries,
      * add small jitter every few thousand steps.
    """
    eps = 1e-12
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_sc = 2.0 * n * M
    it = 0

    while time.time() < deadline:
        it += 1

        # ---- mass‑transfer step ---------------------------------
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # locate the pair (i,j) that realises the maximal product
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target != source:
            max_delta = min(a[source] * 0.5, 0.05)        # allow a slightly larger move
            delta = max_delta * random.random()
            if delta > eps:
                a2 = a.copy()
                a2[source] -= delta
                a2[target] += delta
                a2 = np.maximum(a2, eps)
                a2 /= a2.sum()
                new_M = _fft_self_convolve(a2).max()
                if new_M < M - 1e-13:
                    a = a2
                    conv = _fft_self_convolve(a)
                    M = new_M
                    best_a = a.copy()
                    best_sc = 2.0 * n * M
                    if it % 200 == 0:
                        print(f"[{time.time():.1f}] Hill it={it} new best {best_sc:.6f}")

        # ---- pruning of tiny entries -----------------------------
        if it % 1500 == 0:
            for keep_frac in (0.99, 0.97, 0.95, 0.90):
                k_keep = max(2, int(keep_frac * n))
                idx = np.argpartition(a, -k_keep)[-k_keep:]
                a_pruned = np.zeros_like(a)
                a_pruned[idx] = a[idx]
                a_pruned /= a_pruned.sum()
                sc_pruned = _max_convolution_score(a_pruned)
                if sc_pruned < best_sc - 1e-12:
                    a = a_pruned
                    conv = _fft_self_convolve(a)
                    M = conv.max()
                    n = a.shape[0]
                    best_a = a.copy()
                    best_sc = sc_pruned
                    print(f"[{time.time():.1f}] Prune it={it} keep={keep_frac:.2f} new best {best_sc:.6f}")

        # ---- jitter ------------------------------------------------
        if it % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()

        # early exit if we already hit the target
        if best_sc <= 1.503:
            break

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Main driver:  explore analytically‑parameterised families for several
    lengths, keep the elite candidates and polish each of them with three
    optimisation stages under the given time budget.
    Returns the best step‑function as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 5.0   # safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    elite = []                         # (score, vector)

    # --------------------------------------------------------------
    # 1. Broad exploration – keep the top 8 overall candidates
    lengths = [2000, 3500, 5000, 8000, 12000, 20000]
    for n in lengths:
        if time.time() >= deadline:
            break
        candidates = _generate_candidates_for_length(n)
        for vec in candidates:
            if time.time() >= deadline:
                break
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_score = sc
                best_seq = vec.copy()
                print(f"[{time.time():.1f}] Length {n} new global best {sc:.6f}")
            # maintain elite list of size ≤8
            if len(elite) < 8:
                elite.append((sc, vec.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, vec.copy())
                    elite.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2. Incorporate a previous solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_score = sc
                best_seq = prev.copy()
                print(f"[{time.time():.1f}] Using previous best {sc:.6f}")
            if len(elite) < 8:
                elite.append((sc, prev.copy()))
                elite.sort(key=lambda x: x[0])
            else:
                if sc < elite[-1][0]:
                    elite[-1] = (sc, prev.copy())
                    elite.sort(key=lambda x: x[0])
    except Exception:
        pass

    # --------------------------------------------------------------
    # 3. Fallback in the unlikely case that nothing was generated
    if best_seq is None:
        n0 = 4000
        best_seq = _distribution_uniform(n0)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform score {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Local optimisation from the elite pool
    restarts = min(6, len(elite))
    for ridx in range(restarts):
        if time.time() >= deadline:
            break
        a0 = elite[ridx][1].copy()
        n0 = a0.shape[0]
        print(f"[{time.time():.1f}] Restart {ridx+1}/{restarts} (len={n0}) score {elite[ridx][0]:.6f}")

        # ---- smooth‑max mirror descent ------------------------------
        a_mid, sc_mid = _smooth_max_mirror_descent(a0, deadline)

        # ---- projected sub‑gradient polishing ----------------------
        a_mid, sc_mid = _projected_subgradient(a_mid, deadline)

        # ---- final hill‑climber + pruning ---------------------------
        a_mid, sc_mid = _hill_climb_prune(a_mid, deadline)

        if sc_mid < best_score - 1e-12:
            best_score = sc_mid
            best_seq = a_mid.copy()
            print(f"[{time.time():.1f}] New global best after restart {sc_mid:.6f}")

        # early exit if target achieved
        if best_score <= 1.503:
            break

    # --------------------------------------------------------------
    # 5. Final sanity check and return
    best_seq = np.maximum(best_seq, eps)
    best_seq /= best_seq.sum()
    final_score = _max_convolution_score(best_seq)
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
