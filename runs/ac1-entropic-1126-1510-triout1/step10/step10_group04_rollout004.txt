"""
Propose a non‑negative step‑function (list of floats) that minimizes
``evaluate_sequence``.  The objective for a normalized vector (∑a_i=1) is

        score = 2·n·max(conv(a,a))

so we minimise the L∞‑norm of the self‑convolution.  The algorithm
consists of three stages:

1. **Broad analytic scan** –  families of symmetric densities
   (uniform, arcsine, Beta(p,p), power‑edge, V‑shape, …) are sampled
   for several lengths (up to 12 000).  The best shape is kept.

2. **Refinement by up‑sampling** – the current best vector is repeatedly
   split into more bins (each entry is divided equally among *k*
   copies).  This often lowers the maximum convolution while keeping the
   shape.

3. **Projected exponentiated‑gradient descent** – a sub‑gradient of the
   max‑convolution is computed exactly (the gradient of a peak `k` is
   `g_i = 2·a_{k‑i}` on the admissible range).  The update
   `a←a·exp(-η·g)` is followed by Euclidean projection onto the simplex.
   A linear learning‑rate schedule (≈2 → 0.01) drives the process.
   The best vector seen during the descent is retained.

All steps respect the global time budget (default 1000 s).  Intermediate
scores are printed.  The returned list is already normalised and contains
only non‑negative numbers (tiny eps guarantees strict positivity).
"""

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = s}."""
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _max_convolution_score(a: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a vector already normalised to sum = 1.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * float(conv.max())

# Analytic families – each returns a *probability* vector (sum = 1)
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_triangular(n: int) -> np.ndarray:
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 1:
        right = left[:-1][::-1]
    else:
        right = left[::-1]
    vals = np.concatenate([left, right])
    if vals.size != n:
        vals = np.pad(vals, (0, n - vals.size), constant_values=0)
    vals /= vals.sum()
    return vals


def _distribution_V_shape(n: int) -> np.ndarray:
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre)
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space to avoid overflow for tiny p
    logw = (p - 1) * (np.log(x) + np.log(1 - x))
    w = np.exp(logw)
    if w.sum() == 0 or np.any(np.isinf(w)):
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_power_edge(n: int, alpha: float) -> np.ndarray:
    """
    Edge‑heavy family:
        w_i ∝ ((i+0.5)*(n‑i‑0.5))^{‑α}
    """
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - i - 0.5)
    eps = 1e-12
    vals = np.power(denom + eps, -alpha)
    vals /= vals.sum()
    return vals


def _distribution_power_abs(n: int, beta: float) -> np.ndarray:
    """Symmetric power law |i‑centre|^{β}."""
    centre = (n - 1) / 2.0
    vals = np.abs(np.arange(n, dtype=np.float64) - centre) ** beta
    eps = 1e-12
    vals = np.maximum(vals, eps)
    vals /= vals.sum()
    return vals


# ----------------------------------------------------------------------
def _subgradient_max_convolution(a: np.ndarray):
    """Return a sub‑gradient of max(conv(a,a)) and the current max value."""
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_val = conv.max()
    # treat all indices that are (numerically) maximal
    idxs = np.where(conv >= max_val - 1e-14)[0]
    grad = np.zeros_like(a)
    for k in idxs:
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=np.int64)
        j = k - i_range
        grad[i_range] += 2.0 * a[j]
    if idxs.size > 1:
        grad /= float(idxs.size)
    return grad, max_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low ``evaluate_sequence`` score.
    Returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    eps = 1e-12
    best_seq = None
    best_score = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1. Broad scan over analytic families
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000, 12000]
    for n in lengths:
        if time.time() >= deadline:
            break

        # ---- basic families -------------------------------------------------
        for vec in (
            _distribution_uniform(n),
            _distribution_arcsine(n),
            _distribution_triangular(n),
            _distribution_V_shape(n),
        ):
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} basic   score={sc:.6f}")

        # ---- Beta(p,p) sweep -----------------------------------------------
        p_vals = np.arange(0.05, 2.01, 0.05)
        for p in p_vals:
            if time.time() >= deadline:
                break
            vec = _distribution_beta_sym(n, p)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} Beta(p={p:.2f}) score={sc:.6f}")

        # ---- Power‑edge sweep -----------------------------------------------
        for alpha in (0.55, 0.6, 0.7, 0.8, 0.9):
            if time.time() >= deadline:
                break
            vec = _distribution_power_edge(n, alpha)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} edge(α={alpha:.2f}) score={sc:.6f}")

        # ---- Power‑abs sweep -------------------------------------------------
        for beta in (0.5, 1.0, 2.0, 3.0):
            if time.time() >= deadline:
                break
            vec = _distribution_power_abs(n, beta)
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} abs(β={beta:.1f}) score={sc:.6f}")

        # ---- a few random Dirichlet draws (very edge‑heavy) -----------------
        for _ in range(3):
            if time.time() >= deadline:
                break
            alpha = random.choice([0.02, 0.05, 0.08])
            half = (n + 1) // 2
            half_vec = np.random.dirichlet(np.full(half, alpha))
            if n % 2 == 0:
                vec = np.concatenate([half_vec, half_vec[::-1]])
            else:
                vec = np.concatenate([half_vec, half_vec[-2::-1]])
            vec = np.maximum(vec, eps)
            vec /= vec.sum()
            sc = _max_convolution_score(vec)
            if sc < best_score:
                best_seq = vec.copy()
                best_n = n
                best_score = sc
                print(f"[{time.time():.1f}] init n={n} Dirichlet(α={alpha}) score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2. Incorporate a previously computed global best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, eps)
            prev /= prev.sum()
            sc = _max_convolution_score(prev)
            if sc < best_score:
                best_seq = prev.copy()
                best_n = prev.size
                best_score = sc
                print(f"[{time.time():.1f}] global best from previous  score={sc:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 3. Fallback (very unlikely)
    if best_seq is None:
        best_n = 3000
        best_seq = _distribution_uniform(best_n)
        best_seq /= best_seq.sum()
        best_score = _max_convolution_score(best_seq)
        print(f"[{time.time():.1f}] fallback uniform  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upsampling / refinement (split each entry into k copies)
    a = best_seq.copy()
    for factor in (2, 3, 4, 5, 6):
        if time.time() >= deadline:
            break
        a_ref = np.repeat(a / factor, factor)
        sc = _max_convolution_score(a_ref)
        if sc < best_score:
            best_seq = a_ref.copy()
            a = a_ref
            best_n = a_ref.size
            best_score = sc
            print(f"[{time.time():.1f}] upsample ×{factor}  n={best_n}  score={sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Projected exponentiated‑gradient descent on the exact sub‑gradient
    a = best_seq.copy()
    n = a.shape[0]
    max_iters = 50000
    lr_start = 2.0
    lr_end = 0.01

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # sub‑gradient of the maximal convolution entry
        grad, cur_max = _subgradient_max_convolution(a)

        # linear schedule for the learning rate
        t = it / max_iters
        lr = lr_start * (1 - t) + lr_end * t

        # exponentiated‑gradient update
        a = a * np.exp(-lr * grad)

        # projection onto the simplex (keeps sum = 1)
        a = _project_onto_simplex(a, 1.0)

        # keep strict positivity
        a = np.maximum(a, eps)
        a /= a.sum()

        # occasional reporting
        if (it % 500 == 0) or (it == max_iters - 1):
            sc = 2.0 * n * cur_max
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] gradient it={it}  best={sc:.6f}")

    # ------------------------------------------------------------------
    # 6. Tiny hill‑climber – move a bit of mass from the biggest product pair
    a = best_seq.copy()
    n = a.shape[0]
    hill_iters = 0
    while time.time() < deadline:
        hill_iters += 1
        conv = _fft_self_convolve(a)
        k = int(conv.argmax())
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        i_range = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_range = k - i_range
        products = a[i_range] * a[j_range]
        best_pair_idx = int(np.argmax(products))
        best_i = int(i_range[best_pair_idx])
        best_j = int(j_range[best_pair_idx])

        source = best_i if a[best_i] >= a[best_j] else best_j
        target = random.randint(0, n - 1)
        if target == source:
            continue

        max_delta = min(a[source] * 0.4, 0.015)
        delta = max_delta * random.random()
        if delta < eps:
            continue

        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        a_new = np.maximum(a_new, eps)
        a_new /= a_new.sum()

        new_conv = _fft_self_convolve(a_new)
        new_max = new_conv.max()
        if new_max + 1e-12 < conv.max():
            a = a_new
            sc = 2.0 * n * new_max
            if sc < best_score - 1e-12:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.time():.1f}] hill it={hill_iters}  best={sc:.6f}")

        # occasional jitter to escape plateaus
        if hill_iters % 5000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a += jitter
            a = np.maximum(a, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            # no need to recompute best_score here; it will be caught later
            print(f"[{time.time():.1f}] jitter applied at hill iter {hill_iters}")

    final_score = 2.0 * best_seq.shape[0] * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={best_seq.shape[0]})")
    return best_seq.tolist()
