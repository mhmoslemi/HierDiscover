"""
Propose a non‑negative step function (list of floats) that minimizes
`evaluate_sequence`.  The algorithm consists of five stages:

1️⃣  Very broad exploration of many analytic families (uniform,
    arcsine, symmetric Beta, asymmetric Beta, power‑law, mixtures and
    Dirichlet draws) for several lengths up to 50 000.  The best
    candidate (normalised to sum = 1) is kept.

2️⃣  A long smooth‑max mirror‑descent run (exponentiated‑gradient) that
    minimises a soft‑max approximation of the max‑convolution.  The
    temperature is increased geometrically up to 1e6 while the learning
    rate decays, yielding many precise updates.

3️⃣  Adam‑style projected sub‑gradient descent on the true max‑convolution
    with a back‑tracking line‑search.  Momentum helps to escape shallow
    plateaus.

4️⃣  Aggressive mass‑transfer hill‑climbing that moves a tiny amount of
    probability from the largest contributor to the current peak toward
    the smallest entry.

5️⃣  A short simulated‑annealing polishing stage (random pair moves with
    Metropolis acceptance).  All steps respect the time budget (default
    1000 s) and the constraints of `evaluate_sequence`.
"""

import time
import random
import math
import numpy as np

# Helper utilities -------------------------------------------------------
def _project_onto_simplex(v, s=1.0):
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum(x) = s}.
    Reference: Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·n‑1) using a real FFT."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


def _fft_convolve(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    size = _next_pow2(a.size + b.size - 1)
    A = np.fft.rfft(a, n=size)
    B = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(A * B, n=size)
    return conv[: a.size + b.size - 1]


def _normed_score(a: np.ndarray) -> float:
    """
    Score for a vector that already satisfies sum(a) = 1.
    Equivalent to `evaluate_sequence` without the cheap‑checks.
    """
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    return 2.0 * n * conv.max()

# Analytic families (all return a vector with sum = 1) --------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric Beta(p,p) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # log‑density for numerical stability
    logw = (p - 1.0) * np.log(x) + (p - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_beta(n: int, p: float, q: float) -> np.ndarray:
    """Asymmetric Beta(p,q) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logw = (p - 1.0) * np.log(x) + (q - 1.0) * np.log(1.0 - x)
    logw -= np.max(logw)
    w = np.exp(logw)
    if not np.isfinite(w).all() or w.sum() == 0.0:
        return _distribution_uniform(n)
    w /= w.sum()
    return w


def _distribution_powerlaw(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: w_i ∝ ((i+0.5)*(n-i-0.5))^{-p}."""
    idx = np.arange(n, dtype=np.float64)
    vals = ((idx + 0.5) * (n - idx - 0.5)) ** (-p)
    vals += 1e-15                     # avoid exact zeros
    vals /= vals.sum()
    return vals


def _distribution_mixture_beta_uniform(n: int, p: float, lam: float) -> np.ndarray:
    """Mixture λ·Beta(p,p) + (1‑λ)·Uniform, re‑normalised."""
    beta = _distribution_beta_sym(n, p)
    uni = _distribution_uniform(n)
    mix = lam * beta + (1.0 - lam) * uni
    mix /= mix.sum()
    return mix


def _distribution_dirichlet_random(n: int, concentration: float = 1.0) -> np.ndarray:
    """Draw from Dirichlet(α,…,α) with given concentration."""
    vec = np.random.dirichlet(np.full(n, concentration))
    vec /= vec.sum()
    return vec

        a /= a.sum()

        # occasional reporting / best‑update
        if (it % max(1, total_iters // 20) == 0) or (it == total_iters - 1):
            cur_sc = _normed_score(a)
            if cur_sc < best_score - 1e-12:
                best_score, best_seq = cur_sc, a.copy()
                print(f"[{time.time():.1f}] mirror it={it}/{total_iters} new best {cur_sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Projected Adam‑style sub‑gradient descent on the true max‑convolution
    a = best_seq.copy()
    velocity = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    base_lr = 0.8
    sub_it = 0
    while time.time() < deadline:
        sub_it += 1
        conv = _fft_self_convolve(a)
        cur_max = conv.max()
        # active set (indices where the max is attained, tolerance scaled)
        tol = 1e-12 * (abs(cur_max) + 1.0)
        active = np.flatnonzero(conv >= cur_max - tol)

        # sub‑gradient (sum of contributions from all active indices)
        w = np.zeros_like(conv)
        w[active] = 1.0 / len(active)    # average over ties
        grad = 2.0 * _fft_convolve(a[::-1], w)[n - 1 : n - 1 + n]

        # Adam momentum (bias‑corrected)
        velocity = beta1 * velocity + (1.0 - beta1) * grad
        v_hat = velocity / (1.0 - beta1 ** sub_it)

        # back‑tracking line‑search
        step = base_lr
        while True:
            cand = a - step * v_hat
            cand = _project_onto_simplex(cand, 1.0)
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_sc = _normed_score(cand)
            if cand_sc < best_score - 1e-13 or step < 1e-8:
                break
            step *= 0.5

        a = cand
        if cand_sc < best_score - 1e-12:
            best_score, best_seq = cand_sc, a.copy()
            print(f"[{time.time():.1f}] subgrad it={sub_it} new best {cand_sc:.6f}")

        # small random jitter every 500 iterations
        if sub_it % 500 == 0:
            jitter = np.random.randn(n) * 1e-6
            a = np.maximum(a + jitter, eps)
            a /= a.sum()

        if time.time() + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 5️⃣  Targeted mass‑transfer hill‑climbing
    a = best_seq.copy()
    conv = _fft_self_convolve(a)
    M = conv.max()
    hill_it = 0
    while time.time() < deadline:
        hill_it += 1
        k = int(conv.argmax())
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)

        # locate the pair (i,j) with the largest product contributing to that peak
        best_i, best_j = i_start, k - i_start
        max_prod = a[best_i] * a[best_j]
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            if prod > max_prod:
                max_prod = prod
                best_i, best_j = i, j

        source = best_i if a[best_i] >= a[best_j] else best_j
        # choose a tiny destination (global minimum)
        target = int(np.argmin(a))
        if target == source:
            # pick second‑smallest if clash
            sorted_idx = np.argsort(a)
            target = sorted_idx[1]

        max_delta = min(a[source] * 0.5, 0.02)
        delta = max_delta * random.random()
        if delta <= eps:
            continue

        a2 = a.copy()
        a2[source] -= delta
        a2[target] += delta
        a2 = np.maximum(a2, eps)
        a2 /= a2.sum()
        new_M = _fft_self_convolve(a2).max()
        if new_M < M - 1e-13:
            a = a2
            conv = _fft_self_convolve(a)
            M = new_M
            best_seq = a.copy()
            best_score = 2.0 * n * M
            if hill_it % 100 == 0:
                print(f"[{time.time():.1f}] hill it={hill_it} new best {best_score:.6f}")

        # occasional jitter to avoid stagnation
        if hill_it % 2000 == 0:
            jitter = np.random.randn(n) * 1e-5
            a = np.maximum(a + jitter, eps)
            a /= a.sum()
            conv = _fft_self_convolve(a)
            M = conv.max()
            print(f"[{time.time():.1f}] hill jitter at it={hill_it}")

    # ------------------------------------------------------------------
    # 6️⃣  Very short simulated‑annealing polish (if time permits)
    if time.time() < deadline:
        a = best_seq.copy()
        cur_score = best_score
        T0 = 0.02
        Tf = 1e-5
        total_sa = 2000
        for it in range(total_sa):
            if time.time() >= deadline:
                break
            # temperature schedule (exponential decay)
            T = T0 * (Tf / T0) ** (it / total_sa)

            i, j = random.sample(range(n), 2)
            if a[i] < eps or a[j] < eps:
                continue
            delta = min(a[i], a[j]) * 0.01 * random.random()
            if delta < eps:
                continue
            # move delta from i to j
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = np.maximum(cand, eps)
            cand /= cand.sum()
            cand_score = _normed_score(cand)
            if cand_score < cur_score or random.random() < math.exp((cur_score - cand_score) / T):
                a = cand
                cur_score = cand_score
                if cur_score < best_score - 1e-12:
                    best_score, best_seq = cur_score, a.copy()
                    print(f"[{time.time():.1f}] SA it={it} new best {cur_score:.6f}")

    final_score = 2.0 * n * _fft_self_convolve(best_seq).max()
    print(f"Final best score ≈ {final_score:.6f} (length={n})")
    return best_seq.tolist()
