def _objective(a):

    if s <= 0:

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence.

Algorithm
---------
1. **Rich analytic pool** – For each chosen length n we evaluate
   several parametric families (uniform, arcsine, symmetric beta,
   edge–weighted) and a few random Dirichlet / sparse seeds.  For the
   beta and edge families a cheap golden‑section search over the shape
   parameter p (and a small grid over a mixing weight λ with the
   uniform distribution) is performed, keeping the best candidates.

2. **Local continuous optimisation** – The best few seeds are refined
   by exponentiated‑gradient (mirror) descent on the exact
   L∞‑convolution objective.  The sub‑gradient is taken at the index
   where the self‑convolution attains its maximum.  A simple decreasing
   step‑size (η₀/√t) is used and the iterate is always re‑normalised
   to the probability simplex.

3. **Resolution boost** – The best shape found on a moderate size
   (≈2000–3000) is up‑sampled to larger lengths (5000, 10000, 20000)
   by linear interpolation and re‑normalisation.  This often reduces
   the discrete convolution peak because the underlying continuous
   density is represented more faithfully.

4. **Discrete polishing** – A fast greedy “mass‑transfer’’ walk:
   repeatedly move a random fraction of mass from one index to another.
   If the objective improves the move is kept.  This step is run until
   the global time budget is exhausted.

All steps respect the overall time limit (default 1000 s).  The
function returns a plain Python list; ``evaluate_sequence`` will take
care of clipping and normalisation.

The implementation uses only NumPy and the standard library.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_self_convolve(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(a,a)) for a vector a with sum = 1.
    a is assumed non‑negative.
    """
    s = a.sum()
    if s <= 0.0:
        return float("inf")
    a = a / s
    n = a.shape[0]
    conv = _fft_self_convolve(a)
    max_b = conv.max()
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Analytic families ----------------------------------------------------
def _distribution_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _distribution_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    vals /= vals.sum()
    return vals


def _distribution_beta_sym(n: int, p: float) -> np.ndarray:
    """Symmetric beta (Beta(p,p)) on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid log(0) – clamp p>0 in caller
    w = np.exp((p - 1) * np.log(x) + (p - 1) * np.log(1.0 - x))
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_edge_weighted(n: int, p: float) -> np.ndarray:
    """U‑shaped family: w_i ∝ ((i+0.5)*(n‑i‑0.5))^(−p)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    j = (n - np.arange(n, dtype=np.float64) - 0.5)
    w = (i * j) ** (-p)
    if w.sum() == 0.0:
        w = np.full_like(w, 1.0 / n)
    else:
        w /= w.sum()
    return w


def _distribution_random_sparse(
    n: int, rng: np.random.Generator, density: float = 0.01
) -> np.ndarray:
    """Sparse random vector with a given density of non‑zeros."""
    k = max(2, int(n * density))
    idx = rng.choice(n, size=k, replace=False)
    vals = np.zeros(n, dtype=np.float64)
    vals[idx] = rng.random(k)
    if vals.sum() == 0.0:
        vals[0] = 1.0
    vals /= vals.sum()
    return vals


def _mix_with_uniform(base: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination lam·U + (1‑lam)·base, where U is uniform."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    return lam * uniform + (1.0 - lam) * base


# ----------------------------------------------------------------------
# Golden‑section search over a single scalar parameter -----------------
def _golden_section_search(
    gen_func, a: float, b: float, tol: float = 1e-4, max_iter: int = 30
) -> tuple[float, float]:
    """Return (best_x, best_f) for the min of gen_func(x) on [a,b]."""
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    c = b - (b - a) / phi
    d = a + (b - a) / phi
    fc = _objective(gen_func(c))
    fd = _objective(gen_func(d))
    iter_cnt = 0
    while (b - a) > tol and iter_cnt < max_iter:
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - (b - a) / phi
            fc = _objective(gen_func(c))
        else:
            a, fc = c, fd
            c = d
            d = a + (b - a) / phi
            fd = _objective(gen_func(d))
        iter_cnt += 1
    if fc < fd:
        return c, fc
    else:
        return d, fd


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) on the exact objective -----
def _mirror_descent(
    a: np.ndarray,
    per_deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    lr0: float = 0.6,
    eval_interval: int = 200,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on max‑convolution objective."""
    n = a.shape[0]
    it = 0
    while time.time() < per_deadline:
        it += 1

        # compute full self‑convolution
        conv = _fft_self_convolve(a)
        idx = int(conv.argmax())

        # gradient of max‑conv entry (2*a_{idx-i})
        ind = idx - np.arange(n)
        mask = (ind >= 0) & (ind < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[ind[mask]]

        # exponentiated update
        eta = lr0 / math.sqrt(it)
        a = a * np.exp(-eta * grad)

        # re‑normalise to simplex
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / total

        if it % eval_interval == 0:
            cur = _objective(a)
            if cur < best_score - 1e-14:
                best_score = cur
                best_seq = a.copy()
                print(
                    f"[{time.time():.1f}] mirror it={it} new global best {best_score:.6f}"
                )
        if it >= 500000:   # safety cap
            break
    return best_seq, best_score


# ----------------------------------------------------------------------
# Greedy pairwise mass‑transfer walk -----------------------------------
def _pairwise_swap_local(
    a: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    max_trials: int | None = None,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass from i to j; keep if it improves."""
    n = a.shape[0]
    trials = 0
    while time.time() < deadline:
        if max_trials is not None and trials >= max_trials:
            break
        trials += 1

        src = np.where(a > 1e-12)[0]
        if src.size == 0:
            break
        i = np.random.choice(src)

        # pick a distinct target
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = a[i] * np.random.rand() * 0.35
        if delta < 1e-14:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        cur = _objective(new_a)
        if cur < best_score - 1e-14:
            a = new_a
            best_score = cur
            best_seq = a.copy()
            print(
                f"[{time.time():.1f}] swap i={i}→j={j} δ={delta:.6g} new best {best_score:.6f}"
            )
    return best_seq, best_score


# ----------------------------------------------------------------------
# Upsample a distribution by linear interpolation ---------------------
def _upsample(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Interpolate seq (assumed to sum to 1) to length new_len."""
    old_len = seq.shape[0]
    old_x = np.linspace(0.0, 1.0, old_len, endpoint=False) + 0.5 / old_len
    new_x = np.linspace(0.0, 1.0, new_len, endpoint=False) + 0.5 / new_len
    new_seq = np.interp(new_x, old_x, seq, left=0.0, right=0.0)
    new_seq[new_seq < 0] = 0.0
    if new_seq.sum() == 0.0:
        new_seq = np.full(new_len, 1.0 / new_len, dtype=np.float64)
    else:
        new_seq = new_seq / new_seq.sum()
    return new_seq
